\begin{defn}{Spacetime}{}\\
Spacetime is an $4$-dimensional orientable pseudo-Riemannian manifold $(M,g)$ where $g$ has signature $(3,1)$. 
\end{defn}

\begin{defn}{The Einstein Field Equation}{}\\
Let $(M,g)$ be spacetime manifold. An Einstein field equation consists of the following data. 
\begin{itemize}
\item A $(0,2)$ tensor field $T$ such that $T^{ij}=T^{ji}$ and $\sum_{i=1}^{\dim(M)}\nabla_iT_j^i=0$ called the stress-energy-momentum tensor. 
\item A constant $K$. 
\item The field equation $$R_{ij}-\frac{1}{2}Rg_{ij}=\frac{8\pi G}{c^4}T_{ij}$$ where $G$ is the gravitational constant and $c$ is the speed of light. 
\end{itemize}
\end{defn}
