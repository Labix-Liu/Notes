\subsection{Symmetric Algebras over Graded Modules}
We have seen the tensor, symmetric and exterior algebra of an $R$-module. When the underlying $R$-module is graded, these algebra become bigraded and the theory differs slightly. 

\begin{defn}{The Graded Braid Map}{}\\
Let $R$ be a commutative graded ring. Let $M$ be a graded $R$-module. Define the graded braid map $\tau:M\otimes_RM\to M\otimes_RM$ by $$\tau(x\otimes y)=(-1)^{ij}y\otimes x$$ for $x\in M_i$ and $y\in M_j$. 
\end{defn}

\begin{defn}{The Symmetric Algebra of a Graded Module}{}\\
Let $R$ be a commutative graded ring. Let $M$ be a graded $R$-module. Define the symmetric algebra of $M$ to be the quotient $$S^\text{gr}(M)=\frac{T^\text{gr}(M)}{\langle x\otimes y-\tau(x\otimes y)\;|\;x,y\in M\rangle}$$
\end{defn}

\begin{prp}{}{}\\
Let $R$ be a commutative graded ring. Let $M$ be a graded $R$-module. Then $S^\text{gr}(M)$ is a graded commutative $R$-module. 
\end{prp}

\begin{prp}{}{}\\
Let $R$ be a commutative graded ring. Let $M$ be a graded $R$-module. Then there is a graded $R$-isomorphism $$S^\text{gr}(M)\cong S(M_\text{even})\otimes_R\Lambda(M_\text{odd})$$
\end{prp}

\subsection{Exterior Algebras over Graded Modules}
\begin{defn}{The Exterior Algebra of a Graded Module}{}\\
Let $R$ be a commutative graded ring. Let $M$ be a graded $R$-module. Define the exterior algebra of $M$ to be the quotient $$\Lambda^\text{gr}(M)=\frac{T^\text{gr}(M)}{\langle x\otimes y+\tau(x\otimes y),x\otimes x\;|\;x,y\in M\rangle}$$
\end{defn}

\begin{prp}{}{}\\
Let $R$ be a commutative graded ring. Let $M$ be a graded $R$-module. Then there is a graded $R$-isomorphism $$\Lambda^\text{gr}(M)\cong\Lambda(M_\text{even})\otimes_RS(M_\text{odd})$$
\end{prp}

\section{Graded Lie Algebras}
\begin{defn}{Graded Lie Algebras}{}\\
Let $R$ be a commutative ring. Let $M=\bigoplus_{i=0}^\infty M_i$ be a graded $R$-module. We say that $M$ is a graded Lie algebra if there exists an $R$-module homomorphism $[-,-]:M_i\times_j\to M_{i+j}$ such that the following are true. 
\begin{itemize}
\item $[x,y]=(-1){ij+1}[y,x]$ for all $x\in M_i$ and $y\in M_j$. 
\item Jacobi Identity: $[x,[y,z]]=[[x,y],z]+(-1)^{ij}[y,[x,z]]$. 
\item $[x,[x,x]]=0$. 
\end{itemize}
\end{defn}

\section{The Theory of Semigroups}
\subsection{Basic Properties of Semigroups}
\begin{defn}{Semigroups}{}\\
A semigroup is a pair $(S,\ast)$ where $S$ is a set and $\ast:S\times S\to S$ is a binary operation on $S$ such that $$(a\ast b)\ast c=a\ast(b\ast c)$$ for all $a,b,c\in S$. 
\end{defn}

\begin{defn}{Semigroup Homomorphisms}{}\\
Let $S,T$ be semigroups. Let $\phi:S\to T$ be a function. We say that $\phi$ is a semigroup homomorphism if $$\phi(ab)=\phi(a)\phi(b)$$ for all $a,b\in S$. 
\end{defn}

\begin{defn}{Commutative Semigroups}{}\\
Let $S$ be a semigroup. We say that $S$ is commutative if $$a\ast b=b\ast a$$ for all $a,b\in S$. 
\end{defn}

\begin{defn}{Cancellative Semigroups}{}\\
Let $S$ be a semigroup. We say that $S$ is cancellative if the following are true. 
\begin{itemize}
\item $ab=ac$ implies $b=c$ for all $a,b,c\in S$. 
\item $ba=ca$ implies $b=c$ for all $a,b,c\in S$. 
\end{itemize}
\end{defn}

\begin{defn}{Finitely Generated Semigroups}{}\\
Let $S$ be a semigroup. We say that $S$ is finitely generated if there exists $a_1,\dots,a_n\in S$ such that $$a=a_1^{k_1}\cdots a_n^{k_n}$$ for all $a\in M$ and some $k_1,\dots,k_n\in\N$. 
\end{defn}

\subsection{The Semigroup Ring}
\begin{defn}{The Semigroup Ring}{}\\
Let $S$ be a semigroup. Let $R$ be a ring. Define the semigroup ring of $S$ over $R$ to be the left $R$-module $$R[S]=\bigoplus_{s\in S}R\cdot t^s$$ together with multiplication defined on basis by $t^{s_1}\cdot t^{s_2}=t^{s_1+s_2}$
\end{defn}

\begin{lmm}{}{}\\
Let $S$ be a semigroup. Let $R$ be a ring. Then the following are true. 
\begin{itemize}
\item $R[S]$ is a ring. 
\item If $S$ is commutative, then $R[S]$ is a commutative ring. 
\item If $R$ is commutative, then $R[S]$ is an $R$-algebra. 
\item If $R$ and $S$ are commutative, then $R[S]$ is a commutative $R$-algebra. 
\end{itemize}
\end{lmm}

\section{The Theory of Monoids}
\subsection{Basic Properties of Monoids}
\begin{defn}{Monoids}{}\\
A monoid is a pair $(M,\ast)$ where $M$ is a set and $\ast:M\times M\to M$ is a binary operation on $M$ satisfying the following axioms: 
\begin{itemize}
\item Associativity: $(a\ast b)\ast c=a\ast(b\ast c)$ for all $a,b,c\in M$. 
\item Identity: There exists $1_M\in M$ such that $1_M\ast a=a=a\ast 1_M$ for all $a\in M$. 
\end{itemize}
\end{defn}

\begin{defn}{Monoid Homomorphisms}{}\\
Let $M,N$ be monoids. Let $\phi:M\to N$ be a function. We say that $\phi$ is a monoid homomorphism if the following are true. 
\begin{itemize}
\item $\phi(m_1m_2)=\phi(m_1)\phi(m_2)$ for all $m_1,m_2\in M$. 
\item $\phi(1_M)=1_N$. 
\end{itemize}
\end{defn}

\subsection{The Grothendieck Completion of a Group}
\begin{defn}{Grothendieck Completion}{}\\
Let $M$ be a commutative monoid. Define the Grothendieck completion of $M$ to be the set $$G(M)=\frac{M\times M}{\sim}$$ where $(m_1,m_2)\sim(n_1,n_2)$ if $m_1+n_2+k=m_2+n_1+k$ together with the binary operation $\ast:G(M)\times G(M)\to G(M)$ defined by $$(m_1,m_2)\ast(n_1,n_2)=(m_1+n_1,m_2+n_2)$$
\end{defn}

\begin{lmm}{}{}\\
Let $M$ be a commutative monoid. Then $G(M)$ is an abelian group. 
\end{lmm}

\begin{defn}{The Canonical Homomorphism}{}\\
Let $M$ be a commutative monoid. Define the canonical homomorphism to the Grothendieck completion to be the function $i:M\to G(M)$ given by $i(m)=(m,0)$. 
\end{defn}

\begin{prp}{Universal Property of the Grothendieck Completion}{}\\
Let $M$ be a commutative monoid. Then $G(M)$ and the canonical map $i:M\to G(M)$ satisfies the following universal property: For any abelian group $G$ and any monoid homomorphism $\phi:M\to G$, there exists a unique group homomorphism $\psi:G(M)\to G$ such that the following diagram commutes: \\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	M & {G(M)} \\
	& G
	\arrow["i", from=1-1, to=1-2]
	\arrow["\phi"', from=1-1, to=2-2]
	\arrow["{\exists!\psi}", dashed, from=1-2, to=2-2]
\end{tikzcd}} \\
Moreover, $G(M)$ is the unique (up to unique isomorphism) abelian group satisfying this property. 
\end{prp}

\begin{lmm}{}{}\\
Let $M$ be a cancellative commutative monoid. Then the canonical morphism $i:M\to G(M)$ is injective. 
\end{lmm}

\begin{prp}{}{}\\
There is an adjunction $G:\bold{CMon}\overset{\dashv}{\rightleftarrows}\bold{Ab}:U$ with the forgetful functor. Explicitly, there is a natural isomorphism $$\Hom_\bold{Ab}(G(M),A)\cong\Hom_\bold{CMon}(M,A)$$
\end{prp}

\subsection{Monoid Objects}
\begin{defn}{Monoid Objects}{}\\
Let $(\mC,\otimes,1)$ be a monoidal category. A monoid object in $\mC$ is an object $M\in\mC$ together with the following data: 
\begin{itemize}
\item A morphism $\mu:M\otimes M\to M$ called multiplication. 
\item A morphism $\eta:I\to M$ called the unit. 
\end{itemize}
such that the following diagrams commute: \\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	{M\otimes M\otimes M} & {M\otimes M} && M & {M\otimes M} \\
	{M\otimes M} & M && {M\otimes M} & M
	\arrow["{\mu\otimes\text{id}_M}", from=1-1, to=1-2]
	\arrow["{\text{id}_M\otimes\mu}"', from=1-1, to=2-1]
	\arrow["\mu"', from=1-2, to=2-2]
	\arrow["{(\text{id}_M,\eta)}", from=1-4, to=1-5]
	\arrow["{(\eta,\text{id}_M)}"', from=1-4, to=2-4]
	\arrow["{\text{id}_M}"', from=1-4, to=2-5]
	\arrow["\mu", from=1-5, to=2-5]
	\arrow["\mu", from=2-1, to=2-2]
	\arrow["\mu"', from=2-4, to=2-5]
\end{tikzcd}}\\
\end{defn}

\begin{prp}{}{}\\
Let $(\mC,\otimes,I)$ be a monoidal category. Let $X\in\mC$ be an object. Then $X$ is a monoid object if and only if the Yoneda embedding $h_X:\mC\to\bold{Set}$ given by $C\mapsto\Hom_\mC(C,X)$ factors through the forgetful functor $\bold{Mon}\to\bold{Set}$. 
\end{prp}

\begin{defn}{Morphisms of Monoid Objects}{}\\
Let $(\mC,\otimes,I)$ be a monoidal category. Let $M,N\in\mC$ be monoid objects. Let $\varphi:M\to N$ be a morphsim in $\mC$. We say that $\varphi$ is a morphism of monoid objects if $$\varphi\circ\mu_M=\mu_N\circ(\varphi\otimes\varphi)$$
\end{defn}

\begin{lmm}{}{}\\
Let $(\mC,\otimes,I)$ be a monoidal category. Let $M,N\in\mC$ be monoid objects. Let $\varphi:M\to N$ be a morphsim of monoid objects. Then we have $$\varphi\circ\eta_M=\eta_N\circ\varphi$$
\end{lmm}

\begin{prp}{}{}\\
Let $(\mC,\otimes,I)$ be a monoidal category. Let $M,N\in\mC$ be monoid objects. Let $\varphi:G\to H$ be a morphsim in $\mC$. Then $\varphi$ is a morphism of monoid objects if and only if the Yoneda embedding $$h_\varphi(C):\Hom_\mC(C,G)\to\Hom_\mC(C,H)$$ is a  monoid homomorphism for all $C\in\mC$. 
\end{prp}

\begin{defn}{The Category of Monoid Objects}{}\\
Let $(\mC,\otimes,I)$ be a monoidal category. Define the category of monoid objects $\bold{Mon}(\mC)$ by the following data. 
\begin{itemize}
\item The objcets are the monoid objects of $\mC$. 
\item The morphisms are the morphisms of monoid objects. 
\item Composition is given by composition in $\mC$. 
\end{itemize}
\end{defn}

\begin{eg}{}{}\\
The following are true. 
\begin{itemize}
\item $\bold{Mon}(\bold{Set})=\bold{Mon}$. 
\item $\bold{Mon}(\bold{Mon})=\bold{CMon}$. 
\item $\bold{Mon}(\bold{Ab},\otimes_\Z,\Z)=\bold{Ring}$. 
\item Let $R$ be a commutative ring. Then $\bold{Mon}({_R}\bold{Mod},\otimes_R,R)=\bold{Alg}_R$. 
\item Let $R$ be a commutative ring. Then $\bold{Mon}({_R}\bold{GrMod},\otimes_R,R)=\bold{GrAlg}_R$.

\begin{defn}{Group Objects}{}\\
Let $(\mC,\otimes,1)$ be a monoidal category. A group object in $\mC$ is an monoid object $G\in\mC$ together a morphism $\text{inv}:G\to G$ called the inverse such that the following diagram commutes: \\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	G & {G\otimes G} \\
	{G\otimes G} & G
	\arrow["{(\text{id}_G,\text{inv})}", from=1-1, to=1-2]
	\arrow["{(\text{inv},\text{id}_G)}"', from=1-1, to=2-1]
	\arrow["{\eta\circ p}"', from=1-1, to=2-2]
	\arrow["\mu", from=1-2, to=2-2]
	\arrow["\mu"', from=2-1, to=2-2]
\end{tikzcd}}\\
where $p:G\to I$ is the unique terminal morphism. 
\end{defn}

\begin{prp}{}{}\\
Let $(\mC,\otimes,I)$ be a monoidal category. Let $(M,\mu,\eta)$ be a monoid object in $\mC$. Then the following are equivalent. 
\begin{itemize}
\item $M$ is a group object. 
\item The following diagram is a pullback square: \\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	{M\otimes M\otimes G} & {M\otimes M} \\
	{M\otimes M} & M
	\arrow["{\mu\otimes\text{id}_M}", from=1-1, to=1-2]
	\arrow["{\text{id}_M\otimes\mu}"', from=1-1, to=2-1]
	\arrow["\mu"', from=1-2, to=2-2]
	\arrow["\mu", from=2-1, to=2-2]
\end{tikzcd}}\\
\item The Yoneda embedding $h_M:\mC\to\bold{Set}$ given by $C\mapsto\Hom_\mC(C,M)$ factors through the forgetful functor $\bold{Grp}\to\bold{Set}$. 
\end{itemize}
\end{prp}

\begin{defn}{Morphisms of Group Objects}{}\\

\item $\varphi\circ\text{inv}_G=\text{inv}_H\circ\varphi$
\end{itemize}
\end{lmm}

\begin{prp}{}{}\\
Let $(\mC,\otimes,I)$ be a monoidal category. Let $G,H\in\mC$ be group objects. Let $\varphi:G\to H$ be a morphsim in $\mC$. Then $\varphi$ is a morphism of group objects if and only if the Yoneda embedding $$h_\varphi(C):\Hom_\mC(C,G)\to\Hom_\mC(C,H)$$ is a  group homomorphism for all $C\in\mC$. 
\end{prp}

\begin{itemize}
\item The objcets are the group objects of $\mC$. 
\item The morphisms are the morphisms of group objects.

\section{Comodules and Coalgebras}
\subsection{Coalgebras}
\begin{defn}{Coalgebras}{}\\
Let $R$ be a commutative ring. A coalgebra over $R$ is an $R$-module $C$ together with the following data. 
\begin{itemize}
\item An $R$-module homomorphism $\Delta:C\to C\otimes_RC$ called comultiplication. 
\item An $R$-module homomorphism $\varepsilon:C\to R$ called counit. 
\end{itemize}
such that the following diagrams are commutative: \\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	C & {C\otimes_RC} && C & {C\otimes_RC} \\
	{C\otimes_RC} & {C\otimes_RC\otimes_RC} && {C\otimes_RC} & C
	\arrow["\Delta", from=1-1, to=1-2]
	\arrow["\Delta"', from=1-1, to=2-1]

\subsection{Differential Graded Algebras}
\begin{defn}{Differential Graded Algebras}{}\\
Let $R$ be a commutative ring. A differential graded $R$-algebra consists of the following data. 
\begin{itemize}
\item A $\Z$-graded $R$-algebra $A$. 
\item An $R$-algebra homomorphism of degree $-1$ (called homological dgas) or of degree $1$ (called cohomological dgas). 
\end{itemize}
such that the following are true. 
\begin{itemize}
\item Differential: $d\circ d=0$. 
\item Graded Leibniz: $d(ab)=(da)b+(-1)^{\deg(a)}a(db)$ for $a,b$ homogeneous elements of $A$. 
\end{itemize}
\end{defn}

The differential property means that any (homological) differential graded algebra decomposes into a chain complex whose differential is given by $d$. 

\begin{lmm}{}{}\\
Let $R$ be a commutative ring. Let $(A^\bullet,d)\in\bold{Ch}^\bullet(R)$ be a cochain complex. Then $A^\bullet$ is a cohomological differential graded algebra if and only if it is a monoid object in $\bold{Ch}^\bullet(R)$. 
\end{lmm}

\begin{defn}{Homomorphisms of Differential Graded Algebras}{}\\
Let $R$ be a commutative ring. Let $A,B$ be cohomological differential graded algebras. Let $f:A\to B$ be a function. We say that $f$ is homomorphism if the following are true. 
\begin{itemize}
\item $f$ is a degree $0$ morphism of graded algebras. 
\item $f$ is a cochain map. 
\end{itemize}
\end{defn}

\begin{lmm}{}{}\\
Let $R$ be a commutative ring. Let $(A,d)$ be a cohomological differential graded algebra. Then $H^\ast(A,d)=\bigoplus_{n\in\Z}H^n(A,d)$ is a graded ring with multiplication defined by $$[u]\cdot[v]=[uv]$$ for $u\in H^n(A,d)$ and $v\in H^m(A,d)$. 
\begin{proof}\\
Notice that by the graded Leibniz rule, we have $d(uv)=(du)v+(-1)^nud(v)=0$ since $u$ and $v$ are cocycles. Hence $uv$ is also a cocycle. 
\end{proof}
\end{lmm}

\subsection{Massey Products}
\begin{defn}{Massey Triple Products}{}\\
Let $R$ be a commutative ring. Let $A$ be a differential graded algebra. Let $u\in A^p$, $v\in A^q$ and $w\in A^r$. Suppose that $[u]\cdot[v]=[v]\cdot[w]=0$. Define the Massey triple product of the three elements to be $$\langle[u],[v],[w]\rangle=\{[\overline{s}\cdot w+\overline{u}\cdot t]\in H^{p+q+r-1}(A)\;|\;s\in A^{p+q-1},t\in A^{q+r-1}\text{ and }ds=\overline{u}\cdot v,dt=\overline{v}\cdot w\}$$ where $\overline{a}=(-1)^{1+\deg(a)}a$ for any homogeneous element $a\in A$. 
\end{defn}

\begin{lmm}{}{}\\
Let $R$ be a commutative ring. Let $A$ be a differential graded algebra. Let $u\in A^p$, $v\in A^q$ and $w\in A^r$. Suppose that $[u]\cdot[v]=[v]\cdot[w]=0$. Then we have $$\langle[u],[v],[w]\rangle=\frac{H^{p+q+r-1}(A)}{[u]\cdot H^{q+r-1}(A)+H^{p+q-1}(A)\cdot[w]}$$
\end{lmm}

\begin{prp}{}{}\\

\begin{itemize}
\item $\langle u,v,w\rangle=(-1)^pz$. 
\item $\langle v,w,u\rangle=(-1)^{pq+pr+p}z$. 
\item $\langle w,u,v\rangle=0$. 
\end{itemize}
\end{prp}

\subsection{Koszul Algebras}
\begin{defn}{Koszul Algebras}{}\\
Let $k$ be a field. Let $R$ be a standard graded algebra over $k$. We say that $R$ is a Koszul algebra if $$\dim_k(\text{Tor}_i^R(M,k))=0$$ for all $i\neq j$. 
\end{defn}

Let $k$ be a field. Let $R$ be a Koszul algebra over $R$. Then $R$ admits a minimal graded free resolution $F_\bullet$ and by the constraint of being Koszul, we must have $F_i=R(-i)^{k_i}$ for some $k_i\in\N$. 

\begin{eg}{}{}\\
Let $k$ be a field. Then $k[x_1,\dots,x_n]$ is a Koszul algebra. 
\begin{proof}\\
The Koszul complex $K_\bullet(x_1,\dots,x_n,k)$ gives a minimal graded free resolution that one can check that the $i$th position is given by $R(-i)^{\binom{n}{i}}$. 
\end{proof}
\end{eg}

\begin{eg}{}{}\\
Let $k$ be a field. Then $\frac{k[x,y]}{(x,y)}$ is a Koszul algebra. 
\end{eg}

\begin{prp}{}{}\\
Let $k$ be a field of characteristic $0$. Let $V$ be a finite dimensional vector space over $k$. Then the following are true. 
\begin{itemize}
\item $S(V)$ is a Koszul algebra. 
\item $\Lambda(V)$ is a Koszul algebra. 
\end{itemize}
\end{prp}

\subsection{Linear Resolutions}
We generalize the condition that defines Koszul algebras. 

\begin{defn}{Linear Resolutions}{}\\
Let $k$ be a field. Let $R$ be a standard graded algebra over $k$. Let $M$ be a finitely generated graded $R$-module. Let $F_i=\bigoplus_{j\in\Z}R(-j)^{b_{i,j}(M)}$ be the minimal graded free resolution of $M$. We say that it is linear if there exists $d\in\N$ such that $$b_{i,j}(M)=0$$ for all $j\neq i+d$. 
\end{defn}

\begin{lmm}{}{}\\
Let $k$ be a field. Let $R$ be a standard graded algebra over $k$. Let $M$ be a finitely generated graded $R$-module. If $M$ admits a linear resolution $F_i=\bigoplus_{j\in\Z}R(-j)^{b_{i,j}(M)}$ such that $b_{i,j}(M)=0$ for all $j\neq i+d$, then $M$ is generated at a single degree $d$. 
\end{lmm}

\begin{defn}{Minimal Resolutions of Type $N_{d,p}$}{}\\
Let $R$ be a Notherian $\N$-graded ring such that $R_0$ is a field. Let $M$ be a finitely generated $R$-module. Let $F_i=\bigoplus_{j\in\Z}R(-j)^{b_{i,j}(M)}$ be a minimal graded free resolution of $M$. We say that $M$ is of type $N_{d,p}$ if the following are true. 
\begin{itemize}
\item $M$ is generated in degree $d$. 
\item $b_{i,j}(M)=0$ for all $j\neq i+d$ when $i<p$. 
\end{itemize}
\end{defn}

\begin{eg}{}{}\\
Let $R$ be a Notherian $\N$-graded ring such that $R_0$ is a field. Let $M$ be a finitely generated $R$-module. Let $F_i=\bigoplus_{j\in\Z}R(-j)^{b_{i,j}(M)}$ be a minimal graded free resolution of $M$. 
\begin{itemize}
\item $M$ is of type $N_{d,1}$ if it is generated in degree $d$. 
\item $M$ is of type $N_{d,2}$ if it is generated in degree $d$ and it is linearly presented. 
\item $M$ has a linear resolution if and only if $M$ is of type $N_{d,p}$ for all $p$. 
\end{itemize}
\end{eg}

\begin{defn}{Almost Linear Resolutions}{}\\
Let $R$ be a Notherian $\N$-graded ring such that $R_0$ is a field. Let $M$ be a finitely generated $R$-module. We say that $M$ has an almost linear resolution if $M$ is of type $N_{d,\text{pd}_R(M)}$. 
\end{defn}

\begin{defn}{Componentwise Linear}{}\\
Let $R$ be a Notherian $\N$-graded ring such that $R_0$ is a field. Let $M$ be a finitely generated $R$-module. 
\begin{itemize}
\item We say that $M$ is componentwise linear if $\langle M_d\rangle$ has a liner resolution for all $d$. 
\item We say that $M$ is componentwise of type $N_{d,p}$ if $\langle M_d\rangle$ satisfies $N_{d,p}$ for all $d$. 
\end{itemize}
\end{defn}

\subsection{Koszul Modules}
\begin{defn}{Koszul Modules}{}\\
Let $k$ be a field. Let $R$ be a standard graded algebra over $k$. Let $M$ be a finitely generated graded $R$-module. We say that $M$ is a Koszul module if $M$ admits a componentwise linear resolution. 
\end{defn}

The notion of Koszul modules generalize Koszul algebras. Let $k$ be a field. Let $R$ be a standard graded algebra over $k$. Then clearly $R$ is a Koszul algebra if and only if $k$ is a Kosuzl $R$-module. \\

There is an equivalent way to characterize Koszul modules. 

\begin{defn}{m-Adic Filtration of the Minimal Graded Free Resolution}{}\\
Let $k$ be a field. Let $R$ be a standard graded algebra over $k$. Let $M$ be a finitely generated graded $R$-module. Let $(F_\bullet,d)$ be the minimal graded free resolution of $M$. Define the $m$-adic filtration of $F_\bullet$ to be the subcomplexes given by $$(F^jF)_i=m^{j-i}F_i$$ for all $i\in\Z$ and $j\in\N$. 
\end{defn}

We can then form the associated graded free complex. 

\begin{defn}{The Linear Part}{}\\
Let $k$ be a field. Let $R$ be a standard graded algebra over $k$. Let $M$ be a finitely generated graded $R$-module. Let $(F_\bullet,d)$ be the minimal graded free resolution of $M$. Define the linear part of $F_\bullet$ to be the associated graded complex $$\text{lin}(F)=\text{gr}(F^\bullet F)$$
\end{defn}

Let $M$ be a graded $R$-module. We denote $$M_{\langle\leq n\rangle}=\left\langle\bigoplus_{i=0}^nM_i\right\rangle$$ We can describe the linear part explicitly using the following construction:
\begin{itemize}
\item Define the $d$-linear strand of $F$ to be the chain complex $$(F_{\langle d\rangle})_n=\frac{(F_n)_{\langle\leq d+n\rangle}}{(F_n)_{\langle\leq d-1+n\rangle}}=\frac{\langle\bigoplus_{i=0}^{d+n}(F_n)_i\rangle}{\langle\bigoplus_{i=0}^{d-1+n}(F_n)_i\rangle}$$
\item Then we have $$\text{lin}(F)=\bigoplus_{d\in\Z}F_{\langle d\rangle}$$
\end{itemize}
Intuitively, $\text{lin}(F)$ is simply the complex with $\text{lin}(F)_n=F_n$ and with differential obtained by deleting all nonlinear entries from homogeneous matrices representing the differential of F.

\begin{thm}{Romer 2001}{}\\
Let $k$ be a field. Let $R$ be a standard graded algebra over $k$. Let $M$ be a finitely generated graded $R$-module. Then $M$ is a Koszul module if and only if $M$ admits a minimal graded free resolution $F_\bullet$ such that $\text{lin}(F_\bullet)$ is acyclic. 
\end{thm}

\begin{defn}{m-Adic Filtration of the Minimal Free Resolution}{}\\
Let $(R,m)$ be a Noetherian local ring. Let $M$ be a finitely generated $R$-module. Let $(F_\bullet,d)$ be the minimal free resolution of $M$. Define the $m$-adic filtration of $F_\bullet$ to be the subcomplexes given by $$(F^jF)_i=m^{j-i}F_i$$ for all $i\in\Z$ and $j\in\N$. 
\end{defn}

We can then form the associated graded free complex. 

\begin{defn}{The Linear Part}{}\\
Let $(R,m)$ be a Noetherian local ring. Let $M$ be a finitely generated $R$-module. Let $(F_\bullet,d)$ be the minimal free resolution of $M$. Define the linear part of $F_\bullet$ to be the associated graded complex $$\text{lin}^R(F)=\text{gr}(F^\bullet F)$$
\end{defn}

Explicitly, we have $$\text{lin}^R(F)_n=\text{gr}_m(F_n)(-n)$$

\begin{defn}{Koszul Modules}{}\\

\begin{prp}{}{}\\
Let $(R,m)$ be a Noetherian local ring. Let $M$ be a finitely generated $R$-module. Then the following are equivalent. 
\begin{itemize}
\item $M$ is Koszul. 
\item $\text{lin}^R(F)$ is a minimal graded free resolution of $\text{gr}_m(M)$ over $\text{gr}_m(R)$. 
\item The $\text{gr}_m(R)$-module $\text{gr}_m(M)$ admits a linear resolution. 
\end{itemize}
\end{prp}

Let $k$ be a field. Let $R$ be a standard graded ring over $k$. Let $M$ be a finitely generated graded $R$-module. Then we have $\text{gr}_m(M)\cong M$ and $\text{gr}_m(R)\cong R$ so the above proposition generalizes the notion of Koszul modules from graded rings to Noetherian local rings. 

\begin{defn}{Koszul Local Rings}{}\\
Let $(R,m)$ be a Noetherian local ring. We say that $R$ is Koszul if $R/m$ is a Koszul $R$-module. 
\end{defn}
