\section{Root Systems}
\subsection{Abstract Root Systems and Simple Systems}
\begin{defn}{Abstract Root Systems}{}\\
Let $E$ be a Euclidean space. Let $\Phi\subseteq E$ be a subset. We say that $\Phi$ is a root system if the following are true. 
\begin{itemize}
\item $\Phi$ is finite and $0\notin\Phi$. 
\item If $x\in\Phi$ and $\lambda x\in\Phi$, then $\lambda=\pm1$. 
\item If $x,y\in\Phi$, then $S_x(y)\in\Phi$. 
\end{itemize}
\end{defn}

Note: Our definition of root systems does not include the requirement that $\Phi$ must span $E$, and that $2\frac{\langle x,y\rangle}{\langle y,y\rangle}\in\Z$. 

\begin{defn}{Simple Systems}{}\\
Let $E$ be a Euclidean space. Let $\Phi$ be a root system. Let $\Delta\subseteq\Phi$ be a subset. We say that $\Delta$ is a simple system for $\Phi$ if the following are true. 
\begin{itemize}
\item $\Delta$ is a basis for $\R\langle\Phi\rangle$. 
\item For any $r\in\Phi$, there is a decomposition $$r=\sum_{b\in\Delta}k_b\cdot b$$ where $k_b\in\R$ are either all positive or all negative. 
\end{itemize}
\end{defn}

\begin{lmm}{}{}\\
Let $E$ be a Euclidean space. Let $\Phi$ be a root system. Let $\Delta$ be a simple system of $\Phi$. Let $x,y\in\Phi$ be distinct. Then we have $$\langle x,y\rangle<0$$
\end{lmm}

\begin{lmm}{}{}\\
Let $E$ be a Euclidean space. Let $\Phi$ be a root system. Let $\Delta$ be a simple system of $\Phi$. Then we have $$G_\Phi=\langle S_x\;|\;x\in\Delta\rangle$$
\end{lmm}

\begin{crl}{}{}\\
Let $E$ be a Euclidean space. Let $\Phi$ be a root system. Let $\Delta$ be a simple system of $\Phi$. Then for any $x\in\Phi$, there exists $y\in\Phi$ and $T\in G_\Phi$ such that $T(y)=x$. 
\end{crl}

\subsection{Crystallographic Root Systems}
\begin{defn}{Crystallographic Root Systems}{}\\
Let $E$ be a Euclidean space. Let $\Phi$ be a root system. We say that $\Phi$ is crystallographic if the following are true. 
\begin{itemize}
\item $\Phi$ spans $E$. 
\item For any $x,y\in\Phi$, we have $2\frac{\langle x,y\rangle}{\langle y,y\rangle}\in\Z$. 
\end{itemize}
\end{defn}

\begin{eg}{}{} \\Let $\{e_1,\dots,e_{n+1}\}$ be the standard basis vectors of $\R^{n+1}$. Let $E$ be the inner product space $$E=\{(c_1,\dots,c_{n+1})\subseteq\R^{n+1}\;|\;c_1+\dots+c_n=0\}\cong\R^n$$ equipped with the standard inner product of $\R^{n+1}$. Then the set $$R=\{e_i-e_j\;|\;1\leq i\neq j\leq n+1\}$$ is a root system of $E$. 
\end{eg}

\begin{lmm}{}{} \\Let $E$ be a finite dimensional inner product space over $\R$. Let $R$ be a root system of $E$. Let $\alpha,\beta\in R$. Then the following are true. 
\begin{itemize}
\item The value of $\langle\alpha,\beta\rangle\langle\beta,\alpha\rangle$ must lie in the set $\{0,1,2,3\}$
\item The angle between $\alpha$ and $\beta$ must lie in the set $$\left\{\frac{\pi}{2},\frac{\pi}{3},\frac{2\pi}{3},\frac{\pi}{4},\frac{3\pi}{4},\frac{\pi}{6},\frac{5\pi}{6}\right\}$$
\end{itemize}
\end{lmm}

We can draw a table of the possible values of $\langle\alpha,\beta\rangle$ and the corresponding angle between $\alpha$ and $\beta$: 

\begin{table}[!h]
\centering
\begin{tabular}{|c|c|c|c|}
\hline
$\langle\alpha,\beta\rangle\langle\beta,\alpha\rangle$ & $\langle\alpha,\beta\rangle$ & $\langle\beta,\alpha\rangle$ & $\theta$         \\ \hline
$0$                                                    & $0$                          & $0$                          & $\frac{\pi}{2}$  \\[1.5ex] \hline
\multirow{2}{*}{$1$}                                   & $1$                          & $1$                          & $\frac{\pi}{3}$  \\[1.5ex] \cline{2-4} 
                                                       & $-1$                         & $-1$                         & $\frac{2\pi}{3}$ \\[1.5ex] \hline
\multirow{2}{*}{$2$}                                   & $1$                          & $2$                          & $\frac{\pi}{4}$  \\[1.5ex] \cline{2-4} 
                                                       & $-1$                         & $-2$                         & $\frac{3\pi}{4}$ \\[1.5ex] \hline
\multirow{2}{*}{$3$}                                   & $1$                          & $3$                          & $\frac{\pi}{6}$  \\[1.5ex] \cline{2-4} 
                                                       & $-1$                         & $-3$                         & $\frac{5\pi}{6}$ \\[1.5ex] \hline
\end{tabular}
\end{table}

Here we assume $\|\alpha\|\leq\|\beta\|$ so we exclude for instance the case $\langle\alpha,\beta\rangle=2$ and $\langle\beta,\alpha\rangle=1$. 

\begin{eg}{}{} \\The only root system in $\R^1$ is given by $R=\{\alpha,-\alpha\}$ for any $\alpha\in\R$. 
\end{eg}

\begin{eg}{}{} \\There are exactly four possible root systems in $\R^2$. Let $\alpha,\beta\in\R^2$ be linearly independent such that $\|\alpha\|\leq\|\beta\|$. Let $\theta$ be the angle between $\alpha$ and $\beta$. 
\begin{itemize}
\item $A_1\times A_1=\{\pm\alpha,\pm\beta\}$ and $\theta=\frac{\pi}{2}$. 
\item $A_2=\{\pm\alpha,\pm\beta,\pm(\alpha+\beta)\}$ and $\theta=\frac{2\pi}{3}$. 
\item $B_2=\{\pm\alpha,\pm\beta,\pm(\alpha+\beta),\pm(2\alpha+\beta)\}$ and $\theta=\frac{3\pi}{4}$. 
\item $G_2=\{\pm\alpha,\pm\beta,\pm(\alpha+\beta),\pm(2\alpha+\beta),\pm(3\alpha+\beta),\pm(3\alpha+2\beta)\}$
\end{itemize}
\end{eg}

\begin{defn}{Base of a Crystallographic Root System}{}\\
Let $E$ be a Euclidean space. Let $\Phi$ be a root system. Let $B\subseteq\Phi$ be a subset. We say that $B$ is a base for $\Phi$ if the following are true. 
\begin{itemize}
\item $B$ is a basis for $\R\langle\Phi\rangle=E$. 
\item For any $r\in\Phi$, there is a decomposition $$r=\sum_{b\in\Delta}k_b\cdot b$$ where $k_b\in\Z$ are either all positive or all negative. 
\end{itemize}
\end{defn}

\begin{prp}{}{}\\
Let $E$ be a finite dimensional inner product space over $\R$. Let $\Phi$ be a crystallographic root system of $E$. Then $\Phi$ admits a base. 
\end{prp}

\subsection{Decomposing Root Systems into Irreducibles}
\begin{defn}{Irreducible Root Systems}{} \\Let $(E,(-,-))$ be a finite dimensional inner product space over $\R$. Let $R$ be a crystallographic root system of $E$. We say that $R$ is reducible if there exists two disjoint crystallographic root systems $R_1$ and $R_2$ of $E$ such that $(r_1,r_2)=0$ for $r_1\in R_1$ and $r_2\in R_2$. Otherwise, we say that $R$ is irreducible. 
\end{defn}

\begin{prp}{}{}\\
Let $E$ be a finite dimensional inner product space over $\R$. Let $R$ be a crystallographic root system of $E$. Then the following are true. 
\begin{itemize}
\item There is a decomposition $$R=R_1\amalg\cdots\amalg R_k$$ where each $R_i$ is an irreducible crystallographic root system of $E_i=\R\langle R_i\rangle$
\item There is a decomposition $$E=E_1\oplus\cdots\oplus E_k$$ where $E_i$ is orthogonal to $E_j$ for $i\neq j$. 
\end{itemize}
\end{prp}

\begin{defn}{Isomorphic Root Systems}{} \\Let $E,F$ be a finite dimensional inner product spaces over $\R$. Let $R$ and $S$ be crystallographic root systems of $E$ and $F$ respectively. We say that $R$ and $S$ are isomorphic if there exists an isomorphism $$\phi:E\to F$$ of vector spaces such that the following are true. 
\begin{itemize}
\item $\phi(R)=S$
\item For all $a,b\in R$, $\langle\phi(a),\phi(b)\rangle=\langle a,b\rangle$. 
\end{itemize}
\end{defn}

\begin{eg}{}{} \\The following are true. 
\begin{itemize}
\item $A_1\times A_1=\{\pm\alpha,\pm\beta\}$ is reducible into the product of $A_1$ and $A_1$. 
\item $A_2,B_2,G_2$ are irreducible. 
\end{itemize}
\end{eg}

\subsection{The Weyl Group of a Root System}
\begin{defn}{Weyl Group of a Root System}{}\\
Let $E$ be a Euclidean space. Let $\Phi$ be a crystallographic root system. Define the Weyl group of $\Phi$ to be $$W(\Phi)=G_\Phi=\langle S_x\;|\;x\in\Phi\rangle$$
\end{defn}

\section{Dynkin Diagrams}
\subsection{Dynkin Diagrams and Cartan Matrices}
\begin{defn}{The Dynkin Diagram of a Root System}{} \\Let $E$ be a finite dimensional inner product space over $\R$. Let $R$ be a crystallographic root system of $E$. Let $B$ be a base of $R$. Define a graph $\Delta(R)$ as follows. 
\begin{itemize}
\item There is one vertex $v_b$ for each $b\in B$
\item For any two vertices $v_a$ and $v_b$, there are $d_{a,b}=\langle a,b\rangle\langle b,a\rangle$ number of undirected edges between $v_a$ and $v_b$. 
\end{itemize}
\end{defn}

\begin{prp}{}{} \\Let $E,F$ be a finite dimensional inner product spaces over $\R$. Let $R$ and $S$ be crystallographic root systems of $E$ and $F$ respectively. Then $R\cong S$ if and only if $\Delta(R)\cong\Delta(S)$. 
\end{prp}

\begin{prp}{}{} \\Let $E$ be a finite dimensional inner product space. Let $R$ be a crystallographic root system of $E$. Then $R$ is irreducible if and only if $\Delta(R)$ is a connected graph. 
\end{prp}

\begin{defn}{The Cartan Matrix}{} \\Let $E$ be a finite dimensional inner product space over $\R$. Let $R$ be a crystallographic root system of $E$. Let $B=\{\alpha_1,\dots,\alpha_r\}$ be a base of $R$. Define the Cartan matrix of $R$ to be $$\begin{pmatrix}
\langle\alpha_1,\alpha_1\rangle & \cdots & \langle\alpha_1,\alpha_r\rangle\\
\vdots & \ddots & \vdots\\
\langle\alpha_r,\alpha_1\rangle & \cdots & \langle\alpha_r,\alpha_r\rangle
\end{pmatrix}$$
\end{defn}

\begin{prp}{}{} \\Let $E$ be a finite dimensional inner product space over $\R$. Let $R$ be a crystallographic root system of $E$. Then the Cartan matrix of $R$ is unique up to reordering. 
\end{prp}

\subsection{Classification of Irreducible Root Systems}
\begin{thm}{Classification of Irreducible Root Systems}{} \\Let $E$ be a finite dimensional inner product space over $\R$. Let $R$ be an irreducible crystallographic root system of $E$. Then $\Delta(R)$ is isomorphic to one of the following: 
\begin{itemize}
\item $A_n$ for $n\geq 1$. 
\item $B_n$ for $n\geq 2$. 
\item $C_n$ for $n\geq 3$. 
\item $D_n$ for $n\geq 4$. 
\item $G_2,F_4,E_6,E_7,E_8$. 
\end{itemize}
Conversely, any of the above types can be realized as the Dynkin diagram of some irreducible root system. 
\end{thm}

\section{Reflection Groups}
\subsection{Reflections}
\begin{defn}{Reflections}{}\\
Let $E$ be a Euclidean space. Let $T:E\to E$ be a linear map. We say that $T$ is a reflection if $$\{v\in E\;|\;T(v)=v\}$$ is an $(n-1)$-dimensional subspace of $E$. 
\end{defn}

\begin{defn}{The Reflection Associated to a Vector}{}\\
Let $E$ be a Euclidean space. Let $x\in E$. Define the reflection associated to $E$ to be the map $S_x:E\to E$ given by $$S_x(z)=z-2\frac{\langle x,z\rangle}{\langle x,x\rangle}x$$
\end{defn}

\begin{lmm}{}{}\\
Let $E$ be a Euclidean space. Let $x\in E$. Then the following are true. 
\begin{itemize}
\item $S_x$ is a reflection. 
\item $S_x=S_{\lambda x}$ for any $\lambda\in\R\setminus\{0\}$. 
\end{itemize}
\end{lmm}

\begin{defn}{Root of a Reflection}{}\\
Let $E$ be a Euclidean space. Let $T:E\to E$ be a reflection. Let $x\in E$. We say that $x$ is a root of $T$ if we have $T=S_x$. 
\end{defn}

\begin{prp}{}{}\\
Let $E$ be a Euclidean space. Let $T:E\to E$ be a reflection. Then the following are true. 
\begin{itemize}
\item $T$ admits a root. 
\item The set of roots of $T$ is given by $$\{v\in E\;|\;T(v)=v\}^\perp$$
\end{itemize}
\end{prp}

\begin{lmm}{}{}\\
Let $E$ be a Euclidean space. Let $T:E\to E$ be a reflection. Then $T^2=\text{id}_E$. 
\end{lmm}

\begin{defn}{Finite Reflection Groups}{}\\
Let $E$ be a Euclidean space. Let $G<O(E)$. We say that $G$ is a finite reflection group if the following are true. 
\begin{itemize}
\item $G$ is finite. 
\item $G$ is generated by reflections. 
\end{itemize}
\end{defn}

\begin{eg}{}{}\\
Consider the representation $$D_{2n}\cong\left\langle\begin{pmatrix}
\cos(2\pi/n) & -\sin(2\pi/n)\\
\sin(2\pi/n) & \cos(2\pi/n)
\end{pmatrix}, \begin{pmatrix}
-1 & 0\\
0 & 1
\end{pmatrix}\right\rangle\leq O(\R^2)$$ Then $D_{2n}$ is a finite reflection group. 
\end{eg}

\begin{eg}{}{}\\
Consider $S_n$ as a subgroup of $O(\R^n)$ via the injection $$\sigma\mapsto(e_i\mapsto e_{\sigma(i)})$$ Then $S_n$ is a finite reflection group. 
\begin{proof}\\
Indeed, $S_n$ is generated by transpositions, and transpositions are reflections since for a transposition $\begin{pmatrix}i & j\end{pmatrix}$, the corresponding element in $O(\R^n)$ fixes $e_k$ for $k\neq i,j$ and transposes $e_i-e_j$ with $e_j-e_i$. 
\end{proof}
\end{eg}

\begin{defn}{Root Systems of a Finite Reflection Group}{}\\
Let $E$ be a Euclidean space. Let $G$ be a finite reflection group of $E$. Define the root system of $G$ to be $$\Phi_G=\{x\in E\;|\;S_x\in G\text{ and }\|x\|=1\}$$
\end{defn}

\begin{lmm}{}{}\\
Let $E$ be a Euclidean space. Let $G$ be a finite reflection group of $E$. Then $\Phi_G$ is a root system. 
\end{lmm}

\subsection{The Reflection Group of a Root System}
\begin{defn}{The Finite Reflection Group of a Root System}{}\\
Let $E$ be a Euclidean space. Let $\Phi\subseteq E$ be a root system. Define the finite reflection group associated to $\Phi$ to be $$G_\Phi=\langle S_x\;|\;x\in\Phi\rangle$$
\end{defn}

\begin{lmm}{}{}\\
Let $E$ be a Euclidean space. Let $\Phi\subseteq E$ be a root system. Then $G_\Phi$ is a finite reflection group. 
\end{lmm}

Note: There can be multiple different root systems that give rise to the same finite reflection group. 
Note: $\{S_x\;|\;x\in\Phi_G\}$ might not be the complete set of reflections of $G$.

\section{Coxeter Groups}
\subsection{Coxeter Groups}
\begin{defn}{Coxeter Groups}{}\\
Let $G$ be a group. We say that $G$ is a Coxeter group if $G$ admits a presentation $$G\cong\langle r_1,\dots,r_n\;|\;(r_ir_j)^{m_{ij}}\rangle$$ where $m_{ii}=1$ and $m_{ij}\geq 2$ or $\infty$ for $i\neq j$. 
\end{defn}

\begin{prp}{}{}\\
Let $E$ be a Euclidean space. Let $G$ be a finite reflection group of $E$. Then $G$ is a Coxeter group. 
\end{prp}
