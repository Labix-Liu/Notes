\section{The General Theory of Sheaves}
\subsection{Basic Definition of Presheaves}
As with how we equipped to each variety $V$ its coordinate ring $k[V]$ which are functions on $V$, we want to equip to each spectrum some ring which are functions on them. It will not make sense that we can define functions on spectrums immediately. \\

In its full generality, sheaves are designed to encode all local information into one compact notation. It often has uses in complex geometry for its differential forms and smooth functions. In algebraic geometry it is fundamental for redefining the notion of a variety. 

\begin{defn}{The Category of Open Sets}{} Let $(X,\mT)$ be a topological space. Define the category $\bold{Open}(X)$ to consist of the following data. 
\begin{itemize}
\item The objects $\text{Ob}(\bold{Open}(X))=\mT$ are the open sets of $X$
\item For $U,V\subseteq X$ two open sets of $X$, the morphisms $$\Hom_{\bold{Open}(X)}(U,V)=\begin{cases}
\{\iota:U\hookrightarrow V\text{ the inclusion map}\} & \text{ if } U\subseteq V\\
\emptyset & \text{otherwise}
\end{cases}$$
\item Composition is given as the composition of functions. 
\end{itemize}
\end{defn}

Recall from Category Theory 1 that a presheaf on $\mC$ is just a contravariant functor $\mC\to\bold{Set}$. Using the category of open sets we can now define presheaves on spaces. 

\begin{defn}{Presheaves on a Space}{}\\
Let $X$ be a space. Let $\mC$ be a category. A $\mC$-presheaf on $X$ is a contravariant functor $$\mF:\bold{Open}(X)\to\mC$$
\end{defn}

As per the definition of a functor, for each inclusion map $\iota:V\to U$ of open sets in $X$, passing through a presheaf $\mF$ gives a unique morphism $\mF(\iota):\mF(U)\to\mF(V)$. We often denote $\mF(\iota)$ as $\text{res}_V^U$ or $(-)|_V$. 

\begin{eg}{The Constant Presheaf}{}\\
Let $X$ be a space. Let $S$ be a set. Define the constant presheaf with value $S$ to be the presheaf $\underline{S}_\text{pre}$ defined as follows. 
\begin{itemize}
\item For each open set $U\subseteq X$, we have $$\underline{S}_\text{pre}(U)=S$$
\item For each inclusion $V\subseteq U$ of open sets, define $$\text{res}_V^U=\text{id}_S$$
\end{itemize}
\end{eg}

\begin{defn}{The Stalk of a Presheaf}{} Let $\mF$ be a presheaf on a topological space $(X,\mathcal{T})$. Let $p\in X$. Consider the full subcategory $\mJ_p$ of $\bold{Open}(X)$ consisting of open sets in $X$ that contain $p$. Define the stalk of $\mF$ at $p$ to be the colimit $$\mF_p=\underset{\mJ_p}{\text{colim}}\mF(V_p)$$ 
\end{defn}

\begin{defn}{Morphism of Presheaves}{}\\
Let $X$ be a topological space. Let $\mC$ be a category. Let $\mF,\mG$ be $\mC$-presheaves on $X$. A morphism $\varphi:\mF\to\mG$ is a natural transformation of functors. 
\end{defn}

Notice that the natural transformation $\phi$ here takes every open set $U$ and maps it to a group homomorphism $\phi(U):\mF(U)\to\mG(U)$ if $\mF$ and $\mG$ are presheaves with values in $\bold{Grp}$. 

\begin{defn}{Induced Map of Stalks}{}\\
Let $X$ be a topological space. Let $\mC$ be a category. Let $\mF,\mG$ be $\mC$-presheaves on $X$. Let $p\in X$. Define the induced map of stalks $$\phi_p:\mF_p\to\mG_p$$ to be the map induced by the universal product of colimits. 
\end{defn}

\begin{defn}{Isomorphism of Presheaves}{}\\
Let $X$ be a topological space. Let $X$ be a topological space. Let $\mC$ be a category. Let $\mF,\mG$ be $\mC$-presheaves on $X$. Let $\varphi:\mF\to\mG$ be a morphism of presheaves. We say that $\varphi$ is an isomorphism if $\varphi$ is a natural isomorphism. In this case, we say that $\mF$ and $\mG$ are isomorphic. 
\end{defn}

\subsection{Sheaves with Values in a Set}
\begin{defn}{Sheaves}{} Let $X$ be a space. A $\bold{Set}$-sheaf on $X$ is a $\bold{Set}$-presheaf $\mF:\bold{Open}\to\bold{Sets}$ satisfying two additional properties
\begin{itemize}
\item Identity: If $\{U_i|i\in I\}$ is an open cover of $U$ and $\phi_1,\phi_2\in\mF(U)$ and $\phi_1|_{U_i}=\phi_2|_{U_i}$ for all $i$, then $\phi_1=\phi_2$
\item Gluing: If $\{U_i|i\in I\}$ is an open cover of $U$ and $\phi_i\in\mF(U_i)$ for all $i\in I$ such that $\phi_i|_{U_i\cap U_j}=\phi_j|_{U_i\cap U_j}$ for all $i,j\in I$, then there is exists some $\phi\in\mF(U)$ such that $\phi|_{U_i}=\phi_i$ for all $i\in I$. 
\end{itemize}
\end{defn}

\begin{eg}{}{}\\
Let $X,Y$ be spaces. Define a presheaf $\mF$ on $X$ by $$\mF(U)=\{f:U\to Y\;|\;f\text{ is continuous }\}$$ for $U\subseteq X$ an open set, together with restriction maps. Then $\mF$ is a sheaf. 
\begin{proof}\\
Identity: Suppose that $f,g\in\mF(U)$ for some open set $U\subseteq X$ and $f|_{U_i}=g|_{U_i}$ for $\{U_i\;|\;i\in I\}$ an open cover of $U$. Then for any $x\in U$, there exists $i\in I$ such that $x\in U_i$, and so $$f(x)=f|_{U_i}(x)=g|_{U_i}(x)=g(x)$$ Hence $f=g$. \\~\\

Gluing: Suppose that $f_i\in\mF(U_i)$ for $\{U_i\;|\;i\in I\}$ an open cover of an open set $U\subseteq X$ such that $f_i|_{U_i\cap U_j}=f_j|_{U_i\cap U_j}$. Define $f\in\mF(U)$ by $f(x)=f_i(x)$ for $x\in U_i$. This is well defined since $\{f_i\}$ agree on intersections. It remains to show that $f$ is continuous. Let $W\subseteq Y$ be an open set. I claim that $f^{-1}(W)=\bigcup_{i\in I}f_i^{-1}(W)$. Let $x\in f^{-1}(W)$. Then $f(x)\in W$. Since $x\in W\subseteq U=\bigcup_{i\in I}U_i$, there exists $i\in I$ such that $x\in U_i$. Hence $f_i(x)=f(x)\in W$. Thus $x\in f_i^{-1}(W)$. Conversely, suppose that $x\in f_i^{-1}(W)$. Then clearly $f(x)=f_i(x)\in W$ so that $x\in f^{-1}(W)$. Thus $f^{-1}(W)=\bigcup_{i\in I}f_i^{-1}(W)$. Since each $f_I$ is continuous and $W$ is an open set, $f_i^{-1}(W)$ is open and so the arbitrary union of $f_i^{-1}(W)$ is open. Hence $f^{-1}(W)$ is open. 
\end{proof}
\end{eg}

\begin{eg}{Locally Constant Sheaf}{}\\
Let $X$ be a space. Let $S$ be a set. Define the locally constant presheaf $\underline{S}$ of $X$ with values in $S$ as follows. 
\begin{itemize}
\item For each open set $U\subseteq X$, define $$\underline{S}(U)=\{f:U\to S\;|\;f\text{ is continuous where }S\text{ has the discrete topology }\}$$
\item For each inclusion $V\subseteq U$ of open subsets, define the induced map $\underline{S}(U)\to\underline{S}(V)$ by $f\mapsto f|_V$. 
\end{itemize}
The following are true regarding the locally constant sheaf. 
\begin{itemize}
\item $\underline{S}$ is a sheaf. 
\item Let $U\subseteq X$ be an open set. Let $\mB$ be the set of connected components of $U$. Then $$\underline{S}(U)\cong\prod_{C\in\mB}S$$
\item For all $p\in X$, $\underline{S}_p=S$. 
\end{itemize}
\begin{proof}\\
Let $Y$ be a connected component of $U$. Let $f\in\underline{S}(Y)$. I claim that $f$ is constant. Let $y_0\in Y$. Define $A=\{y\in Y\;|\;f(y)=f(y_0)\}=f^{-1}(f(y_0))$ and $B=\{y\in Y\;|\;f(y)\neq f(y_0)\}$. Clearly $Y=A\cup B$ and $A\cap B=\emptyset$. Both $A$ and $B$ are open sets since $S$ has a discrete topology. Since $Y$ is connected and $A$ is non-empty, this implies that $B=\emptyset$. Hence $Y=A$ and $f$ is constant on $Y$. Thus there is a bijection $\underline{S}(Y)\cong S$. \\

Define a function $\underline{S}(U)\to\prod_{C\in\mB}S$ as follows. For each connected component $C\in\mB$ of $U$, choose $x_C\in C$. Then the map is defined by $f\mapsto(f(x_C))$. This map is injective since $f(x_C)=g(x_C)$ implies that $f(C)=g(C)$ for all connected components $C\in\mB$. Hence $f=g$. It is surjective since for any choice of sequence $(x_C)_{C\in\mB}$, the map $f:U\to S$ defined by $f(x)=x_C$ for $x\in C$ is continuous because $S$ is discrete and $\{x_C\}$ is a closed set, and $f^{-1}(x_C)=C$ is a connected component and so is a closed set. 
\end{proof}
\end{eg}

\begin{prp}{Gluing Sheaves}{}\\
Let $X$ be a space. Let $\{U_i\;|\;i\in I\}$ be an open cover of $X$. Let $\mF_i$ be a sheaf of sets on $U_i$. Let $\phi_{i,j}:\mF_i|_{U_i\cap U_j}\to\mF_j|_{U_i\cap U_j}$ be a collection of morphism of sheaves for ranging $i,j\in I$ such that the following are true. 
\begin{itemize}
\item For each $i,j\in I$, $\phi_{i,j}$ is an isomorphism. 
\item For each $i,j,k\in I$, we have $\phi_{j,k}\circ\phi_{i,j}=\phi_{i,k}$. 
\end{itemize}
Then there exists a unique (up to unique isomoprhism) sheaf of sets $\mF$ on $X$ such that there are isomorphisms $\mF|_{U_i}\overset{\cong}{\to}\mF_i$ for all $i\in I$. 
\end{prp}

\subsection{Sheaves with Values in a Category}
Presheaves are easily generalized to take values in an arbitrary category because they are simply defined to be contravariant functors. 

\begin{defn}{Presheaves with Values in a Category}{} Let $X$ be a space. Let $\mC$ be a category. A presheaf on $X$ with values in $\mC$ is a contravariant functor $$\mF:\bold{Open}(X)\to\mC$$
\end{defn}

However, it is hard to define sheaves with values on an arbitrary category $\mC$. This is because in our definition of sheaves on sets, we made use of the fact that morphisms in $\bold{Set}$ have a well defined notion of restriction. We will define sheaves on a general category by first considering sheaves with an underlying set. 

\begin{thm}{}{} Let $X$ be a space. Let $\mF$ be a presheaf with values in $\bold{Set}$. Then the following are equivalent characterizations. 
\begin{itemize}
\item $\mF$ is a sheaf
\item For any open covering $U=\bigcup_{i\in I}U_i$, the sequence \\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	0 & {\mF(U)} & {\prod_{i\in I}\mF(U_i)} & {\prod_{i,j\in I}\mF(U_i\cap U_j)}
	\arrow[from=1-1, to=1-2]
	\arrow["f", from=1-2, to=1-3]
	\arrow["g", from=1-3, to=1-4]
\end{tikzcd}}\\
where $f$ is given by the product of the unique maps $\mF(U)\to\mF(U_i)$ and $g$ is given by sending $(s_i)_{i\in I}$ to $(s_i|_{U_i\cap U_j}-s_j|_{U_i\cap U_j})_{i,j\in I}$ is exact. 
\item For any open covering $U=\bigcup_{i\in I}U_i$, the following diagram \\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	{\mF(U)} & {\prod_{i\in I}\mF(U_i)} & {\prod_{i,j\in I}\mF(U_i\cap U_j)}
	\arrow["f", from=1-1, to=1-2]
	\arrow["g", shift left, from=1-2, to=1-3]
	\arrow["h"', shift right, from=1-2, to=1-3]
\end{tikzcd}}\\
is an equalizer diagram, where $f$ is given by the product of the unique maps $\mF(U)\to\mF(U_i)$, $g$ is given by sending $(s_i)_{i\in I}$ to $(s_i|_{U_i\cap U_j})_{i,j\in I}$, $h$ is given by sending $(s_i)_{i\in I}$ to $(s_j|_{U_i\cap U_j})_{i,j\in I}$. 
\end{itemize} 
\begin{proof}~\\
We first note that the second and third condition are equivalent. Therefore we just have to show that the first condition is equivalent to the third one. Let $\mF$ be an abelian sheaf. We want to show that $\im(f)=\{x\in\prod_{i\in I}F(U_i)\;|\;g(x)=h(x)\}$. Therefore we begin with $x\in\im(f)$. Then $f(x)=(x_i)_{i\in I}$. But we have that $g((x_i)_{i\in I})=(x_i|_{U_i\cap U_j})_{i,j\in I}$ and $h((x_i)_{i\in I})=(x_j|_{U_i\cap U_j})_{i,j\in I}$. We want to show that for all $i,j\in I$, we have that $x_i|_{U_i\cap U_j}=x_j|_{U_i\cap U_j}$. But $x_i$ is just the restriction of $x$ to $U_i$ and $x_j$ is the restriction of $x$ to $U_j$. But by definition of a sheaf, restricting to $V_i$ and further to $V_{ij}$ is the same as directly restricting to $V_{ij}$. Similarly, restricting to $V_j$ and further to $V_{ij}$ is the same as directly restricting to $V_{ij}$. We conclude that applying $g$ and $h$ to $(x_i)_{i\in I}$ give the same element. Thus $x\in\{x\in\prod_{i\in I}F(U_i)\;|\;g(x)=h(x)\}$. \\

Now let $(y_i)_{i\in I}\in\{x\in\prod_{i\in I}F(U_i)\;|\;g(x)=h(x)\}$. This means that $y_i|_{U_i\cap U_j}=y_j|_{U_i\cap U_j}$ for all $i,j\in I$. By the gluing axiom, there exists $y\in C(U)$ such that $y|_{U_i}=y_i$. Thus $y$ is in the image of $f$ and we conclude that $y\in\im(f)$. \\

Finally, we also need to show that $f$ is injective. Suppose that $x,y\in\mF(U)$ is such that $x_i=y_i$ for all $i\in I$. Then by the identity axiom we conclude that $x=y$. Thus $f$ is injective. Thus the equalizer condition is satisfied. \\

Now suppose that the equalizer condition is satisfied. Then $f$ is injective. This means that for $x,y\in\mF(U)$ such that $x_i=y_i$ for all $i\in I$, then $x=y$. But this is precisely the identity axiom. Suppose that $\im(f)=\text{Eq}(g,h)$. Let $s_i\in\mF(U_i)$ such that $$s_i|_{U_i\cap U_j}=g(s_i)=h(s_i)=s_j|_{U_i\cap U_j}$$ for all $i,j\in I$. Then $(s_i)_{i\in I}\in\text{Eq}(g,h)=\im(f)$. Which means that there exists some $s\in\mF(U)$ such that $s|_{U_i}=s_i$. But this is the content of the gluing axiom. Since the identity and the gluing axiom are satisfied, we conclude that $\mF$ is a sheaf. 
\end{proof}
\end{thm}

Motivated by the above characterization, we define sheaves with values on a category with finite products as follows. 

\begin{defn}{Sheaves with Values in a Category}{} Let $X$ be a space. Let $\mC$ be a category that admits all products. Let $\mF$ be a presheaf on $X$ with values in $\mC$. We say that $\mF$ is a sheaf with values in $\mC$ if for every open covering $U=\bigcup_{i\in I}U_i$, the following diagram \\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	{\mF(U)} & {\prod_{i\in I}\mF(U_i)} & {\prod_{(i,j)\in I\times I}\mF(U_i\cap U_j)}
	\arrow["f", from=1-1, to=1-2]
	\arrow["g", shift left, from=1-2, to=1-3]
	\arrow["h"', shift right, from=1-2, to=1-3]
\end{tikzcd}}\\
is an equalizer diagram, where $f,g,h$ are defined as follows. 
\begin{itemize}
\item The morphism $f$ is the product $$\prod_{i\in I}\text{res}_{U_i}^U:\mF(U)\to\mF(U_i)$$
\item The morphism $g$ is the product $$\prod_{(i,j)\in I\times I}\text{res}_{U_i\cap U_j}^{U_i}:\mF(U_i)\to\mF(U_i\cap U_j)$$
\item The morphism $h$ is the product $$\prod_{(i,j)\in I\times I}\text{res}_{U_i\cap U_j}^{U_j}:\mF(U_j)\to\mF(U_i\cap U_j)$$
\end{itemize}
\end{defn}

Note: the double product is ordered so that the equalizing condition is the same as the gluing condition. 

\begin{prp}{}{}\\
Let $X$ be a space. Let $\mC$ be a category that admits all products. Let $F:\mC\to\bold{Set}$ be a functor such that the following are true. 
\begin{itemize}
\item $F$ is faithful. 
\item $F$ preserves limits. 
\item A morphism $f\in\mC$ is an isomorphism if and only if $F(f)\in\bold{Set}$ is an isomorphism. 
\end{itemize}
Let $\mF$ be a presheaf on $X$ with values in $\mC$. Then $\mF$ is a sheaf with values in $\mC$ if and only if $F\circ\mF$ is a sheaf of sets. 
\begin{proof}\\
Suppose that $\mF$ is a sheaf. Then the following diagram is an equalizer for any open set $U$ and any open cover $\{U_i\;|\;i\in I\}$ of $U$: \\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	{\mF(U)} & {\prod_{i\in I}\mF(U_i)} & {\prod_{(i,j)\in I\times I}\mF(U_i\cap U_j)}
	\arrow["f", from=1-1, to=1-2]
	\arrow["g", shift left, from=1-2, to=1-3]
	\arrow["h"', shift right, from=1-2, to=1-3]
\end{tikzcd}}\\
Since $F$ commutes with limits, the following diagram is an equalizer: \\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	{F\circ\mF(U)} & {\prod_{i\in I}F\circ\mF(U_i)} & {\prod_{(i,j)\in I\times I}F\circ\mF(U_i\cap U_j)}
	\arrow["f", from=1-1, to=1-2]
	\arrow["g", shift left, from=1-2, to=1-3]
	\arrow["h"', shift right, from=1-2, to=1-3]
\end{tikzcd}}\\
Hence $F\circ\mF$ is a sheaf. \\

Conversely, suppose that $F\circ\mF$ is a sheaf. Let $E$ be the equalizer of the following diagram: \\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	{\prod_{i\in I}F\circ\mF(U_i)} & {\prod_{(i,j)\in I\times I}F\circ\mF(U_i\cap U_j)}
	\arrow["g", shift left, from=1-1, to=1-2]
	\arrow["h"', shift right, from=1-1, to=1-2]
\end{tikzcd}}\\
By the universal property of products and the definition of a presheaf, we obtain a morphism $\mF(U)\to E$. Since $F$ preserves limits, we have that $F\circ\mF(U)$ and $F(E)$ are both equalizers of the same diagram, and so the morphism $F\circ\mF(U)\to F(E)$ is an isomorphism. By assumption on $F$, we conclude that $\mF(U)\cong E$. Hence $\mF$ is a sheaf. 
\end{proof}
\end{prp}

The following definition is inspired by the Stacks Project. 

\begin{defn}{Algebraic Categories}{}\\
Let $\mC$ be a category. We say that $\mC$ is an algebraic category if there exists a functor $F:\mC\to\bold{Set}$ such that the following are true. 
\begin{itemize}
\item $\mC$ admits all products and filtered colimits. 
\item $F$ is faithful (so that $\mC$ is a concrete category). 
\item $F$ preserves limits and filtered colimits. 
\item A morphism $f\in\mC$ is an isomorphism if and only if $F(f)\in\bold{Set}$ is an isomorphism. 
\end{itemize}
\end{defn}

This allows us to simplify statements about sheaves with values in categories whose objects are algebraic in nature. Some examples of algebraic categories include: $$\bold{Set},\bold{Grp},\bold{Ab},\bold{Ring},\bold{CRing},\bold{Mod}_R,\bold{Alg}_R$$ They are all equipped with a forgetful functor to sets that is faithful, is the right adjoint of the free functor (take $\text{id}:\bold{Set}\to\bold{Set}$ for $\bold{Set}$) and reflects isomorphisms. \\

We abuse notation and think of as if $\mF(U)$ are sets for each open set $U\subseteq X$ (in reality only $(F\circ\mF)(U)$ are sets). 

\begin{lmm}{}{}\\
Let $X$ be a topological space. Let $\mC$ be an algebraic category. Let $\mF$ be a $\mC$-presheaf on $X$. Let $p\in X$. Then we have $$\mF_p=\frac{\{(U,f)\;|\;U\subseteq X\text{ is an open neighborhood of }p, f\in\mF(U)\}}{\sim}$$ where $(U,f)\sim(V,g)$ if there exists $W\subseteq U\cap V$ an open neighborhood of $p$ such that $f|_W=g|_W$. 
\begin{proof}
If $\mF$ is a $\bold{Set}$-presheaf, the above would be an explicit construction of the filtered colimit. Since $\mC$ is algebraic, $F:\mC\to\bold{Set}$ preserves colimits and by abuse of notation we obtain the following set-theoretic description. 
\end{proof}
\end{lmm}

Think of the definition of stalks as follows: Treat $f$ and $g$ to be sections in $\mF(U_1)$ and $\mF(U_2)$ where $V\subseteq U_1\cap U_2$ is open and contains $x$. Then we think of $f$ and $g$ to be the same function in the stalk as long as they agree on some open set that contains $x$. \\

Indeed, since we do not care about the entirety of the domain of $f$ and $g$, and only care about what happens locally near $x$, it makes sense for us to treat them as a function when they appear to be the same locally. \\

Let $[(s,U)]\in\mF_p$ be an element of the stalk. Consider the following commutative diagram: \\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	{\mF(U)} & {\mG(U)} \\
	{\mF_p} & {\mG_p}
	\arrow[from=1-1, to=1-2]
	\arrow[from=1-1, to=2-1]
	\arrow[from=1-2, to=2-2]
	\arrow[from=2-1, to=2-2]
\end{tikzcd}} \\
The element $s\in\mF(U)$ maps vertically on the left to $[(s,U)]\in\mF_p$. Since the diagram is commutative, we see that the image of the element in $\mF_p$ from the induced map of stalks is $[(\phi(U)(s),U)]$. This is the formula for the induced map. 

\begin{prp}{}{}\\
Let $X$ be a topological space. Let $\mC$ be an algebraic category. Let $\mF,\mG$ be $\mC$-presheaves on $X$. Let $\varphi:\mF\to\mG$ be a morphism. Then $\varphi$ is an isomorphism if and only if for all $p\in X$, the induced map $$\varphi_p:\mF_p\to\mG_p$$ is an isomorphism. 
\begin{proof}\\
Suppose that $\phi$ is an isomorphism of sheaves. Then $\phi$ and $\phi^{-1}$ induces unique morphisms $\phi_p:\mF_p\to\mG_{X,p}$ and $\phi_p^{-1}:\mG_{X,p}\to\mF_p$ making the following diagram commute: \\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	{\mF(U)} & {\mG(U)} & {\mF(U)} & {\mG(U)} \\
	{\mF_p} & {\mG_{X,p}} & {\mF_p} & {\mG(V)}
	\arrow["{\phi(U)}", from=1-1, to=1-2]
	\arrow[from=1-1, to=2-1]
	\arrow["{\phi^{-1}(U)}", from=1-2, to=1-3]
	\arrow[from=1-2, to=2-2]
	\arrow["{\phi^{-1}(U)}", from=1-3, to=1-4]
	\arrow[from=1-3, to=2-3]
	\arrow[from=1-4, to=2-4]
	\arrow["{\exists!}"', dashed, from=2-1, to=2-2]
	\arrow["{\exists!}"', dashed, from=2-2, to=2-3]
	\arrow["{\exists!}"', dashed, from=2-3, to=2-4]
\end{tikzcd}} \\
On the other hand, the identity morphisms $\text{id}_\mF$ and $\text{id}_\mG$ also induces the identity map on the level of stalks. Thus we must have $\phi_p\circ\phi^{-1}_p=\text{id}_{\mG_{X,p}}$ and $\phi^{-1}_p\circ\phi_p=\text{id}_{\mF_p}$ . Hence the induced morphism of stalks is an isomorphism. \\

Now suppose that $\phi_p$ is an isomorphism for each $p\in X$. We show that $\phi(U)$ is bijective for all $U\subseteq X$ open. Let $s\in\mF(U)$ such that $\phi(U)(s)=0$. Then for each $p\in U$, $\phi(U)(s)$ considered as an element in $\mG_{X,p}$ is equal to $0$. Since $\phi_p$ is injective, $s=0$ in $\mF_p$. This means that there exists an open neighbourhood $V_p\subseteq U$ of $p$ such that $s\in\phi(V_p)$ is $0$ for each $p\in U$. Since $U$ is covered by the open neighbourhoods $V_p$ for $p\in U$, by the identity axiom we conclude that $s=0$ in $\phi(U)$. Thus $\phi$ is injective. \\

Now let $t\in\mG(U)$. Let $t_p$ be the corresponding element of $t$ in $\mG_{X,p}$ for each $p\in U$. Since $\phi_p$ is surjective, there exists $s_p\in\mF_p$ such that $\phi_p(s_p)=t_p$ for each $p\in U$. Suppose that on a neighbourhood $V_p$ of $p$, $s_p$ is represented as $r_p$. Then $\phi(r_p)\in\mG(V_p)$ and $t|_{V_p}$ are two elements whose germs in $\mG_{X,p}$ are equal. Hence there exists $W_p\subseteq V_p$ containing $p$ such that $\phi(r_p)=t|_{W_p}$ in $\mG(W_p)$. Now $U$ is covered by open sets of the form $W_p$ for each $p\in U$. Let $p\in W_p$ and $q\in W_q$. Then both $r_p|_{W_p\cap W_q}$ and $r_q|_{W_p\cap W_q}$ are two sections in $\mF(W_p\cap W_q)$ that are sent to $t|_{W_p\cap W_q}\in\mG(W_p\cap W_q)$ by $\phi(W_p\cap W_q)$. By injectivity, they are equal. We conclude that $$r_p|_{W_p\cap W_q}=r_q|_{W_p\cap W_q}$$ By the gluing axiom, we conclude that there exists a section $s\in\mF(U)$ such that $s|_{W_p}=r_p$. Now $\phi(s)$ and $t$ are two sections in $\mG(U)$ such that for any $p\in U$, $\phi(s)|_{W_p}=t|_{W_p}$. By the identity axiom, we conclude that $\phi(s)=t$. 
\end{proof}
\end{prp}

Note that the above proposition is very untrue for presheaves because as one can see, proving both injective and surjective requires to use of the sheaf axioms instead of just the presheaf datum. 

\begin{prp}{}{}\\
Let $X$ be a topological space. Let $\mC$ be an algebraic category. Let $\mF,\mG$ be $\mC$-presheaves on $X$. Let $\varphi,\psi:\mF\to\mG$ be morphisms. Then $\varphi=\phi$ if and only if for all $p\in X$, we have $\varphi_p=\phi_p$. 
\begin{proof}\\
Clearly if $\varphi(U)=\psi(U)$ for all open sets $U\subseteq X$, we must have that the induced map of colimits are equal. So suppose instead that $\varphi_p=\psi_p$ for all $p\in X$. Consider the following commutative diagram: \\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	{\mF(U)} & {\mG(U)} \\
	{\prod_{p\in U}\mF_p} & {\prod_{p\in U}\mG_p}
	\arrow[from=1-1, to=1-2]
	\arrow[from=1-1, to=2-1]
	\arrow[from=1-2, to=2-2]
	\arrow[from=2-1, to=2-2]
\end{tikzcd}} \\
where the top and bottom maps are either $\varphi$ or $\psi$ and their induced product map of stalks. \\

I claim that the vertical maps are injective. Suppose that $s,t\in\mF(U)$ are such that $([(s,U)])_{p\in U}=([(t,U)])_{p\in U}$. This means that for each $p\in U$, we have that $[(s,U)]=[(t,U)]\in\mF_p$. This means that there exists an open subset $p\in V_p\subseteq U$ such that $s|_{V_p}=t|_{V_p}$. Now the collection $\{V_p\;|\;p\in U\}$ forms an open cover of $U$. Since $\mF$ is a sheaf, the identity axiom implies that $s=t$ on $U$. Hence the map is injective. \\

Let $s\in\mF(U)$. Then the left vertical map sends $s$ to $([(s,U)])_{p\in U}$ and $\prod_{p\in U}\varphi_p=\prod_{p\in U}\psi_p$ sends the element to $([(\varphi(U)(s),U)])_{p\in U}=([(\psi(U)(s),U)])_{p\in U}$. Since the right vertical map is injective, we must have $\varphi(U)(s)=\psi(U)(s)$. Hence we have $\varphi=\psi$. 
\end{proof}
\end{prp}

\begin{defn}{Compatible Germs}{}\\
Let $X$ be a space. Let $\mC$ be an algebraic category. Let $\mF$ be an $\mC$-presheaf on $X$. Let $([(s_p,U_p)])_{p\in U}\in\prod_{p\in U}\mF_p$. We say that the element is a compatible germ if for all $p\in U$, there exists an open subset $p\in V_p\subseteq U$ such that $$[(s_p|_{V_p},V_p)]=[(s_q,U_q)]\in\mF_q$$ for all $q\in V_p$. 
\end{defn}

\begin{lmm}{}{}\\
Let $X$ be a space. Let $\mC$ be an algebraic category. Let $\mF$ be an $\mC$-presheaf on $X$. Let $([(s_p,U_p)])_{p\in U}\in\prod_{p\in U}\mF_p$. Then the element is a compatible germ if and only if there exists an open cover $U=\bigcup_{i\in I}U_i$ of $U$ together with elements $f_i\in\mF(U_i)$ such that for all $p\in U_i$ for all $i$, $[(f_i,U_i)]=[(s_p,U_p)]\in\mF_p$. 
\begin{proof}\\
Suppose that $([(s_p,U_p)])_{p\in U}\in\prod_{p\in U}\mF_p$ is a compatible germ. Then $\{U_p\;|\;p\in U\}$ is a valid open cover of $U$ together with the elements $s_p\in\mF(U_p)$ so that the latter condition is satisfied. \\

Conversely, suppose that the latter condition is satisfied. Let $p\in U$. Since $\{U_i\;|\;i\in I\}$ is an open cover of $U$, there exists $i\in I$ such that $p\in U_i$. Then for all $q\in U_i$, we have $[(f_i,U_i)]=[(s_q,U_q)]\in\mF_q$ by assumption. Hence we are done. 
\end{proof}
\end{lmm}

Thus not only are isomorphism of sheaves determined on the level of stalks, the morphism of sheaves themselves are determined on the level of stalks. 

\begin{defn}{Sheafification}{}\\
Let $X$ be a space. Let $\mC$ be an algebraic category. Let $\mF$ be an $\mC$-presheaf on $X$. Define the sheafification $$\mF^\text{sh}:\bold{Open}(X)\to\mA$$ of $\mF$ as follows. 
\begin{itemize}
\item For each open set $U\subseteq X$, $$\mF^\text{sh}(U)=\left\{([(s_p,U_p)])_{p\in U}\in\prod_{p\in U}\mF_p\;\bigg{|}\;([(s_p,U_p)])_{p\in U}\text{ is a compatible germ }\right\}$$
\item For each $V\subseteq U$ an inclusion, there is a unique morphism $\mF^\text{sh}(U)\to\mF^\text{sh}(V)$ that sends $([(s_p,U_p)])_{p\in U}\in\in\mF^\text(U)$ to $$([(s_p,U_p)])_{p\in V}$$
\end{itemize}
\end{defn}

\begin{lmm}{}{}\\
Let $X$ be a space. Let $\mC$ be an algebraic category. Let $\mF$ be an $\mC$-presheaf on $X$. Then $\mF^\text{sh}$ is a $\mC$-sheaf. 
\begin{proof}\\
We first show that $\mF^\text{sh}$ is a sheaf. Let $\{W_i\;|\;i\in I\}$ be an open cover for an open set $U\subseteq X$. \\

For the identity axiom, suppose that $([(s_p,U_p)])_{p\in U}$ and $([t_p,V_p])_{p\in U}$ in $\mF(U)$ are such that their restriction to $\mF^\text{sh}(W_i)$ are equal. This means that $([(s_p,U_p)])_{p\in W_i}=([(s_p,U_p)])_{p\in W_i}$ or in other words, that $[(s_p,U_p)]=[(t_p,V_p)]\in\mF_p$. Then since $\{W_i\;|\;i\in I\}$ covers $U$, we must have $([(s_p,U_p)])_{p\in U}=([t_p,V_p])_{p\in U}$ in $\mF(U)$. \\

For gluing, suppose that we are given $([(s_p,U_p)])_{p\in W_i}\in\mF^\text{sh}(W_i)$ for each $i\in I$ such that restricting to common intersections give equal elements. This means that the element $([(s_p,U_p)])_{p\in U}$ is a well defined element of $\mF^\text{sh}$. 
\end{proof}
\end{lmm}

\begin{eg}{}{}\\
Let $X$ be a space. Let $S$ be a set. Then we have $$\underline{S}=\left(\underline{S}_\text{pre}\right)^\text{sh}$$
\end{eg}

\begin{defn}{Universal Morphism of Sheafification}{}\\
Let $X$ be a space. Let $\mC$ be an algebraic category. Let $\mF$ be an $\mC$-presheaf on $X$. Define the universal morphism of sheafification $\theta_\mF$ as follows. For each open set $U\subseteq X$, define $$\theta_\mF(U):\mF(U)\to\prod_{p\in U}\mF_p$$ by $s\mapsto([(s,U)])_{p\in U}$. 
\end{defn}

\begin{lmm}{}{}\\
Let $X$ be a space. Let $\mC$ be an algebraic category. Let $\mF$ be an $\mC$-presheaf on $X$. Then the following are true. 
\begin{itemize}
\item $\im(\theta_\mF)=\mF^\text{sh}(U)$. 
\item If $\mF$ is a sheaf, then for any open subset $U\subseteq X$, $\theta_\mF(U):\mF\to\mF^\text{sh}$ is injective. 
\end{itemize}
\begin{proof}
\begin{itemize}
\item Clearly any element in the image of the map is a compatible germ. Now conversely, suppose that $([(s_p,U_p)])_{p\in U}$ is a compatible germ. Now $\{U_p\;|\;p\in U\}$ is an open cover of $U$. Let $p,q\in U$ be arbitrary. Let $r\in U_p\cap U_q$. Then there exists an open subset $V_r\subseteq U_p$ such that $[(s_p|_{V_r},V_r)]=[(s_q,U_q)]\in\mF_q$. This means that there exists an open subset $W_r\subseteq V_r\cap U_q$ such that $s_p|_{W_r}=s_q|_{W_r}$. Now since $\{W_r\;|\;r\in U_p\cap U_q\}$ covers $U_p\cap U_q$, we must have that $s_p|_{U_i\cap U_j}=s_q|_{U_i\cap U_j}$ by the identity axiom. Then by the gluability axiom, there exists $s\in\mF(U)$ such that $s|_{U_p}=s_p\in\mF(U_p)$. Now clearly $([(s,U)])_{p\in U}=([(s_p,U_p)])_{p\in U}$ since $s|_{U_p}=s_p\in\mF(U_p)$. Hence the map is surjective. 
\item Suppose that $s,t\in\mF(U)$ are such that $([(s,U)])_{p\in U}=([(t,U)])_{p\in U}$. This means that for each $p\in U$, we have that $[(s,U)]=[(t,U)]\in\mF_p$. This means that there exists an open subset $p\in V_p\subseteq U$ such that $s|_{V_p}=t|_{V_p}$. Now the collection $\{V_p\;|\;p\in U\}$ forms an open cover of $U$. Since $\mF$ is a sheaf, the identity axiom implies that $s=t$ on $U$. Hence the map is injective. 
\end{itemize}
\end{proof}
\end{lmm}

\begin{lmm}{}{}\\
Let $X$ be a space. Let $\mC$ be an algebraic category. Let $\mF$ be an $\mC$-presheaf on $X$. Then $\theta$ induces an isomorphism $$\theta_p:\mF_p\overset{\cong}{\longrightarrow}\mF_p^\text{sh}$$ of stalks for each $p\in X$. 
\begin{proof}\\
By the universal property of stalks as colimits, the morphism $\theta_p$ is defined by $[(s,U)]\mapsto[(([(s,U)])_{p\in U},U)]$. \\

Suppose that $[(([(s,U)])_{q\in U},U)]=[(([(t,U)])_{q\in U},U)]$. By the definition of the equivalence relation, we have that $([(s,U)])_{q\in U}$ and $([(t,U)])_{q\in U}$ agree on $U$. Hence for each $q\in U$, we have $[(s,U)]=[(t,U)]$. Again by the definition of the equivalence relation, we have $s=t$ on $U$. Hence $\theta_p$ is injective. \\

To show that $\theta_p$ is surjective, let $[(([(s_q,U_q)])_{q\in U},U)]\in\mF_p^\text{sh}$. Since $s$ is a compatible germ by definition, there exists an open cover $\{U_i\;|\;i\in I\}$ of $U$ together with elements $f_i\in\mF(U_i)$ such that $[(f_i,U_i)]=[(s_q,U_q)]\in\mF_p$ for all $q\in U_i$ for all $i$. In particular, for our already fixed choice of $p\in X$, there exists $i\in I$ and $f_i\in\mF(U_i)$ such that $[(f_i,U_i)]=[(s_p,U_p)]\in\mF_p$. Now $[(f_i,U_i)]\in\mF_p$ is mapped to $[(([(f_i,U_i)])_{q\in U_i},U_i)]$. I claim that $$[(([(f_i,U_i)])_{q\in U_i},U_i)]=[(([(s_q,U_q)])_{q\in U},U)]\in\mF_p^\text{sh}$$ Indeed, in the intersection $U_i=U_i\cap U$, we have $[(f_i,U_i)]=[(s_q,U_q)]$ for all $q\in U_i$. by definition. Hence we have $([(f_i,U_i)])_{q\in U_i}=([(s_q,U_q)])_{q\in U_i}$ and hence we have surjectivity. 
\end{proof}
\end{lmm}

\begin{prp}{The Universal Property of Sheafification}{}\\
Let $X$ be a space. Let $\mC$ be an algebraic category. Let $\mF$ be an $\mC$-presheaf on $X$. Then the sheafification $\mF^\text{sh}$ of $\mF$ together with the universal morphism $\theta$ satisfies the following universal property. For any sheaf $\mG$ on $X$ and morphism of presheaves $\varphi:\mF\to\mG$, there exists a unique morphism $\varphi^\text{sh}:\mF^\text{sh}\to\mG$ such that the following diagram commutes: \\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	\mF & {\mF^\text{sh}} \\
	& \mG
	\arrow["\theta", from=1-1, to=1-2]
	\arrow["\varphi"', from=1-1, to=2-2]
	\arrow["{\exists!\varphi^\text{sh}}", dashed, from=1-2, to=2-2]
\end{tikzcd}} \\
Moreover, $\mF^+$ is the unique (up to unique isomorphism) sheaf of $X$ satisfying the above property. 
\begin{proof}\\
Let $\mG$ be a sheaf. Let $\phi:\mF\to\mG$ be a morphism of presheaves. To define a morphism of sheaves $\mF^\text{sh}\to\mG$, we first define it on each open set $U$ and prove that the relevant square diagrams commute. So define $\phi^\text{sh}(U):\mF^\text{sh}(U)\to\mG(U)$ for each open set $U\subseteq X$ as follows. For $([(s_p,U_p)])_{p\in U})$ an element in $\prod_{p\in U}\mF_p$, the product map $\prod_{p\in U}\phi_p$ sends the our element to $([\phi(U_p)(s_p),U_p)])_{p\in U}$. Let $([(s_p,U_p)])_{p\in U}$ be a compatible germ. I claim that $([\phi(U_p)(s_p),U_p)])_{p\in U}$ is a compatible germ. By lemma 1.2.6, there exists an open cover $\{U_i\;|\;i\in I\}$ of $U$ together with elements in $f_i\in\mF(U_i)$ such that for all $i\in I$ and all $p\in U_i$, $[(f_i,U_i)]=[(s_p,U_p)]$. Now the elements $\phi(U_i)(f_i)$ for all $i$ proves the claim since $\phi_p$ maps $[(f_i,U_i)]$ to $[(\phi(U_i)(f_i),U_i)]$, $[(s_p,U_p)]\mapsto[(\phi(U_p)(s_p),U_p)]$ and $[(f_i,U_i)]=[(s_p,U_p)]\in\mF_p$ implies that $[(\phi(U_i)(f_i,U_i)]=[(\phi(U_p)(s_p),U_p)]$. By lemma 1.2.9, we obtain a unique element in $\mG(U)$. Hence we obtain a map $\phi^\text{sh}(U):\mF^\text{sh}(U)\to\mG(U)$. \\

Now we show that the composite $\mF\to\mF^\text{sh}\to\mG$ is equal to $\phi$. Notice that given $s\in\mF(U)$, it is sent to $([(s,U)]_{p\in U}$ in $\mF^\text{sh}(U)$ and then to $([(\phi(U)(s),U)])_{p\in U}$ in $\mG(U)$. Evidently, $\phi(U)(s)\in\mG(U)$ is the unique preimage of $([(\phi(U)(s),U)])_{p\in U}$ of the map in lemma 1.2.7. Hence $s\mapsto\phi(U)(s)$ matches up with the map of presheaves $\phi:\mF\to\mG$. \\

It remains to show uniqueness of the morphism $\phi^\text{sh}$. Suppose that $\psi:\mF^\text{sh}\to\mG$ is another morphism of sheaves such that $\phi=\psi\circ\theta$. By the above lemma, $\theta$ is an isomorphism at stalks. Hence for all $p\in X$, we have $$\phi_p\circ\theta_p^{-1}=\psi_p$$ On the other hand, the equation $\phi=\phi^\text{sh}\circ\theta$ gives $$\phi_p\circ\theta_p^{-1}=\phi_p^\text{sh}$$ Equating the two gives $\psi_p=\phi_p^\text{sh}$ for all $p\in X$. Since $\mF^\text{sh}$ and $\mG$ are sheaves, By 1.2.6 we conclude that $\psi=\phi^\text{sh}$ and so we are done. 
\end{proof}
\end{prp}

\begin{lmm}{}{}\\
Let $X$ be a space. Let $\mC$ be an algebraic category. Let $\mF$ be an $\mC$-presheaf on $X$. Then the universal morphism $\theta:\mF\to\mF^\text{sh}$ is an isomorphism. 
\begin{proof}\\
By 1.2.11, we have that $\theta_p$ is an isomorphism for all $p\in X$. Hence by 1.2.4, we have that $\theta$ is an isomorphism. 
\end{proof}
\end{lmm}

Thus the theory of compatible germs gives us an explicit way of understanding the elements of a sheaf by a collection of elements in the stalks. 

\begin{defn}{The Sheafification Functor}{}\\
Let $X$ be a space. Let $\mC$ be an algebraic category. Define the sheafification functor $$(-)^\text{sh}:\bold{P}\mC(X)\to\mC(X)$$ as follows. 
\begin{itemize}
\item For each presheaf $\mF$ on $X$, $\mF^\text{sh}$ is the sheafification of $\mF$. 
\item For each morphism of presheaves $\mF\to\mG$, the morphism of sheaves is given by the universal property of $\mF^\text{sh}$ applied to the composite $\mF\to\mG\to\mG^\text{sh}$. 
\end{itemize}
\end{defn}

\begin{prp}{}{}\\
Let $X$ be a space. Let $\mC$ be an algebraic category. Let $\text{Forget}:\mC(X)\to\bold{P}\mC(X)$ be the forgetful functor. Then there is an adjunction $$(-)^\text{sh}:\bold{P}\mC(X)\overset{\dashv}{\rightleftarrows}\mC(X):\text{Forget}$$
\begin{proof}\\
The universal property of sheafification gives a map from the right hom set to the left hom set. Conversely, precomposition with the universal morphism $\mF\to\mF^\text{sh}$ gives the map to the other direction. Evidently they are inverses of each other. It remains to show that this bijection is natural in both variables. In other words, we ought to show that the following two squares commute: \\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	{\Hom_{\bold{Set(X)}}(\mF^\text{sh},\mG)} & {\Hom_{\bold{PSet}(X)}(\mF,\text{Forget}(\mG))} \\
	{\Hom_{\bold{Set(X)}}(\mH^\text{sh},\mG)} & {\Hom_{\bold{PSet}(X)}(\mH,\text{Forget}(\mG))} \\
	{\Hom_{\bold{Set(X)}}(\mF^\text{sh},\mG)} & {\Hom_{\bold{PSet}(X)}(\mF,\text{Forget}(\mG))} \\
	{\Hom_{\bold{Set(X)}}(\mF^\text{sh},\mI)} & {\Hom_{\bold{PSet}(X)}(\mF,\text{Forget}(\mI))}
	\arrow["\cong", from=1-1, to=1-2]
	\arrow["{-\circ\varphi^\text{sh}}", from=2-1, to=1-1]
	\arrow["\cong"', from=2-1, to=2-2]
	\arrow["{-\circ\varphi}"', from=2-2, to=1-2]
	\arrow["\cong", from=3-1, to=3-2]
	\arrow["{\psi\circ-}"', from=3-1, to=4-1]
	\arrow["{\text{Forget}(\psi)\circ-}", from=3-2, to=4-2]
	\arrow["\cong"', from=4-1, to=4-2]
\end{tikzcd}} \\
for two morphisms $\varphi:\mF\to\mH$ and $\psi:\mG\to\mI$. Since $\bold{Sh}(X)$ is a full subcategory of $\bold{PSh}(X)$, clearly the square on the bottom commutes. For the top square, let $\sigma:\mH^\text{sh}\to\mG$ be a morphism. Following the left and top arrows, we obtain the morphism $\sigma\circ\varphi^\text{sh}\circ\theta_\mF$. Following the bottom and right arrows, we obtain the morphism $\sigma\circ\theta_\mH\circ\varphi$. I claim that $\varphi^\text{sh}\circ\theta_\mF=\theta_\mH\circ\varphi$. Indeed, the definition of $\varphi^\text{sh}$ as in def 2.1.10 is given by the universal property of sheafification of the composite $\theta_\mH\circ\varphi$. Hence precomposing $\varphi^\text{sh}$ with $\theta_\mF$ is precisely $\theta_\mH\circ\varphi$. Thus we are done. 
\end{proof}
\end{prp}

\subsection{The Category of (Pre)Sheaves as an Abelian Category}
\begin{lmm}{}{}\\
Let $X$ be a topological space. Let $\mA$ be an abelian category. Let $\ast$ be the zero object of $\mA$. Then the sheaf $\underline{\ast}$ is the zero object of $\mA(X)$. 
\begin{proof}
Let $\mF\in\mA(X)$ be an abelian sheaf. We first show the existence and uniqueness of a morphism $\underline{\ast}\to\mF$. Suppose that $\varphi:\underline{\ast}\to\mF$ is a morphism. Then for each open set $U\subseteq X$, there is a unique morphism $\varphi(U):\underline{\ast}(U)=\ast\to\mF(U)$. Hence $\varphi$ is unique. Existence is also evident. \\

Similarly, we show the existence and uniqueness of a morphism $\mF\to\underline{\ast}$. Suppose that $\varphi:\mF\to\underline{\ast}$ is a morphism. Then for each open set $U\subseteq X$, there is a unique morphism $\varphi(U):\mF(U)\to\ast=\underline{\ast}(U)$. Hence $\varphi$ is unique. Existence is also evident. 
\end{proof}
\end{lmm}

\begin{defn}{The Product Sheaf}{}\\
Let $X$ be a space. Let $\mC$ be a category that admits all products. Let $\mF,\mG$ be $\mC$-sheaves on $X$. Define the product $\mF\times\mG$ sheaf of $\mF$ and $\mG$ as follows. 
\begin{itemize}
\item For each open set $U\subseteq X$, define $(\mF\times\mG)(U)=\mF(U)\times\mG(U)$. 
\item For each inclusion of open sets $V\subseteq U$, define the morphism $(\mF\times\mG)(U)\to(\mF\times\mG)(V)$ by $\text{res}_V^U\times\text{res}_V^U$. 
\end{itemize}
\end{defn}

\begin{lmm}{}{}\\
Let $X$ be a space. Let $\mC$ be a category that admits all products. Let $\mF,\mG$ be $\mC$-sheaves on $X$. Then $\mF\times\mG$ is the categorical product of $\mF$ and $\mG$ in $\mC(X)$. 
\begin{proof}\\
Firstly, the identity and gluing axiom of $\mF\times\mG$ can be checked coordinate wise. Clearly, there are projection morphisms $\mF\times\mG\to\mF,\mG$ open set wise. Let $\mH$ be a sheaf with morphisms $\mH\to\mF,\mG$. Then by the universal product of products of abelian groups, there is a unique morphism $\mH(U)\to(\mF\times\mG)(U)$ for each open set $U\subseteq X$ such that the composition $\mH(U)\to(\mF\times\mG)(U)\to\mF(U),\mG(U)$ is equal to $\mH(U)\to\mF(U),\mG(U)$ respectively. \subsection{Exact Sequences of Sheaves}
From now on, we focus on the abelian category of sheaves and not presheaves. Therefore we will drop the sheafification symbol for cokernels and images since they mean the abelian categorical object that is the cokernel and the image. \\

Let $X$ be a space. Let $\mA$ be an abelian category that is an algebraic category. Since $\mA(X)$ is an abelian category, it makes sense to talk about chain complexes and in particular, exact sequences of sheaves. Recall that a sequence of sheaves $\mF\to\mG\to\mH$ is exact if $\im(\mF\to\mG)=\ker(\mG\to\mH)$. 

\begin{prp}{}{}\\
Let $X$ be a space. Let $\mA$ be an abelian category that is an algebraic category. Suppose the following \\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	\mF & \mG & \mH
	\arrow[from=1-1, to=1-2]
	\arrow[from=1-2, to=1-3]
\end{tikzcd}} \\~\\
is a sequence of $\mA$-sheaves on $X$. Then the sequence is exact if and only if the following sequence of stalks\\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	{\mF_p} & {\mG_p} & {\mH_p}
	\arrow[from=1-1, to=1-2]
	\arrow[from=1-2, to=1-3]
\end{tikzcd}} \\~\\
is exact for all $p\in X$. 
\begin{proof}
We have that $\im(\mF\to\mG)=\ker(\mG\to\mH)$ and taking stalks gives $\im(\mF_p\to\mG_p)=\ker(\mG_p\to\mH_p)$. Since taking stalks commute with images and kernels, we are done. Conversely, suppose that for all $p\in X$, we have $\im(\mF_p\to\mG_p)=\ker(\mG_p\to\mH_p)$. Since taking stalks commute with kernels and images, this show that $\im(\mF\to\mG)_p=\ker(\mG\to\mH)_p$. But two sheaves are isomorphic if and only if the stalks are isomorphic, we are done. 
\end{proof}
\end{prp}

Thus the functor $$(-)_p:\mA(X)\to\mA$$ of taking stalks is exact.

\item For each inclusion $V\subseteq U$, define the induced map $$\mF(f^{-1}(U))\to\mF(f^{-1}(V))$$ to be the map $\mF(\text{incl.}:f^{-1}(V)\hookrightarrow f^{-1}(U))$. 
\end{itemize}

\begin{prp}{}{}\\
Let $X,Y$ be spaces. Let $f:X\to Y$ be a map. Let $\mC$ be an algebraic category. Let $\mF$ be a $\mC$-sheaf. Then $f_\ast\mF$ is a sheaf on $Y$. 
\begin{proof}\\
It is clear that $f_\ast\mF$ is a presheaf on $Y$. We need to check the identity and gluing axiom. By 1.3.4, without loss of generality we can assume that $\mC=\bold{Set}$. \\

Identity: Let $U\subseteq Y$ be an open set. Let $\{U_i\;|\;i\in I\}$ be an open cover of $U$. Denote the inclusion maps by $\iota_i:U_i\to U$. Let $s,t\in f_\ast\mF(U)$. Suppose that $f_\ast\mF(\iota_i)(s)=f_\ast\mF(\iota_i)(t)$ for all $i\in I$. This means that $s,t\in\mF(f^{-1}(U))$ is such that the images of $s,t$ in $\mF(f^{-1}(U_i))$ are equal for all $i\in I$. Since $\{f^{-1}(U_i)\}$ is an open cover of $f^{-1}(U)$ and $\mF$ is a sheaf on $X$, this means that $s=t$. \\

Gluing: Let $U\subseteq Y$ be an open set. Let $\{U_i\;|\;i\in I\}$ be an open cover of $U$. Suppose that $s_i\in f_\ast\mF(U_i)=\mF(f^{-1}(U_i))$ is such that the images of $s_i$ and $s_j$ in $f_\ast\mF(U_i\cap U_j)=\mF(f^{-1}(U_i\cap U_j))$ agree for all $i,j\in I$. Since $\{f^{-1}(U_i)\}$ is an open cover of $f^{-1}(U)$ and $\mF$ is a sheaf on $X$, we conclude that there exists $s\in\mF(f^{-1}(U))=f_\ast\mF(U)$ such that the image of $s$ in $\mF(f^{-1}(U_i))=f_\ast\mF(U_i)$ is $s_i$ for all $i\in I$. Thus the gluing axiom is satisfied. 
\end{proof}
\end{prp}

Since $W$ is the initial object in the category of open sets containing $W$, we have that he colimit is equal to the terminal object. Hence $(\iota^{-1}\mG_\text{pre})(W)=\mG(W)$. Since $\mG$ is already a sheaf, sheafification gives a natural isomorphism $\iota^{-1}\mG)\cong\iota^{-1}\mG_\text{pre}$ and the latter is equal to $\mG|_U$. 
\end{proof}
\end{lmm}

\begin{prp}{}{}\\

\subsection{Ringed Spaces}
\begin{defn}{Ringed Space}{}\\
A ringed space is a topological space $X$ with a sheaf of rings $\mO_X$ on $X$. 
\end{defn}

\begin{defn}{Morphisms of Ringed Spaces}{} Let $(X,\mathcal{O}_X)$ and $(Y,\mathcal{O}_Y)$ be ringed spaces. A morphism of ringed spaces from $(X,\mathcal{O}_X)$ to $(Y,\mathcal{O}_Y)$ is a pair $(f,f^\#)$ of continuous map $f:X\to Y$ and a map $f^\#:\mathcal{O}_Y\to f_\ast\mathcal{O}_X$ of sheaves of rings on $Y$. 
\end{defn}

\begin{defn}{Locally Ringed Spaces}{}\\
Let $(X,\mO_X)$ be a ringed space. We say that $X$ is a locally ringed space if for all $p\in X$, $\mO_{X,p}$ is a local ring. 
\end{defn}

Suppose that $(X,\mO_X)$ and $(Y,\mO_Y)$ are locally ringed spaces. Given a point $p\in X$, we obtain induced maps $$f^\#:\mO_Y(V)\to f_\ast\mO_X(f^{-1}(V))$$ for any neighbourhood $V$ of $f(p)$. Taking the colimit, we obtain a map $$\mO_{Y,f(p)}=\text{colim}_V\mO_Y(V)\to\text{colim}_V\mO_X(f^{-1}(V))$$ As $V$ ranges through the neighbourhood of $f(p)$, $f^{-1}(V)$ ranges over a subset of the neighbourhoods of $p$ so that we obtain a map $$\mO_{Y,f(p)}=\text{colim}_V\mO_Y(V)\to\text{colim}_V\mO_X(f^{-1}(V))\to\mO_{X,p}$$

\begin{defn}{Morphisms of Locally Ringed Spaces}{} Let $(X,\mathcal{O}_X)$ and $(Y,\mathcal{O}_Y)$ be locally ringed spaces. Then a morphism of locally ringed spaces is a morphism of ringed spaces such that for each $p\in X$, the induced map of local rings $$f_p^\#:\mO_{Y,f(p)}\to\mO_{X,p}$$ is a local homomorphism of local rings. 
\end{defn}

\begin{defn}{The Category of Locally Ringed Spaces}{}\\
Define the category $\bold{LRS}$ of locally ringed spaces as follows. 
\begin{itemize}
\item The objects are locally ringed spaces. 
\item The morphisms are morphisms of locally ringed spaces. \
\item Composition is given by the composition of continuous functions and composition of natural transformations. 
\end{itemize}
\end{defn}

\begin{defn}{Quasi-Coherent Sheaves}{} Let $(X,\mO_X)$ be a ringed space. Let $\mF$ be a sheaf of $\mO_X$-module. We say that $\mF$ is a quasi-coherent sheaf if for all $p\in X$, there exists an open neighbourhood $U\subseteq X$ such that there is an exact sequence: \\

\begin{itemize}
\item The objects are the quasi-coherent sheaves $\mF$ over $(X,\mO_X)$
\item The morphisms are morphisms between sheaves of modules. This means that if $\mF$ and $\mG$ are quasi-coherent $\mO_X$-modules, then a morphism $\mF\to\mG$ is a natural transformation $\lambda:\mF\to\mG$ that respects the module structure. 
\item Composition is given by the composition of sheaves
\end{itemize}
\end{defn}

\begin{defn}{Sheaf of Finite Type}{}  Let $(X,\mO_X)$ be a ringed space. Let $\mF$ be a sheaf of $\mO_X$-module. \begin{itemize}
\item $\mF$ is a sheaf of finite type. 
\item For any $U\subseteq X$ and any morphism $$\varphi:\mO_X^{\otimes n}|_U\to\mF|_U$$ of $\mO_X$-modules, then kernel of $\varphi$ is a sheaf of finite type. 
\end{itemize}
\end{defn}

\subsection{The Global Section Functor}
\begin{defn}{Global Section Functor}{} Let $X$ be a space. Define the global section functor to be the functor $\Gamma(X,-):\bold{Ab}(X)\to\bold{Ab}$ defined as follows. 
\begin{itemize}
\item A sheaf $\mF$ of groups on $X$ is sent to $\Gamma(X,\mF)=\mF(X)$
\item For a morphism of sheaves $\phi:\mF\to\mG$, $\Gamma(X,-)$ sends it to the ring homomorphism $\phi(X):\mF(X)\to\mG(X)$
\end{itemize}
\end{defn}

In general, we can define functor by sending every sheaf to a the local section of a chosen open set. This operation in general is left exact. This means that we have the following proposition. 

\begin{crl}{}{} Let $X$ be a space. The global section functor $\Gamma(X,-):\bold{Ab}(X)\to\bold{Ab}$ is a left exact functor. 
\begin{proof}
The global section functor is a special case of the above proposition. 
\end{proof}
\end{crl}

\begin{defn}{Cohomology Functors}{} Let $X$ be a space. Define the cohomology functors $$H^i(X,-)=R^i\Gamma(X,-):\bold{Ab}(X)\to\bold{Ab}$$ of $X$ to be the right derived functors of $\Gamma(X,-):\bold{Ab}(X)\to\bold{Ab}$. For any sheaf $\mF$ of abelian groups on $X$, the groups $H^i(X,\mF)$ are called the cohomology groups of $\mF$. 
\end{defn}

\begin{prp}{}{} Let $(X,\mO_X)$ be a ringed space. Then the derived functors of $$\Gamma(X,-):{_{\mO_X}\bold{Mod}}(X)\to\bold{Ab}$$ coincide with the derived functors of $\Gamma(X,-):\bold{Ab}(X)\to\bold{Ab}$. 
\end{prp}

Thus there is no distinction in considering the cohomology groups of sheaf of $\mO_X$-modules and when considering them as simply a sheaf of abelian groups. 

\subsection{Flasque Sheaves}
\begin{defn}{Flasque Sheaves}{} A sheaf $\mF$ on a space $X$ is said to be flasque if for every pair of open sets $V\subset U$, the restriction map $\mF(U)\to\mF(V)$ is surjective.

\begin{lmm}{}{} Let $X$ be a space and let $\mF,\mF',\mF''$ be sheaves on $X$ such that there is an exact sequence \\

	0 & \mF & \mF' & \mF'' & 0
	\arrow[from=1-1, to=1-2]
	\arrow[from=1-2, to=1-3]
	\arrow[from=1-3, to=1-4]
	\arrow[from=1-4, to=1-5]
\end{tikzcd}}\\
of sheaves. Then the following are true. 
\begin{itemize}
\item If $\mF$ and $\mF'$ are flasque, then $\mF''$ is flasque. 
\item If $\mF$ is flasque, then for any $U\subseteq X$ open, there is an exact sequence \\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	0 & \mF(U) & \mF'(U) & \mF''(U) & 0
	\arrow[from=1-1, to=1-2]
	\arrow["\phi", from=1-2, to=1-3]
	\arrow["\psi", from=1-3, to=1-4]
	\arrow[from=1-4, to=1-5]
\end{tikzcd}}
\end{itemize}
\begin{proof}
Suppose that $\mF$ and $\mF'$ are flasque. Then by naturality we obtain a commutative diagram \\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	0 & {\mF(U)} & {\mF'(U)} & {\mF''(U)} & 0 \\
	0 & {\mF(V)} & {\mF'(V)} & {\mF''(V)} & 0
	\arrow[from=1-1, to=1-2]
	\arrow["\phi(U)", from=1-2, to=1-3]
	\arrow[two heads, from=1-2, to=2-2]
	\arrow["\psi(U)", from=1-3, to=1-4]
	\arrow[two heads, from=1-3, to=2-3]
	\arrow[from=1-4, to=1-5]
	\arrow[from=1-4, to=2-4]
	\arrow[from=2-1, to=2-2]
	\arrow["\phi(V)", from=2-2, to=2-3]
	\arrow["\psi(V)", from=2-3, to=2-4]
	\arrow[from=2-4, to=2-5]
\end{tikzcd}}\\
for any $V\subseteq U$ open sets. Let $s\in\mF''(V)$. Then by surjectivity of $\psi(V)$ and $\mF'(U)\to\mF'(V)$, there exists $t\in\mF'(U)$ that maps correspondingly by the same maps to $s$. Then using the map $\phi(U)$ we can send $s$ to an element $r\in\mF''(U)$ such that $r$ is sent to $s$ under the map $\mF''(U)\to\mF''(V)$. Thus this map is surjective and $\mF''$ is flasque. \\

Now suppose that $\mF$ is flasque. 
\end{proof}
\end{lmm}

\begin{prp}{}{} Let $X$ be a space and let $\mF$ be a sheaf on $X$. If $\mF$ is flasque then $\mF$ is acyclic for $\Gamma(X,-)$. Explicitly, this means that $$H^i(X,\mF)=0$$ for all $i>0$. 
\begin{proof}
Since $\bold{Ab}(X)$ has enough injective objects, suppose that $\mI$ is an injective object of $\bold{Ab}(X)$ such that $\mF$ is a subobject of $\mI$. Let $\mG$ be the quotient sheaf $\mF/\mI$. Then there is an exact sequence \\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	0 & \mF & \mI & \mG & 0
	\arrow[from=1-1, to=1-2]
	\arrow[from=1-2, to=1-3]
	\arrow[from=1-3, to=1-4]
	\arrow[from=1-4, to=1-5]
\end{tikzcd}}\\
By lemma 3.2.2, $\mI$ is flasque. By 3.2.3, $\mG$ is flasque. By 3.2.3, there is an exact sequence \\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	0 & \mF(X) & \mI(X) & \mG(X) & 0
	\arrow[from=1-1, to=1-2]
	\arrow[from=1-2, to=1-3]
	\arrow[from=1-3, to=1-4]
	\arrow[from=1-4, to=1-5]
\end{tikzcd}}\\
Since $\mI$ is injective, we have that $H^i(X,\mI)=0$ for all $i>0$. By passing to the long exact sequence in cohomology, we deduce that $H^1(X,\mF)=0$ and $H^i(X,\mF)=H^{i-1}(X,\mG)$ for each $i\geq 2$. Applying the same method again gives $H^1(X,\mG)=0$. Thus by induction, we conclude that $H^i(X,\mF)=0$ for all $i>0$. 
\end{proof}
\end{prp}

\end{lmm}

\begin{defn}{Canonical Morphism}{}\\
Let $X$ be a space. Let $\mF\in\bold{Ab}(X)$ be a sheaf on $X$. Define the canonical morphism $\alpha_F:F\to\text{Gd}(\mF)$ by $\alpha_F:F(U)\to\text{Gd}(\mF)$ the formula $$s\mapsto(s_x)_{x\in U}$$
\end{defn}

\begin{defn}{The Godement Functor}{}\\
Let $X$ be a space. Define the Godement functor $\text{Gd}:\bold{Ab}(X)\to\bold{Ab}(X)$ as follows. 
\begin{itemize}
\item Each sheaf $\mF\in\bold{Ab}(X)$ is sent to the godement sheaf $\text{Gd}(\mF)$. 
\item Each morphism $\mF\to\mG$ is sent to the morphism of sheaves defined by the product of the induced map on stalks. 
\end{itemize}
\end{defn}

\begin{prp}{}{}\\
Let $X$ be a space. Let $\mF\in\bold{Ab}(X)$ be a sheaf. Then $\text{Gd}(\mF)$ is flasque. 
\end{prp}

\begin{defn}{Godement Resolution}{}\\
Let $X$ be a space. Let $\mF\in\bold{Ab}(X)$ be a sheaf. Define the Godement resolution $\text{Gd}(\mF)^\bullet$ of $\mF$ as follows. 
\begin{itemize}
\item Define $\text{Gd}(\mF)^0=\mF$. 
\item For each $n\in\N^+$, define $\text{Gd}(\mF)^n=\text{Gd}(\text{coker}(\alpha_{\text{Gd}(\mF)^{n-1}}))$. 
\item Define the morphisms $\text{Gd}(\mF)^n\to\text{Gd}(\mF)^{n+1}$ to be $\alpha_{\text{Gd}(\mF)^n}$. \end{itemize}
\end{defn}

\begin{crl}{}{}\\
Let $X$ be a space. Let $\mF\in\bold{Ab}(X)$ be a sheaf. Then $\text{Gd}(\mF)^\bullet$ is an acyclic resolution for $\Gamma(X,-)$. 
\end{crl}

Consequently, the sheaf cohomology of $\mF$ acquires the formula $$H^i(X;\mF)=H^i(\Gamma(X,\text{Gd}(\mF)^\bullet)$$

\subsection{\v{C}ech Cohomology}
\begin{defn}{\v{C}ech Complex}{} Let $X$ be a topological space and $\mU=\{U_i|i\in I\}$ an open cover of $X$ where $I$ is an indexing set. For any $(i_0,\dots,i_k)\in I^{k+1}$, denote $$U_{i_0,\dots,i_k}=U_{i_0}\cap U_{i_1}\cap\dots\cap U_{i_k}$$ Define for each $k$, $$C^k(X,\mU,\mF)=\prod_{(i_0,\dots,i_k)\in I^{k+1}}\mF(U_{i_0,\dots,i_k})$$ Furthermore, define a boundary map $d:C^k(X,\mU,\mF)\to C^{k+1}(X,\mU,\mF)$ by $$c_{i_0\dots,i_k}\overset{d}{\mapsto}\sum_{s=0}^{k+1}(-1)^s\text{res}(c_{i_0,\dots,\hat{i}_s,\dots,i_{k+1}})$$ Define the \v{C}ech complex to be $(C^\bullet(X,\mU,\mF),d)$. 
\end{defn}

\begin{lmm}{}{} For any space $X$ and any open cover $\mU$ of $X$, $(C^\bullet(X,\mU,\mF),d)$ is indeed a chain complex. 
\end{lmm}

\begin{defn}{\v{C}ech Cohomology}{} Let $(C^\bullet(X,\mU,\mF),d)$ be a \v{C}ech complex. Define the $k$th cohomology group of it to be $$\text{\v{H}}^k(X,\mU,\mF)=\frac{\ker(C^k(X,\mU,\mF)\to C^{k+1}(X,\mU,\mF))}{\im(C^{k-1}(X,\mU,\mF)\to C^k(X,\mU,\mF))}=H(C^\bullet(X,\mU,\mF),d)$$
\end{defn}

\begin{lmm}{}{} For any \v{C}ech complex, we have that $\text{\v{H}}^0(X,\mU,\mF)=\mF(X)$. 
\end{lmm}

\begin{prp}{}{} Let $X$ be a topological space and $\mU$ an open cover of $X$. If the open sets $U_{i_0,\dots,i_k}$ satisfy that $H^k(U_{i_0,\dots,i_k},\mF)=0$ for all $k>0$, then $$H^k(X,\mF)=\text{\v{H}}^k(X,\mU,\mF)$$
\end{prp}

\begin{defn}{The Sheafified \v{C}ech Complex}{}\\
Let $X$ be a space. Let $\mU$ be an open cover of $X$. Let $\mF$ be a sheaf on $X$. Define the sheafified \v{C}ech complex $\mC^\bullet(X,\mU,\mF)$ by the following. 
\begin{itemize}
\item For each $p\in\N$, define $$\mC^p(X,\mU,\mF)=\prod_{i_0<\cdots<i_p}\iota_{U_{i_0,\dots,i_p}}(U_{i_0,\dots,i_p})\in\bold{Ab}(X)$$ where $\iota_U:U\hookrightarrow X$ denotes the inclusion map for an open subset $U\subseteq X$. 
\item Define the boundary map $d:\mC^p(X,\mU,\mF)\to\mC^{p+1}(X,\mU,\mF)$ to be the sheafified version of the boundary map in the \v{C}ech complex. 
\end{itemize}
\end{defn}

Note that by construction, we have $\Gamma(X,\mC^p(X,\mU,\mF))=C^p(X,\mU,\mF)$. 

\begin{lmm}{}{}\\
Let $X$ be a space. Let $\mU$ be an open cover of $X$. Let $\mF$ be a sheaf on $X$. Then the following are true. 
\begin{itemize}
\item $\mC^\bullet(X,\mU,\mF)$ is a resolution of $\mF$. 
\item If $X$ is flasque, then $\mC^\bullet(X,\mU,\mF)$ is a flasque (and hence acyclic) resolution of $\mF$. 
\end{itemize}
\end{lmm}

Note that the above lemma implies that $\text{\v{H}}^i(X,\mU,\mF)=H^i(X,\mF)=0$ if $\mF$ is flasque. 

\begin{lmm}{}{}\\
Let $X$ be a space. Let $\mU$ be an open cover of $X$. Let $\mF$ be a sheaf on $X$. Then there exists a map $$\text{\v{H}}^p(X,\mU,\mF)\to H^p(X,\mF)$$ that is natural in $\mF$ for all $p\in\N$. 
\end{lmm}

\begin{lmm}{Leray}{}\\
Let $X$ be a space. Let $\mU$ be an open cover of $X$. Let $\mF$ be a sheaf on $X$. Suppose that for any $U_{i_0,\dots,i_p}$, we have $H^k(U_{i_0,\dots,i_p},\mF_{U_{i_0,\dots,i_p}})=0$ for all $k,p\in\N$. Then the above natural maps are isomorphisms $$\text{\v{H}}^p(X,\mU,\mF)\cong H^p(X,\mF)$$
\end{lmm}

\begin{prp}{}{}\\
Let $X$ be a space. Let $\mF$ be a sheaf on $X$. Then there is a natural isomorphism $$\varinjlim_{\mU}\text{\v{H}}^1(X,\mU,\mF)\cong H^1(X,\mF)$$
\end{prp}

\begin{prp}{}{}\\
Let $(X,\mO_X)$ be a ringed space. Then there is a natural isomorphism $$\text{Pic}(X)\cong H^1(X,\mO_X^\ast)$$
\end{prp}

\subsection{Hypercohomology}
\begin{defn}{Hypercohomology of a Space}{}\\
Let $X$ be a space. Let $\mF^\bullet$ be a cochain complex of sheaves on $X$. Define the hyper cohomology of $X$ to be the right hyper-derived functor of the global section functor: $$\H^n(X;\mF^\bullet)=\R^n\Gamma(\mF^\bullet)$$
\end{defn}

\begin{prp}{}{}\\
Let $X$ be a space. Let $\mF^\bullet\in\bold{CCh}^+(\bold{Ab}(X))$ be a bounded below cochain complex. Let $\mI^\bullet\in\bold{CCh}^+(\bold{Ab}(X))$ be a bounded below cochain complex such that the following are true. 
\begin{itemize}
\item $I^n$ is injective for all $n$. 
\item There is a quasi-isomorphism $F^\bullet\to\mI^\bullet$. 
\end{itemize}
Then we have $$\H^n(X;\mF^\bullet)=H^i(\Gamma(X,\mI^\bullet))$$
\end{prp}

\begin{prp}{}{}\\
Let $X$ be a space. Let $\mF^\bullet\in\bold{CCh}^+(\bold{Ab}(X))$ be a cochain complex of sheaves on $X$. Then there is a natural isomorphism $$\H^n(X;\mF^\bullet)\cong H^n(\Gamma(X,\text{Tot}^{\prod}(\text{Gd}^\bullet,\mF^\bullet)))$$
\end{prp}

\section{Sites, Sieves and Grothendieck Topologies}
\subsection{Sites on a Category}
We now want to generalize the notion of sheaves to general categories, not just spaces. However, there is no notion of open sets in an arbitrary category. Grothendieck realized that the essence of sheaves comes from the data of the open covers $U=\bigcup_{i\in I}U_i$ for each open set $U$ in a space $X$. 

\begin{defn}{Covers}{} Let $\mC$ be a category. Let $U\in\mC$ be an object. A cover of $U$ is a collection of morphisms $$\{U_i\to U\;|\;i\in I\}$$
\end{defn}

\begin{defn}{Coverages (nLab)}{} Let $\mC$ be a category. A coverage $\text{Cov}(\mC)$ of $\mC$ consists a collection of covers for each $U\in\mC$, such that the following are true. 
\begin{itemize}
\item If $\{f_i:U_i\to U\;|\;i\in I\}$ is a cover of $U$ and $g:V\to U$ is a morphism in $\mC$, then there exists a cover $\{h_j:V_j\to V\}$ of $V$ such that every $g\circ h_j:V_j\to U$ factors through the cover of $U$. Or in other words, for each $h_j:V_j\to V$, there exists $U_i$ and $f_i:U_i\to U$ such that the following diagram commutes: \\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	{V_j} & {U_i} \\
	V & U
	\arrow["k", dashed, from=1-1, to=1-2]
	\arrow["{h_j}"', from=1-1, to=2-1]
	\arrow["{f_i}", from=1-2, to=2-2]
	\arrow["g"', from=2-1, to=2-2]
\end{tikzcd}}\\
\end{itemize}
\end{defn}

\begin{defn}{Coverages (Stacks)}{} Let $\mC$ be a category. A coverage $\text{Cov}(\mC)$ of $\mC$ consists a collection of covers for each $U\in\mC$, such that the following are true. 
\begin{itemize}
\item If $V\to U$ is an isomorphism then $\{V\to U\}\in\text{Cov}(\mC)$
\item If $\{U_i\to U\;|\;i\in I\}\in\text{Cov}(\mC)$ and $\{V_{i,j}\to U_i\;|\;j\in J_i\}\in\text{Cov}(\mC)$ for each $i\in I$, then $\{V_{i,j}\to U_i\to U\;|\;i,j\in I\}\in\text{Cov}(\mC)$
\item If $\{f_i:U_i\to U\;|\;i\in I\}\in\text{Cov}(\mC)$ and $g:V\to U$ is a morphism, then the pullback \\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	{U_i\times_UV} & {U_i} \\
	V & U
	\arrow["\text{proj.}", dashed, from=1-1, to=1-2]
	\arrow["{\text{proj.}}"', from=1-1, to=2-1]
	\arrow["{g}", from=1-2, to=2-2]
	\arrow["f_i"', from=2-1, to=2-2]
\end{tikzcd}}\\
exists for all $i\in I$ and $\{U_i\times_UV\to U\}\in\text{Cov}(\mC)$. 
\end{itemize}
\end{defn}

\begin{defn}{Sites (nLab)}{} A site is a category $\mC$ together with a coverage $\text{Cov}(\mC)$. 
\end{defn}

\begin{defn}{Continuous Functors}{} Let $\mC$ and $\mD$ be two sites. Let $F:\mC\to\mD$ be a functor. We say that $F$ is continuous if the following are true. 
\begin{itemize}
\item $F$ preserves fiber products
\item $F$ sends covers to covers
\end{itemize}
\end{defn}

\begin{defn}{Sheaves on Sites}{} Let $(\mC,\text{Cov}(\mC))$ be a site. Let $\mF$ be a presheaf of sets on $\mC$. We say that $\mF$ if a sheaf on $\mC$ if for all $\{f_i:U_i\to U\;|\;i\in I\}\in\text{Cov}(\mC)$, the following diagram \\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	{\mF(U)} & {\prod_{i\in I}\mF(U_i)} & {\prod_{i,j\in I}\mF(U_i\times_UU_j)}
	\arrow["p", from=1-1, to=1-2]
	\arrow["q", shift left, from=1-2, to=1-3]
	\arrow["r"', shift right, from=1-2, to=1-3]
\end{tikzcd}}\\
is an equalizer diagram, where $p$ is the product of the maps $\mF(f_i)$, $q$ is the product of the maps $\mF(\text{proj}_i:U_i\times_UU_j\to U_i)$ and $r$ is the product of the maps $\mF(\text{proj}_j:U_i\times_UU_j\to U_j)$. 
\end{defn}

\begin{defn}{Abelian Sheaves}{} Let $\mC$ be a site. Define the category of abelian sheaves $$\bold{Ab}(\mC)$$ over $\mC$ to consist of the following data. 
\begin{itemize}
\item The objects are the abelian sheaves on $\mC$
\item The morphisms are the natural transformations of functors
\item Composition is given by the composition of natural transformations
\end{itemize}
\end{defn}

\begin{defn}{Global Sections Functor}{} Let $\mC$ be a site. Define the global section functor $$\Gamma(-,\mC):\bold{Ab}(\mC)\to\bold{Ab}$$ by the following data.

\end{defn}
