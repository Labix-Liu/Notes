\section{Introduction to Presheaves and Sheaves}
\subsection{The Category of Presheaves of Sets}
\begin{defn}{The Category of Open Sets}{}\\
Let $(X,\mT)$ be a topological space. Define the category $\bold{Open}(X)$ to consist of the following data. 
\begin{itemize}
\item The objects are given by the open sets $\mT$ of $X$. 
\item For two open sets $U,V\in\mT$, define the morphisms from $U$ to $V$ by $$\Hom_{\bold{Open}(X)}(U,V)=\begin{cases}
\{\iota:U\hookrightarrow V\} & \text{ if }U\subseteq V\\
\emptyset & \text{ otherwise }
\end{cases}$$
\item Composition is given by the composition of functions. 
\end{itemize}
\end{defn}

\begin{defn}{Presheaves on a Space}{}\\
Let $X$ be a space. A presheaf on $X$ is a contravariant functor $\mF:\bold{Open}(X)\to\bold{Set}$. 
\end{defn}

\begin{eg}{The Constant Presheaf}{}\\
Let $X$ be a space. Let $S$ be a set. Define the constant presheaf with value $S$ to be the presheaf $\underline{S}_\text{pre}$ defined as follows. 
\begin{itemize}
\item For each open set $U\subseteq X$, we have $$\underline{S}_\text{pre}(U)=S$$
\item For each inclusion $V\subseteq U$ of open sets, define $$\text{res}_V^U=\text{id}_S$$
\end{itemize}
\end{eg}

\begin{defn}{The Category of Presheaves}{}\\
Let $X$ be a topological space. Define the category of presheaves on $X$ to be $$\bold{PSh}(X)=\text{Func}(\bold{Open}(X)^\text{op},\bold{Set})$$
\end{defn}

\begin{prp}{}{}\\
Let $X$ be a space. Then $\bold{PSh}(X)$ admits all finite limits. 
\end{prp}

We now have the definition of a morphism of presheaves, which is just a natural transformation, and the definition of isomorphisms, which is just a natural isomorphism. 

\begin{defn}{The Stalk of a Presheaf}{}\\
Let $X$ be a space. Let $\mF$ be a presheaf on $X$. Let $p\in X$. Consider the full subcategory $\mJ_p$ of $\bold{Open}(X)$ consisting of open sets in $X$ that contain $p$. Define the stalk of $\mF$ at $p$ to be the colimit $$\mF_p=\underset{\mJ_p}{\text{colim}}\mF(V_p)$$
\end{defn}

\begin{defn}{Induced Map of Stalks}{}\\
Let $X$ be a topological space. Let $\mF,\mG$ be presheaves on $X$. Let $p\in X$. Define the induced map of stalks $$\phi_p:\mF_p\to\mG_p$$ to be the map induced by the universal product of colimits. 
\end{defn}

\subsection{The Category of Sheaves of Sets}
\begin{defn}{Sheaves}{}\\
Let $X$ be a space. A sheaf on $X$ is a presheaf $\mF:\bold{Open}\to\bold{Sets}$ such that the following are true. 
\begin{itemize}
\item Identity: If $\{U_i|i\in I\}$ is an open cover of $U$ and $\phi_1,\phi_2\in\mF(U)$ and $\phi_1|_{U_i}=\phi_2|_{U_i}$ for all $i$, then $\phi_1=\phi_2$
\item Gluing: If $\{U_i|i\in I\}$ is an open cover of $U$ and $\phi_i\in\mF(U_i)$ for all $i\in I$ such that $\phi_i|_{U_i\cap U_j}=\phi_j|_{U_i\cap U_j}$ for all $i,j\in I$, then there is exists some $\phi\in\mF(U)$ such that $\phi|_{U_i}=\phi_i$ for all $i\in I$. 
\end{itemize}
\end{defn}

\begin{eg}{}{}\\
Let $X,Y$ be spaces. Define a presheaf $\mF$ on $X$ by $$\mF(U)=\{f:U\to Y\;|\;f\text{ is continuous }\}$$ for $U\subseteq X$ an open set, together with restriction maps. Then $\mF$ is a sheaf. 
\begin{proof}\\
Identity: Suppose that $f,g\in\mF(U)$ for some open set $U\subseteq X$ and $f|_{U_i}=g|_{U_i}$ for $\{U_i\;|\;i\in I\}$ an open cover of $U$. Then for any $x\in U$, there exists $i\in I$ such that $x\in U_i$, and so $$f(x)=f|_{U_i}(x)=g|_{U_i}(x)=g(x)$$ Hence $f=g$. \\~\\

Gluing: Suppose that $f_i\in\mF(U_i)$ for $\{U_i\;|\;i\in I\}$ an open cover of an open set $U\subseteq X$ such that $f_i|_{U_i\cap U_j}=f_j|_{U_i\cap U_j}$. Define $f\in\mF(U)$ by $f(x)=f_i(x)$ for $x\in U_i$. This is well defined since $\{f_i\}$ agree on intersections. It remains to show that $f$ is continuous. Let $W\subseteq Y$ be an open set. I claim that $f^{-1}(W)=\bigcup_{i\in I}f_i^{-1}(W)$. Let $x\in f^{-1}(W)$. Then $f(x)\in W$. Since $x\in W\subseteq U=\bigcup_{i\in I}U_i$, there exists $i\in I$ such that $x\in U_i$. Hence $f_i(x)=f(x)\in W$. Thus $x\in f_i^{-1}(W)$. Conversely, suppose that $x\in f_i^{-1}(W)$. Then clearly $f(x)=f_i(x)\in W$ so that $x\in f^{-1}(W)$. Thus $f^{-1}(W)=\bigcup_{i\in I}f_i^{-1}(W)$. Since each $f_I$ is continuous and $W$ is an open set, $f_i^{-1}(W)$ is open and so the arbitrary union of $f_i^{-1}(W)$ is open. Hence $f^{-1}(W)$ is open. 
\end{proof}
\end{eg}

\begin{eg}{Locally Constant Sheaf}{}\\
Let $X$ be a space. Let $S$ be a set. Define the locally constant presheaf $\underline{S}$ of $X$ with values in $S$ as follows. 
\begin{itemize}
\item For each open set $U\subseteq X$, define $$\underline{S}(U)=\{f:U\to S\;|\;f\text{ is continuous where }S\text{ has the discrete topology }\}$$
\item For each inclusion $V\subseteq U$ of open subsets, define the induced map $\underline{S}(U)\to\underline{S}(V)$ by $f\mapsto f|_V$. 
\end{itemize}
The following are true regarding the locally constant sheaf. 
\begin{itemize}
\item $\underline{S}$ is a sheaf. 
\item Let $U\subseteq X$ be an open set. Let $\mB$ be the set of connected components of $U$. Then $$\underline{S}(U)\cong\prod_{C\in\mB}S$$
\item For all $p\in X$, $\underline{S}_p=S$. 
\end{itemize}
\begin{proof}\\
Let $Y$ be a connected component of $U$. Let $f\in\underline{S}(Y)$. I claim that $f$ is constant. Let $y_0\in Y$. Define $A=\{y\in Y\;|\;f(y)=f(y_0)\}=f^{-1}(f(y_0))$ and $B=\{y\in Y\;|\;f(y)\neq f(y_0)\}$. Clearly $Y=A\cup B$ and $A\cap B=\emptyset$. Both $A$ and $B$ are open sets since $S$ has a discrete topology. Since $Y$ is connected and $A$ is non-empty, this implies that $B=\emptyset$. Hence $Y=A$ and $f$ is constant on $Y$. Thus there is a bijection $\underline{S}(Y)\cong S$. \\

Define a function $\underline{S}(U)\to\prod_{C\in\mB}S$ as follows. For each connected component $C\in\mB$ of $U$, choose $x_C\in C$. Then the map is defined by $f\mapsto(f(x_C))$. This map is injective since $f(x_C)=g(x_C)$ implies that $f(C)=g(C)$ for all connected components $C\in\mB$. Hence $f=g$. It is surjective since for any choice of sequence $(x_C)_{C\in\mB}$, the map $f:U\to S$ defined by $f(x)=x_C$ for $x\in C$ is continuous because $S$ is discrete and $\{x_C\}$ is a closed set, and $f^{-1}(x_C)=C$ is a connected component and so is a closed set. 
\end{proof}
\end{eg}

\begin{prp}{}{}\\
Let $X$ be a space. Let $\mF$ be a presheaf. Then the following are equivalent characterizations. 
\begin{itemize}
\item $\mF$ is a sheaf
\item For any open covering $U=\bigcup_{i\in I}U_i$, the sequence \\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	0 & {\mF(U)} & {\prod_{i\in I}\mF(U_i)} & {\prod_{i,j\in I}\mF(U_i\cap U_j)}
	\arrow[from=1-1, to=1-2]
	\arrow["f", from=1-2, to=1-3]
	\arrow["g", from=1-3, to=1-4]
\end{tikzcd}}\\
where $f$ is given by the product of the unique maps $\mF(U)\to\mF(U_i)$ and $g$ is given by sending $(s_i)_{i\in I}$ to $(s_i|_{U_i\cap U_j}-s_j|_{U_i\cap U_j})_{i,j\in I}$ is exact. 
\item For any open covering $U=\bigcup_{i\in I}U_i$, the following diagram \\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	{\mF(U)} & {\prod_{i\in I}\mF(U_i)} & {\prod_{i,j\in I}\mF(U_i\cap U_j)}
	\arrow["f", from=1-1, to=1-2]
	\arrow["g", shift left, from=1-2, to=1-3]
	\arrow["h"', shift right, from=1-2, to=1-3]
\end{tikzcd}}\\
is an equalizer diagram, where $f$ is given by the product of the unique maps $\mF(U)\to\mF(U_i)$, $g$ is given by sending $(s_i)_{i\in I}$ to $(s_i|_{U_i\cap U_j})_{i,j\in I}$, $h$ is given by sending $(s_i)_{i\in I}$ to $(s_j|_{U_i\cap U_j})_{i,j\in I}$. 
\end{itemize} 
\begin{proof}~\\
We first note that the second and third condition are equivalent. Therefore we just have to show that the first condition is equivalent to the third one. Let $\mF$ be an abelian sheaf. We want to show that $\im(f)=\{x\in\prod_{i\in I}F(U_i)\;|\;g(x)=h(x)\}$. Therefore we begin with $x\in\im(f)$. Then $f(x)=(x_i)_{i\in I}$. But we have that $g((x_i)_{i\in I})=(x_i|_{U_i\cap U_j})_{i,j\in I}$ and $h((x_i)_{i\in I})=(x_j|_{U_i\cap U_j})_{i,j\in I}$. We want to show that for all $i,j\in I$, we have that $x_i|_{U_i\cap U_j}=x_j|_{U_i\cap U_j}$. But $x_i$ is just the restriction of $x$ to $U_i$ and $x_j$ is the restriction of $x$ to $U_j$. But by definition of a sheaf, restricting to $V_i$ and further to $V_{ij}$ is the same as directly restricting to $V_{ij}$. Similarly, restricting to $V_j$ and further to $V_{ij}$ is the same as directly restricting to $V_{ij}$. We conclude that applying $g$ and $h$ to $(x_i)_{i\in I}$ give the same element. Thus $x\in\{x\in\prod_{i\in I}F(U_i)\;|\;g(x)=h(x)\}$. \\

Now let $(y_i)_{i\in I}\in\{x\in\prod_{i\in I}F(U_i)\;|\;g(x)=h(x)\}$. This means that $y_i|_{U_i\cap U_j}=y_j|_{U_i\cap U_j}$ for all $i,j\in I$. By the gluing axiom, there exists $y\in C(U)$ such that $y|_{U_i}=y_i$. Thus $y$ is in the image of $f$ and we conclude that $y\in\im(f)$. \\

Finally, we also need to show that $f$ is injective. Suppose that $x,y\in\mF(U)$ is such that $x_i=y_i$ for all $i\in I$. Then by the identity axiom we conclude that $x=y$. Thus $f$ is injective. Thus the equalizer condition is satisfied. \\

Now suppose that the equalizer condition is satisfied. Then $f$ is injective. This means that for $x,y\in\mF(U)$ such that $x_i=y_i$ for all $i\in I$, then $x=y$. But this is precisely the identity axiom. Suppose that $\im(f)=\text{Eq}(g,h)$. Let $s_i\in\mF(U_i)$ such that $$s_i|_{U_i\cap U_j}=g(s_i)=h(s_i)=s_j|_{U_i\cap U_j}$$ for all $i,j\in I$. Then $(s_i)_{i\in I}\in\text{Eq}(g,h)=\im(f)$. Which means that there exists some $s\in\mF(U)$ such that $s|_{U_i}=s_i$. But this is the content of the gluing axiom. Since the identity and the gluing axiom are satisfied, we conclude that $\mF$ is a sheaf. 
\end{proof}
\end{prp}

\begin{prp}{Gluing Sheaves}{}\\
Let $X$ be a space. Let $\{U_i\;|\;i\in I\}$ be an open cover of $X$. Let $\mF_i$ be a sheaf of sets on $U_i$. Let $\phi_{i,j}:\mF_i|_{U_i\cap U_j}\to\mF_j|_{U_i\cap U_j}$ be a collection of morphism of sheaves for ranging $i,j\in I$ such that the following are true. 
\begin{itemize}
\item For each $i,j\in I$, $\phi_{i,j}$ is an isomorphism. 
\item For each $i,j,k\in I$, we have $\phi_{j,k}\circ\phi_{i,j}=\phi_{i,k}$. 
\end{itemize}
Then there exists a unique (up to unique isomoprhism) sheaf of sets $\mF$ on $X$ such that there are isomorphisms $\mF|_{U_i}\overset{\cong}{\to}\mF_i$ for all $i\in I$. 
\end{prp}

\begin{defn}{The Category of Sheaves of Sets}{}\\
Let $X$ be a space. Define the category of sheaves of sets $\bold{Sh}(X)$ on $X$ to be the full subcategory of $\bold{PSh}(X,\bold{Set})$ consisting of sheaves of sets. 
\end{defn}

\begin{prp}{}{}\\
Let $X$ be a space. Then $\bold{Sh}(X)$ admits all finite limits. 
\end{prp}

\begin{prp}{}{}\\
Let $X$ be a space. Let $\mF,\mG$ be sheaves on $X$. Let $\varphi:\mF\to\mG$ be a morphism. Then $\varphi$ is an isomorphism if and only if $\varphi_p:\mF_p\to\mG_p$ is a bijection for all $p\in X$. 
\end{prp}

\begin{prp}{}{}\\
Let $X$ be a space. Let $\mF,\mG$ be sheaves on $X$. Let $\varphi,\psi:\mF\to\mG$ be morphisms. Then $\varphi=\phi$ if and only if $\varphi_p=\psi_p$ for all $p\in X$. 
\end{prp}

\subsection{The Sheafification Functor}
\begin{defn}{Compatible Germs}{}\\
Let $X$ be a space. Let $\mF$ be a presheaf on $X$. Let $([(s_p,U_p)])_{p\in U}\in\prod_{p\in U}\mF_p$. We say that the element is a compatible germ if for all $p\in U$, there exists an open subset $p\in V_p\subseteq U$ such that $$[(s_p|_{V_p},V_p)]=[(s_q,U_q)]\in\mF_q$$ for all $q\in V_p$. 
\end{defn}

\begin{lmm}{}{}\\
Let $X$ be a space. Let $\mF$ be a presheaf on $X$. Let $([(s_p,U_p)])_{p\in U}\in\prod_{p\in U}\mF_p$. Then the element is a compatible germ if and only if there exists an open cover $\bigcup_{i\in I}U_i$ of $U$ and $f_i\in\mF(U_i)$ such that $$[(f_i,U_i)]=[(s_p,U_p)]\in\mF_p$$ for all $i$ and all $p\in U_i$. 
\end{lmm}

\begin{defn}{Sheafification}{}\\
Let $X$ be a space. Let $\mF$ be a presheaf on $X$. Define the sheafification of $\mF$ to be the sheaf $$\mF^\text{sh}:\bold{Open}(X)\to\bold{Set}$$ given as follows. 
\begin{itemize}
\item For each open set $U\subseteq X$, we have $$\mF^\text{sh}(U)=\left\{([(s_p,U_p)])_{p\in U}\in\prod_{p\in U}\mF_p\;\bigg{|}\;([(s_p,U_p)])_{p\in U}\text{ is a compatible germ }\right\}$$
\item For each $V\subseteq U$ an inclusion, there is a unique morphism $\mF^\text{sh}(U)\to\mF^\text{sh}(V)$ given by $$([(s_p,U_p)])_{p\in U}\in\mF^\text{sh}(U)\mapsto([(s_p,U_p])_{p\in V}$$
\end{itemize}
\end{defn}

\begin{lmm}{}{}\\
Let $X$ be a space. Let $\mF$ be a presheaf on $X$. Then $\mF^\text{sh}(X)$ is a sheaf. 
\end{lmm}

\begin{defn}{The Universal Morphism}{}\\
Let $X$ be a space. Let $\mF$ be a presheaf on $X$. Define the universal morphism $\theta_\mF:\mF\to\mF^\text{sh}$ as follows. For each open set $U\subseteq X$, define $$\theta_\mF(U):\mF(U)\to\prod_{p\in U}\mF_p$$ by $s\mapsto([(s,U)])_{p\in U}$. 
\end{defn}

\begin{lmm}{}{}\\
Let $X$ be a space. Let $\mF$ be a presheaf on $X$. Let $U\subseteq X$ be an open set. Then the following are true. 
\begin{itemize}
\item $\im(\theta_\mF(U))=\mF^\text{sh}(U)$. 
\item If $\mF$ is a sheaf, then $\theta_\mF(U):\mF\to\mF^\text{sh}$ is injective. 
\end{itemize}
\end{lmm}

\begin{lmm}{}{}\\
Let $X$ be a space. Let $\mF$ be a sheaf on $X$. Then the following are true. 
\begin{itemize}
\item $\theta$ induces an isomorphism $$\theta_p:\mF_p\overset{\cong}{\longrightarrow}\mF_p^\text{sh}$$ for all $p\in X$. 
\item $\theta:\mF:\mF\to\mF^\text{sh}$ is an isomorphism. 
\end{itemize}
\end{lmm}

\begin{prp}{The Universal Property of Sheafification}{}\\
Let $X$ be a space. Let $\mF$ be a presheaf on $X$. Then the sheafification $\mF^\text{sh}$ together with $\theta_\mF$ satisfies the following universal property. For any sheaf $\mG$ on $X$ and any morphism $\varphi:\mF\to\mG$, there exists a unique morphism $\varphi^\text{sh}:\mF^\text{sh}\to\mG$ such that the following diagram commutes: \\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	\mF & {\mF^\text{sh}} \\
	& \mG
	\arrow["{\theta_\mF}", from=1-1, to=1-2]
	\arrow["\varphi"', from=1-1, to=2-2]
	\arrow["{\exists!\varphi^\text{sh}}", dashed, from=1-2, to=2-2]
\end{tikzcd}}\\
Moreover, $\mF^\text{sh}$ is the unique (up to unique isomorphism) sheaf of $X$ satisfying the above property. 
\end{prp}

\begin{defn}{The Sheafification Functor}{}\\
Let $X$ be a space. Define the sheafification functor $$(-)^\text{sh}:\bold{PSh}(X)\to\bold{Sh}(X)$$ as follows. 
\begin{itemize}
\item Each presheaf $\mF$ is sent to its sheafification $\mF^\text{sh}$. 
\item For each morphism of presheaves $\mF\to\mG$, the morphism of sheaves is given by the universal property of $\mF^\text{sh}$ applied to the composite $\mF\to\mG\to\mG^\text{sh}$. 
\end{itemize}
\end{defn}

\begin{prp}{}{}\\
Let $X$ be a space. Let $\iota:\bold{Sh}(X)\hookrightarrow\bold{PSh}(X)$ be the inclusion functor. Then there is an adjunction $$(-)^\text{sh}:\bold{PSh}(X)\overset{\dashv}{\rightleftarrows}\bold{Sh}(X):\iota$$
\end{prp}

\subsection{Pushforward and Pullback Sheaves}
\begin{defn}{Direct Image Sheaf}{}\\
Let $X,Y$ be spaces. Let $f:X\to Y$ be a map. Let $\mF$ be a presheaf on $X$. Define the direct image of sheaf $$f_\ast\mF:\bold{Open}(Y)\to\mC$$ as follows. 
\begin{itemize}
\item For each open set $U\subseteq Y$, define $f_\ast\mF(U)=\mF(f^{-1}(U))$. 
\item For each inclusion $V\subseteq U$, define the induced map $$\mF(f^{-1}(U))\to\mF(f^{-1}(V))$$ to be the map $\mF(\text{incl.}:f^{-1}(V)\hookrightarrow f^{-1}(U))$. 
\end{itemize}
\end{defn}

\begin{prp}{}{}\\
Let $X,Y$ be spaces. Let $f:X\to Y$ be a map. Let $\mF$ be a sheaf. Then $f_\ast\mF$ is a sheaf on $Y$. 
\begin{proof}\\
It is clear that $f_\ast\mF$ is a presheaf on $Y$. We need to check the identity and gluing axiom. By 1.3.4, without loss of generality we can assume that $\mC=\bold{Set}$. \\

Identity: Let $U\subseteq Y$ be an open set. Let $\{U_i\;|\;i\in I\}$ be an open cover of $U$. Denote the inclusion maps by $\iota_i:U_i\to U$. Let $s,t\in f_\ast\mF(U)$. Suppose that $f_\ast\mF(\iota_i)(s)=f_\ast\mF(\iota_i)(t)$ for all $i\in I$. This means that $s,t\in\mF(f^{-1}(U))$ is such that the images of $s,t$ in $\mF(f^{-1}(U_i))$ are equal for all $i\in I$. Since $\{f^{-1}(U_i)\}$ is an open cover of $f^{-1}(U)$ and $\mF$ is a sheaf on $X$, this means that $s=t$. \\

Gluing: Let $U\subseteq Y$ be an open set. Let $\{U_i\;|\;i\in I\}$ be an open cover of $U$. Suppose that $s_i\in f_\ast\mF(U_i)=\mF(f^{-1}(U_i))$ is such that the images of $s_i$ and $s_j$ in $f_\ast\mF(U_i\cap U_j)=\mF(f^{-1}(U_i\cap U_j))$ agree for all $i,j\in I$. Since $\{f^{-1}(U_i)\}$ is an open cover of $f^{-1}(U)$ and $\mF$ is a sheaf on $X$, we conclude that there exists $s\in\mF(f^{-1}(U))=f_\ast\mF(U)$ such that the image of $s$ in $\mF(f^{-1}(U_i))=f_\ast\mF(U_i)$ is $s_i$ for all $i\in I$. Thus the gluing axiom is satisfied. 
\end{proof}
\end{prp}

\begin{eg}{Skyscraper Sheaf}{}\\
Let $X$ be a space. Let $p\in X$. Let $S$ be a set. Let $\underline{S}$ be the constant sheaf on $\{p\}$. Define the skyscraper sheaf of $X$ at $p$ to be the pushforward $$\iota_\ast\underline{S}(U)=\underline{S}(\iota^{-1}(U))$$ where $\iota:\{p\}\hookrightarrow X$ is the inclusion map. 
\end{eg}

\begin{defn}{The Direct Image Sheaf Functor}{}\\
Let $X,Y$ be spaces. Let $f:X\to Y$ be a map. Let $\mF$ be a sheaf on $X$. Define the direct image sheaf functor $$f_\ast:\bold{Sh}(X)\to\bold{Sh}(Y)$$ by the following data. 
\begin{itemize}
\item For each sheaf $\mF$ of $X$, $f_\ast\mF$ is the direct image sheaf on $Y$. 
\item For each morphism $\varphi:\mF\to\mG$, $f_\ast\varphi:f_\ast\mF\to f_\ast\mG$ is the morphism given by $$\varphi(f^{-1}(U)):\mF(f^{-1}(U))\to\mG(f^{-1}(U))$$
\end{itemize}
\end{defn}

\begin{defn}{Inverse Image Sheaf}{}\\
Let $X,Y$ be spaces. Let $f:X\to Y$ be a map. Let $\mG$ be a sheaf on $Y$. Define the inverse image sheaf $f^{-1}\mG$ to be the sheafification of the presheaf on $X$ defined as follows. 
\begin{itemize}
\item For each open set $U\subseteq X$, let $\mJ(U)\subseteq\bold{Open}(X)$ be the full subcategory consisting of open sets of $Y$ the contain $f(U)$. Then the presheaf sends $U$ to $\underset{W\in\mJ(U)}{\lim}\mG(W)$. 
\item For each inclusion $V\subseteq U$, define the induced map to be induced by the universal property of the limit. 
\end{itemize}
\end{defn}

\begin{prp}{}{}\\
Let $X,Y$ be spaces. Let $f:X\to Y$ be a map. Let $\mG$ be a sheaf on $Y$. Then the inverse image sheaf $f^{-1}\mG$ is a sheaf. 
\end{prp}

\begin{defn}{The Inverse Image Sheaf Functor}{}\\
Let $X,Y$ be spaces. Let $f:X\to Y$ be a map. Let $\mF$ be a sheaf on $X$. Define the inverse image sheaf functor $$f^{-1}:\bold{Sh}(Y)\to\bold{Sh}(X)$$ by the following data. 
\begin{itemize}
\item For each sheaf $\mF$ of $Y$, $f^{-1}\mF$ is the inverse image sheaf on $X$. 
\item For each morphism $\varphi:\mF\to\mG$, the morphism $f^{-1}\mF\to f^{-1}\mG$ is induced by the universal property of limits. 
\end{itemize}
\end{defn}

\begin{eg}{}{}\\
Let $Y$ be a space. Let $p\in Y$. Let $\iota:\{p\}\hookrightarrow Y$ be the inclusion map. Let $\mG$ be a sheaf on $Y$. Then we have $$\left(\iota^{-1}\mG\right)(\{p\})=\mG_p$$
\end{eg}

\begin{prp}{}{}\\
Let $X,Y$ be spaces. Let $f:X\to Y$ be a map. Then there is an adjunction $$f^{-1}:\bold{Sh}(Y)\overset{\dashv}{\rightleftarrows}\bold{Sh}(X):f_\ast$$
\end{prp}

\begin{crl}{}{}\\
Let $X,Y$ be spaces. Let $f:X\to Y$ be a map. Let $\mG$ be a sheaf of $Y$. Then there is a natural isomorphism $$(f^{-1}\mG)_p\cong\mG_{f(p)}$$ for any $p\in X$. 
\begin{proof}
$f^{-1}$ is a left adjoint and stalks are colimits. Since left adjoints preserve colimits we are done. 
\end{proof}
\end{crl}

\section{Sheaves of Modules over a Sheaf of Rings}
\subsection{The Category of Ringed Spaces}
\begin{defn}{Ringed Space}{}\\
A ringed space is a topological space $X$ with a sheaf of rings $\mO_X$ on $X$. 
\end{defn}

\begin{defn}{Morphisms of Ringed Spaces}{}\\
Let $(X,\mathcal{O}_X)$ and $(Y,\mathcal{O}_Y)$ be ringed spaces. A morphism of ringed spaces from $(X,\mathcal{O}_X)$ to $(Y,\mathcal{O}_Y)$ is a pair $(f,f^\#)$ of continuous map $f:X\to Y$ and a map $f^\#:\mathcal{O}_Y\to f_\ast\mathcal{O}_X$ of sheaves of rings on $Y$. 
\end{defn}

\begin{defn}{Commutative Ringed Space}{}\\
Let $(X,\mO_X)$ be a ringed space. We say that $X$ is a commutative ringed space if $\mO_X$ is a sheaf of commutative rings. 
\end{defn}

\begin{defn}{Locally Ringed Spaces}{}\\
Let $(X,\mO_X)$ be a commutative ringed space. We say that $X$ is a locally ringed space if for all $p\in X$, $\mO_{X,p}$ is a local ring. 
\end{defn}

Suppose that $(X,\mO_X)$ and $(Y,\mO_Y)$ are locally ringed spaces. Given a point $p\in X$, we obtain induced maps $$f^\#:\mO_Y(V)\to f_\ast\mO_X(f^{-1}(V))$$ for any neighbourhood $V$ of $f(p)$. Taking the colimit, we obtain a map $$\mO_{Y,f(p)}=\text{colim}_V\mO_Y(V)\to\text{colim}_V\mO_X(f^{-1}(V))$$ As $V$ ranges through the neighbourhood of $f(p)$, $f^{-1}(V)$ ranges over a subset of the neighbourhoods of $p$ so that we obtain a map $$\mO_{Y,f(p)}=\text{colim}_V\mO_Y(V)\to\text{colim}_V\mO_X(f^{-1}(V))\to\mO_{X,p}$$

\begin{defn}{Morphisms of Locally Ringed Spaces}{}\\
Let $(X,\mathcal{O}_X)$ and $(Y,\mathcal{O}_Y)$ be locally ringed spaces. Then a morphism of locally ringed spaces is a morphism of ringed spaces such that for each $p\in X$, the induced map of local rings $$f_p^\#:\mO_{Y,f(p)}\to\mO_{X,p}$$ is a local homomorphism of local rings. 
\end{defn}

\begin{defn}{The Category of Locally Ringed Spaces}{}\\
Define the category $\bold{LRS}$ of locally ringed spaces as follows. 
\begin{itemize}
\item The objects are locally ringed spaces. 
\item The morphisms are morphisms of locally ringed spaces. \
\item Composition is given by the composition of continuous functions and composition of natural transformations. 
\end{itemize}
\end{defn}

\subsection{The Abelian Category Structure of Sheaves of Modules}
\begin{defn}{The Presheaf Kernel}{}\\
Let $(X,\mO_X)$ be a commutative ringed space. Let $\mF,\mG\in\bold{PMod}_{\mO_X}$. Let $\varphi:\mF\to\mG$ be a morphism. Define the kernel $\ker(\varphi)$ of $\varphi$ as follows. 
\begin{itemize}
\item For each open set $U\subseteq X$, define $\ker(\varphi)(U)=\ker(\varphi(U))$. 
\item For each inclusion $V\subseteq U$ of open sets, define the restriction morphism by the universal property of kernels as follows: \\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	{\ker(\varphi(U))} & {\mF(U)} & {\mG(U)} \\
	{\ker(\varphi(V))} & {\mF(V)} & {\mG(V)}
	\arrow[from=1-1, to=1-2]
	\arrow["{\exists!}"', dashed, from=1-1, to=2-1]
	\arrow["{\varphi(U)}", from=1-2, to=1-3]
	\arrow["{\text{res}_V^U}"', from=1-2, to=2-2]
	\arrow["{\text{res}_V^U}", from=1-3, to=2-3]
	\arrow[from=2-1, to=2-2]
	\arrow["{\varphi(V)}"', from=2-2, to=2-3]
\end{tikzcd}} \\~\\
\end{itemize}
\end{defn}

\begin{prp}{}{}\\
Let $(X,\mO_X)$ be a commutative ringed space. Let $\mF,\mG\in\bold{PMod}_{\mO_X}$. Let $\varphi:\mF\to\mG$ be a morphism. Then the following are true. 
\begin{itemize}
\item $\ker(\varphi)$ is the categorical kernel of $\varphi$ in $\bold{PMod}_{\mO_X}$. 
\item If $\mF,\mG\in\bold{Mod}_{\mO_X}$, then $\ker(\varphi)$ is a sheaf and is the categorical kernel of $\varphi$ in $\bold{Mod}_{\mO_X}$. 
\item $\ker(\varphi)_p=\ker(\varphi_p)$. 
\end{itemize}
\begin{proof}\\
\begin{itemize}
\item Let $U\subseteq X$ be an open set. Let $\{U_i\;|\;i\in I\}$ be an open cover of $U$. Let $f,g\in\ker(\varphi)(U)$ such that $\text{res}_{U_i}^U(f)=\text{res}_{U_i}^U(g)$. Consider the following commutative diagram: \\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	{\ker(\varphi(U))} & {\mF(U)} \\
	{\ker(\varphi(U_i))} & {\mF(U_i)}
	\arrow[from=1-1, to=1-2]
	\arrow["{\text{res}_V^U}"', from=1-1, to=2-1]
	\arrow["{\text{res}_V^U}"', from=1-2, to=2-2]
	\arrow[from=2-1, to=2-2]
\end{tikzcd}} \\~\\
Consider the images of $f$ and $g$ in $\mF(U)$. Since $f$ and $g$ map vertically down to the same element in $\ker(\varphi)(U_i)$, by commutativity of the diagram, the images of $f$ and $g$ in $\mF(U)$ map vertically to the same element in $\mF(U_i)$. Since this is true for all $i\in I$, the identity property of the sheaf $\mF$ implies that the image of $f$ and $g$ in $\mF(U)$ are equal. This means that $f-g\in\ker(\varphi(U))$ map to $0\in\mF(U)$. Since $\ker(\varphi)(U)$ is a subgroup of $\mF(U)$, we must have $f=g$. This proves the identity property of $\ker(\varphi)$. 
\item I claim that the induced map $\ker(\phi)_p\to\mF_p\to\mG_p$ is the zero map. Indeed, the map $\ker(\phi)_p\to\mF_p$ is induced by the maps $\ker(\phi)(U)\to\mF(U)$ for each open set $U\subseteq X$, which composes with $\mF(U)\to\mG(U)$ to give the zero map. Hence by the universal property of the kernel of $\phi_p$, we obtain a unique morphism $\ker(\phi)_p\to\ker(\phi_p)$. Since the morphism $\ker(\phi)\to\mF$ is the inclusion map for each open set, the morphism $\ker(\phi)_p\to\mF_p$ is also the inclusion map and so $\ker(\phi_p)\subseteq\ker(\phi_p)$. Conversely, suppose that $[(s,U)]\in\ker(\phi_p)$. Then we have $[(\phi(U)(s),U)]=[(0,V)]$ for some $V\subseteq X$ open. This means that there exists $W\subseteq U\cap V$ open such that $\phi(U)(s)|_W=0|_W$. But we have 
\begin{align*}
0|_W&=\phi(U)(s)|_W\\
&=\text{res}_W^U(\phi(U)(s))\\
&=\phi(W)(\text{res}_W^U(s))\tag{$\phi$ is a morphism}\\
&=\phi(W)(s|_W)
\end{align*}
and so $s|_W\in\ker(\phi)(W)$. By definition of the kernel of $\phi$, we have $[(s|_W,W)]\in\ker(\phi)_p$. But $[(s|_W,W)]=[(s,U)]$ and so we are done. \\

\end{itemize}
\end{proof}
\end{prp}

\end{proof}
\end{prp}

Thus the functor $$(-)(U):\bold{Mod}_{\mO_X}\to\bold{Mod}_{\mO_X(U)}$$ is left exact. It is in general not right exact. 

\begin{prp}{}{}\\
Let $X$ be a topological space. Let $\mF,\mG\in\bold{Mod}_{\mO_X}$. Let $\varphi:\mF\to\mG$ be a morphism. Then the following are equivalent. 
\begin{itemize}
\item $\varphi$ is injective (a monomorphism). 
\item For each $U\subseteq X$ open, $\varphi(U)$ is injective. 
\item For each $p\in X$, $\varphi_p$ is injective. 
\end{itemize}
\begin{proof}
$(1)\iff(2)$ is a special case of 2.3.2. $(1)\iff(3)$ is a special case of 2.3.1. 
\end{proof}
\end{prp}

\begin{prp}{}{}\\
Let $X$ be a topological space. Let $\mF,\mG\in\bold{Mod}_{\mO_X}$. Let $\varphi:\mF\to\mG$ be a morphism. Then the following are equivalent. 
\begin{itemize}
\item $\varphi$ is surjective (an epimorphism). 
\item For each $U\subseteq X$ open and for every $g\in\mG(U)$, there exists an open cover $\{U_i\;|\;i\in I\}$ of $U$ and there exists $f_i\in\mF(U_i)$ such that $\varphi(U_i)(f_i)=g|_{U_i}$. 
\item For each $p\in X$, $\varphi_p$ is surjective. 
\item There exists a base $\mB=\{B_i\;|\;i\in I\}$ of $X$ such that $\varphi(B_i)$ is surjective for all $i\in I$. 
\end{itemize}
\end{prp}

\subsection{Graded Sheaves}
\begin{defn}{Graded Sheaves of $\mO_X$-Modules}{}\\
Let $(X,\mO_X)$ be a commutative ringed space. Let $\mF$ be a sheaf of $\mO_X$-modules. We say that $\mF$ is graded if the following are true. 
\begin{itemize}
\item $\mF$ admits a decomposition $$\mF=\bigoplus_{i=-\infty}^\infty \mF_i$$ where each $\mF_i$ is a sheaf of $\mO_X$-modules. 
\item For each open set $U\subseteq X$, we have that $\mF(U)$ admits a grading $$\mF(U)=\bigoplus_{i=-\infty}^\infty\mF_i(U)$$ as a $\mO_X(U)$-module. 
\item For each inclusion $V\subseteq U$ of open sets in $X$, the induced map $\mF(U)\to\mF(V)$ restricts to maps $\mF_i(U)\to\mF_i(V)$ for all $i$. 
\end{itemize}
\end{defn}

\begin{defn}{Graded Sheaves of $\mO_X$-Algebras}{}\\
Let $(X,\mO_X)$ be a commutative ringed space. Let $\mF$ be a sheaf of $\mO_X$-algebras. We say that $\mF$ is graded if the following are true. 
\begin{itemize}
\item $\mF$ is graded as a $\mO_X$-module. 
\item The map $\mF\otimes_{\mO_X}\mF\to\mF$ restricts to maps $\mF_i\otimes_{\mO_X}\mF_j\to\mF_{i+j}$. 
\end{itemize}
\end{defn}

\subsection{Tensor-Hom Adjunction}
\begin{defn}{Tensor Products}{}\\
Let $(X,\mO_X)$ be a commutative ringed space. Let $\mF,\mG\in\bold{Mod}_{\mO_X}$. Define the tensor product $\mF\otimes_{\mO_X}\mG$ of $\mF$ and $\mG$ to be the sheafification of the sheaf defined as follows. 
\begin{itemize}
\item For each open set $U\subseteq X$, define $(\mF\otimes_{\mO_X}\mG)(U)=\mF(U)\otimes_{\mO_X(U)}\mG(U)$. 
\item For each inclusion of open sets $V\subseteq U$, define the morphism $(\mF\otimes_{\mO_X}\mG)(U)\to(\mF\otimes_{\mO_X}\mG)(V)$ by 
\end{itemize}
\end{defn}

Tensor products commute with stalks. 

\begin{defn}{The Sheaf Hom}{}\\
Let $(X,\mO_X)$ be a commutative ringed space. Let $\mF,\mG\in\bold{Mod}_{\mO_X}$. Define the sheaf hom from $\mF$ to $\mG$ to be the sheaf $$\mathscr{H}om(\mF,\mG)$$ defined as follows. 
\begin{itemize}
\item Each open set $U$ is sent to the abelian group $$\mathscr{H}om(\mF,\mG)(U)=\mathscr{H}om_{\bold{Mod}_{\mO_X|_U}}(\mF|_U,\mG|_U)$$
\item For each inclusion of open sets $V\hookrightarrow U$, define the morphism $$\mathscr{H}om_{\bold{Mod}_{\mO_X}}(\mF|_U,\mG|_U)\to\Hom_{\bold{Mod}_{\mO_X}}(\mF,\mG)(V)$$ as follows. A morphism of sheaves $\phi:\mF|_U\to\mG|_U$ is sent to the restriction $\phi|_V:\mF|_V\to\mG|_V$. 
\end{itemize}
\end{defn}

\begin{defn}{The Dual Sheaf}{}\\
Let $(X,\mO_X)$ be a commutative ringed space. Let $\mF$ be a sheaf of $\mO_X$-modules. Define the dual sheaf of $\mF$ to be $$\mF^\vee=\mathscr{H}om(\mF,\mO_X)$$
\end{defn}

\begin{thm}{The Tensor-Hom Adjunction}{}\\
Let $(X,\mO_X)$ be a commutative ringed space. Let $\mF,\mG,\mH\in\bold{Mod}_{\mO_X}$. Then there is a natural isomorphism $$\Hom_{\bold{Mod}_{\mO_X}}(\mF\otimes_{\mO_X}\mG,\mH)\cong\Hom_{\bold{Mod}_{\mO_X}}(\mF,\mathscr{H}om_{\bold{Mod}_{\mO_X}}(\mG,\mH))$$
\end{thm}

\begin{prp}{}{}\\
Let $(X,\mO_X)$ be a commutative ringed space. Let $\mA,\mB\in\bold{Alg}_{\mO_X}$. Then $\mathscr{H}om_{\bold{Alg}_{\mO_X}}(\mA,\mB)\in\bold{Alg}_{\mO_X}$. 
\end{prp}

\begin{prp}{}{}\\
Let $(X,\mO_X)$ be a commutative ringed space. Let $\mA,\mB,\mC\in\bold{Alg}_{\mO_X}$. Then we have $$\mathscr{H}om_{\bold{Alg}_{\mO_X}}(\mA\otimes_{\mO_X}\mB,\mC)\cong\mathscr{H}om_{\bold{Alg}_{\mO_X}}(\mA,\mC)\times\mathscr{H}om_{\bold{Alg}_{\mO_X}}(\mB,\mC)$$
\end{prp}

\subsection{Pushforward and Pullback Sheaves of Modules}
\begin{lmm}{}{}\\
Let $X,Y$ be spaces. Let $f:X\to Y$ be a map. Then inverse and direct image adjunction restricts to an adjunction $$f^{-1}:\bold{Ab}(Y)\overset{\dashv}{\rightleftarrows}\bold{Ab}(X):f_\ast$$
\end{lmm}

However for modules in general, we need to modify the inverse image functor to incorporate the action of the ring. 

\begin{defn}{Inverse Image Sheaf}{}\\
Let $(X,\mO_X)$ and $(Y,\mO_Y)$ be commutative ringed spaces. Let $f:X\to Y$ be a morphism. Let $\mG$ be a sheaf on $Y$. Define the inverse image sheaf of $\mG$ by $$f^\ast\mG=\mO_X\otimes_{f^{-1}\mO_Y}f^{-1}\mG$$
\end{defn}

Since tensor products are functorial, the inverse image sheaf is also a functorial construction. Also, this is basically a change of rings formula. In particular, the underlying sheaf of abelian groups of $f^{-1}\mG$ and $f^\ast\mG$ are the same. 

\begin{lmm}{}{}\\
Let $X,Y$ be spaces. Let $f:X\to Y$ be a map. Then there is an adjunction $$f^\ast:\bold{Mod}_{\mO_Y}\overset{\dashv}{\rightleftarrows}\bold{Mod}_{\mO_X}:f_\ast$$
\end{lmm}

\begin{lmm}{}{}\\
Let $X,Y$ be spaces. Let $f:X\to Y$ be a map. Then inverse and direct image adjunction restricts to an adjunction $$f^\ast:\bold{Alg}_{\mO_Y}\overset{\dashv}{\rightleftarrows}\bold{Alg}_{\mO_X}:f_\ast$$
\end{lmm}

From operad theory a lot of constructions are carried over eg symmetric products, exterior products. 

\begin{defn}{$k$th Tensor Power}{}\\
Let $(X,\mO_X)$ be a commutative ringed space. Let $\mF$ be a sheaf of $\mO_X$-modules. Define the $k$th tensor power of $\mF$ to be sheafification of the presheaf $$U\mapsto T^k(\mF(U))$$

\end{lmm}

\end{defn}

\begin{defn}{$k$th Symmetric Power}{}\\
Let $(X,\mO_X)$ be a commutative ringed space. Let $\mF$ be a sheaf of $\mO_X$-modules. Define the $k$th symmetric power of $\mF$ to be sheafification of the presheaf $$U\mapsto S^k(\mF(U))$$
\end{defn}

Let $(X,\mO_X)$ be a commutative ringed space. Let $\mF$ be a sheaf of $\mO_X$-modules. If $\mF$ is locally free of rank $n$, then $T^k(\mF)$ is locally free of rank $\binom{n+k-1}{n-1}$. 
\end{lmm}

\begin{defn}{Symmetric Algebra}{}\\

\end{defn}

\begin{defn}{$k$th Exterior Power}{}\\
Let $(X,\mO_X)$ be a commutative ringed space. Let $\mF$ be a sheaf of $\mO_X$-modules. Define the $k$th exterior power of $\mF$ to be sheafification of the presheaf $$U\mapsto\Lambda^k(\mF(U))$$
\end{defn}

\begin{lmm}{}{}\\
Let $(X,\mO_X)$ be a commutative ringed space. Let $\mF$ be a sheaf of $\mO_X$-modules. If $\mF$ is locally free of rank $n$, then the following are true.

\begin{defn}{Exterior Algebra}{}\\
Let $(X,\mO_X)$ be a commutative ringed space. Let $\mF$ be a sheaf of $\mO_X$-modules. Define the exterior algebra of $\mF$ to be sheafification of the presheaf $$U\mapsto\Lambda(\mF(U))$$
\end{defn}

\section{Different Types of Sheaves of Modules}
\subsection{Some Finiteness Properties}
\begin{defn}{Sheaf of Finite Type}{}\\
Let $(X,\mO_X)$ be a commutative ringed space. Let $\mF\in\bold{Mod}_{\mO_X}$. We say that $\mF$ is a sheaf of finite type if there exists an open cover $\{U_i\;|\;i\in I\}$ of $X$ such that there is a surjective morphism $$\mO_X^{\otimes n_i}|_{U_i}\to\mF|_{U_i}$$ for some $n_i\in\N$. 
\end{defn}

\begin{prp}{}{}\\
Let $(X,\mO_X)$ be a commutative ringed space. Let $\mF\in\bold{Mod}_{\mO_X}$ be a sheaf of finite type. Then $\mF_{X,p}$ is a finitely generated $\mO_{X,p}$-module for all $p\in X$. 
\end{prp}

\begin{defn}{Finitely Presented Sheaves}{}\\
Let $(X,\mO_X)$ be a commutative ringed space. Let $\mF\in\bold{Mod}_{\mO_X}$. We say that $\mF$ is finitely presented if there exists an open cover $\{U_i\;|\;i\in I\}$ of $X$ such that there is an exact sequence \\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	{\mO_X^{m_i}|_U} & {\mO_X^{n_i}|_U} & {\mF|_U} & 0
	\arrow[from=1-1, to=1-2]
	\arrow[from=1-2, to=1-3]
	\arrow[from=1-3, to=1-4]
\end{tikzcd}}\\
for some $m_i,n_i\in\N$. 
\end{defn}

\subsection{Quasi-Coherent Sheaves}
Quasi-coherent sheaves and coherent sheaves has extremely abstract definitions. If one is only interested in the case of Algebraic Geometry, then there is a very nice explicit way of defining it, so readers may skip this part. 

\begin{defn}{Quasi-Coherent Sheaves}{}\\
Let $(X,\mO_X)$ be a commutative ringed space. Let $\mF$ be a sheaf of $\mO_X$-module. We say that $\mF$ is a quasi-coherent sheaf if for all $p\in X$, there exists an open neighbourhood $U\subseteq X$ such that there is an exact sequence: \\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	{\mO_X^{\otimes I}|_U} & {\mO_X^{\otimes J}|_U} & {\mF|_U} & 0
	\arrow[from=1-1, to=1-2]
	\arrow[from=1-2, to=1-3]
	\arrow[from=1-3, to=1-4]
\end{tikzcd}}\\
for some countable indexing sets $I$ and $J$. 
\end{defn}

\begin{defn}{The Category of Quasi-coherent Sheaves}{}\\
Let $(X,\mO_X)$ be a commutative ringed space. Define the category of coherent sheaves $\bold{QCoh}_{\mO_X}$ to be the full subcategory of $\bold{Mod}_{\mO_X}$ consisting of quasi-coherent sheaves. 
\end{defn}

\begin{defn}{Coherent Sheaves}{} Let $(X,\mO_X)$ be a commutative ringed space. Let $\mF$ be a sheaf of $\mO_X$-module. We say that $\mF$ is coherent if the following are true. 
\begin{itemize}
\item $\mF$ is a sheaf of finite type. 
\item For any $U\subseteq X$ and any morphism $$\varphi:\mO_X^{\otimes n}|_U\to\mF|_U$$ of $\mO_X$-modules, then kernel of $\varphi$ is a sheaf of finite type. 
\end{itemize}
\end{defn}

\begin{defn}{The Category of Coherent Sheaves}{}\\
Let $(X,\mO_X)$ be a commutative ringed space. Define the category of coherent sheaves $\bold{Coh}_{\mO_X}$ to be the full subcategory of $\bold{QCoh}_{\mO_X}$ consisting of coherent sheaves. 
\end{defn}

\begin{prp}{}{}\\
Let $(X,\mO_X)$ be a commutative ringed space. Let the following \\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	0 & \mF & {\mF'} & {\mF''} & 0
	\arrow[from=1-1, to=1-2]
	\arrow[from=1-2, to=1-3]
	\arrow[from=1-3, to=1-4]
	\arrow[from=1-4, to=1-5]
\end{tikzcd}}\\
be an exact sequence of sheaves. If two of the three sheaves are coherent, then so is the last one. 
\end{prp}

\begin{prp}{}{}\\
Let $(X,\mO_X)$ be a commutative ringed space. Let $\mF\in\bold{Mod}_{\mO_X}$. Then the following are true. 
\begin{itemize}
\item If $\mF$ is coherent, then $\mF$ is finitely presented. 
\item If $\mF$ is finitely presented, then $\mF$ is of finite type. 
\item If $\mF$ is finitely presented, then $\mF$ is quasi-coherent. 
\end{itemize}
\end{prp}

\begin{prp}{}{}\\
Let $(X,\mO_X)$ be a commutative ringed space. Let $\mF\in\bold{Mod}_{\mO_X}$. If $\mO_X$ is coherent, then the following are equivalent. 
\begin{itemize}
\item $\mF$ is coherent. 
\item $\mF$ is finitely presented. 
\item $\mF$ is of finite type. 
\end{itemize}
\end{prp}

\subsection{Reflexive Sheaves}
\begin{defn}{Reflexive Sheaves}{}\\
Let $(X,\mO_X)$ be a commutative ringed space. Let $\mF$ be a sheaf of $\mO_X$-modules. We say that $\mF$ is reflexive if there is an isomorphism of $\mO_X$-modules $$\mF^{\vee\vee}\cong\mF$$
\end{defn}

\subsection{Locally Free Sheaves}
\begin{defn}{Free Sheaves}{}\\
Let $(X,\mO_X)$ be a commutative ringed space. Let $\mF$ be a $\mO_X$-module. We say that $\mF$ is free is there is an isomorphism of $\mO_X$-modules $$\mF\cong\mO_X^{\oplus n}$$
\end{defn}

\begin{defn}{Locally Free Sheaves}{}\\
Let $(X,\mO_X)$ be a commutative ringed space. Let $\mF$ be a sheaf of $\mO_X$-modules. We say that $\mF$ is locally free if $X$ has an open cover $\{U_i\;|\;i\in I\}$ such that for each $U_i$, there is an isomorphism of $\mO_X$-modules $$\mF|_{U_i}\cong\left(\mO_X|_{U_i}\right)^{\oplus n_i}$$
\end{defn}

\begin{lmm}{}{}\\
Let $(X,\mO_X)$ be a commutative ringed space. Let $\mF$ be a locally free sheaf. Then $\mF$ is reflexive. 
\end{lmm}

\begin{defn}{The Rank of a Locally Free Sheaf}{}\\
Let $(X,\mO_X)$ be a commutative ringed space. Let $\mF$ be a locally free sheaf of $\mO_X$-modules. We say that the rank of $\mF$ is $n\in\N$ if there exists an open cover $\{U_i\;|\;i\in I\}$ of $X$ such that there are isomorphisms of $\mO_X$-modules $$\mF|_{U_i}\cong\left(\mO_X|_{U_i}\right)^{\oplus n}$$
\end{defn}

\begin{lmm}{}{}\\
Let $(X,\mO_X)$ be a commutative ringed space. Let $\mF$ be a locally free sheaf of $\mO_X$-modules. If $X$ is connected, then the rank of $\mF$ is well defined. 
\end{lmm}

\begin{prp}{}{} Let $(X,\mO_X)$ be a commutative ringed space. Let $\mF$ be a sheaf of $\mO_X$-modules. Then $\mF$ is locally free of rank $r$ if and only if $$\mF_p\cong\mO_{X,p}^r$$ for all $p\in X$. 
\end{prp}

\begin{prp}{}{}\\
Let $(X,\mO_X)$ be a commutative ringed space. Let $\mF$ be a sheaf of $\mO_X$-modules. If $\mF$ is locally free, then $\mF$ is reflexive. 
\end{prp}

\begin{defn}{The Category of Locally Free Sheaves}{}\\
Let $(X,\mO_X)$ be a commutative ringed space. Define the category of locally free sheaves $$\bold{Loc}_X$$ of rank $n$ over $X$ as follows. 
\begin{itemize}
\item The objects are the locally free sheaves over $X$
\item The morphisms are the morphisms of sheaves of modules
\item Composition is given by the composition of morphisms of sheaves. 
\end{itemize}
\end{defn}

\subsection{Invertible Sheaves}
\begin{defn}{Invertible Sheaf}{}\\
Let $(X,\mO_X)$ be a commutative ringed space. Let $\mF$ be a sheaf of $\mO_X$-modules. We say that $\mF$ is an invertible sheaf if $\mF$ is a locally free sheaf of rank $1$. 
\end{defn}

\begin{prp}{}{}\\
Let $(X,\mathcal{O}_X)$ be a commutative ringed space. Then the following are equivalent characterization of a sheaf of $\mathcal{O}_X$-modules $\mF$
\begin{itemize}
\item $\mF$ is invertible
\item There exists a sheaf $G$ such that $F\otimes_{\mathcal{O}_X}G\cong\mathcal{O}_X$
\item $\mF\otimes_{\mathcal{O}_X}\mF^{\vee}\cong\mathcal{O}_X$
\end{itemize}
\end{prp}

\begin{prp}{}{}\\
Let $(X,\mO_X)$ be a commutative ringed space. Let $\mF$ and $\mG$ be invertible sheaves on $X$. Then $\mF\otimes\mG$ is an invertible sheaf on $X$. 
\end{prp}

\begin{defn}{The Picard Group}{} Let $(X,\mO_X)$ be a commutative ringed space. Define the Picard group of $X$ to be the set $$\text{Pic}(X)=\{\mL\;|\;\mL\text{ is an invertible sheaf on }X\}$$ together with the tensor product $\otimes$. 
\end{defn}

\section{Sheaf Cohomology}
\subsection{Basic Definitions}
\begin{prp}{}{}\\
Let $(X,\mO_X)$ be a commutative ringed space. Then $\bold{Mod}_{\mO_X}$ has enough injectives. 
\end{prp}

\begin{defn}{Global Section Functor}{}\\
Let $(X,\mO_X)$ be a commutative ringed space. Define the global section functor $\Gamma_{\mO_X}(X,-):\bold{Mod}_{\mO_X}\to\bold{Mod}_{\mO_X(X)}$ by the following data. 
\begin{itemize}
\item For $\mF\in\bold{Mod}_{\mO_X}$, we have $\Gamma_{\mO_X}(X,\mF)=\mF(X)$. 
\item For a morphism of sheaves $\phi:\mF\to\mG$, $\Gamma(X,-)$ sends it to the group homomorphism $\phi(X):\mF(X)\to\mG(X)$
\end{itemize}
\end{defn}

In general, we can define functor by sending every sheaf to a the local section of a chosen open set. This operation in general is left exact. This means that we have the following proposition. 

\begin{crl}{}{}\\
Let $(X,\mO_X)$ be a commutative ringed space. The global section functor $\Gamma_{\mO_X}(X,-):\bold{Mod}_{\mO_X}\to\bold{Mod}_{\mO_X(X)}$ is a left exact functor. 
\begin{proof}
The global section functor is a special case of the above proposition. 
\end{proof}
\end{crl}

\begin{defn}{Cohomology Functors}{}\\
Let $X$ be a space. Let $\mF\in\bold{Mod}_{\mO_X(X)}$. Define the sheaf cohomology of $\mF$ to be $$H^i(X,\mF)=R^i\Gamma_{\mO_X}(X,-)(\mF)$$
\end{defn}

\subsection{Flasque Sheaves}
\begin{defn}{Flasque Sheaves}{}\\
Let $X$ be a space. Let $\mF\in\bold{Ab}(X)$. We say that $\mF$ is flasque if for any open sets $V\subseteq U\subseteq X$, we have that $\mF(U)\to\mF(V)$ is surjective. 
\end{defn}

\begin{lmm}{}{}\\
Let $(X,\mO_X)$ be a commutative ringed space. Let $\mF\in\bold{Mod}_{\mO_X}$. Then $\mF$ is flasque. 
\end{lmm}

\begin{lmm}{}{}\\
Let $X$ be a space and let $\mF,\mF',\mF''$ be sheaves on $X$ such that there is an exact sequence \\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	0 & \mF & \mF' & \mF'' & 0
	\arrow[from=1-1, to=1-2]
	\arrow[from=1-2, to=1-3]
	\arrow[from=1-3, to=1-4]
	\arrow[from=1-4, to=1-5]
\end{tikzcd}}\\
of sheaves. Then the following are true. 
\begin{itemize}
\item If $\mF$ and $\mF'$ are flasque, then $\mF''$ is flasque. 
\item If $\mF$ is flasque, then for any $U\subseteq X$ open, there is an exact sequence \\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	0 & \mF(U) & \mF'(U) & \mF''(U) & 0
	\arrow[from=1-1, to=1-2]
	\arrow["\phi", from=1-2, to=1-3]
	\arrow["\psi", from=1-3, to=1-4]
	\arrow[from=1-4, to=1-5]
\end{tikzcd}}
\end{itemize}
\begin{proof}\\
Suppose that $\mF$ and $\mF'$ are flasque. Then by naturality we obtain a commutative diagram \\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	0 & {\mF(U)} & {\mF'(U)} & {\mF''(U)} & 0 \\
	0 & {\mF(V)} & {\mF'(V)} & {\mF''(V)} & 0
	\arrow[from=1-1, to=1-2]
	\arrow["\phi(U)", from=1-2, to=1-3]
	\arrow[two heads, from=1-2, to=2-2]
	\arrow["\psi(U)", from=1-3, to=1-4]
	\arrow[two heads, from=1-3, to=2-3]
	\arrow[from=1-4, to=1-5]
	\arrow[from=1-4, to=2-4]
	\arrow[from=2-1, to=2-2]
	\arrow["\phi(V)", from=2-2, to=2-3]
	\arrow["\psi(V)", from=2-3, to=2-4]
	\arrow[from=2-4, to=2-5]
\end{tikzcd}}\\
for any $V\subseteq U$ open sets. Let $s\in\mF''(V)$. Then by surjectivity of $\psi(V)$ and $\mF'(U)\to\mF'(V)$, there exists $t\in\mF'(U)$ that maps correspondingly by the same maps to $s$. Then using the map $\phi(U)$ we can send $s$ to an element $r\in\mF''(U)$ such that $r$ is sent to $s$ under the map $\mF''(U)\to\mF''(V)$. Thus this map is surjective and $\mF''$ is flasque. \\

Now suppose that $\mF$ is flasque. 
\end{proof}
\end{lmm}

\begin{prp}{}{}\\
Let $X$ be a space and let $\mF$ be a sheaf on $X$. If $\mF$ is flasque then $\mF$ is acyclic for $\Gamma(X,-)$. Explicitly, this means that $$H^i(X,\mF)=0$$ for all $i>0$. 
\begin{proof}\\
Since $\bold{Ab}(X)$ has enough injective objects, suppose that $\mI$ is an injective object of $\bold{Ab}(X)$ such that $\mF$ is a subobject of $\mI$. Let $\mG$ be the quotient sheaf $\mF/\mI$. Then there is an exact sequence \\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	0 & \mF & \mI & \mG & 0
	\arrow[from=1-1, to=1-2]
	\arrow[from=1-2, to=1-3]
	\arrow[from=1-3, to=1-4]
	\arrow[from=1-4, to=1-5]
\end{tikzcd}}\\
By lemma 3.2.2, $\mI$ is flasque. By 3.2.3, $\mG$ is flasque. By 3.2.3, there is an exact sequence \\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	0 & \mF(X) & \mI(X) & \mG(X) & 0
	\arrow[from=1-1, to=1-2]
	\arrow[from=1-2, to=1-3]
	\arrow[from=1-3, to=1-4]
	\arrow[from=1-4, to=1-5]
\end{tikzcd}}\\
Since $\mI$ is injective, we have that $H^i(X,\mI)=0$ for all $i>0$. By passing to the long exact sequence in cohomology, we deduce that $H^1(X,\mF)=0$ and $H^i(X,\mF)=H^{i-1}(X,\mG)$ for each $i\geq 2$. Applying the same method again gives $H^1(X,\mG)=0$. Thus by induction, we conclude that $H^i(X,\mF)=0$ for all $i>0$. 
\end{proof}
\end{prp}

Acyclic sheaves are useful because they help compute sheaf cohomology. Recall from Homological Algebra that $$H^i(X;\mF)=H^i(\Gamma(X,\mI^\bullet))$$ if $\mI^\bullet$ is an acyclic resolution of $\mF$. 

\subsection{Godement Resolution}
\begin{defn}{The Godement Sheaf}{}\\
Let $X$ be a space. Let $\mF\in\bold{Ab}(X)$ be a sheaf on $X$. Define the Godement sheaf $\text{Gd}(F):\bold{Open}(X)\to\bold{Ab}$ as follows. 
\begin{itemize}
\item For each open set $U\subseteq X$, define $$\text{Gd}(F)(U)=\prod_{x\in U}\mF_x$$
\item For each inclusion $V\subseteq U$, define the induced map $\text{Gd}(F)(U)\to\text{Gd}(F)(V)$ to be the restriction map. 
\end{itemize}
\end{defn}

\begin{lmm}{}{}\\
Let $X$ be a space. Let $\mF\in\bold{Ab}(X)$ be a sheaf on $X$. Then $\text{Gd}(F)$ is a flasque sheaf. 
\end{lmm}

\begin{defn}{Canonical Morphism}{}\\
Let $X$ be a space. Let $\mF\in\bold{Ab}(X)$ be a sheaf on $X$. Define the canonical morphism $\alpha_F:F\to\text{Gd}(\mF)$ by $\alpha_F:F(U)\to\text{Gd}(\mF)$ the formula $$s\mapsto(s_x)_{x\in U}$$
\end{defn}

\begin{defn}{The Godement Functor}{}\\
Let $X$ be a space. Define the Godement functor $\text{Gd}:\bold{Ab}(X)\to\bold{Ab}(X)$ as follows. 
\begin{itemize}
\item Each sheaf $\mF\in\bold{Ab}(X)$ is sent to the godement sheaf $\text{Gd}(\mF)$. 
\item Each morphism $\mF\to\mG$ is sent to the morphism of sheaves defined by the product of the induced map on stalks. 
\end{itemize}
\end{defn}

\begin{prp}{}{}\\
Let $X$ be a space. Let $\mF\in\bold{Ab}(X)$ be a sheaf. Then $\text{Gd}(\mF)$ is flasque. 
\end{prp}

\begin{defn}{Godement Resolution}{}\\
Let $X$ be a space. Let $\mF\in\bold{Ab}(X)$ be a sheaf. Define the Godement resolution $\text{Gd}(\mF)^\bullet$ of $\mF$ as follows. 
\begin{itemize}
\item Define $\text{Gd}(\mF)^0=\mF$. 
\item For each $n\in\N^+$, define $\text{Gd}(\mF)^n=\text{Gd}(\text{coker}(\alpha_{\text{Gd}(\mF)^{n-1}}))$. 
\item Define the morphisms $\text{Gd}(\mF)^n\to\text{Gd}(\mF)^{n+1}$ to be $\alpha_{\text{Gd}(\mF)^n}$. \end{itemize}
\end{defn}

\begin{crl}{}{}\\
Let $X$ be a space. Let $\mF\in\bold{Ab}(X)$ be a sheaf. Then $\text{Gd}(\mF)^\bullet$ is an acyclic resolution for $\Gamma(X,-)$. 
\end{crl}

Consequently, the sheaf cohomology of $\mF$ acquires the formula $$H^i(X;\mF)=H^i(\Gamma(X,\text{Gd}(\mF)^\bullet)$$

\subsection{\v{C}ech Cohomology}
\begin{defn}{\v{C}ech Complex}{}\\
Let $X$ be a space. Let $\mU=\{U_i\;|\;i\in I\}$ be an open cover of $X$ where $I$ is an indexing set. Let $\mF\in\bold{Ab}(X)$. Define the \v{C}ech complex $$(C^\bullet(X,\mU,\mF),d)\in\bold{Ch}(\bold{Ab})$$ of $\mF$ with respect to $\mU$ as follows. 
\begin{itemize}
\item The terms of the chain complex is given by $$C^k(X,\mU,\mF)=\prod_{i_0<\cdots<i_k}\mF(U_{i_0}\cap U_{i_1}\cap\dots\cap U_{i_k})$$
\item Define the boundary map $d:C^k(X,\mU,\mF)\to C^{k+1}(X,\mU,\mF)$ as follows. For $s\in C^k(X,\mU,\mF)$ whose components are indicated by $s_{i_0,\dots,i_k}$, the components of $d(s)$ is given by $$(d(s))_{i_0,\dots,i_{k+1}}=\sum_{j=0}^{k+1}(-1)^js_{i_0,\dots,i_{j-1},\widehat{i_j},i_{j+1},\dots,i_{k+1}}|_{U_{i_0,\dots,i_{k+1}}}$$
\end{itemize}
\end{defn}

We use the notation $U_{i_0,\dots,i_k}=U_{i_0}\cap U_{i_1}\cap\dots\cap U_{i_k}$. 

\begin{lmm}{}{} For any space $X$ and any open cover $\mU$ of $X$, $(C^\bullet(X,\mU,\mF),d)$ is indeed a chain complex. 
\end{lmm}

\begin{defn}{\v{C}ech Cohomology}{} Let $(C^\bullet(X,\mU,\mF),d)$ be a \v{C}ech complex. Define the $k$th cohomology group of it to be $$\text{\v{H}}^k(X,\mU,\mF)=\frac{\ker(C^k(X,\mU,\mF)\to C^{k+1}(X,\mU,\mF))}{\im(C^{k-1}(X,\mU,\mF)\to C^k(X,\mU,\mF))}=H(C^\bullet(X,\mU,\mF),d)$$
\end{defn}

\begin{lmm}{}{}\\
Let $(X,\mO_X)$ be a commutative ringed space. Let $\mU$ be an open cover of $X$. Then we have $\text{\v{H}}^0(X,\mU,\mF)=\mF(X)$. 
\end{lmm}

\begin{defn}{The Sheafified \v{C}ech Complex}{}\\
Let $X$ be a space. Let $\mU$ be an open cover of $X$. Let $\mF$ be a sheaf on $X$. Define the sheafified \v{C}ech complex $\mC^\bullet(X,\mU,\mF)$ by the following. 
\begin{itemize}
\item For each $p\in\N$, define $$\mC^p(X,\mU,\mF)=\prod_{i_0<\cdots<i_p}\iota_{U_{i_0,\dots,i_p}}(U_{i_0,\dots,i_p})\in\bold{Ab}(X)$$ where $\iota_U:U\hookrightarrow X$ denotes the inclusion map for an open subset $U\subseteq X$. 
\item Define the boundary map $d:\mC^p(X,\mU,\mF)\to\mC^{p+1}(X,\mU,\mF)$ to be the sheafified version of the boundary map in the \v{C}ech complex. 
\end{itemize}
\end{defn}

Note that by construction, we have $\Gamma(X,\mC^p(X,\mU,\mF))=C^p(X,\mU,\mF)$. 

\begin{lmm}{}{}\\
Let $X$ be a space. Let $\mU$ be an open cover of $X$. Let $\mF$ be a sheaf on $X$. Then the following are true. 
\begin{itemize}
\item $\mC^\bullet(X,\mU,\mF)$ is a resolution of $\mF$. 
\item If $\mF$ is flasque, then $\mC^\bullet(X,\mU,\mF)$ is a flasque (and hence acyclic) resolution of $\mF$. 
\end{itemize}
\end{lmm}

Note that the above lemma implies that $\text{\v{H}}^i(X,\mU,\mF)=H^i(X,\mF)=0$ if $\mF$ is flasque. 

\begin{lmm}{}{}\\
Let $X$ be a space. Let $\mU$ be an open cover of $X$. Let $\mF$ be a sheaf on $X$. Then there exists a map $$\text{\v{H}}^p(X,\mU,\mF)\to H^p(X,\mF)$$ that is natural in $\mF$ for all $p\in\N$. 
\end{lmm}

\begin{lmm}{Leray}{}\\
Let $X$ be a space. Let $\mU$ be an open cover of $X$. Let $\mF$ be a sheaf on $X$. Suppose that for any $U_{i_0,\dots,i_p}$, we have $H^k(U_{i_0,\dots,i_p},\mF_{U_{i_0,\dots,i_p}})=0$ for all $k,p\in\N$. Then the above natural maps are isomorphisms $$\text{\v{H}}^p(X,\mU,\mF)\cong H^p(X,\mF)$$
\end{lmm}

\begin{prp}{}{}\\
Let $X$ be a space. Let $\mF$ be a sheaf on $X$. Then there is a natural isomorphism $$\varinjlim_{\mU}\text{\v{H}}^1(X,\mU,\mF)\cong H^1(X,\mF)$$
\end{prp}

\begin{prp}{}{}\\
Let $(X,\mO_X)$ be a commutative ringed space. Then there is a natural isomorphism $$\text{Pic}(X)\cong H^1(X,\mO_X^\ast)$$
\end{prp}

\subsection{Relation to Singular Cohomology}
\begin{thm}{(Y. Sella 2016)}{}\\
Let $X$ be a locally contractible space. Then there is an isomorphism $$H_\text{sing}^n(X;\Z)\cong H_\text{sheaf}^n(X;\underline{\Z})$$
\end{thm}

\subsection{Hypercohomology}
\begin{defn}{Hypercohomology of a Space}{}\\
Let $X$ be a space. Let $\mF^\bullet$ be a cochain complex of sheaves on $X$. Define the hyper cohomology of $X$ to be the right hyper-derived functor of the global section functor: $$\H^n(X;\mF^\bullet)=\R^n\Gamma(\mF^\bullet)$$
\end{defn}

\begin{prp}{}{}\\
Let $X$ be a space. Let $\mF^\bullet\in\bold{CCh}^+(\bold{Ab}(X))$ be a bounded below cochain complex. Let $\mI^\bullet\in\bold{CCh}^+(\bold{Ab}(X))$ be a bounded below cochain complex such that the following are true. 
\begin{itemize}
\item $\mI^n$ is acyclic for all $n$. 
\item There is a quasi-isomorphism $F^\bullet\to\mI^\bullet$. 
\end{itemize}
Then we have $$\H^n(X;\mF^\bullet)=H^i(\Gamma(X,\mI^\bullet))$$
\end{prp}

\begin{prp}{}{}\\
Let $X$ be a space. Let $\mF^\bullet\in\bold{CCh}^+(\bold{Ab}(X))$ be a cochain complex of sheaves on $X$. Then there is a natural isomorphism $$\H^n(X;\mF^\bullet)\cong H^n(\Gamma(X,\text{Tot}^{\prod}(\text{Gd}^\bullet,\mF^\bullet)))$$
\end{prp}

\end{prp}

\subsection{Ext for Sheaves of Modules}
Let $(X,\mO_X)$ be a commutative ringed space. Let $\mF\in\bold{Mod}_{\mO_X}$. Then we have a functor $\mathscr{H}om_{\mO_X}(\mF,-):\bold{Mod}_{\mO_X}\to\bold{Mod}_{\mO_X}$ between abelian categories. 

\begin{defn}{Ext Functor for Sheaves}{}\\
Let $(X,\mO_X)$ be a commutative ringed space. Let $\mF,\mG\in\bold{Mod}_{\mO_X}$. Define the $i$th ext sheaf of $\mF$ and $\mG$ to be $$\mathscr{E}\text{xt}_{\mO_X}^i(\mF,\mG)=R^i\mathscr{H}\text{om}_{\mO_X}(\mF,-)(\mG)$$
\end{defn}

\begin{lmm}{}{}\\
Let $(X,\mO_X)$ be a commutative ringed space. Let $\mI\in\bold{Mod}_{\mO_X}$ be an injective sheaf. Let $U\subseteq X$ be an open subset. Then $\mI|_U$ is injective in $\bold{Mod}_{\mO_X|_U}$. 
\end{lmm}

\begin{prp}{}{}\\
Let $(X,\mO_X)$ be a commutative ringed space. Let $\mF,\mG\in\bold{Mod}_{\mO_X}$. Let $U\subseteq X$ be open. Then we have $$\mathscr{E}\text{xt}_{\mO_X}^i(\mF,\mG)|_{\mO_X|_U}\cong\mathscr{E}\text{xt}_U^i(\mF|_U,\mG|_U)$$
\end{prp}

\begin{prp}{}{}\\
Let $(X,\mO_X)$ be a commutative ringed space. Let $\mF\in\bold{Mod}_{\mO_X}$. Then we have $$\mathscr{E}\text{xt}_{\mO_X}^i(\mO_X,\mF)=\begin{cases}
\mF & \text{ if }i=0\\
0 & \text{ otherwise }
\end{cases}$$
\end{prp}

\begin{prp}{}{}\\
Let $(X,\mO_X)$ be a commutative ringed space. Let $\mF\in\bold{Mod}_{\mO_X}$. Then we have $$\text{Ext}_{\mO_X}^i(\mO_X,\mF)\cong H^i(X,\mF)$$
\end{prp}

\begin{prp}{}{}\\
Let $(X,\mO_X)$ be a commutative ringed space. Let $\mF,\mG\in\bold{Mod}_{\mO_X}$. Let $\mL_\bullet$ be a resolution of $\mF$ such that each $\mL_i$ is locally free of finite rank. Then we have $$\mathscr{E}\text{xt}_{\mO_X}^i(\mF,\mG)\cong H^i(\mathscr{H}\text{om}_{\mO_X}(\mL_\bullet,\mG))$$
\end{prp}

\begin{prp}{}{}\\
Let $(X,\mO_X)$ be a commutative ringed space. Let $\mF,\mG,\mL\in\bold{Mod}_{\mO_X}$. Suppose that $\mL$ is locally free of finite rank. Then the following are true. 
\begin{itemize}
\item There is an isomorphism $$\text{Ext}_{\mO_X}^i(\mF\otimes_{\mO_X}\mL,\mG)\cong\text{Ext}_{\mO_X}^i(\mF,\mL^\vee\otimes_{\mO_X}\mG)$$
\item There is an isomorphism $$\mathscr{E}\text{xt}_{\mO_X}^i(\mF\otimes_{\mO_X}\mL,\mG)\cong\mathscr{E}\text{xt}_{\mO_X}^i(\mF,\mL^\vee\otimes_{\mO_X}\mG)\cong\mathscr{E}\text{xt}_{\mO_X}^i(\mF,\mG)\otimes_{\mO_X}\mL^\vee$$
\end{itemize}
\end{prp}
