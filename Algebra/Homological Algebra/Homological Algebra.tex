\section{Abelian Categories and Exact Sequences}
\subsection{Additive Categories}
\begin{defn}{Pre-Additive Categories}{}\\
Let $\mC$ be a category. We say that $\mC$ is pre-additive if the following is true. 
\begin{itemize}
\item For any $X,Y\in\mC$, the Hom set $$\Hom_\mC(X,Y)\in\bold{Ab}$$ has the structure of an abelian group. 
\item For any $X,Y,Z\in\mC$, the composition of morphisms $$\circ:\Hom_\mC(X,Y)\times\Hom_\mC(Y,Z)\to\Hom_\mC(X,Z)$$ is bilinear. This means that if $f,g\in\Hom_\mC(X,Y)$ and $h,k\in\Hom_\mC(Y,Z)$ are morphisms, then $$h\circ(f+g)=h\circ f+h\circ g\;\;\;\;\text{ and }\;\;\;\;(h+k)\circ f=h\circ f+k\circ f$$
\end{itemize}
\end{defn}

We also say that $\mC$ is enriched over $\bold{Ab}$. This relates to enriched category theory. 

\begin{defn}{Additive Categories}{} A category $\mA$ is additive if in addition to being pre-additive, it also satisfies the following: 
\begin{itemize}
\item $\mA$ has a zero object, denoted $0$
\item $\mA$ has finite products
\end{itemize}
\end{defn}

\subsection{Abelian Categories}
\begin{defn}{Zero Morphism}{}\\
Let $\mA$ be a category with a zero object. Let $C,D\in\mA$ be objects. Define the zero morphism from $C$ to $D$ to be the the unique morphism $$0:C\overset{\exists!}{\to}0\overset{\exists!}{\to}D$$
\end{defn}

\begin{defn}{Kernels and Cokernels}{}\\
Let $\mA$ be a category with a zero object. Let $C,D\in\mA$ be objects. Let $f:C\to D$ be a morphism. 
\begin{itemize}
\item Define the kernel of $f$ to be $$\ker(f)=\text{Eq}(f,0)$$ if it exists. 
\item Define the cokernel of $f$ to be $$\coker(f)=\text{Coeq}(f,0)$$ if it exists. 
\end{itemize}
\end{defn}

\begin{defn}{Pre-Abelian Category}{}\\
Let $\mA$ be an additive category. We say that $\mA$ is a pre-abelian category if every morphisms in $\mA$ has a kernel and a cokernel. 
\end{defn}

\begin{lmm}{}{}\\
Let $\mA$ be pre-abelian category. Let $f:C\to D$ be a morphism. Then $f$ factors naturally and uniquely as $$C\to\text{coker}(\ker(f)\to C)\to\ker(D\to\text{coker}(f))\to D$$
\begin{proof}
Since $\ker(f)\to C\to D$ is $0$, we invoke the universal property of cokernels to obtain the following commutative diagram: \\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	&& {\text{coker}(\ker(f)\to C)} \\
	{\ker(f)} & C & D & {\text{coker}(f)}
	\arrow["{\exists!}", dashed, from=1-3, to=2-3]
	\arrow[from=2-1, to=2-2]
	\arrow[from=2-2, to=1-3]
	\arrow[from=2-2, to=2-3]
	\arrow[from=2-3, to=2-4]
\end{tikzcd}} \\~\\
Now since $\mA$ is an abelian category, any epimorphism $h$ satisfies $a\circ h=0$ implies $a=0$. Since $C\to\text{coker}(\ker(f)\to C)$ is an epimorphism by 1.2.7 and $C\to\text{coker}(\ker(f)\to C)\to D\to\text{coker}(f)=0$, we conclude that $$\text{coker}(\ker(f)\to C)\to D\to\text{coker}(f)=0$$ Then by the universal property of kernels, we obtain the following commutative diagram: \\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	{\text{coker}(\ker(f)\to C)} \\
	{\ker(D\to\text{coker}(f))} & D & {\text{coker}(f)}
	\arrow["{\exists!}"', dashed, from=1-1, to=2-1]
	\arrow[from=1-1, to=2-2]
	\arrow[from=2-1, to=2-2]
	\arrow[from=2-2, to=2-3]
\end{tikzcd}} \\
It is natural since all morphisms are induced by a universal property. 
\end{proof}
\end{lmm}

\begin{defn}{Abelian Categories}{}\\
Let $\mA$ be a pre-abelian category. We say that $\mA$ is an abelian category if the following are true. 
\begin{itemize}
\item Every monomorphism is the kernel of some morphism. 
\item Every epimorphism is the cokernel of some morphism. 
\end{itemize}
\end{defn}

\begin{lmm}{}{}\\
Let $\mA$ be a pre-abelian category. Then the following are equivalent. 
\begin{itemize}
\item $\mA$ is an abelian category. 
\item Every monomorphism is the kernel of its cokernel, and every epimorphism is the cokernel of its kernel. 
\item For all morphisms $f\in\mA$, the unique morphism $$\text{coker}(\ker(f)\to C)\overset{\cong}{\to}\ker(D\to\text{coker}(f))$$ is an isomorphism. 
\end{itemize}
\begin{proof}
$(2)\implies(3)$: Since $\text{coker}(\ker(f)\to C)=\ker(D\to\text{coker}(f))$ we are done. 
\end{proof}
\end{lmm}

\begin{eg}{}{}\\
Let $R$ be a ring. Then ${_R}\bold{Mod}$ is an abelian category. 
\end{eg}

\begin{prp}{}{}\\
Let $\mA$ be an abelian category. Let $\mC$ be a small category. Then $\mA^\mC$ is an abelian category. 
\end{prp}

\subsection{Injecltivity, Surjectivity and Images}
\begin{defn}{Injectivity and Surjectivity}{} Let $f:X\to Y$ be a morphism in an abelian category. 
\begin{itemize}
\item We say that $f$ is injective if $\ker(f)=0$
\item We say that $f$ is surjective if $\text{coker}(f)=0$
\end{itemize}
\end{defn}

In particular, these notions coincide that of epics and monics in an abelian category. 

\begin{prp}{}{} Let $f:X\to Y$ be a morphism in an abelian category. Then the following are true. 
\begin{itemize}
\item $f$ is injective if and only if $f$ is a monomorphism
\item $f$ is surjective if and only if $f$ is epimorphism
\end{itemize}
\end{prp}

\begin{lmm}{}{}\\
Let $\mA$ be an abelian category. Let $f:X\to Y$ be a morphism. Then the following are true. 
\begin{itemize}
\item The morphism $\ker(f)\to X$ is injective. 
\item The morphism $Y\to\text{coker}(f)$ is surjective. 
\end{itemize}
\begin{proof}
Let $\iota:\ker(f)\to X$ be universal morphism. Let $a,b:Z\to\ker(f)$ be morphisms such that $\iota\circ a=\iota\circ b$. Applying the universal property of kernels. We know that $f\circ\iota\circ a=f\circ\iota\circ b=0$ since $f\circ\iota=0$. Therefore we can apply the universal property of kernels to $\iota\circ a=\iota\circ b$ show that there exists a unique morphism $u:Z\to\ker(f)$ such that $\iota\circ u=\iota\circ a=\iota\circ b$. But since $u$ is unique, we must have $a=u=b$. \\

The proof for cokernels is dual. 
\end{proof}
\end{lmm}

\begin{defn}{The Image of a Morphism}{}\\
Let $\mA$ be an abelian category. Let $C,D\in\mA$. Let $f:C\to D$ be a morphism. Define the image of $f$ to be $$\im(f)=\ker(D\to\text{coker}(f))\cong\text{coker}(\ker(f)\to C)$$
\end{defn}

Indeed, thinking in terms of modules over a ring $R$, the cokernel of an $R$-module homomorphism $f:M\to N$ is $\frac{N}{\im(f)}$ together with the canonical map $N\to\frac{N}{\im(f)}$. Then the kernel of this map is precisely $\im(f)$. On the other hand, the cokernel of the morphism $ker(f)\to M$ is $\frac{M}{\ker(f)}$ since the inclusion map is injective, and by the first isomorphism theorem $\frac{M}{\ker(f)}\cong\im(f)$. 

\begin{lmm}{}{}\\
Let $\mA$ be an abelian category. Let $A,B\in\mA$ be objects. Let $f:A\to B$ be a morphism. Then there exists a natural and unique (up to isomorphism) factorization of $f$ into an epimorphism followed by an monomorphism: \\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	A && B \\
	& {\exists!C}
	\arrow["f", from=1-1, to=1-3]
	\arrow["{\exists!}"', dashed, two heads, from=1-1, to=2-2]
	\arrow["{\exists!}"', dashed, tail, from=2-2, to=1-3]
\end{tikzcd}} \\
Such an object that $f$ factors through is isomorphic to $\im(f)$. 
\begin{proof}\\
We have seen that the map $C\to\text{coker}(\ker(f)\to C)$ is an epimorphism in lmm1.2.10. Similarly, we have seen that the map $\ker(D\to\text{coker}(f))\to D$ is a monomorphism. The factorization is clearly natural. It remains to show that $\im(f)$ is the unique object such that $f$ factors through as an epimorphism followed by a monomorphism. 
\end{proof}
\end{lmm}

\begin{lmm}{}{}\\
Let $\mA$ be an abelian category. Let $A,B\in\mA$ be objects. Let $f:A\to B$ be a morphism. Then the following are true. 
\begin{itemize}
\item If $f$ is injective, then $A\cong\im(f)$. 
\item If $f$ is surjective, then $\im(f)\cong B$. 
\end{itemize}
\begin{proof}\\
We know that $A\to\im(f)$ is surjective by the above. Since $f=A\to\im(f)\to B$ and $f$ is injective, we conclude that $A\to\im(f)$ is injective. \\

Similarly, by the above lemma we know that $\im(f)\to B$ is injective. Since $f=A\to\im(f)\to B$ and $f$ is surjective, we conclude that $\im(f)\to B$ is surjective. 
\end{proof}
\end{lmm}

\subsection{Additive Functors}
\begin{defn}{Additive Functors}{} Let $\mA,\mB$ be abelian categories. We say that a functor $F:\mA\to\mB$ is additive if for every $X,Y\in\mA$, the map $$\Hom_\mA(X,Y)\to\Hom_\mB(F(X),F(Y))$$ is a homomorphism of abelian groups. 
\end{defn}

\begin{prp}{}{}\\
Let $\mA,\mB$ be abelian categories. Let $F:\mA\to\mB$ be a functor. Then the following are equivalent. 
\begin{itemize}
\item $F$ is additive. 
\item $F$ preserves the zero object and binary direct sums. 
\item $F$ preserves finite products. 
\item $F$ preserves finite coproducts. 
\end{itemize}
\end{prp}

\subsection{Exact Sequences}
\begin{defn}{Exact Sequences}{}\\
Let $\mA$ be an abelian category. Let the following be sequence of morphisms in $\mA$: \\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	A & B & C
	\arrow["f", from=1-1, to=1-2]
	\arrow["g", from=1-2, to=1-3]
\end{tikzcd}}\\~\\
We say that the sequence is exact at $B$ if the following are true. 
\begin{itemize}
\item $g\circ f=0$. 
\item Since $g\circ f=0$, we know that $g$ factors through $\text{coker}(f)$. Then the composite $\im(f)=\ker(B\to\text{coker}(f))\to B\overset{g}{\to} C$ is $0$ since $\im(f)\to B\to\text{coker}(f)=0$. By the universal property of kernels, there exists a unique morphism $\im(f)\to\ker(g)$: \\~\\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	{\im(f)=\ker(B\to\text{coker}(f))} \\
	{\ker(g)} & B & {\text{coker}(f)} \\
	&& C
	\arrow["{\exists!}", dashed, from=1-1, to=2-1]
	\arrow[from=1-1, to=2-2]
	\arrow[from=2-1, to=2-2]
	\arrow[from=2-2, to=2-3]
	\arrow["g"', from=2-2, to=3-3]
	\arrow["{\exists!}", dashed, from=2-3, to=3-3]
\end{tikzcd}}\\~\\
We require that this morphism is an isomorphism. 
\end{itemize}
\end{defn}

\begin{defn}{Short Exact Sequence}{}\\
Let $\mA$ be an abelian category. A short exact sequence in $\mA$ is a diagram of the form \\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	0 & A & B & C & 0
	\arrow[from=1-1, to=1-2]
	\arrow["f", from=1-2, to=1-3]
	\arrow["g", from=1-3, to=1-4]
	\arrow[from=1-4, to=1-5]
\end{tikzcd}}\\~\\
such that the sequence is exact in $A$, $B$ and $C$. 
\end{defn}

\begin{lmm}{}{}\\
Let $\mA$ be an abelian category. Then the following are true. 
\begin{itemize}
\item Let $0\to A\to B\to C$ be a sequence of morphisms in $\mA$. Then it is exact if and only if $g\circ f=0$ and $A\cong\ker(g)$. 
\item Let $A\to B\to C\to 0$ be a sequence of morphisms in $\mA$. Then it is exact if and only if $g\circ f=0$ and $C\cong\text{coker}(g)$. 
\end{itemize}
\begin{proof}\\
Suppose that $g\circ f=0$ and $A\cong\ker(g)$. We first show exactness at $A$. Trivially, we have $0\to A\to B=0$. Moreover, $\im(0\to A)=0=\ker(\ker(g)\to B)$. Now we show exactness at $B$. We have $g\circ f=0$ already by assumption, and $$\im(\ker(g)\to B)=\text{coker}(\ker(\ker(g)\to B)\to\ker(g))=\text{coker}(0\to\ker(g))=\ker(g)$$ Hence the sequence is exact. \\

Now suppose that $0\to A\to B\to C$ is an exact sequence. Then clearly we have $g\circ f=0$. To show that $A\cong\ker(g)$, we appeal to the universal property of kernels. Since $A\to B\to C=0$, the morphism $f:A\to B$ must factor through $\ker(B\to C)$, hence we obtain the following diagram: \\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	A &&& B \\
	& {\im(f)} & {\ker(g)}
	\arrow["f", tail, from=1-1, to=1-4]
	\arrow[two heads, from=1-1, to=2-2]
	\arrow[from=1-1, to=2-3]
	\arrow[tail, from=2-2, to=1-4]
	\arrow["\cong"', from=2-2, to=2-3]
	\arrow[from=2-3, to=1-4]
\end{tikzcd}}\\~\\
Since $A\to\im(f)\to\ker(g)$ is surjective, we have that $A\to\ker(g)$ is surjective. Since $f$ is injective, we have that $A\to\ker(g)$ is injective. Hence $A\cong\ker(g)$. \\

The second statement is entirely dual to the first statement and so we omit it. 
\end{proof}
\end{lmm}

\subsection{Properties of Exact Sequences}
\begin{thm}{The Snake Lemma}{}\\
Let $\mA$ be an abelian category. Let the following be a commutative diagram in $\mA$: \\~\\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
 & A\arrow[r, "f"]\arrow[d, "a"] & B\arrow[r, "g"]\arrow[d, "b"] & C\arrow[r]\arrow[d, "c"] & 0\\
0\arrow[r] & A'\arrow[r, "f'"] & B'\arrow[r, "g'"] & C' & 
\end{tikzcd}}\\~\\
If the two rows are exact, then there exists a connecting homomorphism $\delta:\ker(c)\to\text{coker}(a)$ such that the following sequence is exact: \\~\\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	0 & {\ker(f)} & {\ker(a)} & {\ker(b)} & {\ker(c)} \\
	&& {\text{coker}(a)} & {\text{coker}(b)} & {\text{coker}(c)} & {\text{coker}(g')} & 0
	\arrow[from=1-1, to=1-2]
	\arrow[from=1-2, to=1-3]
	\arrow[from=1-3, to=1-4]
	\arrow[from=1-4, to=1-5]
	\arrow[out=0, in=180, looseness=2, from=1-5, to=2-3]
	\arrow[from=2-3, to=2-4]
	\arrow[from=2-4, to=2-5]
	\arrow[from=2-5, to=2-6]
	\arrow[from=2-6, to=2-7]
\end{tikzcd}}\\
\begin{proof}\\
We first show that the first three terms in the sequence are exact. 
\end{proof}
\end{thm}

It is called the snake lemma because of the following diagram: \\
\adjustbox{scale=1.0,center}{\begin{tikzcd}

	&& A & B & C & 0 \\

	\arrow[from=1-5, to=2-5]

	\arrow[from=2-2, to=2-3]
	\arrow[from=2-3, to=2-4]

\begin{lmm}{Short Five Lemma}{}\\
Let $\mA$ be an abelian category. Let the following be a commutative diagram in $\mA$: \\~\\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	0 & A & B & C & 0 \\
	0 & {A'} & {B'} & {C'} & 0
	\arrow[from=1-1, to=1-2]
	\arrow[from=1-2, to=1-3]
	\arrow["f"', from=1-2, to=2-2]
	\arrow[from=1-3, to=1-4]
	\arrow["g"', from=1-3, to=2-3]
	\arrow[from=1-4, to=1-5]
	\arrow["h", from=1-4, to=2-4]
	\arrow[from=2-1, to=2-2]
	\arrow[from=2-2, to=2-3]
	\arrow[from=2-3, to=2-4]
	\arrow[from=2-4, to=2-5]
\end{tikzcd}}\\~\\
If any two of $f,g,h$ are isomorphisms, then the third one is also an isomorphism. 
\begin{proof}\\
Applying the snake lemma gives an exact sequence of the form $$\ker(f)\to\ker(g)\to\ker(h)\to\text{coker}(f)\to\text{coker}(g)\to\text{coker}(h)$$ If $f$ and $g$ are isomorphisms, then the snake lemma shows that $\ker(h)=0$. Hence $h$ is injective. Since $B\to B'\to C'$ is surjective, we have that $B\to C\to C'$ is surjective by commutativity of the diagram, hence $h:C\to C'$ is surjective. Thus $h$ is an isomorphism. \\

If $g$ and $h$ are isomorphisms, then the snake lemma shows that $\text{coker}(f)=0$. Hence $f$ is surjective. Since $A\to B\to B'$ is injective, we have that $A\to A'\to B'$ is also injective by commutativity of the diagram, hence $f:A\to A'$ is injective. Hence $f$ is an isomorphism. \\

If $f$ and $h$ are isomorphisms, then the snake lemma shows that $\ker(g)=\text{coker}(g)=0$. Hence $g$ is an isomorphism. 
\end{proof}
\end{lmm}

\begin{defn}{Split Complexes}{}\\
Let $\mA$ be an abelian category. Let $C\in\bold{Ch}(\mA)$ be a chain complex. We say that $C$ is split if there exists maps $s_n:C_n\to C_{n+1}$ such that $$\partial=\partial\circ s\circ\partial$$ If $C$ is also exact, then we call $C$ split exact. 
\end{defn}

\begin{thm}{The Splitting Lemma}{1.2.6} Let $\mA$ be an abelian category. Let $A,B,C\in\mA$. Then the following are equivalent for a short exact sequence \\~\\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
0\arrow[r] & A\arrow[r, "f"] & B\arrow[r, "g"] & C\arrow[r] & 0
\end{tikzcd}}\\
\begin{itemize}
\item The short exact sequence is split exact. 
\item There is an isomorphism $B\cong A\oplus C$. 
\item There exists a morphism $p:B\to A$ such that $p\circ f=\text{id}_A$. 
\item There exists a morphism $s:C\to B$ such that $g\circ s=\text{id}_C$. 
\end{itemize} 
\end{thm}

\end{prp}

\subsection{Exact Functors and the Freyd-Mitchell Embedding Theorem}
\begin{defn}{Exact Functors}{} Let $\mA,\mB$ be abelian categories. Let $F:\mA\to\mB$ be an additive functor. Let $0\to A\to B\to C\to 0$ be an exact sequence in $\mA$. We say that $F$ is exact if the sequence $$0\to F(A)\to F(B)\to F(C)\to 0$$ is exact. 
\end{defn}

\begin{eg}{}{}\\
Let $R$ be a commutative ring. Let $S\subseteq R$ be a multiplicative subset. Then the functor $$S^{-1}:{_R}\bold{Mod}\to{_S^{-1}R}\bold{Mod}$$ is exact. 
\end{eg}

\begin{lmm}{}{} Let $\mA,\mB$ be abelian categories. Let $F:\mA\to\mB$ be a functor. If $F$ is an equivalence of categories, then $F$ is an exact functor. 
\end{lmm}

\begin{defn}{Abelian Subcategories}{}\\
Let $\mA$ be an abelian category. Let $\mB$ be a subcategory of $\mA$. We say that $\mB$ is an abelian subcategory of $\mB$ if the following are true. 
\begin{itemize}
\item $\mB$ is an abelian category. 
\item The inclusion $\mB\hookrightarrow\mA$ is an exact functor. 
\end{itemize}
\end{defn}

\begin{eg}{}{}\\
Let $R$ be a ring. Then ${_R}\bold{FGMod}$ is an abelian subcategory of ${_R}\bold{Mod}$ if and only if $R$ is Noetherian. 
\end{eg}

The notion of abelian subcategories is rather subtle: since exact sequences in $\mB$ must be exact in $\mA$, this means that for example, kernels formed in $\mB$ must coincide with the kernels formed in $\mA$ with the morphism. This is not true in for example: the category of sheaves and the category of presheaves (they both are abelian categories but the category of sheaves is not an abelian subcategory of the category of presheaves)

\begin{thm}{Freyd-Mitchell Embedding Theorem}{}\\
Let $\mA$ be a small abelian category. Then there exists a ring $R$ and an exact, fully faithful embedding $F:\mA\to{_R\bold{Mod}}$. 
\end{thm}

Such a functor allows us think about diagrams in $\mA$ as if they were diagrams in ${_R\bold{Mod}}$

\subsection{Adjunctions in the Category of Modules}
\begin{thm}{Tensor-Hom Adjunction}{} Let $R$ be a ring and let $M$ be an $R$-module. Then there is an adjunction $$-\otimes_RM:{_R\bold{Mod}}\rightleftarrows{_R\bold{Mod}}:\Hom(M,-)$$ Explicitly, there is a natural isomorphism $$\Hom_{_R\bold{Mod}}(-\otimes_RM,-)\cong\Hom_{_R\bold{Mod}}(-,\Hom_{_R\bold{Mod}}(M,-))$$ 
\end{thm}

More generally, the tensor-hom adjunction is a phenomena that exhibits ${_R}\bold{Mod}$  as a closed monoidal category. 

\begin{prp}{}{} Let $R$ be a ring. Then $({_R}\bold{Mod},\otimes_R,R)$ is a symmetric monoidal category. Moreover, it is a closed monoidal category with internal hom given by the usual hom functor $\text{HOM}_{{_R}\bold{Mod}}(-,-)=\Hom_R(-,-)$. 
\end{prp}

We have seen that the left and right Hom functors $$\Hom(M,-):{_R\bold{Mod}}\to{_R\bold{Mod}}\;\;\;\;\text{ and }\;\;\;\;\Hom(-,M):{_R\bold{Mod}}\to{_R\bold{Mod}}$$ are left exact functors. Using the tensor-hom adjunction, we can deduce the following. 

\begin{prp}{}{} Let $R$ be a ring and let $M$ be an $R$-module. Then both $$-\otimes_R M:{_R\bold{Mod}}\to{_R\bold{Mod}}\;\;\;\;\text{ and }\;\;\;\;M\otimes_R-:{_R\bold{Mod}}\to{_R\bold{Mod}}$$ are right exact functors. 
\end{prp}

We find it important to notice the relation between all four functor here. Notice that because there is a canonical isomorphism $M\otimes_RN\cong N\otimes_RM$, we can replace $-\otimes_RM$ in the natural isomorphism $$\Hom_{_R\bold{Mod}}(-\otimes_RM,-)\cong\Hom_{_R\bold{Mod}}(-,\Hom_{_R\bold{Mod}}(M,-))$$ with $M\otimes_R-$. This leaves us with the question: What is the functor $\Hom(-,M)$ adjoint to? The answer is itself! $$\Hom_R(A,\Hom_R(B,C))\cong\Hom(B,\Hom(A,C))$$ Therefore it is more often to discuss $\Hom(M,-)$ in conjunction with $-\otimes_RM$, while $\Hom(-,M)$ is often left behind. \\~\\

\begin{thm}{Change of Base Ring I}{}\\
Let $R,S$ be commutative rings. Let $\varphi:R\to S$ be a ring homomorphism. Then there is an adjunction $$-\otimes_RS:\bold{Mod}_R\overset{\dashv}{\rightleftarrows}\bold{Mod}_S:-|_R$$
\end{thm}

\begin{thm}{Change of Base Ring II}{}\\
Let $R,S$ be commutative rings. Let $\varphi:R\to S$ be a ring homomorphism. Then there is an adjunction $$-|_R:\bold{Mod}_S\overset{\dashv}{\rightleftarrows}\bold{Mod}_R:\Hom_R(S,-)$$
\end{thm}

\subsection{Graded Objects and Filtrations}
\begin{defn}{Graded Objects}{}\\
Let $G$ be an abelian group. Let $\mA$ be an abelian category. A $G$-graded object in $\mA$ is a collection $$C_\bullet=(C_g)_{g\in G}$$ for $C_g\in\mA$. 
\end{defn}

\begin{defn}{Morphisms of Graded Objects}{}\\
Let $G$ be an abelian group. Let $\mA$ be an abelian category. Let $C_\bullet,D_\bullet$ be $G$-graded objects. A morphism from $C_\bullet$ to $D_\bullet$ is a collection $$f_\bullet=(f_g)_{g\in G}$$ for $f_g:C_g\to D_g\in\mA$. 
\end{defn}

\begin{defn}{The Category of Graded Objects}{}\\
Let $G$ be an abelian group. Let $\mA$ be an abelian category. Define the category $$\bold{Gr}_G(\mA)$$ of $G$-graded objects as follows. 
\begin{itemize}
\item Objects of $\bold{Gr}_G(\mA)$ are graded objects of $\mA$. 
\item Morphisms of $\bold{Gr}_G(\mA)$ are the morphisms of graded objects. 
\item Composition is given by index-wise composition. 
\end{itemize}
\end{defn}

\begin{lmm}{}{}\\
Let $G$ be an abelian group. Let $\mA$ be an abelian category. Then $\bold{Gr}_G(\mA)$ is an abelian category. 
\end{lmm}

\begin{defn}{The Shifting Functor}{}\\
Let $G$ be an abelian group. Let $\mA$ be an abelian category. Let $g_0\in G$. Define the $g$th shifting functor $$-[g_0]:\bold{Gr}_G(\mA)\to\bold{Gr}_G(\mA)$$ as follows. 
\begin{itemize}
\item For each object $C_\bullet$, define $C_\bullet[g_0]=(C_{g+g_0})_{g\in G}$. 
\item For each morphisms $f_\bullet:C_\bullet\to D_\bullet$, define $f_\bullet[g]=(f_{g+g_0})_{g\in G}$. 
\end{itemize}
\end{defn}

\begin{defn}{Morphisms of Given Degree}{}\\
Let $G$ be an abelian group. Let $\mA$ be an abelian category. Let $g_0\in G$. Let $C_\bullet,D_\bullet\in\bold{Gr}_G(\mA)$. A morphism from $C_\bullet$ to $D_\bullet$ of degree $g_0$ is a morphism $C_\bullet\to D_\bullet[g_0]$. 
\end{defn}

\begin{lmm}{}{}\\
Let $G$ be an abelian group. Let $\mA$ be an abelian category. Let $g_0\in G$. Then there is a natural isomorphism $$\Hom_{\bold{Gr}_G(\mA)}(C_\bullet,D_\bullet[g_0])\cong\Hom_{\bold{Gr}_G(\mA)}(C_\bullet[-g_0],D_\bullet)$$ given by $f\mapsto f[-g_0]$. 
\end{lmm}

\begin{defn}{Filtration of an Object}{}\\
Let $\mA$ be an abelian category. Let $A\in\mA$. 
\begin{itemize}
\item A descending filtration of $A$ is a collection $\mF=(F_nA)_{n\geq n_0}$ such that there are monomorphisms $F_{n+1}A\to F_nA$. We denote the filtration by $$0\subseteq\cdots\subseteq F_{n+1}A\subseteq F_nA\subseteq\cdots\subseteq A$$
\item An ascending filtration of $A$ is a collection $\mF=(F_nA)_{n\geq n_0}$ such that there are monomorphisms $F_nA\to F_{n+1}A$. We denote the filtration by $$0\subseteq\cdots\subseteq F_nA\subseteq F_{n+1}A\subseteq\cdots\subseteq A$$
\end{itemize}
\end{defn}

\begin{defn}{Types of Filtrations}{}\\
Let $\mA$ be an abelian category. Let $A\in\mA$. Let $(F_nA)_{n\in\N^+}$ be a filtration of $A$. 
\begin{itemize}
\item We say that the filtration is finite if there exists $s\in\N^+$ such that $F_sA=0$. 
\item We say that the filtration is exhaustive if $\text{colim}_{n\to\infty}F_nA\cong A$. 
\item We say that the filtration if Hausdorff if $\text{lim}_{n\to\infty}F_nA\cong 0$. 
\end{itemize}
\end{defn}

\subsection{The Category of Chain and Cochain Complexes}
\begin{defn}{Chain Complexes}{}\\
Let $\mA$ be an abelian category. A chain complex $(C_\bullet,\partial_\bullet^C)$ in $\mA$ consists of the following data: 
\begin{itemize}
\item A family of objects $C_n\in\mA$ for $n\in\Z$
\item Morphisms $\partial_n^C:C_n\to C_{n-1}$ in $\mA$ for $n\in\Z$. 
\end{itemize}
such that the chain condition: $$\partial_n\circ\partial_{n+1}=0$$ is satisfied for all $n\in\Z$. 
\end{defn}

We often write a chain complex in a diagram: \\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	\cdots & {C_{n+1}} & {C_n} & {C_{n-1}} & \cdots
	\arrow[from=1-1, to=1-2]
	\arrow["{\partial_{n+1}^C}", from=1-2, to=1-3]
	\arrow["{\partial_n^C}", from=1-3, to=1-4]
	\arrow[from=1-4, to=1-5]
\end{tikzcd}}\\
It is also common to denote a chain complex as $C$ with the differential being implicit. \\

A chain complex is a $\Z$-graded object in an abelian category together with a structure map $d:C_\bullet\to C_\bullet[-1]$ satisfying $d\circ d=0$. 

\begin{defn}{Chain Map}{}\\
Let $\mA$ be an abelian category. Let $C,D$ be two chain complexes in $\mA$. A chain map $f:C\to D$ is a family of maps $$f_n:C_n\to D_n$$ in $\mA$ for $n\in\Z$ such that the chain map condition: $$\partial_n^D\circ f_n=f_{n-1}\circ\partial_n^C$$ is satisfied for all $n\in\Z$. 
\end{defn}

In other words, we have the following commutative diagram: \\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	\cdots & {C_{n+1}} & {C_n} & {C_{n-1}} & \cdots \\
	\cdots & {D_{n+1}} & {D_n} & {D_{n-1}} & \cdots
	\arrow[from=1-1, to=1-2]
	\arrow["{\partial_{n+1}^C}", from=1-2, to=1-3]
	\arrow["{f_{n+1}}"', from=1-2, to=2-2]
	\arrow["{\partial_n^C}", from=1-3, to=1-4]
	\arrow["{f_n}"', from=1-3, to=2-3]
	\arrow[from=1-4, to=1-5]
	\arrow["{f_{n-1}}", from=1-4, to=2-4]
	\arrow[from=2-1, to=2-2]
	\arrow["{\partial_{n+1}^D}"', from=2-2, to=2-3]
	\arrow["{\partial_n^D}"', from=2-3, to=2-4]
	\arrow[from=2-4, to=2-5]
\end{tikzcd}}

\begin{lmm}{}{}\\
Let $\mA$ be an abelian category. The following are true regarding chain maps. 
\begin{itemize}
\item Let $C,D,E$ be chain complexes in $\mA$. Let $f:C\to D$ and $g:D\to E$ be chain maps. Then $g\circ f$ defined by $(g\circ f)_n=g_n\circ f_n$ is also a chain map. 
\item Let $C$ be a chain complex in $\mA$. Then $\text{id}:C\to C$ defined by $\text{id}_n=\text{id}_{C_n}$ is also a chain map. 
\end{itemize}
\begin{proof}\\
Since $f$ and $g$ are chain maps, we have that $\partial_n^D\circ f_n=f_{n-1}\circ\partial_n^C$ and $\partial_n^E\circ g_n=g_{n-1}\circ\partial_n^D$ for all $n\in\Z$. Then we have $$\partial_n^E\circ g_n\circ f_n=g_{n-1}\circ\partial_n^D\circ f_n=g_{n-1}\circ f_{n-1}\circ\partial_n^C$$ for all $n\in\Z$ and so we are done. \\

The condition on chain maps is trivial with the identity map. 
\end{proof}
\end{lmm}

\begin{defn}{Category of Chain Complexes}{}\\
Let $\mA$ be an abelian category. Define $$\bold{Ch}(\mA)$$ to be the category of chain complexes where 
\begin{itemize}
\item The objects are chain complexes of objects in $\mA$. 
\item The morphisms are chain maps. 
\item Composition is given by composition of functions pointwise. 
\end{itemize}
\end{defn}

\begin{defn}{Bounded Chain Complexes}{}\\
Let $\mA$ be an abelian category. Let $C\in\bold{Ch}(\mA)$ be a chain complex. 
\begin{itemize}
\item We say that $C$ is bounded above if there exists $N\in\Z$ such that $C_n=0$ for all $n>N$. 
\item We say that $C$ is bounded below if there exists $N\in\Z$ such that $C_n=0$ for all $n<N$. 
\item We say that $C$ is bounded if it is bounded above and bounded below. 
\end{itemize}
\end{defn}

\begin{defn}{Variants of the Category of Chain Complexes}{}\\
Let $\mA$ be an abelian category. 
\begin{itemize}
\item Define the category $$\bold{Ch}_{\geq 0}(\mA)$$ of non-negative chain complexes to be the full subcategory of $\bold{Ch}(\mA)$ consisting of chain complexes that is $0$ in negative degrees. 
\item Define the category $$\bold{Ch}_{\leq 0}(\mA)$$ of non-positive chain complexes to be the full subcategory of $\bold{Ch}(\mA)$ consisting of chain complexes that is $0$ in positive degrees. 
\end{itemize}
\end{defn}

\begin{prp}{}{}\\
Let $\mA$ be an abelian category. Then $\bold{Ch}(\mA)$ is also an abelian category. 
\end{prp}

Thus $\bold{Ch}(\bold{Ch}(\mA))$ also makes sense as a category. The objects are given by chain complexes of chain complexes. 

\begin{defn}{Cochain Complexes}{}\\
Let $\mA$ be an abelian category. A cochain complex $(C^\bullet,\partial_C^\bullet)$ in $\mA$ consists of the following data: 
\begin{itemize}
\item A family of objects $C^n\in\mA$ for $n\in\Z$
\item Morphisms $\partial_C^n:C^n\to C^{n+1}$ in $\mA$ for $n\in\Z$. 
\end{itemize}
such that the chain condition: $$\partial^n\circ\partial^{n-1}=0$$ is satisfied for all $n\in\Z$. 
\end{defn}

We often write a cochain complex in a diagram: \\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	\cdots & {C^{n-1}} & {C^n} & {C^{n+1}} & \cdots
	\arrow[from=1-1, to=1-2]
	\arrow["{\partial_C^{n-1}}", from=1-2, to=1-3]
	\arrow["{\partial_C^n}", from=1-3, to=1-4]
	\arrow[from=1-4, to=1-5]
\end{tikzcd}}\\
Notice that when we write a cochain complex in diagram form, the arrows go the same way but the indexing increases in the opposite direction. \\

Algebraically speaking, there is no difference between a chain complex and a cochain complex, other than the fact that the boundary maps run in the other direction. For $(C^\bullet,\partial^\bullet)$ a cochain complex, we can form a chain complex by setting $C_n=C^{-n}$ and then using the same boundary maps. 

\begin{defn}{Chain Maps}{}\\
Let $\mA$ be an abelian category. Let $C,D$ be two chain complexes in $\mA$. A chain map $f:C\to D$ is a family of maps $$f^n:C^n\to D^n$$ in $\mA$ for $n\in\Z$ such that the chain map condition: $$\partial_D^n\circ f_n=f_{n-1}\circ\partial_C^n$$ is satisfied for all $n\in\Z$. 
\end{defn}

In other words, we have the following commutative diagram: \\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	\cdots & {C^{n-1}} & {C^n} & {C^{n+1}} & \cdots \\
	\cdots & {D^{n-1}} & {D^n} & {D^{n+1}} & \cdots
	\arrow[from=1-1, to=1-2]
	\arrow["{\partial_C^{n-1}}", from=1-2, to=1-3]
	\arrow["{f^{n-1}}"', from=1-2, to=2-2]
	\arrow["{\partial_C^n}", from=1-3, to=1-4]
	\arrow["{f^n}"', from=1-3, to=2-3]
	\arrow[from=1-4, to=1-5]
	\arrow["{f^{n+1}}", from=1-4, to=2-4]
	\arrow[from=2-1, to=2-2]
	\arrow["{\partial_D^{n-1}}"', from=2-2, to=2-3]
	\arrow["{\partial_D^n}"', from=2-3, to=2-4]
	\arrow[from=2-4, to=2-5]
\end{tikzcd}}

\begin{defn}{Category of Cochain Complexes}{}\\
Let $\mA$ be an abelian category. Define the category of cochain complexes $$\bold{CCh}(\mA)$$ to consist of the following data. 
\begin{itemize}
\item The objects are the cochain complexes over $\mA$
\item The morphisms are the chain maps
\item Composition is given by the composition of maps on all levels of the chain map. 
\end{itemize}
\end{defn}

\begin{defn}{Bounded Cochain Complexes}{}\\
Let $\mA$ be an abelian category. Let $C\in\bold{CCh}(\mA)$ be a cochain complex. 
\begin{itemize}
\item We say that $C$ is bounded above if there exists $N\in\Z$ such that $C_n=0$ for all $n>N$. 
\item We say that $C$ is bounded below if there exists $N\in\Z$ such that $C_n=0$ for all $n<N$. 
\item We say that $C$ is bounded if it is bounded above and bounded below. 
\end{itemize}
\end{defn}

\begin{defn}{Variants of the Category of Cochain Complexes}{}\\
Let $\mA$ be an abelian category. 
\begin{itemize}
\item Define the category $$\bold{CCh}_{\geq 0}(\mA)$$ of non-negative cochain complexes to be the full subcategory of $\bold{CCh}(\mA)$ consisting of cochain complexes that is $0$ in negative degrees. 
\item Define the category $$\bold{CCh}_{\leq 0}(\mA)$$ of non-positive chain complexes to be the full subcategory of $\bold{CCh}(\mA)$ consisting of cochain complexes that is $0$ in positive degrees. 
\end{itemize}
\end{defn}

\begin{defn}{The Inclusion Functors}{}\\
Let $\mA$ be an abelian category. Let $n\in\Z$. Define the inclusion functor $$c_n:\bold{\mA}\to\bold{Ch}(\mA)$$ by the following. 
\begin{itemize}
\item For each $X\in\mA$, we have $c_n(X)$ is the chain complex whose $n$th position is $X$ and $0$ for all the others, together with the $0$ maps. 
\item For each morphism $f:X\to Y$, define $c_n(f):c_n(X)\to c_n(Y)$ to be the chain map whose $n$th position is $f$ and $0$ in all other others. 
\end{itemize}
Define the inclusion functor $$c^n:\bold{\mA}\to\bold{CCh}(\mA)$$ by the following. 
\begin{itemize}
\item For each $X\in\mA$, we have $c^n(X)$ is the cochain complex whose $n$th position is $X$ and $0$ for all the others, together with the $0$ maps. 
\item For each morphism $f:X\to Y$, define $c^n(f):c^n(X)\to c^n(Y)$ to be the cochain map whose $n$th position is $f$ and $0$ in all other others. 
\end{itemize}
\end{defn}

The inclusion functors are in general not fully faithful. 

\subsection{Homology and Cohomology}
\begin{defn}{The Homology Groups of a Chain Complex}{}\\
Let $\mA$ be an abelian category. Let $(C_\bullet,\partial_\bullet)$ be a chain complex in $\mA$. 
\begin{itemize}
\item Define the $n$-cycles of the chain complex to be $$Z_n(C_\bullet)=\ker(\partial_n)$$
\item Define the $n$-boundaries of the chain complex to be $$B_n(C_\bullet)=\im(\partial_{n+1})$$
\item Since $\partial_n\circ\partial_{n+1}=0$, we have that $\partial_n$ factors through $\text{coker}(\partial_{n+1})$. Then the composite $B_n(C_\bullet)\to C_n\overset{\partial_n}{\to} C_{n-1}$ is $0$ since $B_n(C_\bullet)\to C_n\to\text{coker}(\partial_{n+1})=0$. By the universal property of kernels, there exists a unique morphism $B_n(C_\bullet)\to Z_n(C_\bullet)$: \\~\\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	{B_n(C_\bullet)} \\
	{Z_n(C_\bullet)} & C_n & {\text{coker}(\partial_{n+1})} \\
	&& C_{n-1}
	\arrow["{\exists!}", dashed, from=1-1, to=2-1]
	\arrow[from=1-1, to=2-2]
	\arrow[from=2-1, to=2-2]
	\arrow[from=2-2, to=2-3]
	\arrow["\partial_n"', from=2-2, to=3-3]
	\arrow["{\exists!}", dashed, from=2-3, to=3-3]
\end{tikzcd}}\\~\\
Define the $n$th homology of the chain complex to be $$H_n(C_\bullet)=\text{coker}(B_n(C_\bullet)\to Z_n(C_\bullet))=\text{coker}(\im(\partial_{n+1})\to\ker(\partial_n))$$
\end{itemize}
\end{defn}

\begin{defn}{The Homology Functor}{}\\
Let $\mA$ be an abelian category. Let $n\in\N$. Define the $n$th homology functor $H_n:\bold{Ch}(\mA)\to\mA$ by as follows. 
\begin{itemize}
\item For a chain complex $C_\bullet$, we have $H_n(C_\bullet)$ is the $n$th homology of the chain complex. 
\item For a chain map $C_\bullet\to C_\bullet'$, we obtain maps $Z_n(C_\bullet)\to Z_n(C_\bullet')$ and $B_n(C_\bullet)\to B_n(C_\bullet')$ since kernels and images are functorial. Since cokernels are also functorial, we obtain a morphism $H_n(C_\bullet)\to H_n(C_\bullet')$. 
\end{itemize}
\end{defn}

\begin{lmm}{}{}\\
Let $\mA$ be an abelian category. Then $H_\ast:\bold{Ch}(\mA)\to\mA$ is an additive functor. 
\end{lmm}

\begin{defn}{Quasi-Isomorphisms}{} Let $C_\bullet,D_\bullet$ be chain complexes. Let $f:C_\bullet\to D_\bullet$ be a chain map. We say that $f$ is a quasi-isomorphism if the induced map in homology $f_\ast:H_n(C_\bullet)\to H_n(D_\bullet)$ is an isomorphism for all $n\in\N$. 
\end{defn}

\begin{eg}{}{}\\
The chain map given by the vertical arrows in $\bold{Ab}$ is a quasi-isomorphism: \\~\\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	0 & \Z & \Z & 0 \\
	0 & 0 & {\Z/2\Z} & 0
	\arrow[from=1-1, to=1-2]
	\arrow["0"', from=1-1, to=2-1]
	\arrow["\times 2", from=1-2, to=1-3]
	\arrow["0"', from=1-2, to=2-2]
	\arrow[from=1-3, to=1-4]
	\arrow["{\text{mod}\;2}", from=1-3, to=2-3]
	\arrow["0", from=1-4, to=2-4]
	\arrow[from=2-1, to=2-2]
	\arrow[from=2-2, to=2-3]
	\arrow[from=2-3, to=2-4]
\end{tikzcd}}\\~\\
In particular, there is no non-zero chain map in the opposite direction. 
\end{eg}

Because cochain complexes is the same data as a chain complex, we can also define homology groups for cochain complexes. We call them the cohomology groups of the cochain complex. 

\begin{defn}{Cohomology}{}\\
Let $\mA$ be an abelian category. Let $C^\bullet$ be a chain complex. Define the $n$th cohomology of $C^\bullet$ to be $$H^n(C^\bullet)=\text{coker}(\im(\delta^n)\to\ker(\delta^{n+1}))$$
\end{defn}

The construction of cohomology groups are functorial because the homology groups of a chain complex also are. 

\begin{defn}{The Cohomology Functor}{} Let $\mA$ be an abelian category. Let $n\in\N$. Define the $n$th cohomology functor to be the functor $$H^n:\bold{CCh}(\mA)\to\mA$$ to consist of the following data. 
\begin{itemize}
\item For each cochain complex $(C^\bullet,\delta^\bullet)$ over $\mA$, define $H^n(C^\bullet,\delta^\bullet)$ to be the $n$th cohomology group of the cochain complex. 
\item For each chain map $f^\bullet:(C^\bullet,\delta^\bullet)\to(D^\bullet,\delta^\bullet)$, the induced map $$f^\ast:H^n(C^\bullet,\delta^\bullet)\to H^n(C^\bullet,\delta^\bullet)$$ is defined by $[z]\mapsto[f^n(z)]$. 
\end{itemize}
\end{defn}

\subsection{Chain Homotopies}
\begin{defn}{Chain Homotopy}{}\\
Let $\mA$ be an abelian category. Let $f_\bullet,g_\bullet:C_\bullet\to C_\bullet'$ be two chain maps in $\bold{Ch}(\mA)$. Then a chain homotopy from $f$ to $g$ is a collection of morphisms $$\eta_n:C_n\to C_{n+1}'$$ in $\mA$ such that $$g_n-f_n=\partial_{n+1}'\eta_n+\eta_{n-1}\partial_n$$ for all $n\in\Z$. In this case, $f$ and $g$ are said to be chain homotopic and we write $f\simeq g$. 
\end{defn}

We depict the data of a homotopy as the following diagram in $\bold{Ch}(\mA)$: \\~\\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	\cdots && {C_{n+1}} && {C_n} && {C_{n-1}} && \cdots \\
	\\
	\cdots && {C_{n+1}'} && {C_n'} && {C_{n-1}'} && \cdots
	\arrow[from=1-1, to=1-3]
	\arrow["{\partial_{n+1}}", from=1-3, to=1-5]
	\arrow["{\partial_n}", from=1-5, to=1-7]
	\arrow[from=3-1, to=3-3]
	\arrow["{\partial_{n+1}'}", from=3-3, to=3-5]
	\arrow["{\partial_n'}", from=3-5, to=3-7]
	\arrow[from=1-3, to=3-1]
	\arrow["{\eta_n}"{description}, from=1-5, to=3-3]
	\arrow["{\eta_{n-1}}"{description}, from=1-7, to=3-5]
	\arrow[from=1-9, to=3-7]
	\arrow["{g_{n+1}-f_{n+1}}"{description}, from=1-3, to=3-3]
	\arrow["{g_n-f_n}"{description}, from=1-5, to=3-5]
	\arrow["{g_{n-1}-f_{n-1}}"{description}, from=1-7, to=3-7]
	\arrow[from=1-7, to=1-9]
	\arrow[from=3-7, to=3-9]
\end{tikzcd}}\\~\\

Note that the notion of chain homotopy is an equivalence relation on $\Hom_{\bold{Ch}(\mA)}(C,D)$ for any two chain complexes $C,D\in\bold{Ch}(\mA)$. 
\begin{itemize}
\item If $f:C\to D$ is a chain homotopy equivalence, then $f\simeq f$ via the chain homotopy $0$. 
\item If $f,g:C\to D$ are chain maps such that $f\simeq g$ via the chain homotopy $\eta$, then $-\eta$ is a chain homotopy such that $g\simeq f$. 
\item If $f,g,h:C\to D$ are chain maps such that $f\simeq g$ via $\eta$ and $g\simeq h$ via $\nu$, then $f\simeq h$ via $\eta+\nu$. 
\end{itemize}

\begin{lmm}{}{}\\
Let $\mA$ be an abelian category. Let $C,D\in\bold{Ch}(\mA)$ be chain complexes in $\mA$. Let $f,g:C\to D$ be chain maps. If $f$ and $g$ are chain homotopic, then $$H_\ast(f)=H_\ast(g)\in\Hom_\mA(H_n(C),H_n(D))$$ for all $n\in\Z$. 
\end{lmm}

\begin{prp}{}{}\\
Let $\mA$ be an abelian group. Let $f_1,g_1:C_\bullet\to D_\bullet$ and $f_2,g_2:D_\bullet\to E_\bullet$ be chain maps in $\bold{Ch}(\mA)$. If $f_1$ and $g_1$ are chain homotopic and $f_2$ and $g_2$ are chain homotopic, then $f_2\circ f_1$ is chain homotopic to $g_2\circ g_1$. 
\begin{proof}
The chain homotopies between $f_1$ and $g_1$ imposes the identity $$\partial\eta+\eta\partial=g_1-f_1$$ for $\eta:C_\bullet\to D_\bullet$ the given chain homotopy. Similarly, for $\nu:D_\bullet\to E_\bullet$ we have the identity $$\partial\nu+\nu\partial=g_2-f_2$$ Then we have that 
\begin{align*}
g_2\circ g_1-f_2\circ f-1&=g_2\circ g_1-g_2\circ f_1+g_2\circ f_1-f_2\circ f_1\\
&=g_2(g_1-f_1)+(g_2-f_2)\circ f_1\\
&=g_2(\partial\eta+\eta\partial)+(\partial\nu+\nu\partial)\circ f_1\\
&=\partial g_2\eta+g_2\eta\partial+\partial\nu f_1+\nu f_1\partial\\
&=\partial(g_2\eta+\nu f_1)+(g_2\eta+\nu f_1)\partial
\end{align*}
Thus $g_2\eta+\nu f_1:C_n\to E_{n+1}$ would be a valid chain homotopy from $g_2\circ g_1$ to $f_2\circ f_1$. 
\end{proof}
\end{prp}

\begin{prp}{}{}\\
Let $\mA$ be an abelian category. Let $C,D\in\bold{Ch}(\mA)$ be chain complexes. Suppose that $f,f',g,g':C\to D$ are chain maps such that $f\simeq f'$ and $g\simeq g'$, then $f+g\simeq f'+g'$. 
\end{prp}

\begin{defn}{Chain Homotopy Equivalence}{}\\
Let $\mA$ be an abelian category. Let $C,D\in\bold{Ch}(\mA)$ be chain complexes in $\mA$. Let $f:C\to D$ be a chain map. We say that $f$ is a chain homotopy equivalence if there exists a chain map $g:D\to C$ such that $$g\circ f\simeq\text{id}_C\;\;\;\text{ and }\;\;\;f\circ g\simeq\text{id}_D$$ In this case, we call $g$ the chain homotopy inverse of $f$. 
\end{defn}

\begin{lmm}{}{}\\
Let $\mA$ be an abelian category. Let $C,D\in\bold{Ch}(\mA)$ be chain complexes in $\mA$. Let $f:C\to D$ be a chain homotopy equivalence. Then the following are true. 
\begin{itemize}
\item The map $$H_\ast(f):H_n(C)\overset{\cong}{\longrightarrow}H_n(D)$$ is an isomorphism for all $n\in\Z$. 
\item If $g$ is a chain homotopy inverse of $f$, then $H_\ast(g)$ is the inverse of $H_\ast(f)$. 
\end{itemize}
\begin{proof}
We know that $b_\bullet\circ a_\bullet\simeq\text{id}_{C_\bullet}$ which means that they induce the same map: $$b_\ast\circ a_\ast=\text{id}_{H_n(C_\bullet)}$$ Similarly the chain homotopies $a_\bullet\circ b_\bullet\simeq\text{id}_{D_\bullet}$ induce the same map $$a_\ast\circ b_\ast:\text{id}_{H_n(D_\bullet)}$$ as the identity. Then these two identities mean that $a_\ast$ is both injective and surjective. 
\end{proof}
\end{lmm}

\begin{prp}{}{} Let $\mA$ be an abelian category. Then chain homotopy equivalence defines an equivalence relation on all chain complexes in $\bold{Ch}(\mA)$. 
\begin{proof}
Clearly any chain complex is chain homotopy equivalent to itself by the identity map. If $C_\bullet$ and $D_\bullet$ are chain homotopy equivalent by the chain maps $a_\bullet:C_\bullet\to D_\bullet$ and $b_\bullet:D_\bullet\to C_\bullet$, then we have the identities $b_\bullet\circ a_\bullet=\text{id}_{C_\bullet}$ and $a_\bullet\circ b_\bullet=\text{id}_{D_\bullet}$. We can then read them in reverse so that $D_\bullet$ and $C_\bullet$ are chain homotopy equivalence by the maps $b_\bullet$ and $a_\bullet$. \\~\\

Suppose further that $D_\bullet$ and $E_\bullet$ are chain homotopy equivalent via the maps $u_\bullet:D_\bullet\to E_\bullet$ and $v_\bullet:E_\bullet\to D_\bullet$. Then the maps $u_\bullet\circ a_\bullet$ and $b_\bullet\circ v_\bullet$ give a chain homotopy equivalence between $C_\bullet$ and $E_\bullet$. Indeed, upon composition, we have that they are chain  homtopic to the identity maps. 
\end{proof}
\end{prp}

\begin{defn}{Contractible Chain Complexes}{}\\
Let $\mA$ be an abelian category. Let $C\in\bold{Ch}(\mA)$ be a chain complex. We say that $C$ is contractible if the identity map $\text{id}_C$ is homotopic to the $0$ map. 
\end{defn}

\begin{eg}{}{}\\
Let $\mA$ be an abelian category. Let $X\in\mA$ be an object. Let $C\in\bold{Ch}(\mA)$ be the chain complex defined as follows: \\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	\cdots & 0 & X & X & 0 & \cdots
	\arrow[from=1-1, to=1-2]
	\arrow[from=1-2, to=1-3]
	\arrow["{\text{id}_X}", from=1-3, to=1-4]
	\arrow[from=1-4, to=1-5]
	\arrow[from=1-5, to=1-6]
\end{tikzcd}}\\~\\
where the two non-zero terms are in positions $n$ and $n+1$ for some $n\in\Z$. Then $C$ is contractible. 
\begin{proof}\\
Without loss of generality we may take $n=0$. Define $\eta:C_n\to C_{n+1}$ as follows: If $n\neq 0$, then $\eta_n$ is the $0$ map. Also $\eta_0=\text{id}_X$. I claim that $\eta$ gives a homotopy $\text{id}_C\simeq 0$. Indeed, if $n\neq 0,1$, then we have $$\partial_{n+1}\circ\eta_n+\eta_{n-1}\circ\partial_n=0\circ 0+0\circ 0=0=\text{id}_{C_n}-0$$ If $n=0$, then we have $$\partial_1\circ\eta_0+\eta_{-1}\circ\partial_0=\text{id}_X\circ\text{id}_X+0\circ 0=\text{id}_X=\text{id}_{C_0}-0$$ and finally if $n=1$, then we have $$\partial_2\circ\eta_1+\eta_0\circ\partial_1=0\circ 0+\text{id}_X\circ\text{id}_X=\text{id}_X=\text{id}_{C_1}-0$$ hence we are done. 
\end{proof}
\end{eg}

\subsection{The Homotopy Category of Cochain Complexes}
\begin{defn}{Homotopy Category of Cochain Complexes}{} Let $\mA$ be an abelian category. Define the homotopy category of chain complexes $$K(\mA)$$ to be the category defined as follows. 
\begin{itemize}
\item The objects are the objects of $\bold{CCh}(\mA)$. 
\item The morphisms are homotopy classes of chain maps. Namely, for $C,D$ chain complexes, define $$\Hom_{K(\mA)}(C,D)=\frac{\Hom_{\bold{Ch}(\mA)}(C,D)}{\sim}$$
\item Composition is given by composition of chain maps. 
\end{itemize}
Define similarly the homotopy categories $K^+(\mA)$, $K^-(\mA)$ and $K^b(\mA)$ for $\bold{CCh}^+(\mA)$, $\bold{CCh}^-(\mA)$ and $\bold{CCh}^b(\mA)$ respectively. 
\end{defn}

Note: $K(\mA)$ is in general no longer an abelian category, but it is still a triangulated category cf section 7. 

\begin{defn}{The Canonical Functor to the Homotopy Category}{}\\
Let $\mA$ be an abelian category. Define the canonical functor $$\bold{Ch}(\mA)\to K(\mA)$$ as follows. 
\begin{itemize}
\item It is the identity map on objects. 
\item For a chain map $f:C\to D$ the image of $f$ is the homotopy class $[f]:C\to D$ of the chain map. 
\end{itemize}
\end{defn}

It is clear that the canonical functor is in general not fully faithful. However, the inclusion functor $\mA\to\bold{Ch}(\mA)\to K(\mA)$ is fully faithful. 

\begin{lmm}{}{}\\
Let $\mA$ be an abelian category. Then the functor $$\mA\overset{c_0}{\longrightarrow}\bold{Ch}(\mA)\to K(\mA)$$ is fully faithful. 
\end{lmm}

\begin{prp}{}{} Let $\mA$ be an abelian category. The homotopy category of cochain complexes satisfy the following universal property. \\~\\

If $F:\bold{Ch}(\mA)\to\mD$ is a functor that sends chain homotopy equivalences to isomorphisms, then $F$ factors uniquely through $K(\mA)$: \\~\\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	{\bold{CCh}(\mA)} & {K(\mA)} \\
	& \mD
	\arrow[from=1-1, to=1-2]

\begin{defn}{Shifted Chain Complexes}{}\\
Let $\mA$ be an abelian category. Let $i\in\Z$. Let $C\in\bold{Ch}(\mA)$ be a chain complex. Define the $i$th shifted chain complex of $C_\bullet$ to be the complex $(C[i],\partial[i])$ defined as follows. 
\begin{itemize}
\item $C[i]_n=C_{n+i}$. 
\item $\partial[i]_n=(-1)^i\partial_{n+i}$
\end{itemize}
\end{defn}

Visually, the chain complex is shifted $i$ times to the left. Note that the shifted chain complex has a different chain map if $i$ is odd. This definition is only possible in an additive category. 

\begin{defn}{The Shifting Functor}{}\\
Let $\mA$ be an abelian category. Let $i\in\Z$. Define the $i$th shifting functor $$-[i]:\bold{Ch}(\mA)\to\bold{Ch}(\mA)$$ as follows. 
\begin{itemize}
\item For a chain complex $C$, define $C[i]$ to be the $i$th shifted chain complex. 
\item For a chain map $f:C\to D$, define $f[i]:C[i]\to D[i]$ by $f[i]_n=f_{n+i}$. 
\end{itemize}
\end{defn}

\begin{lmm}{}{}\\

\item Let $C\in\bold{Ch}(\mA)$ be a chain complex. For any $i\in\Z$ and $n\in\Z$, we have $$H_n(C[i])=H_{n+i}(C)$$
\end{itemize}
\end{lmm}

Let $\mA$ be an abelian category. We have seen that $\bold{Ch}(\mA)$ is also an abelian category, and so it makes sense to talk about chain complexes of chain complexes. \begin{defn}{Double Complexes}{}\\
Let $\mA$ be an abelian category. A double complex in $\mA$ consists of the following data: 
\begin{itemize}
\item A family of objects $C_{p,q}\in\mA$ for $p,q\in\Z$. 
\item Horizontal differentials $d^h:C_{p,q}\to C_{p-1,q}$ in $\mA$ for $p,q\in\Z$. 
\item Vertical differentials $d^v:C_{p,q}\to C_{p,q-1}$ in $\mA$ for $p,q\in\Z$. 
\end{itemize}
such that the following are true. 
\begin{itemize}
\item $d^h\circ d^h=0$. 
\item $d^v\circ d^v=0$. 
\item $d^v\circ d^h=d^h\circ d^v$. 
\end{itemize}
\end{defn}

We picture a double complex as follows: \\~\\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	& \vdots & \vdots & \vdots \\
	\cdots & {C_{p-1,q+1}} & {C_{p,q+1}} & {C_{p+1,q+1}} & \cdots \\
	\cdots & {C_{p-1,q}} & {C_{p,q}} & {C_{p+1,q}} & \cdots \\
	\cdots & {C_{p-1,q-1}} & {C_{p,q-1}} & {C_{p+1,q-1}} & \cdots \\
	& \vdots & \vdots & \vdots
	\arrow[from=1-2, to=2-2]
	\arrow[from=1-3, to=2-3]
	\arrow[from=1-4, to=2-4]
	\arrow[from=2-2, to=2-1]
	\arrow["{d^v}"', from=2-2, to=3-2]
	\arrow["{d^h}"', from=2-3, to=2-2]
	\arrow["{d^v}"', from=2-3, to=3-3]
	\arrow["{d^h}"', from=2-4, to=2-3]
	\arrow["{d^v}", from=2-4, to=3-4]
	\arrow[from=2-5, to=2-4]
	\arrow[from=3-2, to=3-1]
	\arrow["{d^v}"', from=3-2, to=4-2]
	\arrow["{d^h}"', from=3-3, to=3-2]
	\arrow["{d^v}"', from=3-3, to=4-3]
	\arrow["{d^h}"', from=3-4, to=3-3]
	\arrow["{d^v}", from=3-4, to=4-4]
	\arrow[from=3-5, to=3-4]
	\arrow[from=4-2, to=4-1]
	\arrow[from=4-2, to=5-2]
	\arrow["{d^h}", from=4-3, to=4-2]
	\arrow[from=4-3, to=5-3]
	\arrow["{d^h}", from=4-4, to=4-3]
	\arrow[from=4-4, to=5-4]
	\arrow[from=4-5, to=4-4]
\end{tikzcd}}\\~\\
where each square is commutative. Double complexes are also graded objects of an abelian category equipeed with differentials that decrease the first and second degrees respectively. 

\begin{defn}{The Total Complex of a Double Complex}{}\\
Let $\mA$ be an abelian category. Suppose that $\mA$ admits all relevant direct sums. Let $(C_{\bullet,\bullet},d^h,d^v)$ be a double complex. Define the total complex $\text{Tot}^\oplus(C)_\bullet$ to be the chain complex in $\mA$ constructed as follows. 
\begin{itemize}
\item For each $n\in\Z$, define $$\text{Tot}^\oplus(C)_n=\bigoplus_{p+q=n}C_{p,q}$$
\item For each $n\in\Z$, define $d:\text{Tot}^\oplus(C)_n\to\text{Tot}^\oplus(C)_{n-1}$ by $$d=d^v+(-1)^nd^h$$
\end{itemize}
\end{defn}

\begin{defn}{Total Homology of a Double Chain Complex}{}\\
Let $\mA$ be an abelian category. Suppose that $\mA$ admits all relevant products. Let $(C_{\bullet,\bullet},d^h,d^v)$ be a double complex in $\mA$. Define the total homology of $C$ to be the homology $$H_\bullet^\text{Tot}(C_{\bullet,\bullet},d^h,d^v)=H_\bullet(\text{Tot}^\oplus(C)_\bullet,d)$$ of the total complex $(\text{Tot}^\oplus(C)_\bullet,d)$
\end{defn}

\begin{defn}{The Product Total Complex of a Double Complex}{}\\
Let $\mA$ be an abelian category. Let $(C_{\bullet,\bullet},d^h,d^v)$ be a double complex. Define the product total complex $\text{Tot}^{\prod}(C)_\bullet$ to be the chain complex in $\mA$ constructed as follows. 
\begin{itemize}
\item For each $n\in\Z$, define $$\text{Tot}^{\prod}(C)_n=\prod_{p+q=n}C_{p,q}$$
\item For each $n\in\Z$, define $d:\text{Tot}^{\prod}(C)_n\to\text{Tot}^{\prod}(C)_{n-1}$ by $$d=d^v+(-1)^nd^h$$
\end{itemize}
\end{defn}

\begin{defn}{Tensor Product Double Complex}{}\\
Let $(\mA,\otimes_\mA)$ be a monoidal abelian category. Let $(C_\bullet,d^C)$ and $(D_\bullet,d^D)$ be two chain complexes. Define the tensor product of $C_\bullet$ and $D_\bullet$ to be the double complex $C_\bullet\otimes D_\bullet$ with the horizontal differential $d^h:C_p\otimes_\mA D_q\to C_{p-1}\otimes_\mA D_q$ and vertical differential $d^v:C_p\otimes_\mA D_q\to C_p\otimes_\mA D_{q-1}$ is defined by $$d^h=d^C\otimes\text{id}_D\;\;\;\;\text{ and }\;\;\;\;d^v=(-1)^p\text{id}_C\otimes d^D$$ respectively. 
\end{defn}

\begin{defn}{Tensor Product Chain Complex}{} Let $\mA$ be an abelian category. Let $(C_\bullet,d^C)$ and $(D_\bullet,d^D)$ be two chain complexes. Define the tensor product chain complex of $C_\bullet$ and $D_\bullet$ to be the chain complex $$(C\otimes D)_\bullet=\text{Tot}^\oplus(C_\bullet\otimes D_\bullet)$$
\end{defn}

Note that this is different from the tensor product double complex of two chain complexes, although they are closely related. Although the tensor product double complex makes more intuitive sense, the canonical product for $\bold{Ch}(\mA)$ is in fact the tensor product chain complex. Moreover, this product is the symmetric monoidal product of $\bold{Ch}(\mA)$.

\subsection{Left and Right Exact Functors}
\begin{defn}{Left and Right Exact Functors}{} Let $\mA,\mB$ be abelian categories. Let $F:\mA\to\mB$ be an additive functor. Let $0\to A\to B\to C\to 0$ be an exact sequence in $\mA$. 
\begin{itemize}
\item We say that $F$ is right exact if the sequence $$F(A)\to F(B)\to F(C)\to 0$$ is exact. 
\item We say that $F$ is left exact if the sequence $$0\to F(A)\to F(B)\to F(C)$$ is exact. 
\end{itemize}
\end{defn}

\begin{lmm}{}{} Let $\mA,\mB$ be abelian categories. Let $F:\mA\to\mB$ be an additive functor. Then $F$ is exact if and only if $F$ is left exact and right exact. 
\end{lmm}

\begin{prp}{}{} Let $\mA,\mB$ be abelian categories. Let $F:\mA\to\mB$ be a functor. Then the following are true. 
\begin{itemize}
\item $F$ is additive and left exact if and only if $F$ preserves all finite limits. 
\item $F$ is additive and right exact if and only if $F$ preserves all finite colimits. 
\end{itemize}
\end{prp}

Since right adjoints preserves limits, any right adjoint functor between abelian categories must be left exact. Moreover, since limits commutes with limits, the limit functor $\lim_\mJ(-):\mC^\mJ\to\mC$ is also left exact. Dually, left adjoints and the colimit functor must also be right exact. 

\begin{prp}{}{} Let $\mA$ be an abelian category. Let $M\in\mA$. Then the following are true. 
\begin{itemize}
\item The covariant functor $\Hom(M,-):\mA\to\bold{Ab}$ is left exact. 
\item The contravariant functor $\Hom(-,M):\mA\to\bold{Ab}$ is right exact. 
\end{itemize}
\end{prp}

Let $\mC$ be a category and let $\mJ$ be a diagram. Recall that if $\mC$ is complete, then the limit can be thought of as a functor $\lim_\mJ:\mC^\mJ\to\mC$. Dually, if $\mC$ is cocomplete, the colimit can be thought of as a functor $\text{colim}_\mJ:\mC^\mJ\to\mC$. If $\mC$ is an abelian category, it makes sense to ask whether these two functors preserve exact sequences. 

\begin{prp}{}{} Let $\mA$ be an abelian category. Let $\mJ$ be a small category. Then the following are true. 
\begin{itemize}
\item If $\mA$ is complete, then the functor $\lim_\mJ:\mC^\mJ\to\mC$ is left exact
\item If $\mA$ is cocomplete, then the functor $\text{colim}_\mJ:\mC^\mJ\to\mC$ is right exact
\end{itemize}
\end{prp}

\begin{thm}{The FHHF Theorem}{}\\
Let $\mA,\mB$ be abelian categories. Let $F:\mA\to\mB$ be a covariant functor. Then the following are true. 
\begin{itemize}
\item If $F$ is right exact, then there is a natural transformation $F\circ H_n\Rightarrow H_n\circ F$ for all $n\in\N$. 
\item If $F$ is left exact, then there is a natural transformation $H_n\circ F\Rightarrow F\circ H_n$ for all $n\in\N$. 
\item If $F$ is exact, then the above natural transformations isomorphisms and are inverses of each other. 
\end{itemize}
\end{thm}

\begin{prp}{}{}\\
Let $\mA$ be an abelian category. Let $\mJ$ be a filtered category. Let $F:\mJ\to\bold{Ch}(\mA)$ be a functor. Then there is a natural isomorphism $$\colim_\mJ H_n\circ F\cong H_n\left(\colim_\mJ F\right)$$ for all $n\in\N$. 
\begin{proof}\\
Since filtered colimits commute with finite limits, we know that the filtered colimit functor is left exact. Hence it is an exact functor. By the FHHF theorem, filtered colimits commute with homology. 
\end{proof}
\end{prp}

\begin{eg}{}{}\\
Since products are coproducts and coproducts are filtered colimits, the above prp shows that homology commutes with products. 
\end{eg}

\subsection{Injective and Projective Objects}
Injectivity and Projectivity objects are created just for the sake of allowing the $\Hom$ functor to be exact. Therefore its definition is also direct. 

\begin{defn}{Projective and Injective Objects}{} Let $\mA$ be an abelian category. Let $Y\in\mA$ be an object. 
\begin{itemize}
\item We say that $Y$ is injective if the functor $\Hom(-,Y):\mA\to\bold{Ab}$ is exact. 
\item We say that $Y$ is projective if the functor $\Hom(Y,-):\mA\to\bold{Ab}$ is exact. 
\end{itemize}
\end{defn}

\begin{lmm}{}{}\\
Let $\mA$ be an abelian category. Then the following are true. 
\begin{itemize}
\item Let $I\in\mA$ be an objects. Then $I$ is injective if and only if for all objects $X,Y\in\mA$ and injective morphisms $f:X\to Y$ and $u:X\to I$, there exists a morphism $g:Y\to I$ such that the following diagram commutes: \\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	X & I \\
	Y
	\arrow["u", from=1-1, to=1-2]
	\arrow["f"', from=1-1, to=2-1]
	\arrow["\exists"', dashed, from=2-1, to=1-2]
\end{tikzcd}}\\~\\
\item Let $P\in\mA$ be an objects. Then $P$ is projective if and only if for all objects $X,Y\in\mA$ and surjective morphisms $f:X\to Y$ and $u:P\to Y$, there exists a morphism $g:P\to X$ such that the following diagram commutes: \\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	& X \\
	I & Y
	\arrow["f", from=1-2, to=2-2]
	\arrow["\exists", dashed, from=2-1, to=1-2]
	\arrow["u"', from=2-1, to=2-2]
\end{tikzcd}}\\~\\
\end{itemize}
\end{lmm}

\begin{lmm}{}{} Every free module is projective. 
\begin{proof}
Let $R$ be a ring and let $F$ be a free $R$-module. Suppose that $F$ has basis $B$. Let $M,N$ be $R$-modules. Suppose that $f:N\to M$ is surjective and there exists an $R$-module homomorphism $g:R^n\to M$. Since $f$ is surjective, for each $b\in B$, we can choose a pre-image for $g(b)$ in $N$ for all $b\in B$. Call it $n_b$. Now define $h:F\to N$ by $b\mapsto n_b$ and then extend it $R$-linearly. Now if $\sum_{b\in B}k_bb\in F$, we have that 
\begin{align*}
(f\circ h)\left(\sum_{b\in B}k_bb\right)&=f\left(\sum_{b\in B}k_bh(b)\right)\\
&=f\left(\sum_{b\in B}k_bn_b\right)\\
&=f\left(\sum_{b\in B}k_bn_b\right)\\
&=\sum_{b\in B}k_bf(n_b)\\
&=\sum_{b\in B}k_bg(b)\\
&=g\left(\sum_{b\in B}k_bb\right)
\end{align*}
so that $f\circ h=g$. Thus $F$ is projective. 
\end{proof}
\end{lmm}

\begin{prp}{}{} Let $R$ be a ring. Let $P$ be an $R$-module. Then the following are equivalent. 
\begin{itemize}
\item $P$ is projective 
\item For every surjective homomorphism $f:N\twoheadrightarrow M$ and every $R$-module homomorphism $g:P\to M$, there exists a module homomorphism $h:P\to N$ such that the following diagram commutes: \\~\\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
& N\arrow[d, "f", twoheadrightarrow]\\
P\arrow[ru, "\exists h", dashed]\arrow[r, "g"'] & M
\end{tikzcd}} \\
\item $P\oplus Q$ is a free $R$-module for some $R$-module $Q$. 
\end{itemize} 
\begin{proof}~\\
\begin{itemize}
\item $(3)\implies(1)$: Suppose that there exists a module $Q$ such that $P\oplus Q$ is free. Let $f:N\to M$ be a surjective $R$-module homomorphism and let $g:P\to M$ be an $R$-module homomorphism. 
\end{itemize}
\end{proof}
\end{prp}

\begin{prp}{}{} Let $R$ be a ring. Let $P_i$ be an $R$-module for each $i\in I$. Then the direct sum $\bigoplus_{i\in I}P_i$ is projective if and only if each $P_i$ is projective. 
\end{prp}

\begin{eg}{}{} Let $k$ be a field. Then every $k$-module is injective. 
\end{eg}

\begin{prp}{Baer's Criterion}{}\\
Let $R$ be a ring. Let $M$ be a left $R$-module. Then $M$ is injective if and only if for every left ideal $I$ of $R$ and $R$-module homomorphism $\phi:I\to M$, there exists a map $\overline{\phi}$ such that the following diagram commutes: \\~\\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	I & M \\
	R
	\arrow["\phi", from=1-1, to=1-2]
	\arrow[hook, from=1-1, to=2-1]
	\arrow["{\exists\overline{\phi}}"', dashed, from=2-1, to=1-2]
\end{tikzcd}} \\
\end{prp}

\begin{defn}{Enough Injectives and Enough Projectives}{} Let $\mA$ be an abelian category. $\mA$ is said to have enough injectives if every object is the subobject of an injective object. $\mA$ is said to have enough projectives if every object is the quotient of an projective object. 
\end{defn}

There are however equivalent definitions from the categorical point of view. 

\subsection{Resolutions of Objects}
There are in general, four types of resolutions. Namely injective resolutions, projective resolutions, free resolutions and acyclic resolutions. We will study all four of them and their relations in this section. 

\begin{defn}{Injective Resolution}{}\\

\subsection{Derived Functors}
\begin{defn}{Right Derived Functors}{}\\
Let $\mA,\mB$ be abelian categories such that $\mA$ has enough injective. Let $F:\mA\to\mB$ be a left exact functor. Define the right derived functors $$R^iF:\mA\to\mB$$ of $F$ for $i\geq 0$ as follows. 
\begin{itemize}
\item On objects, $R^iF(A)=H^i(F(I^\bullet))$ where $d:A\to I^\bullet$ is an injective resolution of $A$
\item On Morphisms, $R^iF(\phi:A\to B)=H^i(F(\phi^\bullet:I^\bullet\to (I')^\bullet))$ where $\phi^\bullet:I^\bullet\to(I')^\bullet$ is an extension of $\phi$ to resolutions. 
\end{itemize}
\end{defn}

\begin{prp}{}{}\\
Let $\mA,\mB$ be abelian categories such that $\mA$ has enough injective. Let $F:\mA\to\mB$ be a left exact functor. Then the following are true. 
\begin{itemize}
\item $R^0F=F$. 
\item The $n$th right derived functor $R^nF$ is an additive functor from $\mA$ to $\mB$. 
\item If $A\in\mA$ is an injective object, then $R^nF(A)=0$ for all $n>0$. 
\end{itemize}
\end{prp}

\begin{prp}{}{}\\
Let $\mA,\mB$ be abelian categories such that $\mA$ has enough injective. Let $F:\mA\to\mB$ be a left exact functor. For any short exact sequence \\~\\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
0\arrow[r] & A\arrow[r] & B\arrow[r] & C\arrow[r] & 0
\end{tikzcd}}\\~\\
there is a canonical long exact sequence \\~\\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	0 & A & B & C & {R^1(A)} & {R^1(B)} & {R^1(C)} & {R^2(A)} & \cdots
	\arrow[from=1-1, to=1-2]
	\arrow[from=1-2, to=1-3]
	\arrow[from=1-3, to=1-4]
	\arrow[from=1-4, to=1-5]
	\arrow[from=1-5, to=1-6]
	\arrow[from=1-6, to=1-7]
	\arrow[from=1-7, to=1-8]
	\arrow[from=1-8, to=1-9]
\end{tikzcd}}\\~\\
\end{prp}

\begin{defn}{Left Derived Functors}{}\\
Let $\mA,\mB$ be abelian categories such that $\mA$ has enough projectives. Let $F:\mA\to\mB$ be a right exact functor. Define the left derived functors $$L_iF:\mA\to\mB$$ of $F$ $i\geq 0$ as follows. 
\begin{itemize}
\item On objects, $L_iF(A)=H_i(F(P^\bullet))$ where $d:P_\bullet\to A$ is an projective resolution of $A$
\item On Morphisms, $L_iF(\phi:A\to B)=L_i(F(\phi_\bullet:P_\bullet\to (P')_\bullet))$ where $\phi_\bullet:P_\bullet\to(P')_\bullet$ is an extension of $\phi$ to resolutions. 
\end{itemize}
\end{defn}

\begin{prp}{}{}\\
Let $\mA,\mB$ be abelian categories such that $\mA$ has enough projectives. Let $F:\mA\to\mB$ be a right exact functor. Then the following are true. 
\begin{itemize}
\item $L_0F=F$. 
\item The $n$th left derived functor $L_nF$ is an additive functor from $\mA$ to $\mB$. 
\item If $A\in\mA$ is a projective object, then $L_nF(A)=0$ for all $n>0$. 
\end{itemize}
\end{prp}

\begin{prp}{}{}\\
Let $\mA,\mB$ be abelian categories such that $\mA$ has enough projectives. Let $F:\mA\to\mB$ be a right exact functor. For any short exact sequence \\~\\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
0\arrow[r] & A\arrow[r] & B\arrow[r] & C\arrow[r] & 0
\end{tikzcd}}\\~\\
there is a canonical long exact sequence \\~\\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	\cdots & {L_2(C)} & {L_1(A)} & {L_1(B)} & {L_1(C)} & A & B & C & 0
	\arrow[from=1-1, to=1-2]
	\arrow[from=1-2, to=1-3]
	\arrow[from=1-3, to=1-4]
	\arrow[from=1-4, to=1-5]
	\arrow[from=1-5, to=1-6]
	\arrow[from=1-6, to=1-7]
	\arrow[from=1-7, to=1-8]
	\arrow[from=1-8, to=1-9]
\end{tikzcd}}\\~\\
\end{prp}

\begin{defn}{The Ext Functor}{}\\
Let $\mA$ be an abelian category. Let $A\in\mA$. 
\begin{itemize}
\item Suppose that $\mA$ has enough injectives. Define the $n$th Ext functor of $A$ to be the right derived functor $$\text{Ext}_\mA^n(A,-)=R^n(\Hom(A,-))$$
\item Suppose that $\mA$ has enough projectives. Define the $n$th Ext functor of $A$ to be the right derived functor $$\text{Ext}_\mA^n(-,A)=R^n(\Hom(-,A))$$
\end{itemize}
\end{defn}

\subsection{$\delta$-Functors}
\begin{defn}{$\delta$-Functors}{} Let $\mA$ and $\mB$ be abelian categories. A homological $\delta$-functor is a collection $\{T_n:\mA\to\mB\;|\;n\in\N\}$ of additive functors such that for any short exact sequence \\~\\
\adjustbox{scale=1.0,center}{\begin{tikzcd}

\begin{itemize}

\adjustbox{scale=1.0,center}{\begin{tikzcd}

	\arrow[from=1-1, to=1-2]

	\arrow["{\delta_n}", from=1-5, to=1-6]
	\arrow[from=1-6, to=1-7]
\end{tikzcd}}\\~\\
\item If there is a morphism of short exact sequences \\~\\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	0 & A & B & C & 0 \\

	\arrow[from=2-2, to=2-3]
	\arrow[from=2-3, to=2-4]
	\arrow[from=2-4, to=2-5]
\end{tikzcd}}\\~\\
the following diagram commutes: \\~\\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	{T_n(C)} & {T_{n-1}(A)} \\

\subsection{The Ext Functor of ${_R}\bold{Mod}$ and Extensions}
\begin{defn}{Extensions}{} Let $R$ be a ring. Let $M,N$ be $R$-modules. An extension of $M$ by $N$ is a short exact sequence \\~\\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
0\arrow[r] & N\arrow[r] & E\arrow[r] & M\arrow[r] & 0
\end{tikzcd}} \\~\\
of $R$-modules. 
\end{defn}

\begin{defn}{Equivalent Extensions}{} Let $R$ be a ring. Let $M,N$ be $R$-modules. We say that two extensions \\~\\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	0 & N & E & M & 0 \\
	0 & N & F & M & 0
	\arrow[from=1-1, to=1-2]
	\arrow[from=1-2, to=1-3]
	\arrow[from=1-3, to=1-4]
	\arrow[from=1-4, to=1-5]
	\arrow[from=2-1, to=2-2]
	\arrow[from=2-2, to=2-3]
	\arrow[from=2-3, to=2-4]
	\arrow[from=2-4, to=2-5]
\end{tikzcd}} \\~\\
are equivalent if there exists an $R$-module homomorphism $\phi:E\to F$ such that the following diagram commutes: \\~\\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	0 & N & E & M & 0 \\
	0 & N & F & M & 0
	\arrow[from=1-1, to=1-2]
	\arrow[from=1-2, to=1-3]
	\arrow["{\text{id}_N}", from=1-2, to=2-2]
	\arrow[from=1-3, to=1-4]
	\arrow["\phi", from=1-3, to=2-3]
	\arrow[from=1-4, to=1-5]
	\arrow["{\text{id}_M}", from=1-4, to=2-4]
	\arrow[from=2-1, to=2-2]
	\arrow[from=2-2, to=2-3]
	\arrow[from=2-3, to=2-4]
	\arrow[from=2-4, to=2-5]
\end{tikzcd}} \\~\\
\end{defn}

\begin{prp}{}{} Let $R$ be a ring. Let $M,N$ be $R$-modules. Suppose that the following two extensions are equivalent: \\~\\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	0 & N & E & M & 0 \\
	0 & N & F & M & 0
	\arrow[from=1-1, to=1-2]
	\arrow[from=1-2, to=1-3]
	\arrow["{\text{id}_N}", from=1-2, to=2-2]
	\arrow[from=1-3, to=1-4]
	\arrow["\phi", from=1-3, to=2-3]
	\arrow[from=1-4, to=1-5]
	\arrow["{\text{id}_M}", from=1-4, to=2-4]
	\arrow[from=2-1, to=2-2]
	\arrow[from=2-2, to=2-3]
	\arrow[from=2-3, to=2-4]
	\arrow[from=2-4, to=2-5]
\end{tikzcd}} \\~\\
Then $\phi$ is an isomorphism. Moreover, equivalent extensions is an equivalence relation. 
\end{prp}

\begin{defn}{Split Extensions}{} Let $R$ be a ring. Let $M,N$ be $R$-modules. We say that an extension splits if it is equivalent to the following extension \\~\\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
0\arrow[r] & N\arrow[r] & N\oplus M\arrow[r] & M\arrow[r] & 0
\end{tikzcd}} \\~\\
of $R$-modules. The above extension is called a trivial extension. 
\end{defn}

\begin{prp}{}{} Let $R$ be a ring. Let $M,N$ be $R$-modules. There is a bijection $$\frac{\{\text{Extensions of }M\text{ by }N\}}{\cong}\;\;\;\;\overset{1:1}{\longleftrightarrow}\;\;\;\;\text{Ext}_R^1(M,N)$$ where $\cong$ means equivalence of extensions. Moreover, the trivial extension corresponds to the zero element of $\text{Ext}_R^1(M,N)$. 
\end{prp}

\begin{prp}{}{} Let $A,B$ be $R$-modules. Then the following are true regarding the Ext group. 
\begin{itemize}
\item $\text{Ext}_R^0(A,B)\cong\Hom_R(A,B)$
\item $\text{Ext}_R^i(A,B)=0$ for all $i>0$ if $A$ is projective or $B$ is injective
\item $\text{Ext}_R^i(A,B)=0$ for all $i\geq 2$ if $A,B$ are $\Z$-modules. 
\end{itemize}
\end{prp}

Let $R$ be a ring. Recall that the tensor product is a bifunctor $$-\otimes_R-:{_R\bold{Mod}}\times\bold{Mod}_R\to\bold{Ab}$$

\begin{defn}{The Tor Functor}{}\\
Let $R$ be a ring. Let $n\in\N$. 
\begin{itemize}
\item Let $B\in\bold{Mod}_R$. Define the $n$th tor functor of $B$ to be $$\text{Tor}_R^n(-,B)=L_n(-\otimes_RB)$$
\item Let $A\in{_R\bold{Mod}}$. Define the $n$th tor functor of $A$ to be $$\text{Tor}_R^n(A,-)=L_n(A\otimes_R-)$$
\end{itemize}
\end{defn}

\begin{prp}{}{}\\
Let $R$ be a ring. Let $A\in{_R\bold{Mod}}$ and $B\in\bold{Mod}_R$. Then there is an isomorphism $$\text{Tor}_i^R(A,B)=L_i(-\otimes_RB)(A)\cong L_i(A\otimes_R-)(B)$$ that is natural in $A$ and $B$ for all $i\in\N$. 
\end{prp}
While projective modules allows the exactness of $\Hom(M,-)$, flat modules allows the exactness of $-\otimes_RM$. 

\begin{defn}{Flat Modules}{} Let $R$ be a ring. An $R$-module $M$ is said to be flat if for every injective linear map $\phi:K\to L$ of $R$-modules, the map $$\phi\otimes\text{id}_M:K\otimes_RM\to L\otimes_RM$$ is also injective. 
\end{defn}

\begin{prp}{}{} Let $R$ be a ring and $M$ an $R$-module. Then $M$ is a flat module if and only if for every short exact sequence $0\to K\to L\to J\to 0$, the sequence $$0\to K\otimes_RM\to L\otimes_RM\to J\otimes_RM\to 0$$ is also exact. 
\end{prp}

\begin{prp}{}{} Let $R$ be a ring. Then the following are true. 
\begin{itemize}
\item Product: If $A$ and $B$ are flat over $R$ then $A\otimes_R B$ is flat over $R$
\item Base Change: Let $S$ be an $R$-algebra ($R\to S$ a ring hom). Then $M\otimes_RS$ is flat over $S$ for any flat $R$-module $M$
\item Transitivity: Let $S$ be an $R$-algebra such that $S$ is flat over $R$. If $C$ is flat over $S$ then $C$ is flat over $R$. 
\end{itemize}
\end{prp}

We have the following inclusion of modules $$\text{Free Modules}\subset\text{Projective Modules}\subset\text{Flat Modules}\subset\text{Torsion Free Modules}$$ It is important to note that the ``duality'' between projective and injective modules is given from the bifunctor $\Hom(-,-)$ while the ``duality'' given between projective and flat modules is given by the tensor-hom adjunction.

\begin{defn}{Divisible Groups}{} Let $A$ be an abelian group. Let $n\in\N\setminus\{0\}$. We say that $A$ is $n$-divisible if for all $g\in A$, there exists $h\in A$ such that $nh=g$.

\begin{prp}{}{} Let $A$ be an abelian group. Let $n\in\N\setminus\{0\}$. Then the following are equivalent. 
\begin{itemize}
\item $A$ is $n$-divisible. 
\item $nA=A$. 
\item $A$ is an injective $\Z$-module. 
\end{itemize}
\end{prp}

\begin{defn}{The Prüfer p-Group}{} Let $p$ be a prime. Define the Prüfer p-Group to be the quotient group $$\Z(p^\infty)=\frac{\Z[1/p]}{\Z}$$
\end{defn}

\begin{eg}{}{} The following are divisible groups. 
\begin{itemize}
\item $\Q$. 
\item $\Z[1/p]/\Z$. 
\end{itemize}
\end{eg}

\begin{thm}{Structure Theorem for Divisible Groups}{} Let $A$ be an abelian group. If $A$ is divisible, then there is an isomorphism $$A\cong\left(\bigoplus_{p\text{ is prime}}\frac{\Z[1/p]}{\Z}^{k_p}\right)\oplus\Q^I$$ for some choice of indexing sets $I$ and some $k_p\in\N$. 
\end{thm}

\subsection{Koszul Complexes}
\begin{defn}{Koszul Complex}{}\\
Let $R$ be a ring. Let $M$ be a left $R$-module. Let $\varphi:M\to R$ be an $R$-module homomorphism. Define the Koszul complex $K_\bullet(\varphi)$ of $\varphi$ to be the chain complex given as follows. 
\begin{itemize}
\item For each $n\in\N$, define $$K_n(\varphi)=\bigwedge_{i=1}^nM$$ and $K_n(\varphi)=0$ for $n<0$. 
\item Define the differential $d:K_n(M)\to K_{n-1}(M)$ by $$d(m_1\wedge\cdots\wedge m_n)=\sum_{i=1}^n(-1)^{i+1}\varphi(m_i)m_1\wedge\cdots\wedge\widehat{m_i}\wedge\cdots\wedge m_n$$ for $m_1,\dots,m_n\in M$ and then extend it $R$-linearly. 
\end{itemize}
\end{defn}

When $M$ is a finitely generated free $R$-module, it suffices to see where a set of $R$-basis is mapped to. 

\begin{defn}{Koszul Complex on Central Elements}{}\\
Let $R$ be a ring. Let $x_1,\dots,x_n\in Z(R)$ be central elements. Define the Koszul complex on $x_1,\dots,x_n$ to be $$K_\bullet(x_1,\dots,x_n)=K\left((r_1,\dots,r_n)\in R^n\mapsto\sum_{i=1}^nr_ix_i\in R\right)$$
\end{defn}

Let $R$ be a ring. Let $x_1,\dots,x_n\in R$ be central elements. Let $e_1,\dots,e_n$ be an $R$-basis for $R^n$. Then we may describe the differential on the $R$-basis of $\Lambda^k(R^n)$ as follows: $$d(e_{i_1}\wedge\cdots\wedge e_{i_k})=\sum_{j=1}^k(-1)^{j+1}x_je_{i_1}\wedge\cdots\wedge\widehat{e_{i_j}}\wedge\cdots\wedge e_{i_k}$$

\begin{eg}{}{}\\
Let $R$ be a ring. Let $x\in Z(R)$. Then the Koszul complex $K_\bullet(x)$ is the chain complex: \\~\\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	0 & R & R & 0
	\arrow[from=1-1, to=1-2]
	\arrow["x", from=1-2, to=1-3]
	\arrow[from=1-3, to=1-4]
\end{tikzcd}} \\~\\
\end{eg}

\begin{lmm}{}{}\\
Let $R$ be a ring. Let $x,y\in Z(R)$ be central elements. Then we have $$K_\bullet(x,y)=\text{Tot}^\oplus\left(K_\bullet(x)\otimes_R K_\bullet(y)\right)$$
\end{lmm}

\begin{lmm}{}{}\\
Let $R$ be a commutative ring. Let $x_1,\dots,x_n\in Z(R)$ be central elements. Let $A=(a_{ij})\in GL_n(R)$ be an invertible matrix. Then there is an isomorphism $$K_\bullet(x_1,\dots,x_n)\cong K_\bullet\left(\sum_{i=1}^na_{1i}f_i,\dots,\sum_{i=1}^na_{ni}f_i\right)$$
\end{lmm}

For an example, let $R$ be a commutative ring. Let $x,y\in R$. Then the Koszul complex $K(x,y)$ is given by \\~\\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	0 & {R^2\wedge R^2} & {R^2} & R & 0
	\arrow[from=1-1, to=1-2]
	\arrow[from=1-2, to=1-3]
	\arrow[from=1-3, to=1-4]
	\arrow[from=1-4, to=1-5]
\end{tikzcd}}\\~\\
The differentials are given as follows. 
\begin{itemize}
\item The first differential $R^2\to R$ is given by $(r,s)\mapsto rx+sy$. It can also be given as a $1\times 2$ matrix as $\begin{pmatrix}x & y\end{pmatrix}$. Also alternatively, we can write an $R$-basis for $R^2$ with $(1,0)$ and $(0,1)$. Then define the map $R^2\to R$ by $(1,0)\mapsto x$ and $(0,1)\mapsto y$. 
\item Now recall that $R^2\wedge R^2$ has $R$-basis given by $(1,0)\wedge(0,1)$ and so there is an isomorphism $R^2\wedge R^2\cong R$. Then the second differential $R\to R^2$ is given by $1\mapsto(x,-y)$. 
\end{itemize}

\begin{defn}{Koszul (Co)Homology}{} Let $R$ be a ring. Let $x_1,\dots,x_n\in Z(R)$ be central elements. Let $M$ be an $R$-module. Define the Koszul homology of $M$ with respect to $x_1,\dots,x_n$ by $$H_k^\text{Kos}(x_1,\dots,x_n;M)=H_k(K(x_1,\dots,x_n;M))$$ Define the Koszul cohomology of $M$ with respect to the central elements by $$H_\text{Kos}^k(x_1,\dots,x_n;M)=H^k(\Hom_R(K(x_1,\dots,x_n)),M)$$
\end{defn}

\begin{lmm}{}{} Let $R$ be a ring. Let $x_1,\dots,x_n\in Z(R)$ be central elements. Let $M$ be an $R$-module. Then the following are true. 
\begin{itemize}
\item $H_0^\text{Kos}(x_1,\dots,x_n;M)=\frac{M}{(x_1,\dots,x_n)M}$
\item $H_\text{Kos}^0(x_1,\dots,x_n;M)=\{m\in M\;|\;x_im=0\text{ for all }i\}$
\end{itemize}
\end{lmm}

\begin{thm}{Kunneth Theorem}{} Let $R$ be a ring. Let $x_1,\dots,x_n\in R$ be central elements. Let $C_\bullet$ be a chain complex of $R$-modules. Then there is an exact sequence given by \\~\\
\adjustbox{scale=0.9,center}{\begin{tikzcd}
	0 & {H_0(x_1,\dots,x_n;H_q(C_\bullet))} & {H_q^\text{Tot}(K(x_1,\dots,x_n)\otimes_RC_\bullet)} & {H_1(x_1,\dots,x_n;H_{q-1}(C_\bullet))} & 0
	\arrow[from=1-1, to=1-2]
	\arrow[from=1-2, to=1-3]
	\arrow[from=1-3, to=1-4]
	\arrow[from=1-4, to=1-5]
\end{tikzcd}}
\end{thm}

\begin{thm}{Self Duality of the Koszul Complex}{} Let $R$ be a ring. Let $x_1,\dots,x_n\in Z(R)$ be central elements. Let $M$ be a left $R$-module. Then there is a natural isomorphism of chain complexes $$\Hom(K(x_1,\dots,x_n);M)\cong K(x_1,\dots,x_n;M)$$
\end{thm}

The self duality of the Koszul complex means that we could have equally defined the differential to be going up the chain instead of down. In this case, the complex can be defined as \\~\\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	0 & {\bigwedge_{i=1}^nR^n} & {\bigwedge_{i=1}^{n-1}R^n} & \cdots & {R^n} & R & 0
	\arrow[from=1-1, to=1-2]
	\arrow["{d_n}", from=1-2, to=1-3]
	\arrow[from=1-3, to=1-4]
	\arrow[from=1-4, to=1-5]
	\arrow["{d_1}", from=1-5, to=1-6]
	\arrow[from=1-6, to=1-7]
\end{tikzcd}}\\~\\
where the differential $d_k:\bigwedge_{i=1}^{k-1}R^n\to\bigwedge_{i=1}^kR^n$ is now defined by $$d(e_{i_1}\wedge\cdots\wedge e_{i_{k-1}})=\sum_{j=1}^nx_je_j\wedge e_{i_1}\wedge\cdots\wedge e_{i_{k-1}}$$

\begin{crl}{}{} Let $R$ be a ring. Let $x_1,\dots,x_n\in Z(R)$ be central elements. Let $M$ be a left $R$-module. Then there is an isomorphism $$H_p^\text{Kos}(x_1,\dots,x_n;M)\cong H_{\text{Kos}}^{n-p}(x_1,\dots,x_n;M)$$
\end{crl}

\section{Spectral Sequences}
\subsection{General Spectral Sequences}
\begin{defn}{Homological Spectral Sequences}{}\\
Let $\mA$ be an abelian category. Let $r_0\in\N$. A homological spectral sequence $(E_{\bullet,\bullet}^r,d_{\bullet,\bullet}^r)_{r\geq r_0}$ starting on the $r_0$th page consists of the following data.  
\begin{itemize}
\item For each $r\geq r_0$, a $(\Z,\Z)$-bigraded object $E_{\bullet,\bullet}^r$ called the $r$th page. 
\item For all $p,q\in\Z$ and $r\geq r_0$, a morphism $$d_{p,q}^r:E_{p,q}^r\to E_{p-r,q+r-1}^r$$ such that $d^r\circ d^r=0$ called the differential. 
\item For each $p,q\in\Z$ and $r\geq r_0$, an isomorphism $$E_{p,q}^{r+1}=\frac{\ker(d^r:E_{p,q}^r\to E_{p-r,q+r-1}^r)}{\im(d^r:E_{p+r,q-r+1}^r\to E_{p,q}^r)}$$
\end{itemize}
We say that the total degree of $E_{p,q}^r$ is $n=p+q$. 
\end{defn}

\begin{defn}{Cycle and Boundary Notation}{}\\
Let $\mA$ be an abelian category. Let $(E_{\bullet,\bullet}^r,d^r)_{r\geq r_0}$ be a spectral sequence. 
\begin{itemize}
\item For each $p,q\in\Z$, define $Z_{p,q}^r=\ker(d:E_{p,q}^r\to 
E_{p-r,q+r-1}^r)$. 
\item For each $p,q\in\Z$, define $B_{p,q}^r=\im(d:E_{p+r,q-r+1}^r\to E_{p,q}^r)$. 
\end{itemize}
\end{defn}

\begin{defn}{Cohomological Spectral Sequences}{} Let $\mA$ be an abelian category. A cohomological spectral sequence consists of the following data.  
\begin{itemize}
\item A collection of objects $E_r^{\bullet,\bullet}=\{E_r^{p,q}\in\mA\;|\;p,q\in\Z\}$ called pages for each $r\in\N$. So that there is a sequence $$E_1^{\bullet,\bullet},E_2^{\bullet,\bullet},E_3^{\bullet,\bullet},\dots$$ of family of objects
\item A degree $(p,q)$ map $$d_r^{p,q}:E_r^{p,q}\to E_r^{p-r,q+r-1}$$ for each $p,q\in\Z$ and $r\in\N$ such that $d_r\circ d_r=0$
\item Isomorphisms of the form $E_{r+1}^{\bullet,\bullet}=H^\bullet(E_r^{\bullet,\bullet},d_r)$. This means that $$E_{r+1}^{p,q}=\frac{\ker(d_r:E_r^{p,q}\to E_r^{p-r,q+r-1})}{\im(d_r:E_r^{p+r,q-r+1}\to E_r^{p,q})}$$
\end{itemize}
\end{defn}

Notice that cohomological spectral sequences are really the same thing as homological spectral sequences, just reindex the objects by $E_r^{p,q}=E_{-p,-q}^r$. 

\begin{defn}{The Infinity Terms}{}\\
Let $\mA$ be an abelian category. Let $(E_{\bullet,\bullet}^r,d^r)_{r\geq r_0}$ be a spectral sequence. 
\begin{itemize}
\item For each $p,q\in\Z$, define $$Z_{p,q}^\infty=\text{lim}_{r\to\infty}Z_{p,q}^r$$ if the limit exists. 
\item For each $p,q\in\Z$, define $$B_{p,q}^\infty=\text{colim}_{r\to\infty}B_{p,q}^r$$ if the colimit exists. 
\item For each $p,q\in\Z$, define the infinity term to be $$E_{p,q}^\infty=\text{coker}(B_{p,q}^\infty\to Z_{p,q}^\infty)$$ if the colimit and limit exists. 
\end{itemize}
\end{defn}

If $\mA$ is a concrete category, then the maps $Z_{p,q}^{r+1}\to Z_{p,q}^r$ and $B_{p,q}^r\to B_{p,q}^{r+1}$ are inclusion maps so that they are subsets of $E_{p,q}^{r_0}$. Then the limit and colimit evaluates to be the intersection and union respectively. Hence $$E_{p,q}^\infty=\frac{\bigcap_{r=r_0}^\infty Z_{p,q}^r}{\bigcup_{r=r_0}^\infty B_{p,q}^r}$$

\subsection{Stability of Spectral Sequences}
\begin{defn}{Stable Spectral Sequences}{}\\
Let $\mA$ be an abelian category. Let $(E_{\bullet,\bullet}^r,d^r)_{r\geq r_0}$ be a spectral sequence. We say that the spectral sequence is stable at $s\geq r_0$ if there is an isomorphism $$E_{p,q}^\infty\cong E_{p,q}^s$$ for all $p,q\in\Z$. 
\end{defn}

\begin{defn}{Bounded Spectral Sequences}{}\\
Let $\mA$ be an abelian category. Let $(E_{\bullet,\bullet}^r,d^r)_{r\geq r_0}$ be a spectral sequence. 
\begin{itemize}
\item We say that the spectral sequence is bounded if for each $n\in\Z$, there are only finitely many non-zero terms of total degree $n$ in $E^r$ for all $r\in\N^+$. 
\item We say that the spectral sequence is first quadrant if $E_{p,q}^r=0$ for all $p<0$ or $q<0$. 
\end{itemize}
\end{defn}

Clearly if a homological spectral sequence is first quadrant, then it is bounded. 

\begin{lmm}{}{}\\
Let $(E_{\bullet,\bullet}^r,d^r)$ be a bounded homological spectral sequence. Then the following are true. 
\begin{itemize}
\item For each $(p,q)\in\Z^2$, there exists $s\in\N$ such that $E_{p,q}^r\cong E_{p,q}^s$ for all $r\geq s$. 
\item The spectral sequence is stable. 
\end{itemize}
\end{lmm}

\begin{defn}{Collapsing Spectral Sequences}{}\\
Let $\mA$ be an abelian category. Let $(E_{\bullet,\bullet}^r,d^r)_{r\geq r_0}$ be a spectral sequence. We say that it collapses at $s\geq r_0\geq 2$ if there is exactly one non-zero row or column in $E_{\bullet,\bullet}^s$. 
\end{defn}

\begin{lmm}{}{}\\
Let $\mA$ be an abelian category. Let $(E_{\bullet,\bullet}^r,d^r)_{r\geq r_0}$ be a spectral sequence. If the spectral sequence collapses at $s\geq r_0$, then it is stable at $s$. 
\end{lmm}

\subsection{Convergence}
\begin{defn}{Weakly Convergent}{}\\
Let $\mA$ be an abelian category. Let $(E_{\bullet,\bullet}^r,d^r)_{r\geq r_0}$ be a spectral sequence in $\mA$. Let $H_\bullet$ be a $\Z$-graded object in $\mA$. Let $(F_mH_n)_{m\geq r_0}$ be a filtration for $H_n$ for each $n$. 
\begin{itemize}
\item We say that the spectral sequence weakly converges to $H_\bullet$ if the filtrations are exhaustive and there are isomorphisms $$E_{p,q}^\infty\cong\frac{F_pH_{p+q}}{F_{p-1}H_{p+q}}$$ We denote it by $E_{p,q}^r\Rightarrow H_\bullet$. 
\item We say that the spectral sequence converges to $H_\bullet$ if it is weakly convergent to $H_\bullet$ and the filtration is Hausdorff. 
\end{itemize}
\end{defn}

\subsection{The Spectral Sequence of Exact Couples}
\begin{defn}{Exact Couple}{}\\
Let $\mA$ be an abelian category. An exact couple in $\mA$ consists of the following: 
\begin{itemize}
\item Two objects $A,E\in\mA$. 
\item Morphisms $i:A\to A$ and $j:A\to E$ and $k:E\to A$ in $\mA$. 
\end{itemize}
such that the following diagram commutes and is exact: \\~\\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	A && A \\
	& E
	\arrow["i", from=1-1, to=1-3]
	\arrow["j", from=1-3, to=2-2]
	\arrow["k", from=2-2, to=1-1]
\end{tikzcd}}\\~\\
We denote the exact couple by $(A,E,i,j,k)$. Also define the differential of the exact couple to be $d=j\circ k$. 
\end{defn}

Notice that this actually just the data of a long exact sequence. If we look at what happens nearby $E_{p,q}$, we see that we can expand out the triangle ad infinum: \\~\\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	\cdots & {A_{p-a,q-a'}} & {E_{p,q}} & {A_{p+a,q+a'}} & {A_{p+a+b,q+a'+b'}} & \cdots
	\arrow[from=1-1, to=1-2]
	\arrow[from=1-2, to=1-3]
	\arrow[from=1-3, to=1-4]
	\arrow[from=1-4, to=1-5]
	\arrow[from=1-5, to=1-6]
\end{tikzcd}}\\~\\

\begin{defn}{Derived Couple}{}\\
Let $\mA$ be an abelian category. Let $(A,E,i,j,k)$ be an exact couple. Define the derived couple $(A',E',i',j',k')$ of the exact couple by the following data. 
\begin{itemize}
\item Define $A'=\im(i)$. 
\item Define $E'=\text{coker}(\im(d)\to\ker(d))$. 
\item Let the following be the unique factorization of $i$ through $\im(i)$: \\~\\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	A && A \\
	& {\im(i)}
	\arrow["i", from=1-1, to=1-3]
	\arrow["{\exists!b}"', dashed, two heads, from=1-1, to=2-2]
	\arrow["{\exists!a}"', dashed, tail, from=2-2, to=1-3]
\end{tikzcd}}\\
Define $i'=b\circ a$. 
\item Define $j':A'\to E'$ to be the connecting homomorphism in the snake lemma of the following diagram: \\~\\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	& {\im(d)} & A & {A'} & 0 \\
	0 & {\ker(d)} & E & E
	\arrow["k", from=1-2, to=1-3]
	\arrow["{\exists!}"', dashed, from=1-2, to=2-2]
	\arrow["i", from=1-3, to=1-4]
	\arrow["j"', from=1-3, to=2-3]
	\arrow[from=1-4, to=1-5]
	\arrow["0", from=1-4, to=2-4]
	\arrow[from=2-1, to=2-2]
	\arrow[from=2-2, to=2-3]
	\arrow["d", from=2-3, to=2-4]
\end{tikzcd}}\\
(In terms of elements this is given as follows: For $x\in A'$ suppose that $i(y)=x$ for some $y\in A$. Then define $j'(x)=[j(y)]$). 
\item Define $k':E'\to A'$ to be the connecting homomorphism in the snake lemma of the following the diagram: \\~\\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	& E & E & {\text{coker}(d)} & 0 \\
	0 & {\ker(j)} & A & E
	\arrow["d", from=1-2, to=1-3]
	\arrow["0"', from=1-2, to=2-2]
	\arrow[from=1-3, to=1-4]
	\arrow["k"', from=1-3, to=2-3]
	\arrow["d"', from=1-3, to=2-4]
	\arrow[from=1-4, to=1-5]
	\arrow["{\exists!}"', dashed, from=1-4, to=2-4]
	\arrow[from=2-1, to=2-2]
	\arrow[from=2-2, to=2-3]
	\arrow["j"', from=2-3, to=2-4]
\end{tikzcd}}\\
(In terms of elements this is given as follows: $k'([x])=k(x)$). 
\end{itemize}
\end{defn}

\begin{lmm}{}{}\\
Let $\mA$ be an abelian group. Let $(A,E,i,j,k)$ be an exact couple. Then $(A',E',i',j',k')$ is an exact couple. 
\end{lmm}

\begin{defn}{The $n$th Derived Couple}{}\\
Let $\mA$ be an abelian group. Let $(A,E,i,j,k)$ be an exact couple. Define the $n$th derived couple $(A^{(n)},E^{(n)},i^{(n)},j^{(n)},k^{(n)})$ inductively to be the derived couple of the $(n-1)$th derived couple. 
\end{defn}

\begin{prp}{}{}\\
Let $\mA$ be an abelian group. Let $a\in\N$. Let $(A_{\bullet,\bullet},E_{\bullet,\bullet},i,j,k)$ be an exact couple in $\text{Gr}_{\Z^2}(\mA)$ such that $\deg(i)=(1,-1)$ and $\deg(j)=(-a,a)$ and $\deg(k)=(-1,0)$. Then the following are true. 
\begin{itemize}
\item $\deg(i')=(1,-1)$. 
\item $\deg(j')=(-a-1,a+1)$. 
\item $\deg(k')=(-1,0)$. 
\item $\left(E_{\bullet,\bullet}^{(n)},j^{(n)}\circ k^{(n)}\right)_{n\geq a}$ is a homological spectral sequence. 
\end{itemize}
\end{prp}

\subsection{The Spectral Sequence of Filtrations}
\begin{lmm}{}{}\\
Let $\mA$ be an abelian category. Let $C_\bullet\in\bold{Ch}(\mA)$ be a chain complex. Let $(F_pC_\bullet)_{p\in\N^+}$ be an ascending filtration of $C_\bullet$. Consider the following data: 
\begin{itemize}
\item $D_{p,q}=H_{p+q}(F_pC_\bullet)$. 
\item $E_{p,q}=H_{p+q}\left(\frac{F_pC_\bullet}{F_{p-1}C_\bullet}\right)$. 
\item The short exact sequence of chain complexes $0\to F_{p-1}C_\bullet\to F_pC_\bullet\to\frac{F_pC_\bullet}{F_{p-1}C_\bullet}$ induces a long exact sequence: \\~\\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	\cdots & {H_n(F_{p-1}C_\bullet)} & {H_n(F_pC_\bullet)} & {H_n\left(\frac{F_pC_\bullet}{F_{p-1}C_\bullet}\right)} & {H_{n-1}(F_{p-1}C_\bullet)} & \cdots
	\arrow[from=1-1, to=1-2]
	\arrow["i", from=1-2, to=1-3]
	\arrow["j", from=1-3, to=1-4]
	\arrow["k", from=1-4, to=1-5]
	\arrow[from=1-5, to=1-6]
\end{tikzcd}}\\~\\
\end{itemize}
Then $(D_{\bullet,\bullet},E_{\bullet,\bullet},i,j,k)$ is an exact couple. 
\end{lmm}

\begin{defn}{Spectral Sequence of a Filtered Chain Complex}{}\\
Let $\mA$ be an abelian category. Let $C_\bullet\in\bold{Ch}(\mA)$ be a chain complex. Let $(F_pC_\bullet)_{p\in\N^+}$ be an ascending filtration of $C_\bullet$. Define the spectral sequence of the filtered chain complex as follows. 
\begin{itemize}
\item Define $E_{p,q}^0=\left(\frac{F_pC_{p+q}}{F_{p-1}C_{p+q}}\right)$. 
\item For $n\in\N^+$, define $E_{\bullet,\bullet}^n$ to be the spectral sequence associated to the exact couple $(D_{\bullet,\bullet},E_{\bullet,\bullet},i,j,k)$ given above. 
\end{itemize}
\end{defn}

There is a more explicit way of constructing the associated spectral sequence. 

\begin{defn}{Spectral Sequence Arising from Filtered Chain Complexes}{} Let $\mA$ be an abelian category. Let $(C_\bullet,d_\bullet)\in\bold{Ch}(\mA)$ be a chain complex. Let $F_pC$ be a filtered chain complex that is exhaustive. Define the following objects and subobjects in $\mA$. 
\begin{itemize}
\item Define an object $E_{p,q}^0=\frac{F_pC_{p+q}}{F_{p-1}C_{p+q}}$ and a chain complex $E_p^0=\frac{F_pC}{F_{p-1}C}$. 
\item For each $p,r\in\N$, define $$A_p^r=\{c\in F_pC\;|\;d(c)\in F_{p-r}C\}$$ called approximately cycles
\item Write $\eta_p:F_pC\to\frac{F_pC}{F_{p-1}C}=E_p^0$ for the sujection, which is a chain map. 
\item For each $p,r\in\N$, define $$Z_p^r=\eta_p(A_p^r)\subseteq E_p^0$$
\item For each $p,r\in\N$, define $$B_{p-r}^{r+1}=\eta_{p-r}(d(A_p^r))\subseteq E_{p-r}^0$$
\item For each $p\in\N$, define $$Z_p^\infty=\bigcap_{r=1}^\infty Z_p^r\;\;\;\;\text{ and }\;\;\;\; B_p^\infty=\bigcup_{r=1}^\infty B_p^r$$
\end{itemize}
Evidently, there is a tower of subobjects of $E_p^0$ given by $$0=B_p^0\subseteq B_p^1\subseteq\cdots B_p^\infty\subseteq Z_p^\infty\subseteq\cdots Z_p^1\subseteq Z_p^0=E_p^0$$ Thus we finally define $$E_p^r=\frac{Z_p^r}{B_p^r}\;\;\;\text{ for all }r\in R\;\;\;\;\text{ and }\;\;\;\;E_p^\infty=\frac{Z_p^\infty}{B_p^\infty}$$ together with $d:E_p^r\to E_{p-r}^r$ to be the differential induced by the differential of $C$. 
\end{defn}

\begin{lmm}{}{} Let $\mA$ be an abelian category. Let $(C_\bullet,d_\bullet)\in\bold{Ch}(\mA)$ be a chain complex. Let $F_pC$ be a filtered chain complex that is exhaustive. Then the following are true. 
\begin{itemize}
\item For any $p,r\in\N$, $A_p^r+F_{p-1}C=A_{p-1}^{r-1}$
\item For any $p,r\in\N$, $Z_p^r\cong A_p^r/A_{p-1}^{r-1}$
\item For any $p,r\in\N$, there are isomorphisms $$E_p^r=\frac{Z_p^r}{B_p^r}\cong\frac{A_p^r+F_{p-1}C}{d(A_{p+r-1}^{r-1})+F_{p-1}C}\cong\frac{A_p^r}{d(A_{p+r-1}^{r-1})+A_{p-1}^{r-1}}$$
\end{itemize}
\end{lmm}

\begin{thm}{}{} Let $\mA$ be an abelian category. Let $(C_\bullet,d_\bullet)\in\bold{Ch}(\mA)$ be a chain complex. Let $F_pC$ be a filtered chain complex that is exhaustive. Then the map $d:E_p^r\to E_{p-r}^r$ determines an isomorphism $$\frac{Z_p^r}{Z_p^{r+1}}\cong\frac{B_{p-r}^{r+1}}{B_{p-r}^r}$$ Moreover, this concludes that $(E_{p,q}^r,d)$ is a spectral sequence. 
\end{thm}

\begin{defn}{Cartan-Eilenberg Resolution}{} Let $\mA$ be an abelian category with enough projectives. Let $A_\bullet\in\bold{Ch}(\mA)$ be a chain complex as follows: \\~\\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	\cdots & {A_{p-1}} & {A_p} & {A_{p+1}} & \cdots
	\arrow[from=1-1, to=1-2]
	\arrow[from=1-2, to=1-3]
	\arrow[from=1-3, to=1-4]
	\arrow[from=1-4, to=1-5]
\end{tikzcd}}\\~\\
A Cartan-Eilenberg resolution of $A_\bullet$ is an upper half double complex $P_{\bullet,\bullet}$ together with augmentation maps $\varepsilon:P_{P,\bullet}\to A_p$ as follows: \\~\\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	& \vdots & \vdots & \vdots \\
	\cdots & {P_{p-1,1}} & {P_{p,1}} & {P_{p+1,1}} & \cdots \\
	\cdots & {P_{p-1,0}} & {P_{p,0}} & {P_{p+1,0}} & \cdots \\
	\cdots & {A_{p-1}} & {A_p} & {A_{p+1}} & \cdots
	\arrow[from=1-2, to=2-2]
	\arrow[from=1-3, to=2-3]
	\arrow[from=1-4, to=2-4]
	\arrow[from=2-2, to=2-1]
	\arrow[from=2-2, to=3-2]
	\arrow[from=2-3, to=2-2]
	\arrow[from=2-3, to=3-3]
	\arrow[from=2-4, to=2-3]
	\arrow[from=2-4, to=3-4]
	\arrow[from=2-5, to=2-4]
	\arrow[from=3-2, to=3-1]
	\arrow["\varepsilon", from=3-2, to=4-2]
	\arrow[from=3-3, to=3-2]
	\arrow["\varepsilon", from=3-3, to=4-3]
	\arrow[from=3-4, to=3-3]
	\arrow["\varepsilon", from=3-4, to=4-4]
	\arrow[from=3-5, to=3-4]
	\arrow[from=4-2, to=4-1]
	\arrow[from=4-3, to=4-2]
	\arrow[from=4-4, to=4-3]
	\arrow[from=4-5, to=4-4]
\end{tikzcd}}\\~\\
such that the following are true. 
\begin{itemize}
\item If $A_p=0$ then $P_{p,\bullet}=0$
\item For each $p\in\Z$, the chain complex $(B_\bullet,d^v)$ of boundaries where $B_q=d^h(P_{p+1,q})$ form a projective resolution of $A_p$. This means that the vertical chain complexes \\~\\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	\vdots \\
	{d^h(P_{p,0})\subseteq P_{p,1}} \\
	{d^h(P_{p+1,0})\subseteq P_{p,0}} \\
	{A_p}
	\arrow[from=1-1, to=2-1]
	\arrow[from=2-1, to=3-1]
	\arrow["\varepsilon", from=3-1, to=4-1]
\end{tikzcd}}\\~\\
are projective resolutions. 
\item For each $p\in\Z$, the chain complex $(H_\bullet,d^v)$ of the homology groups where $$H_q=H_p(P_{\bullet,q},d^h)$$ form a projective resolution of $H_p(A_\bullet)$. This means that the vertical chain complexes \\~\\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	\vdots \\
	{H_p(P_{\bullet,1},d^h)} \\
	{H_p(P_{\bullet,0},d^h)} \\
	{H_p(A_\bullet)}
	\arrow[from=1-1, to=2-1]
	\arrow["{d^v}", from=2-1, to=3-1]
	\arrow["\varepsilon", from=3-1, to=4-1]
\end{tikzcd}}\\~\\
are projective resolutions. 
\end{itemize}
\end{defn}

\begin{thm}{}{} Let $\mA$ be an abelian category. Then every chain complex $A_\bullet\in\bold{Ch}(\mA)$ has a Cartan-Eilenberg resolution $P_{\bullet,\bullet}\to A_\bullet$. 
\end{thm}

\begin{defn}{Left Hyperderived Functors}{} Let $\mA,\mB$ be abelian categories. Let $F:\mA\to\mB$ be a right exact functor. Define the left hyperderived functors $$\L_i F:\bold{Ch}(\mA)\to\mB$$ by the following. 
\begin{itemize}
\item For $A_\bullet$ a chain complex in $\mA$, choose a Cartan-Eilenberg resolution $P_{\bullet,\bullet}\to A_\bullet$ of $A_\bullet$. Then $$\L_i F(A_\bullet)=H_i(\text{Tot}^\oplus(F(P_{\bullet,\bullet})))$$
\item For $f:A_\bullet\to B_\bullet$ a chain map, choose Cartan-Eilenberg resolutions for both complexes and consider the induced map $\overline{f}:P_{\bullet,\bullet}\to Q_{\bullet,\bullet}$. Define $$\L_i F(f)=H_i(\text{Tot}^\oplus(\overline{f})):\L_iF(A_\bullet)\to\L_iF(B_\bullet)$$
\end{itemize}
\end{defn}


\begin{defn}{Right Hyperderived Functors}{} Let $\mA,\mB$ be abelian categories. Let $F:\mA\to\mB$ be a left exact functor. Define the right hyperderived functors $$\R^i F:\bold{CCh}(\mA)\to\mB$$ by the following. 
\begin{itemize}
\item For $A_\bullet$ a cochain complex in $\mA$, choose a Cartan-Eilenberg resolution $A_\bullet\to I_{\bullet,\bullet}$ of $A_\bullet$. Then $$\R^i F(A_\bullet)=H^i(\text{Tot}^\oplus(F(I_{\bullet,\bullet})))$$
\item For $f:A_\bullet\to B_\bullet$ a chain map, choose Cartan-Eilenberg resolutions for both complexes and consider the induced map $\overline{f}:I_{\bullet,\bullet}\to J_{\bullet,\bullet}$. Define $$\R^i F(f)=H^i(\text{Tot}^\oplus(\overline{f})):\R^iF(A_\bullet)\to\R^iF(B_\bullet)$$
\end{itemize}
\end{defn}

\begin{lmm}{}{} Let $\mA$ and $\mB$ be abelian categories. Let $X\in\mA$ be an object. Then the following are true. 
\begin{itemize}
\item Let $F:\mA\to\mB$ be right exact. By considering $X$ as a chain complex with only non-zero term in degree $0$, the hyperderived left functor $\L_i F(X)=L_iF(X)$ is the same as the usual left derived functor. 
\item Let $F:\mA\to\mB$ be left exact. By considering $X$ as a cochain complex with only non-zero term in degree $0$, the hyperderived right functor $\R^iF(X)=R^iF(X)$ is the same as the usual right derived functor. 
\end{itemize}
\end{lmm}

\begin{lmm}{}{} Let $\mA$ and $\mB$ be abelian categories. Then the following are true. 
\begin{itemize}
\item Let $F:\mA\to\mB$ be right exact. Then the restriction of $\L_i F:\bold{Ch}(\mA)\to\mB$ to $\bold{Ch}^+(\mA)$ are precisely the left derived functor $L_i (H_0F):\bold{Ch}^+(\mA)\to\mB$ of the functor $H_0F:\bold{Ch}(\mA)^+\to\mB$. 
\item Let $F:\mA\to\mB$ be left exact. Then the restriction of $\R^i F:\bold{CCh}(\mA)\to\mB$ to $\bold{CCh}^+(\mA)$ are precisely the left derived functor $R^i (H^0F):\bold{CCh}^+(\mA)\to\mB$ of the functor $H^0F:\bold{CCh}(\mA)^+\to\mB$. 
\end{itemize}
\end{lmm}

\begin{lmm}{}{} Let $\mA$ and $\mB$ be abelian categories. Let the following \\~\\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	0 & A & B & C & 0
	\arrow[from=1-1, to=1-2]
	\arrow[from=1-2, to=1-3]
	\arrow[from=1-3, to=1-4]
	\arrow[from=1-4, to=1-5]
\end{tikzcd}}\\~\\
be a short exact sequence of chain complexes in $\bold{Ch}^+(\mA)$. Let $F:\mA\to\mB$ be right exact. Then there exists a long exact sequence \\~\\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	\cdots & {\L_{i+1}F(C)} & {\L_iF(A)} & {\L_iF(B)} & {\L_iF(C)} & {\L_{i-1}F(A)} & \cdots
	\arrow[from=1-1, to=1-2]
	\arrow["\delta", from=1-2, to=1-3]
	\arrow[from=1-3, to=1-4]
	\arrow[from=1-4, to=1-5]
	\arrow["\delta", from=1-5, to=1-6]
	\arrow[from=1-6, to=1-7]
\end{tikzcd}}\\~\\
\end{lmm}

\begin{prp}{}{} There is always a convergent spectral sequence $$E_{pq}^2=(L_pF)(H_q(A))\Rightarrow\L_{p+q}F(A)$$
\end{prp}

\begin{crl}{}{} Let $\mA,\mB$ be abelian categories. Let $F:\mA\to\mB$ be a right exact functor. Then the following are true. 
\begin{itemize}
\item If $A$ is exact, then $\L_iF(A)=0$ for all $i$. 
\item If $f:A\to B$ is a quasi-isomorphism, then it induces isomorphisms $\L_iF(A)\cong\L_iF(B)$
\end{itemize}
\end{crl}

\section{Triangulated Categories}
\subsection{Axioms of a Triangulated Category}
\begin{defn}{Triangles}{}\\
Let $\mC$ be a category and $T:\mC\to\mC$ an automorphism functor. Let $A,B,C\in\mC$. A triangle on $(A,B,C)$ is a triple $(u,v,w)$ of morphisms in $\mC$ where $u:A\to B$, $v:B\to C$, $w:C\to T(A)$. 
\end{defn}

\begin{defn}{Morphisms of Triangles}{}\\
Let $\mC$ be a category and $T:\mC\to\mC$ an automorphism functor. Let $(u,v,w)$ and $(u',v',w')$ be triangles in $\mC$. A morphism of triangles is a triple $(f,g,h)$ such that the following diagram commutes: \\~\\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	A & B & C & {T(A)} \\
	{A'} & {B'} & {C'} & {T(A')}
	\arrow["u", from=1-1, to=1-2]
	\arrow["f"', from=1-1, to=2-1]
	\arrow["v", from=1-2, to=1-3]
	\arrow["g"', from=1-2, to=2-2]
	\arrow["w", from=1-3, to=1-4]
	\arrow["h"', from=1-3, to=2-3]
	\arrow["{T(f)}", from=1-4, to=2-4]
	\arrow["{u'}"', from=2-1, to=2-2]
	\arrow["{v'}"', from=2-2, to=2-3]
	\arrow["{w'}"', from=2-3, to=2-4]
\end{tikzcd}}\\~\\
\end{defn}

\begin{defn}{Triangulated Categories}{} Let $\mC$ be an additive category. We say that $\mC$ is a triangulated category if there is a functor $T:\mC\to\mC$ and a family $\{(u,v,w)\;|\;u,v,w\in\text{Mor}(\mC)\}$ of triangles called exact triangles such that the following hold. 
\begin{itemize}
\item For any morphism $u:A\to B$, there exists an exact triangle $(u,v,w)$: \\~\\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	A & B & C & {T(A)}
	\arrow["u", from=1-1, to=1-2]
	\arrow["v", from=1-2, to=1-3]
	\arrow["{\exists w}", dashed, from=1-3, to=1-4]
\end{tikzcd}}\\~\\ 
If $(u,v,w)$ is a triangle on $(A,B,C)$ isomorphic to an exact triangle $(u',v',w')$ on $(A',B',C')$, then $(u,v,w)$ is also exact: \\~\\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	A & B & C & {T(A)} \\
	{A'} & {B'} & {C'} & {T(A')}
	\arrow["u", from=1-1, to=1-2]
	\arrow["\cong"', from=1-1, to=2-1]
	\arrow["v", from=1-2, to=1-3]
	\arrow["\cong"', from=1-2, to=2-2]
	\arrow["w", from=1-3, to=1-4]
	\arrow["\cong"', from=1-3, to=2-3]
	\arrow["\cong", from=1-4, to=2-4]
	\arrow["{u'}"', from=2-1, to=2-2]
	\arrow["{v'}"', from=2-2, to=2-3]
	\arrow["{w'}"', from=2-3, to=2-4]
\end{tikzcd}}\\~\\ 
Finally, $(\text{id}_A,0,0)$ is exact: \\~\\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	A & A & 0 & {T(A)}
	\arrow["{\text{id}_A}", from=1-1, to=1-2]
	\arrow[from=1-2, to=1-3]
	\arrow[from=1-3, to=1-4]
\end{tikzcd}}\\~\\ 
\item Rotations: If $(u,v,w)$ is an exact triangle on $(A,B,C)$, then both rotations $(v,w,-T(u))$ and $(-T^{-1}(w),u,v)$ are exact triangles on $(B,C,T(A))$ and $(T^{-1}(C),A,B)$ respectively. 
\item Morphisms: Let the following be exact triangles: \\~\\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	A & B & C & {T(A)} \\
	{A'} & {B'} & {C'} & {T(A')}
	\arrow["u", from=1-1, to=1-2]
	\arrow["v", from=1-2, to=1-3]
	\arrow["w", from=1-3, to=1-4]
	\arrow["{u'}", from=2-1, to=2-2]
	\arrow["{v'}", from=2-2, to=2-3]
	\arrow["{w'}", from=2-3, to=2-4]
\end{tikzcd}}\\~\\
Suppose that there exists morphisms $f:A\to A'$ and $g:B\to B'$ such that $g\circ u=u'\circ f$. Then there exists $h:C\to C'$ such that $(f,g,h)$ is a morphism of triangles: \\~\\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	A & B & C & {T(A)} \\
	{A'} & {B'} & {C'} & {T(A')}
	\arrow["u", from=1-1, to=1-2]
	\arrow["f"', from=1-1, to=2-1]
	\arrow["v", from=1-2, to=1-3]
	\arrow["g"', from=1-2, to=2-2]
	\arrow["w", from=1-3, to=1-4]
	\arrow["{\exists h}"', dashed, from=1-3, to=2-3]
	\arrow["{T(f)}", from=1-4, to=2-4]
	\arrow["{u'}"', from=2-1, to=2-2]
	\arrow["{v'}"', from=2-2, to=2-3]
	\arrow["{w'}"', from=2-3, to=2-4]
\end{tikzcd}}\\~\\

\item The Octahedral Axiom: Let the following be exact triangles: \\~\\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	A & B & {C'} & {T(A)} \\
	B & C & {A'} & {T(B)} \\
	A & C & {B'} & {T(A)}
	\arrow["u", from=1-1, to=1-2]
	\arrow["j", from=1-2, to=1-3]
	\arrow["k", from=1-3, to=1-4]
	\arrow["v", from=2-1, to=2-2]
	\arrow["l", from=2-2, to=2-3]
	\arrow["i", from=2-3, to=2-4]
	\arrow["{v\circ u}", from=3-1, to=3-2]
	\arrow["m", from=3-2, to=3-3]
	\arrow["n", from=3-3, to=3-4]
\end{tikzcd}}\\~\\
Then there exists an exact triangle: \\~\\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	{C'} & {B'} & {A'} & {T(C')}
	\arrow["f", from=1-1, to=1-2]
	\arrow["g", from=1-2, to=1-3]
	\arrow["h", from=1-3, to=1-4]
\end{tikzcd}}\\~\\
such that $l=g\circ m$, $k=n\circ f$, $h=T(j)\circ i$, $i\circ g=T(u)\circ n$ and $f\circ j=m\circ v$. In other words, the following diagram commutes: \\~\\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	&& {B'} \\
	\\
	{C'} &&&& {A'} \\
	\\
	A &&&& C \\
	\\
	&& B
	\arrow["{\exists g}", dashed, from=1-3, to=3-5]
	\arrow["{\exists f}", dashed, from=3-1, to=1-3]
	\arrow["k"', from=3-1, to=5-1]
	\arrow["{h=T(j)\circ i}", from=3-5, to=3-1]
	\arrow["i"', from=3-5, to=7-3]
	\arrow["u"', from=5-1, to=7-3]
	\arrow["l", from=5-5, to=3-5]
	\arrow["j"', from=7-3, to=3-1]
	\arrow["v"', from=7-3, to=5-5]
	\arrow["n"', from=1-3, to=5-1, crossing over]
	\arrow["{v\circ u}", from=5-1, to=5-5, crossing over]
	\arrow["m"', from=5-5, to=1-3, crossing over]
\end{tikzcd}}\\~\\
Where we abused notation by drawing $k:C'\to T(A)$ as a morphism $C'\to A$ etc so that the drawing becomes compact. 
\end{itemize}
\end{defn}

\begin{lmm}{}{} Let $(\mC,T)$ be a triangulated category. Let $(u,v,w)$ be an exact triangle. Then $v\circ u$, $w\circ v$ and $T(u)\circ w$ are $0$ in $\mC$. 
\end{lmm}

\begin{lmm}{}{} Let $(\mC,T)$ be a triangulated category. Let $(f,g,h)$ be a morphism of exact triangles. If both $f$ and $g$ are isomorphisms, then $h$ is an isomorphism. 
\end{lmm}

\begin{defn}{Distinguished Triangles in K($\mA$)}{} Let $\mA$ be an abelian category. We say that a triangle \\~\\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	A & B & C & {T(A)}
	\arrow[from=1-1, to=1-2]
	\arrow[from=1-2, to=1-3]
	\arrow[from=1-3, to=1-4]
\end{tikzcd}}\\~\\
is distinguished in $K(\mA)$ if it is isomorphic to a triangle of the form \\~\\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	X & Y & C(f) & {T(X)}
	\arrow["{f}", from=1-1, to=1-2]
	\arrow[from=1-2, to=1-3]
	\arrow[from=1-3, to=1-4]
\end{tikzcd}}
\end{defn}

\begin{lmm}{}{} Let $\mA$ be an abelian category. Then $K(\mA)$, $K^+(\mA)$, $K^-(\mA)$ and $K^b(\mA)$ are all triangulated categories with distinguished triangles given by the above definition. 
\end{lmm}

\subsection{Morphisms of Triangulated Categories}
\begin{defn}{Morphisms of Triangulated Categories}{} Let $\mC$ and $\mD$ be triangulated categories. A morphism from $\mC$ to $\mD$ is a functor $F:\mC\to\mD$ such that the following are true. 
\begin{itemize}
\item $F$ is an additive functor
\item $F$ commutes with the translation functor. If $T$ is the automorphism of $\mC$ and $S$ is the automorphism of $\mD$, then $$F\circ T=S\circ F$$
\item $F$ sends exact triangles to exact triangles
\end{itemize}
\end{defn}

\subsection{Localization of Categories}
\begin{defn}{Localization of a Category}{} Let $\mC$ be a category and let $S$ be a collection of morphisms in $\mC$. A localization of $\mC$ with respect to $S$ is a category $S^{-1}\mC$ together with a functor $q:\mC\to S^{-1}\mC$ such that the following are true. 
\begin{itemize}
\item For all $s\in S$, $q(s)$ is an isomorphism in $S^{-1}\mC$
\item If $F:\mC\to\mD$ is a functor such that $F(s)$ is an isomorphism for all $s\in S^{-1}\mC$, then there exists a unique functor $G:S^{-1}\mC\to\mD$ such that the following diagram commute: \\~\\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	\mC & {S^{-1}\mC} \\
	& \mD
	\arrow["q", from=1-1, to=1-2]
	\arrow["F"', from=1-1, to=2-2]
	\arrow["{\exists !G}", dashed, from=1-2, to=2-2]
\end{tikzcd}}
\end{itemize}
\end{defn}

\begin{lmm}{}{} Let $\mA$ be an abelian category. Then $K(\mA)$ is a localization of $\mA$ with respect to all homotopy equivalences. 
\end{lmm}

\subsection{Derived Categories}
\begin{defn}{Derived Categories}{} Let $\mA$ be an abelian category and let $\bold{CCh}(\mA)$ be any category of chain complexes of $\mA$. Define the derived category $$D(\mA)=\bold{CCh}(\mA)[\mW^{-1}]$$ of $\mA$ where $\mW$ is all the quasi-isomorphisms in $\bold{CCh}(\mA)$. \\~\\

Respectively, for $\bold{CCh}^+(\mA)$ and $\bold{CCh}^b(\mA)$ define their derived categories to be the localization of the categories with respect to quasi-isomorphisms, denoted by $D^+(\mA)$ and $D^b(\mA)$ respectively. 
\end{defn}

\begin{thm}{}{} Let $\mA$ be an abelian category. Let $K(\mA)$ be the homotopy category of chain complexes of $\mA$. Let $\mW$ be all the quasi-isomorphisms in $\bold{CCh}(\mA)$. Then there is an equivalence of categories $$D(\mA)=K(\mA)[\mW^{-1}]$$
\end{thm}

\begin{defn}{Distinguished Triangles in D($\mA$)}{} Let $\mA$ be an abelian category. We say that a triangle \\~\\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	A & B & C & {T(A)}
	\arrow[from=1-1, to=1-2]
	\arrow[from=1-2, to=1-3]
	\arrow[from=1-3, to=1-4]
\end{tikzcd}}\\~\\
is distinguished in $D(\mA)$ if it is isomorphic to a triangle of the form \\~\\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	X & Y & C(f) & {T(X)}
	\arrow["{f}", from=1-1, to=1-2]
	\arrow[from=1-2, to=1-3]
	\arrow[from=1-3, to=1-4]
\end{tikzcd}}
\end{defn}

\begin{thm}{}{} Let $\mA$ be an abelian category. Then $D(\mA)$ is a triangulated triangles with distinguished triangles given by the above definition. 
\end{thm}

\begin{defn}{Full Subcategory of Complexes}{} Let $\mA$ and $\mB$ be abelian categories such that $\mA$ is a subcategory of $\mB$. Define $$D_\mA(\mB)$$ to be the full triangulated subcategory of complexes with cohomology in $\mA$. 
\end{defn}

\subsection{Derived Categories in Terms of Resolutions}
\begin{defn}{The Category of Injective / Projective Objects}{}\\
Let $\mA$ be an abelian category. 
\begin{itemize}
\item Define the category of injective objects $\bold{Inj}(\mA)$ to be the full subcategory of $\mA$ consisting of injective objects. 
\item Define the category of projective objects $\bold{Proj}(\mA)$ to be the full subcategory of $\mA$ consisting of projective objects. 
\end{itemize}
\end{defn}

\begin{lmm}{}{}\\
Let $\mA$ be an abelian category. Then $\bold{Inj}(\mA)$ and $\bold{Proj}(\mA)$ are abelian categories. 
\end{lmm}

\begin{prp}{}{}\\
Let $\mA$ be an abelian category. Then the following are true. 
\begin{itemize}
\item If $\mA$ has enough injectives, then the induced functor by inclusion $$K^+(\text{Inj}(\mA))\to D^+(\mA)$$ is an equivalence of categories. 
\item If $\mA$ has enough projectives, then the induced functor by inclusion $$K^-(\text{Proj}(\mA))\to D^-(\mA)$$ is an equivalence of categories. 
\end{itemize}
\end{prp}

\subsection{Derived Functors of Derived Categories}
\begin{defn}{Total Right Derived Functors}{} Let $\mA$ and $\mB$ be abelian categories. Let $F:K(\mA)\to K(\mB)$ be a morphism of triangulated categories. A total right derived functor of $F$ is a morphism $$RF:D(\mA)\to D(\mB)$$ together with a natural transformation $\xi:(K(\mA)\to K(\mB)\to D(\mB))\Rightarrow(K(\mA)\to D(\mA)\to D(\mB))$ using the following diagram: \\~\\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	{K(\mA)} & {K(\mB)} \\
	{D(\mA)} & {D(\mB)}
	\arrow["F", from=1-1, to=1-2]
	\arrow["{\exists!}"', from=1-1, to=2-1]
	\arrow["{\exists!}", from=1-2, to=2-2]
	\arrow["RF"', from=2-1, to=2-2]
\end{tikzcd}}\\~\\
from the top-right path to the lower-left path which is universal in the following sense. If $G:D(\mA)\to D(\mB)$ is another morphism equipped with the same natural transformation $\chi$, then there exists a unique natural transformation $$\eta:RF\Rightarrow G$$ such that $\chi=\eta\circ\xi$. 
\end{defn}

\begin{defn}{Total Left Derived Functors}{} Let $\mA$ and $\mB$ be abelian categories. Let $F:K(\mA)\to K(\mB)$ be a morphism of triangulated categories. A total right derived functor of $F$ is a morphism $$LF:D(\mA)\to D(\mB)$$ together with a natural transformation $\xi:(K(\mA)\to D(\mA)\to D(\mB))\Rightarrow(K(\mA)\to K(\mB)\to D(\mB))$ using the following diagram: \\~\\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	{K(\mA)} & {K(\mB)} \\
	{D(\mA)} & {D(\mB)}
	\arrow["F", from=1-1, to=1-2]
	\arrow["{\exists!}"', from=1-1, to=2-1]
	\arrow["{\exists!}", from=1-2, to=2-2]
	\arrow["RF"', from=2-1, to=2-2]
\end{tikzcd}}\\~\\
from the lower-left path to the top-right path which is universal in the following sense. If $G:D(\mA)\to D(\mB)$ is another morphism equipped with the same natural transformation $\chi$, then there exists a unique natural transformation $$\eta:G\Rightarrow LF$$ such that $\chi=\eta\circ\xi$. 
\end{defn}

\begin{lmm}{}{} Let $\mA$ and $\mB$ be abelian categories. Let $F:K(\mA)\to K(\mB)$ be a morphism of triangulated categories. If $F$ is exact, then $F$ is its own left and right total derived functor. 
\end{lmm}

\begin{thm}{}{} Let $\mA,\mB$ be abelian categories. Let $F:\mA\to\mB$ be a functor. Then the following are true. 
\begin{itemize}
\item If $F$ is left exact and $\mA$ has enough injectives, then $$\R^iF=H^i(RF):\bold{CCh}(\mA)\to\mB$$
\item If $F$ is right exact and $\mA$ has enough projectives, then $$\L_iF=H_i(LF):\bold{Ch}(\mA)\to\mB$$
\end{itemize}
\end{thm}

Notice that in particular, if $X\in\mA$ is an object considered as a chain complex in degree $0$, we have seen that $\L_iF=L_iF:\bold{\mA}\to\bold{\mB}$ hence $L_iF=H_i(LF):\bold{\mA}\to\bold{\mB}$. This is similar for the dual version.
