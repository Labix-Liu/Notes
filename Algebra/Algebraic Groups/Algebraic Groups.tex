\section{Introduction to Group Schemes}
\subsection{Group Schemes in General}
\begin{defn}{Group Schemes}{}\\
Let $S$ be a scheme. A group scheme over $S$ is a group object in $\bold{Sch}_S$. 
\end{defn}

\begin{defn}{Types of Group Schemes}{}\\
Let $S$ be a scheme. Let $G$ be a group scheme over $S$. 
\begin{itemize}
\item We say that $G$ is a group variety if $G$ is a variety. 
\item We say that $G$ is an affine group scheme if $G$ is affine. 
\item We say that $G$ is an abelian variety if $G$ is a complete connected group variety. 
\end{itemize}
\end{defn}

Let $X$ be an $S$-scheme. Recall that the functor of points is the functor $h_X:\bold{Sch}_S\to\bold{Set}$ given by $$h_X(Y)=X(Y)=\Hom_{\bold{Sch}_S}(Y,X)$$

\begin{lmm}{}{}\\
Let $S$ be a scheme. Let $G$ be a scheme over $S$. Then $G$ is a group scheme if and only if its functor of points $h_G:\bold{Sch}_S\to\bold{Set}$ factors through the forgetful fucntor $\bold{Grp}\to\bold{Set}$. 
\end{lmm}

\begin{defn}{Commutative Group Schemes}{}\\
Let $S$ be a scheme. A commutative group scheme over $S$ is an abelian group object in $\bold{Sch}_S$. 
\end{defn}

\begin{lmm}{}{}\\
Let $S$ be a scheme. Let $G$ be a group scheme over $S$. Then the following are equivalent. 
\begin{itemize}
\item $G$ is commutative. 
\item $G(X)$ is an abelian group for all $X\in\bold{Sch}_S$. 
\item The functor of points $h_G:\bold{Sch}_S\to\bold{Set}$ factors through the forgetful fucntor $\bold{Ab}\to\bold{Grp}$. 
\end{itemize}
\end{lmm}

\begin{defn}{Morphism of Group Schemes}{}\\
Let $S$ be a scheme. Let $G,H$ be group schemes over $S$. Let $\varphi:G\to H$ be a morphism over $S$. We say that $\varphi$ is a morphism of group schemes if it is a morphism of group objects in $\bold{Sch}_S$. 
\end{defn}

Equivalently, this means that $\varphi$ must respect multiplication: $$\varphi\circ m_G=m_H\circ(\varphi\times\varphi)$$

\begin{lmm}{}{}\\
Let $S$ be a scheme. Let $G,H$ be group schemes over $S$. Let $\varphi:G\to H$ be a morphism over $S$. Then $\varphi$ is a morphism of group schemes if and only if $h_f(X):h_G(X)\to\widetilde{H}(X)$ is a group homomorphism for all $X\in\bold{Sch}_S$. 
\end{lmm}

\begin{defn}{Group Subschemes}{}\\
Let $S$ be a scheme. Let $G$ be a group scheme over $S$. A group subscheme of $G$ is a subset $H\subseteq G$ such that the following are true. 
\begin{itemize}
\item $H$ is a subscheme of $G$. 
\item The inclusion $H\hookrightarrow G$ is a morphism of group schemes. 
\end{itemize}
\end{defn}

\begin{lmm}{}{}\\
Let $S$ be a scheme. Let $G,H$ be group schemes over $S$. Then $H$ is a group subscheme of $G$ if and only if $h_H(X)$ is a subgroup of $h_G(X)$ for all $X\in\bold{Sch}_S$. 
\end{lmm}

\subsection{Group Schemes of Finite Type Over a Finitely Generated Algebra}
Let $k$ be a field. Let $R$ be a finitely generated commutative $k$-algebra. Let $X$ be a scheme of finite type over $R$. We have to be careful about the conditions on the functor of points for $X$ to be a group scheme. There are two functor of points $h_X:\bold{Sch}_R\to\bold{Set}$ and $h_X:\bold{Sch}_R^\text{ft}\to\bold{Set}$. Normally we use the former functor to check if $X$ is a group scheme no matter what conditions we impose on $X$. We need two facts:
\begin{enumerate}
\item $h_X:\bold{Sch}_S\to\bold{Set}$ commutes with filtered colimits of commutative $R$-algebras / $R$-schemes. 
\item Every $R$-scheme is a filtered colimit of its finite type pieces. 
\end{enumerate}~\\

But we can do better. Since $X$ is a finite type scheme, it is covered by finitely many affine finite type schemes. Fact: $h_X$ can be reconstructed from $h_U$ where $U$ ranges over the affine finite type schemes of $X$. Thus it suffices to check $h_X$ over all affine finite type schemes. \\

Finally, using the equivalence between finite generated $R$-algebras and affine finite type schemes over $R$, we have the following: 

\begin{prp}{}{}\\
Let $k$ be a field. Let $R$ be a finitely generated commutative $k$-algebra. Let $X$ be a scheme of finite type over $R$. Then $X$ is a group scheme if and only if the functor $\widetilde{X}:\bold{FGCAlg}_R\to\bold{Set}$ given by $$A\mapsto\Hom_{\bold{Sch}_R}(\text{Spec}(A),X)$$ factors through the forgetful functor $\bold{Grp}\to\bold{Set}$. 
\end{prp}

Note that Milnor book does not go through this because he defines group schemes as group objects directly in the category of schemes of finite type over $R$. \\

The checks for commutativity, subgroup and homomorphism also follows the same reduction. 

\begin{lmm}{}{}\\
Let $k$ be a field. Let $R$ be a finitely generated commutative $k$-algebra. Let $G$ be a group scheme of finite type over $R$. Then the following are equivalent. 
\begin{itemize}
\item $G$ is commutative. 
\item $\widetilde{G}(A)$ is an abelian group for all $A\in\bold{FGCAlg}_R$. 
\item $\widetilde{G}$ factors through the forgetful functor $\bold{Ab}\to\bold{Grp}$. 
\end{itemize}
\end{lmm}

\begin{lmm}{}{}\\
Let $k$ be a field. Let $R$ be a finitely generated commutative $k$-algebra. Let $G,H$ be group schemes of finite type over $R$. Then the following are true. 
\begin{itemize}
\item Let $\varphi:G\to H$ be a morphism of schemes. Then $\varphi$ is a morphism of group schemes if and only if $\widetilde{\varphi}(A)$ is a group homomorphism for all $A\in\bold{FGCAlg}_R$. 
\item $H$ is a group subscheme of $G$ if and only if $\widetilde{H}(A)$ is a subgroup of $\widetilde{G}(A)$ for all $A\in\bold{FGCAlg}_R$. 
\end{itemize}
\end{lmm}

\begin{eg}{}{}\\
Let $k$ be a field. 
\begin{itemize}
\item The fucntor $A\mapsto(A,+)$ is represented by the group scheme $$\G_a(k)=\text{Spec}(k[x])$$
\item The functor $A\mapsto(A^\times,\times)$ is represented by the group scheme $$\G_m(k)=\text{Spec}(k[x,x^{-1}])$$
\item Let $n\in\N$. The functor $A\mapsto\{a\in A\;|\;a^n=1\}$ is represented by the group scheme $$\mu_n=\text{Spec}\left(\frac{k[x]}{(x^n-1)}\right)$$
\item The functor $A\mapsto\text{GL}(n,A)$ is represented by the group scheme $$\text{GL}(n,k)=\text{Spec}\left(\frac{k[x_{i,j}\;|\;1\leq i,j\leq n]\otimes_kk[y]}{(\det((x_{i,j})_{1\leq i,j\leq n})y-1)}\right)$$
\end{itemize}
\end{eg}

\begin{eg}{}{}\\
Let $k$ be a field. 
\begin{itemize}
\item Let $V$ be a finite dimensional vector space over $k$. Then the functor $A\mapsto V\otimes_kA$ is represented by the group scheme $$V_a=\text{Spec}(S(V^\ast))$$
\item Let $V$ be a finite dimensional vector space over $k$. Let $\phi:V\to k^n$ be a $k$-isomorphism. Then $\phi$ induces an isomorphism $$V_a\cong\G_a^{\dim(V)}$$
\item Let $V$ be a finite dimensional vector space over $k$. Define the group scheme $\text{GL}(V)$ to be the representing object of the functor $A\mapsto\text{Aut}_k(V\otimes_kA)$. 
\item Let $V$ be a finite dimensional vector space over $k$. Let $\phi:V\to k^n$ be a $k$-isomorphism. Then the map $T\mapsto\phi\circ T\circ\phi^{-1}$ induces an isomorphism of group schemes $$\text{GL}(V)\cong\text{GL}(n,k)$$
\end{itemize}
\end{eg}

\subsection{Dense Subgroups}
\begin{defn}{Schematically Dense}{}\\
Let $k$ be a field. Let $X$ be a scheme of finite type over $k$. Let $S$ be a subset of $X(k)$. We say that $S$ is schematically dense in $X$ if the map $\mO_X(X)\to\prod_{s\in S}k(s)$ defined by $f\mapsto(f(s))_{s\in S}$ is injective. 
\end{defn}

\begin{prp}{}{}\\
Let $k$ be a field. Let $G$ be a group scheme over $k$. Then the following are equivalent. 
\begin{itemize}
\item $G(k)$ is dense in $G$. 
\item $G(k)$ is schematically dense in $G$. 
\item $G$ is reduced and $G(k)$ is dense in the subspace of closed points of $G$. 
\end{itemize}
\end{prp}

\subsection{The Identity Component}
\begin{lmm}{}{}\\
Let $k$ be a field. Let $G$ be a group scheme of finite type over $k$. Then the following are true. 
\begin{itemize}
\item $G$ is irreducible if and only if $G$ is connected. 
\item $G$ is the disjoint union of its irreducible components. 
\end{itemize}
\end{lmm}

\begin{defn}{Identity Component}{}\\
Let $k$ be a field. Let $G$ be a group scheme of finite type over $k$. Define the identity component $G^\circ$ of $G$ to be the unique irreducible component of $G$ that contains $1_G$. 
\end{defn}

\begin{prp}{}{}\\
Let $k$ be a field. Let $G$ be a group scheme of finite type over $k$. Then $G^\circ$ is a subgroup scheme of $G$. 
\end{prp}

\begin{prp}{}{}\\
Let $k$ be a field. Let $G$ be a group scheme of finite type over $k$. Let $F/k$ be a field extension. Then we have $$G^\circ(F)\cong G(F)^\circ$$
\end{prp}

\subsection{Reducedness and Smoothness}
\begin{prp}{}{}\\
Let $k$ be a field. Let $G$ be a group scheme of finite type over $k$. Then $G$ is separated. 
\end{prp}

\begin{prp}{}{}\\
Let $k$ be a field. Let $G$ be a group scheme of finite type over $k$. Then the following are equivalent. 
\begin{itemize}
\item $G$ is geometrically reduced. 
\item $G$ is smooth. 
\item $G^\circ$ is smooth. 
\item $1_G$ is a smooth point of $G$. 
\item $\dim(T_1G)=\dim(G)$. 
\end{itemize}
\end{prp}

\begin{prp}{}{}\\
Let $k$ be a field. Let $G$ be a group scheme of finite type over $k$. If $G$ is reduced and $k$ is perfect, then $G$ is geometrically reduced. 
\end{prp}

Thus group varieties over a perfect field are smooth. 

\begin{prp}{}{}\\
Let $k$ be a field. Let $G$ be a group scheme of finite type over $k$. Then the following are equivalent. 
\begin{itemize}
\item $G$ is irreducible. 
\item $G$ is connected. 
\item $G$ is geometrically connected. 
\end{itemize}
\end{prp}

\subsection{Correspondence of Subgroup Schemes}
\begin{prp}{}{}\\
Let $k$ be a field. Let $G$ be a group scheme of finite type over $k$. If $G_\text{red}$ is geometrically reduced, then it is a subgroup scheme. 
\end{prp}

\begin{prp}{}{}\\
Let $k$ be a field. Let $G$ be a group scheme of finite type over $k$. Let $H\leq G$ be a subgroup scheme of $G$. Then $H$ is closed in $G$. 
\end{prp}

Thus subgroup schemes of affine group schemes are affine. 

\begin{prp}{}{}\\
Let $k$ be a field. Let $G$ be a group scheme of finite type over $k$. Then there is a one to one correspondence $$\left\{S\leq G(k)\;\bigg{|}\;S\text{ is a closed subgroup }\right\}\;\;\overset{1:1}{\longleftrightarrow}\;\;\left\{H\subseteq G\;\bigg{|}\;\substack{H\text{ is a geometrically reduced}\\\text{subgroup scheme such that}\\H(k)\text{ is schematically dense in }H}\right\}$$ given by $H\mapsto H(k)$ in the backwards direction. 
\end{prp}

Since geometrically reduced implies reduced and all group schemes are separated, we deduce the following corollary. 

\begin{crl}{}{}\\
Let $k$ be a field. Let $G$ be a group scheme of finite type over $k$. Then there is a one to one correspondence $$\left\{S\leq G(k)\;\bigg{|}\;S\text{ is a closed subgroup }\right\}\;\;\overset{1:1}{\longleftrightarrow}\;\;\left\{H\subseteq G\;\bigg{|}\;\substack{H\text{ is a subgroup variety of }G\text{ such}\\\text{that }H(k)\text{ is topologiaclly dense in }G}\right\}$$ given by $H\mapsto H(k)$ in the backwards direction. 
\end{crl}

\begin{crl}{}{}\\
Let $k$ be a separable field. Let $G$ be a group scheme of finite type over $k$. Then there is a one to one correspondence $$\left\{S\leq G(k)\;\bigg{|}\;S\text{ is a closed subgroup }\right\}\;\;\overset{1:1}{\longleftrightarrow}\;\;\left\{H\subseteq G\;\bigg{|}\;H\text{ is a subgroup variety of }G\right\}$$ given by $H\mapsto H(k)$ in the backwards direction. 
\end{crl}

\subsection{The Affine Case and Hopf Algebras}
\begin{prp}{}{}\\
Let $k$ be a field. Let $A$ be a finitely generated commutative algebra over $k$. Let $\Delta:A\to A\otimes_kA$ be a $k$-algebra homomorphism. Then $(\text{Spec}(A),\text{Spec}(\Delta))$ is an affine group scheme over $k$ if and only if $(A,\Delta)$ is a commutative Hopf algebra. 
\end{prp}

\begin{prp}{}{}\\
Let $k$ be a field. There is an equivalence of categories $$\text{Spec}:\bold{FGCHopf}_k\overset{\simeq}{\longrightarrow}\bold{AffGrpSch}_k$$
\end{prp}

\begin{prp}{}{}\\
Let $k$ be a field. Let $G$ be an affine group scheme of finite type over $k$. Then the correspondence between closed subschemes of $G$ and ideal of $\mO_G(G)$ descends to a one to one correspondence $$\left\{H\leq G\;\bigg{|}\;H\text{ is a subgroup scheme of }G\right\}\;\;\overset{1:1}{\longleftrightarrow}\;\;\left\{I\subseteq\mO_G(G)\;\bigg{|}\;I\text{ is a Hopf ideal }\right\}$$
\end{prp}

\begin{thm}{Cartier 1962}{}\\
Let $k$ be a field of characteristic $0$. Let $G$ be an affine group scheme of finite type over $k$. Then $G$ is smooth. 
\end{thm}

\subsection{Anti-Affine Groups}
\begin{defn}{Anti-Affine Groups}{}\\
Let $k$ be a field. Let $G$ be a group scheme of finite type over $k$. We say that $G$ is anti-affine if $\mO_G(G)=k$. 
\end{defn}

\begin{lmm}{}{}\\
Let $k$ be a field. Let $G$ be an abelian variety over $k$. Then $G$ is anti-affine. 
\end{lmm}

\begin{prp}{}{}\\
Let $k$ be a field. Let $G$ be a group scheme of finite type over $k$. If $G$ is anti-affine, then $G$ is smooth and connected. 
\end{prp}

\section{Group Actions and Representations}
\subsection{Group Actions}
\begin{defn}{Group Actions}{}\\
Let $k$ be a field. Let $G$ be a group scheme over $k$. Let $X$ be a scheme of finite type over $k$. A group action of $G$ on $X$ is a morphism $\mu:G\times X\to X$ such that the following diagram commutes: \\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	{G\times G\times X} & {G\times X} && {\ast\times X} & {G\times X} \\
	{G\times X} & X &&& X
	\arrow["{m\times\text{id}_X}", from=1-1, to=1-2]
	\arrow["{\text{id}_G\times\mu}"', from=1-1, to=2-1]
	\arrow["\mu", from=1-2, to=2-2]
	\arrow["{\text{incl.}}", from=1-4, to=1-5]
	\arrow["\cong"', from=1-4, to=2-5]
	\arrow["\mu", from=1-5, to=2-5]
	\arrow["\mu"', from=2-1, to=2-2]
\end{tikzcd}} \\
\end{defn}

\begin{lmm}{}{}\\
Let $k$ be a field. Let $G$ be a group scheme over $k$. Let $X$ be a scheme of finite type over $k$. Let $\mu:G\times X\to X$ be a morphism. Then $\mu$ is a group action if and only if $\widetilde{\mu}:\widetilde{G}\times\widetilde{X}\to\widetilde{X}$ is a natural transformation such that for all $R\in\bold{FGAlg}_k$, $\mu(R):G(R)\times X(R)\to X(R)$ is a group action. 
\end{lmm}

\begin{lmm}{}{}\\
Let $k$ be a field. Let $G$ be a group scheme over $k$. Let $X$ be a scheme of finite type over $k$. Let $\mu:G\times X\to X$ be a group action. Then $\mu$ is faithfully flat. 
\end{lmm}

\begin{lmm}{}{}\\
Let $k$ be a field. Let $G$ be a group scheme over $k$. Let $X$ be a scheme of finite type over $k$. Let $\mu:G\times X\to X$ be a group action. Then the following are true. 
\begin{itemize}
\item If $G$ is smooth, then $\mu$ is smooth. 
\item If $G$ is finite, then $\mu$ is finite. 
\end{itemize}
\end{lmm}

\subsection{Representations of Affine Group Schemes}
\begin{defn}{Representation of a Group Scheme}{}\\
Let $k$ be a field. Let $G$ be a group scheme of finite type over $k$. A representation of $G$ is a vector space $V$ over $k$ together with a morphism of group schemes $\rho:G\to GL(V)$. 
\end{defn}

We will see that for any group scheme $H$, there is a natural isomorphism $$\Hom_{\bold{GrpSch}_k}(G,H)\cong\Hom_{\bold{GrpSch}_k}(G^\text{aff},H)$$ so it suffices to consider affine group schemes. 

\begin{lmm}{}{}\\
Let $k$ be a field. Let $G$ be an affine group scheme over $k$. Let $V$ be a finite dimensional vector space over $k$. If $\rho:G\to GL(V)$ is a natural transformation, then $\rho$ is a homomorphism of group schemes. 
\end{lmm}

\begin{lmm}{}{}\\
Let $k$ be a field. Let $G$ be an affine group scheme over $k$. Let $V$ be a finite dimensional vector space over $k$. Then there is a one to one correspondence $$\{\rho:G\to GL(V)\;|\;\rho\text{ is a representation }\}\;\;\overset{1:1}{\longleftrightarrow}\;\;\left\{\phi:G\times V_a\to V_a\;\bigg{|}\;\substack{\phi\text{ is a group action such that the action}\\G(R)\times V_a(R)\to V_a(R)\text{ is }R\text{-linear}}\right\}$$
\end{lmm}

\begin{prp}{}{}\\
Let $k$ be a field. Let $G$ be an affine group scheme over $k$. Let $V$ be a finite dimensional vector space over $k$. Then there is a one to one correspondence $$\{\rho:G\to GL(V)\;|\;\rho\text{ is a representation }\}\;\;\overset{1:1}{\longleftrightarrow}\;\;\left\{\phi:V\to V\otimes\mO_G(G)\;\bigg{|}\;\substack{\phi\text{ defines a comodule}\\\text{structure on }V}\right\}$$
\end{prp}

\begin{prp}{}{}\\
Let $k$ be a field. Let $G$ be an affine group scheme over $k$. Let $V$ be a vector space over $k$. Let $\rho:G\to GL(V)$ be a representation. Then $\rho$ is a filtered union of its finite dimensional subrepresentation. 
\end{prp}

Therefore it suffices to consider finite dimensional representations of affine group schemes.

\subsection{Characters and Eigenspaces}
\begin{defn}{Characters}{}\\
Let $k$ be a field. Let $G$ be an group scheme of finite type over $k$. Define the group of characters of $G$ to be $$\chi(G)=\Hom_{\bold{GrpSch}_k}^\text{ft}(G,\G_m(k))$$
\end{defn}

\begin{lmm}{}{}\\
Let $k$ be a field. Let $G$ be an group scheme of finite type over $k$. Then $\chi(G)$ is an abelian group. 
\end{lmm}

\begin{prp}{}{}\\
Let $k$ be a field. Let $G$ be an group scheme of finite type over $k$. Then there is a bijection $$\chi(G)\;\;\overset{1:1}{\longleftrightarrow}\;\;G(\mO_G(G))$$ where the latter is the group-like elements of the Hopf algebra. 
\end{prp}

We denote the above map by $a:\chi(G)\to G(\mO_G(G))$. 

\begin{prp}{}{}\\
Let $k$ be a field. Let $G$ be an group scheme of finite type over $k$. Let $\chi_1,\dots,\chi_n\in\chi(G)$ be distinct characters. Then they are linearly independent as elements of the $k$-algebra $\mO_G(G)$. 
\end{prp}

\begin{defn}{Eigenspaces}{}\\
Let $k$ be a field. Let $G$ be a group scheme of finite type over $k$. Let $V$ be a finite dimensional vector space over $k$. Let $\rho:G\to GL(V)$ be a representation. Let $\chi\in\chi(G)$ be a character. Define the eigenspace of $\chi$ by $$V_\chi=\{v\in V\;|\;\rho(v)=v\otimes a(\chi)\}$$
\end{defn}

\subsection{Chevalley's Structure Theorem}
\begin{thm}{Chevalley's Structure Theorem}{}\\
Let $k$ be a field. Let $G$ be a smooth connected group scheme of finite type over $k$. Then $G$ contains a largest smooth connected affine normal subgroup scheme $N$. 
\end{thm}

\begin{thm}{Barsotti-Chevalley Theorem}{}\\
Let $k$ be a perfect field. Let $G$ be a smooth connected group scheme of finite type over $k$. Let $N$ be the largest smooth connected affine normal subgroup scheme. Then $G/N$ is an abelian variety. 
\end{thm}

Relaxing the smoothness condition: 

\begin{thm}{}{}\\
Let $k$ be a field. Let $G$ be a smooth connected group scheme of finite type over $k$. Then there exists a connected affine normal subgroup scheme $N$ of $G$ such that $G/N$ is an abelian variety. 
\end{thm}

\subsection{The Anti-Affine Group Scheme}
\begin{defn}{The Associated Affine Group Scheme}{}\\
Let $k$ be a field. Let $G$ be a group scheme of finite type over $k$. Define the associated affine group scheme to be $$G^\text{aff}=\text{Spec}(\mO_G(G))$$
\end{defn}

\begin{prp}{}{}\\
Let $k$ be a field. Let $G$ be a group scheme of finite type over $k$. Then there is a natural isomorphism $$\Hom_{\bold{GrpSch}_k}(G,H)\cong\Hom_{\bold{GrpSch}_k}(G^\text{aff},H)$$ given by factorizing through the natural map $\phi:G\to G^\text{aff}$ for any affine group scheme $H$ over $k$.

Let $k$ be a field. Let $M$ be an finitely generated abelian group. Then we have $$\widetilde{D(M)}=\Hom_\bold{Ab}(M,(-)^\times)$$
\end{lmm}

By the fundamental theorem of finitely generated abelian groups, we have $$\text{Spec}(k[M])\cong\prod_{i=1}^k\mu_{n_i}\times\G_m(k)^{\rank(M)}$$ if $$M\cong\prod_{i=1}^k\Z/n_i\Z\times\Z^{\rank(M)}$$

\begin{lmm}{}{}\\
Let $k$ be a field. Let $M$ be an finitely generated abelian group. Then there is a group isomorphism $$M\cong\chi(D(M))$$ where for each $A\in\bold{FGCAlg}_k$, the map $$M\to\Hom_\bold{Ab}(\Hom_\bold{Ab}(M,A^\times),A^\times)$$ is given by $m\mapsto(f\mapsto f(m))$. 
\end{lmm}

\begin{prp}{}{}\\
Let $k$ be a field such that $\text{char}(k)=p\neq 0$. Let $M$ be an finitely generated abelian group. Then the following are true. 
\begin{itemize}
\item $D(M)$ is connected if and only if the only torsion in $G$ is $p$-torsion. 
\item $D(M)$ is smooth if and only if $G$ has no $p$-torsion. 
\item $D(M)$ is smooth and connected if and only if $G$ is free. 
\end{itemize}
\end{prp}

For characteristic $0$, we already know that $D(M)$ is smooth by Cartier's theorem. For connectedness, we have: 

\begin{prp}{}{}\\
Let $k$ be a field of characteristic $0$. Let $M$ be an finitely generated abelian group. Then $D(M)$ is connected if and only if $M$ is free. 
\end{prp}

\subsection{Diagonalizable Groups}
\begin{defn}{Diagonlizable Groups}{}\\
Let $k$ be a field. Let $G$ be an affine group scheme over $k$. We say that $G$ is diagonalizable if $$\mO_G(G)=k\langle g\in G\;|\;g\text{ is a group-like element }\rangle$$ as $k$-vector spaces. 
\end{defn}

\begin{defn}{The Category of Diagonalizable Group Schemes}{}\\
Let $k$ be a field. Define the category $\bold{Diag}_k$ to be the full subcategory of $\bold{GrpSch}_k^\text{ft}$ that consists of diagonalizable group schemes. 
\end{defn}

\begin{prp}{}{}\\
Let $k$ be a field. Then the following are true. 
\begin{itemize}
\item $D:\bold{FGAb}\to\bold{Diag}_k$ is an equivalence of categories, with inverse given by $\chi$ sending a group scheme to its abelian group of characters. 
\item $D$ and $\chi$ are exact functors. 
\end{itemize}
\end{prp}

\begin{crl}{}{}\\
Let $k$ be a field. Let $G$ be a diagonalizable affine group scheme over $k$. If $H\leq G$ is a subgroup of $G$, then $H$ and $G/H$ are diagonalizable. 
\end{crl}

\subsection{Diagonalizable Representations}
\begin{defn}{Diagonalizable Representations}{}\\
Let $k$ be a field. Let $G$ be an group scheme of finite type over $k$. Let $\rho:G\to GL(V)$ be a representation of $G$. We say that $\rho$ is diagonalizable if $\rho$ is the sum of $1$-dimensional representations. 
\end{defn}

\begin{prp}{}{}\\
Let $k$ be a field. Let $G$ be an group scheme of finite type over $k$. Then the following are equivalent. 
\begin{itemize}
\item $G$ is diagonalizable. 
\item Every finite dimensional representation of $G$ is diagonalizable. 
\item Every representation of $G$ is diagonalizable. 
\item For every representation $\rho:G\to\text{GL}(V)$ of $G$, there is a decomposition $$V=\bigoplus_{\chi\in\chi(G)}V_\chi$$
\end{itemize}
\end{prp}

\subsection{Tori}
\begin{defn}{Tori}{}\\
Let $k$ be a field. Let $G$ e a group scheme of finite type over $k$. We say that $G$ is a tori if there exists a field extension $F/k$ such that there is an isomorphism $$G(F)\cong\G_m^n$$ for some $n\in\N^+$. 
\end{defn}

\begin{defn}{Split Tori}{}\\
Let $k$ be a field. Let $G$ e a group scheme of finite type over $k$. We say that $G$ is a split tori if there is an isomorphism $$G\cong\G_m^n$$ for some $n\in\N^+$. 
\end{defn}

The split tori are precisely the smooth connected diagonalizable group schemes of finite type over $k$. They correspond to free finitely generated free abelian groups under the equivalence between $D$ and $\chi$. 

\begin{prp}{}{}\\
Let $k$ be a field. Let $T$ be a split tori. Then the following are true. 
\begin{itemize}
\item Let $G$ be a subgroup scheme of $T$. Then $G$ is diagonalizable. 
\item Let $G$ be a group scheme of finite type over $k$. Let $T\to G$ be a quotient map. Then $G$ is a split tori. 
\end{itemize}
\end{prp}

\begin{prp}{}{}\\
Let $k$ be a field. Let $T$ be a split torus over $k$. Then the following are true. 
\begin{itemize}
\item Let $T'$ be a split torus over $k$. Let $\phi:T\to T'$ be a morphism of group schemes. Then $\im(\phi)$ is closed torus in $T'$. 
\item Let $H\leq T$ be a subgroup variety of $T$. If $H$ is irreducible, then $H$ is a torus. 
\item Let $H\leq T$ be a subgroup of $T$. Then $T/H$ is a torus. 
\end{itemize}
\end{prp}

\subsection{The Affine Group Scheme of a Group Ring}
Let $R$ be a commutative ring. Let $G$ be an abelian group. Recall that $R[G]$ is the group algebra whose elements are $R$-linear combination of elements of $G$ and multiplication is induced by the binary operation of $G$. 

\begin{defn}{Group Scheme Associated to Group Algebra}{}\\
Let $k$ be a field. Let $G$ be an abelian group. Define the affine group scheme associated to the group algebra of $G$ to be $\text{Spec}(k[G])$. 
\end{defn}

\begin{defn}{Functor Associated to Group Algebra}{}\\
Let $k$ be a field. Let $G$ be an abelian group. Define the functor $$D(G):\bold{FGAlg}_k\to\bold{Grp}$$ as follows. 
\begin{itemize}
\item For each $k$-algebra $R$, define $D(G)(R)=\Hom_{\bold{Grp}}(G,R^\times)$. 
\item For each $k$-algebra homomorphism, $\varphi:R\to S$ is sent to the group homomorphism $\Hom_{\bold{Grp}}(G,R^\times)\to\Hom_{\bold{Grp}}(G,S^\times)$ which is post composition with $\varphi$. 
\end{itemize}
\end{defn}

\begin{lmm}{}{}\\
Let $k$ be a field. Let $G$ be an abelian group. Then the functor $D(G)$ represents the affine group scheme $\text{Spec}(k[G])$. 
\end{lmm}

\begin{prp}{}{}\\
Let $k$ be a field. Let $G$ be a finitely generated abelian group. Suppose that $\varphi:G\to\bigoplus_{i=1}^r\Z/b_i\Z\oplus\Z^n$ is a group isomorphism. Then $\varphi$ induces an isomorphism of group schemes $$\text{Spec}(k[G])\cong\prod_{i=1}^r\mu_{b_i}\times\G_m^n$$
\end{prp}
