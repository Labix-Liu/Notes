\section{The Symplectic Form}
\subsection{Basic Constructions}
\begin{defn}{Symplectic Manifolds}{}\\
Let $M$ be a smooth manifold. Let $\omega\in\Omega^2(M)$ be a $2$-form. We say that $\omega$ is a symplectic form if the following are true. 
\begin{itemize}
\item $\omega$ is a symplectic structure on $TM$. 
\item $d\omega=0$. 
\end{itemize}
In this case we call $(M,\omega)$ a symplectic manifold. 
\end{defn}

\begin{lmm}{}{}\\
Let $(M,\omega)$ be a symplectic manifold. Then $\dim(M)$ is even. 
\end{lmm}

\begin{defn}{The Liouville Volume Form}{}\\
Let $(M,\omega)$ be a symplectic manifold of dimension $2n$. Define the Liouville volume form of $M$ by $$\mu_M=\frac{\omega^n}{n!}$$
\end{defn}

\begin{prp}{}{}\\
Let $(M,\omega)$ be a symplectic manifold. Then the following are true. 
\begin{itemize}
\item $\mu_M$ is a nowhere-vanishing differential form. 
\item $M$ is orientable. 
\end{itemize}
\end{prp}

\begin{eg}{}{}\\
The following are symplectic manifolds. 
\begin{itemize}
\item $\R^{2n}$ with coordinates $(x_1,\dots,x_n,y_1,\dots,y_n)$ and with symplectic form $$\sum_{i=1}^ndx_i\wedge dy_i$$
\item For all $g\geq 1$, the compact orientable surface $\Sigma_g$ with any $\omega$ such that $0\neq[\omega]\in H_\text{dR}^2(\Sigma_g)$. 
\end{itemize}
\end{eg}

\subsection{Standard Symplectic Coordinates}
\begin{thm}{Moser's Theorem Relative Version}{}\\
Let $M$ be a smooth manifold. Let $X$ be an embedded compact submanifold of $M$. Let $\iota:X\hookrightarrow M$ be the inclusion map. Let $\omega_0$ and $\omega_1$ be symplectic forms on $M$. If $\omega_0|_X=\omega_1|_X$, then there exists $X\subseteq U_0,U_1\subseteq M$ and a map $\varphi:U_0\to U_1$ such that the following are true. 
\begin{itemize}
\item $\iota=\varphi\circ\iota$. 
\item $\varphi$ is a diffeomorphism. 
\item $(\varphi|_X)^\ast\omega_1=\omega_0$. 
\end{itemize}
\end{thm}

\begin{thm}{Darboux (1982)}{}\\
Let $(M,\omega)$ be a symplectic manifold. Let $p\in M$. Then there exists a chart $(U,x^1,\dots,x^n,y^1,\dots,y^n)$ at $p$ such that $$\omega|_U=\sum_{i=1}^ndx^i\wedge dy^i$$
\end{thm}

\begin{defn}{Darboux Coordinates}{}\\
Let $(M,\omega)$ be a symplectic manifold. Let $(U,x^1,\dots,x^n,y^1,\dots,y^n)$ be a local chart of $M$. We say that the chart is Darboux if $$\omega|_U=\sum_{i=1}^ndx^i\wedge dy^i$$
\end{defn}

\subsection{Maps Preserving the Symplectic Structure}
\begin{defn}{Symplectic Maps}{}\\
Let $(M,\omega)$ and $(N,\sigma)$ be symplectic manifolds. Let $f:M\to N$ be a smooth map. 
\begin{itemize}
\item We say that $f$ is symplectic if $$f^\ast\sigma=\omega$$
\item We say that $f$ is a symplectomorphism if $f$ is symplectic and a diffeomorphism. 
\end{itemize}
\end{defn}

\begin{lmm}{}{}\\
Let $(M,\omega)$ and $(N,\sigma)$ be symplectic manifolds. Let $f:M\to N$ be a symplectic map. Then $f$ is an immersion. 
\end{lmm}

\begin{defn}{The Group of Symplectomorphisms}{}\\
Let $(M,\omega)$ be a symplectic manifold. Define the group of symplectomorphisms of $M$ to be the subgroup $$\text{Symp}(M,\omega)=\{f:M\to M\;|\;f\text{ is a symplectomorphism }\}\leq\text{Diff}(M)$$
\end{defn}

\subsection{Symplectic Structure on the Cotangent Bundle}
Through out this section we denote the projection map $T^\ast M\to M$ by $\pi_M$. 

\begin{defn}{The Canonical Hamiltonian Structure}{}\\
Let $M$ be a smooth manifold. Define the canonical Hamiltonian structure $\omega$ on $T^\ast M$ as follows. At the local chart $(U,q^1,\dots,q^n,p^1,\dots,p^n)$, $\omega$ is given by $$\omega|_U=\sum_{i=1}^ndq^i\wedge dp^i$$
\end{defn}

\begin{lmm}{}{}\\
Let $M$ be a smooth manifold. Let $\omega$ be the canonical Hamiltonian structure. Then $(T^\ast M,\omega)$ is a symplectic manifold. 
\end{lmm}

\begin{defn}{The Liouville 1-Form}{}\\
Let $M$ be a smooth manifold. Let $\pi:T^\ast M\to M$ be the bundle map. Define the Lioville $1$-form $\theta\in\Omega^1(T^\ast M)$ as follows. For each $q\in M$ and $p\in T_q^\ast M$, define $$\theta_{(q,p)}=p(\pi_M(v))$$ for any $v\in T_{(q,p)}(T^\ast M)$. 
\end{defn}

\begin{prp}{}{}\\
Let $M$ be a smooth manifold. Let $(U,q^1,\dots,q^n,p^1,\dots,p^n)$ be a chart of $T^\ast M$ where $(q_1,\dots,q_n)\in M$. Then $\theta$ is given in local coordinates by $$\theta|_U=\sum_{i=1}^np^idq^i$$
\end{prp}

\begin{prp}{}{}\\
Let $M$ be a smooth manifold. Let $\theta$ be the Liouville 1-form. Let $\omega$ be the canonical Hamitonian structure. Then we have $$\omega=-d\theta$$
\end{prp}

\begin{defn}{The Lift of a Diffeomorphism}{}\\
Let $M,N$ be smooth manifolds. Let $f:M\to N$ be a diffeomorphism. Define the lift of $f$ to be the map $f_\#:T^\ast M\to T^\ast N$ given by $$f_\#(p,v)=(f(p),w)$$ where $w\in T_p^\ast M$ is the unique element such that $v=(df_p)^\ast w$. 
\end{defn}

\begin{prp}{}{}\\
Let $M,N$ be smooth manifolds. Let $f:M\to N$ be a diffeomorphism. Then the following are true. 
\begin{itemize}
\item We have $f\circ\pi_M=\pi_N\circ f_\#$. 
\item $f_\#^\ast\theta_N=\theta_M$. 
\end{itemize}
\end{prp}

\section{Lagrangian Submanifolds}
\subsection{Basic Constructions}
\begin{defn}{Lagrangian Vector Subspaces}{}\\
Let $k$ be a field. Let $(V,\omega)$ be a symplectic vector space over $k$. Let $W\subseteq V$ be a subspace. We say that $W$ is Lagrangian if the following are true. 
\begin{itemize}
\item $\omega|_W=0$. 
\item $2\dim(W)=\dim(V)$. 
\end{itemize}
\end{defn}

\begin{defn}{Lagrangian Submanifolds}{}\\
Let $(M,\omega)$ be a symplectic manifold. Let $L$ be an embedded submanifold of $M$. We say that $L$ is Lagrangian if $T_pL$ is a Lagrangian subspace of $M$. 
\end{defn}

\begin{lmm}{}{}\\
Let $(M,\omega)$ be a symplectic manifold. Let $L$ be an embedded submanifold of $M$. Then $L$ is Lagrangian if and only if the following are true. 
\begin{itemize}
\item $\iota^\ast\omega=0$ where $\iota:L\hookrightarrow M$ is the inclusion map. 
\item $2\dim(L)=\dim(M)$. 
\end{itemize}
\end{lmm}

\begin{prp}{}{}\\
Let $M$ be a smooth manifold. Let $s\in\Omega^1(M)$. Then $\im(s)$ is a Lagrangian submanifold of $T^\ast M$ if and only if $ds=0$. 
\end{prp}

\begin{thm}{Weinstein Lagrangian Neighborhood Theorem}{}\\
Let $M$ be a smooth manifold. Let $X$ be an embedded compact submanifold of $M$. Let $\iota:X\hookrightarrow M$ be the inclusion map. Let $\omega_0$ and $\omega_1$ be symplectic forms on $M$ such that $X$ is a Lagrangian submanifold of $(M,\omega_0)$ and $(M,\omega_1)$. Then there exists $X\subseteq U_0,U_1\subseteq M$ and a map $\varphi:U_0\to U_1$ such that the following are true. 
\begin{itemize}
\item $\iota=\varphi\circ\iota$. 
\item $\varphi$ is a diffeomorphism. 
\item $(\varphi|_X)^\ast\omega_1=\omega_0$. 
\end{itemize}
\end{thm}

\subsection{Relation to Symplectomorphisms}
\begin{prp}{}{}\\
Let $(M,\omega)$ and $(N,\sigma)$ be symplectic manifolds. Let $f:M\to N$ be a diffeomorphism. Then $f$ is a symplectomorphism if and only if $\Gamma_f$ is a Lagrangian submanifold of $(M\times N,\text{proj}_M^\ast\omega-\text{proj}_N^\ast\sigma)$. 
\end{prp}

Recipe for producing symplectomorphisms. 
\begin{enumerate}
\item Start with a Lagrangian submanifold of $(T^\ast M\times T^\ast N,\omega\times\sigma)$. 
\item Send the Lagrangian through the twist map $\tau:T^\ast M\times T^\ast N\to T^\ast M\times T^\ast N$ given by $(x,v)\mapsto(x,-v)$. The image is then a Lagrangian. 
\item Check whether $\tau(\sigma)$ is the graph of some diffeomorphism $\varphi:M\to N$. 
\item If it is, then $\varphi$ is a symplectomorphism. 
\end{enumerate}

\subsection{Generating Functions}
\begin{defn}{Generating Functions}{}\\
Let $M,N$ be smooth manifolds. Let $f\in\mC^\infty(M\times N)$. Define $$Y_f=\{(p,q,\text{proj}_{T_p^\ast M}d_{(p,q)}f,-\text{proj}_{T_q^\ast M}d_{(p,q)}f)\in M\times N\times T^\ast M\times T^\ast N\;|\;(p,q)\in M\times N\}$$ We say that $f$ is a generating function if there exists a diffeomorphism $\varphi:T^\ast M\to T^\ast N$ such that $Y_f=\Gamma_\varphi$. 
\end{defn}

In this case, we know that $\varphi:T^\ast M\to T^\ast N$ is given by $$\varphi(x,\text{proj}_{T_p^\ast M}d_{(p,q)}f)=(y,-\text{proj}_{T_q^\ast M}d_{(p,q)}f)$$

\section{Calculus on Symplectic Manifolds}
\subsection{Symplectic Vector Fields}
\begin{defn}{Symplectic Vector Fields}{}\\
Let $(M,\omega)$ be a symplectic manifold. Let $X\in\mathfrak{X}(M)$ be a vector field. We say that $X$ is symplectic if $\iota_X(\omega)$ is a closed form. 
\end{defn}

\begin{prp}{}{}\\
Let $(M,\omega)$ be a symplectic manifold. Let $X\in\mathfrak{X}(M)$ be a vector field. Let $\theta^X$ be the flow of $X$. Then $X$ is symplectic if and only if $\theta^X(t,-)$ is a symplectomorphism for all $t$. 
\end{prp}

\subsection{Hamiltonian Vector Fields}
\begin{defn}{The Vector Field of a Function}{}\\
Let $(M,\omega)$ be a symplectic manifold. Let $H\in\mC^\infty(M)$. Define the associated vector field of $H$ to be the unique vector field $X_H\in\mathfrak{X}(M)$ such that $$\iota_{X_H}(\omega)=df$$ where $\iota$ is the interior product. 
\end{defn}

\begin{prp}{}{}\\
Let $(M,\omega)$ be a symplectic manifold. Let $H\in\mC^\infty(M)$. Let $(U,x^1,\dots,x^n,y^1,\dots,y^n)$ be a Darboux chart. Then the vector field of $H$ is given locally at the chart by $$(X_H)|_U=\sum_{i=1}^n\left(\frac{\partial H}{\partial y^i}\frac{\partial}{\partial x^i}-\frac{\partial H}{\partial x^i}\frac{\partial}{\partial y^i}\right)$$
\end{prp}

\begin{defn}{Hamiltonian Vector Fields}{}\\
Let $(M,\omega)$ be a symplectic manifold. Let $X\in\mathfrak{X}(M)$ be a vector field. We say that $X$ is Hamiltonian if there exists a smooth map $H\in\mC^\infty(M)$ such that $$X=X_H$$
\end{defn}

\begin{prp}{}{}\\
Let $(M,\omega)$ be a symplectic manifold. Let $X\in\mathfrak{X}(M)$ be a vector field. If $X$ is Hamiltonian, then $X$ is symplectic. 
\end{prp}

\begin{thm}{Liouville's Theorem}{}\\
Let $(M,\omega)$ be a symplectic manifold. Let $X\in\mathfrak{X}(M)$ be a complete Hamiltonian vector field. Let $\theta^X$ be the flow of $X$. Then we have $$\theta^X(t,-)^\ast\mu_M=\mu_M$$
\end{thm}

\subsection{The Poisson Structure}
\begin{defn}{The Poisson Bracket}{}\\
Let $(M,\omega)$ be a symplectic manifold. Define the poisson bracket of $M$ to be the map $\{-,-\}:\mC^\infty(M)\times\mC^\infty(M)\to\mC^\infty(M)$ by $$\{f,g\}=X_f(g)$$
\end{defn}

\begin{lmm}{}{}\\
Let $(M,\omega)$ be a symplectic manifold. Then the following are true. 
\begin{itemize}
\item $(M,\{-,-\})$ is a Lie algebra. 
\item Leibniz rule: $\{f,gh\}=\{f,g\}h+g\{f,h\}$. 
\end{itemize}
\end{lmm}

\begin{prp}{}{}\\
Let $(M,\omega)$ be a symplectic manifold. Then the map $\mC^\infty(M)\to\mathfrak{X}(M)$ given by $f\mapsto X_f$ is a Lie algebra homomorphism. 
\end{prp}

\subsection{Integrable Systems}
\begin{defn}{Hamiltonian Systems}{}\\
A Hamiltonian system is a triple $(M,\omega,H)$ where $(M,\omega)$ is a symplectic manifold and $H\in\mC^\infty(M)$. 
\end{defn}

\begin{defn}{Integrable Systems}{}\\
Let $(M,\omega,H)$ be a Hamiltonian system. Suppose that $\dim(M)=2n$. We say that it is integrable if there exists $f_i\in\mC^\infty(M)$ for $1\leq i\leq n$ and $f_1=H$ such that $$\{f_i,f_j\}=0$$ for all $i,j$. 
\end{defn}

\begin{thm}{Arnold-Liouville Theorem}{}\\
Let $(M,\omega,H,f_2,\dots,f_n)$ be an integrable system. Let $c\in\R$. Then $(H,f_2,\dots,f_n)^{-1}(c)$ is a Lagrangian submanifold of $M$. Moreover, if $M$ is compact and connected, then the following are true. 
\begin{itemize}
\item $(H,f_2,\dots,f_n)^{-1}(c)$ is diffeomorphic to $T^k$ for some $k\in\N^+$. 
\item Angle coordinates. 
\end{itemize}
\end{thm}

\section{Symplectic and Hamiltonian Actions}
\subsection{Symplectic Actions}
\begin{defn}{Symplectic Actions}{}\\
Let $(M,\omega)$ be a symplectic manifold. Let $G$ be a Lie group. Let $\varphi:G\to\text{Diff}(M)$ be a smooth action. We say that the action is symplectic if $$\im(\varphi)\subseteq\text{Symp}(M)$$
\end{defn}

\begin{lmm}{}{}\\
Let $(M,\omega)$ be a symplectic manifold. Then there is a one to one bijection $$\{X\in\mathfrak{X}(M)\;|\;X\text{ is symplectic and complete }\}\;\;\overset{1:1}{\longleftrightarrow}\;\;\{\varphi:\R\times M\to M\;|\;\varphi\text{ is a symplectic action }\}$$ given as follows. 
\begin{itemize}
\item For a complete and symplectic vector field $X$, the map $t\mapsto\text{exp}(tX)$ is a symplectic action. 
\item For a symplectic action $\varphi$, the map $p\mapsto\frac{d\varphi(t,p)}{dt}$ is a complete and symplectic vector field. 
\end{itemize}
\end{lmm}

\subsection{Moment Maps}
\begin{defn}{Moment Maps}{}\\
Let $(M,\omega)$ be a symplectic manifold. Let $G$ be a Lie group with a symplectic action on $M$. A moment map of $M$ is a map $$\mu:M\to\text{Lie}(G)^\ast\cong\R^{\dim(G)}$$ such that the following are true. 
\begin{itemize}
\item Hamilton's condition: For all $\xi\in\text{Lie}(G)$, the fundamental vector field $\xi_M$ is Hamiltonian via the map $\langle\mu,\xi\rangle:M\to\R$. 
\item Equivariance: For all $g\in G$, we have $$\mu(g\cdot x)=\text{Ad}_g^\ast(\mu(x))$$ for all $x\in M$. 
\end{itemize}
\end{defn}

\begin{defn}{Hamiltonian Actions}{}\\
Let $(M,\omega)$ be a symplectic manifold. Let $G$ be a Lie group with a symplectic action on $M$. We say that the action is Hamiltonian if it admits a moment map $\mu:M\to\text{Lie}(G)$. 
\end{defn}

\subsection{Symplectic Quotients}
\begin{defn}{Regular Values}{}\\
Let $(M,\omega)$ be a symplectic manifold. Let $G$ be a Lie group with a Hamiltonian action on $M$. Let $\xi\in\text{Lie}(G)^\ast$. We say that $\xi$ is a regular value if the induced map $$d\mu_p:T_pM\to\text{Lie}(G)^\ast$$ is surjective for all $p\in\mu^{-1}(\xi)$. 
\end{defn}

\begin{defn}{Coadjoint Stabilizer}{}\\
Let $(M,\omega)$ be a symplectic manifold. Let $G$ be a Lie group with a Hamiltonian action on $M$. Let $\xi\in\text{Lie}(G)^\ast$ be a regular value. Define the coadjoint stabilizer of $\xi$ by $$G_\xi=\{g\in G\;|\;\text{Ad}_g^\ast(\xi)=\xi\}$$
\end{defn}

\begin{defn}{Symplectic Quotients}{}\\
Let $(M,\omega)$ be a symplectic manifold. Let $G$ be a Lie group with a Hamiltonian action on $M$. Let $\xi\in\text{Lie}(G)^\ast$ be a regular value. Define the symplectic quotient of $M$ by $G$ to be the orbit space $$M//G=\frac{\mu^{-1}(\xi)}{G_\xi}$$
\end{defn}

\begin{thm}{The Marsden–Weinstein–Meyer Theorem}{}\\
Let $(M,\omega)$ be a symplectic manifold. Let $G$ be a Lie group with a Hamiltonian action on $M$. Let $\xi\in\text{Lie}(G)^\ast$ be a regular value. If $G_\xi$ acts freely and properly on $\mu^{-1}(\xi)$, then $(M//G,\overline{\omega})$ is a symplectic manifold where $\overline{\omega}$ is given by $$\iota^\ast\omega=\pi^\ast\overline{\omega}$$ where $\iota:\mu^{-1}(\xi)\hookrightarrow M$ is the inclusion map and $\pi:\mu^{-1}:\to M//G$ is the quotient map. 
\end{thm}

\begin{prp}{}{}\\
Let $(M,\omega)$ be a symplectic manifold. Let $G$ be a Lie group with a Hamiltonian action on $M$. Let $\xi\in\text{Lie}(G)^\ast$ be a regular value. Suppose that $G_\xi$ acts freely and properly on $\mu^{-1}(\xi)$. Then we have $$\dim(M//G)=\dim(M)-2\dim(G_\xi)$$
\end{prp}

\subsection{Hamiltonian Torus Actions}
\begin{thm}{The Atiyah–Guillemin–Sternberg convexity theorem (1982)}{}\\
Let $(M,\omega)$ be a compact connected symplectic manifold. Let $T^k\times M\to M$ be a Hamiltonian action. Then $\mu(M)$ is a convex polytope in $\text{Lie}(T^k)^\ast$ given by $$\mu(M)=\text{Conv}(\text{Fix}_{T^k}(M))$$
\end{thm}

\begin{defn}{Toric Symplectic Manifold}{}\\
Let $(M,\omega)$ be a symplectic manifold of dimension $2n$. Let $T^n\times M\to M$ be a Hamiltonian torus action. We say that $M$ is a toric symplectic manifold if $M$ is compact connected and the action is faithful. 
\end{defn}

Delzant polytope: simple, rational, smooth

\begin{thm}{Delzant (1988)}{}\\
There is a one to one bijection $$\frac{\{(M,\omega)\;|\;(M,\omega)\text{ is a toric symplectic manifold }\}}{\cong}\;\;\overset{1:1}{\leftrightarrow}\;\;\{P\;|\;P\text{ is a Delzant polytope }\}$$
\end{thm}
