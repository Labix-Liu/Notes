\section{Vector Bundles}
\subsection{Basic Definitions}
\begin{defn}{Vector Bundles}{}\\
Let $F$ be a field. Let $E,B$ be spaces with $B$ connected. Let $p:E\to B$ be a continuous map. We say that $p$ is a vector bundle of dimension $d\in\N$ over $F$ if the following are true. 
\begin{itemize}
\item $p:E\to B$ is surjective. 
\item Local Triviality: There exists a collection of homeomorphisms $$\{(U_i\subseteq B,\phi_i:p^{-1}(U_i)\to U_i\times F^k)\;|\;i\in I\}$$ called local trivialization charts such that $B=\bigcup_{i\in I}U_i$ and the following diagram commutes: \\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	{p^{-1}(U_i)} && {U_i\times F^k} \\
	& U_i
	\arrow["{\phi_i}", from=1-1, to=1-3]
	\arrow["p"', from=1-1, to=2-2]
	\arrow["\pi", from=1-3, to=2-2]
\end{tikzcd}}\\
where $\pi$ is the projection by forgetting the second variable. 
\end{itemize}
We say that $B$ is the base space, $E$ the total space. It is denoted as $(F,E,B)$. 
\end{defn}

The local trivialization means that locally at a neighbourhood, the vector bundle looks the same the open set times $F^k$. In particular, there is also a notion of trivial bundle which means that the bundle is globally just $B\times\R^r$. 

\begin{defn}{The Fiber of a Bundle}{}\\
Let $F$ be a field. Let $B$ be a space. Let $p:E\to B$ be a vector bundle over $F$. Let $b\in B$. Define the fiber of $E$ at $B$ to be $$E_b=p^{-1}(b)$$
\end{defn}

\begin{eg}{The Trivial Bundle}{}\\
Let $B$ be a space. Let $F$ be a field. The trivial bundle $p:B\times F^k\to B$ defined by $p(b,v)=b$ is a vector bundle. The bundle is denoted by $e^k$. 
\end{eg}

\begin{eg}{The Mobius Strip}{}\\
Let $E=\frac{I\times\R}{\sim}$ where $(0,x)\sim(1,-x)$ for $x\in\R$. Then the projection map $E\to\frac{I}{0\sim 1}\cong S^1$ is a vector bundle of rank $1$. 
\end{eg}

\begin{lmm}{}{}\\
Let $F$ be a field. Let $p:E\to B$ be a vector bundle of dimension $d\in\N$ over $F$. Then the local trivialization charts induces a homeomorhpsim $$E_b\cong F^d$$ for all $b\in B$. 
\end{lmm}

\begin{defn}{Morphism of Vector Bundles}{}\\
Let $p_1:E_1\to B_1$ and $p_2:E_2\to B_2$ be vector bundles. A morphism of these vector bundles is given by is a pair of continuous maps $f:E_1\to E_2$ and $g:B_1\to B_2$ such that the following diagram commutes \\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
E_1\arrow[r, "f"]\arrow[d, "p_1"] & E_2\arrow[d, "p_2"]\\
B_1\arrow[r, "g"] & B_2
\end{tikzcd}} \\
and $f|_{p^{-1}(b)}$ is a linear map for all $b\in B$. 
\end{defn}

\begin{defn}{Isomorphism of Vector Bundles}{} A bundle homomorphism from $E_1$ to $E_2$ is an isomorphism if there exists an inverse bundle homomorphism from $E_2$ to $E_1$. In this case, we say that $E_1$ and $E_2$ are isomorphic. 
\end{defn}

\begin{lmm}{}{}\\
Let $B$ be space. Let $p_1:E_1\to B$ and $p_2:E_2\to B$ be vector bundles. Let $f:E_1\to E_2$ be a morphism of vector bundles. Then $f$ is an isomorphism if and only if $f|_{p_1^{-1}(b)}$ is an isomorphism for all $b\in B$. 
\begin{proof}\\
If $f$ is an isomorphism, then the inverse of $f|_{p_1^{-1}(b)}$ is clearly given by $f^{-1}|_{p_2^{-1}(b)}$. \\

Now suppose that $f|_{p_1^{-1}(b)}$ is an isomorphism for all $b\in B$. Then clearly $f$ is bijective. I claim that $p_2=p_1\circ f^{-1}$. Let $x\in E_2$. Then there exists $y\in E_1$ such that $f(y)=x$. Then $p_1(f^{-1}(x))=p_1(y)$. But $p_1=p_2\circ f$ by definition of morphisms of vector bundles. Hence $p_1(y)=p_2(f(y))=p_2(x)$ and we conclude. \\

It remains to show that $f^{-1}$ is continuous. Let $U\subseteq E_1$ be an open set. Then there exists local trivialization charts $(V_i,\phi_i)$ of $E_1$ such that the $V_i$'s cover $p_1(U)$. Then we have $U=\bigcup_{i\in I}U\cap V_i$ and $f^{-1}(U)$ is open if and only if $f^{-1}(U\cap V_i)$ is open. But since restrictions of local trivialization charts are also local trivialization charts, it suffices to prove that $f$ is continuous on local trivialization charts. \\

So let $(U,\phi)$ be a local trivialization chart of $E_1$. Since $E_2$ is also covered by local trivialization charts, WLOG assume that $(U,\psi)$ is a local trivialization chart of $E_2$. Now the map $\psi\circ f\circ\phi^{-1}:U\times F^k\to U\times F^n$ sends $(x,v)$ to $(x,g(x)(v))$ for some function $g(x):F^k\to F^n$. Since the map is a homeomorphism, the map $v\mapsto g(x)(v)$ must be a homeomorphism, and hence $k=n$. Since restriction to each fiber is a bijection, $g(x)$ must be an invertible linear map and hence continuous. Hence $\phi\circ f^{-1}\circ\psi^{-1}$ is continuous. Thus $f$ is continuous. 
\end{proof}
\end{lmm}

\begin{defn}{Sections}{}\\
Let $k$ be a field. Let $p:E\to B$ be a vector bundle over $k$. A section of $p$ is a map $s:B\to E$ such that $p\circ s=\text{id}_B$. 
\end{defn}

\begin{lmm}{}{}\\
Let $k$ be a field. Let $p:E\to B$ be a vector bundle over $k$. Then the following are true. 
\begin{itemize}
\item Let $s_1,s_2:B\to E$ be sections of $p$. Then $s_1+s_2$ is a section of $p$. 
\item Let $s:B\to E$ be a section of $p$. Let $\lambda\in k$. Then $\lambda s$ is a section of $p$. 
\end{itemize}
\end{lmm}

\begin{defn}{The Space of Sections}{}\\
Let $F$ be a field. Let $p:E\to B$ be a vector bundle over $F$. Define the $F$-vector space of sections to be the set $$s(E)=\{s:B\to E\;|\;s\text{ is a section of }p\}$$ together with addition and scalar multiplication defined pointwise. 
\end{defn}

\begin{lmm}{}{}\\
Let $F$ be a field. Let $p:E\to B$ be a vector bundle over $F$. Then $E$ is isomorphic to the trivial bundle if and only if there exists global sections $s_1,\dots,s_n\in s(E)$ such that $s_1(b),\dots,s_n(b)$ is a basis for $p^{-1}(b)$ for all $b\in B$. 
\end{lmm}

\begin{eg}{}{}\\
The Mobius vector bundle $M\to S^1$ is non-trivial. 
\begin{proof}\\
Suppose that it is trivial. By the above lemma, there exists a global section $s:S^1\to M$ such that $s(e^{2\pi it})\neq[(t,0)]$ for all $t\in[0,1]$. Writing $M=\frac{I\times\R}{\sim}$ where $(0,x)\sim(1,-x)$, we see that the map $s$ is of the form $s(e^{2\pi it})=[(t,f(t))]$ for some continuous map $f:I\to\R$ such that $f(0)=-f(1)$ by definition of the equivalence relation. But by the intermediate value theorem, there exists $c\in I$ such that $f(c)=0$, contradicting the fact that $s$ is nowhere vanishing. 
\end{proof}
\end{eg}

\begin{defn}{Set of Isomorphism Classes of Vector Bundles}{}\\
Let $X$ be a space. Let $F$ be a field. Denote the set of isomorphism classes of vector bundles by $$\text{Vect}_n^F(X)=\frac{\{p:E\to B\;|\;E\text{ is a vector bundle of }B\text{ over }F\}}{\sim}$$ where $E\sim E'$ if $E$ and $E'$ are isomorphic bundles. 
\end{defn}

\subsection{The Cocycle Conditions}
Given two charts $(U_\alpha,\phi_\alpha)$ and $(U_\beta,\phi_\beta)$ of a vector bundle, $$\phi_\beta\circ\phi_\alpha^{-1}:(U_\alpha\cap U_\beta)\times F^k\to(U_\alpha\cap U_\beta)\times F^k$$ is a well defined function. In particular, by fixing a point in $U_\alpha\cap U_\beta$, we obtain a linear map. 

\begin{defn}{Transition Functions}{}\\
Let $p:E\to B$ be an $F$-vector bundle of rank $k$. Let $(U_\alpha,\phi_\alpha)$ and $(U_\beta,\phi_\beta)$ be local trivialization. Define $g_{\alpha,\beta}:U_\alpha\cap U_\beta\to\text{GL}(n,F)$ by $$x\mapsto\phi_\beta\circ\phi_\alpha^{-1}(x,-):F^k\to F^k$$ In other words, $g_{\alpha,\beta}$ is such that $$\phi_\beta\circ\phi_\alpha^{-1}(x,v)=(x,g_{\alpha,\beta}(x)v)$$ For $x\in U_\alpha\cap U_\beta$ and $v\in F^k$. 
\end{defn}

\begin{prp}{}{}\\
Let $p:E\to B$ be a $K$-vector bundle of rank $k$. The transition functions of the vector bundle satisfies the following. 
\begin{itemize}
\item Cocycle condition: $g_{\gamma,\alpha}(x)\circ g_{\beta,\gamma}(x)\circ g_{\alpha,\beta}(x)=\text{id}$ on $U_\alpha\cap U_\beta\cap U_\gamma$ for all $x\in U_\alpha\cap U_\beta\cap U_\gamma$. 
\item $g_{\alpha\alpha}(x)=I_r$ for all $x\in U_\alpha$. 
\end{itemize}
\begin{proof}\\
We have $$\phi_\alpha\circ\phi_\gamma^{-1}\circ\phi_\gamma\circ\phi_\beta^{-1}\circ\phi_\beta\circ\phi_\alpha^{-1}=\text{id}$$ Hence $$(x,(g_{\gamma,\alpha}(x)\circ g_{\beta,\gamma}(x)\circ g_{\alpha,\beta}(x))(v))=(x,v)$$ for all $x\in U_\alpha\cap U_\beta\cap U_\gamma$ and all $v\in F^k$. Hence the cocycle condition is achieved. \\

Clearly $g_{\alpha,\alpha}(x,-)$ is the identity by definition. 
\end{proof}
\end{prp}

The cocycle conditions completely determine a vector bundle. 

\begin{defn}{Associated Vector Bundle}{}\\
Let $F$ be a field. Let $B$ be a space. Let $\{U_i\;|\;i\in I\}$ be an open cover of $B$. Let $g_{i,j}:U_i\cap U_j\to\text{GL}(k,F)$ be a continuous functions for $i,j\in I$ such that $g_{k,i}(x)\circ g_{j,k}(x)\circ g_{i,j}(x)=\text{id}$ for all $x\in U_i\cap U_j\cap U_k$ and $g_{i,i}(x)=\text{id}$ for all $x\in U_i$. Define the associated vector bundle to be the space $$E_g=\frac{\coprod_{i\in I}U_i\times F^k}{\sim}$$ where $(i,b,v)\sim(j,b,g_{i,j}(b)(v))$ if $b\in U_i\cap U_j$, together with the induced projection map $\text{proj}:E_g\to B$. 
\end{defn}

\begin{lmm}{}{}\\
Let $F$ be a field. Let $B$ be a space. Let $\{U_i\;|\;i\in I\}$ be an open cover of $B$. Let $g_{i,j}:U_i\cap U_j\to\text{GL}(k,F)$ be a continuous functions for $i,j\in I$. Then $p:E_g\to B$ is a vector bundle of rank $k$. 
\begin{proof}\\
Firstly note that the projection map $E_g\to B$ is well defined by the universal property of quotient. We then need to prove that the local triviality condition is satisfied. Clearly the composition $U_i\times F^k\hookrightarrow\coprod_{i\in I}U_i\times F^k\to E_g$ composed with the canonical projection map commutes. It is a homeomorphism following the universal property of quotients. 
\end{proof}
\end{lmm}

\begin{lmm}{}{}\\
Let $F$ be a field. Let $B$ be a space. Let $p:E\to B$ be a vector bundle over $F$. Let $g=\{g_{i,j}\;|\;i,j\in I\}$ be the transition functions. Then the transition functions induce an isomorphism $$E_g\cong E$$ of vector bundles. 
\end{lmm}

\subsection{Operations on Vector Bundles}
\begin{defn}{Whitney Sum}{} Let $p_1:E_1\to B$ and $p_2:E_2\to B$ be two vector bundles. Define the direct sum of the vector bundles to be $$E_1\oplus E_2=\{(v_1,v_2)\in E_1\times E_2\;|\;p_1(v_1)=p_2(v_2)\}$$ together with the projection $p:E_1\oplus E_2\to B$ defined by $(v_1,v_2)\mapsto p_1(v)=p_2(v)$. 
\end{defn}

\begin{eg}{}{}\\
Let $p:M\to S^1$ be the Mobius vector bundle. Then $M\oplus M\cong e^2$. 
\begin{proof}\\
Define a morphism $M\oplus M\to e^2$ by $([(t,x)],[(t,y)])\mapsto(e^{2\pi it},x,y)$. Notice that that restricted to the open cover $U_1=\{e^{2\pi it}\;|\;0\leq t\leq 1/2\}$ and $U_2=\{e^{2\pi it}\;|\;1/2\leq t\leq 1\}$ of $S^1$. we see that the local trivialization charts of $M\oplus M$ coincides with that of our morphism. Hence the morphism is a well defined morphism of vector bundles. Moreover, it is an isomorphism of vector spaces on each fiber. 
\end{proof}
\end{eg}

\begin{defn}{Pullback Bundle}{}\\
Let $F$ be a field. Let $X,Y$ be spaces. Let $p:E\to Y$ be a vector bundle over $F$. Let $f:X\to Y$ be a map. Define the pullback bundle of $f$ to be the space $$f^\ast(E)=\{(x,e)\in X\times E\;|\;f(x)=p(e)\}$$ together with the projection map $p^\ast:f^\ast(E)\to X$ given by $p^\ast(x,e)=x$. 
\end{defn}

\begin{defn}{The Hom Bundle}{}\\
Let $F$ be a field. Let $B$ be a space. Let $p_1:E_1\to B$ and $p_2:E_2\to B$ be vector bundles over $F$. Define the Hom bundle of $E_1$ and $E_2$ to be the space $$\Hom(E_1,E_2)=\coprod_{b\in B}\Hom_F(p_1^{-1}(b),p_2^{-1}(b))$$ together with the coproduct of the maps $\Hom_F(p_1^{-1}(b),p_2^{-1}(b))\to\{b\}$. 
\end{defn}

\begin{defn}{The Dual Bundle}{}\\
Let $F$ be a field. Let $B$ be a space. Let $p:E\to B$ be a vector bundle over $F$. Define the dual bundle of $p$ to be the vector bundle $$E^\ast=\Hom(E,e^1)$$
\end{defn}

\begin{defn}{The Tensor Product Bundle}{}\\
Let $F$ be a field. Let $B$ be a space. Let $p_1:E_1\to B$ and $p_2:E_2\to B$ be vector bundles over $F$. Define the tensor product bundle to be the space $$E_1\otimes E_2=\coprod_{b\in B}p_1^{-1}(b)\otimes_F p_2^{-1}(b)$$ together with the coproduct of the maps $p_1^{-1}(b)\otimes p_2^{-1}(b)\to\{b\}$. 
\end{defn}

\begin{defn}{The Exterior Product Bundle}{}\\
Let $F$ be a field. Let $B$ be a space. Let $p:E\to B$ be a vector bundle. Define the $n$th exterior product bundle of $E$ to be $$\Lambda^n(E)=\coprod_{b\in B}\Lambda^n(p^{-1}(b))$$  together with the coproduct of the maps $\Lambda^n(p_1^{-1}(b))\to\{b\}$. 
\end{defn}

\subsection{Homotopy Theory of Vector Bundles}
\begin{thm}{The Homotopy Theorem}{}\\
Let $X$ be a paracompact space. Let $p:E\to X\times I$ be a vector bundle. Then there is a bundle isomorphism $$p^{-1}(X\times\{0\})\cong p^{-1}(X\times\{1\})$$
\end{thm}

\begin{prp}{}{}\\
Let $X,Y$ be paracompact spaces. Let $p:E\to Y$ be a vector bundle. Let $f,g:X\to Y$ be homotopic maps. Then there is an isomrophism $$f^\ast(E)\cong g^\ast(E)$$
\end{prp}

\begin{crl}{}{}\\
Let $X,Y$ be paracompact spaces. Let $f:X\to Y$ be a map. If $f$ is a homotopy equivalence, then the pullback map induces natural bijection $$f^\ast:\text{Vect}_n^F(Y)\overset{1:1}{\to}\text{Vect}_n^F(X)$$
\end{crl}

\begin{crl}{}{}\\
Let $B$ be a paracompact space. Let $p:E\to B$ be a vector bundle. If $B$ is contractible, then $E$ is trivial. 
\end{crl}

\subsection{Almost Complex Structures}
\begin{defn}{Linear Complex Structure}{}\\
Let $V$ be a vector space over $\R$. A linear complex structure on $V$ is a linear map $J:V\to V$ such that $J^2=-\text{id}_V$. 
\end{defn}

\begin{prp}{}{}\\
Let $V$ be a vector space over $\R$. Let $J$ be a linear complex structure on $V$. Then $V$ is a $\C$-vector space whose action is given by $$(x+iy)\cdot v=xv+yJ(v)$$ for $v\in V$ and $x+iy\in\C$. 
\end{prp}

\begin{prp}{}{}\\
Let $V$ be a finite dimensional vector space over $\R$. Then the following are true. 
\begin{itemize}
\item $V$ admits a linear complex structure if and only if $\dim_\R(V)$ is even. 
\item In this case, we have $\dim_\C(V)=2\dim_\R(V)$. 
\end{itemize}
\end{prp}

\begin{eg}{}{}\\
Let $J:\R^{2n}\to\R^{2n}$ be the map defined in block matrix form $$J=\begin{pmatrix}
0 & -I_n\\
I_n & 0
\end{pmatrix}$$
Then $J$ defines a linear complex structure on $\R^{2n}$. In particular, if $e_1,\dots,e_{2n}$ are the standard basis vectors for $\R^{2n}$, then $\frac{1}{2}(e_1-ie_{2n+1}),\dots,\frac{1}{2}(e_n-ie_{2n})$ is a basis for $V$ as a $\C$-vector space. 
\end{eg}

\begin{prp}{}{}\\
Let $V$ be a vector space over $\R$. Let $J$ be a linear complex structure on $V$. Let $T:V\to V$ be an $\R$-linear transformation. Then $T$ is a $\C$-linear transformation if and only if $J\circ T=T\circ J$. 
\end{prp}

The linear complex structure on an $\R$-vector space gives an intrinsic way of giving the $\R$-vector space the structure of a $\C$-vector space. However, we already know an external way of giving $\R$-vector spaces $\C$-multiplication: namely by extension of scalars. 

\begin{prp}{}{}\\
Let $V$ be a vector space over $\R$. Let $J$ be a linear complex structure on $V$. Then the following are true. 
\begin{itemize}
\item $J$ extends to a linear complex structure on $V\otimes_\R\C$ by $J(v\otimes z)=J(v)\otimes z$. 
\item The eigenvalues of $J:V\otimes_\R\C\to V\otimes_\R\C$ is given by $\pm i$ and the eigenspaces of $J$ are given by $$E_i=\{v\otimes 1-J(v)\otimes i\;|\;v\in V\}\;\;\;\text{ and }\;\;\;E_{-i}=\{v\otimes 1+J(v)\otimes i\;|\;v\in V\}$$
\item There is a natural isomorphism of $\C$-vector space $V\cong E_i$ given by $v\mapsto v\otimes 1-J(v)\otimes i$. 
\item The conjugation map $\overline{(-)}:V\otimes_R\C\to V\otimes_\R\C$ given by $\overline{v\otimes z}=v\otimes\overline{z}$ restricts to an isomorphism $$\overline{(-)}|_{E_i}:E_i\overset{\cong}{\to}E_{-i}$$ whose inverse is given by $\overline{(-)}|_{E_{-i}}$. 
\end{itemize}
\end{prp}

\begin{prp}{}{}\\
Let $V$ be a vector space over $\R$. Let $J$ be a linear complex structure on $V$. Then $J^T$ defines a linear complex structure on $V^\ast$. 
\end{prp}

\begin{defn}{Almost Complex Structure}{}\\
Let $B$ be a space. Let $p:E\to B$ be a vector bundle over $\R$. An almost complex structure on $p$ is a vector bundle isomorphism $j:E\to E$ such that $j\circ j=\text{id}$. 
\end{defn}

\begin{prp}{}{}\\
Let $B$ be a space. Let $p:E\to B$ be a vector bundle over $\R$. Let $J$ be an almost complex structure on $p$. Then the following are true. 
\begin{itemize}
\item The rank of $E$ over $\R$ is even. 
\item $E$ is a complex vector bundle whose action is given by $$(x+iy)\cdot v=xv+yJ_b(v)$$ for $v\in E_b$ and $x+iy\in\C$ for each $b\in B$. 
\end{itemize}
\end{prp}

\begin{prp}{}{}\\
Let $B$ be a space. Let $p:E\to B$ be a vector bundle over $\C$ of rank $n$. Then $E$ is a real vector bundle of rank $2n$. 
\end{prp}

\section{Sequences of Vector Bundles}
\subsection{Subbundles and Quotient Bundles}
\begin{defn}{Vector Subbundles}{}\\
Let $F$ be a field. Let $B$ be a space. Let $p:E\to B$ be a vector bundle over $F$. A subbundle of $E$ is a subset $E'\subseteq E$ such that $p|_{E'}:E'\to B$ is a vector bundle over $F$. 
\end{defn}

\begin{lmm}{}{}\\
Let $F$ be a field. Let $B$ be a space. Let $p:E\to B$ be a vector bundle over $F$. Let $E'$ be a subbundle of $E$. Let $\phi_i:p^{-1}(U_i)\to U_i\times F^k$ be a local chart of $E'$. Then the following are true. 
\begin{itemize}
\item There exists a map $\psi_i:p^{-1}(U_i)\to U_i\times F^{n-k}$ such that the map $p^{-1}(U_i)\to U_i\times F^n$ given by $x\mapsto(x,\text{proj}_{F^k}(\phi_i(x)),\text{proj}_{F^{n-k}}(\psi_i(x)))$ is a local chart of $E$. 
\item The transition functions of $E$ are given in block matrix $$g_{ij}=\begin{pmatrix}
g_{ij}' & l_{ij}\\
0 & k_{ij}
\end{pmatrix}$$
for some maps $l_{ij}:U_i\cap U_j\to M_{k,n-k}(F)$ and $k_{ij}:U_i\cap U_j\to\text{GL}(n-k,F)$. 
\item The maps $k_{ij}$ satisfy the cocycle condition and $k_{ii}(x)=I_{n-k}$ for all $x\in U_i$. 
\end{itemize}
\end{lmm}

\begin{defn}{The Quotient Bundle}{}\\
Let $F$ be a field. Let $B$ be a space. Let $p:E\to B$ be a vector bundle over $F$. Let $E'$ be a subbundle of $E$. Define the quotient bundle $E/E'$ to be the associated vector bundle to the collection of functions $k_{ij}$ given by the above. 
\end{defn}

\begin{lmm}{}{}\\
Let $F$ be a field. Let $B$ be a space. Let $p:E\to B$ be a vector bundle over $F$. Let $E'$ be a subbundle of $E$. There are linear isomorphisms $$(E/E')_b\cong E_b/E_b'$$ for all $b\in B$. 
\end{lmm}

\begin{prp}{Universal Property of Quotient Bundles}{}\\
Let $F$ be a field. Let $B$ be a space. Let $p:E\to B$ be a vector bundle over $F$. Let $E'$ be a subbundle of $E$. Then the quotient bundle and the quotient map $q:E\to E/E'$ satisfies the following universal property. For any vector bundle $G$ over $F$ and any morphism $f:E\to G$ such that $f(E')$ is the trivial bundle, there exists a unique morphism $\widetilde{f}:E/E'\to G$ such that the following diagram commutes: \\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	E & {E/E'} \\
	& G
	\arrow["q", from=1-1, to=1-2]
	\arrow["f"', from=1-1, to=2-2]
	\arrow["{\widetilde{f}}", from=1-2, to=2-2]
\end{tikzcd}}\\
Moreover, $E/E'$ is the unique (up to unique isomorphism) vector bundle that satisfies the above. 
\end{prp}

\subsection{Short Exact Sequences of Vector Bundles}
\begin{defn}{Short Exact Sequences of Vector Bundles}{}\\
Let $F$ be a field. Let $B$ be a space. Let $E,G,H$ be vector bundles over $B$. Let $f:E\to G$ and $g:G\to H$ be morphisms. We say that the sequence \\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	0 & E & G & H & 0
	\arrow[from=1-1, to=1-2]
	\arrow["f", from=1-2, to=1-3]
	\arrow["g", from=1-3, to=1-4]
	\arrow[from=1-4, to=1-5]
\end{tikzcd}}\\
is a short exact sequence of vector bundles if the induced linear maps \\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	0 & {E_b} & {G_b} & {H_b} & 0
	\arrow[from=1-1, to=1-2]
	\arrow["{f|_{E_b}}", from=1-2, to=1-3]
	\arrow["{g|_{E_b}}", from=1-3, to=1-4]
	\arrow[from=1-4, to=1-5]
\end{tikzcd}}\\
are short exact sequences of vector spaces for all $b\in B$. 
\end{defn}

\begin{lmm}{}{}\\
Let $F$ be a field. Let $B$ be a space. Suppose that the following is a short exact sequence of vector bundles over $B$: \\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	0 & E & G & H & 0

\section{Metrics on Vector Bundles}
\subsection{}
\begin{defn}{Metrics}{}\\
Let $F$ be a field. Let $B$ be a space. Let $p:E\to B$ be a vector bundle. 
\begin{itemize}
\item Suppose that $F=\R$. A metric on $E$ is a smooth section of $s\in(E\otimes E)^\ast\cong E^\ast\otimes E^\ast$ such that $s_p:E_b\times E_b\to\R$ is an inner product for all $b\in B$. 
\item Suppose that $F=\C$. A metric on $E$ is a smooth section of $s\in(E\otimes E)^\ast\cong E^\ast\otimes E^\ast$ such that $s_p:E_b\times E_b\to\R$ is a Hermitian product for all $b\in B$. 
\end{itemize}
\end{defn}

\begin{lmm}{}{}\\
Let $F=\R$ or $\C$. Let $B$ be a space. Let $p:E\to B$ be a vector bundle. If $B$ is paracompact, then $E$ admits a metric. 
\end{lmm}

Note: ses of vector bundles of paracompact spaces are split exact. 
Note: vector bundles have reduction of structure group to $O(n)$ and $U(n)$ when they admit a metric.

\section{The Topology of Fiber Bundles}
\subsection{Fiber Bundles}
Fiber bundles serve as somewhat of a generalization of both vector bundles and covering spaces, while being a special case of a fibration. It therefore has the properties of a fibration. 

\begin{defn}{Fiber Bundles}{}\\
Let $E,B,F$ be spaces with $B$ connected, and $p:E\to B$ a continuous map. We say that $p$ is a fiber bundle over $F$ if the following are true. 
\begin{itemize}
\item $p:E\to B$ is surjective
\item Local Triviality: There exists a collection $$\{(U_k\subseteq B,\phi_k:p^{-1}(U_k)\overset{\cong}{\to}U_k\times F)\;|\;k\in I\}$$ called local trivialization charts such that the following diagram commutes: \\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	{p^{-1}(U_k)} && {U_k\times F} \\
	& U_k
	\arrow["{\phi_k}", from=1-1, to=1-3]
	\arrow["p"', from=1-1, to=2-2]
	\arrow["\pi", from=1-3, to=2-2]
\end{tikzcd}}\\
where $\pi$ is the projection by forgetting the second variable. 
\end{itemize}
We say that $B$ is the base space, $E$ the total space. It is denoted as $(F,E,B)$ or $F\hookrightarrow E\to B$. 
\end{defn}

Intuitively, we would like a fiber bundle to locally look like the product $B\times F$. 

\begin{lmm}{}{}\\
Let $K$ be a field. Let $p:E\to B$ be a fiber bundle with fiber $F$. Then the local trivialization charts induces a homeomorhpsim $$p^{-1}(b)\cong F$$ for all $b\in B$. 
\end{lmm}

Fiber bundles generalizes vector bundles in the sense that the fibers are no longer vector spaces but instead arbitrary spaces. 

\begin{lmm}{}{}\\
The following are true regarding fiber bundles. 
\begin{itemize}
\item Every vector bundle is a fiber bundle. 
\item Every covering space is a fiber bundle. 
\end{itemize}
\begin{proof}
Indeed if $p:E\to B$ is a vector bundle, then each fiber $p^{-1}(b)$ is an $n$-dimensional vector spaces over a field $F$. Moreover, by definition the local triviality condition is also satisfied. \\

Finally, for any $U\subseteq X$, recall that $$p^{-1}(U)=\coprod_{i\in I}V_i$$ where each $V_i\cong U$. It is clear by definition that $\abs{p^{-1}(x)}\abs{I}$ for any $x\in X$. By giving $I$ the discrete topology, we obtain a homeomorphism $p^{-1}(x)\cong I$. The homeomorphism $p^{-1}(U)=\coprod_{i\in I}V_i$ translates to $$p^{-1}(U)=\coprod_{i\in I}V_i\cong\coprod_{i\in I}U\cong U\times I$$ defined by $\tilde{x}\in V_i\mapsto(p(\tilde{x})=x,i)$. It is thus clear that the local triviality condition is satisfied. 
\end{proof}
\end{lmm}

\begin{eg}{}{}\\
The following maps are fiber bundles. 
\begin{itemize}
\item Let $f_n:S^1\to S^1$ be the map defined by $f_n(z)=z^n$. Then $f_n$ is a fiber bundle with fiber the $n$th roots of unity. 
\item Let $f:\R\to S^1$ be the map defined by $f(t)=e^{2\pi it}$. Then $f$ is a fiber bundle with fiber $\Z$. 
\item Let $p:S^n\to\R\Prj^n$ be the projection map. Then $p$ is a fiber bundle with fiber as a two point set with the discrete topology. 
\end{itemize}
\begin{proof}\\
Indeed, all of these are covering spaces and hence fiber bundles by the above lemma. 
\end{proof}
\end{eg}

\begin{eg}{}{}\\
Let $M=\frac{[-1,1]\times[-1,1]}{(t,-1)\sim(1-t,1)}$ be the Mobius strip. Then the projection map $p:M\to C$ defined by $p(t,s)=s$ is a fiber bundle with fiber $[-1,1]$. 
\end{eg}

\begin{prp}{}{} Every fiber bundle is a Serre fibration. 
\end{prp}

We can provide a partial converse for the fact that every fiber bundle is a Serre fibration. 

\begin{prp}{}{} Let $p:E\to B$ be a fiber bundle. If $B$ is paracompact, then $p$ is a (Hurewicz) fibration. 
\end{prp}

We there fore have inclusions $$\substack{\text{Fiber}\\\text{Bundles}}\subset\substack{\text{Serre}\\\text{Fibrations}}\subset\substack{\text{(Hurewicz)}\\\text{Fibrations}}$$

\begin{defn}{Map of Fiber Bundles}{}\\
Let $(F_1,E_1,B_1)$ and $(F_2,E_2,B_2)$ be fiber bundles. A map of fiber bundles is a pair of basepoint preserving continuous maps $(\tilde{f}:E_1\to E_2,f:B_1\to B_2)$ such that the following diagram commutes: \\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	{E_1} & {E_2} \\
	{B_1} & {B_2}
	\arrow["{\tilde{f}}", from=1-1, to=1-2]
	\arrow["{p_1}"', from=1-1, to=2-1]
	\arrow["{p_2}", from=1-2, to=2-2]
	\arrow["f"', from=2-1, to=2-2]
\end{tikzcd}}\\
A map of fiber bundles $(\tilde{f},f)$ is said to be an isomorphism if there is a map $(\tilde{g}:E_2\to E_1,g:B_2\to B_1)$ such that $\tilde{g}$ is the inverse of $\tilde{f}$ and $g$ is the inverse of $f$. 
\end{defn}

Such a map of fiber bundles determine a continuous of the fibers $F_1\cong p_1^{-1}(b_1)\to p_2^{-1}(b_2)\cong F_2$. Moreover, such a morphism preserves fibers. Indeed, If $p_1^{-1}(b)$ is a fiber of $B$, then using the commutativity of the diagram we have that $$p_2(\overline{f}(p_1^{-1}(b)))=f(p_1(p_1^{-1}(b)))=f(b)$$ which implies that $$p_2^{-1}(f(b))=\overline{f}(p^{-1}(b))$$ or in other words, the fiber at $f(b)$ is the same as the fiber at $b$ applied with $\overline{f}$. 

\begin{defn}{Isomorphic Fiber Bundles}{}\\
Let $p_1:E_1\to B_1$ and $p_2:E_2\to B_2$ be two fiber bundles. Let $(\widetilde{f},f):p_1\to p_2$ be a map of fiber bundles. We say that $(\widetilde{f},f)$ is an isomorphism if there exists a map $(\widetilde{g},g):p_2\to p_1$ of fiber bundles such that $\widetilde{f}$ is the inverse of $\widetilde{g}$ and $f$ is the inverse of $g$. In this case, we say that $p_1$ and $p_2$ are isomorphic. 
\end{defn}

There are two important special cases of fiber bundles that will appear time and time again. 

\begin{defn}{Trivial Bundles}{}\\
We say that a fiber bundle $(F,E,B)$ is trivial if $(F,E,B)$ is isomorphic to the trivial fibration $B\times F\to B$. 
\end{defn}

\begin{defn}{The Pullback Bundle}{} Let $p:E\to B$ be a fiber bundle with fiber $F$. Let $f:B'\to B$ be a continuous function. Define the pullback of $p$ by $f$ to be the space $$f^\ast(E)=\{(b',e)\in B'\times E\;|\;p(e)=f(b')\}$$
\end{defn}

\begin{thm}{}{} Let $p:E\to B$ be a fiber bundle. Suppose that $f,g:X\to B$ are homotopic maps. Then the pull back bundles $$f^\ast(E)\cong g^\ast(E)$$ are equivalent. 
\end{thm}

\subsection{Sections of a Bundle}
\begin{defn}{Sections}{} Let $(F,E,B)$ be a fiber bundle. A section on the fiber bundle is a map $s:B\to E$ such that $$p\circ s=\text{id}_B$$
\end{defn}

\begin{defn}{Local Sections}{} Let $(F,E,B)$ be a fiber bundle. Let $U\subset B$ be an open set. A local section of the fiber bundle on $U$ is a map $s:U\to B$ such that $$p\circ s=\text{id}_U$$
\end{defn}

\subsection{Sphere Bundles}
We now consider a special type of fibrations where the fibers are given by $S^1$. We define a class of fiber bundles of the form $S^1\to S^{2n+1}\to\C\Prj^n$. When $n=1$, this is the Hopf fibration. 

\begin{defn}{Sphere Bundles}{} A sphere bundle is a fiber bundle $p:E\to B$ for which its fibers are the $n$-sphere $S^n$. 
\end{defn}

\begin{eg}{}{}\\
Let $n\in\N$. Consider $S^{2n+1}$ lying inside $\C^{n+1}$. The projection map $\C^{n+1}\setminus\{0\}\to\C\Prj^n$ induces a fiber bundle $$S^1\hookrightarrow S^{2n+1}\to\C\Prj^n$$
\begin{proof}\\
Clearly the projection map $p:S^{2n+1}\to\C\Prj^n$ is surjective. Consider the open cover $\{U_i\;|\;0\leq i\leq n\}$ where $U_i=\{[z_0:\cdots:z_n]\;|\;z_i\neq 0\}$. Define the local trivialization charts $\phi_i:p^{-1}(U_i)\to U_i\times S^1$ by $(z_0,\dots,z_n)\to([z_0:\cdots:z_n],z_i/\abs{z_i})$. It is continuous since it is the product of the composition map and the projection map to the sphere $S^1$. Its inverse is the continuous map $([z_0,\dots,z_n],w)=w\abs{z_i}z_i^{-1}(z_0,\dots,z_n)$ since it is induced by the universal property of quotients. It is clear that the required diagram commutes, that is $p=\text{proj}\circ\phi_i$ where $\text{proj}:U_i\times S^1\to U_i$ is the projection map to the first coordinate. 
\end{proof}
\end{eg}

\begin{eg}{}{}\\
For each $n\in\N$, we have a map of fiber bundles: \\~\\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	{S^{2n+1}} & {S^{2n+3}} \\
	{\C\Prj^n} & {\C\Prj^{n+1}}
	\arrow[hook, from=1-1, to=1-2]
	\arrow[from=1-1, to=2-1]
	\arrow[from=1-2, to=2-2]
	\arrow[hook, from=2-1, to=2-2]
\end{tikzcd}}\\whose horizontal maps are induced by the inclusion $\C^n\hookrightarrow\C^{n+1}$ into the first $n$ coordinates. Taking colimit of the diagram, we have that $$S^1\hookrightarrow S^\infty\to\C\Prj^\infty$$ is a fiber bundle. 
\begin{proof}\\
Since the local triviality condition is a local condition, for each $x\in\C\Prj^\infty$, $x$ lives in $\C\Prj^n$ for some $n\in\N$ and since $S^1\hookrightarrow S^{2n+1}\to\C\Prj^n$ is a fiber bundle, there exists a chart for which the local triviality condition is satisfied. 
\end{proof}
\end{eg}

\begin{eg}{Hopf Fibration / Hopf Bundle}{}\\
Consider the above example with $n=1$. Restricting the total space to $S^3$, the fiber bundle $p:S^3\to S^2$ with fiber $S^1$ is called the Hopf fibration / Hopf bundle. 
\end{eg}
