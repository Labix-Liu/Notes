\section{Triangulated Manifolds}
\begin{defn}{Triangulated Manifolds}{}\\
Let $M$ be a topological manifold. We say that $M$ is a triangulated manifold if $M$ is triangulated as a topological space. 
\end{defn}

\begin{thm}{Edwards}{}\\
Let $n\geq 5$. Let $\mK$ be a simplicial complex of dimension $n$. Then $\abs{\mK}$ is a topological manifold of dimension $n$ if and only if the following are true. 
\begin{itemize}
\item $H_\ast(\text{lk}(\sigma))\cong H_\ast(S^{n-\abs{\sigma}})$ for all $\emptyset\neq\sigma\in\mK$. 
\item $\text{lk}(v)$ is simply connected for all $v\in V(\mK)$. 
\end{itemize}
\end{thm}

manifold Hauptvermutung: if $\mK$ and $\mL$ are simplicial complexes such that $\abs{\mK}$ and $\abs{\mL}$ are homeomorphic triangulated manifold, then is $\mK$ combinatorially equivalent to $\mK$? This is proven to be false for dimension $\geq 5$. 

\subsection{Piecewise Linear Manifolds}
\begin{prp}{}{}\\
The set $$\Gamma=\{f:U\to V\;|\;U,V\subseteq\R^n\text{ are open }, f\text{ is a PL homeomorphism }\}$$ defines a pseudogroup of transformations on $\R^n$. 
\end{prp}

\begin{defn}{Piecewise Linear Manifolds}{}\\
A piecewise linear manifold is a manifold whose pseudogroup of transformations is given by $\Gamma$. 
\end{defn}

\subsection{Combinatorial Manifolds}
\begin{defn}{Combinatorial Manifolds}{}\\
Let $M$ be a topological manifold. We say that $M$ is a combinatorial manifold if there exists a simplicial complex $\mK$ and a homeomorphism $\abs{\mK}\cong X$ such that $$\text{lk}(\sigma)\cong_\text{PL}\partial\Delta^{\dim(M)-\abs{\sigma}+1}\;\;\text{ or }\;\;\Delta^{\dim(M)-\abs{\sigma}}$$ for all $\sigma\in\mK$. 
\end{defn}

\begin{lmm}{}{}\\
Let $\mK$ be a simplicial complex. Suppose that there exists $n\in\N^+$ such that $$\text{lk}(\sigma)\cong_\text{PL}\partial\Delta^{n-\abs{\sigma}+1}\;\;\text{ or }\;\;\Delta^{n-\abs{\sigma}}$$ for all $\sigma\in\mK$. Then $\abs{\mK}$ is a piecewise linear manifold of dimension $n$. 
\end{lmm}

\begin{thm}{Whitehead 1940}{}\\
Let $M$ be a piecewise linear manifold. Then $M$ is a combinatorial manifold. 
\end{thm}

\begin{thm}{PL Hauptvermutung for combinatorial equivalence}{}\\
Let $\mK$ and $\mL$ be simplicial complexes such that $\abs{\mK}$ and $\abs{\mL}$ are combinatorial manifolds. Then $\abs{\mK}\cong_\text{PL}\abs{\mK}$ if and only if $\mK$ and $\mL$ are combinatorially equivalent. 
\end{thm}

\subsection{Pseudo-Manifolds}
\begin{defn}{Pseudo-Manifolds From Simplicial Complexes}{}\\
Let $\mK$ be a simplicial complex. We say that $\mK$ is a pseudo-manifold of dimension $n$ if the following are true. 
\begin{itemize}
\item All maximal simplicies of $\mK$ are of dimension $n$. 
\item Each $(n-1)$-simplex of $\mK$ is the face of two $n$-simplicies of $\mK$. 
\item For any two $n$-simplicies $\sigma,\sigma'\in\mK$, there exists $n$-simplicies $\sigma=\sigma_0,\dots,\sigma_k=\sigma'$ such that $\sigma_i\cap\sigma_{i+1}$ is an $(n-1)$-simplex for $0\leq i\leq k-1$. 
\end{itemize}
\end{defn}

\begin{defn}{Pseudo-Manifolds in General}{}\\
Let $X$ be a space. We say that $X$ is a pseudo-manifold if $X$ is triangulated by a simplicial complex $\mK$ that is a pseudo-manifold. 
\end{defn}

\begin{lmm}{}{}\\
Let $X$ be a pseudo-manifold of dimension $n$. Then $H_n(X)\cong\Z$ or $0$. 
\end{lmm}

\begin{lmm}{}{}\\
Let $\mK$ be a triangulation of $S^n$. Then the following are true. 
\begin{itemize}
\item $\mK$ is a pseudo-manifold. 
\item If $\mK$ is finite, then the generator of $H_n(\mK)$ is represented by the sum of $n$-simplicies of $\mK$. 
\end{itemize}
\end{lmm}
