\section{Riemannian Metrics}
\subsection{The Riemannian Metric}
\begin{defn}{Riemannian Metric}{}\\
Let $M$ be a smooth manifold. A Riemannian metric on $M$ is a smooth section $$g\in\mC^\infty(M,(TM\otimes TM)^\ast)$$ such that $g_p:T_p(M)\times T_p(M)\to\R$ is an inner product for all $p\in M$. 
\end{defn}

Equivalently, a Riemannian metric is an assignment $p\mapsto g_p$ such that for any smooth sections $s,t\in\Gamma(M,TM)$, the map $p\mapsto g_p(s(p),t(p))$ is smooth. 

\begin{defn}{Riemannian Manifold}{} A Riemannian manifold $(M,g)$ is a manifold $M$ together with a Riemannian metric $g$ on $M$. 
\end{defn}

To describe the Riemannian metric in local coordinates, we need some simple algebra. Let $V$ be a finite dimensional vector space over a field $k$. Let $\{e_1,\dots,e_n\}$ be a basis for $V$. Let $\{e^1,\dots,e^n\}$ be the dual basis for $V^\ast$. Then we may establish a canonical isomorphism $$(V\otimes_kV)^\ast\cong V^\ast\otimes_kV^\ast$$ given by $f(-,-)\mapsto\sum_{i,j=1}^nf(e_i\otimes e_j)e^i\otimes e^j$. Its converse is given by $\varphi\otimes\psi\mapsto((v\otimes w)\mapsto\varphi(v)\psi(w))$. 

Then isomorphism carries over to vector bundles, and so we have an isomorphism $(TM\otimes TM)^\ast\cong TM^\ast\otimes TM^\ast$. This is how the Riemannian metric is a $(0,2)$-tensor field. 

\begin{prp}{}{}\\
Let $(M,g)$ be a Riemannian manifold. Let $p\in M$. Let $(U,x^1,\dots,x^n)$ be a chart at $p$. Then $g_p$ as an element of $T_p(M)^\ast\otimes T_p(M)^\ast$ is given locally on the chart by $$g_p=\sum_{1\leq i,j\leq n}g_{i,j}(p)\;(dx^i)|_p\otimes(dx^j)_p$$ where $g_{i,j}:U\to\R$ are smooth functions such that $g_{i,j}(p)=g_p\left(\frac{\partial}{\partial x^i}\bigg{|}_p,\frac{\partial}{\partial x^j}\bigg{|}_p\right)$. 
\end{prp}

Let $v,w\in T_p(M)$ be tangent vectors. Then locally on the chart $(U,x^1,\dots,x^n)$, we may compute the Riemannian metric by $$g_p(v,w)=\sum_{1\leq i,j\leq n}g_{i,j}(p)(dx^i)|_p(v)(dx^j)|_p(w)$$

\begin{thm}{}{}\\
Every smooth manifold admits a Riemannian metric and hence is a Riemannian manifold. 
\end{thm}

\begin{defn}{Isometries}{} Let $(M,g)$ and $(N,h)$ be two Riemannian manifolds. We say that $(M,g)$ and $(N,h)$ are isometric if there exists a diffeomorphisms $f:M\to N$ such that $$h\circ f=g$$ In this case $f$ is said to be an isometry. 
\end{defn}

\begin{thm}{(Nash)}{}\\
Let $(M,g)$ be a Riemannian manifold of dimension $n$. Then there exists an isometric embedding $f:M\to\R^{(n+2)(n+3)/2}$ with the standard Euclidean metric. 
\end{thm}

\begin{defn}{Local Isometries}{} Let $(M,g)$ and $(N,h)$ be two Riemannian manifolds. We say that they are locally isometric if for all $p\in M$, there exists an open neighbourhood $U\subseteq M$ of $p$ and $V\subseteq N$ open and an isometry $f:U\to V$. 
\end{defn}

\begin{defn}{Flat Manifolds}{} Let $(M,g)$ be a Riemannian manifold. We say that $(M,g)$ is flat if it is locally isometric to $\R^n$ with the standard metric. 
\end{defn}

In general, not every Riemannian manifold is flat. This can be shown once we discuss curvatures and torsions. However, this is true when $n=1$. 

\begin{lmm}{}{} Every 1 dimensional Riemannian manifold is flat. 
\end{lmm}

\subsection{Lengths and Angles}
\begin{defn}{Length of a Tangent Vector}{} Let $(M,g)$ be a Riemannian manifold. Let $v\in T_p(M)$ be a tangent vector for $p\in M$. Define the length of $v$ to be $$\abs{v}_g=\sqrt{g_p(v,v)}$$
\end{defn}

\begin{defn}{Angle between two Tangent Vectors}{} Let $(M,g)$ be a Riemannian manifold. Let $p\in M$. For $v,w\in T_pM$ two tangent vectors, define the angle between $v$ and $w$ to be the unique $\theta\in[0,\pi]$ such that $$\cos(\theta)=\frac{g_p(v,w)}{\abs{v}_g\abs{w}_g}$$
\end{defn}

\begin{defn}{Orthogonal Tangent Vectors}{} Let $(M,g)$ be a Riemannian manifold. Let $p\in M$. We say that two tangent vectors $v,w\in T_pM$ are orthogonal if $$g_p(v,w)=0$$
\end{defn}

\subsection{The Musical Isomorphisms}
\begin{defn}{The Flat Map}{} Let $(M,g)$ be a Riemannian manifold. Let $p\in M$. For each $X\in T_pM$, define the flat map $$\flat:T_pM\to T_p^\ast M$$ by sending $X\in T_pM$ to the map $X^\flat:T_pM\to\R$ by $X^\flat(Y)=g_p(X,Y)$. 
\end{defn}

\begin{thm}{The Musical Isomorphism}{} Let $(M,g)$ be a Riemannian manifold. Let $p\in M$. Then the flat map $$\flat:T_pM\to T_p^\ast M$$ is an isomorphism. 
\end{thm}

\begin{defn}{The Sharp Map}{} Let $(M,g)$ be a Riemannian manifold. Let $p\in M$. Define the sharp map $$\#:T_p^\ast M\to T_pM$$ to be the inverse of the flat map. 
\end{defn}

\begin{eg}{}{}\\
Let $(M,g)$ be a Riemannian manifold. Let $p\in U$. Let $(U,x^1,\dots,x^n)$ be a chart at $p$. Let $g$ be given locally at the chart by $$g_p=\sum_{1\leq i,j\leq n}g_{i,j}(p)(dx^i)_p\otimes(dx^j)_p$$ Then the following are true. 
\begin{itemize}
\item The flat map is given locally by $$\left(\frac{\partial}{\partial x^i}\bigg{|}_p\right)^\flat=\sum_{j=1}^ng_{i,j}(p)(dx^j)_p$$
\item The sharp map is given locally by $$\left((dx^i)_p\right)^\#=\sum_{j=1}^ng^{i,j}(p)\frac{\partial}{\partial x^j}\bigg{|}_p$$ where $(g^{i,j}(p))_{1\leq i,j\leq n}=\left((g_{i,j}(p))_{1\leq i,j\leq n}\right)^{-1}$. 
\end{itemize}
\end{eg}

\subsection{The Induced Metric on Cotangent Bundles}
\begin{defn}{The Induced Metric on the Cotangent Bundle}{}\\
Let $(M,g)$ be a Riemannian manifold. Define the induced metric on $T^\ast M$ to be the section $g^{-1}\in\Gamma(M,(T^\ast M\otimes T^\ast M)^\ast)$ by $$g^{-1}(\alpha,\beta)=g(\alpha^\#,\beta^\#)$$
\end{defn}

\begin{prp}{}{}\\
Let $(M,g)$ be a Riemannian manifold. Let $(U,x^1,\dots,x^n)$ be a chart of $M$. Suppose that the components of $g^{-1}$ on the local chart is given by $g^{i,j}$. Then we have $$(g^{i,j})=(g_{i,j})^{-1}$$
\end{prp}

\subsection{Riemannain Manifolds as a Metric Space}
\begin{defn}{Piecewise Regular Paths}{}\\
Let $(M,g)$ be a Riemannian manifold. A piecewise regular path in $M$ consists of smooth functions $\gamma_i:[a_i,b_i]\to M$ for $1\leq i\leq k$ such that the following are true. 
\begin{itemize}
\item For $1\leq i\leq k-1$, we have $\gamma_i(b_i)=\gamma_{i+1}(a_i)$. 
\item For $1\leq i\leq k$, we have $\abs{\gamma_i'(t)}\neq 0$ for all $t\in[a_i,b_i]$. 
\end{itemize}
\end{defn}

\begin{defn}{Length of a Curve}{}\\
Let $(M,g)$ be a Riemannian manifold. Let $\gamma:[a,b]\to M$ be a piecewise regular path in $M$. Define the length of $\gamma$ to be $$L(\gamma)=\int_a^b\abs{\gamma'(t)}_g\;dt$$
\end{defn}

\begin{defn}{Riemannian Distance Function}{}\\
Let $(M,g)$ be a Riemannian manifold. Define the Riemannian distance function of $M$ to be the function $d_g:M\times M\to\R$ given by $$d_g(p,q)=\inf\{L(\gamma)\;|\;\gamma:I\to M\text{ is a piecewise regular path }\}$$
\end{defn}

\begin{prp}{}{}\\
Let $(M,g)$ be a Riemannian manifold. Then the following are true. 
\begin{itemize}
\item $d_g$ is a metric on $M$. 
\item The topology of $M$ is given by the metric topology induced by $d_g$. 
\end{itemize}
\end{prp}

\section{Introduction to Connections}
\subsection{Connections on Vector Bundles}
Recall that for a smooth vector bundle $p:E\to M$, we denote the space of smooth sections on $E$ by $\Gamma(E)$. Moreover, $\mathfrak{X}(M)$ is the space of smooth sections on the tangent bundle $TM$. 

\begin{defn}{Connections}{}\\
Let $M$ be a smooth manifold. Let $p:E\to M$ be a smooth vector bundle. A connection on $p$ is a map $$\nabla:\mathfrak{X}(M)\times\mC^\infty(E)\to\mC^\infty(E)$$ where we denote $\nabla(V,T)$ by $\nabla_V(T)$, such that the following are true. 
\begin{itemize}
\item $C^\infty(M)$-linearity in first variable: For each $T\in\mC^\infty(E)$, the map $V\mapsto\nabla_V(T)$ is $C^\infty(M)$-linear. This means that $$\nabla_{fV+hW}(T)=f\nabla_V(T)+g\nabla_W(T)$$ for $V,W\in\mathfrak{X}(M)$, $f,g\in C^\infty(M)$. 
\item $\R$-linearity in second variable: For each $V\in\mathfrak{X}(M)$, the map $T\mapsto\nabla_V(T)$ is $\R$-linear. This means that $$\nabla_V(\lambda T+\mu S)=\lambda\nabla_V(T)+\mu\nabla_V(S)$$
\item Product rule: The map $\nabla$ satisfies the following product rule: $$\nabla_V(fT)=V(f)\cdot T+f\nabla_V(T)$$
\end{itemize}
\end{defn}

\begin{prp}{}{}\\
Let $M$ be a smooth manifold. Let $p:E\to M$ be a smooth vector bundle. Then $E$ admits a connection. 
\end{prp}

\begin{defn}{Set of Connections}{}\\
Let $M$ be a smooth manifold. Let $p:E\to M$ be a smooth vector bundle. Define the set of connections of $E$ to be $$\mA(E)=\{\nabla:\mathfrak{X}(M)\times\mC^\infty(E)\to\mC^\infty(E)\;|\;\nabla\text{ is a connection }\}$$
\end{defn}

\begin{prp}{}{}\\
Let $M$ be a smooth manifold. Let $p:E\to M$ be a smooth vector bundle. A connection $\nabla$ on $E$ is equivalently an $\R$-linear map $$\nabla:\mC^\infty(E)\to\mC^\infty(T^\ast M\otimes E)=\Omega^1(M)\otimes_{\mC^\infty(M)}\mC^\infty(E)$$ such that $$\nabla(f\cdot s)=df\otimes s+f\cdot\nabla s$$ for all $f\in\mC^\infty(M)$ and $s\in\mC^\infty(E)$. 
\end{prp}

\begin{defn}{The Connection Forms}{}\\
Let $M$ be a smooth manifold. Let $p:E\to M$ be a smooth vector bundle. Let $\nabla$ be a connection on $E$. Let $U\subseteq M$ be an open subset. Let $e_1,\dots,e_r$ be a local frame of $E$ on $U$. Define the connection $1$-forms of $\nabla$ on $U$ to be the unique $1$-forms $\omega_j^i:TU\to\R$ such that $$\nabla_Xe_j=\sum_{i=1}^r\omega_j^i(X)e_j$$ for all $X\in\mathfrak{U}$. 
\end{defn}

\begin{prp}{}{}\\
Let $M$ be a smooth manifold. Let $p:E\to M$ be a smooth vector bundle. Let $\nabla$ be a connection on $E$. Let $U\subseteq M$ be an open subset. Let $e,\overline{e}$ be local frames of $E$ on $U$. Suppose that $g:U\to\text{GL}(r,\R)$ is a transition function such that $\overline{e}=ge$. Suppose that $\omega_j^i$ and $\overline{\omega}_j^i$ are the connection forms of $\nabla$ with respect to $e$ and $\overline{e}$. Then we have $$\overline{\omega}_j^i=g^{-1}\omega g+g^{-1}dg$$
\end{prp}

\subsection{Directional Derivatives of a Vector Bundle}
\begin{defn}{Covariant Derivatives Along a Curve}{}\\
Let $M$ be a smooth manifold. Let $p:E\to M$ be a smooth vector bundle. Let $\gamma:[0,1]\to M$ be a smooth curve. Let $\nabla$ be a connection on $E$. Define the covariant derivative $D_t:\mC^\infty(\gamma^\ast(E))\to\mC^\infty(\gamma^\ast(E))$ of $\gamma$ to be the pullback of the connection $\nabla$ on $\gamma$. 
\end{defn}

\begin{prp}{}{}\\
Let $M$ be a smooth manifold. Let $p:E\to M$ be a smooth vector bundle. Let $\gamma:[0,1]\to M$ be a smooth curve. Let $\nabla$ be a connection on $E$. Then the covariant derivative is given in local charts $(U,\phi=(x^1,\dots,x^n))$ for $\gamma(t_0)\in M$ by the formula $$(D_tV)_{\gamma(t_0)}=\sum_{i=1}^n\left(\frac{\partial a_i}{\partial x^i}\bigg{|}_{\gamma(t_0)}\frac{\partial}{\partial x^i}\bigg{|}_{\gamma(t_0)}+a_i(\gamma(t_0))\nabla\left(\gamma'(t_0),\frac{\partial}{\partial x^i}\bigg{|}_{\gamma(t_0)}\right)\right)$$ where $V$ is given locally on the same chart by $$V_{\gamma(t)}=\sum_{i=1}^na_i(\gamma(t))\frac{\partial}{\partial x^k}\bigg{|}_{\gamma(t)}$$
\end{prp}

\begin{defn}{Parallel Sections}{}\\
Let $M$ be a smooth manifold. Let $p:E\to B$ be a smooth vector bundle. Let $\gamma:[0,1]\to M$ be a smooth curve. Let $\nabla$ be a connection on $E$. Define the parallel sections of $\gamma$ along $\nabla$ to be $\ker(D_t)$. 
\end{defn}

In other words, a parallel section is a local section $s:\im(\gamma)\to E$ such that $D_t(s)=0$. 

\begin{prp}{}{}\\
Let $M$ be a smooth manifold. Let $p:E\to B$ be a smooth vector bundle. Let $\gamma:[0,1]\to M$ be a smooth curve. Let $\nabla$ be a connection on $E$. Let $X\in E_{\gamma(0)}$. Then there exists a unique parallel section $s:[0,1]\to E$ such that $s(0)=X$. 
\end{prp}

\begin{defn}{Parallel Transports}{}\\
Let $M$ be a smooth manifold. Let $p:E\to B$ be a smooth vector bundle. Let $\gamma:[0,1]\to M$ be a smooth curve. Let $\nabla$ be a connection on $E$. Define the parallel transport map $$P_\gamma:E_{\gamma(0)}\to E_{\gamma(1)}$$ by $X\mapsto s_X(1)$ where $s_X:[0,1]\to E$ is the unique parallel section such that $s_X(0)=X$. 
\end{defn}

\begin{lmm}{}{}\\
Let $M$ be a smooth manifold. Let $p:E\to B$ be a smooth vector bundle. Let $\gamma:[0,1]\to M$ be a smooth curve. Let $\nabla$ be a connection on $E$. Then $P_\gamma$ is a well defined linear map. 
\end{lmm}

\subsection{The Curvature Tensor}
\begin{defn}{The Curvature Tensor}{}\\
Let $M$ be a smooth manifold. Let $p:E\to M$ be a smooth vector bundle. Let $\nabla$ be a connection on $p$. Define the curvature of $\nabla$ to be the smooth section $R(\nabla)\in\mC^\infty(\text{End}(E)\otimes\Lambda^2T^\ast M)$ given by $$R(\nabla)(X,Y)(s)=[\nabla_X,\nabla_Y](s)-\nabla_{[X,Y]}(s)$$ for $X,Y\in\mathfrak{X}(M)$ and $s\in\mC^\infty(E)$. 
\end{defn}

Notice that the definition makes sense: Appealing to the tensor characterization lemma, a section of $\mC^\infty(\text{End}(E)\otimes\Lambda^2T^\ast M)$ corresponds to a homomorphism $\Lambda^2TM\to\mC^\infty(\text{End}(E))$. Applying the lemma to $\mC^\infty(\text{End}(E))=\mC^\infty(E^\ast\otimes E)$ again, we see that the curvature tensor corresponds to a map $$\Lambda^2TM\to\Hom_{\mC^\infty(M)}(\mC^\infty(E),\mC^\infty(E))$$

\begin{defn}{The Curvature Form}{}\\
Let $M$ be a smooth manifold. Let $p:E\to M$ be a smooth vector bundle. Let $\nabla$ be a connection on $E$. Let $U\subseteq M$ be an open subset. Let $e_1,\dots,e_r$ be a local frame of $E$ on $U$. Define the curvature forms of $\nabla$ on $U$ to be the unique two forms $\Omega_j^i\in\mC^\infty(U,E)$ such that $$R(\nabla)(X,Y)(e_j)=\sum_{i=1}^r\Omega_j^i(X,Y)e_j$$ for all $X,Y\in\mathfrak{U}$. 
\end{defn}

\begin{thm}{The Second Structural Equation}{}\\
Let $M$ be a smooth manifold. Let $p:E\to M$ be a smooth vector bundle. Let $\nabla$ be a connection on $E$. Let $U\subseteq M$ be an open subset. Let $e_1,\dots,e_r$ be a local frame of $E$ on $U$. Then we have $$\Omega_j^i=d\omega_j^i+\sum_{k=1}^r\omega_k^i\wedge\omega_j^k$$
\end{thm}

Another way to think about it: The matrix $\Omega=(\Omega_j^i)$ is an element of $\Omega^2(U,\mathfrak{gl}(n,\R))$ (indeed locally $\text{End}(E)$ is isomorphic to $\mathfrak{gl}(n,\R)$). Gluing along local patches give $\Omega=d\omega+\omega\wedge\omega$. One can prove that $R(X,Y)=\Omega(X,Y)$. Indeed, $\Omega$ comes from the coordinate description of $R$, so patching it up should recover $R$. 

\begin{prp}{}{}\\
Let $M$ be a smooth manifold. Let $p:E\to M$ be a smooth vector bundle. Let $\nabla$ be a connection on $E$. Let $U\subseteq M$ be an open subset. Let $e,\overline{e}$ be local frames of $E$ on $U$. Suppose that $g:U\to\text{GL}(r,\R)$ is a transition function such that $\overline{e}=ge$. Suppose that $\Omega_j^i$ and $\overline{\Omega}_j^i$ are the curvature forms of $\nabla$ with respect to $e$ and $\overline{e}$. Then we have $$\overline{\Omega}_j^i=g^{-1}\Omega g$$
\end{prp}

\begin{defn}{The Lie Bracket on Smooth Vector Fields}{} Let $M$ be a smooth manifold. Let $X,Y\in\mathfrak{X}(M)$ be smooth vector fields. Define the Lie bracket of $X$ and $Y$ to be the vector field $[X,Y]$ given by the formula $$[X,Y]_p(f)=X(Y_p(f))=Y(X_p(f))$$ for $p\in M$ and $f\in\mC_{M,p}^\infty$.

\end{prp}

\section{Geometry on Riemannian Manifolds}
\subsection{Geodesics on Riemannian Manifolds}
Let $M$ be a smooth manifold. Let $\gamma:I\to M$ be a smooth curve on $M$. Then we can compute its differential to obtain a tangent vector $$\gamma'(t)\in T_{\gamma(t)}M$$ for each $t\in I$. This defines a vector field in $\mathfrak{X}(\gamma)$. 

\begin{defn}{Geodesics}{} Let $M$ be a smooth manifold. Let $\nabla$ be an affine connection on $M$. A geodesic on $M$ is a curve $\gamma:I\to M$ such that $$D_t(\gamma'(t))=0$$ where $D_t$ is the associated covariant derivative of $\nabla$ and $\gamma$. 
\end{defn}

We have seen geodesics for metric spaces, but these are slightly different. We require geodesics on manifolds to only minimize the distance locally. 

\subsection{Minimizing Curves}
\begin{defn}{Minimizing Curves}{}\\
Let $(M,g)$ be a Riemannian manifold. Let $\gamma:[a,b]\to M$ be a smooth curve in $M$. We say that $\gamma$ is minimizing if $$L(\gamma)=d_g(\gamma(a),\gamma(b))$$
\end{defn}

\begin{defn}{Locally Minimizing Curves}{}\\
Let $(M,g)$ be a Riemannian manifold. Let $\gamma:[a,b]\to M$ be a smooth curve in $M$. We say that $\gamma$ is locally minimizing if for all $t\in[a,b]$, there exists a neighbourhood $[c,d]\subseteq[a,b]$ of $t$ such that $\gamma|_{[c,d]}$ is a minimizing curve. 
\end{defn}

\begin{prp}{}{}\\
Let $(M,g)$ be a Riemannian manifold. Let $\gamma:[a,b]\to M$ be a smooth curve in $M$. Then the following are true. 
\begin{itemize}
\item If $\gamma$ is a minimizing curve, then $\gamma$ is a geodesic. 
\item If $\gamma$ is a geodesic, then $\gamma$ is a locally minimizing curve. 
\end{itemize}
\end{prp}

\subsection{Gauss-Bonnet Theorem}
\begin{thm}{The Gauss-Bonnet Formula}{} Let $(M,g)$ be an oriented smooth $2$-manifold. Let $\gamma$ be a positively oriented curved polygon in $M$ and let $\Omega$ be its interior. Then $$\int_\Omega K\,dA+\int_\gamma\kappa_N\,ds+\sum_{i=1}^k\varepsilon_i=2\pi$$ where 
\begin{itemize}
\item $K$ is the Gaussian curvature of $g$
\item $dA$ is the Riemannian volume form
\item $\varepsilon_i$ are the exterior angles of $\gamma$
\item The second integral is taken with respect to arc length
\end{itemize}	
\end{thm}

\begin{thm}{Gauss-Bonnet Theorem}{} Let $(M,g)$ be an smooth compact $2$-dimensional Riemannian manifold. Let $K$ be the Gaussian curvature of $M$ and let $k_g$ be the geodesic curvature of $\partial M$. Then $$\int_MK\,dA+\int_{\partial M}k_g\,ds=2\pi\chi(M)$$
\end{thm}

\begin{crl}{}{} Let $(M,g)$ be an smooth compact $2$-dimensional Riemannian manifold without boundary. Let $K$ be the Gaussian curvature of $M$. Then $$\int_MK\,dA=2\pi\chi(M)$$
\end{crl}

\subsection{Integrating Functions on Riemannian Manifolds}
\begin{defn}{The Riemannian Volume Form}{}\\
Let $(M,g)$ be an orientable Riemannian manifold. Define the volume form $dV$ of $M$ to be given locally at $p\in M$ on the chart $(U,x^1,\dots,x^n)$ by $$(dV)_p=\sqrt{\det(g_{i,j}(p))}(dx^1)_p\wedge\cdots\wedge(dx^n)_p$$ where $g$ is given locally at the chart by $g_p=\sum_{1\leq i,j\leq n}g_{i,j}(p)(dx^i)_p\otimes(dx^j)_p$. 
\end{defn}

Let $(M,g)$ be a compact orientable Riemannian manifold. Let $f:M\to\R$ be a compactly supported smooth function. Then we may define the integral of $f$ over $M$ to be the value $\int_MfdV$. 

\begin{defn}{The Volume of a Compact Orientable Manifold}{}\\
Let $(M,g)$ be a compact orientable Riemannian manifold. Define the volume of $M$ to be $$\text{Vol}(M)=\int_M1dV$$
\end{defn}

\begin{defn}{The Induced Volume Form at the Boundary}{}\\
Let $(M,g)$ be an orientable Riemannian manifold with boundary. Define the induced volume form on $\partial M$ to be $$dS=\iota^\ast dV$$
\end{defn}

\begin{lmm}{}{}\\
Let $(M,g)$ be an orientable Riemannian manifold with boundary. Let $(U,\phi)$ be a boundary chart of $M$. Then locally on $U$, the induced volume form is given by $$dS=\sqrt{\det(g_{ij}|_{\partial M})}dx^1\wedge\cdots\wedge dx^n$$
\end{lmm}

\subsection{Differential Operators}
\begin{defn}{The Gradient of a Function}{}\\
Let $(M,g)$ be a Riemannian manifold. Let $f:M\to\R$ be a smooth function. Define the gradient of $f$ to be $$\text{grad}(f)=(df)^\#$$
\end{defn}

By definition, the gradient of $f$ is the unique section of $TM$ such that $$(df)_p(w)=g_p(\text{grad}(f)_p,w)$$ for all $p\in M$ and $w\in T_pM$. 

\begin{prp}{}{}\\
Let $(M,g)$ be a Riemannian manifold. Let $(U,x^1,\dots,x^n)$ be a chart in $M$. Let $f:M\to\R$ be a smooth function. Let $g$ be given locally at the chart by $$g_p=\sum_{1\leq i,j\leq n}g_{i,j}(p)(dx^i)_p\otimes(dx^j)_p$$ Let $(g^{i,j}(p))=(g_{i,j}(p))^{-1}$ be the inverse of the matrix. Then locally at $U$, the gradient of $f$ at $p$ is given by $$\text{grad}(f)(p)=\sum_{1\leq i,j\leq n}\frac{\partial f}{\partial x^i}\bigg{|}_pg^{i,j}(p)\frac{\partial}{\partial x^j}\bigg{|}_p$$
\end{prp}

\begin{eg}{}{}\\
Let $f:\R^n\to\R$ be a smooth function. Then the gradient of $f$ at $p\in\R^n$ is given by $$\text{grad}(f)(p)=\sum_{i=1}^n\frac{\partial f}{\partial x_i}\bigg{|}_p\frac{\partial}{\partial x_i}\bigg{|}_p$$
\end{eg}

\begin{defn}{The Divergence of a Vector Field}{}\\
Let $(M,g)$ be a Riemannian manifold. Let $X:M\to TM$ be a smooth vector field. Define the divergence of $X$ to be the unique smooth function $\text{div}(X):M\to\R$ such that for each $p\in M$ and $(U,x^1,\dots,x^n)$ a chart at $p$, we have $$(d(X\lrcorner dV))_p=(\text{div}(X)(p))(dV)_p$$ where $dV$ is the local top form given by $(dV)_p=\sqrt{\det(g_{i,j}(p))}(dx^1)_p\wedge\cdots\wedge(dx^n)_p$ and $g$ is given locally at the chart by $g_p=\sum_{1\leq i,j\leq n}g_{i,j}(p)(dx^i)_p\otimes(dx^j)_p$. 
\end{defn}

\begin{prp}{}{}\\
Let $(M,g)$ be a Riemannian manifold. Let $(U,x^1,\dots,x^n)$ be a chart in $M$. Let $X:M\to TM$ be a smooth vector field. Suppose that $g$ and $X$ are given locally at the chart by $$g_p=\sum_{1\leq i,j\leq n}g_{i,j}(p)(dx^i)_p\otimes(dx^j)_p\;\;\;\text{ and }\;\;\;X(p)=\sum_{i=1}^na_i\frac{\partial}{\partial x^i}\bigg{|}_p$$ respectively. Then locally at $U$, the divergence of $X$ at $p\in U$ is given by $$\text{div}(X)(p)=\frac{1}{\sqrt{\det(g_{i,j}(p))}}\sum_{k=1}^n\frac{\partial(a_k\circ\sqrt{\det(g_{i,j})})}{\partial x^k}\bigg{|}_p$$
\end{prp}

\begin{eg}{}{}\\
Let $X:\R^n\to\R^n$ be a smooth vector field whose components are given by $X_i:\R^n\to\R$. Then the divergence of $X$ at $p\in\R^n$ is given by $$\text{div}(X)(p)=\sum_{i=1}^n\frac{\partial X_i}{\partial x_i}\bigg{|}_p$$
\end{eg}

\begin{defn}{The Laplacian of a Function}{}\\
Let $(M,g)$ be a Riemannian manifold. Let $f:M\to\R$ be a smooth function. Define the Laplacian of $f$ to be the function $\Delta f:M\to\R$ given by $$\Delta f=\text{div}(\text{grad}(f))$$
\end{defn}

\begin{prp}{}{}\\
Let $(M,g)$ be a Riemannian manifold. Let $f:M\to\R$ be a smooth function. Let $(U,x^1,\dots,x^n)$ be a chart in $M$. Let $g$ be given locally at the chart by $$g_p=\sum_{1\leq i,j\leq n}g_{i,j}(p)(dx^i)_p\otimes(dx^j)_p$$ Let $(g^{i,j}(p))=(g_{i,j}(p))^{-1}$ be the inverse of the matrix. Then locally at $U$, the laplcian of $f$ at $p$ is given by $$\Delta(f)(p)=\frac{1}{\sqrt{\det(g_{i,j}(p))}}\sum_{1\leq k,l\leq n}\frac{\partial(g^{k,l}\circ\sqrt{\det(g_{i,j})}\circ\frac{\partial f}{\partial x^l})}{\partial x^k}\bigg{|}_p$$
\end{prp}

\begin{eg}{}{}\\
Let $f:\R^n\to\R$ be a smooth function. Then the laplacian of $f$ at $p\in\R^n$ is given by $$\Delta(f)(p)=\sum_{i=1}^n\frac{\partial^2f}{\partial x_i^2}\bigg{|}_p$$
\end{eg}

\subsection{Curl and The Case of Dimension 3}
\begin{defn}{The $\beta$-Operator}{}\\
Let $(M,g)$ be an oriented Riemannian manifold of dimension $3$. Define the $\beta$-operator to be the map $\beta:TM\to\bigwedge_{i=1}^2(T^\ast M)$ by $$\beta(X)=X\lrcorner dV$$
\end{defn}

\begin{lmm}{}{}\\
Let $(M,g)$ be an oriented Riemannian manifold of dimension $3$. Then $\beta$ is a smooth bundle isomorphism. 
\end{lmm}

\begin{defn}{The Curl of a Vector Field}{}\\
Let $(M,g)$ be an oriented Riemannian manifold of dimension $3$. Let $X:M\to TM$ be a smooth vector field. Define the curl of $X$ to be the vector field $\text{curl}(X):M\to TM$ given by $$\text{curl}(X)=\beta^{-1}(d(X^\flat))$$
\end{defn}

\begin{prp}{}{}\\
Let $(M,g)$ be an oriented Riemannian manifold of dimension $3$. Then the following diagram is commutative: \\

\end{prp}

\subsection{Calculus on Manifolds}
Let $M$ be an oriented smooth manifold of dimension $n$. let $p\in\partial M$. Then $T_p(\partial M)$ is an $(n-1)$-dimensional vector subspace of $T_p(M)$. Recall that a boundary chart of $M$ at $p\in M$ is a chart $(U,\phi)$ such that the following are true. 
\begin{itemize}
\item $\phi(U)\subseteq\{(x_1,\dots,x_n)\in\R^n\;|\;x_n\geq 0\}$. 
\item $\phi^{-1}(\{(x_1,\dots,x_n)\in\R^n\;|\;x_n=0\})\subseteq\partial M$. 
\item $\phi(p)=0$. 
\end{itemize}

\begin{defn}{The Unit Normal Vector Field}{}\\
Let $(M,g)$ be an oriented Riemannian manifold with boundary. Define the outward point unit normal vector field to be the unique smooth section $\nu:\partial M\hookrightarrow M\to TM$ such that the following are true. 
\begin{itemize}
\item $g_p(\nu(p),w)=0$ for all $w\in T_pM$ and all $p\in M$. 
\item $g_p(\nu(p),\nu(p))=1$ for all $p\in M$. 
\item For any $p\in M$ and any boundary chart $(U,\phi)$, the $n$th coordinate of $(\phi_\ast)(\nu_p)$ is strictly negative. 
\end{itemize}
\end{defn}

\begin{lmm}{}{}\\
Let $(M,g)$ be an oriented Riemannian manifold of dimension $n$ with boundary. Let $(U,\phi)$ be the boundary chart of $M$. Then locally at $U\cap\partial M$, the unit normal vector field is given by $$\nu=\frac{1}{\sqrt{g^{nn}}}\sum_{j=1}^ng^{jn}\frac{\partial}{\partial x^j}$$
\end{lmm}

\begin{lmm}{}{}\\
Let $(M,g)$ be an orientable Riemannian manifold with boundary. Then we have $$dS=\iota^\ast(\nu^\flat\lrcorner dV)$$
\end{lmm}

\begin{thm}{The Gradient Theorem}{}\\
Let $(M,g)$ be a Riemannian manifold. Let $\gamma:[0,1]\to M$ be a smooth curve. Let $f\in\mC^\infty(M)$ be a smooth function. Then we have $$\int_\gamma\;df=f(\gamma(1))-f(\gamma(0))$$
\end{thm}

\begin{thm}{The Divergence Theorem}{}\\
Let $(M,g)$ be a compact oriented Riemannian manifold with boundary. Let $\iota:\partial M\hookrightarrow M$ be the inclusion map. Let $X\in\mathfrak{X}(M)$. Then we have $$\int_M\text{div}(X)\;dV_g=\int_{\partial M}g(X,\nu)\;dS_g$$
\end{thm}

\subsection{The Hodge Star Operator}
\begin{defn}{The Hodge Star Operator}{}\\
Let $(M,g)$ be an orientable Riemannian manifold. Let $n=\dim(M)$. 
\begin{itemize}
\item Let $p\in M$. Define $\ast_p:\Lambda^kT_p^\ast(M)\to\Lambda^{n-k}T_p^\ast(M)$ to be the unique $\C$-linear isomorphism such that $$\alpha\wedge(\ast_p\beta)=g_p(\alpha,\beta)(dV)_p$$ where $\alpha,\beta\in\Lambda^kT_p^\ast(M)$ and $\text{Vol}$ is the volume form of $M$ associated to $g$, and $\langle-,-\rangle_p$ is the induced metric on $\Lambda^k(T_p^\ast(M))$. 
\item Define the Hodge star operator of $M$ to be the map $$\ast\in\Gamma(M,\Hom(\Omega^k(M),\Omega^{n-k}(M)))$$ given by $p\mapsto\ast_p$. 
\end{itemize}
\end{defn}

\begin{lmm}{}{}\\
Let $(M,g)$ be an orientable Riemannian manifold. Then the following are true regrading the Hodge star operator. 
\begin{itemize}
\item $\ast f=fdV$ for any smooth function $f\in\mC^\infty(M)$. 
\item $\ast\ast\omega=(-1)^{k(n-k)}\omega$ for any smooth $k$-form $\omega\in\Omega^k(M)$. 
\end{itemize}
\end{lmm}

\begin{lmm}{}{}\\
Let $(M,g)$ be an orientable Riemannian manifold. Let $X:M\to TM$ be a smooth vector field. Then the following are true. 
\begin{itemize}
\item $X\lrcorner dV=\ast(X^\flat)$. 
\item $\text{div}(X)=(\ast\circ d\circ\ast)(X^\flat)$
\end{itemize}
\end{lmm}

\begin{lmm}{}{}\\
Let $(M,g)$ be an orientable Riemannian manifold of dimension $3$. Let $X:M\to TM$ be a smooth vector field. Then we have $$\text{curl}(X)=(\ast\circ d(X^\flat))^\#$$
\end{lmm}

\subsection{The $L^2$-Norm}
\begin{defn}{The $L^2$-Norm}{}\\
Let $(M,g)$ be an orientable Riemannian manifold. Define the $L^2$-norm on $M$ to be the map $\langle -,-\rangle_{L^2}:\Omega^k(M)\times\Omega^k(M)\to\R$ by the formula $$\langle\alpha,\beta\rangle_{L^2}=\int_M\langle\alpha,\beta\rangle_gdV$$ where $\langle-,-\rangle_g$ is the induce inner product on $k$-forms. 
\end{defn}

\begin{lmm}{}{}\\
Let $(M,g)$ be an orientable Riemannian manifold. Then $\langle-,-\rangle_{L^2}$ defines an inner product on the compactly supported smooth $k$-forms on $M$ for any $k\in\N$. 
\end{lmm}

\begin{lmm}{}{}\\
Let $(M,g)$ be an orientable Riemannian manifold. Let $\alpha,\beta\in\Omega^k(M)$ be compactly supported smooth $k$-forms. Then we have $$\langle\alpha,\beta\rangle_{L^2}=\int_M\alpha\wedge\ast\beta$$
\end{lmm}

\subsection{The Adjoint Differential Operator}
\begin{defn}{The Adjoint Differential Operator}{}\\
Let $(M,g)$ be an orientable Riemannian manifold. Define the operator $d^\ast:\Omega^k(M)\to\Omega^{k-1}(M)$ by $$d^\ast(\omega)=(-1)^{n(k+1)+1}(\ast\circ d\circ\ast)(\omega)$$ for $\omega\in\Omega^k(M)$. Also define $d^\ast(\omega)=0$ for $\omega\in\Omega^0(M)$ by convention. 
\end{defn}

\begin{lmm}{}{}\\
Let $(M,g)$ be an orientable Riemannian manifold. Then we have $d^\ast\circ d^\ast=0$. 
\end{lmm}

\begin{prp}{}{}\\
Let $(M,g)$ be a compact orientable Riemannian manifold. Then we have $$\langle d^\ast\omega,\eta\rangle_{L^2}=\langle\omega,d\eta\rangle_{L^2}$$ for any $\omega\in\Omega^{k+1}(M)$ and $\eta\in\Omega^k(M)$. 
\end{prp}

\begin{defn}{The Laplace-Beltrami Operator}{}\\
Let $M$ be an orientable Riemannian manifold. Define the Laplace-Beltrami operator on $M$ to be the map $\Delta:\Omega^k(M)\to\Omega^k(M)$ given by $$\Delta=d^\ast\circ d+d\circ d^\ast$$
\end{defn}

\begin{lmm}{}{}\\
Let $M$ be an orientable Riemannian manifold. Then the following are true regarding the Laplace-Beltrami operator. 
\begin{itemize}
\item $\ast\circ\Delta=\Delta\circ\ast$. 
\item $\langle\Delta\alpha,\beta\rangle_{L^2}=\langle\alpha,\Delta\beta\rangle_{L^2}$ for any compactly supported differential forms $\alpha,\beta\in\Omega^k(M)$. 
\end{itemize}
\end{lmm}

\subsection{The Hodge Theorem for Riemannian Manifolds}
\begin{defn}{Harmonic Forms}{}\\
Let $M$ be an orientable Riemannian manifold. Let $\alpha\in\Omega^k(M)$. We say that $\alpha$ is harmonic if $\alpha\in\ker(\Delta)$. 
\end{defn}

\begin{lmm}{}{}\\
Let $M$ be an orientable Riemannian manifold. Let $\alpha\in\Omega^k(M)$. Then $\alpha$ is harmonic if and only if $d(\alpha)=0$ and $d^\ast(\alpha)=0$. 
\end{lmm}

\begin{defn}{The Space of Harmonic Forms}{}\\
Let $M$ be an orientable Riemannian manifold. Define the space of harmonic $k$-forms to be $$\mH^k(M)=\ker(\Delta)$$
\end{defn}

\begin{thm}{Hodge Theorem}{}\\
Let $M$ be an compact orientable Riemannian manifold. Then the following are true. 
\begin{itemize}
\item $\mH^k$ is finite dimensional for all $k\in\N$. 
\item There is a decomposition $$\Omega^k(M)\cong\mH^k(M)\oplus\im(\Delta)$$
\end{itemize}
\end{thm}

\begin{crl}{}{}\\
Let $M$ be an orientable Riemannian manifold. Then the quotient map induces an isomorphism $$\mH^k(M)\cong H_\text{dR}^k(M)$$ for all $k\in\N$. 
\end{crl}

\section{Holonomy}
\subsection{Holonomy Groups}
\begin{defn}{Holonomy Groups}{}\\
Let $(M,g)$ be a Riemaniann manifold. Let $p:E\to M$ be a smooth vector bundle. Let $\nabla$ be a connection on $E$. Let $x\in M$. Define the holonomy group of $E$ at $x$ with respect to $\nabla$ to be $$\text{Hol}_x(\nabla)=\{P_\gamma\;|\;\gamma(0)=\gamma(1)=x\}\subseteq\text{GL}(E_x)$$
\end{defn}

\begin{lmm}{}{}\\
Let $(M,g)$ be a Riemaniann manifold. Let $p:E\to M$ be a smooth vector bundle. Let $\nabla$ be a connection on $E$. Let $x\in M$. Then $\text{Hol}_x(\nabla)$ is a Lie subgroup of $\text{GL}(E_x)$. 
\end{lmm}

\begin{prp}{}{}\\
Let $(M,g)$ be a Riemaniann manifold. Let $p:E\to M$ be a smooth vector bundle. Let $\nabla$ be a connection on $E$. Then the following are true. 
\begin{itemize}
\item Let $\gamma:[0,1]\to M$ be a path. Then we have $$\text{Hol}_{\gamma(1)}(\nabla)=P_\gamma\text{Hol}_{\gamma(0)}(\nabla)P_\gamma^{-1}$$
\item If $M$ is simply connected, then $\text{Hol}_x(\nabla)$ is connected for all $x\in M$. 
\end{itemize}
\end{prp}

So for simply connected spaces, the holonomy groups are independent of base point up to isomorphism. But the specific embedding into $\text{GL}(n,F)$ does depend on the base point. 

\begin{defn}{}{}\\
Let $M$ be a simply connected Riemannian manifold. Let $p:E\to M$ be a smooth vector bundle over $\R$. Let $\nabla$ be a connection on $E$. Define the holonomy group of $E$ with respect to $\nabla$ to be $$\text{Hol}(\nabla)=\text{Hol}_x(\nabla)$$ for any base point $x\in M$. 
\end{defn}

\begin{prp}{}{}\\
Let $(M,g)$ be a Riemaniann manifold. Let $p\in M$. Let $\phi:T_pM\to\R^n$ be an $\R$-linear isomorphism. Then we have $$\text{GL}(\phi)(\text{Hol}_p(\nabla^g))\leq O(n)$$
\end{prp}

\begin{prp}{}{}\\
Let $(M,g)$ and $(N,h)$ be Riemannian manifolds. Let $p\in M$ and $q\in N$. Then we have $$\text{Hol}_{(p,q)}(\nabla^{g\times h})=\text{Hol}_p(\nabla^g)\times\text{Hol}_p(\nabla^h)$$
\end{prp}

\begin{prp}{}{}\\
Let $(M,g)$ be a Riemannian manifold. Let $p\in M$. If $M$ is irreducible, then the representation $\text{Hol}_p(\nabla^g)\to\text{GL}(T_pM)$ is irreducible. 
\end{prp}

\begin{thm}{Berger}{}\\
Let $(M,g)$ be a Riemannian manifold of dimension $n$. Suppose that $M$ is simply connected, irreducible and not symmetric. Let $p\in M$. Then exactly one of the following cases hold. 
\begin{itemize}
\item $\text{Hol}_p(\nabla^g)\cong SO(n)$. 
\item $n=2m$ for $m\geq 2$ and $\text{Hol}_p(\nabla^g)\cong U(m)\leq SO(n)$. 
\item $n=2m$ for $m\geq 2$ and $\text{Hol}_p(\nabla^g)\cong SU(m)\leq SO(n)$. 
\item $n=4m$ for $m\geq 2$ and $\text{Hol}_p(\nabla^g)\cong SP(m)\leq SO(n)$. 
\item $n=4m$ for $m\geq 2$ and $\text{Hol}_p(\nabla^g)\cong SP(m)\times SP(1)\leq SO(n)$. 
\item $n=7$ and $\text{Hol}_p(\nabla^g)\cong G_2\leq SO(7)$. 
\item $n=8$ and $\text{Hol}_p(\nabla^g)\cong\text{Spin}(7)\leq SO(8)$. 
\end{itemize}
\end{thm}
