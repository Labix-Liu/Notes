\begin{defn}{Monomial Notation}{}\\
Let $R$ be a commutative ring. Let $u=(u_1,\dots,u_n)\in\N^n$. Define $$x^u=x_1^{u_1}\cdots x_n^{u_n}\in R[x_1,\dots,x_n]$$
\end{defn}

\subsection{The Semi-group Ring}
\begin{defn}{The Semi-Group Ring}{}\\
Let $S$ be a monoid. Let $R$ be a ring. Define the semi-group ring of $S$ over $R$ to be the left $R$-module $$R[S]=\bigoplus_{s\in S}R\cdot s$$ together with multiplication defined on basis by $t^{s_1}\cdot t^{s_2}=t^{s_1+s_2}$
\end{defn}

\begin{lmm}{}{}\\
Let $S$ be a monoid. Let $R$ be a ring. Then the following are true. 
\begin{itemize}
\item $R[S]$ is a ring. 
\item If $S$ is commutative, then $R[S]$ is a commutative ring. 
\item If $R$ and $S$ are commutative, then $R[S]$ is an $R$-algebra. 
\end{itemize}
\end{lmm}

\subsection{Polyhedral Cones}
\begin{defn}{The Cone of a Set of Vectors}{}\\
Let $k$ be a field. Let $V$ be a finite dimensional vector space over $k$. Let $S=\{v_1,\dots,v_s\}\subseteq V$ be finite set of vectors. Define the cone spanned by $S$ to be $$\text{Cone}(S)=\left\{\sum_{k=1}^s\lambda_kv_k\;\bigg{|}\;\lambda_k\geq 0\right\}$$
\end{defn}

\begin{defn}{Cones in a Vector Space}{}\\
Let $k$ be a field. Let $V$ be a finite dimensional vector space over $k$. Let $C\subseteq V$ be a subset of $V$. We say that $C$ is a cone of $V$ if there exists a finite subset $S\subseteq V$ of vectors such that $$C=\text{Cone}(S)$$
\end{defn}

\begin{lmm}{}{}\\
Let $k$ be a field. Let $C\subseteq k^n$ be a subset of $V$. Then $C$ is a cone if and only if there exists half planes $H_1,\dots,H_d\subseteq k^n$ such that $$C=H_1\cap\cdots\cap H_d$$
\end{lmm}

\begin{defn}{Dual Cone}{}\\
Let $k$ be a field. Let $V$ be a finite dimensional vector space over $k$. Let $C\subseteq V$ be a subset of $V$. Then define the dual cone of $C$ to be $$C^\vee=\{f\in V^\ast\;|\;\langle f,v\rangle\geq 0\text{ for all }v\in C\}$$
\end{defn}

\subsection{Affine Semi-groups}
\begin{defn}{Affine Semi-group}{}\\
Let $S$ be a monoid. We say that $S$ is an affine semi-group if there exists an injective monoid homomorphism $S\to\N^d$ for some $d\in\N$. 
\end{defn}

\subsection{Lattice Ideals}
Let $S$ be a commutative monoid. Write $G(S)$ the grothendieck completion of $S$. It is an abelian group. 

\begin{defn}{Lattice Ideal}{}\\
Let $L\subseteq\Z^n$ be a lattice. Let $R$ be a commutative ring. Define the lattice ideal of $L$ in $R[x_1,\dots,x_n]$ to be $$I_L=R\langle x^u-x^v\;|\;u,v\in\N^n\text{ and }u-v\in L\rangle$$
\end{defn}

\begin{prp}{}{}\\
Let $k$ be a field. Let $S$ be a finitely generated commutative monoid. Let $s_1,\dots,s_n$ be a set of generators of $S$. Let $\phi:\Z^n\to G(S)$ be the group homomorphism defined by $e_i\mapsto s_i$. Then there is a $k$-algebra isomorphism $$k[S]\cong\frac{k[x_1,\dots,x_n]}{I_{\ker(\phi)}}$$
\end{prp}

\begin{prp}{}{}\\
Let $k$ be a field. Let $S$ be a finitely generated commutative monoid. Let $s_1,\dots,s_n$ be a set of generators of $S$. Let $\phi:\Z^n\to G(S)$ be the group homomorphism defined by $e_i\mapsto s_i$. Then the following are equivalent. 
\begin{itemize}
\item $I_{\ker(\phi)}$ is prime. 
\item $k[S]$ is an integral domain. 
\item $G(S)$ is a free abelian group. 
\item There exists an injective monoid homomorphism $S\to\N^d$ for some $d\in\N$. 
\end{itemize}
\end{prp}

\section{Constructing Toric Varieties}
\subsection{The Standard $n$-Torus}
\begin{defn}{The Standard $n$-Torus}{}\\
Let $k$ be a field. Let $n\in\N\setminus\{0\}$. Define the standard $n$-torus over $k$ to be the affine group variety $$\G_m^n=\text{Spec}(k[x,x^{-1}])^n$$
\end{defn}

\begin{defn}{Tori}{}\\
Let $k$ be a field. Let $V$ be an affine algebraic group. We say that $V$ is a torus if there exists a field extension $K/k$ such that the algebraic group $V$ over $K$ is isomorphic to $\G_m^n=\text{Spec}(K[x,x^{-1}])^n$ for some $n\in\N\setminus\{0\}$. 
\end{defn}

\begin{defn}{Split Tori}{}\\
Let $k$ be a field. Let $V$ be a torus over the field extension $K/k$. We say that $V$ is split if $K=k$. 
\end{defn}

If $K$ is separably closed or algebraically closed, then all tori are split. \\

Recall that the functor $D:\bold{FGAb}\to\bold{Diag}$ sending a finitely generated abelian group $G$ to $D(G)=\text{Spec}(k[G])$ is an equivalence of categories. Evidently split tori corresponds to the free abelian groups of finite rank under the equivalence. \\

The following are some more properties of a split torus $T$:
\begin{itemize}
\item $T$ is diagonalizable (in $T$ is a torus if and only if $T$ is smooth connected diagonalizable)
\item Every representation of $G$ is diagonalizable. 
\item If $V$ is a representation of $T$, then we can decompose $V$ into its eigenspaces: $$V=\bigoplus_{\chi\in\chi(T)}V_\chi$$
\end{itemize}

\begin{lmm}{}{}\\
Let $k$ be a field. Let $T$ be split torus over $k$. Then the following are true. 
\begin{itemize}
\item Let $T'$ be a torus over $k$. Let $\phi:T\to T'$ be a morphism of group schemes. Then $\im(\phi)$ is a torus. 
\item Let $H\leq T$ be a subgroup variety of $T$. If $H$ is irreducible, then $H$ is a torus. 
\item Let $H\leq T$ be a subgroup of $T$. Then $T/H$ is a torus. 
\end{itemize}
\end{lmm}

\begin{defn}{Character Lattices}{}\\
Let $k$ be a field. Let $T$ be a torus over $k$. Define the character lattice of $T$ to be the group of characters $$\chi(T)=\Hom_{\bold{GrpSCh}_k}(T,\G_m)$$
\end{defn}

\begin{defn}{One-parameter Subgroups}{}\\
Let $k$ be a field. Let $T$ be a torus over $k$. A one-parameter subgroup of $T$ is a homomorphism of group schemes $\G_m\to T$. 
\end{defn}

Since $\chi$ is the inverse to $D$, we have the following result. 

\begin{lmm}{}{}\\
Let $k$ be a field. Let $T$ be a split torus over $k$. Then the following are true. 
\begin{itemize}
\item The map $$\chi(T)\times\Hom_{\bold{GrpSch}_k}(\G_m,T)\to\Z$$ defined by $(\chi,\lambda)\to\chi\circ\lambda$ is a perfect pairing. 
\item A choice of an isomorphism $T\cong\G_m^n$ induces a isomorphisms $\chi(T)\cong\Z^n\cong\Hom_{\bold{GrpSch}_k}(\G_m,T)$ such that the above perfect pairing is the standard Euclidean dot product. 
\end{itemize}
\end{lmm}

\begin{prp}{}{}\\
Let $k$ be a field. The functor $$\chi(-_{k^\text{sep}}):\bold{Tori}_k\to\bold{FGFreeAb(with continuous action of )}\text{Gal}(k^\text{sep}/k)$$ is an equivalence of categories. 
\end{prp}

\subsection{Constructing Toric Varieties}
\begin{defn}{Toric Varieties}{}\\
Let $k$ be a field. Let $X$ be an irreducible variety. We say that $X$ is a toric variety if the following are true. 
\begin{itemize}
\item $(\C^\ast)^n$ is a Zariski open subset of $X$. 
\item The action of $(\C^\ast)^n$ on itself extends to $X$. 
\end{itemize}
We denote the underlying torus of $X$ by $T_X$. 
\end{defn}

\begin{defn}{Affine Toric Varieties Associated to Characters}{}\\
Let $k$ be a field. Let $n\in\N\setminus\{0\}$. Let $\mA=\{\chi_1,\dots,\chi_m\}\subset\chi(\G_m^n)$ be characters of the standard $n$-torus. Define the affine toric varieties associated to $\mA$ to be $$Y_\mA=\overline{\im(\Phi_\mA)}$$ where $\Phi_\mA:\G_m^n\to\A_\C^m$ is the map defined by $\Phi_\mA(t)=(\chi_1(t),\dots,\chi_m(t))$
\end{defn}

\begin{prp}{}{}\\
Let $k$ be a field. Let $n\in\N\setminus\{0\}$. Let $\mA\subset\chi(\G_m^n)$ be a finite subseet of characters of the standard $n$-torus. Then the following are true. 
\begin{itemize}
\item $Y_\mA$ is an affine toric variety. 
\item The underlying torus of $Y_\mA$ has character lattice $\Z\mA$. 
\item $\dim(Y_\mA)=\text{rank}(\Z\mA)$
\end{itemize}
\end{prp}

\begin{defn}{Toric Ideals}{}\\
Let $L\subseteq\Z^n$ be a lattice. Let $R$ be a commutative ring. Let $I_L\subseteq R[x_1,\dots,x_n]$ be the lattice ideal of $L$. We say that $I_L$ is a toric ideal if $I_L$ is a prime ideal. 
\end{defn}

\begin{prp}{}{}\\
Let $k$ be a field. Let $X$ be an irreducible affine variety over $k$. Then the following are equivalent. 
\begin{itemize}
\item $X$ is a affine toric variety. 
\item There exists a lattice $L$ and a finite subset $\mA\subseteq L$ such that $X=Y_\mA$. 
\item There exists a toric ideal $I$ of $k[x_1,\dots,x_n]$ such that $X=\text{Spec}\left(\frac{k[x_1,\dots,x_n]}{I}\right)$. 
\item There exists an affine semigroup $S$ such that $X=\text{Spec}(k[S])$. 
\end{itemize}
\end{prp}

\begin{prp}{}{}\\
Let $k$ be an algebraically closed field. Let $S$ be an affine semi-group. Then there is a bijection $$\{p\in\text{Spec}(k[S])\;|\;p\text{ is a closed point }\}\;\;\overset{1:1}{\longleftrightarrow}\;\;\{\phi:S\to(\C,\cdot)\;|\;\phi\text{ is a monoid homomorphism }\}$$ given as follows. For each $p\in\text{Spec}(k[S])$ a closed point, $p$ corresponds to the monoid homomorphism $\gamma_p:S\to\C$ defined by $\gamma_p(m)=\chi^m(p)$
\end{prp}

\begin{prp}{}{}\\
Let $k$ be an algebraically closed field. Let $X=\text{Spec}(k[S])$ be an affine torus over $k$. Then the action of the torus $T_X$ on $X$ is induced by the $k$-algebra homomorphism $$k[S]\to k[\chi(T_X)]\otimes_k k[S]$$ given by $\gamma_p\mapsto\gamma_p\otimes\gamma_p$ where here we identify the closed points on $k[S]$ by the above prp. 
\end{prp}
