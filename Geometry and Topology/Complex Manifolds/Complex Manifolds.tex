\section{Complex Manifolds}
\subsection{Basic Definitions}
\begin{defn}{The Holomorphic Pseudogroup of Transformations}{}\\
Define the holomorphic Pseudogroup of transformations to be $$\Gamma_\text{Holo}=\{f:U\to V\;|\;f\text{ is biholomorphic and }U,V\subseteq\C^n\text{ are open }\}$$
\end{defn}

\begin{defn}{Complex Manifolds}{}\\
A complex manifold $M$ is a manifold whose atlas is given by $\Gamma_\text{Holo}$. 
\end{defn}

\begin{prp}{}{} Every complex manifold is a smooth real manifold. 
\begin{proof}
Let $(U,\phi=z_1,\dots,z_n))$ be a chart of a complex manifold $M$. Write $z_k=x_k+iy_k$ where $x_k$ and $y_k$ are smooth real valued functions on $U$. Then $(x_1,\dots,x_n,y_1,\dots,y_n)$ gives a homeomorphism between $U$ and an open subset of $\R^{2n}$. 
\end{proof}
\end{prp}

\begin{lmm}{}{}\\
Let $M$ be a complex manifold. Let $U\subseteq M$ be an open subset. Then $U$ is a complex manifold. 
\end{lmm}

\subsection{Complex Submanifolds}
\begin{defn}{Complex Submanifolds}{} Let $M$ be an $n$ dimensional manifold. An $m$ dimensional submanifold is a subset $Y$ of $M$ such that for every $y\in Y$, there exists a chart $(U,\phi)$ of $M$ such that $$\phi(Y\cap U)=\phi(U)\cap\{0\}^{n-m}\times\C^m=\{(z_1,\dots,z_n)\in\phi(U)|z_{n-m+1}=\dots=z_n=0\}$$
\end{defn}

\section{Almost Complex Manifolds}
\subsection{Linear Complex Structures}
\begin{defn}{Linear Complex Structure}{}\\
Let $V$ be a vector space over $\R$. A linear complex structure on $V$ is a linear map $J:V\to V$ such that $J^2=-\text{id}_V$. 
\end{defn}

\begin{prp}{}{}\\
Let $V$ be a vector space over $\R$. Let $J$ be a linear complex structure on $V$. Then $V$ is a $\C$-vector space whose action is given by $$(x+iy)\cdot v=xv+yJ(v)$$ for $v\in V$ and $x+iy\in\C$. 
\end{prp}

\begin{prp}{}{}\\
Let $V$ be a finite dimensional vector space over $\R$. Then the following are true. 
\begin{itemize}
\item $V$ admits a linear complex structure if and only if $\dim_\R(V)$ is even. 
\item In this case, we have $\dim_\C(V)=2\dim_\R(V)$. 
\end{itemize}
\end{prp}

\begin{eg}{}{}\\
Let $J:\R^{2n}\to\R^{2n}$ be the map defined in block matrix form $$J=\begin{pmatrix}
0 & -I_n\\
I_n & 0
\end{pmatrix}$$
Then $J$ defines a linear complex structure on $\R^{2n}$. In particular, if $e_1,\dots,e_{2n}$ are the standard basis vectors for $\R^{2n}$, then $\frac{1}{2}(e_1-ie_{2n+1}),\dots,\frac{1}{2}(e_n-ie_{2n})$ is a basis for $V$ as a $\C$-vector space. 
\end{eg}

\begin{prp}{}{}\\
Let $V$ be a vector space over $\R$. Let $J$ be a linear complex structure on $V$. Let $T:V\to V$ be an $\R$-linear transformation. Then $T$ is a $\C$-linear transformation if and only if $J\circ T=T\circ J$. 
\end{prp}

The linear complex structure on an $\R$-vector space gives an intrinsic way of giving the $\R$-vector space the structure of a $\C$-vector space. However, we already know an external way of giving $\R$-vector spaces $\C$-multiplication: namely by extension of scalars. 

\begin{prp}{}{}\\
Let $V$ be a vector space over $\R$. Let $J$ be a linear complex structure on $V$. Then the following are true. 
\begin{itemize}
\item $J$ extends to a linear complex structure on $V\otimes_\R\C$ by $J(v\otimes z)=J(v)\otimes z$. 
\item The eigenvalues of $J:V\otimes_\R\C\to V\otimes_\R\C$ is given by $\pm i$ and the eigenspaces of $J$ are given by $$E_i=\{v\otimes 1-J(v)\otimes i\;|\;v\in V\}\;\;\;\text{ and }\;\;\;E_{-i}=\{v\otimes 1+J(v)\otimes i\;|\;v\in V\}$$
\item There is a natural isomorphism of $\C$-vector space $V\cong E_i$ given by $v\mapsto v\otimes 1-J(v)\otimes i$. 
\item The conjugation map $\overline{(-)}:V\otimes_R\C\to V\otimes_\R\C$ given by $\overline{v\otimes z}=v\otimes\overline{z}$ restricts to an isomorphism $$\overline{(-)}|_{E_i}:E_i\overset{\cong}{\to}E_{-i}$$ whose inverse is given by $\overline{(-)}|_{E_{-i}}$. 
\end{itemize}
\end{prp}

\begin{prp}{}{}\\
Let $V$ be a vector space over $\R$. Let $J$ be a linear complex structure on $V$. Then $J^T$ defines a linear complex structure on $V^\ast$. 
\end{prp}

\subsection{Almost Complex Structures on Vector Bundles}
\begin{defn}{Almost Complex Structure}{}\\
Let $p:E\to B$ be a vector bundle. An almost complex structure on $p$ is a section $J:B\to\Hom(E,E)$. 
\end{defn}

\begin{prp}{}{}\\
Let $p:E\to B$ be a vector bundle. Let $J$ be an almost complex structure on $p$. Then $p$ is a complex vector bundle whose action is given by $$(x+iy)\cdot v=xv+yJ_b(v)$$ for $v\in E_b$ and $x+iy\in\C$ for each $b\in B$. 
\end{prp}

\subsection{Almost Complex Manifolds}
\begin{defn}{Almost Complex Manifolds}{}\\
An almost complex manifold is a smooth manifold $M$ together with an almost complex structure $J:M\to\Hom(TM,TM)$ that is smooth. 
\end{defn}

Even though every even dimensional vector space has a linear complex structure, they may not glue smoothly into an almost complex structure for a smooth manifold. This is a question of the reduction of the structure group from $GL_{2n}(\R)$ to the subgroup $GL_n(\C)$.

\section{Complex Vector Bundles}
\subsection{Hermitian and Holomorphic Vector Bundles}
\begin{defn}{Hermitian Structure}{} Let $p:E\to B$ be a vector bundle. A Hermitian structure $H$ on $E$ is a Hermitian product on each fibre varying smoothly on $E$. This means that for $x\in M$, $H:E_x\times E_x\to\C$ satisfies the following.
\begin{itemize}
\item $H(u,v)$ is $\C$-linear for every $v\in E_x$
\item $H(u,v)=\overline{H(v,u)}$
\item $H(u,u)>0$ for all $u\neq 0$
\item $H(u,v)$ is a smooth function on $M$ for every smooth sections $u,v$ of $E$
\end{itemize}
\end{defn}

\begin{defn}{Holomorphic Vector Bundle}{} Let $p:E\to M$ be a vector bundle over a complex manifold $M$. We say the vector bundle is holomorphic (equipped with a holomorphic structure) if the trivializations $$\tau_i:p^{-1}(U_i)\overset{\cong}{\to}U_i\times\C^n$$ has transition matrices $\tau_{ij}=\tau_j\circ\tau_i$ that have holomorphic coefficients. 
\end{defn}

\section{Tangent Spaces}
\subsection{The Complex Structure of Tangent Spaces on Almost Complex Manifolds}
\begin{defn}{Holomorphic and Antiholomorphic Tangent Space}{}\\
Let $(M,J)$ be an almost complex manifold. 
\begin{itemize}
\item Define $T_p^{1,0}(M)$ to be the eigenspace of $i$ of the linear complex structure $J_p$ on $T_{\C,p}(M)$. 
\item Define $T_p^{0,1}(M)$ to be the eigenspace of $-i$ of the linear complex structure $J_p$ on $T_{\C,p}(M)$. 
\end{itemize}
\end{defn}

We have seen above that $$T_pM\otimes_R\C\cong T_p^{(1,0)}(M)\oplus T_p^{(0,1)}(M)$$ as complex vector spaces. 

\begin{lmm}{}{}\\
Let $(M,J)$ be an almost complex manifold. Let $p\in M$. Let $(U,x^1,\dots,x^n,y^1,\dots,y^n)$ be a smooth chart at $p$. Suppose that $J_p\left(\frac{\partial}{\partial x^i}\bigg{|}_p\right)=\frac{\partial}{\partial y^i}\bigg{|}_p$ for $1\leq i\leq n$. Then the following are true. 
\begin{itemize}
\item A $\C$-basis for $T_p^{1,0}(M)$ is given by $$\left\{\frac{1}{2}\left(\frac{\partial}{\partial x^1}-i\frac{\partial}{\partial y^1}\right)\bigg{|}_p,\dots,\frac{1}{2}\left(\frac{\partial}{\partial x^n}-i\frac{\partial}{\partial y^n}\right)\bigg{|}_p\right\}$$
\item A $\C$-basis for $T_p^{0,1}(M)$ is given by $$\left\{\frac{1}{2}\left(\frac{\partial}{\partial x^1}+i\frac{\partial}{\partial y^1}\right)\bigg{|}_p,\dots,\frac{1}{2}\left(\frac{\partial}{\partial x^n}+i\frac{\partial}{\partial y^n}\right)\bigg{|}_p\right\}$$
\end{itemize}
\end{lmm}

Assuming that $J_p$ sends the basis vector $\frac{\partial}{\partial x^i}\bigg{|}_p$ to $\frac{\partial}{\partial y^i}\bigg{|}_p$ loses no generality since we can always choose appropriate coordinate functions so that the almost complex structure locally at $U$ looks like ours one. \\

Recall from complex analysis that we defined partial complex derivatives by $$\frac{\partial}{\partial z_k}=\frac{1}{2}\left(\frac{\partial}{\partial x_k}-i\frac{\partial}{\partial y_k}\right)\;\;\;\text{ and }\;\;\;\frac{\partial}{\partial\overline{z_k}}=\frac{1}{2}\left(\frac{\partial}{\partial x_k}+i\frac{\partial}{\partial y_k}\right)$$

\begin{defn}{The Basis for Tangent Space given by the Almost Complex Structure}{}\\
Let $(M,J)$ be an almost complex manifold. Let $p\in M$. Let $(U,x^1,\dots,x^n,y^1,\dots,y^n)$ be a smooth chart at $p$. Define the tangent vector $$\frac{\partial}{\partial z^k}\bigg{|}_p:\mC_{M,p}^\infty\to\R$$ by $$\frac{\partial}{\partial z^k}\bigg{|}_p=\left(\frac{\partial}{\partial x^k}-i\frac{\partial}{\partial y^k}\right)\bigg{|}_p$$ Define the tangent vector $$\frac{\partial}{\partial\overline{z^k}}\bigg{|}_p:\mC_{M,p}^\infty\to\R$$ by $$\frac{\partial}{\partial\overline{z^k}}\bigg{|}_p=\left(\frac{\partial}{\partial x^k}+i\frac{\partial}{\partial y^k}\right)\bigg{|}_p$$
\end{defn}

\subsection{Decomposing the Tangent Space of Complex Manifolds}
Let $M$ be a complex manifold. Let $p\in M$. Regarded as a smooth manifold, we have the notion of tangent space $T_pM=\text{Der}(C_{M,p}^\infty(M,\R),\R)$. Since we are on a complex manifold, we ought to consider smooth functions with values in $\C$ instead. 

\begin{lmm}{}{}\\
Let $M$ be a complex manifold. Let $p\in M$. Let $(U,x^1,\dots,x^n,y^1,\dots,y^n)$ be a smooth chart at $p$. Then an $\R$-basis for $T_pM$ is given by $$\left\{\frac{\partial}{\partial x^1}\bigg{|}_p,\dots,\frac{\partial}{\partial x^n}\bigg{|}_p,\frac{\partial}{\partial y^1}\bigg{|}_p,\dots,\frac{\partial}{\partial y^n}\bigg{|}_p\right\}$$
\end{lmm}

\begin{defn}{The Complex Tangent Space}{}\\
Let $M$ be a complex manifold. Let $p\in M$. Define the complex tangent space of $M$ at $p$ to be $$T_{\C,p}(M)=\text{Der}_\C(C_{M,p,\C}^\infty,\C)$$
\end{defn}

Let $u:\C\to\C$ be a function. Identifying $\C$ with $\R^2$, we obtain the notion of smooth functions. And smoothness of $u$ can be checked on the components of $u$. This gives a decomposition of tangent spaces. 

\begin{lmm}{}{}\\
Let $M$ be a complex manifold. Let $p\in M$. Then we have a natural $\C$-vector space isomorphism $$T_{\C,p}(M)\cong T_p(M)\otimes_\R\C$$
\end{lmm}

In particular, the real dimension of $T_{\C,p}(M)$ is $4\dim_\C(M)$, and a $\C$-basis for $T_{\C,p}(M)$ is given by lemma 3.5.1. This is the effect of considering $T_{\C,p}(M)$ as an extension of scalars from $T_p(M)$. 

\begin{prp}{}{}\\
Let $M$ be a complex manifold. Then $M$ is an almost complex manifold. 
\end{prp}

We know from Complex Analysis that not every smooth function is holomorphic. A smooth function must satisfy the Cauchy-Riemann equations in order to be holomorphic. We will see in a moment that the complex tangent space contains the holomorphic tangent space defined earlier. 

\subsection{The Holomorphic Tangent Space}
Let $M$ be a smooth manifold. Recall that the tangent space of $M$ is defined to be $$\text{Der}_\R(C_{M,p}^\infty,C_{M,p}^\infty/m_p)$$ where $C_{M,p}^\infty$ is the stalk of the sheaf $C_M^\infty(-)$ of smooth functions on $M$ at $p$. Now if $M$ is a complex manifold, the canonical maps we care about are holomorphic maps and the sheaf of holomorphic functions $\mO_M(-)$. Therefore we define the holomorphic tangent space in a similar way. 

\begin{defn}{The Holomorphic Tangent Space}{}\\
Let $M$ be a complex manifold. Let $p\in M$. Define the Holomorphic Tangent space of $M$ at $p$ to be $$T_p^{\text{Holo}}(M)=\text{Der}_\C\left(\mO_{M,p},\C\right)$$ where $\C$ is a $\mO_{M,p}$-module via the isomorphism $\C\cong\frac{\mO_{M,p}}{m_p}$. 
\end{defn}

Immediately one is curious of the relation between $T_p^\text{Holo}(M)$ and $T_p(M)$ since every complex manifold is a smooth manifold of double the dimension. 

\begin{prp}{}{}\\
Let $M$ be a complex manifold. Let $p\in M$. Then we have $T_p^{1,0}(M)=T_p^\text{Holo}(M)$. 
\end{prp}

\subsection{The Induced Map and the Jacobians}
Let $M,N$ be complex manifolds. Let $f:M\to N$ be a smooth map. Let $p\in M$. Let $(U,x^1,\dots,x^n,y^1,\dots,y^n)$ be a smooth chart at $p$. Let $(V,u^1,\dots,u^m,v^1,\dots,v^m)$ be a smooth chart at $f(p)$. Recall that the $\R$-linear map $(f_\ast)_p:T_p(M)\to T_p(N)$ is given by the Jacobian which can be compactly written into block matrix form $$(f_\ast)_p=\begin{pmatrix}
\frac{\partial u^i\circ f}{\partial x^j}\bigg{|}_p & \frac{\partial u^i\circ f}{\partial y^j}\bigg{|}_p\\
\frac{\partial v^i\circ f}{\partial x^j}\bigg{|}_p & \frac{\partial v^i\circ f}{\partial y^j}\bigg{|}_p
\end{pmatrix}$$

We may ask what the Jacobian looks like under our newly acquired basis from the almost complex structure. 

\begin{defn}{The Holomorphic Jacobian}{}\\
Let $M,N$ be complex manifolds. Let $f:M\to N$ be a smooth map. Let $p\in M$. Let $(U,x^1,\dots,x^n,y^1,\dots,y^n)$ be a smooth chart at $p$. Let $(V,u^1,\dots,u^m,v^1,\dots,v^m)$ be a smooth chart at $f(p)$. Define the holomorphic Jacobian of $f$ with respect to the charts to be $$J(f)=\begin{pmatrix}
\frac{\partial(u^1+iv^1)\circ f}{\partial z^1}\bigg{|}_p & \cdots & \frac{\partial(u^1+iv^1)\circ f}{\partial z^n}\bigg{|}_p\\
\vdots & \ddots & \vdots\\
\frac{\partial(u^m+iv^m)\circ f}{\partial z^1}\bigg{|}_p & \cdots & \frac{\partial(u^m+iv^m)\circ f}{\partial z^n}\bigg{|}_p
\end{pmatrix}$$
\end{defn}

\begin{prp}{}{}\\
Let $M,N$ be complex manifolds. Let $f:M\to N$ be a smooth map. Let $p\in M$. Let $(U,x^1,\dots,x^n,y^1,\dots,y^n)$ be a smooth chart at $p$. Let $(V,u^1,\dots,u^m,v^1,\dots,v^m)$ be a smooth chart at $f(p)$. Then the differential $(f_\ast)_p$ under the basis induced from the almost complex structure is given in block matrix form by $$(f_\ast)_p=\begin{pmatrix}
J(f) & 0\\
0 & \overline{J(f)}
\end{pmatrix}$$
\end{prp}

\begin{prp}{}{}\\
Let $M,N$ be complex manifolds. Let $f:M\to N$ be a holomorphic map. Let $p\in M$. Let $(U,x^1,\dots,x^n,y^1,\dots,y^n)$ be a smooth chart at $p$. Let $(V,u^1,\dots,u^m,v^1,\dots,v^m)$ be a smooth chart at $f(p)$. Then $(f_\ast)_p:T_p(M)\to T_p(N)$ restricts to an $\C$-linear map $T_p^{1,0}(M)\to T_{f(p)}^{1,0}(N)$. 
\end{prp}

\subsection{Dolbeault Cohomology}
\begin{defn}{Dolbeault Operators}{}\\
Let $M$ be a complex manifold. Define the Dolbeault operators on $M$ by $$\partial=\text{proj}_{p+1,q}\circ d:\Omega^{p,q}\to\Omega^{p+1,q}\;\;\;\text{ and }\;\;\;\overline{\partial}=\text{proj}_{p,q+1}\circ d:\Omega^{p,q}\to\Omega^{p,q+1}$$
\end{defn}

\begin{prp}{}{}\\
Let $M$ be a complex manifold. Let $p\in M$. Let $(U,z^1,\dots,z^n)$ be a local chart at $p$. Suppose that $\omega\in\Omega_M^{p,q}$ is given locally by $$\omega=\sum_{\abs{I}=p,\abs{J}=q}f_{I,J}dz^I\wedge d\overline{z^J}$$ where $f_{I,J}:U\to\C$ are smooth functions. Then the following are true. 
\begin{itemize}
\item $\partial\omega$ is given locally on the same chart by $$\partial\omega=\sum_{\abs{I}=p,\abs{J}=q}\sum_k\frac{\partial f_{I,J}}{\partial z^l}dz^l\wedge dz^I\wedge\overline{z^J}$$
\item $\overline{\partial}\omega$ is given locally on the same chart by $$\overline{\partial}\omega=\sum_{\abs{I}=p,\abs{J}=q}\sum_k\frac{\partial f_{I,J}}{\partial\overline{z^l}}d\overline{z^l}\wedge dz^I\wedge\overline{z^J}$$
\end{itemize}
\end{prp}

\begin{prp}{}{}\\
Let $M$ be a complex manifold. Then the following are true. 
\begin{itemize}
\item $d=\partial+\overline{\partial}$. 
\item $\partial\circ\partial=\overline{\partial}\circ\overline{\partial}=0$. 
\item $\partial\circ\overline{\partial}=-\overline{\partial}\circ\partial$. 
\item Let $\omega\in\Omega_{M,\C}^k(M)$ and $\eta\in\Omega_{M,\C}^l(M)$. Then we have $$\partial(\omega\wedge\eta)=\partial(\omega)\wedge\eta+(-1)^k\omega\wedge\partial(\eta)\;\;\;\text{ and }\;\;\;\overline{\partial}(\omega\wedge\eta)=\overline{\partial}(\omega)\wedge\eta+(-1)^k\omega\wedge\overline{\partial}(\eta)$$
\end{itemize}
\end{prp}

We can now write the bicomplex in a grid: \\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	\vdots & \vdots & \vdots \\
	{\Omega^{0,2}(M)} & {\Omega^{1,2}(M)} & {\Omega^{2,2}(M)} & \cdots \\
	{\Omega^{0,1}(M)} & {\Omega^{1,1}(M)} & {\Omega^{2,1}(M)} & \cdots \\
	{\mO_M(M)} & {\Omega^{1,0}(M)} & {\Omega^{2,0}(M)} & \cdots
	\arrow[from=2-1, to=1-1]
	\arrow["\partial", from=2-1, to=2-2]
	\arrow[from=2-2, to=1-2]
	\arrow["\partial", from=2-2, to=2-3]
	\arrow[from=2-3, to=1-3]
	\arrow[from=2-3, to=2-4]
	\arrow["{\overline{\partial}}", from=3-1, to=2-1]
	\arrow["\partial", from=3-1, to=3-2]
	\arrow["{\overline{\partial}}", from=3-2, to=2-2]
	\arrow["\partial", from=3-2, to=3-3]
	\arrow["{\overline{\partial}}"', from=3-3, to=2-3]
	\arrow[from=3-3, to=3-4]
	\arrow["{\overline{\partial}}", from=4-1, to=3-1]
	\arrow["\partial", from=4-1, to=4-2]
	\arrow["{\overline{\partial}}", from=4-2, to=3-2]
	\arrow["\partial", from=4-2, to=4-3]
	\arrow["{\overline{\partial}}"', from=4-3, to=3-3]
	\arrow[from=4-3, to=4-4]
\end{tikzcd}}\\
Note that the squares here are not commutative, and this is not the notation we have defined for bicomplexes. If we augment the differential in the total complex by accounting the anti-commutativity of the squares, we would obtain isomorphisms $$H_\text{dR}^k(M;\C)=H^k(\text{Tot}(\Omega^{\bullet,\bullet}(M),\partial,\overline{\partial}))$$

\begin{defn}{Holomorphic de Rham Cohomology}{}\\
Let $M$ be a complex manifold. Define the $n$th holomorphic de Rham cohomology group of $M$ to be $$H_\text{Holo}^n(M)=H^n(\Omega_M^{\bullet,0}(M),\partial)$$
\end{defn}

\begin{defn}{Dolbeault Complex}{} Let $M$ be a complex manifold. Let $p\in\N$. Define the $p$th Dolbeault complex of $M$ to be the cochain complex \\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
\cdots\arrow[r] & \Omega^{p,q-1}(M)\arrow[r, "\overline{\partial}"] & \Omega^{p,q}(M)\arrow[r, "\overline{\partial}"] & \Omega^{p,q+1}(M)\arrow[r, "\overline{\partial}"] & \cdots
\end{tikzcd}}~\\
denoted $\Omega^{p,\bullet}(M)$. 
\end{defn}

\begin{defn}{Dolbeault Cohomology}{} Define the dobeault cohomology of a complex manifold $M$ to be $$H^{p,q}(M)=\frac{\ker(\bar{\partial}:\Omega^{p,q}(M)\to\Omega^{p,q+1}(M)}{\im(\bar{\partial}:\Omega^{p,q-1}(M)\to\Omega^{p,q}(M))}=H^q(\Omega^{p,\bullet}(M),\overline{\partial})$$
\end{defn}

We may relate Dolbeault cohomology with sheaf cohomology. 

\begin{thm}{}{}\\
Let $M$ be a complex manifold. Then we have an isomorphism $$H^{p,q}(M)\cong H^q(M,\Omega_M^p)$$
\end{thm}

\begin{crl}{}{}\\
Let $M$ be a compact complex manifold. Then $H^{0,0}(M)=\C$. 
\end{crl}

\begin{lmm}{}{}\\
Let $M$ be a complex manifold. Then $H^{0,q}(M)=0$ for all $q>\dim(M)$. 
\end{lmm}

\begin{thm}{Serre Duality}{}\\
Let $M$ be a compact complex manifold. Then we have $$H^{p,q}(M)\cong H^{\dim(X)-p,\dim(X)-q}(M)$$
\end{thm}

\section{Hermitian Manifolds}
\subsection{Hermitian Metric}
\begin{defn}{Hermitian Metric on a Vector Bundle}{}\\
Let $M$ be a smooth manifold. Let $p:E\to M$ be a complex vector bundle. A Hermitian metric on $M$ is a smooth section $g\in\Gamma(M,(E\otimes\overline{E})^\ast)$ such that $g_p:E_p\times E_p\to\R$ is a Hermitian inner product for all $p\in M$. 
\end{defn}

Equivalently, a Hermitian metric is an assignment $p\mapsto\langle-,-\rangle_p$ such that for any smooth sections $s,t\in\Gamma(M,E)$, the map $p\mapsto\langle s(p),t(p)\rangle_p$ is smooth. 

\begin{defn}{Hermitian Manifolds}{}\\
Let $M$ be a complex manifold. We say that $M$ is Hermitian if $T^{1,0}(M)$ admits a Hermitian metric. 
\end{defn}

\begin{prp}{}{}\\
Let $(M,h)$ be a Hermitian manifold. Let $p\in M$. Let $(U,z^1,\dots,z^n)$ be a complex chart at $p$. Then $h_p$ is given locally on the chart by $$h_p=\sum_{i,j=1}^nh_{i,j}|_p\;(dz^i)_p\otimes(d\overline{z^j})_p$$ where $h_{i,j}|_p=h_p\left(\frac{\partial}{\partial z^i}\bigg{|}_p,\frac{\partial}{\partial z^j}\bigg{|}_p\right)$. 
\end{prp}

\begin{prp}{}{}\\
Let $M$ be a smooth manifold. Let $p:E\to M$ be a complex vector bundle. Then $E$ admits a Hermitian metric. 
\end{prp}

\begin{crl}{}{}\\
Let $M$ be a complex manifold. Then $M$ is a Hermitian manifold. 
\end{crl}

\subsection{The Associated Form}
\begin{defn}{The Associated Form}{}\\
Let $(M,h)$ be a Hermitian manifold. Define the associated form of $h$ to be $$\omega=\frac{i}{2}(h-\overline{h})$$
\end{defn}

In local coordinates, $\omega$ is expressed as $$\omega=\frac{i}{2}\sum_{\alpha,\beta=1}^n h_{\alpha\overline{\beta}}dz_{\alpha}\wedge d\overline{z}_\beta$$ if $M$ is a complex manifold of complex dimension $n$. 

\begin{defn}{The Associated Riemannian Metric}{}\\
Let $(M,h)$ be a Hermitian manifold. Define the associated Riemannian metric of $h$ to be $$g=\frac{1}{2}(h+\overline{h})=$$
\end{defn}

In local coordinates, $g$ is expressed as $$g(u,v)=\frac{1}{2}\sum h_{\alpha\overline{\beta}}(dz_\alpha\otimes d\overline{z}_\beta+d\overline{z}_\beta\otimes dz_\alpha)$$

\begin{lmm}{}{}\\
Let $(M,h)$ be a Hermitian manifold. Then we have $\omega$ is a $(1,1)$-form. 
\end{lmm}

\begin{lmm}{}{}\\
Let $(M,h)$ be a Hermitian manifold. Then we have $$h=g-i\omega$$
\end{lmm}

\begin{prp}{}{}\\
Let $(M,h)$ be a Hermitian manifold whose associated almost complex structure is given by $J$. Then the following are true. 
\begin{itemize}
\item $h(Ju,Jv)=h(u,v)$
\item $g(Ju,Jv)=g(u,v)$
\item $\omega(Ju,Jv)=\omega(u,v)$
\end{itemize}
\end{prp}

\begin{prp}{}{}\\
Let $(M,h)$ be a Hermitian manifold whose associated almost complex structure is given by $J$. Then the following are true. 
\begin{itemize}
\item $\omega(u,v)=g(Ju,v)$. 
\item $g(u,v)=\omega(u,Jv)$. 
\end{itemize}
\end{prp}

Notice that the three equations $$h=g-i\omega\;\;\;\text{ and }\;\;\;\omega(u,v)=g(Ju,v)\;\;\;\text{ and }\;\;\;g(u,v)=\omega(u,Jv)$$ completely determines the triple $(g,h,\omega)$. 

\begin{prp}{}{}\\
Let $M$ be a complex manifold whose associated almost complex structure is given by $J$. Let $g$ be a Riemannian metric on $M$. Then the following are true. 
\begin{itemize}
\item $h=g-i\omega$ is a Hermitian form on $M$ where $\omega(u,v)=g(Ju,v)$. 
\item $\omega$ is the associated form of $h$. 
\end{itemize}
\end{prp}

\begin{prp}{}{}\\
Let $M$ be a complex manifold whose associated almost complex structure is given by $J$. Let $\omega$ be a $(1,1)$-form on $M$ that is non-degenerate, and preserves $J$ and is positive definite in the sense that $\omega(u,Jv)>0$ for all non-zero $u,v$. Then the following are true. 
\begin{itemize}
\item $g(u,v)=\omega(u,Jv)$ is a Riemannian metric on $M$. 
\item $h=g-i\omega$ is a Hermitian metric on $M$. 
\end{itemize}
\end{prp}

\section{Hodge Theory}
\subsection{Hodge Theory for Smooth Manifolds}
Let $V$ be a finite dimensional inner product over $\R$. The inner product induces an inner product on $\Lambda^k(V)$ given by $$\langle\alpha_{i_1}\wedge\cdots\wedge\alpha_{i_k},\beta_{j_1}\wedge\cdots\wedge\beta_{j_k}\rangle=\det\begin{pmatrix}
\langle\alpha_{i_1},\beta_{j_1}\rangle & \cdots & \langle\alpha_{i_1}\wedge\beta_{j_k}\rangle\\
\vdots & \ddots & \vdots\\
\langle\alpha_{i_k},\beta_{j_1}\rangle & \cdots & \langle\alpha_{i_k}\wedge\beta_{j_k}\rangle
\end{pmatrix}$$

\begin{defn}{The $L^2$-Norm}{}\\
Let $(M,g)$ be an orientable Riemmanian manifold. Let $\alpha,\beta\in\Omega^k(M)$ be differential forms of compact support. Define the $L^2$ norm on $M$ to be $$\langle\alpha,\beta\rangle_{L^2}=\int_M\langle\alpha,\beta\rangle\text{Vol}$$ where $\langle-,-\rangle$ is the induced metric on $\Omega^k(M)$ by $g$. 
\end{defn}

\begin{lmm}{}{}\\
Let $(M,g)$ be an orientable Riemmanian manifold. Then $\langle-,-\rangle_{L^2}$ defines a norm on the vector space of compactly supported differential forms on $M$. 
\end{lmm}

\begin{defn}{The Hodge Star Operator}{}\\
Let $(M,g)$ be an orientable Riemannian manifold. Let $n=\dim(M)$. 
\begin{itemize}
\item Let $p\in M$. Define $\ast_p:\Lambda^kT_p^\ast(M)\to\Lambda^{n-k}T_p^\ast(M)$ to be the unique $\C$-linear isomorphism such that $$\alpha\wedge(\ast_p\beta)=\langle\alpha,\beta\rangle_p\text{Vol}_p$$ where $\alpha,\beta\in\Lambda^kT_p^\ast(M)$ and $\text{Vol}$ is the volume form of $M$ associated to $g$, and $\langle-,-\rangle_p$ is the induced metric on $\Lambda^k(T_p^\ast(M))$. 
\item Define the Hodge star operator of $M$ to be the map $$\ast\in\Gamma(M,\Hom(\Omega^k(M),\Omega^{n-k}(M)))$$ given by $p\mapsto\ast_p$. 
\end{itemize}
\end{defn}

\begin{lmm}{}{}\\
Let $(M,g)$ be an orientable Riemannian manifold. Then we have $$\ast^2=(-1)^{k(n-k)}$$ on $\Omega^k(M)$. 
\end{lmm}

\begin{lmm}{}{}\\
Let $(M,g)$ be an orientable Riemmanian manifold. Let $\alpha,\beta\in\Omega^k(M)$ be differential forms. Then we have $$\langle\alpha,\beta\rangle_{L^2}=\int_M\alpha\wedge\ast\beta$$
\end{lmm}

\begin{defn}{Adjoint Differential Operator}{}\\
Let $(M,g)$ be an orientable Riemannian manifold. Define the operator $d^\ast:\Omega^{k+1}(M)\to\Omega^k(M)$ by $$d^\ast=(-1)^k(\ast^{-1}\circ d\circ\ast)$$
\end{defn}

\begin{lmm}{}{}\\
Let $(M,g)$ be an orientable Riemannian manifold. Then we have $$d^\ast=(-1)^{n(k+1)+1}(\ast\circ d\circ\ast)$$
\end{lmm}

\begin{prp}{}{}\\
Let $(M,g)$ be an orientable Riemannian manifold. Let $\alpha\in\Omega^k(M)$ and $\beta\in\Omega^{k+1}(M)$ be differential forms with compact support. Then we have $$\langle\alpha,d^\ast\beta\rangle_{L^2}=\langle d\alpha,\beta\rangle_{L^2}$$
\end{prp}

\begin{defn}{The Laplace-Beltrami Operator}{}\\
Let $M$ be an orientable Riemannian manifold. Define the Laplace-Beltrami operator on $M$ to be the map $\Delta:\Omega^k(M)\to\Omega^k(M)$ given by $$\Delta=d^\ast\circ d+d\circ d^\ast$$
\end{defn}

\begin{lmm}{}{}\\
Let $M$ be an orientable Riemannian manifold. Then the following are true regarding the Laplace-Beltrami operator. 
\begin{itemize}
\item $\ast\circ\Delta=\Delta\circ\ast$. 
\item $\langle\Delta\alpha,\beta\rangle_{L^2}=\langle\alpha,\Delta\beta\rangle_{L^2}$ for any compactly supported differential forms $\alpha,\beta\in\Omega^k(M)$. 
\end{itemize}
\end{lmm}

\begin{defn}{Harmonic Forms}{}\\
Let $M$ be an orientable Riemannian manifold. Let $\alpha\in\Omega^k(M)$. We say that $\alpha$ is harmonic if $\alpha\in\ker(\Delta)$. 
\end{defn}

\begin{lmm}{}{}\\
Let $M$ be an orientable Riemannian manifold. Let $\alpha\in\Omega^k(M)$. Then $\alpha$ is harmonic if and only if $d(\alpha)=0$ and $d^\ast(\alpha)=0$. 
\end{lmm}

\begin{defn}{The Space of Harmonic Forms}{}\\
Let $M$ be an orientable Riemannian manifold. Define the space of harmonic $k$-forms to be $$\mH^k(M)=\ker(\Delta)$$
\end{defn}

\begin{thm}{Hodge Theorem}{}\\
Let $M$ be an compact orientable Riemannian manifold. Then the following are true. 
\begin{itemize}
\item $\mH^k$ is finite dimensional for all $k\in\N$. 
\item There is a decomposition $$\Omega^k(M)\cong\mH^k(M)\oplus\im(\Delta)$$
\end{itemize}
\end{thm}

\begin{crl}{}{}\\
Let $M$ be an orientable Riemannian manifold. Then the quotient map induces an isomorphism $$\mH^k(M)\cong H_\text{dR}^k(M)$$ for all $k\in\N$. 
\end{crl}

\subsection{Hodge Star Operator for Complex Manifolds}
\begin{defn}{The Hodge Star Operator}{}\\
Let $(M,h)$ be a complex Hermitian manifold. Let $n=\dim(M)$. 
\begin{itemize}
\item Let $p\in M$. Define $\ast_p:\Lambda^kT_p^\ast(M)\to\Lambda^{n-k}T_p^\ast(M)$ to be the unique $\C$-linear isomorphism such that $$\alpha\wedge\overline{\ast_p\beta}=\langle\alpha,\beta\rangle_p\text{Vol}_p=\det\begin{pmatrix}
h_p(\alpha_{i_1},\beta_{j_1}) & \cdots & h_p(\alpha_{i_1}\wedge\beta_{j_k})\\
\vdots & \ddots & \vdots\\
h_p(\alpha_{i_k},\beta_{j_1}) & \cdots & h_p(\alpha_{i_k}\wedge\beta_{j_k})
\end{pmatrix}\text{Vol}_p$$ where $\alpha,\beta\in\Lambda^kT_p^\ast(M)$ and $\text{Vol}$ is the volume form of $M$, and $\langle-,-\rangle_p$ is the induced metric on $\Lambda^k(T_p^\ast(M))$. 
\item Define the Hodge star operator of $M$ to be the map $$\ast\in\Gamma(M,\Hom(\Omega^k(M),\Omega^{n-k}(M)))$$ given by $p\mapsto\ast_p$. 
\end{itemize}
\end{defn}

\begin{lmm}{}{}\\
Let $M$ be a complex manifold. Then we have $$\ast^2=(-1)^{k(n-k)}$$ on $\Omega^k(M)$. 
\end{lmm}

\begin{defn}{Adjoint Dolbeault Operators}{}\\
Let $M$ be a complex manifold. 
\begin{itemize}
\item Define the operator $d^\ast:\Omega^{k+1}(M)\to\Omega^k(M)$ by $$d^\ast=(-1)^k(\ast^{-1}\circ d\circ\ast)$$
\item Define the operators $\partial^\ast:\Omega^{k+1}(M)\to\Omega^k(M)$ by $\partial^\ast=-(\ast\circ\overline{\partial}\circ\ast)$. 
\item Define the operators $\overline{\partial}^\ast:\Omega^{k+1}(M)\to\Omega^k(M)$ by $\overline{\partial}^\ast=-(\ast\circ\partial\circ\ast)$. 
\end{itemize}
\end{defn}

\begin{defn}{The $L^2$-Norm}{}\\
Let $(M,h)$ be a complex Hermitian manifold. Let $\alpha,\beta\in\Omega^k(M)$ be differential forms of compact support. Define the $L^2$ norm on $M$ to be $$\langle\alpha,\beta\rangle_{L^2}=\int_M\langle\alpha,\beta\rangle\text{Vol}$$ where $\langle-,-\rangle$ is the induced metric on $\Omega^k(M)$ from $h$. 
\end{defn}

\begin{lmm}{}{}\\
Let $M$ be a complex manifold. If $M$ is compact, then $\langle-,-\rangle_{L^2}$ defines a norm on $\Omega^k(M)$. 
\end{lmm}

\begin{lmm}{}{}\\
Let $M$ be a complex manifold. Let $\alpha,\beta\in\Omega^k(M)$ be differential forms of compact support. Then we have $$\langle\alpha,\beta\rangle_{L^2}=\int_M\alpha\wedge\overline{\ast\beta}$$
\end{lmm}

\begin{prp}{}{}\\
Let $M$ be a complex manifold. Let $\alpha\in\Omega^k(M)$ and $\beta\in\Omega^{k+1}(M)$ be differential forms of compact support. Then the following are true. 
\begin{itemize}
\item $d$ and $d^\ast$ are formal adjoints as linear operators over the $L^2$ norm. This means that $$\langle\alpha,d^\ast\beta\rangle_{L^2}=\langle d\alpha,\beta\rangle_{L^2}$$
\item $\partial$ and $\partial^\ast$ are formal adjoints as linear operators over the $L^2$ norm. This means that $$\langle\alpha,\partial^\ast\beta\rangle_{L^2}=\langle\partial\alpha,\beta\rangle_{L^2}$$
\item $\overline{\partial}$ and $\overline{\partial}^\ast$ are formal adjoints as linear operators over the $L^2$ norm. This means that $$\langle\alpha,\overline{\partial}^\ast\beta\rangle_{L^2}=\langle\overline{\partial}\alpha,\beta\rangle_{L^2}$$
\end{itemize}
\end{prp}
