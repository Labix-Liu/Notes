\section{Complex Manifolds}
\subsection{Basic Definitions}
\begin{defn}{The Holomorphic Pseudogroup of Transformations}{}\\
Define the holomorphic Pseudogroup of transformations to be $$\Gamma_\text{Holo}=\{f:U\to V\;|\;f\text{ is biholomorphic and }U,V\subseteq\C^n\text{ are open }\}$$
\end{defn}

\begin{defn}{Complex Manifolds}{}\\
A complex manifold $M$ is a manifold whose atlas is given by $\Gamma_\text{Holo}$. 
\end{defn}

\begin{prp}{}{} Every complex manifold is a smooth real manifold. 
\begin{proof}
Let $(U,\phi=z_1,\dots,z_n))$ be a chart of a complex manifold $M$. Write $z_k=x_k+iy_k$ where $x_k$ and $y_k$ are smooth real valued functions on $U$. Then $(x_1,\dots,x_n,y_1,\dots,y_n)$ gives a homeomorphism between $U$ and an open subset of $\R^{2n}$. 
\end{proof}
\end{prp}

\begin{lmm}{}{}\\
Let $M$ be a complex manifold. Let $U\subseteq M$ be an open subset. Then $U$ is a complex manifold. 
\end{lmm}

\begin{defn}{Almost Complex Manifolds}{}\\
An almost complex manifold is a smooth manifold $M$ together with an almost complex structure on $TM$. 
\end{defn}

Even though every even dimensional vector space has a linear complex structure, they may not glue smoothly into an almost complex structure for a smooth manifold. This is a question of the reduction of the structure group from $GL_{2n}(\R)$ to the subgroup $GL_n(\C)$. 

\subsection{The Sheaf of Holomorphic Forms}
\begin{defn}{The Sheaf of Holomorphic Forms}{}\\
Let $M$ be a complex manifold. Define the sheaf of holomorphic form $\mO_M$ as follows. 
\begin{itemize}
\item For each open set $U\subseteq M$, define $\mO_M(U)=\{f:U\to\C\;|\;f\text{ is holomorphic }\}$. 
\item For each inclusion of open sets $V\hookrightarrow U$, the map $\mO_M(U)\to\mO_M(V)$ is given by the restriction map $f\mapsto f|_V$. 
\end{itemize}
\end{defn}

\begin{lmm}{}{}\\
Let $M$ be a compact connected complex manifold. Then we have $$\mO_M(M)\cong\C$$
\end{lmm}

\begin{thm}{Oka's (Coherence) Theorem}{}\\
Let $M$ be a complex manifold. Then $\mO_M$ is a coherent sheaf. 
\end{thm}

\begin{defn}{The Sheaf of Meromorphic Forms}{}\\
Let $M$ be a complex manifold. Define the sheaf of meromorphic form $\mM_M$ as follows. 
\begin{itemize}
\item For each open set $U\subseteq M$, define $\mM_M(U)=\text{Frac}(\mO_M(U))$. 
\item For each inclusion of open sets $V\hookrightarrow U$, the map $\mM_M(U)\to\mM_M(V)$ is given by the restriction map $f\mapsto f|_V$. 
\end{itemize}
\end{defn}

\subsection{Complex Submanifolds}
\begin{defn}{Complex Submanifolds}{} Let $M$ be an $n$ dimensional manifold. An $m$ dimensional submanifold is a subset $Y$ of $M$ such that for every $y\in Y$, there exists a chart $(U,\phi)$ of $M$ such that $$\phi(Y\cap U)=\phi(U)\cap\{0\}^{n-m}\times\C^m=\{(z_1,\dots,z_n)\in\phi(U)|z_{n-m+1}=\dots=z_n=0\}$$
\end{defn}

\section{More on Vector Bundles}
\subsection{Holomorphic Vector Bundles}
\begin{defn}{Holomorphic Vector Bundles}{}\\
Let $M$ be a complex manifold. A holomorphic vector bundle is a vector bundle $p:E\to B$ over $\C$ such that the following are true. 
\begin{itemize}
\item $E$ is a complex manifold. 
\item $p$ is holomorphic. 
\item The local trivialization charts $\phi:p^{-1}(U)\overset{\cong}{\to}U\times\C^r$ are biholomorphic. 
\end{itemize}
\end{defn}

\begin{prp}{}{}\\
Let $M$ be a complex manifold. Let $E$ be a complex manifold and $p:E\to M$ a vector bundle over $\C$. Then $p$ is a holomorphic vector bundle if and only if the transition maps $g_{i,j}:U_i\cap U_j\to\text{GL}(n,\C)$ are holomorphic. 
\end{prp}

\begin{defn}{Holomorphic Sections}{}\\
Let $M$ be complex manifold. Let $p:E\to M$ be a holomorphic vector bundle. Let $U\subseteq M$. Define the holomorphic sections from $U$ to $E$ to be $$\Gamma^\text{hol}(U,E)=\{s\in\Gamma(U,E)\;|\;s\text{ is holomorphic }\}$$
\end{defn}

\subsection{Vector Bundles and Sheaves}
\begin{thm}{Serre's Correspondence for Topological Manifolds}{}\\
Let $M$ be a topological manifold. Then there is an equivalence of categories $$\bold{Vect}^\R(M)\simeq\bold{LF}_{\mC_M^0}$$
\end{thm}

\begin{thm}{Serre's Correspondence for Smooth Manifolds}{}\\
Let $M$ be a smooth manifold. Then there are equivalence of categories $$\bold{SmVect}^\R(M)\simeq\bold{LF}_{\mC_M^\infty}\;\;\;\text{ and }\;\;\;\bold{SmVect}^\C(M)\simeq\bold{LF}_{\mC_M^\infty\otimes_\R\C}$$
\end{thm}

\begin{thm}{Serre's Correspondence for Complex Manifolds}{}\\
Let $M$ be a complex manifold. Then there is an equivalence of categories $$\bold{HolVect}^\C(M)\simeq\bold{LF}_{\mO_M}$$
\end{thm}

\begin{thm}{Serre-Swan Theorem for Manifolds}{}\\
Let $M$ be a smooth manifold. Let $p:E\to M$ be a smooth vector bundle over $M$. Then the following are true. 
\begin{itemize}
\item $\mC^\infty(E)$ is a finitely generated and projective $\mC^\infty(M)$-module. 
\item Suppose that $M$ is connected. There is an equivalence of categories $$\mC^\infty:\bold{SmVect}^\R(M)\overset{\simeq}{\to}\bold{proj}(\mC^\infty(M))$$
\end{itemize}
\end{thm}

\begin{thm}{Serre-Swan Theorem}{}\\
Let $X$ be a compact Hausdorff space. Then there are equivalence of categories $$\Gamma:\bold{Vect}^\R(X)\overset{\simeq}{\to}\bold{proj}(\mC^0(X))\;\;\;\text{ and }\;\;\;\Gamma:\bold{Vect}^\C(X)\overset{\simeq}{\to}\bold{proj}(\mC^0(X)\otimes_\R\C)$$
\end{thm}

\section{Tangent Spaces}
\subsection{The Complex Structure of Tangent Spaces on Almost Complex Manifolds}
\begin{defn}{Holomorphic and Antiholomorphic Tangent Space}{}\\
Let $(M,J)$ be an almost complex manifold. 
\begin{itemize}
\item Define $T_p^{1,0}(M)$ to be the eigenspace of $i$ of the linear complex structure $J_p$ on $T_{\C,p}(M)$. 
\item Define $T_p^{0,1}(M)$ to be the eigenspace of $-i$ of the linear complex structure $J_p$ on $T_{\C,p}(M)$. 
\end{itemize}
\end{defn}

We have seen above that $$T_pM\otimes_R\C\cong T_p^{(1,0)}(M)\oplus T_p^{(0,1)}(M)$$ as complex vector spaces. 

\begin{lmm}{}{}\\
Let $(M,J)$ be an almost complex manifold. Let $p\in M$. Let $(U,x^1,\dots,x^n,y^1,\dots,y^n)$ be a smooth chart at $p$. Suppose that $J_p\left(\frac{\partial}{\partial x^i}\bigg{|}_p\right)=\frac{\partial}{\partial y^i}\bigg{|}_p$ for $1\leq i\leq n$. Then the following are true. 
\begin{itemize}
\item A $\C$-basis for $T_p^{1,0}(M)$ is given by $$\left\{\frac{1}{2}\left(\frac{\partial}{\partial x^1}-i\frac{\partial}{\partial y^1}\right)\bigg{|}_p,\dots,\frac{1}{2}\left(\frac{\partial}{\partial x^n}-i\frac{\partial}{\partial y^n}\right)\bigg{|}_p\right\}$$
\item A $\C$-basis for $T_p^{0,1}(M)$ is given by $$\left\{\frac{1}{2}\left(\frac{\partial}{\partial x^1}+i\frac{\partial}{\partial y^1}\right)\bigg{|}_p,\dots,\frac{1}{2}\left(\frac{\partial}{\partial x^n}+i\frac{\partial}{\partial y^n}\right)\bigg{|}_p\right\}$$
\end{itemize}
\end{lmm}

Assuming that $J_p$ sends the basis vector $\frac{\partial}{\partial x^i}\bigg{|}_p$ to $\frac{\partial}{\partial y^i}\bigg{|}_p$ loses no generality since we can always choose appropriate coordinate functions so that the almost complex structure locally at $U$ looks like ours one. \\

Recall from complex analysis that we defined partial complex derivatives by $$\frac{\partial}{\partial z_k}=\frac{1}{2}\left(\frac{\partial}{\partial x_k}-i\frac{\partial}{\partial y_k}\right)\;\;\;\text{ and }\;\;\;\frac{\partial}{\partial\overline{z_k}}=\frac{1}{2}\left(\frac{\partial}{\partial x_k}+i\frac{\partial}{\partial y_k}\right)$$

\begin{defn}{The Basis for Tangent Space given by the Almost Complex Structure}{}\\
Let $(M,J)$ be an almost complex manifold. Let $p\in M$. Let $(U,x^1,\dots,x^n,y^1,\dots,y^n)$ be a smooth chart at $p$. Define the tangent vector $$\frac{\partial}{\partial z^k}\bigg{|}_p:\mC_{M,p}^\infty\to\R$$ by $$\frac{\partial}{\partial z^k}\bigg{|}_p=\left(\frac{\partial}{\partial x^k}-i\frac{\partial}{\partial y^k}\right)\bigg{|}_p$$ Define the tangent vector $$\frac{\partial}{\partial\overline{z^k}}\bigg{|}_p:\mC_{M,p}^\infty\to\R$$ by $$\frac{\partial}{\partial\overline{z^k}}\bigg{|}_p=\left(\frac{\partial}{\partial x^k}+i\frac{\partial}{\partial y^k}\right)\bigg{|}_p$$
\end{defn}

\subsection{Decomposing the Tangent Space of Complex Manifolds}
Let $M$ be a complex manifold. Let $p\in M$. Regarded as a smooth manifold, we have the notion of tangent space $T_pM=\text{Der}(C_{M,p}^\infty(M,\R),\R)$. Since we are on a complex manifold, we ought to consider smooth functions with values in $\C$ instead. 

\begin{lmm}{}{}\\
Let $M$ be a complex manifold. Let $p\in M$. Let $(U,x^1,\dots,x^n,y^1,\dots,y^n)$ be a smooth chart at $p$. Then an $\R$-basis for $T_pM$ is given by $$\left\{\frac{\partial}{\partial x^1}\bigg{|}_p,\dots,\frac{\partial}{\partial x^n}\bigg{|}_p,\frac{\partial}{\partial y^1}\bigg{|}_p,\dots,\frac{\partial}{\partial y^n}\bigg{|}_p\right\}$$
\end{lmm}

\begin{defn}{The Complex Tangent Space}{}\\
Let $M$ be a complex manifold. Let $p\in M$. Define the complex tangent space of $M$ at $p$ to be $$T_{\C,p}(M)=\text{Der}_\C(C_{M,p,\C}^\infty,\C)$$
\end{defn}

Let $u:\C\to\C$ be a function. Identifying $\C$ with $\R^2$, we obtain the notion of smooth functions. And smoothness of $u$ can be checked on the components of $u$. This gives a decomposition of tangent spaces. 

\begin{lmm}{}{}\\
Let $M$ be a complex manifold. Let $p\in M$. Then we have a natural $\C$-vector space isomorphism $$T_{\C,p}(M)\cong T_p(M)\otimes_\R\C$$
\end{lmm}

In particular, the real dimension of $T_{\C,p}(M)$ is $4\dim_\C(M)$, and a $\C$-basis for $T_{\C,p}(M)$ is given by lemma 3.5.1. This is the effect of considering $T_{\C,p}(M)$ as an extension of scalars from $T_p(M)$. 

\begin{prp}{}{}\\
Let $M$ be a complex manifold. Then $M$ is an almost complex manifold. 
\end{prp}

We know from Complex Analysis that not every smooth function is holomorphic. A smooth function must satisfy the Cauchy-Riemann equations in order to be holomorphic. We will see in a moment that the complex tangent space contains the holomorphic tangent space defined earlier. 

\subsection{The Holomorphic Tangent Space}
Let $M$ be a smooth manifold. Recall that the tangent space of $M$ is defined to be $$\text{Der}_\R(C_{M,p}^\infty,C_{M,p}^\infty/m_p)$$ where $C_{M,p}^\infty$ is the stalk of the sheaf $C_M^\infty(-)$ of smooth functions on $M$ at $p$. Now if $M$ is a complex manifold, the canonical maps we care about are holomorphic maps and the sheaf of holomorphic functions $\mO_M(-)$. Therefore we define the holomorphic tangent space in a similar way. 

\begin{defn}{The Holomorphic Tangent Space}{}\\
Let $M$ be a complex manifold. Let $p\in M$. Define the Holomorphic Tangent space of $M$ at $p$ to be $$T_p^{\text{Holo}}(M)=\text{Der}_\C\left(\mO_{M,p},\C\right)$$ where $\C$ is a $\mO_{M,p}$-module via the isomorphism $\C\cong\frac{\mO_{M,p}}{m_p}$. 
\end{defn}

Immediately one is curious of the relation between $T_p^\text{Holo}(M)$ and $T_p(M)$ since every complex manifold is a smooth manifold of double the dimension. 

\begin{prp}{}{}\\
Let $M$ be a complex manifold. Let $p\in M$. Then there is a biholomorphism $T_p^{1,0}(M)=T_p^\text{Holo}(M)$. 
\end{prp}

The way in which we may decompose the complexified tangent space into its holomorphic and antiholomorphic part works also for almost complex manifolds. In fact this motivates the theory for almost complex manifolds in the first place. 

\subsection{The Induced Map and the Jacobians}
Let $M,N$ be complex manifolds. Let $f:M\to N$ be a smooth map. Let $p\in M$. Let $(U,x^1,\dots,x^n,y^1,\dots,y^n)$ be a smooth chart at $p$. Let $(V,u^1,\dots,u^m,v^1,\dots,v^m)$ be a smooth chart at $f(p)$. Recall that the $\R$-linear map $(f_\ast)_p:T_p(M)\to T_p(N)$ is given by the Jacobian which can be compactly written into block matrix form $$(f_\ast)_p=\begin{pmatrix}
\frac{\partial u^i\circ f}{\partial x^j}\bigg{|}_p & \frac{\partial u^i\circ f}{\partial y^j}\bigg{|}_p\\
\frac{\partial v^i\circ f}{\partial x^j}\bigg{|}_p & \frac{\partial v^i\circ f}{\partial y^j}\bigg{|}_p
\end{pmatrix}$$

We may ask what the Jacobian looks like under our newly acquired basis from the almost complex structure. 

\begin{defn}{The Holomorphic Jacobian}{}\\
Let $M,N$ be complex manifolds. Let $f:M\to N$ be a smooth map. Let $p\in M$. Let $(U,x^1,\dots,x^n,y^1,\dots,y^n)$ be a smooth chart at $p$. Let $(V,u^1,\dots,u^m,v^1,\dots,v^m)$ be a smooth chart at $f(p)$. Define the holomorphic Jacobian of $f$ with respect to the charts to be $$J(f)=\begin{pmatrix}
\frac{\partial(u^1+iv^1)\circ f}{\partial z^1}\bigg{|}_p & \cdots & \frac{\partial(u^1+iv^1)\circ f}{\partial z^n}\bigg{|}_p\\
\vdots & \ddots & \vdots\\
\frac{\partial(u^m+iv^m)\circ f}{\partial z^1}\bigg{|}_p & \cdots & \frac{\partial(u^m+iv^m)\circ f}{\partial z^n}\bigg{|}_p
\end{pmatrix}$$
\end{defn}

\begin{prp}{}{}\\
Let $M,N$ be complex manifolds. Let $f:M\to N$ be a smooth map. Let $p\in M$. Let $(U,x^1,\dots,x^n,y^1,\dots,y^n)$ be a smooth chart at $p$. Let $(V,u^1,\dots,u^m,v^1,\dots,v^m)$ be a smooth chart at $f(p)$. Then the differential $(f_\ast)_p$ under the basis induced from the almost complex structure is given in block matrix form by $$(f_\ast)_p=\begin{pmatrix}
J(f) & 0\\
0 & \overline{J(f)}
\end{pmatrix}$$
\end{prp}

\begin{prp}{}{}\\
Let $M,N$ be complex manifolds. Let $f:M\to N$ be a holomorphic map. Let $p\in M$. Let $(U,x^1,\dots,x^n,y^1,\dots,y^n)$ be a smooth chart at $p$. Let $(V,u^1,\dots,u^m,v^1,\dots,v^m)$ be a smooth chart at $f(p)$. Then $(f_\ast)_p:T_p(M)\to T_p(N)$ restricts to an $\C$-linear map $T_p^{1,0}(M)\to T_{f(p)}^{1,0}(N)$. 
\end{prp}

\section{Complex Differential Forms}
\subsection{Complexified Differential Forms}
\begin{defn}{The Sheaf of Complex Valued Forms}{}\\
Let $M$ be a complex manifold. Define the sheaf of complex valued $k$-forms $\Omega_{M,\C}^k$ of $M$ to be the sheaf of smooth sections of $\Lambda^k(T_\C(M))$. Explicitly, we have $$\Omega_{M,\C}^k(U)=\Gamma^\infty(U,\Lambda^k(T_\C(M))$$
\end{defn}

\begin{lmm}{}{}\\
Let $M$ be a complex manifold. Then $\Omega_{M,\C}^k$ is a fine sheaf of $\mC_{M,\C}^\infty$-modules of rank $\binom{2n}{k}$. 
\end{lmm}

\begin{lmm}{}{}\\
Let $M$ be a complex manifold. Then there is an isomorphism $$\Omega_{M,\C}^k\cong\Omega_{M,\R}^k\otimes_{\underline{\R}}\underline{\C}$$
\end{lmm}

\begin{lmm}{}{}\\
Let $M$ be a complex manifold. Then there is an isomorphism $$\Omega_{M,\C}^k\cong\Lambda_{\mC_{M,\C}^\infty}^k\Omega_{M,\C}^1$$
\end{lmm}

\begin{defn}{The Complexified Differential}{}\\
Let $M$ be a complex manifold. 
\begin{itemize}
\item Define the differential $d_\C:\Omega_{M,\C}^\bullet(M)\to\Omega_{M,\C}^{\bullet}(M)$ of $M$ to be the $\mC_{M,\C}^\infty(M)$-module homomorphism $$d_\C=d\otimes\text{id}_\C$$ where $d:\Omega^{\bullet}(M)\to\Omega^{\bullet}(M)$ is the differential of $M$ considered as a smooth manifold. 
\item Define the $\mC_{M,\C}^\infty$-module homomorphism $d_\C:\Omega_{M,\C}^\bullet\to\Omega_{M,\C}^\bullet$ as follows. For each open set $U$, the $\mC_{M,\C}^\infty(U)$-module homomorphism $d_\C(U)$ is the differential of $U$ considered as a complex manifold. 
\end{itemize}
\end{defn}

\begin{defn}{The Complex de Rham Cohomology Groups}{}\\
Let $M$ be a complex manifold. Define the $n$th holomorphic de Rham cohomology groups of $M$ to be $$H_\text{dR}^n(M;\C)=H^n(\Omega_{M,\C}^\bullet(M),d_\C)$$
\end{defn}

\begin{prp}{}{}\\
Let $M$ be a complex manifold. Then there are natural isomorphisms $$H_\text{dR}^n(M;\C)\cong H_\text{dR}^n(M;\R)\otimes_\R\C\cong\H^n(M,\Omega_{M,\C}^\bullet)$$ for all $n\in\N$. 
\end{prp}

\subsection{Mixed Differential Forms}
Let $M$ be a complex manifold. The $\C$-vector space $\Omega_\C^1(M)$ consists of smooth maps from $M$ to $T_\C^\ast M$. The splitting $T_\C^\ast M=TM^+\oplus TM^-$ together with sections on operations of vector bundles means that giving a section of $T_\C^\ast M$ is the same as giving a section of $TM^+$ and $TM^-$. Thus we acquire a splitting $\Omega_\C^1(M)=\Omega^{1,0}(M)\oplus\Omega^{0,1}(M)$. 

Let $M$ be a complex manifold. The splitting $T_\C^\ast M=TM^+\oplus TM^-$. Properties of the exterior product that translates to vector bundles as $$\Lambda^n(T_\C^\ast M)=\bigoplus_{p+q=n}\Lambda^p(TM^+)\otimes\Lambda^q(TM^-)$$

\begin{defn}{The Sheaf of Mixed Differential Forms}{}\\
Let $M$ be a complex manifold. Define the sheaf of $(p,q)$-forms $\Omega_M^{p,q}$ to be the sheaf of smooth sections of $\Lambda^p(T^{1,0})^\ast\otimes\Lambda^q(T^{0,1})^\ast$. Explicitly, we have $$\Omega_M^{p,q}(U)=\Gamma^\infty(U,\Lambda^p(T^{1,0})^\ast\otimes\Lambda^q(T^{0,1})^\ast)$$
\end{defn}

\begin{lmm}{}{}\\
Let $M$ be a complex manifold. Then $\Omega_M^{p,q}$ is a fine sheaf of $\mC_{M,\C}^\infty$-modules of rank $\binom{n}{p}\binom{n}{q}$. 
\end{lmm}

\begin{lmm}{}{}\\
Let $M$ be a complex manifold. Then there is an isomorphism $$\Omega_{M,\C}^n=\bigoplus_{p+q=n}\Omega_M^{p,q}$$
\end{lmm}

\begin{defn}{Dolbeault Operators}{}\\
Let $M$ be a complex manifold. Define the Dolbeault operators on $M$ by $$\partial=\text{proj}_{p+1,q}\circ d:\Omega^{p,q}\to\Omega^{p+1,q}\;\;\;\text{ and }\;\;\;\overline{\partial}=\text{proj}_{p,q+1}\circ d:\Omega^{p,q}\to\Omega^{p,q+1}$$
\end{defn}

\begin{prp}{}{}\\
Let $M$ be a complex manifold. Let $p\in M$. Let $(U,z^1,\dots,z^n)$ be a local chart at $p$. Suppose that $\omega\in\Omega_M^{p,q}$ is given locally by $$\omega=\sum_{\abs{I}=p,\abs{J}=q}f_{I,J}dz^I\wedge d\overline{z^J}$$ where $f_{I,J}:U\to\C$ are smooth functions. Then the following are true. 
\begin{itemize}
\item $\partial\omega$ is given locally on the same chart by $$\partial\omega=\sum_{\abs{I}=p,\abs{J}=q}\sum_k\frac{\partial f_{I,J}}{\partial z^l}dz^l\wedge dz^I\wedge\overline{z^J}$$
\item $\overline{\partial}\omega$ is given locally on the same chart by $$\overline{\partial}\omega=\sum_{\abs{I}=p,\abs{J}=q}\sum_k\frac{\partial f_{I,J}}{\partial\overline{z^l}}d\overline{z^l}\wedge dz^I\wedge\overline{z^J}$$
\end{itemize}
\end{prp}

\begin{prp}{}{}\\
Let $M$ be a complex manifold. Then the following are true. 
\begin{itemize}
\item $d=\partial+\overline{\partial}$. 
\item $\partial\circ\partial=\overline{\partial}\circ\overline{\partial}=0$. 
\item $\partial\circ\overline{\partial}=-\overline{\partial}\circ\partial$. 
\item Let $\omega\in\Omega_{M,\C}^k(M)$ and $\eta\in\Omega_{M,\C}^l(M)$. Then we have $$\partial(\omega\wedge\eta)=\partial(\omega)\wedge\eta+(-1)^k\omega\wedge\partial(\eta)\;\;\;\text{ and }\;\;\;\overline{\partial}(\omega\wedge\eta)=\overline{\partial}(\omega)\wedge\eta+(-1)^k\omega\wedge\overline{\partial}(\eta)$$
\end{itemize}
\end{prp}

\begin{defn}{Dolbeault Complex}{} Let $M$ be a complex manifold. Let $p\in\N$. Define the $p$th Dolbeault complex of $M$ to be the cochain complex \\
\adjustbox{scale=1.0,center}{\begin{tikzcd}

\end{tikzcd}}~\\
denoted $\Omega^{p,\bullet}(M)$. 
\end{defn}

\begin{defn}{Dolbeault Cohomology}{} Define the dobeault cohomology of a complex manifold $M$ to be $$H^{p,q}(M)=\frac{\ker(\bar{\partial}:\Omega^{p,q}(M)\to\Omega^{p,q+1}(M)}{\im(\bar{\partial}:\Omega^{p,q-1}(M)\to\Omega^{p,q}(M))}=H^q(\Omega^{p,\bullet}(M),\overline{\partial})$$
\end{defn}

\begin{lmm}{}{}\\
Let $M$ be a complex manifold. Then there is an isomorphism $$H^{p,q}(M)\cong\H(M,\Omega_M^{p,\bullet})$$
\end{lmm}

\subsection{Holomorphic Differential Forms}
\begin{defn}{The Sheaf of Holomorphic Differential Forms}{}\\
Let $M$ be a complex manifold. Define the sheaf of holomorphic $k$-forms $\Omega_M^k$ of $M$ to be the sheaf of holomorphic sections of $\Lambda^k(T^{1,0})^\ast$. Explicitly, we have $$\Omega_M^k(U)=\Gamma^\text{hol}(U,\Lambda^k(T^{1,0})^\ast)$$
\end{defn}

\begin{lmm}{}{}\\
Let $M$ be a complex manifold. Then $\Omega_M^k$ is a sheaf of $\mO_M$-modules. 
\end{lmm}

Notice that this is a sheaf of $\mO_M$-modules and not $\mC^\infty(M)$. In general $\Omega_M^k$ is strictly smaller than $\Omega_M^{k,0}$. 

\begin{lmm}{}{}\\
Let $M$ be a complex manifold. Then $(\Omega_M^\bullet,\partial)$ is a cochain complex. 
\end{lmm}

\begin{defn}{Holomorphic de Rham Cohomology}{}\\
Let $M$ be a complex manifold. Define the $k$th holomorphic de Rham cohomology group of $M$ to be $$H_\text{hol}^k(M)=\H^k(M,\Omega_M^\bullet)$$
\end{defn}

$\Omega_M^k$ is coherent but not fine. Hence this is genuinely a different object from $H^k(\Omega_M^k(M),\partial)$. 

\begin{prp}{}{}\\
Let $M$ be a complex manifold. Then there is an isomorphism $$H_\text{dR}^k(M;\C)\cong H_\text{hol}^k(M)$$
\end{prp}

\begin{thm}{Dolbeault's Theorem}{}\\
Let $M$ be a complex manifold. Then we have an isomorphism $$H^{p,q}(M)\cong H^q(M,\Omega_M^p)$$
\end{thm}

\begin{crl}{}{}\\
Let $M$ be a compact complex manifold. Then $H^{0,0}(M)=\C$. 
\end{crl}

\begin{lmm}{}{}\\
Let $M$ be a complex manifold. Then $H^{0,q}(M)=0$ for all $q>\dim(M)$. 
\end{lmm}

\subsection{Orientability}
\begin{prp}{}{}\\
Let $M$ be a complex manifold. Then $M$ is orientable. 
\end{prp}

\begin{crl}{}{}\\
Let $M,N$ be complex manifolds. Let $f:M\to N$ be a holomorphic map. Then $f$ preserves orientation. 
\end{crl}

\section{Hermitian Manifolds}
\subsection{The Hermitian Metric}
\begin{defn}{Hermitian Manifolds}{}\\
Let $M$ be a complex manifold. We say that $M$ is Hermitian if $T^{1,0}(M)$ admits a Hermitian metric. 
\end{defn}

\begin{prp}{}{}\\
Let $(M,h)$ be a Hermitian manifold. Let $p\in M$. Let $(U,z^1,\dots,z^n)$ be a complex chart at $p$. Then $h_p$ is given locally on the chart by $$h_p=\sum_{i,j=1}^nh_{i,j}|_p\;(dz^i)_p\otimes(d\overline{z^j})_p$$ where $h_{i,j}|_p=h_p\left(\frac{\partial}{\partial z^i}\bigg{|}_p,\frac{\partial}{\partial z^j}\bigg{|}_p\right)$. 
\end{prp}

\begin{prp}{}{}\\
Let $M$ be a smooth manifold. Let $p:E\to M$ be a complex vector bundle. Then $E$ admits a Hermitian metric. 
\end{prp}

\subsection{The Associated Form}
\begin{defn}{The Associated Form}{}\\
Let $(M,h)$ be a Hermitian manifold. Define the associated form of $h$ to be $$\omega=\frac{i}{2}(h-\overline{h})$$
\end{defn}

In local coordinates, $\omega$ is expressed as $$\omega=\frac{i}{2}\sum_{\alpha,\beta=1}^n h_{\alpha\overline{\beta}}dz_{\alpha}\wedge d\overline{z}_\beta$$ if $M$ is a complex manifold of complex dimension $n$. 

\begin{defn}{The Associated Riemannian Metric}{}\\
Let $(M,h)$ be a Hermitian manifold. Define the associated Riemannian metric of $h$ to be $$g=\frac{1}{2}(h+\overline{h})=$$
\end{defn}

In local coordinates, $g$ is expressed as $$g(u,v)=\frac{1}{2}\sum h_{\alpha\overline{\beta}}(dz_\alpha\otimes d\overline{z}_\beta+d\overline{z}_\beta\otimes dz_\alpha)$$

\begin{lmm}{}{}\\
Let $(M,h)$ be a Hermitian manifold. Then we have $\omega$ is symplectic structure on $TM$. 
\end{lmm}

\begin{lmm}{}{}\\
Let $(M,h)$ be a Hermitian manifold. Then we have $$h=g-i\omega$$
\end{lmm}

\begin{prp}{}{}\\
Let $(M,h)$ be a Hermitian manifold whose associated almost complex structure is given by $J$. Then the following are true. 
\begin{itemize}
\item $h(Ju,Jv)=h(u,v)$
\item $g(Ju,Jv)=g(u,v)$
\item $\omega(Ju,Jv)=\omega(u,v)$
\end{itemize}
\end{prp}

\begin{prp}{}{}\\
Let $(M,h)$ be a Hermitian manifold whose associated almost complex structure is given by $J$. Then the following are true. 
\begin{itemize}
\item $\omega(u,v)=g(Ju,v)$. 
\item $g(u,v)=\omega(u,Jv)$. 
\end{itemize}
\end{prp}

Notice that the three equations $$h=g-i\omega\;\;\;\text{ and }\;\;\;\omega(u,v)=g(Ju,v)\;\;\;\text{ and }\;\;\;g(u,v)=\omega(u,Jv)$$ completely determines the triple $(g,h,\omega)$. 

\begin{prp}{}{}\\
Let $M$ be a complex manifold whose associated almost complex structure is given by $J$. Let $g$ be a Riemannian metric on $M$. Then the following are true. 
\begin{itemize}
\item $h=g-i\omega$ is a Hermitian form on $M$ where $\omega(u,v)=g(Ju,v)$. 
\item $\omega$ is the associated form of $h$. 
\end{itemize}
\end{prp}

\begin{prp}{}{}\\
Let $M$ be a complex manifold whose associated almost complex structure is given by $J$. Let $\omega$ be a $(1,1)$-form on $M$ that is non-degenerate, and preserves $J$ and is positive definite in the sense that $\omega(u,Jv)>0$ for all non-zero $u,v$. Then the following are true. 
\begin{itemize}
\item $g(u,v)=\omega(u,Jv)$ is a Riemannian metric on $M$. 
\item $h=g-i\omega$ is a Hermitian metric on $M$. 
\end{itemize}
\end{prp}

\begin{prp}{}{}\\
Let $(M,h)$ be a complex Hermitian manifold. Let $\omega$ be its associated $2$-form. Then the volume form of $M$ is given by $$dV=\frac{1}{n!}\omega^n$$
\end{prp}

\section{Complex Hodge Theory}
Let $(M,h)$ be a (almost) complex manifold. Then $M$ is necessarily an orientable Riemannian manifold as seen above. Therefore Hodge theory applies to $M$. In particular, if $d$ is the exterior derivative of $M$, then there is an isomorphism $$\mH^k(M)\cong H_\text{dR}^k(M)$$ where the left hand side is the space of Harmonic $k$-forms on $M$. \\

However, since $M$ is (almost) complex, the exterior derivative splits into its holomorphic $\partial$ and its antiholomorphic $\overline{\partial}$ counterpart, and so does the cohomology groups. In this section we will show that Hodge theory for $\overline{\partial}$ also holds true. 

\subsection{Complex Properties of the Hodge Star Operator}
\begin{prp}{}{}\\
Let $(M,h)$ be a complex Hermitian manifold. Let $n=\dim(M)$. Then the following are true. 
\begin{itemize}
\item $\ast\omega\in\Omega^{(n-p,n-q)}$ for any $\omega\in\Omega^{p,q}(M)$. 
\item For any $\phi,\psi\in\Omega^{p,q}(M)$, we have $$\langle\phi,\psi\rangle_{L^2}dV=\phi\wedge\ast\psi$$
\item For any $\psi\in\Omega^{(p,q)}(M)$, we have $\ast\ast\phi=(-1)^{p+q}\psi$. 
\end{itemize}
\end{prp}

\subsection{Adjoints of Differential Operators}
Recall that for a Riemannian manifold, the exterior derivative $d:\Omega^n(M)\to\Omega^{n+1}(M)$ admits an adjoint that is characterized by the formula $d^\ast=-\ast d\ast$. In the theory of complex manifolds, the splitting $d=\partial+\overline{\partial}$ means that $\partial$ and $\overline{\partial}$ also admits adjoints. 

\begin{defn}{The Adjoint of $\partial$}{}\\
Let $(M,h)$ be a complex Hermitian manifold. Define the operator $\partial^\ast:\Omega^{p,q}(M)\to\Omega^{p-1,q}(M)$ to be the operator such that $$\langle\partial^\ast\omega,\eta\rangle_{L^2}=\langle\omega,\partial\eta\rangle_{L^2}$$ for all $\omega\in\Omega^{p,q}(M)$ and $\eta\in\Omega^{p-1,q}(M)$. 
\end{defn}

\begin{prp}{}{}\\
Let $(M,h)$ be a complex Hermitian manifold. Then the following are true. 
\begin{itemize}
\item $\partial^\ast=-\ast\partial\ast$. 
\item $\partial^\ast\circ\partial^\ast=0$. 
\end{itemize}
\end{prp}

\begin{defn}{The Adjoint of $\overline{\partial}$}{}\\
Let $(M,h)$ be a complex Hermitian manifold. Define the operator $\overline{\partial}^\ast:\Omega^{p,q}(M)\to\Omega^{p,q-1}(M)$ to be the operator such that $$\langle\overline{\partial}^\ast\omega,\eta\rangle_{L^2}=\langle\omega,\overline{\partial}\eta\rangle_{L^2}$$ for all $\omega\in\Omega^{p,q}(M)$ and $\eta\in\Omega^{p,q-1}(M)$. 
\end{defn}

\begin{prp}{}{}\\
Let $(M,h)$ be a complex Hermitian manifold. Then the following are true. 
\begin{itemize}
\item $\overline{\partial}^\ast=-\ast\overline{\partial}\ast$. 
\item $\overline{\partial}^\ast\circ\overline{\partial}^\ast=0$. 
\end{itemize}
\end{prp}

\begin{defn}{The Complex Laplacian Operators}{}\\
Let $(M,h)$ be a complex Hermitian manifold. Define the complex Laplacian operators $\Delta_{\partial},\Delta_{\overline{\partial}}:\Omega^{p,q}(M)\to\Omega^{p,q}(M)$ by $$\Delta_\partial=\partial\circ\partial^\ast+\partial^\ast\circ\partial\;\;\;\text{ and }\;\;\;\Delta_{\overline{\partial}}=\overline{\partial}\circ\overline{\partial}^\ast+\overline{\partial}^\ast\circ\overline{\partial}$$
\end{defn}

\begin{defn}{Complex Harmonic Forms}{}\\
Let $M$ be a complex Hermitian manifold. Let $\omega\in\Omega^{p,q}(M)$ be a differential form. We say that $\omega$ is harmonic if $\omega\in\ker(\Delta_{\overline{\partial}})$. 
\end{defn}

\begin{defn}{Space of Harmonic Forms}{}\\
Let $M$ be a complex Hermitian manifold. Define the space of harmonic $(p,q)$-forms to be $$\mH^{p,q}(M)=\ker(\Delta_{\overline{\partial}})$$
\end{defn}

\begin{thm}{Complex Hodge Theorem}{}\\
Let $M$ be a complact complex Hermitian manifold. Then the following are true. 
\begin{itemize}
\item $\mH^{p,q}(M)$ is finite dimensional for all $p,q\in\N$. 
\item There is a decomposition $$\Omega^{p,q}(M)\cong\mH^{p,q}(M)\oplus\overline{\partial}(\Omega^{p,q-1}(M))\oplus\overline{\partial}^\ast(\Omega^{p,q+1}(M))$$
\end{itemize}
\end{thm}

\begin{crl}{}{}\\
Let $M$ be a compact complex Hermitian manifold. Then the quotient map induces an isomorphism $$\mH^{p,q}(M)\cong H^{p,q}(M)$$
\end{crl}

\begin{prp}{}{}\\
Let $M$ be a compact complex Hermitian manifold. Let $n=\dim(M)$. Then the following are true. 
\begin{itemize}
\item $H^{n,n}(M)=\C$. 
\item There is a perfect pairing $H^{p,q}(M)\times H^{n-p,n-q}(M)\to\C$ given by $$(\alpha,\beta)\mapsto\int_M\alpha\wedge\beta$$
\end{itemize}
\end{prp}
