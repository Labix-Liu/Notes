\section{Introduction to Manifolds}
\subsection{Pseudogroup of Transformations}
\begin{defn}{The Set of Homeomorphisms of Open Subsets}{}\\
Let $(X,\mT)$ be a topological space. Define the set of homeomorphisms of $X$ between open subsets to be $$H(X)=\{f:U\to V\;|\;U,V\in\mT, f\text{ is a homeomorphism }\}$$
\end{defn}

We establish the definition of manifolds in a slightly non-standard way via the use of pesudogroups. 

\begin{defn}{Pseudogroup of Transformations}{}\\
Let $(X,\mT)$ be a topological space. A pseudogroup of transformations on $X$ is a subset of homeomorphisms $$\Gamma\subseteq H(X)$$ such that the following are true. 
\begin{itemize}
\item (Cover) There exists and indexing set $I$ and $f_i:U_i\to V_i\in\Gamma$ for $i\in I$ such that $X=\bigcup_{i\in I}U_i$. 
\item (Restriction) If $f:U\to V\in\Gamma$ and $W\subseteq U$, then $f|_W\in\Gamma$. 
\item (Local) If $f:U\to V\in H(X)$ and there exists an open cover $U=\bigcup_{i\in I}U_i$ such that $f|_{U_i}\in\Gamma$, then $f\in\Gamma$. 
\item (Inverse) If $f\in\Gamma$ then $f^{-1}\in\Gamma$. 
\item (Composition) If $f_1:U_1\to V_1\in\Gamma$ and $f':V_2\to W_2\in\Gamma$ such that $V_1\cap V_2\neq\emptyset$, then $$f_2\circ f_1:f_1^{-1}(V_1\cap V_2)\to f_2(V_1\cap V_2)$$ lies in $\Gamma$
\end{itemize}
\end{defn}

Manifolds is more of less some sort of a surface living in $\R^n$. To translate notions of differntiability to a manifold we want to cover the manifold with open sets isomorphic to $\R^m$. The pseudogroup of transformation is essentially determining what structure we want to investigate. \\
Denote $\Gamma_{C^r}$ the pseudogroup where all transformations are $C^r$. Different values of $r$ encode different information on the manifold since the underlying transformations between the open sets that cover the manifold change. 

\begin{defn}{$C^r$-Differentiable Maps}{} Let $U,V\subseteq\R^n$ be open. Denote $C^r(U,V)$ the set of all $r$ times differentiable maps from $U$ to $V$ such that the $r$th derivative is continuous. \\
By convention, $C^0(U,V)$ is the set of all continuous maps from $U$ to $V$ and $C^k(\R^n)=C^k(\R^n,\R^n)$. $C^\infty(U,V)$ denotes the set of all infinitely differentiable (smooth) maps from $U$ to $V$. 
\end{defn}

\begin{eg}{}{}\\
The following are pseudogroups of transformations. 
\begin{itemize}
\item $\Gamma(\R^n,\mC^0)=H(\R^n)$. 
\item $\Gamma(\R^n,\mC^k)=\{f:U\to V\in H(\R^n)\;|\;f\in\mC^k(U,V)\text{ and }f^{-1}\in\mC^k(V,U)\}$ for $1\leq k\leq\infty$. 
\item $\Gamma(\R^n,\mC^\infty)=\{f:U\to V\in H(\R^n)\;|\;f\in\mC^\infty(U,V)\text{ and }f^{-1}\in\mC^\infty(V,U)\}$. 
\item $\Gamma(\R^n,\mA)=\{f:U\to V\in H(\R^n)\;|\;f,f^{-1}\text{ are analytic }\}$. 
\item $\Gamma(H^n,\mC^\infty)=\{f:U\to V\in H(H^n)\;|\;f\in\mC^\infty(U,V)\text{ and }f^{-1}\in\mC^\infty(V,U)\}$. 
\end{itemize}
\end{eg}

\subsection{Manifolds and their Structures}
\begin{defn}{Atlas}{}\\
Let $M$ be a topological space. An atlas of $M$ compatible with a pseudogroup of transformations $\Gamma$ is a family of pairs $\{(U_i,\phi_i)\;|\;i\in I\}$, called charts, such that the following are true. 
\begin{itemize}
\item $\{U_i\;|\;i\in I\}$ is an open cover of $M$. 
\item Each $\phi_i$ is a homeomorphism of $U_i$ onto an open set of $S$
\item Whenever $U_i\cap U_j$ is nonempty, the mapping $\phi_j\circ\phi_i^{-1}$ of $\phi_i(U_i\cap U_j)$ onto $\phi_j(U_i\cap U_j)$ is an element of $\Gamma$
\end{itemize}
\end{defn}

This definition complements well with our explanation. For an atlas to make sense with a pseudogroup, the target (range) should be the underlying $\R^m$. The $U_i$ should at least cover $M$ and be part of $M$. And finally areas with overlapping should have maps that allow transition between two covers, which is what the third condition is saying. 

\begin{defn}{Compatible Charts}{} Let $(U,\phi)$ be a chart of a space $M$. We say that it is compatible with another chart $(V,\varphi)$ if $\varphi\circ\phi^{-1}:\phi(U\cap V)\to\varphi(U\cap V)$ is in $\Gamma$. \\
We say that $(U,\phi)$ is compatible with an atlas $\{(U_i,\phi_i)\;|\;i\in I\}$ of $M$ if $(U,\phi)$ is compatible with every chart in the atlas. 
\end{defn}

To prove that two atlases $A$ and $A'$ are not equal, one should find a chart in $A$ that is incompatible with a chart in $A'$, or vice versa. 

\begin{defn}{Complete Atlas}{} A complete atlas on a topological space $M$ compatible with $\Gamma$ is a atlas of $M$ compatible with $\Gamma$ which is not contained in any other atlas $M$ compatible with $\Gamma$. 
\end{defn}

Complete atlases are also called maximal atlases. 

\begin{lmm}{}{} Every atlas of $M$ compatible with $\Gamma$ is contained in a unique complete atlas. 
\begin{proof}
Let $A=\{(U_i,\phi_i)\}$ be an atlas of $M$ compatible with $\Gamma$. Let $\tilde{A}$ be the family of all pairs $(U_i,\phi_i)$ such that $\phi:U_i\subseteq M\to S$ is a homeomorphism such that $\phi_j\circ\phi_i^{-1}:\phi_i(U_i\cap U_j)\to\phi_j(U_i\cap U_j)$ is an element of $\Gamma$ whenever $U_i\cap U_j$ is nonempty. Then $\tilde{A}$ is the complete atlas of $M$ containing $A$. 
\end{proof}
\end{lmm}

However note that this is not to say that $M$ has a unique $\Gamma$-structure. \\

By collecting atlas and using the complete atlas of a topological space, we can talk about all possible transitions between the open covers. 

\begin{defn}{Manifolds}{}\\
Let $X$ be a space. Let $\Gamma$ be a pseudogroup of transformations over $X$. A $\Gamma$-manifold is a pair $(M,\mA)$ where
\begin{itemize}
\item $M$ is a second countable Hausdorff space. 
\item $\mA$ is a complete atlas compatible with $\Gamma$. 
\end{itemize}
\end{defn}

A chart on a real manifold $M$ of dimension $n$ is actually an open set together with $n$ coordinate functions. Indeed if $(U,\phi)$ is a chart of $M$, then $\phi:U\to\R^n$ is a map to $\R^n$. If we are interested in the particular components of $\phi$, we write $\phi=(x^1,\dots,x^n)$ where $x^i:U\to\R$ is the coordinate function sending a point on $U$ to its $i$th coordinate. All the $x^i$s combine to recover $\phi$. If we would like to talk about the actual coordinate on $\R^n$ instead of the coordinate function, we write $x_i$ instead. 

\begin{defn}{Transition Functions}{}\\
Let $M$ be a manifold. Let $\{(U_i,\phi_i)\;|\;i\in I\}$ be an atlas on $M$. The transition functions on $M$ are the functions $$\phi_j\circ\phi_i^{-1}:\phi_i(U_i\cap U_j)\to\phi_j(U_i\cap U_j)$$ for $U_i\cap U_j\neq\emptyset$. 
\end{defn}

We now discuss some topological properties of manifolds whose charts are locally $\R^n$, $\C^n$ or $H^n$. 

\subsection{Maps between Manifolds}
\begin{defn}{Morphism of Manifolds}{}\\
Let $\Gamma$ be a pseudogroup of transformations. Let $M,N$ be $\Gamma$-manifolds whose atlases are given by $\{(U_i,\phi_i)\;|\;i\in I\}$ and $\{(V_j,\psi_j)\;|\;j\in J\}$ respectively. Let $f:M\to N$ be a function. We say that $f$ is a morphism if $$\psi_j\circ f\circ\phi_i^{-1}\in\Gamma$$ whenever $f(U_i)\cap V_j\neq\emptyset$. 
\end{defn}

\begin{lmm}{}{}\\
Let $M,N,K$ be manifolds. Let $f:M\to N$ and $g:N\to K$ be morphisms. Then $g\circ f$ is a morphism. 
\end{lmm}

\begin{lmm}{}{}\\
Let $M$ be a manifold. Then $\text{id}_M:M\to M$ is a morphism. 
\end{lmm}

\begin{defn}{Isomorphisms}{}\\
Let $M,N$ be manifolds. Let $f:M\to N$ be a morphism. We say that $f$ is an isomorphism if there exists a morphism $g:N\to M$ such that $$g\circ f=\text{id}_M\;\;\;\text{ and }\;\;\;f\circ g=\text{id}_N$$
\end{defn}

\begin{defn}{The Group of Automorphisms}{}\\
Let $M$ be a manifold. Define the group of automorphisms of $M$ to be the set $$\text{Aut}(M)=\{f:M\to M\;|\;f\text{ is an isomorphism }\}$$
\end{defn}

\section{New Manifolds from Old}
\subsection{Submanifolds}
\begin{defn}{Embedded Submanifolds}{}\\
Let $M$ be a manifold. An embedded submanifold of $M$ is a subset $N\subseteq M$ such that for all $p\in N$, there exists a chart $(U,\phi=(x^1,\dots,x^n))$ of $M$ at $p$ such that $x_{i_1}|_{U\cap N}=\cdots=x_{i_{n-k}}|_{U\cap N}=0$ vanishes as functions for some fixed $k\in\N$. 
\end{defn}

Intuitively, the condition that some coordinates function of charts on $N$ vanish means that $N$ really have charts in $\R^{n-m}$ instead of $\R^n$. This condition is more commonly written as $$\phi(U\cap N)=\phi(U)\cap\R^k\times\{0\}^{n-k}$$

\begin{prp}{}{}\\
Let $M$ be a manifold. Let $N\subseteq M$ be an embedded submanifold. Then the following are true. 
\begin{itemize}
\item $N$ is a manifold in its own right whose atlas is given by $$\{(U_i\cap N,\phi|_{U_i\cap N})\;|\;i\in I\}$$ where $\{(U_i,\phi_i)\;|\;i\in I\}$ is the atlas of $M$. 
\item If $n-k$ coordinate function of $M$ vanishes at $N$, then $\dim(N)=k$. 
\item The inclusion map $N\hookrightarrow M$ is a morphism of manifolds. 
\end{itemize}
\end{prp}

It is not possible to characterize embedded submanifolds using simply inclusions $N\hookrightarrow M$. 

\begin{eg}{Alexander Horned Ball}{}\\
The Alexander horned ball is a topological $3$-manifold equipped with an inclusion map of sets to $S^3$, but is not an embedded submanifold of $S^3$. 
\end{eg}

\subsection{Product Manifolds}
\begin{prp}{}{}\\
Let $M,N$ be manifolds. Then the following are true. 
\begin{itemize}
\item $M\times N$ is a manifold of dimension. 
\item If $(U,\phi)$ is a chart of $M$ and $(V,\psi)$ is a chart of $N$, then $(U\times V,\phi\times\psi)$ is a chart of $M\times N$. 
\end{itemize}
\end{prp}

\subsection{Quotient Manifolds}
\begin{prp}{}{}\\
Let $G$ be a discrete group. Let $M$ be a $\Gamma$-manifold that is a $G$-space. Suppose that the following are true. 
\begin{itemize}
\item The action of $G$ on $M$ is free and proper. 
\item $g\cdot -:M\to M\in\text{Aut}(M)$ for all $g\in G$. 
\end{itemize}
Then the following are true. 
\begin{itemize}
\item The orbit space $M/G$ is a $\Gamma$-manifold. 
\item $\dim(M/G)=\dim(M)$. 
\end{itemize}
\end{prp}

\section{Real Manifolds}
\subsection{Topological Properties of Real Manifolds}
\begin{defn}{Types of Real Manifolds}{}\\
Let $\Gamma$ be a pseudogroup of transformations over $\R^n$. Let $M$ be a $\Gamma$-manifold. 
\begin{itemize}
\item We say that $M$ is a topological manifold if $\Gamma=\Gamma(\R^n,\mC^0)$. 
\item Let $1\leq k<\infty$. We say that $M$ is a $C^k$-differentiable manifold if $\Gamma=\Gamma(\R^n,\mC^k)$. 
\item We say that $M$ is a smooth manifold if $\Gamma=\Gamma(\R^n,\mC^\infty)$. 
\item We say that $M$ is a real analytic manifold if $\Gamma=\Gamma(\R^n,\mA)$. 
\end{itemize}
\end{defn}

\begin{defn}{Dimension}{}\\
Let $\Gamma$ be a pseudogroup of transformations over $\R^n$. Let $M$ be a $\Gamma$-manifold. Define the dimension of $M$ to be $\dim(M)=n$. 
\end{defn}

\begin{lmm}{}{}\\
Let $M$ be a manifold. Then $M$ is locally compact. 
\end{lmm}

\begin{defn}{The Space of Differentiable Functions}{}\\
Let $k\in\N\cup\{\infty\}$. Let $M$ be a $\mC^k$-differentiable manifold. Define the space of $\mC^k$-differentiable functions of $M$ to be the set $$\mC^k(M)=\{f:M\to\R\;|\;f\text{ is a }\mC^k\text{-differentiable function }\}$$ together with pointwise addition and pointwise multiplication. 
\end{defn}

\subsection{Differentiable Structure on Real Manifolds}
\begin{lmm}{}{} The following are true regarding the different types of manifolds. 
\begin{itemize}
\item Every smooth manifold is a differentiable manifold of class $C^r$ for any $r\in\N$
\item Every differentiable manifold of class $C^r$ is a differentiable manifold of class $C^{r-1}$ for $r\geq 1$
\item Every differentiable manifold of class $C^r$ is a topological manifold
\end{itemize}
\end{lmm}

\begin{prp}{}{}\\
Let $k\geq 1$. Let $M$ be a $C^k$ manifold with atlas $\mA$. Then $\mA$ is compatible with a unique $C^\infty$ atlas. 
\end{prp}

\begin{prp}{}{}\\
Let $k\geq 1$. Let $M,N$ be smooth manifolds. Then $M$ and $N$ are diffeomorphic if and only if $M$ and $N$ are $C^k$-diffeomorphic. 
\end{prp}

\subsection{Analytic Structure on Real Manifolds}
\begin{thm}{Grauert-Whitney}{}\\
Every smooth manifold admits a compatible real analytic structure. 
\end{thm}

\subsection{The Low Dimension Case}
\begin{thm}{Kervaire-Milnor}{}\\
Let $M$ be a topological manifold of dimension $n\leq 3$. Then $M$ admits a unique smooth structure. 
\end{thm}

\begin{prp}{}{}\\
Let $M,N$ be topological manifolds of dimension both $2$ or $3$. Then $M$ and $N$ are $\mC^0$-diffeomorphic if and only if $M$ and $N$ are diffeomorphic. 
\end{prp}

\subsection{Manifolds with Boundaries}
\begin{defn}{Manifolds with Boundaries}{}\\
Let $k\in\N\cup\{\infty\}$. A $\mC^k$-manifold with boundary is a $\Gamma(H^n,\mC^k)$-manifold. 
\end{defn}

\begin{lmm}{}{}\\
Let $k\in\N\cup\{\infty\}$. Let $M$ be a $\mC^k$-manifold. Then $M$ is a $\mC^k$-manifold with boundary. 
\end{lmm}

\begin{defn}{Dimension}{}\\
Let $M$ be a $\Gamma(H^n,\mC^k)$-manifold. Define the dimension of $M$ to be $\dim(M)=n$. 
\end{defn}

\begin{defn}{Boundary Charts}{}\\
Let $M$ be a $\mC^k$-manifold with boundary. Let $p\in\partial M$. Let $(U,\phi)$ be a chart at $p$. We say that $(U,\phi)$ is a boundary chart at $p$ if the following are true. 
\begin{itemize}
\item $\phi(U)\subseteq\{(x_1,\dots,x_n)\in\R^n\;|\;x_n\geq 0\}$. 
\item $\phi^{-1}(\{(x_1,\dots,x_n)\in\R^n\;|\;x_n=0\})\subseteq\partial M$. 
\item $\phi(p)=0$. 
\end{itemize}
\end{defn}

\begin{defn}{The Boundary of a Manifold}{}\\
Let $M$ be a manifold with boundary. Define the boundary of $M$ to be $$\partial M=\{p\in M\;|\;p\text{ admits a boundary chart }\}$$
\end{defn}

\begin{lmm}{}{}\\
Let $M$ be a manifold with boundary. Then the following are true. 
\begin{itemize}
\item $\partial M\cap M^\circ=\emptyset$. 
\item $\partial M$ is a closed subset of $M$. 
\end{itemize}
\end{lmm}

\begin{prp}{}{}\\
Let $M$ be a $\mC^k$-manifold with boundary. Then $\partial M$ is a $\mC^k$-manifold of dimension $\dim(M)-1$ with atlas $$\{(U_\alpha|_{\partial M},\phi_\alpha|_{\partial M})\}$$ where $\{(U_\alpha,\phi_\alpha)\}$ the atlas on $M$. 
\end{prp}
