\section{Simplicies}
\subsection{Basic Definitions}
\begin{defn}{Affinely Independent}{}{}\\
Let $n,m\in\N$. We say that a set of points $\{v_0,\dots,v_m\}\subset\R^n$ is affinely independent if $v_1-v_0,\dots,v_n-v_0$ are linearly independent. 
\end{defn}

\begin{defn}{$n$-Simplexes}{}\\
Let $n\in\N$. Let $v_0,\dots,v_n\in\R^n$ be a set of affinely independent points. An $n$-simplex is the set of points $$\Delta^n=\left\{\sum_{k=0}^nt_kv_k\bigg{|}\sum_{k=0}^nt_k=1\text{ and }t_k\geq 0\text{ for all }k=0,\dots,n\right\}$$ We write $\Delta^n=[v_0,\dots,v_n]$ to indicate the spanning vectors. The standard $n$-simplex is just the $n$-simplex whose vertices are the standard basis vectors for $\R^{n+1}$. 
\end{defn}

Note that the vertices here are an ordered set so that an orientation is inherently defined. Realistically, the order of the vertices does not change the homology groups which we will define later. 

\begin{defn}{Properties of $n$-Simplexes}{} Let $\Delta^n=[v_0,\dots,v_n]$ be an $n$-simplex. 
\begin{itemize}
\item The $k$th face of $\Delta^n$ is a the $n-1$ simplex $\Delta_k^n=\Delta^n\cap\{x_k=0\}$. We use $[v_0,\dots,\hat{v}_k,\cdots,v_n]$ to indicate the $k$th $n-1$dimensional face
\item The boundary of $\Delta$, written $\partial\Delta^n$ is the union of all its proper faces
\item The interior is defined to be $(\Delta^n)^\circ=\Delta^n\setminus\partial\Delta^n$
\end{itemize}
\end{defn}

It is easy to see that any face of an $n$-simplex is an $n-1$ simplex in its own right. 

\begin{lmm}{}{} Any two $k$-simplexes where one in $\R^m$ and one in $\R^n$ are homeomorphic. 
\end{lmm}

\subsection{Barycentric Subdivision}
Barycentric subdivisions is the main ingredient for proving one of the major results of singular homology. It is defined first through $\Delta$-set, and then passed on to singular $n$-simplexes, which we will see in the next chapter. 

\begin{defn}{Barycenter}{} Let $v=[v_0,\dots,v_n]$ be an $n$-simplex. Define the barycenter of $v$ to be the point $$B(v)=\frac{1}{n+1}\sum_{i=0}^nv_i$$
\end{defn}

In fact, we can find the barycenter inductively. Knowing the barycenter $b_i$ on the $i$th face $[v_0,\dots,\hat{v}_i,\dots,v_n]$, let $l_i$ be the line connecting $b_i$ and $v_i$. Then $b$ is the intersection of all these lines $l_i$. 

\begin{defn}{Barycentric Cone}{} Let $v$ be an $n$-simplex. Let $w=[w_1,\dots,w_n]\subseteq v$ be an $(n-1)$-simplex in $v$. Define the barycentric cone of $w$ to be the $n$-simplex $$\mB(w)=[B(v),w_1,\dots,w_n]\subset v$$ 
\end{defn}

Intuitively, this means that given a face of $\Delta^n$ which is an $(n-1)$-simplex, its barycentric cone is the $n$-simplex constructed from the vertices of the face together with the barycenter. 

\begin{defn}{The Barycentric Map}{} Let $v$ be an $n$-simplex. Define the barycentric map of $v$ to be the map $$\mB:C_{n-1}(v)\to C_n(v)$$ defined on the basis by $\mB([w_1,\dots,w_n])=[B(v),w_1,\dots,w_n]$. 
\end{defn}

\begin{defn}{Barycentric Subdivision}{} Let $\Delta^n$ be the standard $n$-simplex. Define inductively the barycentric subdivision $$S^\Delta(\Delta^n)\in C_n(\Delta^n)$$ of $\Delta^n$ by the following recursive formula. 
\begin{itemize}
\item When $n=0$, $S^\Delta(\Delta^0)=\Delta^0$
\item When $n>0$, the barycentric subdivision of $\Delta^n$ is given by $$S^\Delta(\Delta^n)=\sum_{i=0}^n(-1)^i\mB S^\Delta(\partial_i\Delta^n)$$
\end{itemize}
\end{defn}

To elicit a few examples, consider the case of $n=1$. Then we have 
\begin{align*}
S(\Delta^1)&=\mB S(\partial\Delta^1)\\
&=\mB S(\partial_0\Delta^1)-\mB S(\partial_1\Delta^1)\\
&=\mB[v_1]-\mB[v_0]\\
&=[b,v_1]-[b,v_0]
\end{align*}
where $b=\frac{1}{2}(v_0+v_1)$. Intuitively, we are breaking up the $n$-simplex $\Delta^n$ into tiny pieces of $n$-simplexes using the center of mass of each subsimplex in $\Delta^n$. 

\begin{lmm}{}{} Let $\sigma=[w_1,\dots,w_n]\subseteq\Delta^n$ be an $(n-1)$-simplex. Then the following are true. 
\begin{itemize}
\item $\partial(\mB(\sigma))+\mB(\partial\sigma)=\sigma$ 
\item $\partial S(\Delta^n)=S(\partial\Delta^n)$
\end{itemize} 
\begin{proof}
We have that 
\begin{align*}
\partial\mB[w_1,\dots,w_n]&=\partial[b,w_1,\dots,w_n]\\
&=\sum_{i=0}^n(-1)^i\partial_i[b,w_1,\dots,w_n]\\
&=[w_1,\dots,w_n]-\mB(\partial[w_1,\dots,w_n])
\end{align*}
And thus the first identity is satisfied. \\
We prove the second item inductively. When $n=0$, we have $0$ on both sides. When $n>0$, we have
\begin{align*}
\partial S\Delta^n&=\partial\mB(S(\partial\Delta^n))\\
&=\text{id}(S\partial\Delta^n)-\mB(\partial(S\partial\Delta^n))\tag{First Identity}\\
&=S\partial\Delta^n-\mB(S\partial^2\Delta^n)\tag{Induction}\\
&=S\partial\Delta^n
\end{align*}
\end{proof}
\end{lmm}

\begin{lmm}{}{} Let $v=[v_0,\dots,v_n]$ be a simplex. Let $[w_0,\dots,w_n]$ be a component in the barycentric subdivision of $v$. Then we have $$\text{diam}([w_0,\dots,w_n])\leq\frac{n}{n+1}\text{diam}([v_0,\dots,v_n])$$ 
\begin{proof}
If $n=0$, then the claim is true since $[w_0]=[v_0]$ has diameter $0$. Assume that $n>0$. We first prove the following subclaim: For every $x\in[x_0,\dots,x_n]$ an $n$-simplex, its maximum distance to points in the simplex is attained at a vertex $x_i$. Indeed if $y\in[x_0,\dots,x_n]$ with $\|x-y\|$ maximal, then we can write $y=\sum_{i=1}^nt_ix_i$ with $\sum_{i=1}^nt_i=1$ and $t_i\geq 0$. Then we have that 
\begin{align*}
\|x-y\|&=\left\|x-\sum_{i=1}^nt_ix_i\right\|\\
&=\left\|\sum_{i=1}^nt_i(x-x_i)\right\|\\
&\leq\sum_{i=1}^nt_i\|x-x_i\|\\
&\leq\max\|x-x-i\|
\end{align*}
with equality if $y$ is one of the vertices with $\|x-x_i\|$ being maximal. By applying the above twice, we have that the diameter of $[w_0,\dots,w_n]$ is the length of the longest edge $[x,y]$ in $[w_0,\dots,w_n]$. Now there are two cases. \\

Case 1: None of $x,y$ is the barycenter $b$ of $[w_0,\dots,w_n]$. Then they must be the vertices of a simplex in the barycentric subdivision of one of the faces $[v_0,\dots,\hat{v_i},\dots,v_n]$. By induction, we have that 
\begin{align*}
\text{diam}([w_0,\dots,w_n])&=\|x-y\|\\
&\leq\frac{n-1}{n}\text{diam}([v_0,\dots,\hat{v_i},\dots,v_n])\\
&\leq\frac{n}{n+1}\text{diam}([v_0,\dots,v_n])
\end{align*}
and so we are done. \\

Case 2: Without loss of generality, $x=b$. \\
Then $y$ lies on some face of $[v_0,\dots,v_n]$ and the claim above implies that we can take $y=v_i$ for some vertex $v_i$ of that face. Let $b_i$ be the barycenter of $[v_0,\dots,\hat{v_i},\dots,v_n]$. Then we have that the barycenter of $[v_0,\dots,v_n]$ is $$b=\frac{1}{n+1}\sum_{j=1}^nv_j=\frac{1}{n+1}v_i+\frac{n}{n+1}b_i$$ We thus have that 
\begin{align*}
\text{diam}([w_0,\dots,w_n])&=\|v_i-b\|\\
&\leq\frac{n}{n+1}\|v_i-b_i\|\\
&\leq\frac{n}{n+1}\text{diam}([v_0,\dots,v_n])
\end{align*}
and so we conclude. 
\end{proof}
\end{lmm}

\section{Simplicial and Delta Complexes}
\subsection{Delta Complexes}
\begin{defn}{$\Delta$-Complexes}{}\\
A $\Delta$-complex $(S_\bullet,d_\bullet)$ consists of the following data. 
\begin{itemize}
\item For each $n\in\N$, a set $S_n$ of $n$-simplicies. 
\item For each $n\in\N$ and $0\leq i\leq n$, a map $d_i^n:S_n\to S_{n-1}$ such that $$d_i^{n-1}\circ d_j^n=d_{j-1}^{n-1}\circ d_i^n$$ called the face relation is satisfied whenever $i<j$. 
\end{itemize}
\end{defn}

In particular, $d_i$ sends an $n$-simplex to its $i$th face, which is an $n-1$ simplex. This means that for $s=[v_1,\dots,v_n]\in S_n$, $d_i(s)=[v_1,\dots,\hat{v}_i,\cdots,v_n]$. 

\begin{defn}{Geometric Realization of $\Delta$-Complexes}{}\\
Let $S=(S_\bullet,d_\bullet)$ be a $\Delta$-set. The geometric realization of $S$ is a topological space $X$ that is built up inductively as follows: 
\begin{itemize}
\item The $0$-skeleton $X^0$ is a discrete set with points in $S_0$
\item Given the $n-1$ skeleton $X^{n-1}$ and $S_{n-1}$. Now define $$X^n=\left(X^{n-1}\cup\coprod_{\alpha\in I_n}\Delta_\alpha^n\right)/\sim$$ where $\sim$ is the equivalence relation $\Delta_\beta^{n-1}\sim\partial_k\Delta_\alpha^n$ given from the face maps $d_\bullet$. (Intuitively, each face of the $n$-simplex in $S_n$ gets identified with a $n-1$ simplex in $X^{n-1}$)
\item Define $X=\bigcup_nX^n$. The minimal $n$ for which $X=X^n$ is called the dimension of $n$
\end{itemize}
We also write $X$ as $\abs{S}$ to indicate the $\Delta$-set. 
\end{defn}

\begin{defn}{Map of Delta-Complexes}{} Let $S=(S_\bullet,d_\bullet)$ and $S'=(S_\bullet',d_\bullet')$ be two $\Delta$-sets. A map of $\Delta$-sets $f:S\to S'$ is a family of maps $f_n:S_n\to S_n'$ for each $n\in\N$ such that for all $0\leq i\leq n$, we have that $$d_i'\circ f_n=f_{n-1}\circ d_i:S_n\to S_{n-1}'$$
\end{defn}

\subsection{Simplicial Complexes}
\begin{defn}{Simplicial Complexes}{} Let $\mK$ be a set of simplexes. We say that $\mK$ is a simplicial complex if the following are true. 
\begin{itemize}
\item For any $S\in\mK$, every face $\partial_k S$ of $S$ lies in $\mK$. 
\item If $S_1$ and $S_2$ are two simplexes in $\mK$, then either $S_1\cap S_2$ is empty or $S_1\cap S_2$ is a common face of both $S_1$ and $S_2$. 
\end{itemize}
\end{defn}

\begin{thm}{}{} Every simplicial complex is a $\Delta$-complex. 
\end{thm}

\begin{defn}{Simplicial Maps}{}\\
Let $\mK,m\mL$ be simplicial complexes. Let $f:\mK\to\mL$ be a function. We say that $f$ is a simplicial map if the following are true. 
\begin{itemize}
\item There exists a simplex $T\in\mL$ such that $f(S)\subseteq T$. 
\item For all $S\in\mK$, $f|_S$ is affine linear. 
\end{itemize}
\end{defn}

\begin{defn}{Piecewise Linear Maps}{}\\
Let $\mK,m\mL$ be simplicial complexes. Let $f:\mK\to\mL$ be a function. We say that $f$ is piecewise linear if there is a subcomplex $\mK'\subseteq\mK$ such that $f|_{\mK'}$ is a simplicial map. 
\end{defn}

\subsection{Abstract Simplicial Complexes}
\begin{defn}{Abstract Simplicial Complex}{}\\
Let $T$ be a set. Let $S\subseteq\mP(T)$. We say that $S$ is an abstract simplicial complex if for all $U\in S$ and $V\subseteq U$, we have $V\in S$. 
\end{defn}

We mostly consider the case where $T$ is finite. In this case, we may replace the underlying set $T$ by $\{1,\dots,n\}$ by a suitable $n\in\N$. 

\begin{eg}{}{}\\
The set $S=\mP(\{1,2,3\})\cup\mP(\{1,3,4\})\cup\mP(\{2,4\})$ is an abstract simplicial complex. 
\end{eg}

\begin{defn}{Geometric Realization of Abstract Simplicial Complexes}{}\\
Let $n\in\N$. Let $S$ be an abstract simplicial complex on $\{1,\dots,n\}$. Define the geometric realization of $S$ to be the space $$\abs{S}=\bigcup_{I\in S}\Delta^I$$ where $\Delta^I$ is the $\abs{I}$-simplex with vertices $e_i$ for $i\in I$. 
\end{defn}

By definition, the geometric realization $\abs{S}$ is a subcomplex of the standard $n$-simplex $[e_1,\dots,e_{n+1}]\subseteq\R^{n+1}$. 

\begin{eg}{}{}\\
Let $n\in\N$. Consider the set $S=\mP(\{1,\dots,n+1\})$. Then $\abs{S}$ is the standard $n$-simplex. 
\end{eg}

\subsection{Cubical Embeddings of Simplicial Complexes}
\begin{defn}{The Cubical Embedding}{}\\
Let $n\in\N$. Define the cubical embedding $i_c:\Delta^n\to[0,1]^{n+1}$ as follows. 
\begin{enumerate}
\item For $1\leq k\leq n$, the barycenter of the $k$-dimensional face $B([e_{i_1},\dots,e_{i_k}])$ is sent to $(u_1,\dots,u_{n+1})\in[0,1]^{n+1}$ where $u_j=0$ if $j\in\{i_1,\dots,i_k\}$ and $u_j=1$ otherwise. 
\item Extend this into a piecewise linear embedding $i_c:\Delta^n\to[0,1]^{n+1}$ as follows. For each simplex $[p_1,\dots,p_l]$ in the barycentric subdivision of $\Delta^n$, $[i_c(p_1),\dots,i_c(p_l)]$ lies on a face of $[0,1]^{n+1}$. Hence there exists an simplicial map sending $[p_1,\dots,p_l]$ to $[i_c(p_1),\dots,i_c(p_l)]$. 
\end{enumerate}
\end{defn}

\begin{lmm}{}{}\\
Let $n\in\N$. Then $\partial[0,1]^n$ is a simplicial complex. 
\end{lmm}

\begin{defn}{Cubical Embeddings of Abstract Simplicial Complexes}{}\\
Let $\mK$ be an abstract simplicial complex. Define the cubical embedding of $\mK$ to be the subspace $$\text{cc}(\mK)=i_c(\abs{\mK})$$
\end{defn}

\section{Polyhedrons and Polytopes}
\subsection{Convex Geometry}
\begin{defn}{Convex Sets}{}\\
Let $C\subseteq\R^n$. We say that $C$ is convex if for all $x,y\in C$ and $t\in[0,1]$, we have $(1-t)x+ty\in C$. 
\end{defn}

\begin{defn}{Convex Hull}{}\\
Let $S\subseteq\R^n$ be a subset. Define the convex hull of $S$ to be $$\text{Conv}(S)=\left\{\sum_{k=1}^n\lambda_kx_k\;\bigg{|}\;x_k\in S\text{ and }\sum_{k=1}^n\lambda_k=1\right\}$$
\end{defn}

\begin{prp}{}{}\\
Let $S\subseteq\R^n$ be a subset. Then the following are sets are equal. 
\begin{itemize}
\item The convex hull $\text{Conv}(S)$. 
\item $\bigcap_{C\subseteq\R^n\text{ is convex}}C$. 
\item The smallest convex set containing $S$. 
\end{itemize}
\end{prp}

\subsection{Polyhedrons and its Properties}
Polytopes exists more generically in Euclidean space. We lose nothing discussing polytopes in $\R^n$ since Euclidean spaces maybe identified with $\R^n$ canonically. 

\begin{defn}{Halfspaces}{}\\
A halfspace of $\R^n$ is a subset of $\R^n$ of the form $$\{x\in\R^n\;|\;\langle a,x\rangle+b\geq 0\;|\;\}$$ for some $a,b\in\R^n$. 
\end{defn}

\begin{defn}{Polyhedrons}{}\\
A polyhedron in $\R^n$ is an intersection of finitely many halfspaces. 
\end{defn}

\begin{defn}{Polytopes}{}\\
A polytope in $\R^n$ is a polyhedron that is bounded. 
\end{defn}

\begin{prp}{}{}\\
Let $P\subseteq\R^n$ be a subset. Then $P$ is a polytope if and only if there exists a finite subset $S\subseteq\R^n$ such that $P=\text{Conv}(S)$. 
\end{prp}

\begin{defn}{Dimension}{}\\
Let $P\subseteq\R^n$ be a polytope. Define the dimension of $P$ to be the dimension of the smallest affine subspace containing $P$. 
\end{defn}

\begin{defn}{Affinely Isomorphic}{}\\
Let $P_1\subseteq\R^{n_1}$ and $P_2\subseteq\R^{n_2}$ be polytopes. We say that $P_1$ and $P_2$ are affinely isomorphic if there exists an affine transformation $f:\R^{n_1}\to\R^{n_2}$ such that $f|_{P_1}$ is a bijection. 
\end{defn}

Note that $f$ itself need not be an affine isomorphism (i.e. an isomorphism of affine spaces). 

\subsection{Faces}
\begin{defn}{Supporting Hyperplane}{}\\
Let $P\subseteq\R^n$ be a polyhedron. Let $H\subseteq\R^n$ be a hyperplane. We say that $H$ is a supporting hyperplane for $P$ if the following are true. 
\begin{itemize}
\item $P\cap H\neq\emptyset$. 
\item $P$ lies entirely in one of the halfspaces defined by $H$. 
\end{itemize}
\end{defn}

\begin{defn}{Faces}{}\\
Let $P\subseteq\R^n$ be a polyhedron. A face of $P$ is a subset of $P$ defined by $$F=P\cap H$$ where $H$ is a supporting hyperplane for $P$. 
\end{defn}

\begin{lmm}{}{}\\
Let $P\subseteq\R^n$ be a polyhedron. Let $F\subseteq P$ be a face. Then $F$ is a polytope. 
\end{lmm}

\begin{defn}{Vertices, Edges and Facets}{}\\
Let $P\subseteq\R^n$ be a polyhedron. Let $F\subseteq P$ be a face. 
\begin{itemize}
\item We say that $F$ is a vertex if $\dim(F)=0$. 
\item We say that $F$ is an edge if $\dim(F)=1$. 
\item We say that $F$ is a facet if $\dim(F)=\dim(P)-1$. 
\end{itemize}
\end{defn}

\begin{defn}{The Face Poset}{}\\
Let $P\subseteq\R^n$ be a polyhedron. Define the face poset $\mF(P)$ of $P$ to be the set of faces of $P$ together with inclusion. 
\end{defn}

\begin{defn}{Combinatorially Equivalent}{}\\
Let $P_1\subseteq\R^{n_1}$ and $P_2\subseteq\R^{n_2}$ be polytopes. We say that $P_1$ and $P_2$ are combinatorially equivalent if there exists a inclusion preserving bijection between $\mF(P_1)$ and $\mF(P_2)$. In this case we write $P_1\simeq P_2$. 
\end{defn}

\section{Cones and Fans}
\subsection{Cones and its Properties}
\begin{defn}{Cones}{}\\
Let $C\subseteq\R^n$. We say that $C$ is a cone if $x\in C$ implies that $sx\in C$ for all $s\geq 0$. 
\end{defn}

\begin{defn}{Polyhedral Cones}{}\\
A cone in $\R^n$ is a subset $C\subseteq\R^n$ such that if $c_1,\dots,c_n\in C$, then $\sum_{i=1}^n\lambda_ic_i\in C$ for all $\lambda_1,\dots,\lambda_n\geq 0$. 
\end{defn}

\begin{defn}{Conical Hull}{}\\
Let $S\subseteq\R^n$ be finite. Define the conical hull of $S$ to be $$\text{Cone}(S)=\{\sum_{s\in S}\lambda_ss\;|\;\lambda_s\geq 0\}$$
\end{defn}

\begin{prp}{}{}\\
Let $C\subseteq\R^n$ be a subset. Then the following are equivalent. 
\begin{itemize}
\item $C$ is a polyhedral cone. 
\item There exists a finite subset $S\subseteq\R^n$ such that $C=\text{Cone}(S)$. 
\item $C$ is the intersection of finitely many halfspaces containing $0$ in their boundary. 
\end{itemize}
\end{prp}

Therefore cones are also polyhedrons. 

\begin{defn}{Dimension of a Cone}{}\\
Let $C\subseteq\R^n$ be a cone. Define the dimension of $C$ to be the dimension of the smallest affine subspace containing $C$. 
\end{defn}

\begin{defn}{Rational Cones}{}\\
Let $C\subseteq\R^n$ be a cone. We say that $C$ is rational if there exists a finite set $S\subseteq\Z^n$ such that $C=\text{Cone}(S)$. 
\end{defn}

\begin{defn}{Regular Cones}{}\\
Let $C\subseteq\R^n$ be a cone. We say that $C$ is regular if there exists a finite set $S\subseteq\Z^n$ such that the following are true. 
\begin{itemize}
\item $C=\text{Cone}(S)$. 
\item Elements of $S$ are $\Z$-linearly independent. 
\end{itemize}
\end{defn}

\subsection{The Dual Cone}
\begin{defn}{The Dual of a Cone}{}\\
Let $C\in\R^n$ be a cone. Define the dual of $C$ to be $$C^\vee=\{x\in\R^n\;|\;\langle x,c\rangle\geq 0\text{ for all }c\in C\}$$
\end{defn}

\begin{defn}{Associated Hyperplane and Halfspace of a Linear Functional}{}\\
Let $\phi\in(\R^n)^\ast$. Define the associated hyperplane of $\phi$ to be $$H_\phi=\{x\in\R^n\;|\;\phi(x)=0\}$$ Define the associated halfspace of $\phi$ to be $$H_\phi^+=\{x\in\R^n\;|\;\phi(x)\geq 0\}$$
\end{defn}

\begin{lmm}{}{}\\
Let $C\in\R^n$ be a cone. Let $H$ be a hyperplane in $\R^n$. Then $H$ is a supporting hyperplane of $C$ if and only if there exists $\phi\in(\R^n)^\ast$ such that $H=H_\phi$. 
\end{lmm}

\begin{lmm}{}{}\\
Let $C\in\R^n$ be a polyhedral cone. Then the following are true. 
\begin{itemize}
\item $C^\vee$ is a polyhedral cone. 
\item $(C^\vee)^vee=C$. 
\item $C=\bigcap_{i=1}^nH_{\phi_i}$ for $\phi_i\in(\R^n)^\ast$ if and only if $C^\vee=\text{Cone}(\phi_1,\dots,\phi_n)$. 
\end{itemize}
\end{lmm}

\begin{lmm}{Separation Lemma}{}\\
Let $C_1,C_2\subseteq\R^n$ be cones such that $F=C_1\cap C_2$ is a face of both $C_1$ and $C_2$. Then there exists $\phi\in C_1^\vee\cap(-C_2)^\vee$ such that $$F=C_1\cap H_\phi=C_2\cap H_\phi$$
\end{lmm}

\begin{defn}{Strongly Convex Cones}{}\\
Let $C\subseteq\R^n$ be a cone. We say that $C$ is strongly convex if $C$ does not contain a line. 
\end{defn}

\begin{lmm}{}{}\\
Let $C\subseteq\R^n$ be a cone. Then the following are equivalent. 
\begin{itemize}
\item $C$ is strongly convex. 
\item $\{0\}$ is a face of $C$. 
\item $C\cap(-C)=\{0\}$. 
\item $\dim(C^\vee)=n$. 
\end{itemize}
\end{lmm}

\subsection{Polar Sets}
\begin{defn}{Polar Sets}{}\\
Let $S\subseteq\R^n$ be a subset. Define the polar set of $S$ to be $$S^\ast=\{f\in(\R^n)^\ast\;|\;f(x)\leq 1\text{ for all }x\in S\}$$
\end{defn}

\begin{prp}{}{}\\
Let $P\subseteq\R^n$ be a polytope. If $0\in P$, then $P^\ast$ is a polytope. 
\end{prp}

\begin{eg}{}{}\\
The polar set of the cube $C^3=\{(a,b,c)\in\R^3\;|\;-1\leq a,b,c\leq 1\}$ is the octahedron $$(C^3)^\ast=\{(x,y,z)\in(\R^3)^\ast\;|\;\pm x+\pm y+\pm z\leq 1\}$$
\end{eg}

$P=(P^\ast)^\ast$. 

$\mF(P)^\text{op}\cong\mF(P^\ast)$. 

\subsection{Simple and Simplicial Polytopes}
\begin{defn}{Simple Polytopes}{}\\
Let $P\subseteq\R^n$ be a polytope. We say that $P$ is a simple polytope if for all vertices $v\in P$, there exists $n$ edges $e_1,\dots,e_n\subseteq P$ such that $v\in\partial e_i$ for all $i$.

\subsection{The Hilbert Series of Stanley-Reisner Rings}
\begin{defn}{The Face Vector}{}\\
Let $S$ be an abstract simplicial complex on $\{1,\dots,n\}$. Define the face vector $(f_{-1},f_0,\dots,f_n)$ of $S$ by the formula $$f_i=\text{The number of }i\text{-dimensional faces of }S$$ for $0\leq i\leq n$ and by convention, $f_{-1}=1$. 
\end{defn}

\begin{defn}{The $h$-Vector}{}\\
Let $S$ be an abstract simplicial complex on $\{1,\dots,n\}$. Let $(f_0,\dots,f_n)$ be the face vector of $S$. Define the $h$-vector $(h_0,\dots,h_n)$ of $S$ by the formula $$h_i=\sum_{k=0}^i(-1)^{i-k}\binom{n+1-k}{i-k}f_{k-1}$$
\end{defn}

\begin{lmm}{}{}\\
Let $S$ be an abstract simplicial complex on $\{1,\dots,n\}$. Let $(f_0,\dots,f_n)$ be the face vector of $S$. Let $(h_0,\dots,h_n)$ be the $h$-vector of $S$. Then the following are true. 
\begin{itemize}
\item We have $$\sum_{k=0}^{n+1}f_{k-1}(t-1)^{d-k}=\sum_{j=0}^{n+1}h_jt^{n+1-j}$$ for variable $t$. 
\item For $0\leq i\leq n$, we have $$f_i=\sum_{k=0}^{i+1}\binom{n+1}{i+1-k}h_k$$
\end{itemize}
\end{lmm}

\begin{thm}{Stanley}{}\\
Let $k$ be a field. Let $n\in\N$. Let $S$ be an abstract simplicial complex on $\{1,\dots,n\}$. Let $(f_0,\dots,f_n)$ be the face vector of $S$. Let $(h_0,\dots,h_n)$ be the $h$-vector of $S$. Then the Hilbert series of $k[S]$ is given by $$HS_{k[S]}(t)=\sum_{i=0}^{n+1}f_{k-1}\left(\frac{t^2}{1-t^2}\right)^k=\frac{1}{(1-t^2)^{n+1}}\sum_{i=0}^{n+1}h_it^{2i}$$
\end{thm}

\begin{itemize}

\item The Hilbert series for the Stanley-Reisner ring of $\partial\Delta^n$ is given by $$HS_{k[\partial\Delta^n]}(t)=\frac{1}{(1-t^2)^{n+1}}\sum_{i=0}^{n+1}t^{2i}$$
\end{itemize}
\end{eg}

\end{lmm}

Recall the construction of the cubical embedding $i_c:\Delta^n\to[0,1]^{n+1}$. 
\begin{enumerate}
\item For $1\leq k\leq n$, the barycenter of the $k$-dimensional face $B([e_{i_1},\dots,e_{i_k}])$ is sent to $(u_1,\dots,u_n)\in[0,1]^{n+1}$ where $u_j=0$ if $j\in\{i_1,\dots,i_k\}$ and $u_j=1$ otherwise. 
\item Extend this into a piecewise linear embedding $i_c:\Delta^n\to[0,1]^{n+1}$ as follows. For each simplex $[p_1,\dots,p_l]$ in the barycentric subdivision of $\Delta^n$, $[i_c(p_1),\dots,i_c(p_l)]$ lies on a face of $[0,1]^{n+1}$. Hence there exists an simplicial map sending $[p_1,\dots,p_l]$ to $[i_c(p_1),\dots,i_c(p_l)]$. 
\end{enumerate}

Since every finite abstract simplicial complex can be realized as a subcomplex of $\Delta^n$ for large $n$, we also get the cubical embedding of a finite abstract simplicial complex, which is denoted by $\text{cc}(\K)$. 

\begin{defn}{Moment-Angle Complex}{}\\
Let $n\in\N$. Let $\mK$ be an abstract simplicial complex over $\{1,\dots,n\}$. Define the moment-angle complex of $\mK$ to be $$\mZ_\mK=\mu^{-1}(\text{cc}(\mK))\subseteq D^n$$
\end{defn}

Equivalently, it is the pullback of the inclusion map $\mZ_\mK=\text{cc}(\mK)\to[0,1]^n$ and the quotient map $\mu:D^n\to[0,1]^n$. 

\begin{eg}{}{}\\
Let $n\in\N$. Then the following are true. 
\begin{itemize}
\item $\mZ_\emptyset=(S^1)^n$. 
\item $\mZ_{\Delta^n}=D^{n+1}$. 
\end{itemize}
\end{eg}

\begin{lmm}{}{}\\
Let $n\in\N$. Let $\mK$ be an abstract simplicial complex over $\{1,\dots,n\}$. The following are true regarding the moment-angle complex of $\mK$. 
\begin{itemize}
\item $\mZ_\mK$ is a $(S^1)^n$-invariant subspace. 
\item The homeomorphism $\frac{D^n}{(S^1)^n}\cong[0,1]^n$ induced by the map $\mu(z_1,\dots,z_n)=(\abs{z_1}^2,\dots,\abs{z_n}^2)$ descends to a homeomorphism $$\frac{\mZ_\mK}{(S^1)^n}\cong\text{cc}(\mK)$$
\end{itemize}
\end{lmm}

Recall that $\text{cc}(\mK)$ is equal to the union of a finite number of faces of $[0,1]^n$. Also recall the notation $$C_I=\{(x_1,\dots,x_n)\in[0,1]^n\;|\;x_i=0\text{ if }i\in I\}$$ for a face of the cube $[0,1]^n$ that contains the origin and $I\subseteq\{1,\dots,n\}$. 

\begin{lmm}{}{}\\
Let $n\in\N$. Let $\mK$ be an abstract simplicial complex over $\{1,\dots,n\}$. Suppose that $\text{cc}(\mK)=\bigcup_{I\in\mK}C_I$. Then there is a homeomorphism $$\mZ_\mK=\bigcup_{I\in\mK}\mu^{-1}(C_I)=\bigcup_{I\in\mK}\{(x_1,\dots,x_n)\in D^n\;|\;\abs{z_j}^2=1\text{ for all }j\in\{1,\dots,n\}\setminus I\}$$ where $\mu$ is the map $\mu(z_1,\dots,z_n)=(\abs{z_1}^2,\dots,\abs{z_n}^2)$. 
\end{lmm}

For $I\subseteq\{1,\dots,n\}$, the condition that $\abs{z_j}^2=1$ for $j\in\{1,\dots,n\}\setminus I$ means that for those coordinates in $D^n$, they only consist of the boundary which is $S^1$. Hence we may think of $\mZ_\mK$ as the union of products $$\mZ_\mK=\bigcup_{I\in\mK}\left(\prod_{i\in I}D^2\times\prod_{i\in\{1,\dots,n\}\setminus I}S^1\right)$$ But note that this product is ambiguous on which coordinates should be $S^1$ and which should be $D^2$ (because we have a chosen embedding into $\C^n$ although swapping the order of the products give homeomorphic spaces). \end{proof}
\end{eg}

\begin{defn}{The Standard $n$-Torus}{}\\
Let $k$ be a field. Let $n\in\N\setminus\{0\}$. Define the standard $n$-torus over $k$ to be the affine group variety $$\G_m^n=\text{Spec}(k[x,x^{-1}])^n$$

\begin{defn}{Tori}{}\\
Let $k$ be a field. Let $V$ be an affine algebraic group. We say that $V$ is a torus if there exists a field extension $K/k$ such that the algebraic group $V$ over $K$ is isomorphic to $\G_m^n=\text{Spec}(K[x,x^{-1}])^n$ for some $n\in\N\setminus\{0\}$. 
\end{defn}

\begin{defn}{Split Tori}{}\\
Let $k$ be a field. Let $V$ be a torus over the field extension $K/k$. We say that $V$ is split if $K=k$. 
\end{defn}

If $K$ is separably closed or algebraically closed, then all tori are split. \\

\item If $V$ is a representation of $T$, then we can decompose $V$ into its eigenspaces: $$V=\bigoplus_{\chi\in\chi(T)}V_\chi$$
\end{itemize}

\subsection{Constructing Toric Varieties}
\begin{defn}{Toric Varieties}{}\\
Let $X$ be an irreducible variety over $\C$. We say that $X$ is a toric variety if the following are true. 
\begin{itemize}
\item $(\C^\ast)^n$ is a Zariski open subset of $X$. 
\item The action of $(\C^\ast)^n$ on itself extends to $X$. 
\end{itemize}
We denote the underlying torus of $X$ by $T_X$. 
\end{defn}

\begin{defn}{Affine Toric Varieties Associated to Characters}{}\\
Let $m\in\N^+$. Let $\mA=\{\chi_1,\dots,\chi_m\}\subset\chi(\G_m^n)$ be characters of the standard $n$-torus. Define the affine toric variety associated to $\mA$ to be $$Y_\mA=\overline{\im(\Phi_\mA)}$$ where $\Phi_\mA:\G_m^n\to\A_\C^m$ is the map defined by $\Phi_\mA(t)=(\chi_1(t),\dots,\chi_m(t))$
\end{defn}

\begin{prp}{}{}\\
Let $m\in\N^+$. Let $\mA\subset\chi(\G_m^n)$ be a finite subset of characters of the standard $n$-torus. Then the following are true. 
\begin{itemize}
\item $Y_\mA$ is an affine toric variety. 
\item The underlying torus of $Y_\mA$ has character lattice $\Z\mA$. 
\item $\dim(Y_\mA)=\text{rank}(\Z\mA)$
\end{itemize}
\end{prp}

Let $L\subseteq\Z^n$ be a lattice. Let $R$ be a commutative ring. Let $I_L\subseteq R[x_1,\dots,x_n]$ be the lattice ideal of $L$. We say that $I_L$ is a toric ideal if $I_L$ is a prime ideal. 
\end{defn}

\item $X$ is a affine toric variety. 
\item There exists a lattice $L$ and a finite subset $\mA\subseteq L$ such that $X=Y_\mA$. 
\item There exists a toric ideal $I$ of $\C[x_1,\dots,x_n]$ such that $X=\text{Spec}\left(\frac{\C[x_1,\dots,x_n]}{I}\right)$. 
\item There exists an affine semigroup $S$ such that $X=\text{Spec}(\C[S])$. 
\end{itemize}
\end{prp}

\begin{prp}{}{}\\
Let $k$ be an algebraically closed field. Let $S$ be an affine semi-group. Then there is a bijection $$\{p\in\text{Spec}(k[S])\;|\;p\text{ is a closed point }\}\;\;\overset{1:1}{\longleftrightarrow}\;\;\{\phi:S\to(\C,\cdot)\;|\;\phi\text{ is a monoid homomorphism }\}$$ given as follows. For each $p\in\text{Spec}(k[S])$ a closed point, $p$ corresponds to the monoid homomorphism $\gamma_p:S\to\C$ defined by $\gamma_p(m)=\chi^m(p)$
\end{prp}

\begin{prp}{}{}\\
Let $k$ be an algebraically closed field. Let $X=\text{Spec}(k[S])$ be an affine torus over $k$. Then the action of the torus $T_X$ on $X$ is induced by the $k$-algebra homomorphism $$k[S]\to k[\chi(T_X)]\otimes_k k[S]$$ given by $\gamma_p\mapsto\gamma_p\otimes\gamma_p$ where here we identify the closed points on $k[S]$ by the above prp. 
\end{prp}

\subsection{The Dictionary Between Combinatorics and Toric Varieties}
\begin{defn}{The Associated Semigroup of a Cone}{}\\
Let $C\subseteq\R^n$ be a rational cone. Define the associated semigroup of $C$ to be $$S_C=C^\vee\cap(\Z^n)^\ast\subseteq(\R^n)^\ast$$
\end{defn}

\begin{lmm}{Gordon's Lemma}{}\\
Let $C\subseteq\R^n$ be a rational polyhedral cone. Then $S_C$ is finitely generated and hence is an affine semigroup. 
\end{lmm}

\begin{defn}{Affine Toric Variety Associated to Rational Cone}{}\\
Let $\sigma\subseteq\R^n$ be a rational polyhedral cone. Define the affine toric variety associated to $\sigma$ to be $$V_\sigma=\text{Spec}(\C[S_\sigma])$$
\end{defn}

\begin{prp}{}{}\\
There is a one-to-one correspondence $$\{X\text{ is an affine toric variety over }\C\}\;\;\overset{1:1}{\longleftrightarrow}\;\;\{C\text{ is a rational polyhedral cone }\}$$ given by sending a rational polyhedral cone $\sigma\subseteq\R^n$ to $\text{Spec}(\C[S_\sigma])$. 
\end{prp}

\begin{prp}{}{}\\
Let $\sigma\subseteq\R^n$ be a rational polyhedral cone. Then the following are equivalent. 
\begin{itemize}
\item $\text{Spec}(\C[S_\sigma])$ is normal. 
\item $S_\sigma$ is a saturated affine semi-group. 
\item $\sigma$ is strongly convex. 
\item $\dim(\text{Spec}(\C[S_\sigma])=n$. 
\end{itemize}
\end{prp}

\begin{prp}{}{}\\
Let $X=\text{Spec}(\C[S])$ be an affine toric variety over $\C$ where $S$ is an affine semigroup. Then the normalization of $X$ is given by $$\tilde{X}=\text{Spec}(\C[S_{\text{Cone}(S)^\vee}])$$ together with the normalization map given by the inclusion $\C[S]\hookrightarrow\C[S_{\text{Cone}(S)^\vee}]$. 
\end{prp}

\begin{prp}{}{}\\
Let $\sigma\subseteq\R^n$ be a strongly convex rational polyhedral cone. Then $\text{Spec}(\C[S_\sigma])$ is smooth if and only if $\sigma$ is regular. 
\end{prp}

Thus we have constructed a dictionary 
\begin{align*}
\text{Affine Toric Varieties}\;\;&\overset{1:1}{\longleftrightarrow}\;\;\text{Rational Polyhedral Cones}\\
\text{Affine Normal Toric Varieties}\;\;&\overset{1:1}{\longleftrightarrow}\;\;\text{Strongly Convex Rational Polyhedral Cones}\\
\text{Affine Smooth Toric Varieties}\;\;&\overset{1:1}{\longleftrightarrow}\;\;\text{Regular Rational Polyhedral Cones}
\end{align*}
