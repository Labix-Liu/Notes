\section{Polyhedrons and Polytopes}
\subsection{Convex Geometry}
\begin{defn}{Convex Sets}{}\\
Let $C\subseteq\R^n$. We say that $C$ is convex if for all $x,y\in C$ and $t\in[0,1]$, we have $(1-t)x+ty\in C$. 
\end{defn}

\begin{defn}{Convex Hull}{}\\
Let $S\subseteq\R^n$ be a subset. Define the convex hull of $S$ to be $$\text{Conv}(S)=\left\{\sum_{k=1}^n\lambda_kx_k\;\bigg{|}\;x_k\in S\text{ and }\sum_{k=1}^n\lambda_k=1\right\}$$
\end{defn}

\begin{prp}{}{}\\
Let $S\subseteq\R^n$ be a subset. Then we have $$\text{Conv}(S)=\bigcap_{\substack{S\subseteq C\subseteq\R^n\\C\text{ is convex }}}C$$
\end{prp}

\begin{thm}{Caratheodory's Theorem}{}\\
Let $S\subseteq\R^n$ be a subset. Then there exists $x_1,\dots,x_{n+1}\in S$ such that $$\text{Conv}(S)=\text{Conv}(x_1,\dots,x_{n+1})$$
\end{thm}

\subsection{Polyhedrons and its Properties}
Polytopes exists more generically in Euclidean space. We lose nothing discussing polytopes in $\R^n$ since Euclidean spaces maybe identified with $\R^n$ canonically. 

\begin{defn}{Halfspaces}{}\\
A halfspace of $\R^n$ is a subset of $\R^n$ of the form $$\{x\in\R^n\;|\;\langle a,x\rangle+b\geq 0\}$$ for some $a,b\in\R^n$. 
\end{defn}

\begin{defn}{Polyhedrons}{}\\
Let $P\subseteq\R^n$. We say that $P$ is a polyhedon if there exists finitely many half spaces $H_1,\dots,H_k$ such that $$P=\bigcap_{i=1}^kH_i$$
\end{defn}

\begin{defn}{Polytopes}{}\\
Let $P\subseteq\R^n$ be a polyhedron. We say that $P$ is a polytope if $P$ is bounded. 
\end{defn}

\begin{thm}{Weyl-Minkowski}{}\\
Let $P\subseteq\R^n$ be a subset. Then $P$ is a polytope if and only if there exists a finite subset $S\subseteq\R^n$ such that $P=\text{Conv}(S)$. 
\end{thm}

\begin{defn}{Dimension}{}\\
Let $P\subseteq\R^n$ be a polytope. Define the dimension of $P$ to be the dimension of the smallest affine subspace containing $P$. 
\end{defn}

\subsection{The Faces of a Polyhedron}
\begin{defn}{Supporting Hyperplane}{}\\
Let $P\subseteq\R^n$ be a polyhedron. Let $H\subseteq\R^n$ be a hyperplane. We say that $H$ is a supporting hyperplane for $P$ if the following are true. 
\begin{itemize}
\item $P\cap H\neq\emptyset$. 
\item $P$ lies entirely in one of the halfspaces defined by $H$. 
\end{itemize}
\end{defn}

\begin{defn}{Faces}{}\\
Let $P\subseteq\R^n$ be a polyhedron. A face of $P$ is a subset of $P$ defined by $$F=P\cap H$$ where $H$ is a supporting hyperplane for $P$. 
\end{defn}

\begin{lmm}{}{}\\
Let $P\subseteq\R^n$ be a polyhedron. Let $F\subseteq P$ be a face. Then $F$ is a polytope. 
\end{lmm}

\begin{lmm}{}{}\\
Let $P\subseteq\R^n$ be a polyhedron. Let $F,G\subseteq P$ be faces of $P$. Then $F\cap G$ is a face of $P$. 
\end{lmm}

\begin{lmm}{}{}\\
Let $P\subseteq\R^n$ be a polyhedron. Let $F\subseteq P$ be a face of $P$. Let $G\subseteq F$ be a face of $F$. Then $G$ is a face of $P$. 
\end{lmm}

\begin{defn}{Vertices, Edges and Facets}{}\\
Let $P\subseteq\R^n$ be a polyhedron. Let $F\subseteq P$ be a face. 
\begin{itemize}
\item We say that $F$ is a vertex if $\dim(F)=0$. Denote the set of vertices of $P$ by $V(P)$. 
\item We say that $F$ is an edge if $\dim(F)=1$. 
\item We say that $F$ is a facet if $\dim(F)=\dim(P)-1$. 
\end{itemize}
\end{defn}

\begin{defn}{The Face Poset}{}\\
Let $P\subseteq\R^n$ be a polyhedron. Define the face poset $\mF(P)$ of $P$ to be the set of faces of $P$ together with inclusion. 
\end{defn}

\begin{defn}{The Face Vector}{}\\
Let $P\subseteq\R^n$ be a polytope of dimension $d$. Define the face vector $(f_0,\dots,f_{d-1})$ of $S$ by the formula $$f_i=\text{The number of }i\text{-dimensional faces of }P$$ for $0\leq i\leq d-1$ and by convention, $f_{-1}=f_d=1$. 
\end{defn}

\begin{lmm}{Euler's Formula}{}\\
Let $P\subseteq\R^n$ be a polytope of dimension $d$. Then we have $$\sum_{i=0}^{d-1}(-1)^if_i=1+(-1)^d$$
\end{lmm}

\begin{defn}{Combinatorially Equivalent}{}\\
Let $P_1\subseteq\R^{n_1}$ and $P_2\subseteq\R^{n_2}$ be polytopes. We say that $P_1$ and $P_2$ are combinatorially equivalent if there exists a inclusion preserving bijection between $\mF(P_1)$ and $\mF(P_2)$. In this case we write $P_1\simeq P_2$. 
\end{defn}

\begin{defn}{Edge Graph}{}\\
Let $P\subseteq\R^n$ be a polytope. Define the edge graph $G(P)$ of $P$ by the following data. 
\begin{itemize}
\item $V(G(P))=V(P)$. 
\item $E(G(P))=\{(p,q)\;|\;\text{ There exists an edge }E\text{ such that }\partial E=\{p,q\}\}$. 
\end{itemize}
\end{defn}

\subsection{Dual Polytopes}
\begin{defn}{Polar Dual}{}\\
Let $P\subseteq\R^n$ be a polytope. Define the polar dual of $P$ to be $$P^\ast=\{y\in\R^n\;|\;\langle x,y\rangle\leq 1\text{ for all }x\in P\}$$
\end{defn}

\begin{prp}{}{}\\
Let $P\subseteq\R^n$ be a polytope. Suppose that $0\in P$. Then the following are true. 
\begin{itemize}
\item $P^\ast$ is a polytope. 
\item $P=(P^\ast)^\ast$. 
\item Let $F\subseteq P$ be a face. Then $F^\ast$ is a face of $P^\ast$ of dimension $\dim(P)-1-\dim(F)$. 
\item There is a inclusion reversing bijection $\mF(P)\overset{1:1}{\leftrightarrow}\mF(P)^\text{op}$ given by $F\mapsto F^\ast$. 
\end{itemize}
\end{prp}

\begin{defn}{Dual Graph}{}\\
Let $P\subseteq\R^n$ be a polytope. Define the dual graph $G^\ast(P)$ as follows. 
\begin{itemize}
\item $V(G^\ast(P))=\{F\subseteq P\;|\;F\text{ is a facet of }P\}$. 
\item $E(G^\ast(P))=\{(F,G)\;|\;F\cap G\text{ is a face of dimension }\dim(P)-2\}$. 
\end{itemize}
\end{defn}

\begin{lmm}{}{}\\
Let $P\subseteq\R^n$ be a polytope such that $0\in P$. Then $G(P^\ast)=G^\ast(P)$. 
\end{lmm}

\section{Operations on Polytopes}
\subsection{Minkowski Sum}
\begin{defn}{Minkowski Sum}{}\\
Let $P,Q\subseteq\R^n$ be polytopes. Define the Minkowski sum of $P$ and $Q$ by $$P+Q=\{x+y\in\R^n\;|\;x\in P,y\in Q\}$$
\end{defn}

\begin{prp}{}{}\\
Let $P,Q\subseteq\R^n$ be polytopes. Suppose that $P=\text{Conv}(v_1,\dots,v_p)$ and $Q=\text{Conv}(w_1,\dots,w_q)$. Then $P+Q$ is a polytope and $$P+Q=\text{Conv}\left(\{v_i+w_j\;|\;1\leq i\leq p,1\leq j\leq q\}\right)$$
\end{prp}

\subsection{The Join of Polytopes}
\begin{defn}{The Join of Two Polytopes}{}\\
Let $P\subseteq\R^n$ and $Q\subseteq\R^m$ be polytopes. Define the join of $P$ and $Q$ as follows. Embed $P$ into $\R^{n+m+1}$ by $x\mapsto(x,0,\dots,0)$ and embed $Q$ into $\R^{n+m+1}$ by $y\mapsto(0,\dots,0,y,1)$. Then the join of $P$ and $Q$ is $$P\ast Q=\{(1-\lambda)x+\lambda y\;|\;x\in P,y\in Q,0\leq\lambda\leq 1\}$$
\end{defn}

\begin{prp}{}{}\\
Let $P$ and $Q$ be polytopes. Then the following are true. 
\begin{itemize}
\item $\dim(P\ast Q)=\dim(P)+\dim(Q)+1$. 
\item $\mF(P\ast Q)=\mF(P)\ast\mF(Q)$. 
\item $V(P\ast Q)=V(P)\amalg V(Q)$. 
\end{itemize}
\end{prp}

\begin{eg}{}{}\\
Let $n,m\in\N^+$. Then we have $$\Delta^n\ast\Delta^m=\Delta^{n+m+1}$$
\end{eg}

\begin{eg}{}{}\\
Let $P$ be a polytope. Then the following are true. 
\begin{itemize}
\item $P\ast\Delta^1\cong\Sigma P$. 
\item $P\ast\Delta^0\cong SP$. 
\end{itemize}
\end{eg}

\section{Symmetries of a Polytope}
\subsection{Congruent Polytopes}
\begin{defn}{Congruent Polytopes}{}\\
Let $P\subseteq\R^n$ and $Q\subseteq\R^m$ be polytopes. We say that $P$ and $Q$ are congruent if there exists a function $\varphi:P\to Q$ such that the following are true. 
\begin{itemize}
\item $\varphi$ is a bijection. 
\item $\|\varphi(y)-\varphi(x)\|=\|y-x\|$ for all $x,y\in P$. 
\end{itemize}
\end{defn}

\begin{prp}{}{}\\
Let $P\subseteq\R^n$ and $Q\subseteq\R^m$ be polytopes. Then $P$ and $Q$ are congruent if and only if there exists an isometry $\varphi:\R^n\to\R^m$ such that $\varphi(P)=Q$. 
\end{prp}

\begin{defn}{Symmetries of a Polytope}{}\\
Let $P\subseteq\R^n$ be a polytope. A symmetry of $P$ is a congruence $\varphi:P\to P$. Denote the group of symmetries of $P$ by $$\text{Sym}(P)=\{\varphi:P\to P\;|\;\varphi\text{ is a congruence }\}\leq\text{Isom}(\R^n)$$
\end{defn}

\begin{lmm}{}{}\\
Let $P\subseteq\R^n$ be a polytope. Let $F\subseteq P$ be a face of $P$. Let $\varphi:P\to P$ be a symmetry. Then $\varphi(F)$ is a face of dimension $\dim(F)$. 
\end{lmm}

\begin{prp}{}{}\\
Let $P\subseteq\R^n$ be a polytope. Let $\tau:\R^n\to\R^n$ by the map defined by $\tau(x)=x-\overline{P}$. Then there is an isomorphism $$\text{Sym}(P)\cong\tau\text{Sym}(P)\tau^{-1}$$
\end{prp}

WLOG we may translate any polytope $P$ such that $\overline{P}=0\in\R^n$.

Let $P\subseteq\R^n$ be a polytope such that $\overline{P}=0$. Then we have $$\text{Sym}(P)\leq O(\text{AffSpan}(P))\cong O(\dim(P),\R)$$
\end{prp}

\subsection{Affine Isomorphisms}
\begin{defn}{Affinely Isomorphic}{}\\
Let $P_1\subseteq\R^{n_1}$ and $P_2\subseteq\R^{n_2}$ be polytopes. We say that $P_1$ and $P_2$ are affinely isomorphic if there exists an affine transformation $f:\R^{n_1}\to\R^{n_2}$ such that $f|_{P_1}$ is a bijection. 
\end{defn}

Note that $f$ itself need not be an affine isomorphism (i.e. an isomorphism of affine spaces). 

\begin{defn}{Affine Symmetries of a Polytope}{}\\
Let $P\subseteq\R^n$ be a polytope. An affine symmetry of $P$ is an affine isomorphism $f:P\to P$. Denote the group of affine symmetries of $P$ by $$\text{Aff}(P)=\{f:P\to P\;|\;f\text{ is an affine isomorphism }\}\leq\text{Aff}(\R^n)$$
\end{defn}

\begin{prp}{}{}\\
Let $P\subseteq\R^n$ be a polytope. Let $\tau:\R^n\to\R^n$ by the map defined by $\tau(x)=x-\overline{P}$. Then there is an isomorphism $$\text{Aff}(P)\cong\tau\text{Aff}(P)\tau^{-1}$$
\end{prp}

\begin{prp}{}{}\\
Let $P\subseteq\R^n$ be a polytope such that $\overline{P}=0$. Then we have $$\text{Aff}(P)\leq O(\text{AffSpan}(P))\cong\text{GL}(\dim(P),\R)$$
\end{prp}

\begin{prp}{}{}\\
Let $P\subseteq\R^n$ be a polytope. Then we have $$\text{Sym}(P)\leq\text{Aff}(P)\leq\text{Aut}(\mF(P))$$
\end{prp}

\section{The Duality between Simple and Simplicial Polytopes}
\subsection{Simplicial Polytopes}
\begin{defn}{Simplicial Polytopes}{}\\
Let $P\subseteq\R^n$ be a polytope. We say that $P$ is a simplicial polytope if all its facets are simplicies. 
\end{defn}

\begin{lmm}{}{}\\
Let $P\subseteq\R^n$ be a simplicial polytope. Then $\partial P$ is a simplicial complex. 
\end{lmm}

\begin{lmm}{}{}\\
Let $P\subseteq\R^n$ be a simplicial polytope. Then $\partial P$ is a triangulation of $S^{\dim(P)-1}$. 
\end{lmm}

\begin{prp}{The Dehn-Sommerville Equation}{}\\
Let $P\subseteq\R^n$ be a simplicial polytope of dimension $d$. Then the following are true: $$\sum_{j=k}^{d-1}(-1)^j\binom{j+1}{k+1}f_j=(-1)^{d-1}f_k$$ for $-1\leq k\leq d-2$. 
\end{prp}

In particular, $\partial P$ is a simplicial complex. The notion of face vector coincides, and they enjoy both the Dehn-Sommerville equation and its relation with the $h$-vector. 

\begin{eg}{}{}\\
Substituting $k=-1$ in the Dehn-Somemerville equation recovers Euler's identity. 
\end{eg}

\begin{eg}{}{}\\
Let $P$ be a simplicial $3$-polytope. Then the Dehn-Sommmerville equations give $$2=f_0-f_1+f_2\;\;\text{ and }\;\;2f_1=3f_2$$ Substituting gives $f_1=3f_0-6$. Hence the $f$-vector of $P$ is given by $$f(P)=(f_0,3f_0-6,2f_0-4)$$
\end{eg}

\begin{eg}{}{}\\
Let $P$ be a simplicial $4$-polytope. Then the Dehn-Sommmerville equations give $$0=f_0-f_1+f_2-f_3\;\;\text{ and }\;\;2f_0+3f_2=2f_1+4f_3\;\;\text{ and }\;\;2f_3=f_2$$ Substituting gives $f_1=3f_0-6$. Hence the $f$-vector of $P$ is given by $$f(P)=(f_0,f_1,2(f_1-f_0),f_1-f_0)$$
\end{eg}

\subsection{Simple Polytopes}
\begin{defn}{Simple Polytopes}{}\\
Let $P\subseteq\R^n$ be a polytope. We say that $P$ is a simple polytope if for all vertices $v\in P$, there exists $n$ edges $e_1,\dots,e_n\subseteq P$ such that $v\in\partial e_i$ for all $i$. 
\end{defn}

\begin{lmm}{}{}\\
Let $P_1\subseteq\R^n$ and $P_2\subseteq\R^m$ be simple polytopes. Then the Cartesian product $P_1\times P_2$ is a simple polytope. 
\end{lmm}

\subsection{The Duality Theorem}
\begin{prp}{}{}\\
Let $P\subseteq\R^n$ be a polytope. Then $P$ is simple if and only $P^\ast$ is simplicial. 
\end{prp}

Duality of $d$-dimensional faces with $(n-d)$-dimensional faces. 

\begin{prp}{}{}\\
Let $P\subseteq\R^n$ be a polytope. If $P$ is both simple and simplicial, then $P$ belongs to one of the following types of polytopes. 
\begin{itemize}
\item $n=2$ (and so $P$ is a polygon). 
\item $P$ is a simplex. 
\end{itemize}
\end{prp}

\subsection{The Nerve Complex}
\begin{defn}{The Nerve Complex}{}\\
Let $P\subseteq\R^n$ be a simple polytope. Define the nerve complex of $P$ to be the simplicial complex $$\mK_P=\partial(P^\ast)$$
\end{defn}

\begin{lmm}{}{}\\
Let $P_1\subseteq\R^n$ and $P_2\subseteq\R^m$ be simple polytopes. Then we have $$\mK_{P_1\times P_2}\cong\mK_{P_1}\ast\mK_{P_2}$$
\end{lmm}

\section{Extremal Polytopes}
\subsection{Cyclic Polytopes}
\begin{defn}{Cyclic Polytopes}{}\\
Let $\gamma:\R\to\R^d$ be the function $\gamma(t)=(t,\dots,t^d)$. Let $t_1,\dots,t_n\in\R$ such that $t_i<t_{i+1}$ for $1\leq i\leq n-1$. Define the cyclic polytope of dimension $d$ with $n$ vertices by $$C(n,d)=\text{Conv}(\{\gamma(t_1),\dots,\gamma(t_n)\})$$
\end{defn}

Note: different choices of $t_1,\dots,t_n$ lead to combinatorially equivalent cyclic polytopes. 

\begin{lmm}{}{}\\
Let $n,d\in\N^+$. Then $C(n,d)$ is a simplicial polytope. 
\end{lmm}

\begin{thm}{Gale Evenness Condition}{}\\
Let $n,d\in\N^+$. Let $t_1,\dots,t_n\in\R$ such that $$C(n,d)=\text{Conv}(\{\gamma(t_1),\dots,\gamma(t_n)\})$$ Then $\text{Conv}(\{t_{i_1},\dots,t_{i_d}\})$ is a facet of $C(n,d)$ if and only if $i_k-i_j$ is an odd number for $1\leq j,k\leq d$. 
\end{thm}

\begin{prp}{}{}\\
Let $n,d\in\N^+$. Then we have $$f_{\floor{d/2}-1}(C(n,d))=\binom{n-\ceil{d/2}}{\floor{d/2}}+\binom{n-\floor{d/2}-1}{\ceil{d/2}-1}$$
\end{prp}

\begin{lmm}{}{}\\
Let $n,d\in\N^+$. Then we have $$f_i(C(n,d))=\binom{n}{i+1}$$ for $0\leq i\leq\floor*{\frac{n}{2}}-1$. 
\end{lmm}

\begin{crl}{}{}\\
Let $n,d\in\N^+$. Then we have $$f_i(C(n,d))=\sum_{k=0}^{\floor*{\frac{n}{2}}}\binom{k}{d-1-i}\binom{n-d+k-1}{k}+\sum_{j=0}^{\floor*{\frac{n-1}{2}}}\binom{d-j}{d-1-i}\binom{n-d+j-1}{j}$$ for $0\leq i\leq d$, where we set $\binom{p}{q}=0$ if $p<q$ or $q<0$. 
\end{crl}

\begin{thm}{Upper Bound Theorem}{}\\
Let $P$ be a polytope of dimension $d$. Suppose that $f_0(P)=n$. Then we have $$f_k(P)\leq f_k(C(n,d))$$ for all $k$. 
\end{thm}

\subsection{Stacked Polytopes}
\begin{defn}{Vertex Pulling}{}\\
Let $P\subseteq\R^n$ be a polytope. Let $F\subseteq P$ be a face of $P$. Let $v\in\R^n\setminus P$ such that the following are true. 
\begin{itemize}
\item Suppose that $H$ is a supporting hyperplane of $F$. Then $v$ and $P$ are on different sides in $\R^n\setminus H$. 
\item Suppose that $G\subseteq P$ is a facet such that $G\neq F$, and $H$ is a supporting hyperplane of $G$. Then $v$ and $P$ are on the same side $\R^n\setminus H$. 
\end{itemize}
Define the pulling of $P$ by $v$ along $F$ to be the polytope $$P'=\text{Conv}(P\cup\{v\})$$
\end{defn}

\begin{defn}{Stacked Polytopes}{}\\
Let $P\subseteq\R^n$ be a polytope. We say that $P$ is stacked if there exists a sequence of polytopes $\Delta^{\dim(P)}=S_0,\dots,S_k=P$ such that $S_{i+1}$ is obtained from $S_i$ by vertex pulling for all $0\leq i\leq k-1$. 
\end{defn}

\begin{prp}{}{}\\
Let $P\subseteq\R^n$ be a stacked polytope. Then the following are true. 
\begin{itemize}
\item $P$ is obtained from $\Delta^{\dim(P)}$ by $\abs{V(P)}-\dim(P)-1$ vertex pullings. 
\item $P$ is simplicial. 
\item The $f$-vector of $P$ is given by $$f_k(P)=\binom{\dim(P)}{k}V(P)-\binom{\dim(P)+1}{k+1}k$$ for $1\leq k\leq\dim(P)-1$. 
\end{itemize}
\end{prp}

\begin{prp}{}{}\\
Let $P\subseteq\R^n$ be a simplicial polytope such that $\dim(P)\geq 2$. Then the following are equivalent. 
\begin{itemize}
\item $P$ is a stacked polytope. 
\item $G^\ast(P)$ is a tree. 
\item If $F\in\text{MF}(\partial P)$ is a missing face, then $\dim(F)=\floor*{\frac{\dim(P)}{2}}$. 
\end{itemize}
\end{prp}

When $\dim(P)=3$, then $P$ is a stacked polytope if and only if $\partial P$ is a flag complex. 

\begin{thm}{Lower Bound Theorem}{}\\
Let $P$ be a simplicial polytope such that $\dim(P)\geq 3$. Suppose that $f_0(P)=n$. Let $Q$ be a stacked polytope with $n$ vertices. Then we have $$f_k(P)\geq f_k(Q)$$ for all $k$. 
\end{thm}

\subsection{Regular Polytopes}
\begin{defn}{Flags of a Polytope}{}\\
Let $P\subseteq\R^n$ be a polytope. A flag of $P$ is a maximal chain in $\mF(P)$. Denote the set of flags of $P$ by $$\text{Flag}(P)=\{\Phi\subseteq\mF(P)\;|\;\Phi\text{ is a flag of }P\}$$
\end{defn}

Since symmetries of a polytope sends faces to faces of the same dimension, we obtain a group action of $\text{Sym}(P)$ on $\text{Flag}(P)$. 

\begin{defn}{Regular Polytopes}{}\\
Let $P\subseteq\R^n$ be a polytope. We say $P$ is a regular polytope if $\text{Sym}(P)$ acts transitively on $\text{Flag}(P)$. 
\end{defn}

\begin{lmm}{}{}|\
Let $P\subseteq\R^n$ be a regular polytope. Then $P^\ast$ is regular. 
\end{lmm}

\begin{eg}{Regular n-Polygons}{}\\
Let $n\geq 3$. Define the regular $n$-gon to be $$P_n=\text{Conv}\left(\left\{\left(\cos\left(\frac{2\pi k}{n}\right),\sin\left(\frac{2\pi k}{n}\right)\right)\;\bigg{|}\;0\leq k\leq n-1\right\}\right)$$ Then there is an isomorphism $$\text{Sym}(P_n)\cong D_{2n}$$
\end{eg}

\begin{prp}{}{}\\
Let $P$ be a regular polytope of dimension $1$. Then $P$ is congruent to $[-1,1]$. 
\end{prp}

\begin{prp}{}{}\\
Let $P$ be a regular polytope of dimension $2$. Then $P$ is congruent to $P_n$ for some $n\geq 3$. 
\end{prp}

\begin{prp}{}{}\\
Let $P$ be a regular polytope of dimension $3$. Then $P$ is congruent to exactly one of the following. 
\begin{itemize}
\item The tetrahedron. 
\item The cube. 
\item The octahedron. 
\item The dodecahedron. 
\item The icosahedron. 
\end{itemize}
\end{prp}

\section{Piecewise Linear Structure}
\subsection{Relation between Polyhedra and Simplicial Complexes}
\begin{defn}{Locally Finite Collections}{}\\
Let $X$ be a space. Let $P\subseteq\mP(X)$ be a collection of subsets. We say that $P$ is locally finite if for each $p\in X$, there exists a neighbourhood $U\subseteq X$ of $p$ such that the set $$\{V\in P\;|\;U\cap V\neq\emptyset\}$$ is finite. 
\end{defn}

\begin{prp}{}{}\\
Let $P$ be a polyhedra. Then $P$ is a locally finite union of simplexes. 
\end{prp}

Thus if $P$ is a polytope, then $P$ has the underlying structure of a simplicial complex. 

\begin{prp}{}{}\\
Let $P\subseteq\R^n$ be a polytope. Then there is a PL homeomorphism $$P\cong_\text{PL}D^{\dim(P)}$$
\end{prp}

\subsection{Piecewise Linear Maps between Polyhedra}
\begin{prp}{}{}\\
Let $P,Q$ be polyhedra. Let $f:P\to Q$ be a function. Then the following are equivalent. 
\begin{itemize}
\item $f$ is piecewise linear. 
\item For each $p\in P$, there exists $L\subseteq P$ such that $f(\lambda p+\mu x)=\lambda f(p)+\mu f(x)$ for all $\lambda,\mu\geq 0$ and $\lambda+\mu=1$ and $x\in L$. 
\item $\Gamma(f)=\{(x,f(x))\;|\;x\in P\}$ is a polyhedron. 
\item There exists a locally finite decomposition into polyhedra $P=\bigcup_{\alpha\in I}P_\alpha$ such that for each $\alpha\in I$, the restriction $f|_{P_\alpha}$ is affine linear. 
\end{itemize}
\end{prp}

\begin{lmm}{}{}\\
Let $P,Q$ be polyhedra. Let $f:P\to Q$ be a piecewise linear function. If $f$ is a homeomorphism, then $f$ is a piecewise linear homeomorphism. 
\end{lmm}

\subsection{Polytopal Spheres}
\begin{defn}{Polytopal Spheres}{}\\
Let $S$ be a simplicial complex. We say that $S$ is a polytopal sphere if $S$ is the boundary of a polytope. 
\end{defn}

Together with results from Simplicial Topology, we have a series of inclusions $$\text{Polytopal Spheres}\subseteq\text{Combinatorial Spheres}\subseteq\text{Simplicial Spheres}$$

When $\dim=2$, all inclusions are equalities. When $\dim=3$, the first inclusion is strict and the second inclusion is an equality. At $\dim=4$, the second inclusion is an open problem, at $\dim\geq 5$, the inclusion is strict. If the number of vertices is $\leq\dim+4$, then all inclusions are equalities.

\subsection{Cones and its Properties}
\begin{defn}{Cones}{}\\
Let $C\subseteq\R^n$. We say that $C$ is a cone if $x\in C$ implies that $sx\in C$ for all $s\geq 0$. 
\end{defn}

\begin{defn}{Polyhedral Cones}{}\\
A cone in $\R^n$ is a subset $C\subseteq\R^n$ such that if $c_1,\dots,c_n\in C$, then $\sum_{i=1}^n\lambda_ic_i\in C$ for all $\lambda_1,\dots,\lambda_n\geq 0$. 
\end{defn}

\begin{defn}{Conical Hull}{}\\
Let $S\subseteq\R^n$ be finite. Define the conical hull of $S$ to be $$\text{Cone}(S)=\left\{\sum_{s\in S}\lambda_ss\;\bigg{|}\;\lambda_s\geq 0\right\}$$
\end{defn}

\begin{prp}{}{}\\
Let $C\subseteq\R^n$ be a subset. Then the following are equivalent. 
\begin{itemize}
\item $C$ is a polyhedral cone. 
\item There exists a finite subset $S\subseteq\R^n$ such that $C=\text{Cone}(S)$. 
\item $C$ is the intersection of finitely many halfspaces containing $0$ in their boundary. 
\end{itemize}
\end{prp}

Therefore cones are also polyhedrons. 

\begin{defn}{Dimension of a Cone}{}\\
Let $C\subseteq\R^n$ be a cone. Define the dimension of $C$ to be the dimension of the smallest affine subspace containing $C$. 
\end{defn}

\begin{defn}{Rational Cones}{}\\
Let $C\subseteq\R^n$ be a cone. We say that $C$ is rational if there exists a finite set $S\subseteq\Z^n$ such that $C=\text{Cone}(S)$. 
\end{defn}

\begin{defn}{Regular Cones}{}\\
Let $C\subseteq\R^n$ be a cone. We say that $C$ is regular if there exists a finite set $S\subseteq\Z^n$ such that the following are true. 
\begin{itemize}
\item $C=\text{Cone}(S)$. 
\item Elements of $S$ are $\Z$-linearly independent. 
\end{itemize}
\end{defn}

\begin{defn}{Simplicial Cones}{}\\
Let $C\subseteq\R^n$ be a cone. We say that $C$ is simplicial if there exists a finite set $S\subseteq\R^n$ such that the following are true. 
\begin{itemize}
\item $C=\text{Cone}(S)$. 
\item Elements of $S$ are $\R$-linearly independent. 
\end{itemize}
\end{defn}

\subsection{The Dual Cone}
\begin{defn}{The Dual of a Cone}{}\\
Let $C\in\R^n$ be a cone. Define the dual of $C$ to be $$C^\vee=\{x\in\R^n\;|\;\langle x,c\rangle\geq 0\text{ for all }c\in C\}$$
\end{defn}

\begin{defn}{Associated Hyperplane and Halfspace of a Linear Functional}{}\\
Let $\phi\in(\R^n)^\ast$. Define the associated hyperplane of $\phi$ to be $$H_\phi=\{x\in\R^n\;|\;\phi(x)=0\}$$ Define the associated halfspace of $\phi$ to be $$H_\phi^+=\{x\in\R^n\;|\;\phi(x)\geq 0\}$$
\end{defn}

\begin{lmm}{}{}\\
Let $C\in\R^n$ be a cone. Let $H$ be a hyperplane in $\R^n$. Then $H$ is a supporting hyperplane of $C$ if and only if there exists $\phi\in C^\vee$ such that $H=H_\phi$. 
\end{lmm}

\begin{lmm}{}{}\\
Let $C\in\R^n$ be a polyhedral cone. Then the following are true. 
\begin{itemize}
\item $C^\vee$ is a polyhedral cone. 
\item $(C^\vee)^\vee=C$. 
\item $C=\bigcap_{i=1}^nH_{\phi_i}$ for $\phi_i\in(\R^n)^\ast$ if and only if $C^\vee=\text{Cone}(\phi_1,\dots,\phi_n)$. 
\item If $C$ is strongly convex, then $C^\vee$ is strongly convex. 
\end{itemize}
\end{lmm}

\begin{lmm}{Separation Lemma}{}\\
Let $C_1,C_2\subseteq\R^n$ be cones such that $F=C_1\cap C_2$ is a face of both $C_1$ and $C_2$. Then there exists $\phi\in C_1^\vee\cap(-C_2)^\vee$ such that $$F=C_1\cap H_\phi=C_2\cap H_\phi$$
\end{lmm}

\begin{defn}{Strongly Convex Cones}{}\\
Let $C\subseteq\R^n$ be a cone. We say that $C$ is strongly convex if $C$ does not contain a line. 
\end{defn}

\begin{lmm}{}{}\\
Let $C\subseteq\R^n$ be a cone. Then the following are equivalent. 
\begin{itemize}
\item $C$ is strongly convex. 
\item $\{0\}$ is a face of $C$. 
\item $C\cap(-C)=\{0\}$. 
\item $\dim(C^\vee)=n$. 
\end{itemize}
\end{lmm}

\subsection{Fans and its Properties}
\begin{defn}{Fans}{}\\
Let $n\in\N^+$. A fan in $\R^n$ is a finite set of cones $\Sigma$ such that the following are true. 
\begin{itemize}
\item Each $C\in\Sigma$ is a strongly convex rational polyhedral cone. 
\item For each cone $C\in\Sigma$ and each face $F$ of $C$, we have $F\in\Sigma$. 
\item For cones $C_1,C_2\in\Sigma$, we have $C_1\cap C_2$ are faces of $C_1$ and $C_2$. 
\end{itemize}
We write $\Sigma(d)=\{C\in\Sigma\;|\;\dim(C)=d\}$ for the $d$-dimensional cones in the fan. 
\end{defn}

\begin{defn}{The Support of a Fan}{}\\
Let $n\in\N^+$. Let $\Sigma$ be a fan in $\R^n$. Define the support of $\Sigma$ to be $$\abs{\Sigma}=\cup_{C\in\Sigma}C$$
\end{defn}
