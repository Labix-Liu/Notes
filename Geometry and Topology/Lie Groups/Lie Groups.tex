\section{Introduction to Lie Groups}
\subsection{Lie Groups}
\begin{defn}{Lie Groups}{} \\A Lie group $G$ is a group $G$ that is also a smooth manifold such that the following are true. 
\begin{itemize}
\item The multiplication map $\cdot:G\times G\to G$ defined by $$(g,h)\mapsto gh$$ is a smooth map of manifolds. 
\item The inverse map $(-)^{-1}:G\times G$ defined by $$g\mapsto g^{-1}$$ is a smooth map of manifolds. 
\end{itemize}
\end{defn}

Some immediate examples of Lie groups include the following: 
\begin{itemize}
\item $(\R^n,+)$ and $(\R^n\setminus\{0\},\times)$. 
\item $U(n), SU(n), O(n), SO(n), SL(n), PSL(n)$ for any $n\in\N$. 
\end{itemize}

\begin{eg}{}{}\\
Let $G$ be a finite group equipped with the discrete topology. Then $G$ is a Lie group. 
\end{eg}

\begin{eg}{}{}\\
Let $n\in\N$. Then $\text{GL}(n,\R)$ is a Lie group. 
\begin{proof}\\
We know that $\text{GL}(n,\R)$ is an open submanifold of $M_n(\R)\cong\R^2$. Multiplication is smooth because multiplication is given by polynomials in each entry, which are smooth. The inverse is smooth since the formula for the inverse consists of determinants of sub matrices, and they are a finite sum and product of terms of the matrices. 
\end{proof}
\end{eg}

\begin{eg}{}{}\\
The following are Lie groups. 
\begin{itemize}
\item $S^1$. 
\item $T^n$ for $n\in\N^+$. 
\end{itemize}
\end{eg}

\begin{defn}{Lie Group Homomorphism}{}\\
Let $G,H$ be Lie groups. Let $\phi:G\to H$ be a function. We say that $f$ is a Lie group homomorphism if the following are true. 
\begin{itemize}
\item $f$ is a group homomorphism. 
\item $f$ is a smooth map. 
\end{itemize}
\end{defn}

\begin{defn}{Lie Group Isomorphisms}{}\\
Let $G,H$ be Lie groups. Let $\phi:G\to H$ be a Lie group homomorphism. We say that $\phi$ is a Lie group isomorphism if there exists a Lie group homomorphism $\psi:H\to G$ such that $$\psi\circ\phi=\text{id}_G\;\;\;\text{ and }\;\;\;\phi\circ\psi=\text{id}_H$$
\end{defn}

\begin{lmm}{}{}\\
Let $G,H$ be Lie groups. Let $\phi:G\to H$ be a Lie group homomorphism. Then $\phi$ is an isomorphism if and only if $\phi$ is bijective. 
\end{lmm}

\begin{defn}{Lie Subgroups}{}\\
Let $G$ be a Lie group. A Lie subgroup of $G$ is a subset $H\subseteq G$ such that the following are true. 
\begin{itemize}
\item $H$ is a Lie group. 
\item The inclusion map $H\hookrightarrow G$ is a Lie group homomorphism. 
\end{itemize}
\end{defn}

\begin{prp}{}{}\\
Let $G$ be a Lie group. Let $H\leq G$ be a subgroup. If $H$ is an embedded submanifold of $G$, then $H$ is a Lie subgroup of $G$. 
\end{prp}

\begin{thm}{Closed Subgroup Theorem}{}\\
Let $G$ be a Lie group. Let $H\leq G$ be a closed subgroup. Then $H$ is a Lie subgroup of $G$. 
\end{thm}

Recall that any open subset of a topological group is closed. Then we have the following. 

\begin{crl}{}{}\\
Let $G$ be a Lie group. Let $H\leq G$ be an open subgroup of $G$. Then $H$ is a Lie subgroup of $G$. 
\end{crl}

\begin{eg}{}{}\\
Let $n\in\N^+$. Then $\text{O}(n,\R),\text{SO}(n,\R),\text{SL}(n,\R)$ are Lie subgroups of $\text{GL}(n,\R)$. 
\end{eg}

\begin{eg}{}{}\\
The following are true regarding the determinant map. 
\begin{itemize}
\item $\det:GL_n(\R)\to\R$ is a Lie group homomorphism. 
\item The map $(\det_\ast)_I:T_I(GL_n(\R))\to T_1(\R)$ is given by $(\det_\ast)_I(X)=\text{tr}(X)$. 
\end{itemize}
\end{eg}

\begin{thm}{The First Isomorphism Theorem for Lie Groups}{}\\
Let $G,H$ be Lie groups. Let $f:G\to H$ be a Lie group homomorphism. If $\im(G)$ is a closed subgroup of $H$, then $f$ induces an isomorphism of Lie groups $$\frac{G}{\ker(f)}\cong\im(G)$$
\end{thm}

Let $G$ be a topological group. Recall that $G^0$ is the connected component of $G$ containing the identity. 

\begin{lmm}{}{}\\
Let $G$ be a Lie group. Then the following are true. 
\begin{itemize}
\item $G^0\trianglelefteq G$ is a normal subgroup. 
\item $G^0$ is a Lie group. 
\item $G/G^0$ is discrete. 
\end{itemize}
\end{lmm}

\subsection{Relation between Lie Groups and Lie Algebras}
For a group $G$, denote the left multiplication map of $h\in G$ by $l_h$. If $G$ is a Lie group, we have seen that $l_h$ is a smooth map, and so it induces a differential $(l_h)_\ast$. 

\begin{defn}{Left Invariant Vector Field}{} \\Let $G$ be a Lie group and $X$ a vector field on $G$. We say that $X$ is left invariant if $$(l_h)_\ast(X_g)=X_{hg}$$ for all $X_g\in T_g(G)$. 
\end{defn}

\begin{prp}{}{} \\Let $G$ be a Lie group. The vector space of left invariant vector fields of $G$ is a Lie algebra of dimension $\dim(G)$. 
\end{prp}

\begin{defn}{Lie Algebra of a Lie Group}{}\\
Let $G$ be a Lie group. Define the Lie algebra of $G$ to be $\R$-vector space $$\text{Lie}(G)=\{X:G\to TG\;|\;X\text{ is a left invariant vector field }\}$$ together with the Lie bracket on vector fields. 
\end{defn}

\begin{prp}{}{}\\
Let $G$ be a Lie group. Then there is an $\R$-vector space isomorphism $$\text{Lie}(G)\cong T_{1_G}(G)$$ given by $X\mapsto X(1_G)$. 
\end{prp}

Recall that given a homomorphism of Lie groups $\phi:G\to H$, it induces a differential $\phi_\ast:T_g(G)\to T_{\phi(g)}(H)$. 

\begin{prp}{}{}\\
Let $G,H$ be Lie groups. Let $f:G\to H$ be a Lie group homomorphism. Then the induced map $$(f_\ast)_{1_G}:T_{1_G}(G)\to T_{1_H}(H)$$ is a Lie algebra homomorphism. 
\end{prp}

In other words, we constructed a functor from Lie groups to Lie algebras sending a Lie group to the tangent space at the identity. 

\begin{eg}{}{}\\
Let $G$ be a closed subgroup of $GL_n(\R)$. Then we have $$\text{Lie}(G)=\{M\in M_n(\R)\;|\;e^{tX}\in G\text{ for all }t\in\R\}$$
\end{eg}

\begin{prp}{}{}\\
Let $G,H$ be Lie groups. Let $f:G\to H$ be a Lie group homomorphism. Then the following are true. 
\begin{itemize}
\item $\text{Lie}(\ker(f))=\ker(df)$. 
\item If $f(G)$ is a closed subgroup, then $\text{Lie}(\im(f))=\im(df)$. 
\end{itemize}
\end{prp}

\subsection{The Exponential Map}
\begin{defn}{One Parameter Subgroups}{}\\
Let $G$ be a Lie group. A one-parameter subgroup of $G$ is a Lie group homomorphism $\gamma:\R\to G$. 
\end{defn}

\begin{lmm}{}{}\\
Let $G$ be a Lie group. Let $X\in T_{1_G}(G)$. Then there exists a unique one-parameter subgroup $\gamma:\R\to G$ of $G$ such that $$X_{\gamma}=X$$
\end{lmm}

\begin{defn}{The Exponential Map}{}\\
Let $G$ be a Lie group. Define the exponential map $\text{exp}:\text{Lie}(G)\to G$ by $$\text{exp}(X)=\gamma(1)$$ where $\gamma:\R\to G$ is the unique one-parameter subgroup of $G$ whose tangent vector at $1_G$ is $X$. 
\end{defn}

\begin{prp}{}{}\\
Let $G,H$ be Lie groups. Let $f:G\to H$ be a Lie group homomorphism. Let $X\in T_{1_G}(G)$. Then we have $$\text{exp}(f_\ast(X))=f(\text{exp}(X))$$
\end{prp}

Recall that a finite dimensional vector space $V$ over $\R$ has the structure of a manifold by identifying $V$ with $\R^n$. Since $\R^n$ is a group, $V$ is a Lie group whose associated Lie algebra is then $\R^n\cong V$. 

\begin{prp}{}{}\\
Let $G$ be a Lie group. Then the following are true. 
\begin{itemize}
\item $\text{exp}:\text{Lie}(G)\to G$ is a smooth map. 
\item $(\text{exp}_\ast)_0:\text{Lie}(G)\to\text{Lie}(G)$ is the identity map. 
\item There exist open subsets $0\in U\subseteq\text{Lie}(G)$ and $1_G\in V\subseteq G$ such that $\text{exp}|_U$ is a diffeomorphism. 
\end{itemize}
\end{prp}

\begin{lmm}{}{}\\
Let $G$ be an abelian Lie group. Then $\text{exp}:\text{Lie}(G)\to G$ is a Lie group homomorphism. 
\end{lmm}

\begin{prp}{}{}\\
Let $G$ be a Lie group. Suppose that one of the following is true. 
\begin{itemize}
\item $G$ is connected and compact. 
\item $G$ is connected and nilpotent. 
\item $G=\text{GL}_n(\R)$. 
\end{itemize}
Then $\text{exp}:\text{Lie}(G)\to G$ is surjective. 
\end{prp}

\subsection{The Correspondence between Lie Groups and Lie Algebras}
\begin{thm}{Lie's Third Theorem}{}\\
Let $\mathfrak{g}$ be a finite dimensional Lie algebra. Then there exists a simple connected Lie group $G$ such that $\text{Lie}(G)\cong\mathfrak{g}$. 
\end{thm}

\begin{thm}{The Homomorphisms Theorem}{}\\
Let $G,H$ be Lie groups such that $G$ is simple connected. Let $\phi:\text{Lie}(G)\to\text{Lie}(H)$ be a Lie algebra homomrphisms. Then there exists a unique Lie group homomorphism $f:G\to H$ such that $\phi=df$. 
\end{thm}

Abstract nonsense: The functor $F$ from the category of connected Lie groups to the category of finite dimensional Lie algebras admit a left adjoint $\Gamma$ whose unit map $\mathfrak{g}\to F(\Gamma(\mathfrak{g}))$ is an isomorphism, and whose counit map $\Gamma(F(G))\to G$ is the universal covering space for $G$. 

\begin{thm}{The Subgroups-Subalgebras Theorem}{}\\
Let $G$ be a Lie groups. Let $\mathfrak{h}\leq\text{Lie}(G)$ be a Lie subalgebra. Then there exists a unique connected Lie subgroup $H\leq G$ such that $\mathfrak{H}=\text{Lie}(H)$. 
\end{thm}

\subsection{Smooth Actions}
\begin{defn}{}{}\\
Let $G$ be a Lie group. Let $M$ be a smooth manifold. An action of $G$ on $M$ is a function $\cdot:G\times M\to M$ such that the following are true. 
\begin{itemize}
\item $G$ acts on $M$ as a group. 
\item $\cdot$ is a smooth map. 
\end{itemize}
\end{defn}

If $G$ is acts on a manifold $M$ as a Lie group, then in particular, it is also a topological group acting on a topological space continuously.

\begin{prp}{}{}\\
Let $G$ be a connected abelian Lie group. Then $\text{exp}:\text{Lie}(G)\cong\R^{\dim(G)}\to G$ is the universal covering space of $G$. 
\end{prp}

\begin{prp}{}{}\\
Let $G$ be a connected abelian Lie group. Then there is a Lie group isomorphism $$G\cong T^k\times\R^{\dim(G)-k}$$ for some $k\in\N$. 
\end{prp}

\subsection{Properties of Tori}
\begin{crl}{}{}\\
Let $G$ be a compact connected abelian Lie group. Then there is a Lie group isomorphism $G\cong T^{\dim(G)}$. 
\end{crl}

\begin{defn}{Generator of a Torus}{}\\
Let $t\in T^k$. We say that $t$ is a generator of $T^k$ if the smallest closed subgroup containing $t$ is $T^k$. 
\end{defn}

\begin{prp}{}{}\\
Every torus admits a generator. 
\end{prp}

\begin{prp}{}{}\\
Let $t=[(t_1,\dots,t_k)]\in(\R/\Z)^k$. Then $t$ is a generator if and only if $1,t_1,\dots,t_k$ are linearly independent over $\Q$. 
\end{prp}

\subsection{Maximal Tori}
\begin{defn}{Maximal Torus}{}\\
Let $G$ be a compact Lie group. A maximal torus of $G$ is a Lie subgroup $T\leq G$ such that the following are true. 
\begin{itemize}
\item $T$ is a torus. 
\item $T$ is not properly contained in a larger Lie subgroup of $G$ that is a torus. 
\end{itemize}
\end{defn}

\begin{prp}{}{}\\
Let $G$ be a compact connected Lie group. Let $T$ be a maximal torus of $G$. Then the following are true. 
\begin{itemize}
\item If $T'\leq G$ is another maximal torus of $G$, then $T$ and $T'$ are conjugates in $G$. 
\item We have $$G=\bigcup_{T\text{ is a maximal torus of }G}T$$
\item We have $$Z(G)=\bigcap_{T\text{ is a maximal torus of }G}T$$
\end{itemize}
\end{prp}

\section{Homotopy Theory of Lie Groups}
\begin{prp}{}{}\\
Let $G$ be a connected Lie group. Then the following are true. 
\begin{itemize}
\item $G$ admits a universal covering space $\widetilde{G}$ that is a smooth manifold. 
\item The structure map $p:\widetilde{G}\to G$ is a morphism of Lie groups. 
\item We have $\pi_1(G)\cong\ker(p)$. 
\end{itemize}
\end{prp}

\begin{eg}{}{}\\
The universal covering space of $T^n$ is $\R^n$, which is also a Lie group and whose covering map is a Lie group homomorphism. 
\end{eg}

\begin{eg}{}{}\\
The map $\text{exp}:\R\to\R^\ast$ given by $t\mapsto e^t$ is a universal covering map of $\R^\ast$. 
\end{eg}
