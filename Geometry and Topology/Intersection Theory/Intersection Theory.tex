\section{Intersection Theory}
\subsection{The Divisor of a Rational Function}
\begin{defn}{The Order of a Function at a Subvariety}{}\\
Let $k$ be a field. Let $X$ be a scheme of finite type over $k$. Let $W$ be an irreducible subvariety of $X$. Let $f/g\in K(W)^\ast$ for $f,g\in\mO_X(W)$ and $g\neq 0$. Let $V\subseteq W$ be an irreducible subvariety of codimension $1$. Define the order of $f$ at $V$ to be $$\text{ord}_V(f/g)=l_{\mO_X(W)}\left(\frac{\mO_X(W)}{(f)}\right)-l_{\mO_X(W)}\left(\frac{\mO_X(W)}{(g)}\right)$$
\end{defn}

\begin{defn}{The Divisor of a Rational Function}{}\\
Let $k$ be a field. Let $X$ be a scheme of finite type over $k$. Let $W$ be an irreducible subvariety of $X$. Let $f\in K(W)^\ast$. Define the divisor of $f$ to be $$\text{div}(f)=\sum_{\substack{V\text{ is a codimension }1\\\text{irreducible subvariety of }W}}\text{ord}_V(f)\cdot V$$
\end{defn}

\subsection{The Chow Groups}
\begin{defn}{The Group of $n$-Cycles}{}\\
Let $k$ be a field. Let $X$ be a scheme of finite type over $k$. Let $n\in\N$. Define the group of $n$-cycles on $X$ to be the free abelian group $$Z_n(X)=\Z\langle V\;|\;\substack{V\text{ is an }n\text{-dimensional}\\\text{irreducible subvariety of }X}\rangle$$
\end{defn}

\begin{lmm}{}{}\\
Let $k$ be a field. Let $X$ be a scheme of finite type over $k$. Let $W$ be an irreducible subvariety of $X$. Let $f\in K(W)^\ast$. Then $\text{div}(f)\in Z_{\dim(W)-1}(X)$. 
\end{lmm}

\begin{defn}{Cycles Rationally Equivalent to Zero}{}\\
Let $k$ be a field. Let $X$ be a scheme of finite type over $k$. Let $V\in Z_n(X)$. We say that $V$ is rationally equivalent to $0$ if there exists finitely many $n+1$-dimensional irreducible subvarieties $W_1,\dots,W_r$ of $X$ and $f_i\in K(W_i)^\ast$ such that $$V=\sum_{i=1}^n\text{div}(f_i)$$
\end{defn}

\begin{prp}{}{}\\
Let $k$ be a field. Let $X$ be a scheme of finite type over $k$. Let $V\in Z_n(X)$ be a cycle. Then $V$ is rationally equivalent to $0$ i and only if there exists $(n+1)$-dimensional irreducible subvarieties $W_1,\dots,W_r$ of $X\times\Prj_k^1$ such that the projection maps $p_i:W_i\to\Prj_k^1$ are dominant and $V=\sum_{i=1}^r[p_i^{-1}(0)]-[p_i^{-1}(\infty)]$
\end{prp}

\begin{defn}{The Chow Group}{}\\
Let $k$ be a field. Let $X$ be a scheme of finite type over $k$. Define the $n$th Chow group of $X$ to be $$\text{CH}_n(X)=\frac{Z_n(X)}{\sim}$$ where we say that $V_1\sim V_2$ if $V_1-V_2$ is rationally equivalent to $0$. In this case, we say that $V_1$ and $V_2$ are rationally equivalent. 
\end{defn}

\subsection{Pushfoward and Pullback of Cycles}
\begin{defn}{The Push Forward Map}{}\\
Let $k$ be a field. Let $X,Y$ be schemes of finite type over $k$. Define the push forward map $f_\ast:Z_n(X)\to Z_n(Y)$ by $$f_\ast(V)=\deg(V/f(V))f(V)$$ where $$\deg(V/f(V))=\begin{cases}
[k(V):k(f(V))] & \text{ if }\dim(V)=\dim(f(V))\\
0 & \text{ otherwise }
\end{cases}$$
\end{defn}

\begin{prp}{}{}\\
Let $k$ be a field. Let $X,Y$ be schemes of finite type over $k$. Then the push forward map descends to a well defined homomorphism $f_\ast:\text{CH}_n(X)\to\text{CH}_n(Y)$. 
\end{prp}

\begin{prp}{}{}\\
Let $k$ be a field. Let $X$ be a scheme of finite type over $k$. Let $i:Y\hookrightarrow X$ be a closed subscheme of $X$. Let $j:X\setminus Y\hookrightarrow X$ be the inclusion map. Then there is an exact sequence \\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	{\text{CH}_n(Y)} & {\text{CH}_n(X)} & {\text{CH}_n(X\setminus Y)} & 0
	\arrow["{i_\ast}", from=1-1, to=1-2]
	\arrow["{j\ast}", from=1-2, to=1-3]
	\arrow[from=1-3, to=1-4]
\end{tikzcd}} \\
\end{prp}
