\section{Spectra and its Structure Sheaf}
\subsection{Spectra and its Topology}
Recall that for $A$ a commutative ring, we defined $$\text{Spec}(A)=\{P\subseteq A\;|\;P\text{ is a prime ideal of }A\}$$ We can similarly define the notion of vanishing loci of $\text{Spec}(A)$. 

\begin{defn}{Zero Locus}{} Let $A$ be a commutative ring. Let $T\subseteq A$. Define the vanishing locus of $T$ to be $$\V^\text{S}(T)=\{p\in\text{Spec}(A)\;|\;T\subseteq p\}$$
\end{defn}

\begin{prp}{}{} Let $A$ be a commutative ring. Let $T\subseteq A$ be a subset. Let $I=(T)$ be the ideal generated by $T$. Then $$\V^\text{S}(T)=\V^\text{S}(I)$$
\end{prp}

The proposition shows that we only need to concern ourselves with the zero set of ideals of $A$. 

\begin{lmm}{}{} Let $A$ be a commutative ring. The following are true. 
\begin{itemize}
\item $\V^\text{S}(1)=\emptyset$
\item $\V^\text{S}(0)=\text{Spec}(A)$
\item Let $\{a_i|i\in I\}$ be a countable set of ideals of $A$, then $$\V^\text{S}\left(\sum_{i\in I}a_i\right)=\bigcap_{i\in I}\V^\text{S}(a_i)$$
\item Let $\{a_1,\dots,a_n\}$ be a finite set of ideals of $A$, then $$\V^\text{S}\left(\bigcap_{k=1}^na_k\right)=\bigcup_{k=1}^n\V^\text{S}(a_k)$$
\end{itemize}
\end{lmm}

\begin{defn}{Zariski Topology on Spec}{} Let $A$ be a commutative ring. Define the Zariski topology on $\text{Spec}(A)$ to be the topology where the closed sets are exactly sets of the form $\V^\text{S}(I)$ for $I\subseteq A$ an ideal of $A$. 
\end{defn}

\begin{eg}{}{} The Zariski topology on $\text{Spec}(\Z)$ is given as follows. 
\begin{itemize}
\item The points are given by $\text{Spec}(\Z)=\{(p)\;|\;p\text{ is a prime }\}\cup\{(0)\}$
\item The closed subsets are of the form $$\V^s((n))=\{(p)\in\text{Spec}(\Z)\;|\;p\text{ divides }n\}$$
\end{itemize}
\end{eg}

\begin{eg}{}{} The Zariski topology on $\text{Spec}(\C[x])$ is given as follows. 
\begin{itemize}
\item The points are given by $\text{Spec}(\C[x])=\{(x-a)\;|\;a\in\C\}\cup\{(0)\}$
\item The closed subsets are precisely of the form $$\V^s((p))=\{(x-a)\;|\;a\text{ is a root of }p\}$$ for $p\in\C[x]$ since $\C[x]$ is a PID. 
\end{itemize}
\end{eg}

\begin{lmm}{}{} Let $k$ be an algebraically closed field. Let $V$ be an affine variety. Then the bijection given by the Hilbert's nullstellensatz $$V\cong\text{maxSpec}(k[V])$$ is a homeomorphism where $\text{maxSpec}(k[V])$ inherits the subspace topology of $\text{Spec}(k[V])$. 
\end{lmm}

Let $k$ be an algebraically closed field. For a closed subset $V\subseteq\A_k^n$, we have the following. 

\begin{itemize}
\item There is a bijection $$V\;\;\overset{1:1}{\longleftrightarrow}\;\;\text{maxSpec}(V)\subseteq\text{Spec}(V)$$ The geometry of $V$ is enhanced because we have more points in general, given by the irreducible subvarieties. 
\item In the Zariski topology of $V$, closed subsets are affine subvarieties of $V$. In the Zariski topology of $\text{Spec}(V)$, closed subsets are collections of irreducible subvarieties. 
\item For a closed subset $V$ of $\text{Spec}(k[V])$, we have $C=\V^s(I)$ for some $I\subseteq k[V]$ an ideal. There is a bijection of sets $$C=\V^s(I)\;\;\overset{1:1}{\longleftrightarrow}\;\;\{W\subseteq\V(I)\;|\;W\text{ is irreducible }\}$$
\item In particular, for an irreducible polynomial $f\in k[x_1,\dots,x_n]$, $(f)\in\text{Spec}(k[V])$ is a point. The closure of $(f)$ is given by $$\overline{(f)}=\{m\in\text{maxSpec}(k[V])\;|\;(f)\subseteq m\}\cup\{(f)\}$$
\end{itemize}

\begin{prp}{}{} Let $A$ be a commutative ring. Let $I\subseteq A$ be an ideal of $A$. Then $$\text{Spec}(A/I)=\V^\text{S}(I)$$ where equal refers to the two spaces are equal. 
\end{prp}

We can also explicitly write out the open sets and a basis for the Zariski topology. 

\begin{defn}{Distinguished Open Sets}{} Let $A$ be a commutative ring. Let $S\subseteq A$. Define the distinguished open set of $S$ to be $$D(S)=\{p\in\text{Spec}(A)|S\not\subseteq p\}$$ Let $f\in A$. Then the collection $$D(f)=\{p\in\text{Spec}(A)|f\notin p\}$$ for $f$ varying in $A$ are called basic open sets. 
\end{defn}

Notice that from the definition we can directly see that $\V(S)$ and $D(S)$ partitions $\text{Spec}(A)$ for every $S\subset A$. Moreover, if $S$ generates the ideal $a$, $\V(a)=\V(M)$ hence we will only feed in ideals of $A$ into $\V(-)$ from now on. 

\begin{prp}{}{}\\
Let $R$ be a commutative ring. Then the following are true. 
\begin{itemize}
\item The open sets of $\text{Spec}(R)$ are of the form $D(S)$ for $S\subseteq R$. 
\item The basic open sets form a basis for the Zariski topology. 
\end{itemize}
\end{prp}

\subsection{Hilbert's Nullstellensatz for Spec}
\begin{defn}{Ideals from a Zero Locus}{} Let $A$ be a commutative ring. Let $V\subseteq\text{Spec}(R)$. Define $$\I(V)=\{f\in A\;|\;f\in p\text{ for all }p\in V\}$$
\end{defn}

\begin{thm}{Scheme-theoretic Nullstellensatz}{} Let $A$ be a commutative ring. Let $J$ be an ideal of $A$. Then $\I(\V(J))=\sqrt{J}$. 
\end{thm}

\begin{prp}{}{} Let $A$ be a commutative ring. Then $\V^\text{S}(-)$ and $\I^\text{S}(-)$ induce an inclusion reversing bijection $$\left\{\substack{\text{Radical ideals}\\\text{of }A}\right\}\;\;\overset{\text{1:1}}{\longleftrightarrow}\;\;\left\{\substack{\text{Closed Subsets of}\\\text{Spec}(A)}\right\}$$
\end{prp}

\begin{prp}{}{} Let $A$ be a commutative ring. Then $\V^\text{S}(-)$ and $\I^\text{S}(-)$ induces a bijection $$\left\{\substack{\text{Irreducible Components}\\ \text{of Spec}(A)}\right\}\;\;\overset{\text{1:1}}{\longleftrightarrow}\;\;\left\{\substack{\text{Minimal prime}\\\text{ideals of }A}\right\}$$
\end{prp}

\subsection{The Structure Sheaf of a Spectrum}
We now define the structure sheaf on a spectrum so that they form a ringed space. 

\begin{defn}{Structure Sheaf}{} Let $A$ be a commutative ring and $\text{Spec}(A)$ the spectrum of $A$ as a topological space. Define the structure sheaf on $\text{Spec}(A)$ to be the functor $\mO_{\text{Spec}(A)}:\bold{Open}(\text{Spec}(A))\to\bold{Rings}$ defined as follows. 
\begin{itemize}
\item For each $U\subseteq X$ open, define $$\mO_{\text{Spec}(A)}(U)=\left\{s:U\to\coprod_{p\in U}A_p\;\bigg{|}\;\substack{\forall p\in U,\;s(p)\in A_p\text{ and }\\\exists U_p\subset U\text{ s.t. }q\in U_p\text{ implies }s(q)\in A_p}\right\}$$
\item For $V\subseteq U$ an inclusion, the unique morphism $\mO_{\text{Spec}(A)}(U)\to\mO_{\text{Spec}(A)}(V)$ sends $s\in\mO_{\text{Spec}(A)}(U)$ to the restriction $$s|_V:V\to\coprod_{p\in V}A_p$$
\end{itemize}
\end{defn}

Note that each $s$ as a function from $U$ simply means that $s$ is indexed by $U\subseteq\text{Spec}(A)$. Alternatively we can write each element of $\mO(U)$ as $s=(s_p)_{p\in U}$ such that $s_p\in A_p$. 

\begin{thm}{}{} Let $A$ be a commutative ring. Then the structure sheaf $$\mO_{\text{Spec}(A)}:\bold{Open}(\text{Spec}(A))\to\bold{Set}$$ defined above is indeed a sheaf on $\text{Spec}(A)$. 
\end{thm}

The structure sheaf allows $\text{Spec}(A)$ to be a ringed space. The ringed space on any spectrum is in fact a locally ringed space. But this is not true for the ringed space on varieties in the classical sense. 

\begin{prp}{}{} Let $A$ be a commutative ring. Then the following are true regarding the ringed space $(\text{Spec}(A),\mO_{\text{Spec}(A)})$. 
\begin{itemize}
\item For any $p\in\text{Spec}(A)$, there is an isomorphism $\mO_{\text{Spec}(A),p}\cong A_p$ on the level of stalks. 
\item $(\text{Spec}(A),\mO_{\text{Spec}(A)})$ is a locally ringed space. 
\item For any element $f\in A$, there is an isomorphism $\mO_{\text{Spec}(A)}(D(f))\cong A_f$
\item There is an isomorphism $\mO_{\text{Spec}(A)}(\text{Spec}(A))\cong A$ on the global level. 
\end{itemize} 
\begin{proof}~\\
\begin{itemize}
\item Define a homomorphism $\phi_p:\mO_{\text{Spec}(A),p}\to A_p$ as follows. For $s\in\mO_{\text{Spec}(A),p}$ a local section in a neighbourhood of $p$ to $s(p)\in A_p$. This is well defined: If $(U,s)\sim(V,t)$, then there exists a neighbourhood $W\subseteq U\cap V$ of $p$ such that $s|_W=t|_W$. Hence $s(p)=t(p)$. It is clear that $\phi_p$ is a ring homomorphism by definition of the sheaf. It remains to show that $\phi_p$ is a bijection. \\

Assume that $a/f\in A_p$. Then $D(f)$ is an open neighbourhood of $p$ and $a/f$ becomes a section in $\mO_{\text{Spec}(A)}(D(f))$. Hence $\phi_p$ is surjective. Now suppose that $s$ and $t$ be two local sections in $\mO_{\text{Spec}(A),p}$ such that $s(p)=t(p)$. Assume that $s$ is local on $U$ and $t$ is local on $V$, then $s(p)=a/f$ and $t(p)=b/g$ in $W\subseteq U\cap V$ for some $a,b\in A$ and $f,g\in A\setminus p$. Since $s(p)=t(p)$, we conclude that there exists $h\in A\setminus p$ such that $h(ag-bf)=0$. For any $q\in D(f)\cap D(g)\cap D(h)$, $h(ag-bf)=0$ still holds in $A_q$ hence $a/f=b/g$ in $D(f)\cap D(g)\cap D(h)$, which is a neighbourhood of $p$. Hence $s=t$ in a neighbourhood of $p$. Thus $s$ and $t$ have the same stalk. Thus $\phi_p$ is injective. 

\item From the above we immediately conclude that every stalk of the ringed space is a local ring. Hence $(\text{Spec}(A),\mO_{\text{Spec}(A)})$ is a locally ringed space. 

\item Define a map $\phi:A_f\to\mO_{\text{Spec}(A)}(D(f))$ by sending $a/f^n\in A_f$ to $\left(s:D(f)\to\coprod_{p\in D(f)}A_p\right)\in\mO_{\text{Spec}(A)}(D(f))$ that assigns each $p\in D(f)$ to $a/f^n\in A_p$. This makes sense since $p\in D(f)$ implies $f\notin p$ so that $a/f^n\in A_p$. It is clear that this is a ring homomorphism. It remains to show that $\phi_p$ is a bijection. \\

Suppose that $\phi(a/f^n)=\phi(b/f^m)$. For each $p\in D(f)$, $\phi(a/f^n)(p)=\phi(b/f^m)(p)$ implies that $a/f^n=b/f^m$ hence there exists some $h\in A$ such that $h(f^ma-f^nb)$. Notice that the annihilator $\text{Ann}_A(f^ma-f^nb)$ is such that $h$ lies in it. Since $h\notin p$, we have that $\text{Ann}_A(f^ma-f^nb)$ is not a subset of $p$. This is true for any $p\in D(f)$ hence $V(\text{Ann}_A(f^ma-f^nb))\cap D(f)=\emptyset$. We conclude that $f\in\sqrt{\text{Ann}_A(f^ma-f^nb)}$ so $f^l(f^ma-f^nb)=0$ for some $l$. Since $f$ is invertible in $A_f$, we can multiply the inverse on both sides to obtain $a/f^n=b/f^m$ and so $\phi$ is injective. \\

Let $s\in\mO_{\text{Spec}(A)}(D(f))$. 

\item Using the above applied to $f=1_A$, we conclude that $\mO_{\text{Spec}(A)}(\text{Spec}(A))\cong A$
\end{itemize}
\end{proof}
\end{prp}

\subsection{The Spec Functor}
\begin{defn}{Induced Map on Spectrum}{} Let $R,S$ be commutative rings. Let $\phi:R\to S$ be a ring homomorphism. Define the function $$\phi^\flat:\text{Spec}(S)\to\text{Spec}(R)$$ by the formula $P\mapsto\phi^{-1}(P)$
\end{defn}

\begin{prp}{}{} Let $A,B,C$ be commutative rings. Let $\phi:A\to B$ and $\psi:B\to C$ be ring homomorphisms. Then the following are true. 
\begin{itemize}
\item $(\psi\circ\phi)^\flat=\phi^\flat\circ\psi^\flat$
\item $(\text{id}_A)^\flat=\text{id}_{\text{Spec}(A)}$
\end{itemize}
\end{prp}

\begin{lmm}{}{}  Let $R,S$ be commutative rings. Let $\phi:R\to S$ be a ring homomorphism.  Then the induced map $$\phi^\flat:\text{Spec}(S)\to\text{Spec}(R)$$ is continuous. 
\end{lmm}

\begin{prp}{}{} Let $V,W$ be affine varieties over $\C$. Let $F:V\to W$ be a morphism of affine varieties. Let $m_x$ be the maximal ideal of $\C[V]$ corresponding to the point $x$. Then $$(F^\ast)^\flat(m_x)=m_{F(x)}$$
\end{prp}

We thus now have the following nice diagram: \\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	{\text{maxSpec}(\C[V])} & & {\text{maxSpec}(\C[W])} \\
	V & & W
	\arrow["{\phi^\flat|_{\text{maxSpec}(\C[V])}}", from=1-1, to=1-3]
	\arrow["{1:1}"', leftrightarrow, from=1-1, to=2-1]
	\arrow["{1:1}", leftrightarrow, from=1-3, to=2-3]
	\arrow["\phi"', from=2-1, to=2-3]
\end{tikzcd}}\\

\begin{prp}{}{} Let $V$ be an affine variety over $\C$. Let $X\subseteq V$ be an affine irreducible subvariety. Let $P_X$ be the prime ideal of $\C[V]$ corresponding to $X$. Let $Y=\overline{F(X)}$. Then $$(F^\ast)^\flat(P_X)=P_Y$$
\end{prp}

\begin{eg}{}{} Let $\sigma:\C[x]\to\C[[t]]$ be a map. Let $\iota:\C[[t]]\to\C((t))$ be the inclusion map. Consider the following diagram: \\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	{\C[x]} && {\C((t))} \\
	& {\C[[t]]} \\
	\begin{array}{c} \substack{\text{Spec}(\C[x])=\\\{(0)\}\cup\{(x-a)\;|\;a\in\C\}} \end{array} && {\text{Spec}(\C((t)))=\{(0)\}} \\
	& {\text{Spec}(\C[[t]])=\{(0),(t)\}}
	\arrow["\sigma", from=1-1, to=1-3]
	\arrow["\iota"', hook, from=2-2, to=1-3]
	\arrow["{\sigma^\flat}"', from=3-3, to=3-1]
	\arrow["{\iota^\flat}", from=3-3, to=4-2]
	\arrow["\rho", dashed, from=4-2, to=3-1]
\end{tikzcd}}\\
\begin{itemize}
\item Let $\sigma$ be the map $x\mapsto t$. Then $\rho$ exists. 
\item Let $\sigma$ be the map $x\mapsto t+1$. Then $\rho$ exists. 
\item Let $\sigma$ be the map $x\mapsto 1/t$. Then $\rho$ does not exist. 
\item Let $\sigma$ be the map $x\mapsto t^2$. Then $\rho$ exists. 
\item Let $\sigma$ be the map $x\mapsto e^t$. Then $\rho$ exists. 
\end{itemize} 
\begin{proof}
We know that the $\text{Spec}$ functor is fully faithful. This means that $\rho$ exists if and only if there exists a map $\C[x]\to\C[[t]]$ making the top diagram commute. \\

For the first two cases, the images of the map lie inside $\C[[t]]$. Hence the map $\C[x]\to\C[[t]]$ exists, and hence $\rho$ exists. In the first case, we have $\rho((t))=(x)$ and in the second case we have $\rho((t))=(x-1)$. \\

For the third case, no map makes the top diagram commute hence $\rho$ does not exist. \\

For the fourth and fifth cases, the images of the map lie inside $\C[[t]]$. Hence $\rho$ exists. In the fourth case we have $\rho((t))=(x)$ and in the fifth case we have $\rho((t))=(x-1)$. Indeed, $f\in\rho^{-1}(t)$ if and only if $t$ divides $f(e^t)$. This means that $g(t)=f(e^t)$ has no constant term, or that $g(0)=0$. This means that $f(1)=0$, hence $x-1$ divides $f$. 
\end{proof}
\end{eg}

\begin{defn}{The Spec Functor}{} Define the spec functor $$\text{Spec}:\bold{CRing}\to\bold{LRS}$$ to consist of the following data. 
\begin{itemize}
\item A commutative ring $A$ is sent to the locally ringed space $(\text{Spec}(A),\mO_{\text{Spec}(A)})$. 
\item For a ring homomorphism $\phi:A\to B$, we have that $\text{Spec}(\phi)$ consists of a continuous map $\phi^\flat:\text{Spec}(B)\to\text{Spec}(A)$ and a morphism of sheaves $$(\phi^\flat)^\#:\mO_{\text{Spec}(B)}\to(\phi^\flat)_\ast\mO_{\text{Spec}(A)}$$ defined as follows. 
Fixing $p\in\text{Spec}(A)$, for each $q\in(\phi^\flat)^{-1}(p)$, $\phi$ induces a local homomorphism $\phi_{p,q}:A_p\to B_q$. Ranging over $q$, we obtain a map $$\phi_p:A_p\to\coprod_{q\in(\phi^\flat)^{-1}(p)}B_q$$ For $U\subseteq\text{Spec}(B)$ an open set, ranging over $p\in U$ gives a map $$\coprod_{p\in U}\phi_p:\coprod_{p\in U}A_p\to\coprod_{q\in(\phi^\flat)^{-1}(U)}B_q$$ The morpnism then sends $f\in\mO_{\text{Spec}(B)}(U)$ to the composite \\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	{(\phi^\flat)^{-1}(U)} & U & {\coprod_{p\in U}A_p} && {\coprod_{q\in(\phi^\flat)^{-1}(U)}B_q}
	\arrow["{\phi^\flat}", from=1-1, to=1-2]
	\arrow["f", from=1-2, to=1-3]
	\arrow["{\coprod_{p\in U}\phi_p}", from=1-3, to=1-5]
\end{tikzcd}} \\
\end{itemize}
\end{defn}

\begin{prp}{}{}\\
The spec functor is fully faithful. 
\end{prp}

\subsection{The Associated Sheaf of Modules}
We restate the definition here of a sheaf of modules here. Let $\mA$ be a sheaf of rings over $X$. Let $U$ be an open set of $X$. A sheaf of $\mA$-modules over $X$ is a sheaf $\mF$ such that each $\mF(U)$ is an $\mA(U)$-modules. Moreover, for each inclusion of open sets $V\subseteq U$, the restriction homomorphism $\mathcal{F}(U)\to\mathcal{F}(V)$ is such that the following diagram commutes: \\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
\mA(U)\times\mF(U)\arrow[r, "\text{action}"]\arrow[d, "\text{res}_{U,V}\times\text{res}_{U,V}"'] &\mF(U)\arrow[d, "\text{res}_{U,V}"]\\
\mA(V)\times\mF(V)\arrow[r, "\text{action}"] & \mF(V)
\end{tikzcd}}\\

For a scheme $(X,\mO_X)$ that is also a locally ringed space, we denote the category of sheaves of $\mO_X$-modules as $\text{Mod}_{\mO_X}$. \\

It is the analogue of a module over a ring for the following reason. 

\begin{defn}{Associated Sheaf}{} Let $A$ be a commutative ring. Let $M$ be an $A$-module. Define a sheaf $\widetilde{M}$ on $\text{Spec}(A)$ as follows. 
\begin{itemize}
\item For each open set $U\subseteq\text{Spec}(A)$, define $$\widetilde{M}(U)=\left\{s:U\to\coprod_{p\in U}M_p\;\bigg{|}\;\substack{\forall p\in U,\;s(p)\in M_p\text{ and }\exists\;U_p\subseteq U\text{ s.t. }\\q\in V\text{ implies }s(q)=\frac{m}{f}\in M_q\text{ for }f\in A,m\in M}\right\}$$
\item For $V\subseteq U$ an inclusion, define the unique morphism $\widetilde{M}(U)\to\widetilde{M}(V)$ given by restriction. 
\end{itemize}
\end{defn}

\begin{lmm}{}{} Let $M$ be an $A$-module. Then the associated sheaf is a sheaf of $\mathcal{O}_{\text{Spec}(A)}$-modules. 
\end{lmm}

\begin{prp}{}{} Let $M$ be an $A$-module. Then the following are true regarding the associated sheaf $\widetilde{M}$. 
\begin{itemize}
\item For each $p\in\text{Spec}(A)$, there is an isomorphism $\widetilde{M}_p\cong M_p$
\item For any $f\in A$, there is an isomorphism $\widetilde{M}(D(f))\cong M_f$ of $A_f$-modules

\end{itemize}
\end{prp}

\begin{itemize}

\end{itemize}

\end{prp}

Recall that sheaf $\mF$ of $\mO_X$-modules is quasicoherent if for all $p\in X$, there exists an open neighbourhood $U\subseteq X$ such that there is an exact sequence: \\
\adjustbox{scale=1.0,center}{\begin{tikzcd}

\begin{itemize}
\item $\mF$ is a sheaf of finite type. 
\item For any $U\subseteq X$ and any morphism $$\varphi:\mO_X^{\otimes n}|_U\to\mF|_U$$ of $\mO_X$-modules, then kernel of $\varphi$ is a sheaf of finite type. 
\end{itemize}

\subsection{Projective Spectra and its Topology}
\begin{defn}{The Projective Spectrum}{} Let $S$ be a commutative graded ring. Define the projective spectrum of $S$ to be $$\text{Proj}(S)=\{P\in\text{Spec}(S)\;|\;P\text{ is homogeneous and }S_+\not\subseteq P\}$$ where $S_+$ is the irrelevant ideal. 
\end{defn}

\begin{defn}{The Homogeneous Vanishing Set}{}\\
Let $S$ be a commutative graded ring. Let $I$ be a homogeneous ideal of $S$. Define the vanishing set of $I$ to be $$\V^H(I)=\{P\in\text{Proj}(S)\;|\;I\subseteq S\}$$
\end{defn}

\begin{prp}{}{}\\
Let $S$ be a commutative graded ring. Then the following are true. 
\begin{itemize}
\item Let $\{I_j\;|\;j\in J\}$ be a set of homogeneous ideals of $S$. Then we have $$\V^H\left(\sum_{j\in J}I_j\right)=\bigcap_{j\in J}\V^H(I_j)$$
\item Let $I_1,I_2$ be homogeneous ideals of $S$. Then we have $$\V^H(I_1I_2)=\V^H(I_1)\cup\V^H(I_2)$$
\end{itemize}
\end{prp}

Similar to that of $\text{Spec}(A)$ we can endow a topology on $\text{Proj}(S)$. 

\begin{defn}{Zariski Topology}{}\\
Let $S$ be a commutative graded ring. Define the Zariski topology on $\text{Proj}(S)$ to be the topology where the closed sets are exactly sets of the form $\V^H(I)$ for $I\subseteq S$ a homogeneous ideal. 
\end{defn}

\begin{defn}{Projective Space over a Ring}{}\\
Let $R$ be a commutative ring. Define the projective $n$ space over $A$ to be $$\Prj_A^n=\text{Proj}(A[x_0,\dots,x_n])$$
\end{defn}

\subsection{The Structure Sheaf on a Projective Spectrum}
\begin{defn}{The Structure Sheaf on Projective Spectra}{} Let $S$ be a graded ring. Construct $\mathcal{O}_{\text{Proj}(S)}:\bold{Open}(\text{Proj}(S))\to\bold{Rings}$ as follows. 
\begin{itemize}
\item For $U\subseteq\text{Proj}(S)$ an open set, define $$\mO_{\text{Proj}(S)}(U)=\left\{s:U\to\coprod_{p\in U}S_{(p)}\;\bigg{|}\;\substack{\forall p\in U,\;s(p)\in S_{(p)}\text{ and }\exists\;U_p\subseteq U\\\text{ s.t. }q\in U_p\text{ implies }s(q)=a/f\in S_{(q)}\text{ for }a\text{ and }f\text{ homogenous}}\right\}$$
\item For $V\subseteq U$ the inclusion, define the unique map $\mO_{\text{Proj}(S)}(U)\to\mO_{\text{Proj}(S)}(V)$ by the restriction of elements. 
\end{itemize}
Then $\mO_{\text{Proj}(S)}$ is a sheaf on $S$. 
\end{defn}

\begin{prp}{}{}\\
Let $S$ be a commutative graded ring. Then the following are true. 
\begin{itemize}
\item For any $p\in\text{Proj}(S)$, there is a graded ring isomorphism $\mO_{\text{Proj}(S),p}\cong(S_P)_0$. 
\item For any homogeneous element $f\in S_+$, there is an isomorphism $\mO_{\text{Proj}(S)}(D_+(f))\cong S_{(f)}$
\end{itemize}
\end{prp}

\subsection{The Associated Sheaf of Graded Modules}
\begin{defn}{Sheaves of Modules on Graded Rings}{} Let $S$ be a graded ring. Let $M$ be a graded $S$-module. Consider the module $$M_{(p)}=T^{-1}M$$ where $T$ is the multiplicative system of homogenous elements of $S$ not in $p$. Define the sheaf associated to $M$ on $\text{Proj}(S)$, $$\widetilde{M}:\bold{Open}(\text{Proj}(S))\to\bold{Rings}$$ as follows. 
\begin{itemize}
\item For each $U\subseteq\text{Proj}(S)$ open, define $$\mO_{\text{Proj}(S)}(U)=\left\{s:U\to\coprod_{p\in U}M_{(p)}\;\bigg{|}\;\substack{\forall p\in U,\;s(p)\in M_p\text{ and }\exists\;U_p\subseteq U\text{ s.t. }q\in V\text{ implies }\\s(q)=\frac{m}{f}\in M_q\text{ for }f\in S\text{ and }m\in M\text{ homogenous}}\right\}$$
\item For $V\subseteq U$ an inclusion, define the unique morphism $\widetilde{M}(U)\to\widetilde{M}(V)$ by restriction. 
\end{itemize}
\end{defn}

\begin{prp}{}{} Let $S$ be a graded ring and let $M$ be a graded module over $S$. Then the following are true regarding the sheaf of modules $\widetilde{M}$. 
\begin{itemize}
\item For any $p\in\text{Proj}(S)$, there is an isomorphism $\widetilde{M}_p\cong M_{(p)}$
\item For any homogenous $f\in S_+$, there is an isomorphism $$\widetilde{M}|_{D_+(f)}\cong\widetilde{M_{(f)}}$$ via the isomorphism of $D_+(f)$ with $\text{Spec}(S_{(f)})$
\end{itemize}
\end{prp}

\begin{lmm}{}{} Let $S$ be a graded ring and let $M$ be a graded module over $S$. Then $\widetilde{M}$ is a $\mO_{\text{Proj}(S)}$-module. Moreover, if $S$ is Noetherian, then $\widetilde{M}$ is a coherent $\mO_{\text{Proj}(S)}$-module. 
\end{lmm}

For a graded ring $S$, recall that we can shift the grading of $S$ up and down, and it will still be a graded ring. This is the shifted $S(n)$ where $n$ denotes shifting up $n$ times. (Note that this is not true for algebras because $S$ is an algebra over $S_0$, if the grading is shifted then $S(n)$ is an algebra over $S(n)_n$. 

\begin{defn}{The Twisting Sheaf of Serre}{} Let $S$ be a graded ring. Let $X=\text{Proj}(S)$. For any $n\in\Z$, define the sheaf $$\mO_X(n)=\widetilde{S(n)}$$ We call $\mO_X(1)$ the twisting sheaf of Serre. For any sheaf of $\mO_X$-modules $\mF$, denote $$\mF(n)=\mF\otimes_{\mO_X}\mO_X(n)$$
\end{defn}

The twisting sheaf of Serre is important because it is the prototypical example of an invertible sheaf on $\text{Proj}(S)$. 

\begin{prp}{}{} Let $S$ be a graded ring and let $X=\text{Proj}(S)$. Suppose that $S$ is generated by $S_1$ as an $S_0$-algebra. Then the following are true. 
\begin{itemize}
\item The sheaf $\mO_X(n)$ is invertible. 
\item If $M$ is a graded $S$-module, then $\widetilde{M}(n)\cong\widetilde{M(n)}$
\item There is an isomorphism $\mO_X(n)\otimes\mO_X(m)\cong\mO_X(n+m)$
\end{itemize}
\end{prp}

\begin{defn}{Graded Module Associated to a Sheaf of Modules}{} Let $S$ be a graded ring and let $X=\text{Proj}(S)$. Let $\mF$ be a sheaf of $mO_X$-module. Define the graded $S$-module associated to $\mF$ to be the group $$\Gamma_\ast(\mF)=\bigoplus_{n\in\Z}\Gamma(X,\mF(n))$$ together with the structure of graded $S$-module as follows. If $s\in S_d$, then $s$ determines a global section $s\in\Gamma(X,\mO_X(d))$ naturally. For any $t\in\Gamma(X,\mF(n))$, define $s\cdot t\in\Gamma(X,\mF(n+d))$ by sending $s\otimes t\in\mF(n)\otimes\mO_X(d)$ to $\mF(n+d)$ by the isomorphism $$\mF(n)\otimes\mO_X(d)\cong\mF(n+d)$$
\end{defn}

\begin{lmm}{}{} Let $A$ be a ring and let $S=A[x_0,\dots,x_n]$ for $r\geq 1$. Then there is an isomorphism $$\Gamma_\ast(\mO_{\text{Proj}(S)})\cong S$$
\end{lmm}

Note that this is not true if $S$ is not a polynomial ring. 

\begin{prp}{}{}\\
Let $S$ be a graded ring. There is an adjunction given by $$\widetilde{(-)}:\bold{GrMod}_S\rightleftarrows\bold{QCoh}_{\mO_{\text{Proj}(S)}}:\Gamma_\bullet$$ Explicitly, there is a natural isomrophism $$\Hom_{\bold{QCoh}_{\mO_{\text{Proj}(S)}}}(\widetilde{M},\mF)\cong\Hom_{\bold{GrMod}_S}(M,\Gamma_\bullet(\mF))$$ for $M$ a graded $S$-module and $\mF$ a $\mO_{\text{Proj}(S)}$-module. 
\end{prp}

\subsection{Affine and Projective Schemes}
\begin{defn}{Affine Schemes}{}\\
Let $(X,\mO_X)$ be a locally ringed space. We say that $X$ is an affine scheme if there exists a commutative ring $R$ such that $(X,\mO_X)\cong(\text{Spec}(R),\mO_{\text{Spec}(R)})$. 
\end{defn}

\begin{prp}{}{}\\
Let $X$ be an affine scheme. Then $X$ is compact. 
\end{prp}

\begin{defn}{The Category of Affine Schemes}{}\\
The category of affine schemes $\bold{AffSch}$ is the full subcategory of $\bold{LRS}$ consisting of affine schemes. 
\end{defn}

\begin{defn}{Projective Scheme}{}\\
Let $(X,\mO_X)$ be a locally ringed space. We say that $X$ is a projective scheme if there exists a commutative graded ring $S$ such that $(X,\mO_X)\cong(\text{Proj}(S),\mO_{\text{Proj}(S)})$. 
\end{defn}

\subsection{Schemes in General}
\begin{defn}{Schemes}{}\\
Let $(X,\mO_X)$ be a locally ringed space. We say that $X$ is a scheme if for all $p\in X$, there exists an open neighborhood $U\subseteq X$ of $p$ such that $(U,\mO_X|_U)$ is an affine scheme. 
\end{defn}

\begin{prp}{}{}\\
Let $S$ be a commutative graded ring. Then $(\text{Proj}(S),\mO_{\text{Proj}(S)})$ is a scheme. 
\end{prp}

\begin{thm}{Gluing Schemes}{} Let $(X_i,\mathcal{O}_{X_i})$ for $i\in I$ be a family of schemes. Denote $U_{ij}$ an open subset of $X_i$ for $i,j\in I$. Suppose that there is a system of morphisms $$\theta_{ij}:(U_{ij},\mathcal{O}_{X_i}|_{U_{ij}})\to(U_{ji},\mathcal{O}_{X_j}|_{U_{ji}})$$ such that $\theta_{ii}=\text{id}$, $\theta_{ij}\circ\theta_{jk}=\theta_{ik}$. Then there exists a scheme $(X,\mathcal{O}_X)$ and an open cover $X=\bigcup_{i\in I}X_i'$ and a family of isomorphisms $\varphi_i:(X_i',\mathcal{O}_X|_{X_i'})\to(X_i,\mathcal{O}_{X_i})$ such that $$(\varphi_j|_{X_i\cap X_j})^{-1}\circ\theta_{ij}\varphi_i|_{X_i\cap X_j}=\text{id}$$ for all $i,j\in I$. 
\end{thm}

\begin{defn}{The Category of Schemes}{}\\
The category $\bold{Sch}$ of schemes is the full subcategory of $\bold{LRS}$ consisting of schemes. 
\end{defn}

Explicitly, a morphism of schemes $f:X\to Y$ consists of the following data. 
\begin{itemize}
\item A continuous map $X\to Y$. 
\item A morphism of sheaves $\mO_Y\to f_\ast\mO_X$. 
\end{itemize}

\begin{prp}{}{} There is an adjunction $$\Gamma:\bold{Sch}\rightleftarrows\bold{CRing}:\text{Spec}$$ Explicitly, there is a isomorphism $$\Hom_\bold{CRing}(\Gamma(X,\mO_X),R)\cong\Hom_\bold{Sch}((X,\mO_X),(\text{Spec}(R),\mO_{\text{Spec}(R)}))$$ that is natural in $(X,\mO_X)$ and $R$. 
\end{prp}

\begin{prp}{}{} Let $X$ and $Y$ be two schemes over $S$. Then the fibered product $X\times_SY$ exists and is unique up to unique isomorphism. 
\begin{proof}
We first prove the theorem for the case of affine schemes. Let $X=\text{Spec}(A)$, $Y=\text{Spec}(B)$ and $S=\text{Spec}(C)$. I claim that $\text{Spec}(A\otimes_CB)$ is the fibered product of $X$ and $Y$ over $S$. Using the equivalence of categories, we have that 
\begin{align*}
\Hom_\bold{AffSch}(Z,\text{Spec}(A\otimes_CB))&\cong\Hom_\bold{Rings}(A\otimes_CB,\Gamma(Z))\\
&\cong\Hom_\bold{Rings}(A,\Gamma(Z))\times_{\Hom_\bold{Rings}(C,\Gamma(Z))}\Hom_\bold{Rings}(B,\Gamma(Z))\\
&\cong\Hom_\bold{AffSch}(Z,\text{Spec}(A))\times_{\Hom_\bold{AffSch}(Z,\text{Spec}(C))}\Hom_\bold{AffSch}(Z,\text{Spec}(B))
\end{align*}
Thus we have proved that $\text{Spec}(A\otimes_CB)$ is the fiber product of $X$ and $Y$ over $S$. 
\end{proof}
\end{prp}

\begin{defn}{Projective Space over a Scheme}{}\\
Let $X$ be a scheme. Let $n\in\N\setminus\{0\}$. Define the projective $n$-space over $X$ to be $$\Prj_X^n=\Prj_\Z^n\times_\Z X$$
\end{defn}

\begin{defn}{The Residue Field}{}\\
Let $X$ be a scheme. Let $x\in X$. Define the residue field of $x$ to be $$k(x)=\frac{\mO_{X,x}}{m}$$ where $m$ is the unique maximal ideal of $\mO_{X,x}$. 
\end{defn}

\subsection{Open and Closed Subschemes}
\begin{defn}{Open Immersion}{}\\
Let $X,Y$ be schemes. Let $f:X\to Y$ be a morphism. We say that $f$ is an open immersion if $f$ induces an isomorphism $X\cong f(X$ where $f(X)$ is an open subset of $Y$. 
\end{defn}

\begin{defn}{Open Subschemes}{}\\
Let $(X,\mO_X)$ be a scheme. An open subscheme of $X$ is a scheme $(U,\mO_X|_U)$ where $U\subseteq X$ is an open subset. 
\end{defn}

\begin{defn}{Closed Immersion}{}\\
Let $(X,\mO_X)$ and $(Y,\mO_Y)$ be schemes. Let $f:X\to Y$ be a morphism. We say that $f$ is a closed immersion if $f$ induces an isomorphism $X\cong f(X)$ where $f(X)$ is a closed subset of $Y$, and $f^\#$ is surjective. 
\end{defn}

\begin{defn}{Closed Subschemes}{}\\
Let $X$ be a scheme. A closed subscheme of $X$ is an equivalence class of closed immersions $f:Y\to X$ where we say that $f:Y\to X$ and $f':Y'\to X$ are equivalent if there exists an isomorphism $i:Y'\to Y$ such that $f'=f\circ i$. 
\end{defn}

\begin{prp}{}{}\\
Let $A$ be a ring. Then the following are true. 
\begin{itemize}
\item For any ideal $I\subseteq A$, the projection map induces a closed subscheme structure $\text{pr}^\flat:\text{Spec}(A/I)\to\text{Spec}(A)$. 
\item Every closed subscheme of $\text{Spec}(A)$ is affine. Moreover, if $f:Y\to X$ is a closed subscheme of $X$, then there exists an ideal $I\subseteq A$ such that $f$ is equivalent to $\text{pr}^\flat:\text{Spec}(A/I)\to\text{Spec}(A)$. 
\end{itemize}
\end{prp}

\subsection{Quasi-Coherent Modules on Schemes}
\begin{prp}{}{} Let $(X,\mO_X)$ be a scheme. Let $\mF$ be a $\mO_X$-module. Then the following are equivalent. 
\begin{itemize}
\item $\mF$ is quasi-coherent. 
\item For all affine open subschemes $U$, $\mF|_U\cong\widetilde{M}$ for some $\mO_X(U)$-module $M$. 
\item $X$ can be covered by open affine subsets $U_i=\text{Spec}(A_i)$ such that for each $i$, there is an $A_i$-module $M_i$ with $\mathcal{F}|_{U_i}\cong\widetilde{M}_i$. 
\end{itemize}
\end{prp}

\begin{prp}{}{} Let $(X,\mO_X)$ be a scheme. Let $\mF$ be a sheaf of $\mO_X$-modules. Then $\mF$ is a coherent sheaf if and only if $X$ can be covered by open affine subsets $U_i=\text{Spec}(A_i)$ such that for each $i$, there is a finitely generated $A_i$-module $M_i$ with $\mathcal{F}|_{U_i}\cong\widetilde{M}_i$. 
\end{prp}


\subsection{The Ideal Sheaf}
\begin{defn}{Sheaf of Ideals}{} Let $(X,\mO_X)$ be a scheme. Let $\mI$ be a sheaf of $\mO_X$-modules. We say that $\mI$ is a sheaf of ideals if for each open subset $U\subseteq X$, we have that $\mI(U)$ is an ideal of $\mO_X(U)$. 
\end{defn}

\begin{defn}{The Ideal Sheaf}{}\\
Let $X$ be a scheme. Let $Y\subseteq X$ be a closed subscheme of $X$. Define the ideal sheaf of $Y$ to be the kernel $$\mI_Y=\ker(i^\#:\mO_X\to i_\ast(\mO_Y))$$
\end{defn}

\begin{lmm}{}{}\\
Let $X$ be a scheme. Then the following are true. 
\begin{itemize}
\item Let $Y\subseteq X$ be a closed subscheme of $X$. Then $\mI_Y$ is a quasi-coherent sheaf. 
\item Let $\mI$ be a sheaf of ideals of $X$. Then there exists a closed subscheme $Y\subseteq X$ of $X$ such that $\mI\cong\mI_Y$. 
\end{itemize}
\end{lmm}

\begin{lmm}{}{}\\
Let $X$ be a scheme. Let $Y\subseteq X$ be a closed subscheme of $X$. If $X$ is Noetherian, then $\mI_Y$ is a coherent sheaf. 
\end{lmm}

\begin{prp}{}{}\\
Let $R$ be a commutative ring. Then there is a one to one correspondence $$\{I\subseteq R\;|\;I\text{ is an ideal of }R\}\;\;\overset{1:1}{\longleftrightarrow}\;\;\{Y\subseteq\text{Spec}(R)\;|\;Y\text{ is a closed subscheme of Spec}(R)\}$$ given by $I\mapsto\text{Spec}(R/I)$. 
\end{prp}

\begin{prp}{}{}\\
Let $S$ be a commutative graded ring. Then there is a one to one correspondence $$\{I\subseteq S\;|\;I\text{ is a homogeneous ideal of }S\}\;\;\overset{1:1}{\longleftrightarrow}\;\;\{Y\subseteq\text{Spec}(S)\;|\;Y\text{ is a closed subscheme of Proj}(S)\}$$ given by $I\mapsto\text{Proj}(S/I)$. 
\end{prp}

\subsection{Reduced Schemes}
\begin{defn}{Reduced Schemes}{}\\
Let $X$ be a scheme. We say that $X$ is reduced for every open set $U\subseteq X$, $\mO_X(U)$ is a reduced ring. 
\end{defn}

\begin{lmm}{}{}\\
Let $X$ be a scheme. Then $X$ is reduced if and only if for all $p\in X$, $\mO_{X,p}$ is a reduced ring. 
\end{lmm}

\begin{defn}{The Reduced Scheme}{}\\
Let $(X,\mO_X)$ be a scheme. Define the reduced scheme $(\mO_X)_\text{red}$ of $X$ to be the sheafification of the presheaf defined as follows. 
\begin{itemize}
\item For each open set $U\subseteq X$, define $(\mO_X)_\text{red}(U)=\frac{\mO_X(U)}{N(\mO_X(U))}$. 
\item For an inclusion of open sets $V\subseteq U$, define the induced map $$(\mO_X)_\text{red}(U)\to(\mO_X)_\text{red}(V)$$ as follows. There is a restriction map $\mO_X(U)\to\mO_X(V)$. Projecting to the quotient, we have the map $\mO_X(U)\to(\mO_X)_\text{red}(V)$. Since nilpotents are sent to nilpotents under ring homomorphisms, the map descends to the quotient $(\mO_X)_\text{red}(U)\to(\mO_X)_\text{red}(V)$. 
\end{itemize}
\end{defn}

\begin{defn}{The Canonical Map of the Reduced Scheme}{}\\
Let $X$ be a scheme. Define the canonical map $X_\text{red}\to X$ as follows. 
\begin{itemize}
\item The map of the underlying topological space is given by the identity map. 
\item The morphism of sheaves $\mO_X\to\text{id}_\ast((\mO_X)_\text{red})=(\mO_X)_\text{red}$ is given by the quotient map $\mO_X(U)\to(\mO_X)_\text{red}(U)=\frac{\mO_X(U)}{N(\mO_X(U))}$ for any open set $U$. 
\end{itemize}
\end{defn}

\begin{prp}{}{}\\
Let $X$ be a scheme. Then the reduced scheme of $X$ satisfies the following universal property. For any reduced scheme $Y$ and morphism $f:Y\to X$, there exists a unique morphism $f_\text{red}:Y\to X_\text{red}$ such that the following diagram commutes: \\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	Y & {X_\text{red}} \\
	& X
	\arrow["{\exists!f_\text{red}}", dashed, from=1-1, to=1-2]
	\arrow["f"', from=1-1, to=2-2]
	\arrow[from=1-2, to=2-2]
\end{tikzcd}}\\
Moreover, $X_\text{red}$ is the unique reduced scheme satisfying the above. 
\end{prp}

\subsection{Dimension and Codimension}
\begin{defn}{The Dimension of a Scheme}{}\\
Let $X$ be a scheme. Define the dimension of $X$ to be dimension of the underlying space of $X$. Explicitly, we have $$\dim(X)=\sup_{\substack{Z_0,\dots,Z_n\subseteq X\\\text{closed irreducible subsets}}}\{n\in\N\;|\;Z_0\subset Z_1\subset\dots\subset Z_n\}$$
\end{defn}

\begin{prp}{}{}\\
Let $X$ be a scheme. Then the following numbers are equal. 
\begin{itemize}
\item $\dim(X)$. 
\item $\max\{\dim(V_i)\;|\;1\leq i\leq n\}$ where $V_1,\dots,V_n$ are the irreducible components of $X$. 
\item $\sup\{\dim(\mO_{X,p})\;|\;p\in X\}$. 
\end{itemize}
\end{prp}

\begin{prp}{}{}\\
Let $X,Y$ be schemes. Let $i:Y\to X$ be a closed immersion. If $X$ is integral and $\dim(X)=\dim(Y)$, then $i$ is an isomorphism. 
\end{prp}

\end{prp}

\begin{prp}{}{}\\
Let $k$ be a field. Let $X,Y$ be schemes locally of finite type over $k$. Then we have $$\dim(X\times_kY)=\dim(X)+\dim(Y)$$
\end{prp}

\begin{defn}{The Codimension of a Scheme}{}\\
Let $X$ be a scheme. Let $Y\subseteq X$ be a closed subset of $X$. Define the codimension of $Y$ in $X$ to be $$\text{codim}_X(Y)=\sup_{\substack{Z\subseteq Y\text{ is a}\\\text{closed irreducible subsets}}}\{n\in\N\;|\;Z=Z_0\subset\cdots\subset Z_n\subset X\text{ are closed irreducible subsets }\}$$
\end{defn}

\begin{prp}{}{}\\
Let $k$ be a field. Let $X$ be an irreducible scheme of finite type over $k$. Let $Y$ be a closed subset of $X$. Then we have $$\dim(Y)+\text{codim}_X(Y)=\dim(X)$$
\end{prp}

\section{Points on A Scheme}
\subsection{The Functor of Points}
Let $k$ be a field. Recall the Yoneda Embedding to be the functor $$\bold{Sch}_k\to\Hom_{\bold{Cat}^{\text{op}}}(\bold{Sch}_k,\bold{Set})$$ given by the following data. 
\begin{itemize}
\item A scheme $X$ over $k$ is assigned to the functor $\Hom_{\bold{Sch}_k}(-,X):\bold{Sch}_k\to\bold{Set}$. 
\item A morphism $f:X\to Y$ over $k$ is assigned to the natural transformation $\Hom_{\bold{Sch}_k}(-,Y)\to\Hom_{\bold{Sch}_k}(-,X)$ given by precomposition with $f$. 
\end{itemize}

The Yoneda lemma then says that this is a fully faithful functor. One can restrict the Yoneda embedding to the category of affine schemes, and using the equivalence of the category of finitely generated schemes of finite type over $k$ with the category of finitely generated $k$-algebras, we obtain the following series of definitions. 

\begin{defn}{R-Points of a Scheme}{}\\
Let $k$ be a field. Let $X$ be a scheme over $k$. Let $R$ be a finitely generated $k$-algebra. Define the $R$-points of $X$ to be $$X(R)=\Hom_{\bold{Sch}_k}(\text{Spec}(R),X)$$
\end{defn}

\begin{defn}{The Functor of Points}{}\\
Let $k$ be a field. Let $X$ be a scheme over $k$. Define the functor $$\widetilde{X}:\bold{FGAlg}_k\to\bold{Set}$$ as follows. 
\begin{itemize}
\item For a finitely generated $k$-algebra $R$, $\widetilde{X}(R)=X(R)$ is the $R$-points of $X$. 
\item For a $k$-algebra homomorphism $\varphi:R\to S$, $X(\varphi):X(R)\to X(S)$ is given by precomposition with $\varphi^\flat:\text{Spec}(S)\to\text{Spec}(R)$. 
\end{itemize}
\end{defn}

\begin{defn}{An Embedding of Schemes}{}\\
Let $k$ be a field. Define the functor $$\widetilde{(-)}:\bold{Sch}_k\to\Hom_{\bold{Cat}}(\bold{FGAlg}_k,\bold{Set})$$ as follows. 
\begin{itemize}
\item For a $k$-scheme $X$, $\widetilde{X}$ is the functor sending $R$ to $X(R)$. 
\item For a morphism of $k$-schemes $f:X\to Y$, $\widetilde{X}\to\widetilde{Y}$ is the natural transformation defined as follows. For each $R\in\bold{FGAlg}_k$, the map $X(R)\to Y(R)$ is given by post composition with $f$. 
\end{itemize}
\end{defn}

It turns out that for fully faithfulness of the Yoneda embedding for the category of schemes over $k$, it is enought to check it on affine schemes. 

\begin{lmm}{}{}\\
Let $k$ be a field. Then the functor $\widetilde{(-)}:\bold{Sch}_k\to\Hom_{\bold{Cat}}(\bold{Alg}_k,\bold{Set})$ is fully faithful. 
\end{lmm}

In other words, the $R$ points of a scheme for all $R$ determine the scheme itself. \\

We now specialize it to the case where $R=k$ the base field. 

\begin{eg}{}{}\\
The $\C$-points of $\text{Spec}(\C[x])$ are in bijection with $\C$. 
\begin{proof}\\
The $\C$-points of the affine scheme are in bijection with the ring homomorphisms $\C[x]\to\C$. Clearly for any $a\in\C$, the evaluation homomorphism $\varphi_a:\C[x]\to\C$ is a ring homomorphism. On the other hand, given a ring homomorphism $\phi:\C[x]\to\C$, it is surjective since $\phi(1)=1$ by definition. Then there is a ring isomorphism $\frac{\C[x]}{\ker(\phi)}\cong\C$ given by the first isomorphism theorem. Since $\C$ is a field, $\ker(\phi)$ is a maximal ideal and by the correspondence of maximal ideals of $\C[x]$, we must have $\ker(\phi)=(x-a)$ for some $a\in\C$. Hence $\phi=\varphi_a$. Hence we are done. 
\end{proof}
\end{eg}

Notice that $\text{Spec}(\C[x])$ has an extra point $(0)$ which is not a $\C$-point of the affine scheme. 

\begin{prp}{}{}\\
Let $k$ be a field. Let $X$ be a scheme. Then there is a natural bijection $$X(k)\cong\{x\in X\;|\;\text{the canonical homomorphism }k\to k(x)\text{ is an isomorphism }\}$$
\end{prp}

\begin{lmm}{}{}\\
Let $k$ be an algebraically closed field. Let $X$ be a variety over $k$. Then there is a natural bijection $$X(k)\cong\{x\in X\;|\;x\text{ is a closed point of }X\}$$
\end{lmm}

\subsection{Generic Points and Function Fields}
\begin{defn}{Generic Points}{}\\
Let $X$ be a scheme. Let $Z$ be an irreducible closed subset of $X$. Let $p\in X$. We say that $p$ is a generic point for $Z$ if $Z=\overline{\{p\}}$. 
\end{defn}

\begin{lmm}{}{} Let $X$ be a scheme. Let $Z$ be an irreducible closed subset of $X$. Then $Z$ admits a unique generic point. 
\end{lmm}

\begin{prp}{}{} Let $X$ be a scheme. If $X$ is irreducible, then $X$ has a unique generic point. More generally, if $X$ has irreducible components $S_i$ for $i\in I$, then each $S_i$ has a unique generic point. 
\end{prp}

\begin{defn}{Function Field of an Integral Scheme}{}\\
Let $X$ be an integral scheme. Let $\nu$ be its unique generic point. Define the function field of $X$ to be $$k(X)=\mO_{X,\nu}$$
\end{defn}

\begin{lmm}{}{}\\
Let $X$ be an integral scheme. Let $\nu$ be its unique generic point. Then $k(X)$ is a field. 
\end{lmm}

\begin{prp}{}{}\\
Let $R$ be a commutative ring. Then $k(\text{Spec}(R))=Q(R)$. 
\end{prp}

\subsection{The Fibers of a Morphism}
\begin{defn}{The Fibers of a Morphism}{}\\
Let $X,Y$ be schemes. Let $f:X\to Y$ be a morphism. Let $y\in Y$ be a point. Define the fiber of $f$ at $y$ to be the pullback $$X_y=X\times_Y\frac{\mO_{Y,y}}{m}$$ where $m$ is the unique maximal ideal of $\mO_{Y,y}$. 
\end{defn}

\begin{prp}{}{}\\
Let $X,Y$ be schemes. Let $f:X\to Y$ be a morphism. Let $y\in Y$ be a point. Then there is a natural bijection of sets $$X_y\cong f^{-1}(y)$$
\end{prp}

\section{Relative Properties of Schemes}
\subsection{Morphisms of Finite Types}
\begin{defn}{Morphisms Locally of Finite Type}{}\\
Let $X,Y$ be schemes. Let $f:X\to Y$ be a morphism. We say that $f$ is locally of finite type if for every open affine subset $V=\text{Spec}(B)$ of $Y$, there is an open affine cover $\{U_i=\text{Spec}(A_i)\;|\;i\in I\}$ for $f^{-1}(V)$ such that $A_i$ is a finitely generated $B$ algebra for all $i$. 
\end{defn}

\begin{prp}{}{}\\
Let $k$ be a field. Let $X,Y$ be schemes locally of finite type over $k$. Then $X\times_kY$ is locally of finite type over $k$. 
\end{prp}

\begin{defn}{Morphisms of Finite Type}{}\\
Let $X,Y$ be schemes. Let $f:X\to Y$ be a morphism. We say that $f$ is locally of finite type if for every open affine subset $V=\text{Spec}(B)$ of $Y$, there is a finite open affine cover $\{U_i=\text{Spec}(A_i)\;|\;1\leq i\leq n\}$ for $f^{-1}(V)$ such that $A_i$ is a finitely generated $B$ algebra for all $i$. 
\end{defn}

It is easy to see that we have the following results. 
\begin{itemize}
\item Let $k$ be a field. There is an equivalence of categories between the category of finitely generated $k$-algebras and the category of schemes of finite type over $k$ given by the spec functor and the global section functor. 
\item The restricted Yoneda embedding from the category of schemes of finite type over $k$ to the the category of functors $$\Hom_{\bold{Cat}}(\bold{FGAlg}_k,\bold{Set})$$ is fully faithful. 
\end{itemize}

\begin{lmm}{}{}\\
Let $X,Y$ be schemes. Let $f:X\to Y$ be a morphism. If $f$ is a closed immersion, then $f$ is a morphism of finite type. 
\end{lmm}

\begin{prp}{}{}\\
Let $X,Y,Z,S$ be schemes. Then the following are true. 
\begin{itemize}
\item If $f:X\to Y$ and $g:Y\to Z$ are morphisms of finite type, then $g\circ f$ is a morphism of finite type. 
\item If $f:X\to S$ and $g:Y\to S$ are morphisms of finite type, then $X\times_SY\to S$ is a morphism of finite type. 
\end{itemize}
\end{prp}

\begin{lmm}{}{}\\
Let $k$ be a field. Let $X$ be a scheme of finite type over $k$. Then $X(k)$ is Noetherian. 
\end{lmm}

\subsection{Separated Morphisms}
Separatedness is essentially the analog of the Hausdorff condition for schemes. Recall that a topological space $X$ is Hausdorff if and only if the digonal morphism to $X\times X$ is closed. 

\begin{defn}{Diagonal Morphisms}{} Let $f:X\to Y$ be a morphism of schemes. The diagonal morphism is the unique morphism $\delta:X\to X\times_Y X$ whose composition with both projection maps $p_1,p_2:X\times_Y X\to X$ is the identity map of $X$. 
\end{defn}

\begin{defn}{Separated Morphisms and Schemes}{} Let $f:X\to Y$ be a morphism of schemes. We say that $f$ is separated (or $X$ is separated over $Y$) if the diagonal morphism $\delta$ is a closed immersion. A scheme $X$ is separated if it is separated over $\text{Spec}(\Z)$. 
\end{defn}

\begin{prp}{}{} If $f:X\to Y$ is a morphism of affine schemes, then $f$ is separated. 
\end{prp}

\begin{prp}{}{} Let $f:X\to Y$ be a morphism of schemes. Then $f$ is separated if and only if the image of the diagonal morphism is a closed subset of $X\times_Y X$. 
\end{prp}

\begin{thm}{Valuative Criterion of Separatedness}{}\\
Let $X,Y$ be schemes. Let $f:X\to Y$ be a morphism. Suppose that $X$ is Noetherian. Then $f$ is proper if and only if for every valuation ring $(R,m)$ and every morphism $\text{Spec}(R/m)\to X$ and $\text{Spec}(R)\to Y$, any morphism $\text{Spec}(R)\to X$ making the following diagram commutes: \\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	{\text{Spec}(R/m)} & X \\
	{\text{Spec}(R)} & Y
	\arrow[from=1-1, to=1-2]
	\arrow["{\text{proj.}^\flat}"', from=1-1, to=2-1]
	\arrow["f", from=1-2, to=2-2]
	\arrow["{\exists!}"{description}, dashed, from=2-1, to=1-2]
	\arrow[from=2-1, to=2-2]
\end{tikzcd}} \\
is unique. 
\end{thm}

\begin{prp}{}{} Let $X$ and $Y$ be Noetherian schemes. Then any open or closed immersions $f:X\to Y$ are separated. 
\end{prp}

\subsection{Proper Morphisms}
\begin{defn}{Universally Closed Morphisms}{}\\
Let $X,Y$ be schemes. Let $f:X\to Y$ be a morphism. We say that $f$ is universally close if for any scheme $X'$ and any morphism $g:X'\to Y$, the induced morphism $X\times_YX'\to X'$ is a closed morphism. 
\end{defn}

\begin{defn}{Proper Morphisms}{}\\
Let $X,Y$ be schemes. Let $f:X\to Y$ be a morphism. We say that $f$ is a proper morphism if $f$ is separated, of finite type and is universally closed. 
\end{defn}

Let $X$ be a scheme over a field $k$. The condition for the structure morphism $X\to\text{Spec}(k)$ to be proper is to test that it is separated, of finite type and that for any scheme $Y$ over $k$, the morphism $X\times_kY\to Y$ is closed. 

\begin{thm}{Valuative Criterion for Properness}{}\\
Let $X,Y$ be schemes. Let $f:X\to Y$ be a morphism of finite type. Suppose that $X$ is Noetherian. Then $f$ is proper if and only if for every valuation ring $(R,m)$ and every morphism $\text{Spec}(R/m)\to X$ and $\text{Spec}(R)\to Y$, there exists a unqiue morphism $\text{Spec}(R)\to X$ such that the following diagram commutes: \\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	{\text{Spec}(R/m)} & X \\
	{\text{Spec}(R)} & Y
	\arrow[from=1-1, to=1-2]
	\arrow["{\text{proj.}^\flat}"', from=1-1, to=2-1]
	\arrow["f", from=1-2, to=2-2]
	\arrow["{\exists!}"{description}, dashed, from=2-1, to=1-2]
	\arrow[from=2-1, to=2-2]
\end{tikzcd}} \\
\end{thm}

\begin{prp}{}{}\\
Let $X,Y$ be Noetherian schemes. Let $f:X\to Y$ be a closed immersion. Then $f$ is proper. 
\end{prp}

\begin{prp}{}{}\\
Let $X,Y,Z$ be Noetherian schemes. Let $f:X\to Y$ and $g:Y\to Z$ be morphisms. Then the following are true. 
\begin{itemize}
\item If $f$ and $g$ are proper, then $g\circ f$ is proper. 
\item If $g\circ f$ is proper and $g$ is separated, then $f$ is proper. 
\end{itemize}
\end{prp}

\begin{prp}{}{}\\
Let $X_1,X_2,Y_1,Y_2$ be Noetherian schemes. Let $f_1:X_1\to Y_1$ and $f_2:X_2\to Y_2$ be morphisms. If $f_1$ and $f_2$ are proper, then $$f_1\times f_2:X_1\times X_2\to Y_1\times Y_2$$ is proper. 
\end{prp}

\subsection{Finite Morphisms}
\begin{defn}{Finite Morphisms}{}\\
Let $X,Y$ be schemes. Let $f:X\to Y$ be a morphism. We say that $f$ is finite if for every open affine subset $V=\text{Spec}(B)$ of $Y$, $f^{-1}(V)=\text{Spec}(A)$ is affine and $A$ is a finitely generated $B$-module. 
\end{defn}

\begin{prp}{}{}\\
Let $X,Y$ be schemes. Let $f:X\to Y$ be a morphism. Then $f$ is finite if and only if there exists an open affine cover $\{U_i=\text{Spec}(B_i)\;|\;i\in I\}$ of $Y$ such that for each $i$,$f^{-1}(U_i)=\text{Spec}(A_i)$ is affine and $A_i$ is a finitely generated $B_i$-module. 
\end{prp}

\begin{prp}{}{}\\
Let $X,Y$ be schemes. Let $f:X\to Y$ be a morphism. If $f$ is finite, then the following are true. 
\begin{itemize}
\item $f$ is proper. 
\item For all $y\in Y$, $f^{-1}(y)$ is finite. 
\item $f$ is closed. 
\end{itemize}
\end{prp}

\begin{prp}{}{}\\
Let $X$ be a scheme. Let $Y$ be a locally Noetherian scheme. Let $f:X\to Y$ be a morphism. Then $f$ is finite if and only if $f$ is proper and for all $y\in Y$, $f^{-1}(y)$ is finite. 
\end{prp}

\section{Relation to Classical Varieties}
\subsection{Varieties Revisited}
\begin{defn}{The Associated Scheme of a Variety}{}\\
Let $k$ be a field. Let $X$ be a variety over $k$. Define the associated scheme of $X$ as follows. 
\begin{itemize}
\item The topological space is given by $t(X)=\{Z\subseteq X\;|\;Z\text{ is irreducible and closed }\}$ whose closed subsets are of the form $t(Z)$ for $Z\subseteq X$ an irreducible and closed subset of $X$. 
\item Let $\alpha:X\to t(X)$ be the continuous map $\alpha(p)=\overline{\{p\}}$. Then the sheaf on $t(X)$ is the pushforward sheaf $\alpha_\ast(\mO_X)$. 
\end{itemize}
\end{defn}

\begin{defn}{The Associated Scheme of a Variety}{}\\
Let $k$ be a field. Define the functor $$t:\bold{Var}_k\to\bold{Sch}_k$$ as follows. 
\begin{itemize}
\item Each variety $X$ over $k$ is associated to the scheme $t(X)$. 
\item For a morphism of varieties $f:X\to Y$, $t(f)$ is the map defined by $t(f)(Z)=\overline{f(Z)}$ for $Z\subseteq X$ an irreducible closed subset of $X$. 
\end{itemize}
\end{defn}

\begin{prp}{}{}\\
Let $k$ be an algebraically closed field. Then the following are true. 
\begin{itemize}
\item The functor $t:\bold{Var}_k\to\bold{Sch}_k$ is fully faithful. 
\item The essential image of $t$ consists of the Noetherian, reduced and separated schemes of finite type over $k$. 
\item Restricted to the essential image, the inverse of $t$ is given by the functor defined by $X\mapsto(X(k),\mO_X|_{X(k)})$. 
\end{itemize}
\end{prp}

This motivates the following definition. 

\begin{defn}{Varieties}{} Let $k$ be a field. A variety over $k$ is a scheme $X$ such that $X$ is an reduced separated scheme of finite type over $k$. 
\end{defn}

\begin{lmm}{}{}\\
Let $k$ be a field. Let $X$ be a variety over $k$. Let $x\in X$. Then the following are equivalent. 
\begin{itemize}
\item $x$ is a closed point of $X$. 
\item The field extension $k\to\frac{\mO_{X,x}}{m}$ is finite. 
\item The field extension $k\to\frac{\mO_{X,x}}{m}$ is algebraic. 
\end{itemize}
\end{lmm}

\begin{defn}{The Category of Varieties}{} Define the category of varieties $\text{Var}_k$ over a field $k$ as follows. 
\begin{itemize}
\item The objects are varieties $X$ over $k$
\item The morphisms are morphisms of schemes $X\to Y$ over $k$. 
\item Composition is given by the composition of morphisms. 
\end{itemize}
\end{defn}

In virtue of the above proposition, this new category of varieties is equivalent to the category of classical varieties. 

\begin{prp}{}{}\\
Let $k$ be a field. Let $X,Y$ be varieties over $X$. Then the following are true. 
\begin{itemize}
\item $X\times Y$ is a variety over $k$. 
\item If $X,Y$ are affine, then $X\times Y$ is affine. 
\item If $X,Y$ are projective, then $X\times Y$ is projective. 
\end{itemize}
\end{prp}

\begin{prp}{}{}\\
Let $k$ be a field. Let $X,Y$ be varieties over $k$. Let $\phi:X\to Y$ be a morphism of schemes over $k$. Then $\phi$ is separated and of finite type over $k$. 
\end{prp}

\begin{defn}{Subvarities}{} Let $X$ be a variety. A subvariety of $X$ is a closed subscheme of $X$ such that $X$ is also a variety. 
\end{defn}

\subsection{Complete Varieties}
\begin{defn}{Complete Varieties}{} Let $k$ be a field. We say that a variety $X$ over $k$ is complete if it is proper over $k$. 
\end{defn}

\begin{lmm}{}{}\\
Let $k$ be a field. Let $X,Y$ be a varieties over $k$. If $X$ and $Y$ are complete, then $X\times Y$ is complete. 
\end{lmm}

\begin{prp}{}{}\\
Let $k$ be a field. Let $X$ be a variety over $k$. Let $Y\subseteq X$ be a subvariety of $X$. Then the following are true. 
\begin{itemize}
\item If $X$ is complete and $Y$ is closed in $X$, then $Y$ is complete. 
\item If $Y$ is complete, then $Y$ is closed in $X$. 
\end{itemize}
\end{prp}

\begin{lmm}{}{}\\
Let $k$ be a field. Let $X$ be an affine variety over $k$. If $X$ is complete, then $\dim(X)=0$. 
\end{lmm}

\begin{prp}{}{}\\
Let $k$ be a field. Let $X$ be a variety over $k$. If $X$ is projective, then $X$ is complete. 
\end{prp}

\section{The Theory of Divisors}
\subsection{Weil Divisors}
\begin{defn}{Regular in Codimension $1$}{}\\
Let $X$ be a scheme. We say that $X$ is regular in codimension $1$ if for all irreducible codimension $1$ subschemes $Y$ of $X$, the local ring $\mO_{X,Y}$ is a regular local ring. 
\end{defn}

\begin{defn}{Weil Divisors}{}\\
Let $X$ be a Noetherian integral separated scheme that is regular in codimension one. Define the group of Weil divisors to be the free abelian group $$\text{Div}(X)=\Z\langle Y\subseteq X\;|\;Y\text{ is a closed integral subscheme of codimension one }\rangle$$ generated by closed integral subschemes of $X$ of codimension one. 
\end{defn}

\begin{defn}{Divisors of Functions}{}\\
Let $X$ be a Noetherian integral separated scheme that is regular in codimension one. For a closed integral subscheme $Y$ of codimension one, denote $\eta$ the unique generic point of $Y$ and denote $v_Y$ the valuation of the discrete valuation ring $\mO_{X,\eta}$. Define the divisor of $f\in k(X)\setminus\{0\}$ to be $$(f)=\sum_Yv_Y(f)\cdot Y$$
\end{defn}

\begin{lmm}{}{}\\
Let $X$ be a Noetherian integral separated scheme that is regular in codimension one. Let $f\in k(X)\setminus\{0\}$. Then $(f)\in\text{Div}(X)$. 
\end{lmm}

The lemma shows that divisors of functions are well defined. 

\begin{defn}{Principal Divisors}{} Let $X$ be a Noetherian integral separated scheme which is regular in codimension $1$. We say that a divisor $D$ on $X$ is principal if $D=(f)$ for some function $f$. 
\end{defn}

\begin{defn}{The Divisor Class Group}{} Let $X$ be a Noetherian integral separated scheme which is regular in codimension $1$. Let $\text{Prin}(X)$ be the subgroup of all principal divisors of $X$. Define the divisor class group of $X$ as $$\text{Cl}(X)=\frac{\text{Div}(X)}{\text{Prin}(X)}$$ Two elements of the same coset are said to be linearly equivalent. 
\end{defn}

\begin{defn}{Degree Homomorphism}{} Let $X$ be a Noetherian integral separated scheme which is regular in codimension $1$. Define the degree homomorphism $$\deg:\text{Div}(X)\to\Z$$ by $D=\sum_Pn_P\cdot P\mapsto\sum_Pn_P$. For each divisor $D$, define the degree of $D$ to be $$\deg(D)=\sum_Pn_P$$
\end{defn}

\subsection{Cartier Divisors}
\begin{defn}{The Sheaf of Total Quotient Rings}{} Let $(X,\mO_X)$ be a scheme. Define a presheaf $K_X:\bold{Open}(X)\to\bold{Ring}$ as follows. 
\begin{itemize}
\item Each $U\subseteq X$ open subset is associated to the total quotient ring $K(U)=Q(\mO_X(U))$ . 
\item For each inclusion $U\subseteq V$, define $K(V)\to K(U)$ by just the restriction map. 
\end{itemize}
Define the sheaf of total quotient rings to be the associated sheaf of the presheaf $K$. 
\end{defn}

\begin{defn}{Invertible Elements of a Sheaf}{} Let $(X,\mF)$ be a ringed space. Define the sheaf of invertible elements $$\mF^\ast:\bold{Open}\to\bold{Grp}$$ of $\mF$ as follows. 
\begin{itemize}
\item For $U\subseteq X$ an open set, define $\mF^\ast(U)=\text{The Group of Units of }\mF(U)$
\item For $U\subseteq V$, define $\mF^\ast(V)\to\mF^\ast(U)$ to just be the restriction map. 
\end{itemize}
\end{defn}

\begin{lmm}{}{}\\
Let $(X,\mO_X)$ be a scheme. Then there is an exact sequence of $\mO_X$-modules given by \\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	0 & {\mO_X^\ast} & {K_X^\ast} & {\frac{K_X^\ast}{\mO_X^\ast}} & 0
	\arrow[from=1-1, to=1-2]
	\arrow[from=1-2, to=1-3]
	\arrow[from=1-3, to=1-4]
	\arrow[from=1-4, to=1-5]
\end{tikzcd}} \\
\end{lmm}

\begin{defn}{Cartier Divisors}{}\\
Let $X$ be a scheme. Define the group of Cartier divisors to be the abelian group $$\Gamma(X,K_X^\ast/\mO_X^\ast)$$
\end{defn}

Explicitly, a Cartier divisor consists of an open cover $\{U_u\;|\;i\in I\}$ and regular functions $f_i\in K_X(U_i)$ such that for each $i,j\in I$, there exists $s\in\mO_X^\ast(U_i\cap U_j)$ such that $f_i=sf_j$. 

\begin{defn}{Principal Cartier Divisors}{}\\
Let $(X,\mO_X)$ be a scheme. A Cartier divisor $f$ of $X$ is said to be principal if it is the image of the natural map $$\Gamma(X,K_X^\ast)\to\Gamma(X,K_X^\ast/\mO_X^\ast)$$
\end{defn}

\begin{defn}{The Cartier Divisor Class Group}{}\\
Let $(X,\mO_X)$ be a scheme. Define the Cartier divisor class group to be the abelian group $$\text{CaCl}(X)=\frac{\Gamma(X,K_X^\ast/\mO_X^\ast)}{\sim}$$ where $D_1\sim D_2$ if and only if $D_1-D_2$ is a principal divisor. 
\end{defn}

\begin{prp}{}{} Let $X$ be an integral, separated Noetherian scheme such that all local rings are UFD. Then the following are true. 
\begin{itemize}
\item There is an isomorphism $$\text{Div}(X)\cong\Gamma(X,K_X^\ast/\mO_X^\ast)$$
\item There is an isomorphism $$\text{Cl}(X)\cong\text{CaCl}(X)$$
\end{itemize}
\end{prp}

\subsection{Relation to Invertible Sheaves}
\begin{defn}{The Associated Sheaf of a Divisor}{}\\
Let $(X,\mO_X)$ be a scheme. Let $D=\{(U_i,f_i\in K_X^\ast)\;|\;i\in I\}$ be a Cartier divisor on $X$. Define the associated sheaf $\mL(D)$ of $D$ to be the $\mO_X$-module by gluing the schemes $(U_i,\mO_X(U_i)(f_i^{-1}))$ for each $i$. 
\end{defn}

\begin{prp}{}{}\\
Let $X$ be a scheme. Let $D$ be a Cartier divisor of $X$. Then $\mL(D)$ is an invertible sheaf. 
\end{prp}

The proposition motivates us to investigate the relations between Cartier divisors and invertible sheaves on a scheme $X$. This leads to a very fruitful and satisfying result. 

\begin{thm}{}{} Let $X$ be a scheme. For any Cartier divisor $D$ of $X$, the association $D\mapsto\mL(D)$ gives a bijection $$\{\text{Cartier Divisors on }X\}\;\;\;\;\overset{1:1}{\longleftrightarrow}\;\;\;\;\{\text{Invertible Subsheaves of }\mK\}$$
\end{thm}

\begin{prp}{}{} Let $X$ be a scheme. Let $D_1$ and $D_2$ be Cartier divisors of $X$. Then there is an isomorphism $$\mL(D_1-D_2)\cong\mL(D_1)\otimes\mL(D_2)^{-1}$$
\end{prp}

\begin{prp}{}{}\\
Let $X$ be a scheme. Then the map $$\text{CaCl}(X)\to\text{Pic}(X)$$ defined by $[D]\mapsto\mL(D)$ is an injective group homomorphism. 
\end{prp}

When $X$ is integral, Cartier divisors and the Picard group is entirely the same invariant for $X$. 

\begin{prp}{}{}\\
Let $X$ be a scheme. Then the there is an isomorphism of groups $$\text{CaCl}(X)\cong\text{Pic}(X)$$ defined by $[D]\mapsto\mL(D)$. 
\end{prp}

\begin{crl}{}{} Let $X$ be a scheme. If $X$ is Noetherian, integral, separated and that all local rings are UFDs, then the above isomorphism $$\text{Cl}(X)\cong\text{Pic}(X)$$ is natural in $X$. 
\end{crl}

\begin{crl}{}{} Let $X=\Prj_k^n$ be the projective space over some field $k$. Then every invertible sheaf on $X$ is isomorphic to $\mO_X(m)$ for some $m\in\Z$. 
\end{crl}

\section{Classifying Maps to Projective Space}
\subsection{Base Points}
\subsection{Relation to Invertible Sheaves}

\begin{defn}{The Jacobian Matrix at a Point}{}\\
Let $k$ be a field. Let $X$ be a scheme of finite type over $k$. Let $p\in X(k)$ be a $k$-valued point. Let $\text{Spec}\left(\frac{k[x_1,\dots,x_n]}{(f_1,\dots,f_r)}\right)$ be an affine neighbourhood containing $p$. Define the Jacobian matrix of $X$ at $p$ to be the matrix $$J_pX=\begin{pmatrix}
\frac{\partial f_1}{\partial x_1}\bigg{|}_p & \cdots & \frac{\partial f_1}{\partial x_n}\bigg{|}_p\\
\vdots & \ddots & \vdots\\
\frac{\partial f_r}{\partial x_1}\bigg{|}_p & \cdots & \frac{\partial f_r}{\partial x_n}\bigg{|}_p
\end{pmatrix}$$
\end{defn}

\subsection{Smooth Morphisms}
\begin{defn}{Smooth Morphisms}{}\\
Let $k$ be a field. Let $X,Y$ be schemes of finite type over $k$. Let $f:X\to Y$ be a morphism over $k$. We say that $f$ is smooth of relative dimension $n$ if the following are true. 
\begin{itemize}
\item $f$ is flat. 
\item If $X'\subseteq X$ and $Y'\subseteq Y$ are irreducible components such that $f(X')\subseteq Y'$, then $\dim(X')=\dim(Y')+n$. 
\item For all $p\in X$, we have $$\dim_{k(p)}(\Omega_{X/Y}^1\otimes k(p))=n$$
\end{itemize}
\end{defn}

Notice that if $X$ is irreducible, then the third criterion is equivalent to saying that $\dim_k(\Omega_{X/Y}^1)=n$. 

\begin{prp}{}{}\\
Let $k$ be an algebraically closed field. Let $X,Y$ be schemes of finite type over $k$. Let $f:X\to Y$ be a morphism over $k$. Then the following are equivalent. 
\begin{itemize}
\item $f$ is smooth of relative dimension $\dim(X)-\dim(Y)$. 
\item $\Omega_{X/Y}^1$ is locally free of rank $\dim(X)-\dim(Y)$. 
\item For all closed points $p\in X$, the induced map $T_pX\to T_{f(p)}Y$ is surjective. 
\end{itemize}
\end{prp}

\begin{defn}{Smooth Schemes}{}\\
Let $k$ be a field. Let $X$ be a scheme of finite type over $k$. We say that $X$ is a smooth scheme if $X$ is smooth over $k$. 
\end{defn}

\begin{prp}{}{}\\
Let $k$ be a field. Let $X$ be a scheme of finite type over $k$. Then the following are true. 
\begin{itemize}
\item $X$ is smooth. 
\item For all $p\in X$, the open affine scheme $\text{Spec}\left(\frac{k[x_1,\dots,x_n]}{(f_1,\dots,f_r)}\right)$ containing $p$ is such that $\rank(J_pX)=n-\dim(X)$. 
\item For all closed points $p\in X$, the open affine scheme $\text{Spec}\left(\frac{k[x_1,\dots,x_n]}{(f_1,\dots,f_r)}\right)$ containing $p$ is such that $\rank(J_pX)=n-\dim(X)$. 
\end{itemize}
\end{prp}

\begin{lmm}{}{}\\
Let $k$ be a field. Let $X$ be a scheme of finite type over $k$. If $X$ is smooth, then $X$ is regular. 
\end{lmm}

The main difference between smooth schemes and regular schemes is that the Jacobian rank criterion is satisfied for all (closed) points for smooth schemes, while the same criterion is only necessary on $k$-valued points for regular schemes. \\

When base field is algebraically closed, $k$-valued points are the same as closed points, and so we have the following. 

\begin{prp}{}{}\\
Let $k$ be an algebraically closed field. Let $X$ be a scheme of finite type over $k$. Then the following are true. 
\begin{itemize}
\item $X$ is smooth if and only if $X$ is regular. 
\item Suppose that $X$ is irreducible and separated over $k$. Then $X$ is smooth if and only if $X$ is a regular variety over $k$. 
\end{itemize}
\end{prp}

\begin{defn}{The Order of a Function at a Subvariety}{}\\
Let $k$ be a field. Let $X$ be a scheme of finite type over $k$. Let $W$ be an irreducible subvariety of $X$. Let $f/g\in K(W)^\ast$ for $f,g\in\mO_X(W)$ and $g\neq 0$. Let $V\subseteq W$ be an irreducible subvariety of codimension $1$. Define the order of $f$ at $V$ to be $$\text{ord}_V(f/g)=l_{\mO_X(W)}\left(\frac{\mO_X(W)}{(f)}\right)-l_{\mO_X(W)}\left(\frac{\mO_X(W)}{(g)}\right)$$
\end{defn}

\begin{defn}{The Divisor of a Rational Function}{}\\
Let $k$ be a field. Let $X$ be a scheme of finite type over $k$. Let $W$ be an irreducible subvariety of $X$. Let $f\in K(W)^\ast$. Define the divisor of $f$ to be $$\text{div}(f)=\sum_{\substack{V\text{ is a codimension }1\\\text{irreducible subvariety of }W}}\text{ord}_V(f)\cdot V$$
\end{defn}

\begin{defn}{The Group of $n$-Cycles}{}\\
Let $k$ be a field. Let $X$ be a scheme of finite type over $k$. Let $n\in\N$. Define the group of $n$-cycles on $X$ to be the free abelian group $$Z_n(X)=\Z\langle V\;|\;\substack{V\text{ is an }n\text{-dimensional}\\\text{irreducible subvariety of }X}\rangle$$
\end{defn}

\begin{lmm}{}{}\\
Let $k$ be a field. Let $X$ be a scheme of finite type over $k$. Let $W$ be an irreducible subvariety of $X$. Let $f\in K(W)^\ast$. Then $\text{div}(f)\in Z_{\dim(W)-1}(X)$. 
\end{lmm}

\begin{defn}{Cycles Rationally Equivalent to Zero}{}\\
Let $k$ be a field. Let $X$ be a scheme of finite type over $k$. Let $V\in Z_n(X)$. We say that $V$ is rationally equivalent to $0$ if there exists finitely many $\dim(V)+1$-dimensional irreducible subvarieties $W_1,\dots,W_r$ of $X$ and $f_i\in K(W_i)^\ast$ such that $$V=\sum_{i=1}^n\text{div}(f_i)$$
\end{defn}

\begin{defn}{The Chow Group}{}\\
Let $k$ be a field. Let $X$ be a scheme of finite type over $k$. Define the $n$th Chow group of $X$ to be $$\text{CH}_n(X)=\frac{Z_n(X)}{\sim}$$ where we say that $V_1\sim V_2$ if $V_1-V_2$ is rationally equivalent to $0$. In this case, we say that $V_1$ and $V_2$ are rationally equivalent. 
\end{defn}
