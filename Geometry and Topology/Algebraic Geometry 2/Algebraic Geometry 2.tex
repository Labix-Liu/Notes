\section{Spectra and its Structure Sheaf}
\subsection{The Zariski Topology and Hilbert's Nullstellensatz for Spectra}
Recall that for $A$ a commutative ring, we defined $$\text{Spec}(A)=\{P\subseteq A\;|\;P\text{ is a prime ideal of }A\}$$ We can similarly define the notion of vanishing loci of $\text{Spec}(A)$. 

\begin{defn}{Zero Locus}{} Let $R$ be a commutative ring. Let $T\subseteq R$. Define the vanishing locus of $T$ to be $$\V^\text{S}(T)=\{p\in\text{Spec}(R)\;|\;T\subseteq p\}$$
\end{defn}

\begin{lmm}{}{} Let $R$ be a commutative ring. Then the following are true. 
\begin{itemize}
\item If $T\subseteq R$ is a subset and $I=(T)$ is the ideal generated by $T$, then $$\V^\text{S}(T)=\V^\text{S}(I)$$
\item If $I\subseteq R$ is an ideal of $R$, then $$\V^\text{S}(\sqrt{I})=\V^\text{S}(I)$$
\end{itemize}
\begin{proof}
Clearly since $T\subseteq I$, we have $\V^\text{S}(I)\subseteq\V^\text{S}(T)$. Conversely, suppose that $P\in\V^\text{S}(T)$. Then $T\subseteq P$. But $I=(T)$ is the smallest ideal containing $T$. Hence $(T)\subseteq P$. \\

Let $P\in\V^\text{S}(\sqrt{I})$. Then $I\subseteq\sqrt{I}\subseteq P$ and so $P\in\V^\text{S}(I)$. Conversely, suppose that $P\in\V^\text{S}(I)$. Then $I\subseteq P$. Since $\sqrt{I}$ is the intersection of all prime ideals containing $I$, we must have $\sqrt{I}\subseteq P$. Hence $P\in\V^\text{S}(\sqrt{I})$. 
\end{proof}
\end{lmm}

The proposition shows that we only need to concern ourselves with the zero set of radical ideals of $A$. 

\begin{lmm}{}{}\\
Let $R$ be a commutative ring. Let $I,J$ be ideals of $R$. Then the following are true. 
\begin{itemize}
\item If $I\subseteq J$ are ideals of $R$, then $\V^\text{S}(J)\subseteq\V^\text{S}(I)$. 
\item We have $\V^\text{S}(IJ)=\V^\text{S}(I\cap J)=\V^\text{S}(I)\cup\V^\text{S}(J)$. 
\end{itemize}
\begin{proof}\\
Let $P\in\V^\text{S}(I_2)$. Then $I_2\subseteq P$. Hence $I_1\subseteq P$ and $P\in\V^\text{S}(I_1)$. \\

From Commutative Algebra 1, we have $$\V^\text{S}(IJ)=\V^\text{S}(\sqrt{IJ})=\V^\text{S}(\sqrt{I\cap J})=\V^\text{S}(I\cap J)$$
\end{proof}
\end{lmm}

\begin{lmm}{}{} Let $A$ be a commutative ring. The following are true. 
\begin{itemize}
\item $\V^\text{S}(1)=\emptyset$
\item $\V^\text{S}(0)=\text{Spec}(A)$
\item Let $\{a_i\;|\;i\in I\}$ be a set of ideals of $A$, then $$\V^\text{S}\left(\sum_{i\in I}a_i\right)=\bigcap_{i\in I}\V^\text{S}(a_i)$$
\item Let $\{a_1,\dots,a_n\}$ be a finite set of ideals of $A$, then $$\V^\text{S}\left(\bigcap_{k=1}^na_k\right)=\bigcup_{k=1}^n\V^\text{S}(a_k)$$
\end{itemize}
\begin{proof}
The first two are clear. We have that 
\begin{align*}
P\in\V^\text{S}\left(\sum_{i\in I}a_i\right)&\iff\sum_{i\in I}a_i\subseteq P\\
&\iff a_i\subseteq P\text{ for all }i\in I\\
&\iff P\in\V^\text{S}(a_i)\text{ for all }i\in I\\
&\iff P\in\bigcap_{i\in I}\V^\text{S}(a_i)
\end{align*}~\\

It suffices to complete the case $n=2$ and by induction it is true for any finite number of ideals. 
We have that 
\begin{align*}
P\in\V^\text{S}(a_1\cap a_2)&\iff a_1\cap a_2\subseteq P\\
&\iff a_1a_2\subseteq a_1\cap a_2\subseteq P\\
&\iff a_1\subseteq P\text{ or }a_2\subseteq P\\
&\iff P\in\V^\text{S}(a_1)\cup\V^\text{S}(a_2)
\end{align*}
\end{proof}
\end{lmm}
It is easy to see that lemma 1.1.4 shows that the set of vanishing loci of a spectra defines a topology on spectra. 

\begin{defn}{Zariski Topology on Spec}{} Let $A$ be a commutative ring. Define the Zariski topology on $\text{Spec}(A)$ to be the topology where the closed sets are exactly sets of the form $\V^\text{S}(I)$ for $I\subseteq A$ an ideal of $A$. 
\end{defn}

\begin{eg}{}{} The Zariski topology on $\text{Spec}(\Z)$ is given as follows. 
\begin{itemize}
\item The points are given by $\text{Spec}(\Z)=\{(p)\;|\;p\text{ is a prime }\}\cup\{(0)\}$
\item The closed subsets are of the form $$\V^s((n))=\{(p)\in\text{Spec}(\Z)\;|\;p\text{ divides }n\}$$
\end{itemize}
\end{eg}

\begin{eg}{}{} The Zariski topology on $\text{Spec}(\C[x])$ is given as follows. 
\begin{itemize}
\item The points are given by $\text{Spec}(\C[x])=\{(x-a)\;|\;a\in\C\}\cup\{(0)\}$
\item The closed subsets are precisely of the form $$\V^s((p))=\{(x-a)\;|\;a\text{ is a root of }p\}$$ for $p\in\C[x]$ since $\C[x]$ is a PID. 
\end{itemize}
\end{eg}

\begin{defn}{Ideals from a Zero Locus}{}\\
Let $R$ be a commutative ring. Let $S\subseteq\text{Spec}(R)$ be a subset. Define the ideal of associated to $S$ to be $$\I^\text{S}(S)=\{f\in R\;|\;f\in P\text{ for all }P\in S\}$$
\end{defn}

\begin{lmm}{}{}\\
Let $R$ be a commutative ring. Then the following are true. 
\begin{itemize}
\item If $S\subseteq R$ is a subset, then $\I^\text{S}(S)$ is a radical ideal of $R$. 
\item If $S\subseteq T\subseteq R$, then $\I^\text{S}(T)\subseteq\I^\text{S}(S)$. 
\item If $S_1,S_2\subseteq R$, then $\I^\text{S}(S_1\cup S_2)=\I^\text{S}(S_1)\cap\I^\text{S}(S_2)$. 
\end{itemize}
\begin{proof}
Clearly we have $\I^\text{S}(S)=\bigcap_{P\in S}P$ and intersections of ideals is an ideal. Now let $f\in\sqrt{\I^\text{S}(S)}$. Then $f^r\in\I^\text{S}(S)$ for some $r\in\N$. This means that $f^r\in\bigcap_{P\in S}P$. Since all $P$ are prime ideals, $f^r\in P$ implies $f\in P$. Hence $f\in\I^\text{S}(S)$ and so $\I^\text{S}(S)$ is radical. \\

Since $S\subseteq T$, we must have $$\I^\text{S}(T)=\bigcap_{P\in T}\subseteq\bigcap_{P\in S}P=\I^\text{S}(S)$$
\end{proof}
\end{lmm}

\begin{prp}{Hilbert's Nullstellensatz for Spectra}{}\\
Let $R$ be a commutative ring. Then the following are true. 
\begin{itemize}
\item If $I$ is an ideal of $R$, then $\I^\text{S}(\V^\text{S}(I))=\sqrt{I}$. 
\item If $S\subseteq\text{Spec}(R)$ is a subset, then $\V^\text{S}(\I^\text{S}(S))=\overline{S}$. 
\end{itemize}
\begin{proof}
Let $I$ be an ideal of $R$, then we have $$\I^\text{S}(\V^\text{S}(I))=\bigcap_{P\in\V^\text{S}(I)}P=\bigcap_{I\subseteq P\in\text{Spec}(R)}P=\sqrt{I}$$ Now let $S\subseteq\text{Spec}(R)$ be a subset. Notice that a closed subset $T=\V^\text{S}(I)$ for some ideal $I\subseteq R$ contains $S$ if and only if for all $P\in S$, $I\subseteq P$. And this is true if and only if $I\subseteq\I^\text{S}(S)$. Since the largest ideal contained by $\I^\text{S}(S)$ is itself, the smallest closed subset containing $S$ must be $\V^\text{S}(\I^\text{S}(S))$. Hence $\overline{S}=\V^\text{S}(\I^\text{S}(S))$. 
\end{proof}
\end{prp}

\begin{lmm}{}{}\\
Let $R$ be a commutative ring. Let $S\subseteq\text{Spec}(R)$ be a subset. Then we have $$\I^\text{S}(\overline{S})=\I^\text{S}(S)$$
\begin{proof}
Since $S\subseteq T$, we must have $$\I^\text{S}(T)=\bigcap_{P\in T}\subseteq\bigcap_{P\in S}P=\I^\text{S}(S)$$
We have $$\I^\text{S}(\overline{S})=\I^\text{S}(\V^\text{S}(\I^\text{S}(S)))=\I^\text{S}(S)$$ where the first equality comes from item 4 and the second equality comes from item 3. 
\end{proof}
\end{lmm}

The lemma shows that we only need to concern ourselves with closed subsets of $\text{Spec}(R)$ as the input of $\I^\text{S}(-)$. 

\begin{prp}{}{}\\
Let $R$ be a commutative ring. Then the following are true. 
\begin{itemize}
\item There is an inclusion reversing bijection $$\left\{\substack{\text{Radical ideals}\\\text{of }R}\right\}\;\;\overset{\text{1:1}}{\longleftrightarrow}\;\;\left\{\substack{\text{Closed Subsets of}\\\text{Spec}(R)}\right\}$$ induced by $\V^\text{S}(-)$ and $\I^\text{S}(-)$. 
\item There is an inclusion reversing bijection $$\text{Spec}(R)\;\;\overset{\text{1:1}}{\longleftrightarrow}\;\;\left\{\substack{\text{Irreducible Closed Subsets}\\ \text{of Spec}(R)}\right\}$$ induced by $\V^\text{S}(-)$ and $\I^\text{S}(-)$. 
\item There is an inclusion reversing bijection $$\left\{\substack{\text{Minimal prime}\\\text{ideals of }R}\right\}\;\;\overset{\text{1:1}}{\longleftrightarrow}\;\;\left\{\substack{\text{Irreducible Components}\\ \text{of Spec}(R)}\right\}$$ induced by $\V^\text{S}(-)$ and $\I^\text{S}(-)$. 
\end{itemize}
\begin{proof}\\
The first inclusion reversing bijection comes from 1.1.9. \\

Let $P\in\text{Spec}(R)$ be a point. I claim that $\V^\text{S}(P)$ is irreducible. Indeed, $\V^\text{S}(P)=\text{Spec}(R/P)$ and $R/P$ being an integral domain implies that $\V^\text{S}(P)$ is irreducible. Conversely, let $S$ be an irreducible closed subset of $\text{Spec}(R)$. I claim that $\I^\text{S}(S)$ is a point. Since $S$ is closed, $S=\V^\text{S}(I)$ for some radical ideal $I$ of $R$, and so $\I^\text{S}(S)=I$. I claim that $I$ is a prime ideal. Let $fg\in I$. Then $S=\V^\text{S}(I)\subseteq\V^\text{S}(fg)=\V^\text{S}(f)\cup\V^\text{S}(g)$ and we have $$S=\left(\V^\text{S}(f)\cap S\right)\cup\left(\V^\text{S}(g)\cap S\right)$$ Since $S$ is irreducible, without loss of generality suppose that $S=\V^\text{S}(f)\cap S$. Let $P\in S=\V^\text{S}(I)$, then $P\in\V^\text{S}(f)$ implies that $f\in P$. Thus we have $f\in\bigcap_{P\supseteq I}P=\sqrt{I}=I$. Hence $I$ is prime. \\

Since $\V^\text{S}$ and $\I^\text{S}$ are inclusion reversing, the smallest primes ordered by inclusion is in bijection to the largest irreducible closed subsets ordered by inclusion. 
\end{proof}
\end{prp}

\subsection{Topological Properties of Spectra}
\begin{lmm}{}{} Let $k$ be an algebraically closed field. Let $V$ be an affine variety. Then the bijection given by the Hilbert's nullstellensatz $$V\cong\text{maxSpec}(k[V])$$ is a homeomorphism where $\text{maxSpec}(k[V])$ inherits the subspace topology of $\text{Spec}(k[V])$. 
\end{lmm}

Let $k$ be an algebraically closed field. For a closed subset $V\subseteq\A_k^n$, we have the following. 

\begin{itemize}
\item There is a bijection $$V\;\;\overset{1:1}{\longleftrightarrow}\;\;\text{maxSpec}(V)\subseteq\text{Spec}(V)$$ The geometry of $V$ is enhanced because we have more points in general, given by the irreducible subvarieties. 
\item In the Zariski topology of $V$, closed subsets are affine subvarieties of $V$. In the Zariski topology of $\text{Spec}(V)$, closed subsets are collections of irreducible subvarieties. 
\item For a closed subset $V$ of $\text{Spec}(k[V])$, we have $C=\V^s(I)$ for some $I\subseteq k[V]$ an ideal. There is a bijection of sets $$C=\V^s(I)\;\;\overset{1:1}{\longleftrightarrow}\;\;\{W\subseteq\V(I)\;|\;W\text{ is irreducible }\}$$
\item In particular, for an irreducible polynomial $f\in k[x_1,\dots,x_n]$, $(f)\in\text{Spec}(k[V])$ is a point. The closure of $(f)$ is given by $$\overline{(f)}=\{m\in\text{maxSpec}(k[V])\;|\;(f)\subseteq m\}\cup\{(f)\}$$
\end{itemize}

\begin{prp}{}{}\\
Let $R$ be a commutative ring. Let $I\subseteq R$ be an ideal of $R$. Then there is natural homeomorphism $$\text{Spec}(R/I)=\V^\text{S}(I)$$ induced by the correspondence theorem for ideals. 
\begin{proof}\\
By the correspondence theorem for ideals, the two sets are equal. Now let $\V^\text{S}(J/I)$ be a closed subset of $\text{Spec}(R/I)$ where $J$ is an ideal of $R$ (all ideals of $R/I$ are of this form by the correspondence theorem). Then we have $$\V^\text{S}(T)=\{P\in\text{Spec}(R/I)\;|\;T\subseteq P\}\overset{1:1}{\longleftrightarrow}\{P\in\text{Spec}(R)\;|\;J\subseteq P\}=\V^\text{S}(J)$$ Thus the closed sets of the two spaces coincide via the same bijection, namely that given by the correspondence theorem. 
\end{proof}
\end{prp}

We can also explicitly write out the open sets and a basis for the Zariski topology. 

\begin{defn}{Distinguished Open Sets}{}\\
Let $R$ be a commutative ring. Let $S\subseteq R$ be a subset. Define the distinguished open set of $S$ to be $$D(S)=\text{Spec}(R)\setminus\V^S(S)$$ If $S=\{f\}$, then we say that $D(S)$ is a basic open set. 
\end{defn}

\begin{lmm}{}{}\\
Let $R$ be a commutative ring. Then the basic open sets form a basis for the Zariski topology on $\text{Spec}(R)$. 
\begin{proof}\\
Let $S\subseteq R$ be a subset. Then we have 
\begin{align*}
D(S)&=\text{Spec}(R)\setminus\V^\text{S}(S)\\
&=\text{Spec}(R)\setminus\V^\text{S}\left(\sum_{f\in S}(f)\right)\\
&=\text{Spec}(R)\setminus\bigcap_{f\in S}\V^\text{S}((f))\\
&=\bigcup_{f\in S}(D(f))
\end{align*}
and so we are done. 
\end{proof}
\end{lmm}

\begin{lmm}{}{}\\
Let $R$ be a commutative ring. Let $f\in R$. Then there is a natural homeomorphism $$D(f)=\text{Spec}(R_f)$$ induced by the correspondence theorem for prime ideals of localization. 
\begin{proof}
By the correspondence of prime ideals of localization, we have $$\text{Spec}(R_f)\overset{1:1}{\longleftrightarrow}\{P\in\text{Spec}(R)\;|\;P\cap\{f^n\;|\;n\in\N\}\}=D(f)$$ so we are done. Moreover, for any $T\subseteq R$ and $f\notin T$, we have $$D(f)\supseteq\V^\text{S}((T))=\{P\in\text{Spec}(R_f)\;|\;(T)_f\subseteq P\}=\V^\text{S}((T)_f)$$
\end{proof}
\end{lmm}

We will prove some topological properties of spectra. Recall the following notions. 
\begin{itemize}
\item A space $X$ is irreducible if it is not the union of two proper closed subsets. Every space can be uniquely decomposed into its irreducible components, all of which are closed. 
\item A space $X$ is connected if it is not the disjoint union of two open sets. Every space can be uniquely decomposed into its connected components, all of which are closed (but not necessarily open in $X$). Moreover, irreducible spaces are clearly connected (but the converse is not true). 
\item A space $X$ is compact if any open cover has a finite subcover. 
\item A space $X$ is Noetherian if for any descending chain of closed sets $Z_1\supseteq Z_2\supseteq\cdots\supseteq Z_k\supseteq\cdots$, there exists $n\in\N$ such that $Z_n=Z_{n+1}=Z_{n+2}=\cdots$. The Noetherianess property guarantees that the decomposition into irreducible components is finite. 
\end{itemize}

\begin{prp}{}{}\\
Let $R$ be a commutative ring. Then the following are true. 
\begin{itemize}
\item $R$ is an integral domain if and only if $\text{Spec}(R)$ is irreducible. 
\item $\text{Spec}(R)$ is connected if and only if $R$ is not a product of rings. 
\item $\text{Spec}(R)$ is compact. 
\item If $R$ is Noetherian, then $\text{Spec}(R)$ is Noetherian. 
\end{itemize}
\begin{proof}\\
\begin{itemize}
\item Suppose that $\text{Spec}(R)$ is reducible. Then there exists ideals $I,J\neq(0)$ of $R$ such that $\text{Spec}(R)=\V^\text{S}(I)\cup\V^\text{S}(J)=\V^\text{S}(I\cap J)$. Since $P\supseteq I\cap J$ for all prime ideals $P$ of $R$, we have $(0)\neq IJ\subseteq I\cap J\subseteq N(R)$. Thus $N(R)$ is non-zero and so $R$ is not an integral domain. \\

Now suppose that $\text{Spec}(R)$ is irreducible. Let $f,g\in R$ such that $fg=0$. Then we have $$\V^\text{S}((f))\cup\V^\text{S}((g))=\V^\text{S}((fg))=\V^\text{S}((0))=\text{Spec}(R)$$ Since $\text{Spec}(R)$ is irreducible, without loss of generality suppose that $\V^\text{S}((f))=\text{Spec}(R)$. Then $f\in P$ for all prime ideals $P$ of $R$. Hence $f\in N(R)=\sqrt{(0)}$, so $f^n=0$ and $f=0$. 
\item Suppose that $R=S_1\times S_2$ is a product of rings. Then there is a natural isomorphism $\text{Spec}(R)\cong\text{Spec}(S_1)\amalg\text{Spec}(S_2)$ induced by the correspondence theorem for ideals. Hence $R$ is not connected. Suppose that $R$ is not connected. Then there exists non trivial proper open subsets $D(I),D(J)$ such that $\text{Spec}(R)=D(I)\amalg D(J)=\V^\text{S}(J)\amalg\V^\text{S}(I)$. 
\item Since $\{D(f)\;|\;f\in R\}$ is a basis for the Zariski topology, it suffices to consider an open cover consisting of these basis elements. So let $\{D(f_i)\;|\;i\in I\}$ be an open cover of $\text{Spec}(R)$. Then we have $\text{Spec}(R)=\bigcup_{i\in I}D(f_i)$ implies $$\emptyset=\bigcap_{i\in I}\V^\text{S}(f_i)$$ This means that for all $P\in\text{Spec}(R)$, there exists $i\in I$ such that $f_i\notin P$ and $(f_i)\not\subseteq P$. So there is no prime ideal $P$ that contains $\sum_{i\in I}(f_i)$. In particular, there is no maximal ideal that contains this ideal. Hence $\sum_{i\in I}(f_i)=(1)$. By definition of sums of ideals, there exists $x_i\in R$ non-zero for all but finitely many, such that $\sum_{i\in I}x_if_i=1$. So there exists $x_1,\dots,x_n\in R$ and $f_1,\dots,f_n$ such that $\sum_{i=1}^nx_if_i=1$. \\

Suppose for a contradiction that $\emptyset\neq\bigcap_{i=1}^n\V^\text{S}(f_i)$, then there exists a prime ideal $P$ of $R$ such that $f_i\in P$. Hence $\sum_{i=1}^n(f_i)\subseteq P$. But since $\sum_{i=1}^nx_if_i=1$, we have $1\in P$ and this is a contradiction. Hence we must have $\emptyset=\bigcap_{i=1}^n\V^\text{S}(f_i)$. In other words, $\text{Spec}(R)=\bigcup_{i=1}^nD(f_i)$ and so we have found a finite subcover for the given open cover. 

\item Suppose that $R$ is Noetherian. Let $$\V^\text{S}(I_1)\supseteq\V^\text{S}(I_2)\supseteq\cdots\supseteq\V^\text{S}(I_k)\supseteq\cdots$$ be a decreasing sequence of closed subsets of $\text{Spec}(R)$. Then applying the inclusion reversing bijection $\I^\text{S}(-)$ gives an ascending chain of ideals $$\sqrt{I_1}\subseteq\sqrt{I_2}\subseteq\cdots\subseteq\sqrt{I_k}\subseteq\cdots$$ of $R$. Since $R$ is Noetherian, there exists $n\in\N$ such that $\sqrt{I_n}=\sqrt{I_{n+t}}$ for all $t\in\N$. This means that $$\V^\text{S}(I_n)=\V^\text{S}(\sqrt{I_n})=\V^\text{S}(\sqrt{I_{n+t}})=\V^\text{S}(I_{n+t})$$ Thus $\text{Spec}(R)$ is Noetherian. 
\end{itemize}
\end{proof}
\end{prp}

\subsection{The Structure Sheaf of a Spectrum}
We now define the structure sheaf on a spectrum so that they form a locally ringed space. 

\begin{defn}{Structure Sheaf}{}\\
Let $R$ be a commutative ring. Define the structure sheaf $$\mO_{\text{Spec}(R)}:\bold{Open}(\text{Spec}(R))\to\bold{Rings}$$ on $\text{Spec}(R)$ by the following. 
\begin{itemize}
\item For each $U\subseteq\text{Spec}(R)$ open, define $$\mO_{\text{Spec}(R)}(U)=\left\{(s_p)_{p\in U}\in\prod_{p\in U}R_p\;\bigg{|}\;\substack{\forall p,\exists p\in\;U_p\subseteq U\text{ open and }a,f\in R\text{ s.t.}\\q\in U_p\text{ implies }f\notin q\text{ and }s_q=a/f\in A_q}\right\}$$
\item For $V\subseteq U$ an inclusion, the unique morphism $\mO_{\text{Spec}(R)}(U)\to\mO_{\text{Spec}(R)}(V)$ is the projection map $$\text{res}_V^U:\prod_{p\in U}R_p\to\prod_{p\in V}R_p$$
\end{itemize}
\end{defn}

\begin{prp}{}{}\\
Let $R$ be a commutative ring. Then the structure sheaf $$\mO_{\text{Spec}(R)}:\bold{Open}(\text{Spec}(R))\to\bold{Set}$$ defined above is indeed a sheaf on $\text{Spec}(R)$. 
\begin{proof}\\
Identity: Let $U\subseteq\text{Spec}(R)$ be an open set. Let $\{U_i\;|\;i\in I\}$ be an open cover of $U$. Suppose that $(s_p)_{p\in U},(t_p)_{p\in U}\in\mO_{\text{Spec}(R)}(U)$ are such that $(s_p)_{p\in U_i}=(t_p)_{p\in U_i}$ for all $i\in I$. Since the $U_i$ cover $U$, we check that for all $p\in U$, we have $s_p=t_p$ so that $(s_p)_{p\in U}=(t_p)_{p\in U}$. \\

Gluing: Let $U\subseteq\text{Spec}(R)$ be an open set. Let $\{U_i\;|\;i\in I\}$ be an open cover of $U$. Suppose that $((s_i)_p)_{p\in U_i}$ for ranging $i$ is a collection of sections such that $((s_i)_p)_{p\in U_i\cap U_j}=((s_j)_p)_{p\in U_i\cap U_j}$. Define a section in $\mO_{\text{Spec}(R)}(U)$ by the collection $t_p=(s_i)_p$ if $p\in U_i$. Since the collection $((s_i)_p)_{p\in U_i}$ agrees on intersections, $(t_p)_{p\in U}$ is well defined, and moreover restricts to $((s_i)_p)_{p\in U_i}$ for each $i$. Hence we are done.
\end{proof}
\end{prp}

The structure sheaf allows $\text{Spec}(R)$ to be a ringed space. The ringed space on any spectrum is in fact a locally ringed space. But this is not true for the ringed space on varieties in the classical sense. 

\begin{prp}{}{}\\
Let $R$ be a commutative ring. Then the following are true regarding the ringed space $(\text{Spec}(R),\mO_{\text{Spec}(R)})$. 
\begin{itemize}
\item For any $p\in\text{Spec}(R)$, there is a natural isomorphism $\mO_{\text{Spec}(R),p}\cong R_p$ on the level of stalks. 
\item $(\text{Spec}(R),\mO_{\text{Spec}(R)})$ is a locally ringed space. 
\item For any element $f\in R$, there is an isomorphism $\mO_{\text{Spec}(R)}(D(f))\cong R_f$
\item There is an isomorphism $\mO_{\text{Spec}(R)}(\text{Spec}(R))\cong R$ on the global level. 
\end{itemize} 
\begin{proof}
\begin{itemize}
\item Let $p\in\text{Spec}(R)$. For each open set $U\subseteq X$ containing $p$, there is an evident ring homomorphism $\mO_\text{Spec}(R)(U)\to R_p$ that is projection. By the universal property of colimits, this induces a ring homomorphism $\mO_{\text{Spec}(R),p}\to R_p$ defined by $[((s_q)_{q\in V},V)]\mapsto s_p$. It remains to show that this map is a bijection. Suppose that $[((s_q)_{q\in V},V)]$ is such that $s_p=0\in R_p$. Then there exists an open subset $p\in W\subseteq V$ and $f,g\in R$ such that for all $q\in W$, $g\notin q$ and $s_q=f/g$. In particular this means that $s_p=f/g=0$. This is equivalent to there exists $h\in R\setminus p$ such that $hf=0$. Let $S=D(g)\cap D(h)\cap W$. Notice that $p\in S$. Let $q\in S$. In $R_q$, we have $g,h\notin q$ and so $hf=0$ implies that $0=f/g=s_q$ in $R_q$. Hence $(s_q)_{q\in S}=0$. Then we have $[((s_q)_{q\in V},V)]=[((s_q)_{q\in S},S)]=0$ so that the ring homomorphism is injective. For surjectivity, let $s_p\in R_p$. Then there exists $f,g\in R$ and $g\notin p$ such that $s_p=f/g$. Define a section $(f/g)_{p\in D(g)}\in\mO_{\text{Spec}(R)}(D(g))$. It is a well defined element because on the open set $D(g)$, for any $q\in D(g)$ we clearly have $s_q=f/g\in R_q$ so that the compatibility requirement is satisfied. Evidently, we have $[((f/g)_{p\in D(g)},D(g))]\in\mO_{\text{Spec}(R),p}$ is mapped to $s_p=f/g\in R_p$. Hence the ring homomorphism is surjective. 

\item From the above we immediately conclude that every stalk of the ringed space is a local ring. Hence $(\text{Spec}(R),\mO_{\text{Spec}(R)})$ is a locally ringed space. 

\item Define a map $\phi:R_f\to\mO_{\text{Spec}(R)}(D(f))$ by $\phi(r/f^n)=(r/f^n)_{q\in D(f)}$. It is clear that it is a ring homomorphism. It remains to show that the map is bijective. Suppose that $(r/f^n)_{q\in D(f)}=(s/f^m)_{q\in D(f)}$. Then for each $q\in D(f)$, we have $r/f^n=s/f^m\in R_q$. This means that there exists $h\in R\setminus q$ such that $h(rf^m-sf^n)=0$. Now notice that $h\in\text{Ann}_R(rf^m-sf^n)$ and $h\notin q$. Hence $\text{Ann}_R(rf^m-sf^n)\notin q$. Since this is true for all $q\in D(f)$, we must have 
\begin{align*}
\V^\text{S}(\text{Ann}_R(rf^m-sf^n))\cap D(f)&=\emptyset\\
\V^\text{S}(\text{Ann}_R(rf^m-sf^n))\setminus\V^\text{S}((f))&=\emptyset
\end{align*}
and so $\V^\text{S}(\text{Ann}_R(rf^m-sf^n))\subseteq\V^\text{S}((f))$. Taking $\I^\text{S}(-)$ gives $(f)\subseteq\sqrt{(f)}\subseteq\sqrt{\text{Ann}_R(rf^m-sf^n)}$. Then $f\in(f)$ means $f\in\sqrt{\text{Ann}_R(rf^m-sf^n)}$ so there exists $k\in\N$ for which $f^k(rf^m-sf^n)=0$. Thus we have $r/f^n=s/f^m$ in $R_f$. Hence our map is injective. \\

For surjectivity, let $(s_p)_{p\in D(f)}\in\mO_{\text{Spec}(R)}(D(f))$. For each $p\in D(f)$, there exists an open neighbourhood $V_p\subseteq D(f)$ of $p$ and $r_p,g_p\in R$, $g_p\notin p$ such that $s_q=r_p/g_p$ for all $q\in V_p$. Notice that the open sets $\{D(g_p)\;|\;p\in D(f)\}$ cover $D(f)$. Now I claim that this open cover has a finite subcover. Indeed, we know that $D(f)\subseteq\bigcup_{p\in D(f)}D(g_p)$. Taking complements, we have $\bigcap_{p\in D(f)}\V^\text{S}(g_p)\subseteq\V^\text{S}(f)$. Notice that the intersection is equal to $\V^\text{S}(\sum_{p\in D(f)}(g_p))$. Taking $\I^\text{S}(-)$ on both sides give $\sqrt{(f)}\subseteq\sqrt{\sum_{p\in D(f)}(g_p)}$. So there exists $k\in\N$ such that $f^k\in\sum_{p\in D(f)}(g_p)$. By definition of sum of ideals, there exists a finite indexing set $I$ and $a_i\in R$ such that $f^k=\sum_{i\in I}a_ig_i$. Then $f^k\in\sum_{i\in I}(g_i)$, and taking $\V^\text{S}(-)$ gives $\bigcap_{i\in I}\V^\text{S}(g_i)\subseteq\V^\text{S}(f)$ and taking complements give $D(f)\subseteq\bigcup_{i\in I}D(g_i)$. Hence $D(f)\subseteq D(g_1)\cup\cdots\cup D(g_r)$ has a finite subcover. On the open set $D(g_ig_j)=D(g_i)\cap D(g_j)$, for any $q\in D(g_ig_j)$, $s_q=r_i/g_i$ on $D(g_i)$ and $s_q=r_j/g_j$ on $D(g_j)$ so $r_i/g_i=s_q=r_j/g_j$ in $D(g_ig_j)$. Since we have just proven that the map $R_{g_ig_j}\to\mO_{\text{Spec}(R)}(D(g_ig_j))$ is injective, we have $r_i/g_i=r_j/g_j\in R_{g_ig_j}$. So there exists $n_{i,j}\in\N$ such that $(g_ig_j)^{n_{i,j}}(r_ig_j-r_jg_i)=0$. Let $n=\max\{n_{i,j},1\leq i,j\leq r\}$. Then for all $1\leq i,j\leq r$, we have $(g_ig_j)^n(r_ig_j-r_jg_i)=0$. Let $r_i'=r_ig_i^n$ and $g_i'=g_i^{n+1}$. Then rewriting the equation $(g_ig_j)^n(r_ig_j-r_jg_i)=0$ gives $$g_j'r_i'-g_i'r_j'=0$$ Notice that we still have $s_q=\frac{r_i'}{g_i'}$ for all $q\in D(g_i)$. Now since $D(g_i)=D(g_i')$ for $i\in I$ covers $D(f)$, there exists $k\in\N$ and $a_i\in R$ such that $f^k=\sum_{i\in I}a_ig_i'$ similar to what we did above to find the finite subcover. Let $r=\sum_{i\in I}a_ir_i'$. Then we have $$g_j'r=\sum_{i\in I}a_ir_i'g_j'=\sum_{i\in I}a_ir_j'g_i'=f^kr_j'$$ Then for all $j\in I$, we have $\frac{r}{f^k}=\frac{r_j}{g_j'}$ on $D(g_j)=D(g_j')$. Thus we have that the element $\frac{r}{f^k}$ maps to our given element. Thus we are done. 

\item Apply the above by $f=1_R$. 
\end{itemize}
\end{proof}
\end{prp}

\subsection{The Spec Functor}
\begin{defn}{Induced Map on Spectrum}{} Let $R,S$ be commutative rings. Let $\phi:R\to S$ be a ring homomorphism. Define the function $$\phi^\flat:\text{Spec}(S)\to\text{Spec}(R)$$ by the formula $P\mapsto\phi^{-1}(P)$
\end{defn}

\begin{prp}{}{} Let $A,B,C$ be commutative rings. Let $\phi:A\to B$ and $\psi:B\to C$ be ring homomorphisms. Then the following are true. 
\begin{itemize}
\item $(\psi\circ\phi)^\flat=\phi^\flat\circ\psi^\flat$
\item $(\text{id}_A)^\flat=\text{id}_{\text{Spec}(A)}$
\end{itemize}
\end{prp}

\begin{lmm}{}{}\\
Let $R,S$ be commutative rings. Let $\phi:R\to S$ be a ring homomorphism.  Then the induced map $$\phi^\flat:\text{Spec}(S)\to\text{Spec}(R)$$ is continuous. 
\begin{proof}\\
Let $T=\V^\text{S}(I)$ be a closed set in $\text{Spec}(R)$. Then $(\phi^\flat)^{-1}(T)=\{P\in\text{Spec}(S)\;|\;\phi^{-1}(P)=\phi^\flat(P)\in T\}$. I claim that this set is equal to $\V^\text{S}(\phi(I))$. Let $P\in(\phi^\flat)^{-1}(T)$. Then $\phi^{-1}(P)\in T$ implies $I\subseteq\phi^{-1}(P)$ which means that $\phi(I)\subseteq P$. Hence $P\in\V^\text{S}(\phi(I))$. Now suppose that $P\in\V^\text{S}(\phi(I))$. Then $\phi(I)\subseteq P$ implies that $I\subseteq\phi^{-1}(\phi(I))\subseteq\phi^{-1}(P)$. Hence $\phi^{-1}(P)\in T$. Thus the preimage of a closed set is closed. 
\end{proof}
\end{lmm}

\begin{prp}{}{} Let $V,W$ be affine varieties over $\C$. Let $F:V\to W$ be a morphism of affine varieties. Let $m_x$ be the maximal ideal of $\C[V]$ corresponding to the point $x$. Then $$(F^\ast)^\flat(m_x)=m_{F(x)}$$
\end{prp}

We thus now have the following nice diagram: \\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	{\text{maxSpec}(\C[V])} & & {\text{maxSpec}(\C[W])} \\
	V & & W
	\arrow["{\phi^\flat|_{\text{maxSpec}(\C[V])}}", from=1-1, to=1-3]
	\arrow["{1:1}"', leftrightarrow, from=1-1, to=2-1]
	\arrow["{1:1}", leftrightarrow, from=1-3, to=2-3]
	\arrow["\phi"', from=2-1, to=2-3]
\end{tikzcd}}\\

\begin{prp}{}{} Let $V$ be an affine variety over $\C$. Let $X\subseteq V$ be an affine irreducible subvariety. Let $P_X$ be the prime ideal of $\C[V]$ corresponding to $X$. Let $Y=\overline{F(X)}$. Then $$(F^\ast)^\flat(P_X)=P_Y$$
\end{prp}

\begin{eg}{}{} Let $\sigma:\C[x]\to\C[[t]]$ be a map. Let $\iota:\C[[t]]\to\C((t))$ be the inclusion map. Consider the following diagram: \\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	{\C[x]} && {\C((t))} \\
	& {\C[[t]]} \\
	\substack{\text{Spec}(\C[x])=\\\{(0)\}\cup\{(x-a)\;|\;a\in\C\}} && {\text{Spec}(\C((t)))=\{(0)\}} \\
	& {\text{Spec}(\C[[t]])=\{(0),(t)\}}
	\arrow["\sigma", from=1-1, to=1-3]
	\arrow["\iota"', hook, from=2-2, to=1-3]
	\arrow["{\sigma^\flat}"', from=3-3, to=3-1]
	\arrow["{\iota^\flat}", from=3-3, to=4-2]
	\arrow["\rho", dashed, from=4-2, to=3-1]
\end{tikzcd}}\\
\begin{itemize}
\item Let $\sigma$ be the map $x\mapsto t$. Then $\rho$ exists. 
\item Let $\sigma$ be the map $x\mapsto t+1$. Then $\rho$ exists. 
\item Let $\sigma$ be the map $x\mapsto 1/t$. Then $\rho$ does not exist. 
\item Let $\sigma$ be the map $x\mapsto t^2$. Then $\rho$ exists. 
\item Let $\sigma$ be the map $x\mapsto e^t$. Then $\rho$ exists. 
\end{itemize} 
\begin{proof}
We know that the $\text{Spec}$ functor is fully faithful. This means that $\rho$ exists if and only if there exists a map $\C[x]\to\C[[t]]$ making the top diagram commute. \\

For the first two cases, the images of the map lie inside $\C[[t]]$. Hence the map $\C[x]\to\C[[t]]$ exists, and hence $\rho$ exists. In the first case, we have $\rho((t))=(x)$ and in the second case we have $\rho((t))=(x-1)$. \\

For the third case, no map makes the top diagram commute hence $\rho$ does not exist. \\

For the fourth and fifth cases, the images of the map lie inside $\C[[t]]$. Hence $\rho$ exists. In the fourth case we have $\rho((t))=(x)$ and in the fifth case we have $\rho((t))=(x-1)$. Indeed, $f\in\rho^{-1}(t)$ if and only if $t$ divides $f(e^t)$. This means that $g(t)=f(e^t)$ has no constant term, or that $g(0)=0$. This means that $f(1)=0$, hence $x-1$ divides $f$. 
\end{proof}
\end{eg}

\begin{defn}{The Spec Functor}{} Define the spec functor $$\text{Spec}:\bold{CRing}\to\bold{LRS}$$ to consist of the following data. 
\begin{itemize}
\item A commutative ring $A$ is sent to the locally ringed space $(\text{Spec}(A),\mO_{\text{Spec}(A)})$. 
\item For a ring homomorphism $\phi:A\to B$, we have that $\text{Spec}(\phi)$ consists of a continuous map $\phi^\flat:\text{Spec}(B)\to\text{Spec}(A)$ and a morphism of sheaves $$(\phi^\flat)^\#:\mO_{\text{Spec}(A)}\to(\phi^\flat)_\ast\mO_{\text{Spec}(B)}$$ defined as follows. 
Fixing $p\in\text{Spec}(A)$, for each $q\in(\phi^\flat)^{-1}(p)$, $\phi$ induces a local homomorphism $\phi_{p,q}:A_p\to B_q$. Ranging over $q$, we obtain a map $$\phi_p:A_p\to\prod_{q\in(\phi^\flat)^{-1}(p)}B_q$$ For $U\subseteq\text{Spec}(A)$ an open set, ranging over $p\in U$ gives a map $$\prod_{p\in U}\phi_p:\prod_{p\in U}A_p\to\prod_{q\in(\phi^\flat)^{-1}(U)}B_q$$ The morpnism then sends $f\in\mO_{\text{Spec}(A)}(U)$ to the composite \\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	{(\phi^\flat)^{-1}(U)} & U & {\prod_{p\in U}A_p} && {\prod_{q\in(\phi^\flat)^{-1}(U)}B_q}
	\arrow["{\phi^\flat}", from=1-1, to=1-2]
	\arrow["f", from=1-2, to=1-3]
	\arrow["{\prod_{p\in U}\phi_p}", from=1-3, to=1-5]
\end{tikzcd}} \\
\end{itemize}
\end{defn}

\begin{lmm}{}{}\\
Let $R,S$ be commutative rings. Let $\varphi:R\to S$ be a ring homomorphism. Let $f\in R$. Then the morphism $$R_f\cong\mO_{\text{Spec}(R)}(D(f))\to\mO_{\text{Spec}(S)}((\varphi^\flat)^{-1}(D(f)))=\mO_{\text{Spec}(S)}(D(\varphi(f)))\cong S_{\varphi(f)}$$ is the induced morphism of localization $\varphi_f:R_f\to S_{\varphi(f)}$. 
\begin{proof}\\
For any $p\in D(f)$ and $q\in(\varphi^\flat)^{-1}(p)$, notice that the localization morphism $\varphi_p:R_p\to S_q$ is defined by $r/s\mapsto\varphi(r)/\varphi(s)$. Since the following diagram commtues \\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	{R_f} &&&&& {S_{\varphi(f)}} \\
	{\mO_{\text{Spec}(R)}(D(f))} &&&&& {\mO_{\text{Spec}(S)}(D(\varphi(f)))}
	\arrow["{\frac{r}{f^n}\mapsto\frac{\varphi(r)}{\varphi(f)^n}}", from=1-1, to=1-6]
	\arrow["\cong", from=1-1, to=2-1]
	\arrow["\cong"', from=1-6, to=2-6]
	\arrow["{\left(\frac{r}{f^n}\right)_{p\in D(f)}\mapsto\left(\frac{\varphi(r)}{\varphi(f)^n}\right)_{q\in D(\varphi(f))}}", from=2-1, to=2-6]
\end{tikzcd}} \\
by naturality of the vertical isomorphisms, we check that our morphism is indeed in its desired form. 
\end{proof}
\end{lmm}

\begin{prp}{}{}\\
The spec functor is fully faithful. 
\begin{proof}\\
Clearly the functor is injective by since there is a natural isomorphism $\mO_{\text{Spec}(R)}(\text{Spec}(R))\cong R$. For surjectivity, suppose that $(\varphi,\varphi^\#):(\text{Spec}(S),\mO_{\text{Spec}(S)})\to(\text{Spec}(R),\mO_{\text{Spec}(R)})$ is a morphism of spectra. Let $q\in\text{Spec}(S)$. By the universal property of colimits and 1.3.3, the following diagram on the left can be simplified to the diagram on the right: \\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	{\mO_{\text{Spec}(R)}(\text{Spec}(R))} & {\mO_{\text{Spec}(S)}(S)} && R & S \\
	{\mO_{\text{Spec}(R),\varphi(q)}} & {\mO_{\text{Spec}(S),q}} && {R_{\varphi(q)}} & {S_q}
	\arrow["{\varphi^\#(\text{Spec}(R))}", from=1-1, to=1-2]
	\arrow[from=1-1, to=2-1]
	\arrow[from=1-2, to=2-2]
	\arrow[from=1-4, to=1-5]
	\arrow[from=1-4, to=2-4]
	\arrow[from=1-5, to=2-5]
	\arrow[from=2-1, to=2-2]
	\arrow[from=2-4, to=2-5]
\end{tikzcd}} \\
By the universal property of localization, the morphism $R_{\varphi(q)}\to S_q$ allowing the right diagram to commute is unique. Hence the ring homomorphism between stalks is precisely the morphism induced by the universal property of localization. Now since $q$ is the unique maximal ideal of $S_q$ and $\varphi(q)$ is the unique maximal ideal of $R_{\varphi(q)}$, we know that this homomorphism must send the maximal ideal $\varphi(q)$ of $R_{\varphi(q)}$ to $q$ of $S_q$. By the natural bijection between prime ideals of a ring and its localization, $\varphi(q)$ corresponds to the same prime ideal in $R$ and $q$ corresponds to the same prime ideal in $S$. Moreover, the since the diagram is commutative, the map $\varphi^\#(\text{Spec}(R)):R\to S$ must send $\varphi(q)$ to $q$. \\

I claim that the preimage of $(\varphi,\varphi^\#)$ is precisely the morphism $\varphi^\#(\text{Spec}(R)):R\to S$. Indeed, the induced morphism $\text{Spec}(\varphi^\#)$ sends $q$ to $(\varphi^\#)^{-1}(q)$. But we have just established above that $\varphi^\#$ sends $\varphi(q)$ to $q$. Hence $\text{Spec}(\varphi^\#)$ must send $q$ to $\varphi(q)$. Thus $\text{Spec}(\varphi^\#)$ agrees with $(\varphi,\varphi^\#)$ on the level of topological spaces. \\

It remains to show that the two map of sheaves are equal. To do this, it suffices to show that the induced map of stalks are equal. Notice that the two map of sheaves are equal in their global section and their map of topological spaces, hence each of the two map of sheaves induces morphisms $f,g:R_{\varphi(q)}\to S_q$ making the above diagram commute. But we have established that such a morphism allowing the above diagram to commute must be induced by the universal property of localization, and so must be unique. Hence the induced morphism of stalks of the two map of sheaves are equal. Thus we are done. 
\end{proof}
\end{prp}

\section{Projective Spectra and its Structure Sheaf}
\subsection{The Zariski Topology and Hilbert's Nullstellensatz for Proj}
Let $S=\bigoplus_{i=0}^\infty S_i$ be a commutative graded ring. Recall that the irrelevant ideal of $S$ is defined by $$S_+=\bigoplus_{i=1}^\infty S_i$$ More generally, recall that we defined $$I_+=I\cap S_+$$ for a homogeneous ideal $I\subseteq S$ in Rings and Modules. It is also a homogeneous ideal. 

\begin{defn}{The Projective Spectrum}{} Let $S$ be a commutative graded ring. Define the projective spectrum of $S$ to be $$\text{Proj}(S)=\{P\in\text{Spec}(S)\;|\;P\text{ is homogeneous and }S_+\not\subseteq P\}$$ where $S_+$ is the irrelevant ideal. 
\end{defn}

\begin{defn}{The Homogeneous Vanishing Set}{}\\
Let $S$ be a commutative graded ring. Let $I$ be a homogeneous ideal of $S$. Define the vanishing set of $I$ to be $$\V^H(I)=\{P\in\text{Proj}(S)\;|\;I\subseteq S\}$$
\end{defn}

\begin{lmm}{}{}\\
Let $S$ be a commutative graded ring. Then the following are true. 
\begin{itemize}
\item If $I\subseteq J$ are homogeneous ideals of $S$, then $\V^H(J)\subseteq \V^H(I)$. 
\end{itemize}
\end{lmm}

\begin{lmm}{}{}\\
Let $S$ be a commutative graded ring. Let $I\subseteq S$ be a homogeneous ideal. Then $\V^H(I)=\emptyset$ if and only if $I$ is an irrelevant ideal. 
\end{lmm}

This justifies the terminology for irrelevant ideals: They have no geometric meaning at all in terms of the Proj construction. 

\begin{prp}{}{}\\
Let $S$ be a commutative graded ring. Then the following are true. 
\begin{itemize}
\item Let $\{I_j\;|\;j\in J\}$ be a set of homogeneous ideals of $S$. Then we have $$\V^H\left(\sum_{j\in J}I_j\right)=\bigcap_{j\in J}\V^H(I_j)$$
\item Let $I_1,I_2$ be homogeneous ideals of $S$. Then we have $$\V^H(I_1I_2)=\V^H(I_1)\cup\V^H(I_2)$$
\end{itemize}
\end{prp}

Similar to that of $\text{Spec}(R)$ we can endow a topology on $\text{Proj}(S)$. 

\begin{defn}{Zariski Topology on Proj}{}\\
Let $S$ be a commutative graded ring. Define the Zariski topology on $\text{Proj}(S)$ to be the topology where the closed sets are exactly sets of the form $\V^H(I)$ for $I\subseteq S$ a homogeneous ideal. 
\end{defn}

\begin{lmm}{}{}\\
Let $S$ be a commutative graded ring. Let $I$ be a homogeneous ideal of $S$. Then we have $$\V^H(I)=\V^\text{S}(I)\cap\text{Proj}(R)$$
\begin{proof}
Trivial. 
\end{proof}
\end{lmm}

\begin{prp}{}{}\\
Let $S$ be a commutative graded ring. Then $\text{Proj}(S)$ with the Zariski topology is a subspace of $\text{Spec}(S)$ with the Zariski topology. 
\end{prp}

\begin{defn}{Projective Space over a Ring}{}\\
Let $R$ be a commutative ring. Define the projective $n$ space over $A$ to be $$\Prj_A^n=\text{Proj}(A[x_0,\dots,x_n])$$
\end{defn}

\begin{defn}{The Ideal of a Subset of Proj}{}\\
Let $S$ be a commutative graded ring. Let $T\subseteq\text{Proj}(R)$ be a subset. Define the ideal associated to $T$ to be $$\I^H(T)=\{f\in S_+\;|\;f\in P\text{ for all }P\in T\}$$
\end{defn}

\begin{lmm}{}{}\\
Let $S$ be a commutative graded ring. Then the following are true. 
\begin{itemize}
\item If $T\subseteq S$ is a subset, then $\I^H(T)$ is a homogeneous radical ideal. 
\item If $T\subseteq S$ is a subset, then we have $$\I^H(T)=\left(\bigcap_{P\in T}P\right)\cap S_+$$
\item If $T_1\subseteq T_2\subseteq S$, then $\I^H(T_2)\subseteq\I^H(T_1)$. 
\item If $T_1,T_2\subseteq S$, then $\I^H(T_1\cup T_2)=\I^H(T_1)\cap\I^H(T_2)$. 
\end{itemize}
\end{lmm}

\begin{prp}{Hilbert's Nullstellensatz for Proj}{}
Let $S$ be a commutative graded ring. Then the following are true. 
\begin{itemize}
\item If $I\subseteq S_+$ is an ideal of $S$, then $\I^H(\V^H(I))=\sqrt{I}_+$. 
\item If $T\subseteq S$ is a subset, then $\V^H(\I^H(T))=\overline{T}$. 
\end{itemize}
\begin{proof}\\
From Commutative Algebra 1, we have seen that $\sqrt{I}_+=\bigcap_{I\subseteq P\text{ is relevant }}P$. Then we have $$\I^H(\V^H(I))=\left(\bigcap_{P\in\V^H(I)}P\right)\cap S_+=\left(\bigcap_{I\subseteq P\text{ is relevant }}P\right)\cap S_+=(\sqrt{I})_+$$~\\

For any closed set $\V^H(I)$ for some homogeneous ideal $I\subseteq S$, $\V^H(I)$ contains $T$ if and only if $I\subseteq P$ for all $P\in T$. And this is true if and only if $$I\subseteq\I^H(T)=\left(\bigcap_{P\in T}P\right)\cap S_+$$ Since the largest homogeneous ideal contained by $\I^H(T)$ is itself, the smallest closed subset containing $T$ must be $\V^H(\I^S(T))$. Hence $\overline{T}=\V^H(\I^H(T))$. 
\end{proof}
\end{prp}

\begin{prp}{}{}\\
Let $S$ be a commutative graded ring. Then the following are true. 
\begin{itemize}
\item There is an inclusion reversing bijection $$\left\{\substack{\text{Homogeneous radical}\\\text{ideals }I\text {of }S\text{ such that }I=\sqrt{I}\cap S_+}\right\}\;\;\overset{\text{1:1}}{\longleftrightarrow}\;\;\left\{\substack{\text{Closed Subsets of}\\\text{Proj}(S)}\right\}$$ induced by $\V^\text{H}(-)$ and $\I^\text{H}(-)$. 
\item There is an inclusion reversing bijection $$\text{Proj}(S)\;\;\overset{\text{1:1}}{\longleftrightarrow}\;\;\left\{\substack{\text{Irreducible Closed Subsets of}\\\text{Proj}(S)}\right\}$$ induced by $\V^\text{H}(-)$ and $\I^\text{H}(-)$. 
\end{itemize}
\end{prp}

\begin{lmm}{}{}\\
Let $S$ be a commutative graded ring. Let $I\subseteq S_+$ be a homogeneous ideal. Then the following are equivalent. 
\begin{itemize}
\item $\V^H(I)=\emptyset$. 
\item If $(f_i)=I$ and each $f_i$ is homogeneous for $i\in J$, then $\text{Proj}(S)=\bigcup_{i\in J}D_+(f_i)$. 
\item $\sqrt{I}\supseteq S_+$. 
\end{itemize}
\end{lmm}

Thus any ideal $I$ whose radical contains $S_+$ is geometrically irrelevant: Indeed if $\sqrt{I}\supseteq S_+$ then $\V^H(I)$ is empty. 

\subsection{Topological Properties of Proj}
\begin{defn}{Projective Distinguished Open Sets}{}\\
Let $S$ be a commutative graded ring. Let $I\subseteq S$ be a homogeneous ideal. Define the projective distinguished open set of $I$ to be $$D_+(I)=\text{Proj}(S)\setminus\V^H(I)$$ If $I=(f)$ then we call $D_+(I)$ a basic open set. 
\end{defn}

\begin{lmm}{}{}\\
Let $S$ be a commutative graded ring. Then the basic open sets form a basis for the Zariski topology on $\text{Proj}(S)$. 
\end{lmm}

\begin{lmm}{}{}\\
Let $S$ be a commutative graded ring. Let $f\in S$ be a homogeneous element of non-zero degree. Then there is an isomorphism of locally ringed spaces $$D_+(f)\cong\text{Spec}((S_f)_0)$$ given by the correspondence of prime ideals of localization of graded rings. 
\end{lmm}

\begin{prp}{}{}\\
Let $S$ be a commutative graded ring. Then the following are true. 
\begin{itemize}
\item If $S$ is Noetherian, then $\text{Proj}(S)$ is Noetherian. 
\item If $S$ is an integral domain, then $\text{Proj}(S)$ is irreducible. 
\end{itemize}
\begin{proof}
\begin{itemize}
\item
\item I claim that $\text{Proj}(S)=\overline{(0)}$. Indeed, we have $P\in\overline{\{(0)\}}$ if and only if $P\in\V^H(\I^H(\{(0)\}))$ if and only if $\I^H(\{(0)\})\subseteq P$. But since $\I^H(\{(0)\})=\emptyset$, the condition is null and so we have equality. \\

Now suppose that $\text{Proj}(S)=\V^H(I)\cup\V^H(J)$ for some homogeneous ideals $I,J$ of $S$. Without loss of generality suppose that $(0)\in\V^H(I)$. By definition of closure, we must have $\overline{(0)}=\V^H(I)$ so $\text{Proj}(S)$ is irreducible. 
\end{itemize}
\end{proof}
\end{prp}

\subsection{The Structure Sheaf on a Projective Spectrum}
\begin{defn}{The Structure Sheaf on Projective Spectra}{} Let $S$ be a graded ring. Construct $\mathcal{O}_{\text{Proj}(S)}:\bold{Open}(\text{Proj}(S))\to\bold{Rings}$ as follows. 
\begin{itemize}
\item For $U\subseteq\text{Proj}(S)$ an open set, define $$\mO_{\text{Proj}(S)}(U)=\left\{(s_p)_{p\in U}\in\prod_{p\in U}S_{(p)}\;\bigg{|}\;\substack{\forall p,\exists p\in U_p\subseteq U\text{ open and }a,f\in S\text{ homogeneous s.t.}\\q\in U_p\text{ implies }f\notin q\text{ and }s_q=a/f\in S_{(q)}}\right\}$$
\item For $V\subseteq U$ the inclusion, define the unique map $\mO_{\text{Proj}(S)}(U)\to\mO_{\text{Proj}(S)}(V)$ by projection map $$\text{res}_V^U:\prod_{p\in U}S_{(p)}\to\prod_{p\in V}S_{(p)}$$
\end{itemize}
Then $\mO_{\text{Proj}(S)}$ is a sheaf on $S$. 
\end{defn}

\begin{prp}{}{}\\
Let $S$ be a commutative graded ring. Then the following are true. 
\begin{itemize}
\item For any $p\in\text{Proj}(S)$, there is a graded ring isomorphism $\mO_{\text{Proj}(S),p}\cong(S_{(p)})_0$. 
\item For any homogeneous element $f\in S_+$, there is an isomorphism $\mO_{\text{Proj}(S)}(D_+(f))\cong S_{(f)}$
\end{itemize}
\end{prp}

\section{The Category of Schemes}
\subsection{The Definition of Schemes}
\begin{defn}{Affine Schemes}{}\\
Let $(X,\mO_X)$ be a locally ringed space. We say that $X$ is an affine scheme if there exists a commutative ring $R$ such that $(X,\mO_X)\cong(\text{Spec}(R),\mO_{\text{Spec}(R)})$. 
\end{defn}

\begin{defn}{The Affine Space}{}\\
Let $R$ be a commutative ring. Let $n\in\N\setminus\{0\}$. Define the $n$ dimensional affine space over $R$ to be $$\A_R^n=\text{Spec}(R[x_1,\dots,x_n])$$
\end{defn}

\begin{prp}{}{}\\
Let $X$ be an affine scheme. Then $X$ is compact. 
\end{prp}

\begin{defn}{The Category of Affine Schemes}{}\\
The category of affine schemes $\bold{AffSch}$ is the full subcategory of $\bold{LRS}$ consisting of affine schemes. 
\end{defn}

\begin{lmm}{}{}\\
There is an equivalence of categories $$\bold{CRing}\cong\bold{AffSch}^{\text{op}}$$ given by $\text{Spec}$ whose inverse is $\Gamma$. 
\end{lmm}

\begin{defn}{Projective Scheme}{}\\
Let $(X,\mO_X)$ be a locally ringed space. We say that $X$ is a projective scheme if there exists a commutative graded ring $S$ such that $(X,\mO_X)\cong(\text{Proj}(S),\mO_{\text{Proj}(S)})$. 
\end{defn}

\begin{defn}{Schemes}{}\\
Let $(X,\mO_X)$ be a locally ringed space. We say that $X$ is a scheme if for all $p\in X$, there exists an open neighborhood $U\subseteq X$ of $p$ such that $(U,\mO_X|_U)$ is an affine scheme. 
\end{defn}

\begin{prp}{}{}\\
Let $S$ be a commutative graded ring. Then $(\text{Proj}(S),\mO_{\text{Proj}(S)})$ is a scheme. 
\end{prp}

\begin{lmm}{}{}\\
Let $R$ be a commutative ring. Let $f\in R$ be an element. Then there is a natural isomorphism of locally ringed space $$(\text{Spec}(R_f),\mO_{\text{Spec}(R_f)})\cong(D(f),\mO_{\text{Spec}(R)}|_{D(f)})$$
\begin{proof}
We already established a natural homeomorphism $\varphi:D(f)\overset{\cong}{\to}\text{Spec}(R_f)$ given by the correspondence of prime ideals of localization $\varphi(P)=P_f$. It remains to define an isomorphism of sheaves. For each open set $U\subseteq\text{Spec}(R_f)$, define $f(U):\mO_{\text{Spec}(R_f)}(U)\to\mO_{\text{Spec}(R)}|_{D(f)}(\varphi^{-1}(U))$ as follows. For each $p\in U$, there is a natural isomorphism $(R_f)_P\cong R_p$. Since $\mO_{\text{Spec}(R_f)}(U)$ is a subset of $\prod_{p\in U}(R_f)_p$, define $f(U)$ by the product of the isomorphisms $(R_f)_p\cong R_p$. Evidently, $f(U)$ is an isomorphism since $\varphi$ is a homeomorphism. Thus we are done. 
\end{proof}
\end{lmm}

\begin{lmm}{}{}\\
Let $(X,\mO_X)$ be a scheme. Let $U\subseteq X$ be an open set. Then $(U,\mO_X|_U)$ is a scheme. 
\begin{proof}\\
Let $p\in U$. Since $X$ is a scheme, there exists an open neighborhood $V\subseteq X$ of $p$ such that $(V,\mO_X|_V)$ is an affine scheme. Since $V$ has an open cover of basic open sets, there exists a basic open set $D(f)$ such that $p\in D(f)$. But by the above, $D(f)$ is an affine scheme. Hence $p$ lies locally in an affine scheme. Thus $X$ is a scheme. 
\end{proof}
\end{lmm}

\end{proof}

\subsection{Understanding Schemes Locally}
\begin{thm}{Gluing Schemes}{}\\
Let $(X_i,\mathcal{O}_{X_i})$ for $i\in I$ be a family of schemes. Suppose that we have the following data. 
\begin{itemize}
\item For each $i,j\in I$, an open subset $U_{ij}\subseteq X_i$. 
\item Isomorphisms $\theta_{ij}:U_{ij}\overset{\cong}{\to}U_{ji}$. 
\end{itemize}
such that the following are true. 
\begin{itemize}
\item $\theta_{ii}=\text{id}$. 
\item $\theta_{ij}|_{U_{ij}\cap U_{ik}}\circ\theta_{jk}|_{U_{ji}
\cap U_{jk}}=\theta_{ik}|_{U_{ij}\cap U_{ik}}$. 
\end{itemize}
Then there exists a unique scheme $(X,\mathcal{O}_X)$ and an open cover $X=\bigcup_{i\in I}X_i'$ and a family of isomorphisms $\varphi_i:(X_i',\mathcal{O}_X|_{X_i'})\to(X_i,\mathcal{O}_{X_i})$ such that $$(\varphi_j|_{X_i\cap X_j})^{-1}\circ\theta_{ij}\circ\varphi_i|_{X_i\cap X_j}=\text{id}$$ for all $i,j\in I$. 
\end{thm}

\begin{prp}{Gluing Morphisms}{}\\
Let $X,Y$ be a scheme. Let $X=\bigcup_{i\in I}U_i$ be an open cover of $X$. Let $f_i:U_i\to Y$ be morphisms of schemes such that $f_i|_{U_i\cap U_j}=f_j|_{U_i\cap U_j}$ for all $i,j\in I$. Then there exists a unique morphism of schemes $f:X\to Y$ such that $f|_{U_i}=f_i$ for all $i\in I$. 
\begin{proof}\\
Since morphisms of topological spaces and morphisms of sheaves glue uniquely, we are done. \\

We note that the unique morphism of schemes is defined as follows. The underlying map of topological spaces is given by $f(p)=f_i(p)$ for $p\in U_i$. For the morphism of sheaves $\mO_Y\to f_\ast\mO_X$, let $V\subseteq Y$ be an open subset. Then define $\mO_Y(V)\to\mO_X(f^{-1}(V))$ as follows. For each $s\in\mO_Y(V)$, the morphisms $(f_i)^\#(V):\mO_Y(V)\to\mO_{U_i}((f_i)^{-1}(V))$ gives a section $(f_i)^\#(V)(s)\in\mO_{U_i}((f^{-1}(V))=\mO_X(U_i\cap f^{-1}(V))$. Then the identity and gluing axiom guarantees a unique section $f(s)\in\mO_X(f^{-1}(V))$. 
\end{proof}
\end{prp}

\end{eg}

\begin{defn}{The Affine Communication Property}{}\\
Let $X$ be a scheme. Let $P$ be a property of affine open subsets. We say that $P$ has the affine communication property if the following are true. 
\begin{itemize}
\item If $\text{Spec}(R)\subseteq X$ has property $P$, then $\text{Spec}(R_f)\subseteq X$ has property $P$ for any $f\in R$. 
\item If $\text{Spec}(R)\subseteq X$, $R=(f_1,\dots,f_n)$ and $\text{Spec}(R_{f_i})\subseteq X$ has property $P$ for all $i$, then $\text{Spec}(R)\subseteq X$ has property $P$. 
\end{itemize}
\end{defn}

\begin{thm}{The Affine Communication Lemma}{}\\
Let $X$ be a scheme. Let $P$ be a property of affine open subsets of a scheme such that $P$ has the affine communication property. Let $X=\bigcup_{i\in I}\text{Spec}(R_i)$ is an affine open cover such that each $\text{Spec}(R_i)$ has property $P$, then every affine open subset of $X$ has property $P$. 
\end{thm}

\subsection{Properties of the Category of Schemes}
\begin{defn}{The Category of Schemes}{}\\
The category $\bold{Sch}$ of schemes is the full subcategory of $\bold{LRS}$ consisting of schemes. 
\end{defn}

Explicitly, a morphism of schemes $f:X\to Y$ consists of the following data. 
\begin{itemize}
\item A continuous map $X\to Y$. 
\item A morphism of sheaves $\mO_Y\to f_\ast\mO_X$. 
\end{itemize}

\begin{prp}{}{}\\
Let $X,Y$ be schemes. Let $f:X\to Y$ be a morphism of ringed spaces. Then the following are equivalent. 
\begin{itemize}
\item $f$ is a morphism of schemes. 
\item For any affine open subsets $U\subseteq X$ and $V\subseteq Y$ such that $f(U)\subseteq V$, the morphism of ringed spaces $f|_U:U\to V$ is a morphism of the form induced by the Spec functor. 
\item There exists affine open covers $X=\bigcup_{i\in I}U_i$ and $Y=\bigcup_{i\in I}V_i$ such that $f|_{U_i}$ is a morphism of the form induced by the Spec functor. 
\end{itemize}
\begin{proof}\\
$(1)\implies(2)$: Since $f$ is a morphism of locally ringed spaces, $f|_U:U\to V$ is also a morphism of locally ringed spaces (stalks of $f$ are isomorphic to stalks of $f|_U$). Since $\text{Spec}:\bold{CRing}\to\bold{LRS}$ is fully faithful, $f|_U$ must be induced by the Spec functor. \\

$(2)\implies(3)$: Trivial. \\

$(3)\implies(1)$: Since stalks of $f$ are equal to the stalks of $f|_{U_i}$ and morphisms induced by the spec functor are morphisms of locally ringed spaces, $f$ must also be a morphism of locally ringed spaces. 
\end{proof}
\end{prp}

\begin{prp}{}{}\\
There is an adjunction $$\Gamma:\bold{Sch}\overset{\dashv}{\rightleftarrows}\bold{CRing}:\text{Spec}$$
\begin{proof}\\
We wish to show that there is a natural isomorphism $$\Hom_{\bold{Sch}}(X,\text{Spec}(R))\cong\Hom_{\bold{CRing}}(R,\Gamma(X))$$ We first construct the canonical morphism $X\to\text{Spec}(\Gamma(X))$. Let $X=\bigcup_{i\in I}U_i$ be an affine open cover. For each $i\in I$, define $f_i:\text{Spec}(\mO_X(U_i))\cong U_i\to\text{Spec}(\Gamma(X))$ to be the morphism $\text{Spec}(\Gamma(X)=\mO_X(X)\overset{\text{res}_{U_i}^X}{\to}\mO_X(U_i))$. Since $\text{res}_{U_i}^X\circ\text{res}_{U_i\cap U_j}^X=\text{res}_{U_j}^X\circ\text{res}_{U_i\cap U_j}^X$, by 3.2.2 there exists a unique morphism $f:X\to\text{Spec}(\Gamma(X))$ such that $f|_{U_i}=\text{Spec}(\text{res}_{U_i}^X)$. By construction, the global section of the morphism of sheaves $\Gamma(f):\Gamma(X)\to\Gamma(\text{Spec}(\Gamma(X)))\cong\Gamma(X)$ is the identity map. Moreover, notice that if $X$ is affine, the the morphism $X\to\text{Spec}(\Gamma(X))$ is an isomorphism. \\

Now I claim that for any morphism $g:X\to\text{Spec}(R)$, there exists a unique ring homomorphism $R\to\Gamma(X)$ such that the following diagram commutes: \\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	X && {\text{Spec}(R)} \\
	& {\text{Spec}(\Gamma(X))}
	\arrow["g", from=1-1, to=1-3]
	\arrow["f"', from=1-1, to=2-2]
	\arrow["{\exists!}"', dashed, from=2-2, to=1-3]
\end{tikzcd}}\\
and thus completing the proof (surjectivity given by composition with $X\to\text{Spec}(\Gamma(X))$. Since the morphism $X\to\text{Spec}(\Gamma(X))$ is natural and morphisms glue, it suffices to check for the case that $X$ is affine. But if $X$ is affine, then $f$ is an isomorphism hence we are done. 
\end{proof}
\end{prp}

\subsection{Relative Schemes}
\begin{defn}{The Category of Relative Schemes}{}\\
Let $S$ be a scheme. Define the category of schemes over $S$ to be the slice category $$\bold{Sch}_S=\bold{Sch}_{/S}$$
\end{defn}

Recall that fibered products is a name for pullbacks, and pullbacks are the same as products in the appropriate slice category. 

\begin{prp}{}{} Let $X$ and $Y$ be two schemes over $S$. Then the fibered product $X\times_SY$ exists and is unique up to unique isomorphism. 
\begin{proof}
We first prove the theorem for the case of affine schemes. Let $X=\text{Spec}(A)$, $Y=\text{Spec}(B)$ and $S=\text{Spec}(C)$. I claim that $\text{Spec}(A\otimes_CB)$ is the fibered product of $X$ and $Y$ over $S$. Using the equivalence of categories, we have that 
\begin{align*}
\Hom_\bold{AffSch}(Z,\text{Spec}(A\otimes_CB))&\cong\Hom_\bold{Rings}(A\otimes_CB,\Gamma(Z))\\
&\cong\Hom_\bold{Rings}(A,\Gamma(Z))\times_{\Hom_\bold{Rings}(C,\Gamma(Z))}\Hom_\bold{Rings}(B,\Gamma(Z))\\
&\cong\Hom_\bold{AffSch}(Z,\text{Spec}(A))\times_{\Hom_\bold{AffSch}(Z,\text{Spec}(C))}\Hom_\bold{AffSch}(Z,\text{Spec}(B))
\end{align*}
Thus we have proved that $\text{Spec}(A\otimes_CB)$ is the fiber product of $X$ and $Y$ over $S$. 
\end{proof}
\end{prp}

Let $R$ be a commutative ring. Then there is an adjunction $$\Gamma:\bold{Sch}_R\overset{\dashv}{\rightleftarrows}\bold{CAlg}_R:\text{Spec}$$
\end{prp}

\subsection{Points on A Scheme}
Let $k$ be a field. Recall the Yoneda Embedding to be the functor $$\bold{Sch}_k\to\Hom_{\bold{Cat}^{\text{op}}}(\bold{Sch}_k,\bold{Set})$$ given by the following data. 
\begin{itemize}
\item A scheme $X$ over $k$ is assigned to the functor $\Hom_{\bold{Sch}_k}(-,X):\bold{Sch}_k\to\bold{Set}$. 
\item A morphism $f:X\to Y$ over $k$ is assigned to the natural transformation $\Hom_{\bold{Sch}_k}(-,Y)\to\Hom_{\bold{Sch}_k}(-,X)$ given by precomposition with $f$. 
\end{itemize}

The Yoneda lemma then says that this is a fully faithful functor. Because schemes are patched together by affine schemes, a non-trivial statement says that the scheme can be recovered just by considering the Yoneda embedding restricted to affine schemes. 

\begin{defn}{The Functor of Points}{}\\
Let $S$ be a scheme. Let $X$ be a scheme over $S$. Define the functor of points $$\widetilde{X}:\bold{AffSch}_S\to\bold{Set}$$ by the following data. 
\begin{itemize}
\item For an affine scheme $Z$ over $S$, we have $\widetilde{X}(Z)=\Hom_{\bold{Sch}_S}(Z,X)$. 
\item For a morphism of schemes $f:Z\to Y$ over $S$, the morphism $\widetilde{X}(Y)\to\widetilde{X}(Z)$ is induced by precomposition with $f$. 
\end{itemize}
\end{defn}

\begin{prp}{}{}\\
Let $S$ be a scheme. Let $X$ be a scheme over $S$. Then the functor $\widetilde{(-)}:\bold{Sch}_S\to\Hom_{\bold{Cat}}(\bold{AffSch}_S,\bold{Set})$ is fully faithful. 
\end{prp}

If $S=\text{Spec}(R)$ is affine, then we can use the equivalence between $R$-algebras and affine schemes over $S$ to get a functor $$\widetilde{X}:\bold{CAlg}_R\to\bold{Set}$$ If $X=\text{Spec}(A)$ is also affine, then the functor of points collapses to $$\widetilde{\text{Spec}(A)}(\text{Spec}(B))=\Hom_{\bold{CAlg}_R}(A,B)$$

\begin{eg}{}{}\\
The $\C$-points of $\text{Spec}(\C[x])$ are in bijection with $\C$. 
\begin{proof}\\
The $\C$-points of the affine scheme are in bijection with the ring homomorphisms $\C[x]\to\C$. Clearly for any $a\in\C$, the evaluation homomorphism $\varphi_a:\C[x]\to\C$ is a ring homomorphism. On the other hand, given a ring homomorphism $\phi:\C[x]\to\C$, it is surjective since $\phi(1)=1$ by definition. Then there is a ring isomorphism $\frac{\C[x]}{\ker(\phi)}\cong\C$ given by the first isomorphism theorem. Since $\C$ is a field, $\ker(\phi)$ is a maximal ideal and by the correspondence of maximal ideals of $\C[x]$, we must have $\ker(\phi)=(x-a)$ for some $a\in\C$. Hence $\phi=\varphi_a$. Hence we are done. 
\end{proof}
\end{eg}

Notice that $\text{Spec}(\C[x])$ has an extra point $(0)$ which is not a $\C$-point of the affine scheme. 

\begin{eg}{}{}\\
The $\Q$-points of $\text{Spec}\left(\frac{\Z[x,y]}{(x^2+y^2-16)}\right)$ are precisely the rational solutions of $x^2+y^2=16$. 
\begin{proof}\\
Indeed, by the fully faithfulness of spec, we have $$\Hom_{\bold{Sch}}\left(\text{Spec}(\Q),\text{Spec}\left(\frac{\Z[x,y]}{(x^2+y^2-16)}\right)\right)\cong\Hom_{\bold{CRing}}\left(\frac{\Z[x,y]}{(x^2+y^2-16)},\Q\right)$$ Suppose that $(u,v)$ is a rational solution to the equation. Then the morphism $\varphi:\Z[x,y]\to\Q$ defined by $\varphi(f)=f(u,v)$ factors through $\frac{\Z[x,y]}{(x^2+y^2-16)}$ and gives rise to a $\Q$-point. Conversely, given any $\Q$-point $\varphi:\frac{\Z[x,y]}{(x^2+y^2-16)}$, precomposition with the quotient map gives a morphism $\Z[x,y]\to\Q$ which must be of the form of an evaluation homomorphism. 
\end{proof}
\end{eg}

\begin{prp}{}{}\\
Let $k$ be a field. Let $X$ be a scheme. Then there is a natural bijection $$\widetilde{X}(k)\cong\{x\in X\;|\;\text{the canonical homomorphism }k\to k(x)\text{ is an isomorphism }\}$$
\end{prp}

\begin{defn}{Generic Points}{}\\
Let $X$ be a space. Let $Z\subseteq X$ be an irreducible closed subset. Let $p\in Z$. We say that $p$ is a generic point for $Z$ if $Z=\overline{\{p\}}$. 
\end{defn}

The second part of 1.1.12 shows that every irreducible component of an affine scheme has a unique generic point. 

\begin{lmm}{}{}\\
Let $X$ be a scheme. Then the following are true. 
\begin{itemize}
\item If $Z\subseteq X$ is an irreducible closed subset, then $Z$ has a unique generic point. 
\item There is a bijection $$X\;\;\overset{1:1}{\longleftrightarrow}\;\;\{\text{ Irreducible closed subsets of }X\}$$ given by $p\mapsto\overline{\{p\}}$. 
\end{itemize}
\begin{proof}
Let $\{V_i=\text{Spec}(R_i)\;|\;i\in J\}$ be an open affine cover of $X$. Then there exists a subset $\{U_i\;|\;i\in I\}$ of the open cover that forms an open cover for $Z$ and each $Z\cap U_i$ is non-empty. Since $Z$ is irreducible, $Z\cap U_i$ is irreducible. Since $Z$ is non-empty (irreducible spaces are non-empty by assumption), the second case must occur at least in one $i$, and by 1.1.12 there exists a unique $P_i\in\text{Spec}(R_i)$ such that $Z\cap U_i=\overline{\{P_i\}}^{U_i}$ where $\overline{(-)}^{U_i}$ refers to the closure taken in $U_i$. \\

Now I claim that $\overline{T}^{U_i}=\overline{T}^X\cap U_i$ for any subset $T\subseteq U_i$. To prove this, it suffices to show that $\overline{T}^{U_i}$ is the smallest closed subset in $U_i$ that contains $T$. So let $V\subseteq U_i$ be a closed subset containing $T$. By definition of the subspace topology of $U_i$, there exists a closed subset $W\subseteq X$ such that $V=W\cap U_i$. Then we have $$\overline{T}^X=\bigcap_{T\subseteq C\subseteq X\text{ closed }}C\subseteq W$$ since $T\subseteq V=W\cap U_i\subseteq W$. Then intersecting with $U_i$ gives $$\overline{T}^{X}\cap U_i\subseteq W\cap U_i=V$$ hence $\overline{T}^X\cap U_i$ is the closure of $T$ in $U_i$. \\

Going back to the main question, we thus have that $$Z\cap U_i=\overline{\{P_i\}}^X\cap U_i$$ Now I claim that $P_i\in U_j$ for all $j\in I$. Suppose for a contradiction that $P_i\notin U_j$. Then $P_i\in U_i\setminus U_j$. Now since $U_i\cap U_j$ is open, we have that $U_i\setminus U_j$ is closed in $U_i$. By definition of closure, we have $$\overline{\{P_i\}}^X\cap U_i=\overline{\{P_i\}}^{U_i}\subseteq U_i\setminus U_j$$ Then we have $$Z\cap U_i\cap U_j=\overline{\{P_i\}}^{U_i}\cap U_j=\overline{\{P_i\}}^X\cap U_i\cap U_j=\emptyset$$ Taking complements, we get $$Z=(Z\setminus Z\cap U_i)\cup(Z\setminus Z\cap U_j)$$ Since $Z$ is irreducible, without loss of generality we have $Z=Z\setminus Z\cap U_i$, or $Z\cap U_i=\emptyset$. This is a contradiction since we assumed that are open cover is such that $Z\cap U_i$ is non-empty. \\

We now have that $P_i\in U_j$ for all $j\in I$. Let $V\subseteq U_i\cap U_j$ be an affine open set, Then we have $$\overline{\{P_i\}}^V=\overline{\{P_i\}}^X\cap V=Z\cap U_i\cap U_j\cap V=\overline{\{P_j\}}^X\cap V=\overline{\{P_j\}}^V$$ Since $V$ is affine, the generic point is unique by 1.1.12, hence $P_i=P_j$. Write $P=P_i$. \\

Now since the $U_i$'s cover $Z$, we have that $$Z=\bigcup_{i\in I}(Z\cap U_i)=\bigcup_{i\in I}(\overline{\{P_i\}}^X\cap U_i)=\bigcup_{i\in I}(\overline{\{P\}}^X\cap U_i)=\overline{\{P\}}^X$$ Hence $Z$ has a generic point. \\

For uniqueness, suppose that $Z=\overline{\{p\}}^X=\overline{\{q\}}^X$. Let $U$ be an affine open subset $X$. Then $Z\cap U$ is affine and $$\overline{\{p\}}^U=\overline{\{p\}}^X\cap U=\overline{\{q\}}^X\cap U=\overline{\{q\}}^U$$ Since generic points of affine sets are unique, we must have $p=q$. \\

For the second part of the statement, note that the map $p\mapsto\overline{\{p\}}$ is surjective by the existence of generic point, and the map is injective by uniqueness of the generic point. 
\end{proof}
\end{lmm}

\subsection{Interpreting Elements of the Sheaf as Functions}
\begin{defn}{The Residue Field}{}\\
Let $X$ be a scheme. Let $p\in X$. Define the residue field of $p$ to be $$\kappa(p)=\frac{\mO_{X,p}}{m}$$ where $m$ is the unique maximal ideal of $\mO_{X,p}$. 
\end{defn}

\begin{defn}{Values of Elements of a Sheaf}{}\\
Let $(X,\mO_X)$ be a scheme. Let $U\subseteq X$ be an open set. Let $f\in\mO_X(U)$. Let $p\in U$. Define the value of $f$ at $p$ to be the image of $f$ in the morphism $$\mO_X(U)\to\mO_{X,p}\to\kappa(p)$$ We denote the value by $f(p)$. 
\end{defn}

\begin{lmm}{}{}\\
Let $(X,\mO_X)$ be a scheme. Let $U\subseteq X$ be an open set. Let $f\in\mO_X(U)$. Then the set $$S=\{p\in U\;|\;\text{ the value of }f\text{ at }p\text{ is non-zero }\}$$ is an open subset of $X$. 
\begin{proof}\\
Suppose that $p$ lies in the given set. This means that $[(f,U)]\in\mO_{X,p}$ is invertible. So there exists $[(g,V)]\in\mO_{X,p}$ such that $[(f,U)]\cdot[(g,V)]=[(1,X)]$. So there exists an open neighborhood $W_p\subseteq U\cap V$ of $p$ such that $f|_{W_p}\cdot g|_{W_p}=1$. Notice that for any $q\in W_p$, the same equation holds in $\mO_{X,q}$. Hence $S=\bigcup_{p\in S}W_p$ is an open set. 
\end{proof}
\end{lmm}

\begin{defn}{Support of a Section}{}\\
Let $(X,\mO_X)$ be a scheme. Let $f\in\mO_X(X)$ be a global section. Define the support of $f$ to be $$\text{Supp}(f)=\{p\in X\;|\;[(f,X)]\neq 0\in\mO_{X,p}\}$$
\end{defn}

\begin{lmm}{}{}\\
Let $(X,\mO_X)$ be a scheme. Let $f\in\mO_X(X)$ be a global section. Then $\text{Supp}(f)\subseteq X$ is a closed subset. 
\begin{proof}\\
For any $p\notin\text{Supp}(f)$, we have that $[(f,X)]\in\mO_{X,p}$ is zero. So there exists a neighborhood $V_p\subseteq X$ of $p$ such that $f|_{V_p}=0$. Then for all $q\in V_p$, clearly $[(f,X)]=0\in\mO_{X,q}$. Then clearly $U=\bigcup_{p\in X}V_p$ is an open subset of $X$ that is the complement of $\text{Supp}(f)$. Hence $\text{Supp}(f)$ is a closed subset. 
\end{proof}
\end{lmm}

Notice that $\{p\in X\;|\;f(p)=0\}$ is an open subset, while the support of $f$ is closed. They are two different notions.

\section{Topological Properties of Schemes}
\subsection{Reduced Schemes}
In a general scheme, functions are not determined by points. 

\begin{eg}{}{}\\
Consider the spectra $X=\text{Spec}\left(\frac{k[\varepsilon]}{(\varepsilon^2)}\right)$. Then for all $p\in X$, the value of $\varepsilon$ at $p$ is $0$. 
\begin{proof}\\
Notice that the residue field of $X$ at $(\varepsilon)$ is given by $$k((\varepsilon))=\frac{(k[\varepsilon]/(\varepsilon^2))_{(\varepsilon)}}{(\varepsilon}\cong\frac{k[\varepsilon]/(\varepsilon^2)}{(\varepsilon)}\cong k$$ and the $k$-algebra homomorphism $\mO_X(X)=\frac{k[\varepsilon]}{(\varepsilon^2)}\to k=k((\varepsilon))$ is given by $\varepsilon\to 0$. Hence the value of $\varepsilon$ at the only point $(\varepsilon)$ in $X$ is $0$. 
\end{proof}
\end{eg}

So it is not possible to identify the functions $\varepsilon$ and $0$ only by their value at the points of the spectrum. Notice that checking whether a particular element is $0$ on all points is the same as checking whether the element lies in the nilradical of the ring, because it is the intersection of prime ideals. 

\begin{defn}{Reduced Schemes}{}\\
Let $X$ be a scheme. We say that $X$ is reduced for every open set $U\subseteq X$, $\mO_X(U)$ is a reduced ring. 
\end{defn}

\begin{lmm}{}{}\\
Let $X$ be a scheme. Then $X$ is reduced if and only if for all $p\in X$, $\mO_{X,p}$ is a reduced ring. 
\begin{proof}\\
Suppose first that $X$ is reduced. Let $p\in X$. Let $[(f,U)]\in\mO_{X,p}$ be nilpotent. Then $[(f^n,U)]=0=[(0,U)]$ for some $n\in\N$. By definition of the equivalence classes, we must have an open subset $V\subseteq U$ such that $(f|_V)^n=(f^n)|_V=0$ in $\mO_X(V)$. Since $\mO_X(V)$ is nilpotent, we must have that $f|_V=0$. Hence $[(f,U)]=[(f|_V,V)]=[(0|_V,V)]=0\in\mO_{X,p}$. Hence $\mO_{X,p}$ is a reduced ring. \\

Now suppose that for all $p\in X$, $\mO_{X,p}$ is reduced. Let $f\in\mO_X(U)$ be a nilpotent element. For each $p\in U$, $[(f,U)]$ is an element of $\mO_{X,p}$. At the same time, $f^n=0$ in $\mO_X(U)$ implies that $[(f,U)]^n=[(f^n,U)]=0\in\mO_{X,p}$ for all $p\in U$. Since $\mO_{X,p}$ is reduced, there exists a neighborhood $V_p\subseteq U$ of $p$ such that $f|_{V_p}=0$. Now $\{V_p\;|\;p\in U\}$ is an open cover of $U$, and the collection $\{f|_{V_p}\;|\;p\in U\}$ agrees on intersections in the open cover. By the gluing axiom and the identity axiom the unique element in $\mO_X(U)$ that restricts to $f|_{V_p}=0$ for each $p\in U$ must be $f$, and so $f=0$. Hence $\mO_X(U)$ is reduced. 
\end{proof}
\end{lmm}

\begin{lmm}{}{}\\
Let $R$ be a commutative ring. Then $R$ is reduced if and only if $\text{Spec}(R)$ is reduced. 
\begin{proof}\\
Suppose that $R$ is reduced. Let $p\in\text{Spec}(R)$. Then there is a ring isomorphism $\mO_{\text{Spec}(R),p}\cong R_p$. Since the localization of a reduced ring is reduced, we must have that $R_p$ is reduced. Hence by lmm 4.1.3 $\text{Spec}(R)$ is reduced. \\

Conversely, $\text{Spec}(R)$ is a reduced ring, then the ring isomorphism $\mO_{\text{Spec}(R)}(\text{Spec}(R))\cong R$ shows that $R$ is reduced. 
\end{proof}
\end{lmm}

\begin{lmm}{}{}\\
Let $X$ be a reduced scheme. Let $f,g\in\mO_X(X)$. Then $f=g$ if and only if for all $p\in X$, the value of $f$ at $p$ is equal to the value of $g$ at $p$. 
\begin{proof}\\
If $f=g$ then the values must be the same. Now suppose that the values of $f$ and $g$ agree on all points. Without loss of assumption we can assume that $g=0$ and $f$ takes the value of $0$ at all points. 
\end{proof}
\end{lmm}

\begin{defn}{The Reduced Scheme}{}\\
Let $(X,\mO_X)$ be a scheme. Define the reduced scheme $(\mO_X)_\text{red}$ of $X$ to be the sheafification of the presheaf defined as follows. 
\begin{itemize}
\item For each open set $U\subseteq X$, define $(\mO_X)_\text{red}(U)=\frac{\mO_X(U)}{N(\mO_X(U))}$. 
\item For an inclusion of open sets $V\subseteq U$, define the induced map $$(\mO_X)_\text{red}(U)\to(\mO_X)_\text{red}(V)$$ as follows. There is a restriction map $\mO_X(U)\to\mO_X(V)$. Projecting to the quotient, we have the map $\mO_X(U)\to(\mO_X)_\text{red}(V)$. Since nilpotents are sent to nilpotents under ring homomorphisms, the map descends to the quotient $(\mO_X)_\text{red}(U)\to(\mO_X)_\text{red}(V)$. 
\end{itemize}
\end{defn}

\begin{defn}{The Canonical Map of the Reduced Scheme}{}\\
Let $X$ be a scheme. Define the canonical map $X_\text{red}\to X$ as follows. 
\begin{itemize}
\item The map of the underlying topological space is given by the identity map. 
\item The morphism of sheaves $\mO_X\to\text{id}_\ast((\mO_X)_\text{red})=(\mO_X)_\text{red}$ is given by the quotient map $\mO_X(U)\to(\mO_X)_\text{red}(U)=\frac{\mO_X(U)}{N(\mO_X(U))}$ for any open set $U$. 
\end{itemize}
\end{defn}

\begin{prp}{}{}\\
Let $X$ be a scheme. Then the reduced scheme of $X$ satisfies the following universal property. For any reduced scheme $Y$ and morphism $f:Y\to X$, there exists a unique morphism $f_\text{red}:Y\to X_\text{red}$ such that the following diagram commutes: \\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	Y & {X_\text{red}} \\
	& X
	\arrow["{\exists!f_\text{red}}", dashed, from=1-1, to=1-2]
	\arrow["f"', from=1-1, to=2-2]
	\arrow[from=1-2, to=2-2]
\end{tikzcd}}\\
Moreover, $X_\text{red}$ is the unique reduced scheme satisfying the above. 
\end{prp}

\subsection{Integral Schemes}
\begin{defn}{Integral Schemes}{}\\
Let $X$ be a scheme. We say that $X$ is integral if for every open set $U\subseteq X$, $\mO_X(U)$ is an integral domain. 
\end{defn}

\begin{lmm}{}{}\\
Let $X$ be a scheme. If $X$ is integral, then $\mO_{X,p}$ is an integral domain for all $p\in X$. 
\end{lmm}

\begin{prp}{}{}\\
Let $X$ be a scheme. Then $X$ is integral if and only if $X$ is reduced and irreducible. 
\begin{proof}\\
Suppose that $X$ is integral. Then clearly $X$ is reduced. If $X$ is not irreducible, then there exists proper closed subsets $X_1,X_2\subseteq X$ such that $X=X_1\cup X_2$. Let $U_i=X\setminus X_i$. Then we have $$\emptyset=X\setminus(X_1\cup X_2)=U_1\cap U_2$$ and $U_1,U_2$ are non-empty. Then we have $$\mO_X(U_1\amalg U_2)\cong\mO(U_1)\times\mO(U_2)$$ so that $\mO_X(U_1\amalg U_2)$ is not an integral domain, a contradiction. \\

Now suppose that $X$ is reduced and irreducible. Let $U\subseteq X$ be an open set. Let $V\subseteq U$ be an affine open subset of $U$. Since $X$ is irreducible, $V$ is also irreducible and since $V$ is affine, by 1.2.6 we must have that $\mO_X(V)$ is an integral domain. Now consider the morphism $\mO_X(U)\to\mO_X(V)$. I claim that the morphism is injective. Let $f\in\ker(\mO_X(U)\to\mO_X(V))$. Then $f|_V=0$. Notice that this is true for all $W\subseteq U$ affine open subsets. Since $U$ is a scheme, such affine open subsets cover $U$, hence by the gluing axiom, $f=0|_U$. Hence $\mO_X(U)\to\mO_X(V)$ is injective. Now since $\mO_X(V)$ is an integral domain, $\mO_X(U)$ is also an integral domain since it is the subring of an integral domain. Hence $X$ is integral. 
\end{proof}
\end{prp}

\begin{lmm}{}{}\\
Let $R$ be a commutative ring. Then the following are true. 
\begin{itemize}
\item $R$ is an integral domain if and only if $\text{Spec}(R)$ is integral. 
\item If $R$ is a graded integral domain, then $\text{Proj}(R)$ is integral. 
\end{itemize}
\begin{proof}\\
If $R$ is an integral domain, then by 1.2.6, $\text{Spec}(R)$ is irreducible. Since $R$ is an integral domain, $R$ is reduced and by 4.1.4, $\text{Spec}(R)$ is reduced. Conversely, if $\text{Spec}(R)$ is integral, then it is irreducible and by 1.2.6 $R$ must be an integral domain. 
\end{proof}
\end{lmm}

\begin{prp}{}{}\\
Let $k$ be an algebraically closed field. Let $X,Y$ be integral schemes over $k$. Then $X\times_kY$ is integral. 
\end{prp}

\subsection{Rational Functions on a Scheme}
\begin{defn}{The Sheaf of Total Quotient Ring}{}\\
Let $X$ be a scheme. Define the sheaf of total quotient rings $\mK_X$ to be the sheafification of the sheaf $Q_X$ given by the following data. 
\begin{itemize}
\item For each open subset $U\subseteq X$, we have $Q_X(U)=Q(\mO_X(U))$. 
\item For each inclusion of open sets $V\subseteq U$, the map $Q(\mO_X(U))\to Q(\mO_X(V))$ is the map induced by the universal property of localization. 
\end{itemize}
\end{defn}

The generic points corresponding to the irreducible components (those maximal among irreducible closed subsets) play a special role. 

\begin{prp}{}{}\\
Let $X$ be a scheme. Then the following are true. 
\begin{itemize}
\item Let $p\in X$. Then we have $\mK_{X,p}=Q(\mO_{X,p})$. 
\item Let $Z\subseteq X$ be an irreducible component of $X$. Let $\eta$ be its corresponding generic point. Then we have $\mK_{X,\eta}=\mO_{X,\eta}$. 
\end{itemize}
\end{prp}

\begin{prp}{}{}\\
Let $X$ be a reduced scheme. Let $Z\subseteq X$ be an irreducible component of $X$. Let $\eta$ be its corresponding generic point. Then the following are true. 
\begin{itemize}
\item $\mK_{X,\eta}$ is a field. 
\item We have $\mK_{X,\eta}=\mK(Z)$. 
\end{itemize}
\end{prp}

\begin{defn}{Function Field of an Integral Scheme}{}\\
Let $X$ be an integral scheme. Let $\eta$ be its unique generic point. Define the function field of $X$ to be $$k(X)=\mK_X(X)=\mK_{X,\eta}=\mO_{X,\eta}$$
\end{defn}

\begin{lmm}{}{}\\
Let $X$ be an integral scheme. If $\text{Spec}(R)\cong U\subseteq X$ is a non-empty affine open subset of $X$, then there is a natural isomorphism of rings $k(X)\cong\text{Frac}(R)$ given by the localization map. 
\begin{proof}\\
Let $\varphi:U\overset{\cong}{\to}\text{Spec}(R)$ be the scheme isomorphism. The ring homomorphism is given as follows. Then by 2.4.10 in Sheaf Theory, we have natural isomorphisms $$\mO_{X,\eta}\cong\left(\mO_X|_{\text{Spec}(R)}\right)_\eta\cong\mO_{\text{Spec}(R),\varphi(\eta)}\cong R_{\varphi(\eta)}$$ where the last isomorphism follows from 1.3.3. Then there is a natural localization map $\mO_{X,\eta}\cong R_{\varphi(\eta)}\to\text{Frac}(R)$. Now since $X$ is integral, $R\cong\mO_X(\text{Spec}(R))$ is an integral domain and so must have a unique generic point that is the zero ideal $(0)$. Since $\varphi(\eta)$ is also a generic point of $\text{Spec}(R)$, we must have $\varphi(\eta)=(0)$. Then $R_{\varphi(\eta)}=R_{(0)}=\text{Frac}(R)$ so our given map is an isomorphism. 
\end{proof}
\end{lmm}

\subsection{Noetherian Schemes}
\begin{defn}{Locally Noetherian Schemes}{}\\
Let $X$ be a scheme. We say that $X$ is locally Noetherian if there exists an open affine cover $\{U_i=\text{Spec}(A_i)\;|\;i\in I\}$ of $X$ such that each $A_i$ is Noetherian. 
\end{defn}

\begin{prp}{}{}\\
Let $X$ be a scheme. Then $X$ is locally Noetherian if and only if for every affine open subset $U\subseteq X$, we have $\mO_X(U)$ is Noetherian. 
\end{prp}

\begin{defn}{Noetherian Schemes}{}\\
Let $X$ be a scheme. We say that $X$ is Noetherian if $X$ is locally Noetherian and compact. 
\end{defn}

\begin{lmm}{}{}\\
Let $X$ be a scheme. Suppose that $X$ is Noetherian. Then the following are true. 
\begin{itemize}
\item The underlying space of $X$ is Noetherian. 
\item Every open subset of $X$ is compact. 
\end{itemize}
\begin{proof}\\
Since $X$ is Noetherian, there exists Noetherian rings $R_1,\dots,R_n$ such that $X=\bigcup_{i=1}^n\text{Spec}(R_i)$. By 1.2.6, each $\text{Spec}(R_i)$ is Noetherian. We have seen that a finite union of Noetherian subspaces is Noetherian in Algebraic Geometry 1. Hence the underlying space of $X$ is Noetherian. \\

We have also seen in Algebraic Geometry 1 that every open subset of a Noetherian space is compact. 
\end{proof}
\end{lmm}

\begin{lmm}{}{}\\
Let $R$ be a commutative ring. Then $R$ is Noetherian if and only if $\text{Spec}(R)$ is a Noetherian scheme. 
\end{lmm}

If $R$ is Noetherian, then the underlying topological space of $\text{Spec}(R)$ is also Noetherian. But the converse of this is not true in general.

\section{Finiteness Properties of Morphisms}
\subsection{Closure Properties of Morphism Classes}
\begin{defn}{Stable Under Composition}{}\\
Let $P$ be a property of morphisms of schemes. We say that $P$ is stable under composition if the following are true. Suppose that the following is a diagram in $\bold{Sch}$: \\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	X & Y & Z
	\arrow["f", from=1-1, to=1-2]
	\arrow["g", from=1-2, to=1-3]
\end{tikzcd}}\\
If $f$ and $g$ has property $P$, then $g\circ f$ has property $P$. 
\end{defn}

\begin{defn}{Stable Under Base Change}{}\\
Let $P$ be a property of morphisms of schemes. We say that $P$ is stable under composition if the following are true. Suppose that the following is a diagram in $\bold{Sch}$: \\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	{X\times_ZY} & Y \\
	X & Z
	\arrow["k", from=1-1, to=1-2]
	\arrow["h"', from=1-1, to=2-1]
	\arrow["g", from=1-2, to=2-2]
	\arrow["f"', from=2-1, to=2-2]
\end{tikzcd}}\\
If $f$ has property $P$, then $k$ has property $P$. 
\end{defn}

\begin{defn}{Stable Under Products}{}\\
Let $P$ be a property of morphisms of schemes. We say that $P$ is stable under composition if the following are true. Let $S$ be a scheme. Let $X_1,X_2,Y_1,Y_2$ be schemes. Let $f_1:X_1\to Y_1$ and $f_2:X_2\to Y_2$ be morphisms. If $f_1,f_2$ has property $P$, then $$f_1\times_Sf_2:X_1\times_SX_2\to Y_1\times_SY_2$$ has property $P$. 
\end{defn}

\begin{lmm}{}{}\\
Let $P$ be a property of morphisms of schemes. Suppose that $P$ is stable under composition and base change. Then $P$ is stable under products. 
\begin{proof}\\
Consider the following diagram: \\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	& {X\times_SX'} & {X\times_SY'} & {Y\times_SY'} \\
	{X'} & {Y'} && X & Y
	\arrow[from=1-2, to=1-3]
	\arrow[from=1-2, to=2-1]
	\arrow[from=1-3, to=1-4]
	\arrow[from=1-3, to=2-2]
	\arrow[from=1-3, to=2-4]
	\arrow[from=1-4, to=2-5]
	\arrow[from=2-1, to=2-2]
	\arrow[from=2-4, to=2-5]
\end{tikzcd}}\\
I claim that the two squares are pullback squares. Indeed, there are canonical isomorphisms $X'\times_{Y'}X\times_SY'\cong X\times_S X'$ and $X\times_Y Y\times_SY'\cong X\times_SY'$. Then since the property $P$ is stable under composition and base change, we are done. 
\end{proof}
\end{lmm}

\begin{defn}{Local on the Target}{}\\
Let $P$ be a property of morphisms of schemes. We say that $P$ is local on the target if the following are true. 
\begin{itemize}
\item If $f:X\to Y$ has property $P$, then for all open subsets $V\subseteq Y$, the morphism $f^{-1}(V)\to V$ has property $P$. 
\item If $f:X\to Y$ is a morphism and there exists an open cover $Y=\bigcup_{i\in I}V_i$ such that $f^{-1}(V_i)\to V_i$ has property $P$, then $f$ has property $P$. 
\end{itemize}
\end{defn}

\subsection{Affine Morphisms}
\begin{defn}{Affine Morphisms}{}\\
Let $X,Y$ be schemes. Let $f:X\to Y$ be a morphism. We say that $f$ is affine if for all affine open subsets $V\subseteq Y$, we have $f^{-1}(V)$ is affine open. 
\end{defn}

\begin{lmm}{}{}\\
Let $X,Y$ be schemes. Let $f:X\to Y$ be a morphism. Then $f$ is an affine morphism if and only if there exists an affine open cover $Y=\bigcup_{i\in I}V_i$ such that $f^{-1}(V_i)$ is affine open. 
\end{lmm}

\begin{lmm}{}{}\\
The class of affine morphisms satisfy the following properties. 
\begin{itemize}
\item Closed under composition. 
\item Closed under base change. 
\item Local on the Target. 
\item Closed under products. 
\end{itemize}
\end{lmm}

\subsection{Open, Closed and Locally Closed Embeddings}
\begin{defn}{Open Embeddings (Immersion)}{}\\
Let $X,Y$ be schemes. Let $f:X\to Y$ be a morphism. We say that $f$ is an open immersion if $f$ induces an isomorphism $X\cong f(X)$ where $f(X)$ is an open subset of $Y$. 
\end{defn}

\begin{lmm}{}{}\\
The class of open embeddings satisfy the following properties. 
\begin{itemize}
\item Closed under composition. 
\item Closed under base change. 
\item Local on the Target. 
\item Closed under products. 
\end{itemize}
\begin{proof}\\
Let $f:X\to Y$ and $g:Y\to Z$ be open embeddings. Then we know that $X\cong f(X)$ and $Y\cong g(Y)$. Hence $X\cong g(f(X))$. Moreover, since $f(X)$ is an open subset of $Y$, the isomorphism $Y\cong g(Y)$ implies that $g(f(X))$ is an open subset of $g(Y)$, and hence an open subset of $Z$. Hence open embeddings are closed under composition. \\

Let $f:X\to Z$ be an open embedding. Let $g:Y\to Z$ be a morphism. As a space, the pullback of $f$ and $g$ is precisely $g^{-1}(f(X))\subseteq Y$, and evidently its structure sheaf is given by $\mO_Y|_{g^{-1}(f(X))}$. Evidently, the morphism $(g^{-1}(f(X)),\mO_Y|_{g^{-1}(f(X))})\to Y$ is given by inclusion. It is an open embedding since $g^{-1}(f(X))$ is an open subset of $Y$ since $f(X)$ is an open subset of $Z$. \\

Let $f:X\to Y$ be an open embedding. Let $V\subseteq Y$ be open. Since $X\cong f(X)$, we must have that $V\cap f(X)$ is the the image of $f^{-1}(V)$ and is an open subset of $Y$. Conversely, suppose that $f:X\to Y$ is a morphism and $Y=\bigcup_{i\in I}V_i$ is an open cover of $Y$ such that $f^{-1}(V_i)\to V_i$ is an open embedding. Notice that $f^{-1}(V_i)$ for $i\in I$ forms an open cover for $X$. Let $U\subseteq X$ be open. For each $i\in I$, $U\cap f^{-1}(V_i)\subseteq f^{-1}(V_i)$ is open and its image $f(U\cap f^{-1}(V_i))$ is open. Then $f(U)=\bigcup_{i\in I}f(U\cap f^{-1}(V_i))$ is open. Hence $f$ is a homeomorphism of underlying spaces. Since $\mO_X|_U=\mO_U$, it follows that the structures sheaves of $X$ and $f(X)$ are also isomorphic. Hence open embeddings are local on target. \\

Since open embeddings are closed under composition and base change, by the above they must also be closed under products. 
\end{proof}
\end{lmm}

\begin{defn}{Closed Embeddings (Closed Immersions)}{}\\
Let $X,Y$ be schemes. Let $f:X\to Y$ be a morphism. We say that $f$ is a closed embedding if the following are true. 
\begin{itemize}
\item $X\cong f(X)$ and $f(X)\subseteq Y$ is a closed subset. 
\item $\mO_Y\to f_\ast\mO_X$ is surjective. 
\end{itemize}
\end{defn}

\begin{prp}{}{}\\
Let $X,Y$ be schemes. Let $f:X\to Y$ be a morphism. Then $f$ is a closed embedding if and only if the following are true. 
\begin{itemize}
\item $f$ is an affine morphism. 
\item For any affine open subset $V\subseteq Y$, the ring homomorphism $\mO_Y(V)\to\mO_X(f^{-1}(V))$ is surjective. 
\end{itemize}
\end{prp}

\begin{lmm}{}{}\\
Let $X,Y$ be schemes. Let $f:X\to Y$ be a morphism. Then $f$ is a closed embedding if and only if there exists an affine open cover $Y=\bigcup_{i\in I}V_i$ such that $f^{-1}(V_i)\to V_i$ is a closed embedding. 
\begin{proof}\\
If $f$ is a closed embedding, then $f^{-1}(V_i)\to V_i$ must also be a closed embedding. Conversely, if $f^{-1}(V_i)\to V_i$ is a closed embedding for all $i\in I$, then by 7.2.2 $f$ is an affine morphism. Moreover, 
\end{proof}
\end{lmm}

\begin{lmm}{}{}\\
The class of closed embeddings satisfy the following properties. 
\begin{itemize}
\item Closed under composition. 
\end{itemize}
\begin{proof}\\
Since surjectivity and affine morphisms are closed under composition, closed embeddings must also be closed under composition. \\
\end{proof}
\end{lmm}

\begin{lmm}{}{}\\
Let $R,S$ be commutative rings. Let $\varphi:R\to S$ be a ring homomorphism. If $\varphi$ is surjective, then $\text{Spec}(\varphi):\text{Spec}(S)\to\text{Spec}(R)$ is a closed embedding. 
\end{lmm}

\begin{defn}{Locally Closed Embeddings}{}\\
Let $X,Y$ be schemes. Let $f:X\to Y$ be a morphism. We say that $f$ is a locally closed embedding if $f$ factors into a closed embedding followed by an open embedding. 
\end{defn}

\begin{prp}{}{}\\
Let $X,Y$ be schemes. Let $f:X\to Y$ be a locally closed embedding. If $\im(f)$ is a closed subset of $X$, then $f$ is a closed embedding. 
\end{prp}

\subsection{Morphisms of Finite Types}
\begin{defn}{Morphisms Locally of Finite Type}{}\\
Let $X,Y$ be schemes. Let $f:X\to Y$ be a morphism. We say that $f$ is locally of finite type if for every open affine subset $V$ of $Y$, there exists an affine open cover $f^{-1}(V)=\bigcup_{i\in I}U_{ij}$ such that $\mO_X(U_{ij})$ is a finitely generated $\mO_X(V)$-algebra for all $i$. 
\end{defn}

\begin{prp}{}{}\\
Let $X,Y$ be schemes. Let $f:X\to Y$ be a morphism Then the following are equivalent. 
\begin{itemize}
\item $f$ is locally of finite type. 
\item There exists an affine open cover $Y=\bigcup_{i\in I}V_i$ such that there exists an affine open cover $f^{-1}(V_i)=\bigcup_{j\in J}U_{ij}$ such that $\mO_X(U_{ij})$ is a finitely generated $\mO_Y(V_i)$-algebra. 
\item For every affine open subset $V\subseteq Y$ and every affine open subset $U\subseteq f^{-1}(V)$, we have $\mO_X(U)$ is a finitely generated $\mO_Y(V)$-algebra. 
\end{itemize}
\end{prp}

\begin{lmm}{}{}\\
The class of locally of finite type morphisms satisfy the following properties. 
\begin{itemize}
\item Closed under composition. 
\item Closed under base change. 
\item Local on the Target. 
\item Closed under products. 
\end{itemize}
\end{lmm}

\begin{defn}{Morphisms of Finite Type}{}\\
Let $X,Y$ be schemes. Let $f:X\to Y$ be a morphism. We say that $f$ is of finite type if for every affine open subset $V$ of $Y$, there is a finite affine open cover $f^{-1}(V)=\bigcup_{i=1}^nU_i$ for $f^{-1}(V)$ such that $mO_X(U_i)$ is a finitely generated $\mO_Y(V)$-algebra for $1\leq i\leq n$. 
\end{defn}

It is easy to see that we have the following results. 
\begin{itemize}
\item Let $R$ be a commutative ring. Then the adjunction $\Gamma:\bold{Sch}_R\overset{\dashv}{\rightleftarrows}\bold{CAlg}_R:\text{Spec}$ restricts to an adjunction between schemes of finite type over $R$ and finitely generated algebras over $R$. 
\item Let $R$ be a commutative ring. The fully faithful functor $\widetilde{(-)}:\bold{Sch}_R\to\Hom_{\bold{Cat}}(\bold{CAlg}_R,\bold{Set})$ restricts to a fully faithful functor from schemes of finite types to $\Hom_{\bold{Cat}}(\bold{FGCAlg}_R,\bold{Set})$. 
\end{itemize}

\begin{lmm}{}{}\\
The class of finite type morphisms satisfy the following properties. 
\begin{itemize}
\item Closed under composition. 
\item Closed under base change. 
\item Local on the Target. 
\item Closed under products. 
\end{itemize}
\end{lmm}

\begin{lmm}{}{}\\
Let $k$ be a field. Let $X$ be a scheme of finite type over $k$. Then $X(k)$ is Noetherian. 
\end{lmm}

\subsection{Finite Morphisms}
\begin{defn}{Finite Morphisms}{}\\
Let $X,Y$ be schemes. Let $f:X\to Y$ be a morphism. We say that $f$ is finite if the following are true. 
\begin{itemize}
\item $f$ is an affine morphism. 
\item For every affine open subset $V\subseteq Y$, we have $\mO_X(f^{-1}(V))$ is a finitely generated $\mO_Y(V)$-module. 
\end{itemize}
\end{defn}

\begin{lmm}{}{}\\
Let $X,Y$ be schemes. Let $f:X\to Y$ be a morphism. Then $f$ is finite if and only if the following are true. 
\begin{itemize}
\item $f$ is an affine morphism. 
\item There exists an affine open cover $Y=\bigcup_{i\in I}V_i$ of $Y$ such that $\mO_X(f^{-1}(V_i))$ is a finitely generated $\mO_Y(V_i)$-module for all $i\in I$. 
\end{itemize}
\end{lmm}

\begin{lmm}{}{}\\
Let $X,Y$ be schemes. Let $f:X\to Y$ be a closed embedding. Then $f$ is finite. 
\begin{proof}\\
We have seen that $f$ is affine. Moreover, for any affine open subset $V\subseteq Y$, we have that $\mO_Y(V)\to\mO_X(f^{-1}(V))$ is surjective. Then $\mO_X(f^{-1}(V))$ is generated by $1$ as an $\mO_Y(V)$-module. Indeed, for any $f\in\mO_X(f^{-1}(V))$, there exists $g\in\mO_Y(V)$ such that $g$ is mapped to $f$. Then by definition of the module structure, we have $f=g\cdot 1$. Hence $\mO_X(f^{-1}(V))$ is generated by $1$. 
\end{proof}
\end{lmm}

\begin{lmm}{}{}\\
The class of finite morphisms satisfy the following properties. 
\begin{itemize}
\item Closed under composition. 
\item Closed under base change. 
\item Local on the Target. 
\item Closed under products. 
\end{itemize}
\begin{proof}\\
Let $f:X\to Y$ and $g:Y\to Z$ be finite morphisms. We know that $g\circ f$ is affine. Also, let $W\subseteq Z$ be affine open. Then $\mO_Y(g^{-1}(W))$ is a finitely generated $\mO_Z(W)$-module, and $\mO_X(f^{-1}(g^{-1}(W)))$ is a finitely generated $\mO_Y(g^{-1}(W))$-module. Hence $\mO_X(f^{-1}(g^{-1}(W)))$ is a finitely generated $\mO_Z(W)$-module. Hence $g\circ f$ is finite. 
\end{proof}
\end{lmm}

\begin{prp}{}{}\\
Let $X,Y$ be schemes. Let $f:X\to Y$ be a morphism. If $f$ is finite, then the following are true. 
\begin{itemize}
\item For all $y\in Y$, $f^{-1}(y)$ is finite. 
\item $f$ is closed. 
\end{itemize}
\end{prp}

\begin{eg}{}{}\\
Let $k$ be a field. The inclusion $\A_k^2\setminus\{(x,y)\}\hookrightarrow\A_k^2$ has finite fibers but is a finite morphism. 
\begin{proof}\\
Clearly it has finite fibers. However, the inverse image of $\A_k^2$, which is $\A_k^2\setminus\{(x,y)\}$ is not affine. Hence it is not an affine morphism and is not finite. 
\end{proof}
\end{eg}

\begin{eg}{}{}\\
The inclusion $\A_\C^1\setminus\{(x)\}\hookrightarrow\A_\C^1$ has finite fibers and is affine, but it is not a finite morphism. 
\begin{proof}\\
Clearly it has finite fibers. It is also affine since $\A_\C^1\setminus\{(x)\}$ is isomorphic to $\text{Spec}(\C[u,v]/(uv-1))$ via the morphism $\A_\C^1\setminus\{(x)\}\to\text{Spec}(\C[u,v]/(uv-1))$ that is adjoint to the identity $\C[x,y]/(uv-1)\to\Gamma(\A_C^1\setminus\{(x)\})\cong\frac{\C[u,v]}{(uv-1)}$. \\

Now the morphism $\text{Spec}(\C[u,v]/(uv-1))\to\A_\C^1$ is given by the ring homomorphism $\C[x]\to\frac{\C[u,v]}{(uv-1)}$ defined by $x\mapsto(u,1/u)$. Then $\frac{\C[u,v]}{(uv-1)}\cong\C[u,1/u]$ is not a finitely generated $\C[x]$-module. 
\end{proof}
\end{eg}

\subsection{Integral Morphisms}
\begin{defn}{Integral Morphisms}{}\\
Let $X,Y$ be schemes. Let $f:X\to Y$ be a morphism. We say that $f$ is integral if the following are true. 
\begin{itemize}
\item $f$ is affine. 
\item For all affine open subsets $V\subseteq Y$, we have $\mO_Y(V)$ is integral over $\mO_X(f^{-1}(V))$. 
\end{itemize}
\end{defn}

\begin{prp}{}{}\\
Let $X,Y$ be schemes. Let $f:X\to Y$ be a morphism. Then $f$ is integral if and only if the following are true. 
\begin{itemize}
\item $f$ is affine. 
\item There exists an affine open cover $Y=\bigcup_{i\in I}V_i$ such that $\mO_Y(V_i)$ is integral over $\mO_X(f^{-1}(V_i))$ for all $i\in I$. 
\end{itemize}
\end{prp}

\begin{lmm}{}{}\\
The class of integral morphisms satisfy the following properties. 
\begin{itemize}
\item Closed under composition. 
\item Closed under base change. 
\item Local on the Target. 
\item Closed under products. 
\end{itemize}
\end{lmm}

\begin{prp}{}{}\\
Let $X,Y$ be schemes. Let $f:X\to Y$ be a morphism. Then $f$ is finite if and only if the following are true. 
\begin{itemize}
\item $f$ is a morphism of finite type. 
\item $f$ is integral. 
\end{itemize}
\begin{proof}\\
Recall from Commutative Algebra 1 that a given a morphism of commutative rings $B\to A$, we have $A$ is a finitely generated $B$-module if and only if $A$ is a finitely generated $B$-algebra that is integral over $B$. Thus we are done. 
\end{proof}
\end{prp}

\section{Separated and Proper Morphisms}
\subsection{Separated Morphisms}
Separatedness is essentially the analog of the Hausdorff condition for schemes. Recall that a topological space $X$ is Hausdorff if and only if the digonal morphism to $X\times X$ is closed. 

\begin{eg}{}{}\\
Let $U_0$ and $U_1$ be the canonical affine open cover of $\Prj^1$. Let $V_0=D(x_1/x_0)$ and $V_1=D(x_0/x_1)$ be the localization of $U_1,U_2$ at $x_1/x_0$ and $x_0/x_1$ respectively. The gluing morphism $V_0\to V_1$ defined by multiplication by $x_0/x_1$ recovers $\Prj^1$. Now consider the gluing morphism given by mapping $x_1/x_0$ to $x_0/x_1$. This is an isomorphism and and we obtain a scheme $X$ that consists of $k^\ast$ together with two points at infinity similar to $\Prj^1$. Now consider the maps $k[t]\to k[x_1/x_0]$ and $k[t]\to k[x_0/x_1]$ given by mapping $t$ to $x_1/x_0$ and $x_0/x_1$ respectively. They glue into a morphism $X\to\text{Spec}(k[t])$ whose fiber over $(t)\in\text{Spec}(k[t])$ is given by two distinct points $(x_1/x_0)$ and $(x_0/x_1)$. For any open sets $U_1,U_2$ that contains $(x_1/x_0)$ and $(x_0/x_1)$ respectively in the classical topology, we must have $U_1\cap U_2\neq\emptyset$ so that $X$ is not Hausdorff in the classical topology. 
\end{eg}

\begin{defn}{Diagonal Morphisms}{} Let $f:X\to Y$ be a morphism of schemes. The diagonal morphism is the unique morphism $\delta:X\to X\times_Y X$ whose composition with both projection maps $p_1,p_2:X\times_Y X\to X$ is the identity map of $X$. 
\end{defn}

\begin{lmm}{}{}\\
Let $X,Y$ be schemes. Let $f:X\to Y$ be a morphism. Then $\delta:X\to X\times_YX$ is a locally closed embedding. 
\end{lmm}

\begin{defn}{Separated Morphisms and Schemes}{} Let $f:X\to Y$ be a morphism of schemes. We say that $f$ is separated (or $X$ is separated over $Y$) if the diagonal morphism $\delta$ is a closed immersion. A scheme $X$ is separated if it is separated over $\text{Spec}(\Z)$. 
\end{defn}

\begin{prp}{}{} Let $f:X\to Y$ be a morphism of schemes. Then $f$ is separated if and only if the image of the diagonal morphism is a closed subset of $X\times_Y X$. 
\end{prp}

\begin{prp}{}{}\\
Let $X,Y$ be affine schemes. Let $f:X\to Y$ be a morphism. Then $f$ is separated. 
\end{prp}

\begin{thm}{Valuative Criterion of Separatedness}{}\\
Let $X,Y$ be schemes. Let $f:X\to Y$ be a morphism. Suppose that $X$ is Noetherian. Then $f$ is proper if and only if for every valuation ring $(R,m)$ and every morphism $\text{Spec}(R/m)\to X$ and $\text{Spec}(R)\to Y$, any morphism $\text{Spec}(R)\to X$ making the following diagram commutes: \\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	{\text{Spec}(R/m)} & X \\
	{\text{Spec}(R)} & Y
	\arrow[from=1-1, to=1-2]
	\arrow["{\text{proj.}^\flat}"', from=1-1, to=2-1]
	\arrow["f", from=1-2, to=2-2]
	\arrow["{\exists!}"{description}, dashed, from=2-1, to=1-2]
	\arrow[from=2-1, to=2-2]
\end{tikzcd}} \\
is unique. 
\end{thm}

\begin{lmm}{}{}\\
The class of separated morphisms satisfy the following properties. 
\begin{itemize}
\item Closed under composition. 
\item Closed under base change. 
\item Local on the Target. 
\item Closed under products. 
\end{itemize}
\end{lmm}

\begin{prp}{}{} Let $X$ and $Y$ be Noetherian schemes. Then any open or closed immersions $f:X\to Y$ are separated. 
\end{prp}

\begin{lmm}{}{}\\
Let $R$ be a commutative ring. Let $X$ be a scheme over $R$. Then $X$ is separated over $\text{Spec}(R)$ if and only if $X$ is separated over $\text{Spec}(\Z)$. 
\end{lmm}

\begin{lmm}{}{}\\
Let $X$ be scheme. If $X$ is separated, then the intersection of affine open subsets are affine open. 
\end{lmm}

\subsection{Proper Morphisms}
\begin{defn}{Universally Closed Morphisms}{}\\
Let $X,Y$ be schemes. Let $f:X\to Y$ be a morphism. We say that $f$ is universally close if for any scheme $X'$ and any morphism $g:X'\to Y$, the induced morphism $X\times_YX'\to X'$ is a closed morphism. 
\end{defn}

\begin{defn}{Proper Morphisms}{}\\
Let $X,Y$ be schemes. Let $f:X\to Y$ be a morphism. We say that $f$ is a proper morphism if $f$ is separated, of finite type and is universally closed. 
\end{defn}

Let $X$ be a scheme over a field $k$. The condition for the structure morphism $X\to\text{Spec}(k)$ to be proper is to test that it is separated, of finite type and that for any scheme $Y$ over $k$, the morphism $X\times_kY\to Y$ is closed. 

\begin{thm}{Valuative Criterion for Properness}{}\\
Let $X,Y$ be schemes. Let $f:X\to Y$ be a morphism of finite type. Suppose that $X$ is Noetherian. Then $f$ is proper if and only if for every valuation ring $(R,m)$ and every morphism $\text{Spec}(R/m)\to X$ and $\text{Spec}(R)\to Y$, there exists a unqiue morphism $\text{Spec}(R)\to X$ such that the following diagram commutes: \\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	{\text{Spec}(R/m)} & X \\
	{\text{Spec}(R)} & Y
	\arrow[from=1-1, to=1-2]
	\arrow["{\text{proj.}^\flat}"', from=1-1, to=2-1]
	\arrow["f", from=1-2, to=2-2]
	\arrow["{\exists!}"{description}, dashed, from=2-1, to=1-2]
	\arrow[from=2-1, to=2-2]
\end{tikzcd}} \\
\end{thm}

\begin{lmm}{}{}\\
The class of proper morphisms satisfy the following properties. 
\begin{itemize}
\item Closed under composition. 
\item Closed under base change. 
\item Local on the Target. 
\item Closed under products. 
\end{itemize}
\end{lmm}

\begin{prp}{}{}\\
Let $X,Y$ be schemes. Let $f:X\to Y$ be a finite morphism. Then $f$ is proper. 
\end{prp}

\begin{prp}{}{}\\
Let $X,Y$ be affine schemes. Let $f:X\to Y$ be a morphism. Then $f$ is proper if and only if $f$ is finite. 
\end{prp}

\section{Relation to Classical Varieties}
\subsection{Varieties Revisited}
\begin{defn}{The Associated Scheme of a Variety}{}\\
Let $k$ be a field. Let $X$ be a variety over $k$. Define the associated scheme of $X$ as follows. 
\begin{itemize}
\item The topological space is given by $t(X)=\{Z\subseteq X\;|\;Z\text{ is irreducible and closed }\}$ whose closed subsets are of the form $t(Z)$ for $Z\subseteq X$ an irreducible and closed subset of $X$. 
\item Let $\alpha:X\to t(X)$ be the continuous map $\alpha(p)=\overline{\{p\}}$. Then the sheaf on $t(X)$ is the pushforward sheaf $\alpha_\ast(\mO_X)$. 
\end{itemize}
\end{defn}

\begin{defn}{The Associated Scheme of a Variety}{}\\
Let $k$ be a field. Define the functor $$t:\bold{Var}_k\to\bold{Sch}_k$$ as follows. 
\begin{itemize}
\item Each variety $X$ over $k$ is associated to the scheme $t(X)$. 
\item For a morphism of varieties $f:X\to Y$, $t(f)$ is the map defined by $t(f)(Z)=\overline{f(Z)}$ for $Z\subseteq X$ an irreducible closed subset of $X$. 
\end{itemize}
\end{defn}

\begin{prp}{}{}\\
Let $k$ be an algebraically closed field. Then the following are true. 
\begin{itemize}
\item The functor $t:\bold{Var}_k\to\bold{Sch}_k$ is fully faithful. 
\item The essential image of $t$ consists of the Noetherian, reduced and separated schemes of finite type over $k$. 
\item Restricted to the essential image, the inverse of $t$ is given by the functor defined by $X\mapsto(X(k),\mO_X|_{X(k)})$. 
\item $X$ is an affine variety if and only if $t(X)$ is an affine scheme. 
\item $X$ is a projective variety if and only if $t(X)$ is a projective scheme. 
\end{itemize}
\end{prp}

This motivates the following definition. 

\begin{defn}{Varieties}{} Let $k$ be a field. A variety over $k$ is a scheme $X$ such that $X$ is an reduced separated scheme of finite type over $k$. 
\end{defn}

\begin{lmm}{}{}\\
Let $k$ be a field. Let $X$ be a variety over $k$. Let $x\in X$. Then the following are equivalent. 
\begin{itemize}
\item $x$ is a closed point of $X$. 
\item The field extension $k\to\frac{\mO_{X,x}}{m}$ is finite. 
\item The field extension $k\to\frac{\mO_{X,x}}{m}$ is algebraic. 
\end{itemize}
\end{lmm}

\begin{crl}{}{}\\
Let $k$ be an algebraically closed field. Let $X$ be a variety over $k$. Then there is a natural bijection $$X(k)\cong\{x\in X\;|\;x\text{ is a closed point of }X\}$$
\end{crl}

\begin{defn}{The Category of Varieties}{} Define the category of varieties $\text{Var}_k$ over a field $k$ as follows. 
\begin{itemize}
\item The objects are varieties $X$ over $k$
\item The morphisms are morphisms of schemes $X\to Y$ over $k$. 
\item Composition is given by the composition of morphisms. 
\end{itemize}
\end{defn}

In virtue of the above proposition, this new category of varieties is equivalent to the category of classical varieties. 

\begin{prp}{}{}\\
Let $k$ be a perfect field. Let $X,Y$ be varieties over $X$. Then the following are true. 
\begin{itemize}
\item $X\times Y$ is a variety over $k$. 
\item If $X,Y$ are affine, then $X\times Y$ is affine. 
\item If $X,Y$ are projective, then $X\times Y$ is projective. 
\end{itemize}
\end{prp}

\begin{prp}{}{}\\
Let $k$ be an algebraically closed field. Let $X,Y$ be varieties over $X$. Then the following are true. 
\begin{itemize}
\item If $X,Y$ are irreducible, then $X\times Y$ is irreducible. 
\item If $X,Y$ are connected, then $X\times Y$ is connected. 
\end{itemize}
\end{prp}

\begin{prp}{}{}\\
Let $k$ be a field. Let $X,Y$ be varieties over $k$. Let $\phi:X\to Y$ be a morphism of schemes over $k$. Then $\phi$ is separated and of finite type over $k$. 
\end{prp}

\subsection{Complete Varieties}
\begin{defn}{Complete Varieties}{} Let $k$ be a field. We say that a variety $X$ over $k$ is complete if it is proper over $k$. 
\end{defn}

\begin{lmm}{}{}\\
Let $k$ be a field. Let $X,Y$ be a varieties over $k$. If $X$ and $Y$ are complete, then $X\times Y$ is complete. 
\end{lmm}

\begin{prp}{}{}\\
Let $k$ be a field. Let $X$ be a variety over $k$. Let $Y\subseteq X$ be a subvariety of $X$. Then the following are true. 
\begin{itemize}
\item If $X$ is complete and $Y$ is closed in $X$, then $Y$ is complete. 
\item If $Y$ is complete, then $Y$ is closed in $X$. 
\end{itemize}
\end{prp}

\begin{lmm}{}{}\\
Let $k$ be a field. Let $X$ be an affine variety over $k$. If $X$ is complete, then $\dim(X)=0$. 
\end{lmm}

\begin{prp}{}{}\\
Let $k$ be a field. Let $X$ be a variety over $k$. If $X$ is projective, then $X$ is complete. 
\end{prp}

\section{Sheaf of Modules over a Scheme}
\subsection{The Associated Sheaf of Modules}
We restate the definition here of a sheaf of modules here. Let $\mA$ be a sheaf of rings over $X$. Let $U$ be an open set of $X$. A sheaf of $\mA$-modules over $X$ is a sheaf $\mF$ such that each $\mF(U)$ is an $\mA(U)$-modules. Moreover, for each inclusion of open sets $V\subseteq U$, the restriction homomorphism $\mathcal{F}(U)\to\mathcal{F}(V)$ is such that the following diagram commutes: \\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
\mA(U)\times\mF(U)\arrow[r, "\text{action}"]\arrow[d, "\text{res}_{U,V}\times\text{res}_{U,V}"'] &\mF(U)\arrow[d, "\text{res}_{U,V}"]\\
\mA(V)\times\mF(V)\arrow[r, "\text{action}"] & \mF(V)
\end{tikzcd}}\\

For a scheme $(X,\mO_X)$ that is also a locally ringed space, we denote the category of sheaves of $\mO_X$-modules as $\text{Mod}_{\mO_X}$. \\

It is the analogue of a module over a ring for the following reason. 

\begin{defn}{Associated Sheaf}{}\\
Let $R$ be a commutative ring. Let $M$ be an $R$-module. Define a sheaf $\widetilde{M}$ on $\text{Spec}(R)$ as follows. 
\begin{itemize}
\item For each open set $U\subseteq\text{Spec}(R)$, define $$\widetilde{M}(U)=\left\{(s_p)_{p\in U}\in\prod_{p\in U}M_p\;\bigg{|}\;\substack{\forall p,\exists p\in\;U_p\subseteq U\text{ open and }m\in M,f\in R\text{ s.t.}\\q\in U_p\text{ implies }f\notin q\text{ and }s_q=m/f\in M_q}\right\}$$
\item For $V\subseteq U$ an inclusion, define the unique morphism $\widetilde{M}(U)\to\widetilde{M}(V)$ given by restriction. 
\end{itemize}
\end{defn}

\begin{lmm}{}{}\\
Let $M$ be an $R$-module. Then the associated sheaf is a sheaf of $\mathcal{O}_{\text{Spec}(R)}$-modules. 
\begin{proof}\\
Same proof as 1.3.2. 
\end{proof}
\end{lmm}

\begin{prp}{}{}\\
Let $R$ be a commutative ring. Let $M$ be an $R$-module. Then the following are true regarding the associated sheaf $\widetilde{M}$. 
\begin{itemize}
\item For each $p\in\text{Spec}(R)$, there is a natural isomorphism $\widetilde{M}_p\cong M_p$
\item For any $f\in R$, there is a natural isomorphism $\widetilde{M}(D(f))\cong M_f$ of $R_f$-modules
\item There is a natural isomorphism $\widetilde{M}(\text{Spec}(R))\cong M$. 
\end{itemize}
\begin{proof}
Same proof as 1.3.3. 
\end{proof}
\end{prp}

\begin{defn}{The Functor of Associated Sheaf}{}\\
Let $R$ be a commutative ring. Define the associated sheaf functor $$\widetilde{(-)}:\bold{Mod}_R\to\bold{Mod}_{\mO_{\text{Spec}(R)}}$$ as follows. 
\begin{itemize}
\item For each $R$-module $M$, define $\widetilde{M}$ to be the associated sheaf. 
\item For each $R$-module homomorphism $\varphi:M\to N$, define the induced morphism $\widetilde{\varphi}$ on each open set $U\subseteq X$ as follows. For each $p\in U$, there is an induced $R$-module homomorphism $M_p\to N_p$. Define $\widetilde{\varphi}$ by the restriction of the product $$\prod_{p\in U}M_p\to\prod_{p\in U}N_p$$ to $\widetilde{M}(U)$. 
\end{itemize}
\end{defn}

We have to check that the image of the restriction of the product lies inside $\widetilde{N}(U)$. 

\begin{prp}{}{}\\
Let $R$ be a commutative ring. Then $\widetilde{(-)}$ is fully faithful. 
\begin{proof}
Let $\varphi,\psi:M\to N$ be $R$-module homomorphisms. Since the $R$-module isomorphism $M=M_1\cong\widetilde{M}(\text{Spec}(R))$ is natural, we obtain a commutative diagram: \\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	M & N \\
	{\widetilde{M}(\text{Spec}(R))} & {\widetilde{N}(\text{Spec}(R))}
	\arrow["{\varphi,\psi}", from=1-1, to=1-2]
	\arrow["\cong"', from=1-1, to=2-1]
	\arrow["\cong", from=1-2, to=2-2]
	\arrow["{\widetilde{\varphi}=\widetilde{\psi}}"', from=2-1, to=2-2]
\end{tikzcd}}\\
Hence $\varphi=\psi$. Thus $\widetilde{(-)}$ is faithful. Now let $f:\widetilde{M}\to\widetilde{N}$ be a morphism of $\mO_\text{Spec}(R)$-modules. Consider the following diagram: \\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	&&& {M_p} \\
	{\widetilde{M}(\text{Spec}(R))} && {\widetilde{M}_p} \\
	& {\widetilde{M}(U)} && {N_p} \\
	{\widetilde{N}(\text{Spec}(R))} && {\widetilde{N}_p} \\
	& {\widetilde{N}(U)}
	\arrow["{f(\text{Spec}(R))_p}", from=1-4, to=3-4]
	\arrow[from=2-1, to=1-4]
	\arrow[from=2-1, to=2-3]
	\arrow[from=2-1, to=3-2]
	\arrow["{f(\text{Spec}(R))}"', from=2-1, to=4-1]
	\arrow["{\exists!}", dashed, from=2-3, to=1-4]
	\arrow["{\exists!f_p}", dashed, from=2-3, to=4-3]
	\arrow[from=3-2, to=2-3]
	\arrow["{f(U)}"{description, pos=0.7}, from=3-2, to=5-2]
	\arrow[from=4-1, to=3-4]
	\arrow[from=4-1, to=4-3]
	\arrow[from=4-1, to=5-2]
	\arrow["{\exists!}"', dashed, from=4-3, to=3-4]
	\arrow[from=5-2, to=4-3]
\end{tikzcd}}\\
It is commutative by the naturality of colimits and localization. Thus the morphisms $\widetilde{M}(U)\to\widetilde{N}(U)$ is given by the product of the localizations $f(\text{Spec}(R))_p$ for each $p\in U$. By construction of $\widetilde{(-)}$, the morphisms $\widetilde{f(\text{Spec}(R))}(U)$ is also given by the product of the localizations. Thus the morphisms $\widetilde{f(\text{Spec}(R)}$ and $f$ are the same. Thus the preimage of $f$ is given by $f(\text{Spec}(R))$. Hence the functor $\widetilde{(-)}$ is full. 
\end{proof}
\end{prp}

\begin{prp}{}{} Let $R$ be a commutative ring. Then there is an adjunction $$\widetilde{(\;\cdot\;)}:\bold{Mod}_R\rightleftarrows\bold{Mod}_{\mO_{\text{Spec}(R)}}:\Gamma$$
\end{prp}

\begin{prp}{}{}\\
Let $R$ be a commutative ring. Then $\widetilde{(-)}$ and $\Gamma$ are exact functors. 
\end{prp}

\subsection{The Associated Sheaf of Graded Modules}
\begin{defn}{Sheaves of Modules on Graded Rings}{} Let $S$ be a graded ring. Let $M$ be a graded $S$-module. Consider the module $$M_{(p)}=T^{-1}M$$ where $T$ is the multiplicative system of homogenous elements of $S$ not in $p$. Define the sheaf associated to $M$ on $\text{Proj}(S)$, $$\widetilde{M}:\bold{Open}(\text{Proj}(S))\to\bold{Rings}$$ as follows. 
\begin{itemize}
\item For each $U\subseteq\text{Proj}(S)$ open, define $$\mO_{\text{Proj}(S)}(U)=\left\{s:U\to\coprod_{p\in U}M_{(p)}\;\bigg{|}\;\substack{\forall p\in U,\;s(p)\in M_p\text{ and }\exists\;U_p\subseteq U\text{ s.t. }q\in V\text{ implies }\\s(q)=\frac{m}{f}\in M_q\text{ for }f\in S\text{ and }m\in M\text{ homogenous}}\right\}$$
\item For $V\subseteq U$ an inclusion, define the unique morphism $\widetilde{M}(U)\to\widetilde{M}(V)$ by restriction. 
\end{itemize}
\end{defn}

\begin{prp}{}{} Let $S$ be a graded ring and let $M$ be a graded module over $S$. Then the following are true regarding the sheaf of modules $\widetilde{M}$. 
\begin{itemize}
\item For any $p\in\text{Proj}(S)$, there is an isomorphism $\widetilde{M}_p\cong M_{(p)}$
\item For any homogenous $f\in S_+$, there is an isomorphism $$\widetilde{M}|_{D_+(f)}\cong\widetilde{M_{(f)}}$$ via the isomorphism of $D_+(f)$ with $\text{Spec}(S_{(f)})$
\end{itemize}
\end{prp}

\begin{lmm}{}{} Let $S$ be a graded ring and let $M$ be a graded module over $S$. Then $\widetilde{M}$ is a $\mO_{\text{Proj}(S)}$-module. 
\end{lmm}

For a graded ring $S$, recall that we can shift the grading of $S$ up and down, and it will still be a graded ring. This is the shifted $S(n)$ where $n$ denotes shifting up $n$ times. (Note that this is not true for algebras because $S$ is an algebra over $S_0$, if the grading is shifted then $S(n)$ is an algebra over $S(n)_n$. 

\begin{defn}{The Twisting Sheaf of Serre}{} Let $S$ be a graded ring. Let $X=\text{Proj}(S)$. For any $n\in\Z$, define the sheaf $$\mO_X(n)=\widetilde{S(n)}$$ We call $\mO_X(1)$ the twisting sheaf of Serre. For any sheaf of $\mO_X$-modules $\mF$, denote $$\mF(n)=\mF\otimes_{\mO_X}\mO_X(n)$$
\end{defn}

The twisting sheaf of Serre is important because it is the prototypical example of an invertible sheaf on $\text{Proj}(S)$. 

\begin{prp}{}{} Let $S$ be a graded ring and let $X=\text{Proj}(S)$. Suppose that $S$ is generated by $S_1$ as an $S_0$-algebra. Then the following are true. 
\begin{itemize}
\item The sheaf $\mO_X(n)$ is invertible. 
\item If $M$ is a graded $S$-module, then $\widetilde{M}(n)\cong\widetilde{M(n)}$
\item There is an isomorphism $\mO_X(n)\otimes\mO_X(m)\cong\mO_X(n+m)$
\end{itemize}
\end{prp}

\begin{defn}{Graded Module Associated to a Sheaf of Modules}{} Let $S$ be a graded ring and let $X=\text{Proj}(S)$. Let $\mF$ be a sheaf of $mO_X$-module. Define the graded $S$-module associated to $\mF$ to be the group $$\Gamma_\ast(\mF)=\bigoplus_{n\in\Z}\Gamma(X,\mF(n))$$ together with the structure of graded $S$-module as follows. If $s\in S_d$, then $s$ determines a global section $s\in\Gamma(X,\mO_X(d))$ naturally. For any $t\in\Gamma(X,\mF(n))$, define $s\cdot t\in\Gamma(X,\mF(n+d))$ by sending $s\otimes t\in\mF(n)\otimes\mO_X(d)$ to $\mF(n+d)$ by the isomorphism $$\mF(n)\otimes\mO_X(d)\cong\mF(n+d)$$
\end{defn}

\begin{lmm}{}{} Let $A$ be a ring and let $S=A[x_0,\dots,x_n]$ for $r\geq 1$. Then there is an isomorphism $$\Gamma_\ast(\mO_{\text{Proj}(S)})\cong S$$
\end{lmm}

Note that this is not true if $S$ is not a polynomial ring. 

\subsection{Sheaves on Distinguished Affine Base}
\begin{defn}{The Distinguished Affine Base}{}\\
Let $X$ be a scheme. Define the distinguished affine base $\bold{Open}_{D}(X)$ of $X$ to be the full subcategory of $\bold{Open}(X)$ consisting of affine open sets of the form $\text{Spec}(R)$ and $\text{Spec}(R_f)$ for all $f\in R$. 
\end{defn}

\begin{lmm}{}{}\\
Let $X$ be a scheme. Then $\bold{Open}_D(X)$ is a filtered subcategory of $\bold{Open}(X)$. 
\end{lmm}

Recall from Sheaf Theory that sheaves are uniquely defined on filtered subcategories of the category of open sets. 

\begin{crl}{}{}\\
Let $X$ be a space. Let $\mC$ be an algebraic category. Then there is an equivalence of categories $$\mC(X)=\mC_{\bold{Open}_D(X)}(X)$$
\end{crl}

Thus it suffices to define the values of a sheaf or a morphism of sheaves only on a distinguished base of $X$. 

\subsection{The Support of a Module}
Let $R$ be a commutative ring. Let $M$ be a finitely generated $R$-module. Let $P_1,\dots,P_k$ be the complete list of distinct minimal prime ideals over $\text{Ann}_R(M)$. Recall that the support of the module $M$ is given by $$\text{Supp}(M)=\bigcup_{i=1}^k\V^S(P_i)$$ Now let $P_1,\dots,P_k$ be the prime ideals appearing in the disassembly of $M$. Then we have the relation $$\text{Ass}(M)\subseteq\{P_1,\dots,P_k\}\subseteq\text{Supp}(M)=\V^S(\text{Ann}_R(M))$$ Moreover, the minimal primes in the four sets coincide. Here we explain the geometric intuition for these notions. \\

By definition, $\text{Supp}(M)$ is the set of points of $\text{Spec}(R)$ for which $\widetilde{M}_P=M_P$ is non-zero. Thinking of $\widetilde{M}$ as a ring of functions on $\text{Spec}(R)$, this is the same as saying it suffices to define the functions for points in the support. The fact that $\text{Supp}(M)=\V^S(\text{Ann}_R(M))$ means that $\text{Supp}(M)$ is a closed subscheme itself. 

\begin{prp}{}{}\\
Let $R$ be a commutative ring. Let $M$ be a finitely generated $R$-module. If $R$ is Noetherian, then we have $$\text{Supp}(M)=\overline{\text{Ass}(M)}$$
\end{prp}

Recall that the associated primes record the zero divisors of $M$ if $R$ is Noetherian: $$\bigcup_{P\in\text{Ass}(M)}P=\{r\in R\;|\;r\text{ is a zero divisor of }M\}$$

\begin{prp}{}{}\\
Let $R$ be a Noetherian commutative ring. Let $M$ be a finitely generated $R$-module. Then the irreducible components of $\text{Supp}(M)$ is given by $$\text{Supp}(M)=\bigcup_{P\in\text{Ass}(M),P\text{ is minimal }}\V^S(P)$$
\end{prp}

What about the non minimal primes? Those are the embedded primes. 

\begin{defn}{Embedded Primes}{}\\
Let $R$ be a Noetherian commutative ring. Let $M$ be a finitely generated $R$-module. Let $P\in\text{Ass}(M)$. We say that $P$ is an embedded prime if $P$ is not minimal in $\text{Ass}(M)$. 
\end{defn}

Intuition: The only elements of a ring that an embedded prime can annihilate are nilpotents. So if a ring is reduced, it has no embedded primes. 

\subsection{The Relative Spec and Proj Functors}
\begin{defn}{The Relative Spec Functor}{}\\
Let $Y$ be a scheme. Let $\mA$ be a quasi-coherent sheaf of $\mO_Y$-algebras. Define the relative spec of $\mF$ to be the unique scheme $X=\text{Spec}_Y(\mA)$ given by the following gluing data. 
\begin{itemize}
\item The affine schemes $\text{Spec}(\mA(V))$ for each affine open subsets $V\subseteq Y$. 
\item For affine open subsets $V_i,V_j\subseteq Y$, an open subset $V_{ij}=\text{Spec}(\mA(V_i\cap V_j))$. 
\item The isomorphisms $V_{ij}\cong V_{ji}$ given by the identity. 
\end{itemize}
together with the morphism $f:X\to Y$ given by the gluing data $\text{Spec}(\mA(V))\to Y$ induced by $\mO_Y(V)\to\mA(V)$. 
\end{defn}

\begin{defn}{The Relative Proj Functor}{}\\
Let $Y$ be a scheme. Let $\mS=\bigoplus_{n\geq 0}\mS_n$ be a quasi-coherent sheaf of graded $\mO_Y$-algebras. Define the relative proj of $\mS$ to be the unique scheme $X=\text{Proj}_Y(\mS)$ given by the following gluing data.
\begin{itemize}
\item The schemes $\text{Proj}(\mS(V))$ for each affine open subset $V\subseteq Y$.
\item For affine open subsets $V_i,V_j\subseteq Y$, an open subset $V_{ij}=\text{Proj}(\mS(V_i\cap V_j))$.
\item The isomorphisms $V_{ij}\cong V_{ji}$ given by the identity.
\end{itemize}
together with the morphism $f:X\to Y$ given by the gluing data $\text{Proj}(\mS(V))\to Y$ induced by $\mO_Y(V)\to\mS_0(V)$.
\end{defn}

\section{Vector Bundles over a Scheme}
\subsection{Vector Bundles and Sections}
\begin{defn}{Vector Bundles}{}\\
Let $E,X$ be schemes. Let $\pi:E\to X$ be a map. We say that $\pi$ is a vector bundle of rank $n$ if the following are true. 
\begin{itemize}
\item $\pi$ is surjective. 
\item There exists a collection of isomorphisms $$\{(U_i\subseteq B,\phi_i:\pi^{-1}(U_i)\overset{\cong}{\longrightarrow} \A_{U_i}^n)\}$$ called local trivialization charts such that $B=\bigcup_{i\in I}U_i$ and the following diagram commutes: \\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	{p^{-1}(U_i)} && {\A_{U_i}^n} \\
	& U_i
	\arrow["{\phi_i}", from=1-1, to=1-3]
	\arrow["p"', from=1-1, to=2-2]
	\arrow["\pi", from=1-3, to=2-2]
\end{tikzcd}}\\
\item For each pair $i,j\in I$ and each open affine subscheme $V=\text{Spec}(A)\subseteq U_i\cap U_j$, the $A$-algebra automorphism $\theta_V\in\text{Aut}_A(A[x_1,\dots,x_n])$ corresponding to $$\phi_i|_{\pi^{-1}(V)}\circ\left(\phi_j|_{\pi^{-1}(V)}\right)^{-1}:\A_V^n\to\A_V^n$$ is of the form $$\theta_V(x_i)=\sum_{j=1}^na_{ij}x_j$$
\end{itemize}
\end{defn}

Note: The topological definition does not directly translate over because every point of a topological space $X$ has the same value field and a continuous map is completely determined by its value at each point. For schemes, each point does not have a fixed residue field, even if the scheme lies over a field. Moreover, points do not determine morphism of schemes. The fix is to instead of checking pointwise linear isomorphism, we check it on all possible affine open sets of the intersections. \\

If $E$ and $X$ are $k$-schemes, then we have $\A_{U_i}^n=U_i\times_k\A_k^n$ so the definition looks more familiar to the topological case. 

\begin{prp}{}{}\\
Let $E,X$ be schemes. Let $\pi:E\to X$ be a map. Then $\pi$ is a vector bundle if and only if the following are true. 
\begin{itemize}
\item $\pi$ is surjective. 
\item There exists a collection of isomorphisms $$\{(U_i\subseteq B,\phi_i:\pi^{-1}(U_i)\overset{\cong}{\longrightarrow} \A_{U_i}^n)\}$$ called local trivialization charts such that $B=\bigcup_{i\in I}U_i$ and the following diagram commutes: \\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	{p^{-1}(U_i)} && {\A_{U_i}^n} \\
	& U_i
	\arrow["{\phi_i}", from=1-1, to=1-3]
	\arrow["p"', from=1-1, to=2-2]
	\arrow["\pi", from=1-3, to=2-2]
\end{tikzcd}}\\
\item For each pair $i,j\in I$, there exists $g_{ij}\in\text{GL}(n,\mO_X(U_i\cap U_j))$ called transition functions such that the following diagram commutes: \\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	{\A_{U_i\cap U_j}^n} && {\A_{U_i\cap U_j}^n} \\
	& {\pi^{-1}(U_i\cap U_j)}
	\arrow["{\text{id}\times g_{ij}}", from=1-1, to=1-3]
	\arrow["{\phi_i}", from=2-2, to=1-1]
	\arrow["{\phi_j}"', from=2-2, to=1-3]
\end{tikzcd}}\\
\end{itemize}
\end{prp}

\begin{lmm}{}{}\\
Let $E,X$ be schemes. Let $\pi:E\to X$ be a vector bundle. Let $\{(U_i,\phi_i)\;|\;i\in I\}$ be the local trivialization charts. Then the following are true. 
\begin{itemize}
\item $g_{ik}=g_{jk}\circ g_{ij}$ in $U_i\cap U_j\cap U_k$. 
\item $g_{ii}=1_{\text{GL}(n,\mO_X(U_i))}$. 
\end{itemize}
\end{lmm}

\begin{defn}{Morphisms of Vector Bundles}{}\\
Let $E,F,X$ be schemes. Let $\pi_E:E\to X$ and $\pi_F:F\to X$ be vector bundles of rank $n$ and $m$ respectively. Let $\{(U_i,\phi_i)\;|\;i\in I\}$ and $\{(U_i,\psi_i)\;|\;i\in I\}$ be local trivialization charts of $E$ and $F$ respectively with the same cover. Let $f:E\to F$ be a morphism over $X$. We say that $f$ is a morphism of vector bundles if one of the following conditions are true. 
\begin{itemize}
\item For any open affine subset $V=\text{Spec}(A)\subseteq U_i$, the map of $A$-algebras $\tau_V:A[y_1,\dots,y_m]\to A[x_1,\dots,x_n]$ corresponding to $$\psi_i|_{\pi_F^{-1}(V)}\circ f\circ \phi_i|_{\pi_E^{-1}(V)}:\A_V^n\to\A_V^m$$ is of the form $$\tau_V(y_i)=\sum_{j=1}^nh_{ij}x_j$$
\item We have $$\psi_i\circ f\circ\phi_i^{-1}=\text{id}_{U_i}\times h_i:U_i\times\A^n\to U_i\times\A^m$$ for some $h_i\in M_{m\times n}(\mO_X(U_i))$. Moreover, we have $$g_{ij}^F\cdot h=h\cdot g_{ij}^E$$ for all $i,j\in I$. 
\end{itemize}
\end{defn}

\begin{defn}{Isomorphism of Vector Bundles}{}\\
Let $E,F,X$ be schemes. Let $\pi_E:E\to X$ and $\pi_F:F\to X$ be vector bundles. Let $f:E\to F$ be a morphism. We say that $f$ is an isomorphism if there exists a morphism $g:F\to E$ such that $$g\circ f=\text{id}_E\;\;\;\text{ and }\;\;\;f\circ g=\text{id}_F$$
\end{defn}

\begin{prp}{}{}\\
Let $E,F,X$ be schemes. Let $\pi_E:E\to X$ and $\pi_F:F\to X$ be vector bundles of rank $n$. Let $f:E\to F$ be a morphism with local matrices $\{h_i\;|\;i\in I\}$. Then $f$ is an isomorphism if and only if $h_i\in\text{GL}(n,\mO_X(U_i))$. 
\end{prp}

\begin{defn}{The Category of Vector Bundles}{}\\
Let $X$ be a scheme. Let $n\in\N$. Define the category of vector bundles $$\bold{Vect}_n(X)$$ to consist of the following data. 
\begin{itemize}
\item The objects are the rank $n$ vector bundles over $X$. 
\item The morphisms are the morphisms of vector bundles. 
\item Composition is given by the composition of morphisms of schemes. 
\end{itemize}
\end{defn}

There is a one to one correspondence between vector bundles up to isomorphism and transition functions up to equivalence. 

\subsection{The Sheaf of Local Sections}
\begin{defn}{Local Sections}{}\\
Let $E,X$ be schemes. Let $\pi:E\to X$ be a vector bundle. Let $U\subseteq X$ be open. A local section of $E$ at $U$ is a map $s:U\to X$ such that $\pi\circ s=\text{id}_U$. 
\end{defn}

\begin{defn}{Sheaf of Local Sections}{}\\
Let $E,X$ be schemes. Let $\pi:E\to X$ be a vector bundle. Define the sheaf of local sections $$\mF(E):\bold{Open}(X)\to\bold{Set}$$ as follows. 
\begin{itemize}
\item For each $U\subseteq X$ open, define $$\mF(E)(U)=\{s:U\to E\;|\;s\text{ is a local section }\}$$
\item For each inclusion of open sets $U\hookrightarrow V$, define $\mF(E)(V)\to\mF(E)(U)$ by restriction of local sections $s\mapsto s|_U$. 
\end{itemize}
\end{defn}

\subsection{The Equivalence with Locally Free Sheaves}
\begin{prp}{}{}\\
Let $X$ be a scheme. Then there is an equivalence of categories $$\bold{Vect}_n(X)\simeq\bold{Loc}_n(X)$$ given as follows. 
\begin{itemize}
\item For each vector bundle $\pi:E\to X$, the associated locally free sheaf is the sheaf of local sections $\mF(E)$. 
\item For each locally free sheaf $\mA$ of $X$, the associated vector bundle is given by $\text{Spec}_X(\text{Sym}(\mA^\vee))\to X$. 
\end{itemize}
\end{prp}

\subsection{Open Subschemes}
\begin{defn}{Open Subschemes}{}\\
Let $(X,\mO_X)$ be a scheme. An open subscheme of $X$ is a scheme $(U,\mO_X|_U)$ where $U\subseteq X$ is an open subset. 
\end{defn}

\subsection{Closed Subschemes}
Given two closed embeddings $f:X\to Y$ and $g:Z\to Y$, even if $f(X)=g(Z)$, the subscheme structure can be different. 

\begin{eg}{}{}\\
Let $f:\text{Spec}(k[x]/(x))\to\text{Spec}(k[x])$ be induced by the quotient map $k[x]\to k[x]/(x)$. Let $g:\text{Spec}(k[x]/(x^2))\to\text{Spec}(k[x])$ be the morphism induced by the quotient map $k[x]\to k[x]/(x^2)$. Then the following are true. 
\begin{itemize}
\item $\im(f)=\im(g)$ as topological spaces. 
\item $f_\ast\mO_{\text{Spec}(k[x]/(x))}$ and $g_\ast\mO_{\text{Spec}(k[x]/(x^2))}$ are not isomorphic as sheaves. 
\end{itemize}
\begin{proof}\\
Clearly, $f$ is the map defined by $f((0))=(x)$ and $g$ is the map defined by $g((x))=(x)$. Hence their images are equal. Moreover, the subsheaf structures are different since their global sections are different. 
\end{proof}
\end{eg}

Let $X$ be a scheme. Let $Y\subseteq X$ be a subset. If $Y$ is open, then $Y$ naturally inherits the structure of a scheme. If $Y$ is closed, then it is not obvious how the $\mO_X$ is inherited by $Y$. Indeed, by the above example, a closed subset of a scheme can have multiple subscheme structures. Thus the definition of a closed subscheme must have its structure sheaf built in (compare it to the definition of open subschemes). 

\begin{defn}{Closed Subschemes}{}\\
Let $X$ be a scheme. A closed subscheme of $X$ is a scheme $Y$ together with a closed embedding $f:Y\to X$ such that $Y\subseteq X$. 
\end{defn}

\begin{defn}{The Ideal Sheaf}{}\\
Let $X$ be a scheme. Let $f:Y\to X$ be a subscheme of $X$. Define the ideal sheaf of $Y$ to be $$\mI_Y=\ker(\mO_X\to f_\ast\mO_Y)$$
\end{defn}

\begin{prp}{}{}\\
Let $X$ be a scheme. Then the following are true. 
\begin{itemize}
\item Let $f:Y\to X$ be a closed subscheme. Then $\mI_Y$ is a quasi-coherent sheaf of ideals. 
\item Let $\mI$ be a quasi-coherent sheaf of ideals of $X$. Then there exists a closed subscheme $Y\subseteq X$ such that $\mI\cong\mI_Y$. 
\end{itemize}
\end{prp}

\begin{prp}{}{}\\
Let $R$ be a commutative ring. Then there is a one to one correspondence $$\{I\subseteq R\;|\;I\text{ is an ideal of }R\}\;\;\overset{1:1}{\longleftrightarrow}\;\;\{Y\subseteq\text{Spec}(R)\;|\;Y\text{ is a closed subscheme of Spec}(R)\}$$ given by $I\mapsto\text{Spec}(R/I)$. 
\end{prp}

\begin{prp}{}{}\\
Let $S$ be a commutative graded ring. Then there is a one to one correspondence $$\{I\subseteq S\;|\;I\text{ is a homogeneous ideal of }S\}\;\;\overset{1:1}{\longleftrightarrow}\;\;\{Y\subseteq\text{Spec}(S)\;|\;Y\text{ is a closed subscheme of Proj}(S)\}$$ given by $I\mapsto\text{Proj}(S/I)$. 
\end{prp}

\section{Geometric Properties of Schemes}
\subsection{Dimension and Codimension}
\begin{defn}{The Dimension of a Scheme}{}\\
Let $X$ be a scheme. Define the dimension of $X$ to be dimension of the underlying space of $X$. Explicitly, we have $$\dim(X)=\sup_{\substack{Z_0,\dots,Z_n\subseteq X\\\text{closed irreducible subsets}}}\{n\in\N\;|\;Z_0\subset Z_1\subset\dots\subset Z_n\}$$
\end{defn}

\begin{eg}{}{}\\
Let $R$ be a ring. Then we have $\dim(\text{Spec}(R))=\dim(R)$. 
\end{eg}

\begin{prp}{}{}\\
Let $X$ be a scheme. Then the following numbers are equal. 
\begin{itemize}
\item $\dim(X)$. 
\item $\max\{\dim(V_i)\;|\;1\leq i\leq n\}$ where $V_1,\dots,V_n$ are the irreducible components of $X$. 
\item $\sup\{\dim(\mO_{X,p})\;|\;p\in X\}$. 
\end{itemize}
\end{prp}

\begin{prp}{}{}\\
Let $X,Y$ be schemes. Let $i:Y\to X$ be a closed immersion. If $X$ is integral and $\dim(X)=\dim(Y)$, then $i$ is an isomorphism. 
\end{prp}

\begin{prp}{}{}\\
Let $k$ be a field. Let $X$ be an irreducible scheme of finite type over $k$. Then all maximal chains of closed irreducible subsets have length $\dim(X)$. 
\end{prp}

\begin{prp}{}{}\\
Let $k$ be a field. Let $X,Y$ be schemes locally of finite type over $k$. Then we have $$\dim(X\times_kY)=\dim(X)+\dim(Y)$$
\end{prp}

\begin{defn}{The Codimension of a Scheme}{}\\
Let $X$ be a scheme. Let $Y\subseteq X$ be a closed subset of $X$. Define the codimension of $Y$ in $X$ to be $$\text{codim}_X(Y)=\sup_{\substack{Z\subseteq Y\text{ is a}\\\text{closed irreducible subsets}}}\{n\in\N\;|\;Z=Z_0\subset\cdots\subset Z_n\subset X\text{ are closed irreducible subsets }\}$$
\end{defn}

\begin{prp}{}{}\\
Let $X$ be a scheme. Let $Y\subseteq X$ be an irreducible closed subset. Let $\eta$ be the generic point of $Y$. Then $$\text{codim}_X(Y)=\dim(\mO_{X,\eta})$$
\end{prp}

\begin{prp}{}{}\\
Let $k$ be a field. Let $X$ be an irreducible scheme of finite type over $k$. Let $Y$ be an irreducible closed subset of $X$. Then we have $$\dim(Y)+\text{codim}_X(Y)=\dim(X)$$
\end{prp}

\subsection{Normality}
\begin{defn}{Normal Schemes}{}\\
Let $X$ be a scheme. We say that $X$ is normal if for all $p\in X$, $\mO_{X,p}$ is a normal domain. 
\end{defn}

\begin{lmm}{}{}\\
Let $R$ be a commutative ring. If $R$ is normal, then $\text{Spec}(R)$ is a normal scheme. 
\begin{proof}\\
Let $P\in\text{Spec}(R)$. Since $R$ is normal, the localization $R_P$ is normal. Hence $\mO_{X,P}\cong R_P$ is normal. Thus $\text{Spec}(R)$ is a normal scheme. 
\end{proof}
\end{lmm}

\begin{eg}{}{}\\
Let $k$ be a field. Let $n\in\N\setminus\{0\}$. Then $\A_k^n,\Prj_k^n$ are normal schemes. 
\begin{proof}\\
Since $k$ is a normal domain, $k[x_1,\dots,x_n]$ is also a normal domain by Commutative Algebra 1. Hence $\A_k^n$ is normal. As for $\Prj_k^n$, notice that the stalk of any point of $\Prj_k^n$ can be computed as the stalk of any affine chart $U_i\cong\A_k^n$ containing the point. Since $\A_k^n$ is normal, all stalks are normal domains. Hence all stalks of $\Prj_k^n$ are normal domains, and $\Prj_k^n$ is a normal scheme. 
\end{proof}
\end{eg}

\begin{eg}{}{}\\
Let $R$ be a UFD such that $2\in R$ is invertible. Let $d\in R$ be such that $x^2-d\in R[x]$ is irreducible. Then $\text{Spec}\left(\frac{R[x]}{(x^2-r)}\right)$ is normal if and only if $d\in R$ is squarefree. 
\begin{proof}\\
We have seen from Commutative Algebra 1 7.5.8 that $d\in R$ is squarefree implies that $\frac{R[x]}{(x^2-d)}$ is normal. Hence $\text{Spec}\left(\frac{R[x]}{(x^2-r)}\right)$ is normal. \\

Conversely, suppose that $d\in R$ is not squarefree. If $d$ is a square, then $\frac{R[x]}{(x^2-d)}$ is not an integral domain and so cannot be normal. Otherwise, without loss of generality we only need to consider the ring $\frac{R[x]}{(x^2-dy^2)}$ where $d$ is squarefree and $y\in R$. In this case, $x^2-dy^2$ is still irreducible, and since $R[x]$ is a UFD, $(x^2-dy^2)$ is a prime ideal. But I claim that it is not normal. Consider $\frac{x}{y}\in\text{Frac}\left(\frac{R[x]}{(x^2-dy^2)}\right)=\frac{\text{Frac}(R)[x]}{(x^2-dy^2)}$. It is integral over $\frac{R[x]}{(x^2-dy^2)}$ since it is the root of the monic polynomial $p(t)=t^2-d\in\frac{R[x]}{(x^2-dy^2)}[t]$. Since normality is a local property, there exists a prime ideal $P$ of $\frac{R[x]}{(x^2-dy^2)}$ such that $\mO_{\text{Spec}(R[x]/(x^2-dy^2)),P}\cong R_P$ is not normal. Hence $\text{Spec}\left(\frac{R[x]}{(x^2-r)}\right)$ is not normal. 
\end{proof}
\end{eg}

\begin{eg}{}{}\\
Let $k$ be an algebraically closed field such that $\text{char}(k)\neq 2$. Then the following rings are normal. 
\begin{itemize}
\item $\text{Spec}\left(\frac{\Z[x]}{(x^2-n)}\right)$ where $n$ is squarefree and $n\equiv2,3\;(\bmod\;4)$. 
\item $\text{Spec}\left(\Z\left[\frac{1+\sqrt{n}}{2}\right]\right)$ where $n$ is squarefree and $n\equiv1\;(\bmod\;4)$. 
\item $\text{Spec}\left(\frac{k[x_1,\dots,x_n]}{(x_1^2+\dots+x_m^2)}\right)$ where $n\geq m\geq 3$. 
\item $\text{Spec}\left(\frac{k[w,x,y,z]}{(wz-xy)}\right)$. 
\end{itemize}
\begin{proof}\\
From Commutative Algebra 1 7.5.9, we have seen that the first two rings are normal. From 7.5.10 we have seen that the last two rings are normal. 
\end{proof}
\end{eg}

\begin{prp}{}{}\\
Let $X$ be a normal scheme. Let $U\subseteq X$ be open such that $\text{codim}_X(U)\geq 2$. Let $\mF$ be a reflexive sheaf of $\mO_X$-modules. Then we there is an isomorphism $$\mF\cong i_\ast(\mF|_U)$$ where $i:U\hookrightarrow X$ is the inclusion map. 
\end{prp}

\begin{defn}{The Normalization of a Scheme}{}\\
Let $X$ be an integral scheme. The normalization of $X$ consists of a scheme $\widetilde{X}$ and a morphism $\pi:\widetilde{X}\to X$ that satisfies the following universal property. For every normal integral scheme $Z$ and every dominant morphism $\phi:Z\to X$, there exists a unique morphism $\widetilde{\phi}:Z\to\widetilde{X}$ such that the following diagram commutes: \\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	Z & {\widetilde{X}} \\
	& X
	\arrow["{\exists!\widetilde{\phi}}", dashed, from=1-1, to=1-2]
	\arrow["\phi"', from=1-1, to=2-2]
	\arrow["\pi", from=1-2, to=2-2]
\end{tikzcd}} \\
\end{defn}

\begin{prp}{}{}\\
Let $X$ be an integral scheme. Then the normalization of $X$ exists and is unique (up to unique isomorphism), given by the following gluing data: 
\begin{itemize}
\item For each affine open subset $U\subseteq X$, define $\overline{U}=\text{Spec}(\overline{\mO_X(U)})$ (normalization of $\mO_X(U)$ in its fraction field). 
\item For two affine open subsets $U,V\subseteq X$, define $\theta_{U,V}:\overline{U\cap V}\to\overline{U\cap V}$ by the identity map. 
\end{itemize}
\end{prp}

\subsection{Regularity}
\begin{defn}{Regular Schemes}{}\\
Let $X$ be a scheme. Let $p\in X$. We say that $X$ is regular at $p$ if $\mO_{X,p}$ is a regular local ring. We say that $X$ is regular if $X$ is regular at all points $p\in X$. 
\end{defn}

\begin{lmm}{}{}\\
Let $X$ be a scheme. If $X$ is regular, then $X$ is normal. 
\end{lmm}

\begin{defn}{The Jacobian Matrix at a Point}{}\\
Let $k$ be a field. Let $X$ be a scheme of finite type over $k$. Let $p\in X$. Let $\text{Spec}\left(\frac{k[x_1,\dots,x_n]}{(f_1,\dots,f_r)}\right)$ be an affine neighbourhood containing $p$. Define the Jacobian matrix of $X$ at $p$ to be the matrix $$J_pX=\begin{pmatrix}
\frac{\partial f_1}{\partial x_1}\;\bmod p & \cdots & \frac{\partial f_1}{\partial x_n}\;\bmod p\\
\vdots & \ddots & \vdots\\
\frac{\partial f_r}{\partial x_1}\;\bmod p & \cdots & \frac{\partial f_r}{\partial x_n}\;\bmod p
\end{pmatrix}$$
\end{defn}
v
\begin{prp}{}{}\\
Let $k$ be a field. Let $X$ be a scheme of finite type over $k$. Let $p\in X(k)$ be a $k$-valued point. Then the following are equivalent. 
\begin{itemize}
\item $p$ is a regular point of $X$. 
\item $\dim(T_pX)=\dim(X)$. 
\item If $\text{Spec}\left(\frac{k[x_1,\dots,x_n]}{(f_1,\dots,f_r)}\right)$ is an affine open cover containing $p$, then $\rank(J_pX)=n-\dim(X)$
\end{itemize}
\end{prp}

\subsection{Smoothness}
The Jacobian criterion for checking regularity is only applicable for $k$-valued points. Motivated by this, we define the notion of a smooth scheme to be the schemes that satisfy the Jacobian criterion at all points. 

\begin{defn}{Smooth Schemes}{}\\
Let $k$ be a field. Let $X$ be a scheme locally of finite type over $k$. We say that $X$ is smooth if for all $p\in X$, the affine open subscheme $\text{Spec}\left(\frac{k[x_1,\dots,x_n]}{(f_1,\dots,f_r)}\right)$ containing $p$ is such that $\rank(J_pX)=n-\dim(X)$. 
\end{defn}

The main difference between smooth schemes and regular schemes is that the Jacobian rank criterion is satisfied for all (closed) points for smooth schemes, while the same criterion is only necessary on $k$-valued points for regular schemes. 

\begin{lmm}{}{}\\
Let $k$ be a field. Let $X$ be a scheme of finite type over $k$. Then the following are true. 
\begin{itemize}
\item If $X$ is smooth, then $X$ is regular. 
\item If $k$ is algebraically closed, then $X$ is smooth if and only if $X$ is regular. 
\end{itemize}
\end{lmm}

\begin{eg}{}{}\\
The affine scheme $\text{Spec}\left(\frac{\C[x,y]}{(x^3-y^2)}\right)$ is smooth at all points except $(0,0)$. 
\begin{proof}\\
For any point $p$, we have $$J_pX=(3x^2\;\;\;-2y)$$ For a point of the form $p=(x-a,y-b)$, we have $J_pX$ is rank $1$ if and only if $a\neq 0$ and $b\neq 0$, which only happens when $p=(x,y)$. We also check that for $p=(0)$ the generic point, we have $J_p(X)=(3x^2\;\;\;-2y)$ in the residue field $K((0))=\C(x,y)/(x^3-y^2)$. It is non-zero since $x$ and $y$ are non-zero elements in the residue field. Hence we are done. 
\end{proof}
\end{eg}

\subsection{The Smooth Locus}
In practice, smooth schemes and smooth varieties are the most well behaved. When dealing with schemes that are singular, we may consider the set of all smooth points instead. 

\begin{defn}{Smooth Locus}{}\\
Let $k$ be a field. Let $X$ be a scheme of finite type over $k$. Define the smooth locus of $X$ to be $$X_\text{sm}=\{x\in X\;|\;x\text{ is a smooth point }\}$$
\end{defn}

\begin{prp}{}{}\\
Let $k$ be a field. Let $X$ be a scheme of finite type over $k$. Then $X_\text{sm}$ is open in $X$. 
\end{prp}

\begin{prp}{}{}\\
Let $k$ be a field. Let $X$ be a scheme of finite type over $k$. If $X$ is a normal, then we have $$\text{codim}_X(X\setminus X_\text{sm})\geq 2$$
\end{prp}

Hence reflexive sheaves and line bundles of a normal scheme is the same content as those on the smooth locus. This makes normal schemes the next most well behaved schemes after smooth schemes. 

\begin{thm}{Generic Smoothness}{}\\
Let $k$ be a perfect field. Let $X$ be an integral scheme of finite type over $k$. Then $X_\text{sm}$ is dense in $X$ and $\dim(X_\text{sm})=\dim(X)$. 
\end{thm}

\begin{defn}{Cohen-Macaulay Schemes}{}\\
Let $(X,\mO_X)$ be a scheme. We say that $X$ is Cohen-Macaulay if $\mO_{X,p}$ is Cohen-Macaulay for all $p\in X$. 
\end{defn}

\begin{prp}{}{}\\
Let $X$ be a locally Noetherian scheme. 
\begin{itemize}
\item If $\dim(X)=0$, then $X$ is Cohen-Macaulay. 
\item Suppose that $\dim(X)=1$. Then $X$ is Cohen-Macaulay if and only if $X$ has no embedded primes. 
\end{itemize}
\end{prp}

\begin{defn}{Gorenstein Schemes}{}\\
Let $(X,\mO_X)$ be a scheme. We say that $X$ is Gorenstein if $\mO_{X,p}$ is Gorenstein for all $p\in X$. 
\end{defn}

\subsection{Serre's $R_n$ and $S_n$ Criterion}
\begin{defn}{Serre's $R_n$ Criterion}{}\\
Let $X$ be a scheme. Let $n\in\N$. We say that $X$ satisfies $R_n$ if $\mO_{X,p}$ satisfies $R_n$ for all $p\in X$. 
\end{defn}

\begin{defn}{Serre's $S_n$ Criterion}{}\\
Let $X$ be a scheme. Let $n\in\N$. We say that $X$ satisfies $S_n$ if $\mO_{X,p}$ satisfies $S_n$ for all $p\in X$. 
\end{defn}

\begin{prp}{}{}\\
Let $X$ be a locally Noetherian scheme. Then the following are true. 
\begin{itemize}
\item $X$ is reduced if and only if $X$ satisfies $R_0$ and $S_1$. 
\item $X$ is normal if and only if $X$ satisfies $R_1$ and $S_2$. 
\item $X$ is Cohen-Macaulay if and only if $X$ satisfies $S_n$ for all $n\geq 0$. 
\end{itemize}
\end{prp}

\section{The Theory of Divisors}
\subsection{Weil Divisors}
\begin{defn}{Weil Divisors}{}\\
Let $X$ be a locally Noetherian integral scheme. Define the group of Weil divisors to be the free abelian group $$\text{Div}(X)=\Z\langle Y\subseteq X\;|\;Y\text{ is a closed integral subscheme of codimension one }\rangle$$ generated by closed integral subschemes of $X$ of codimension one. 
\end{defn}

\begin{eg}{}{}\\
Let $p=[1:0]\in]\Prj^1$. Let $n>0$. Then we have $\mO_X(np)(U_x)\cong (x/y)^{-n}\cdot k[x/y]$ and $\mO_X(np)(U_y)\cong k[(y/x)]$. 
\end{eg}

\begin{defn}{Divisors of Functions}{}\\
Let $X$ be a locally Noetherian integral scheme. For each closed integral subscheme $Y$ of codimension one, let $\eta$ be the generic point of $Y$. Let $f\in K(X)^\ast$. Define the divisor of $f$ to be $$\text{div}(f)=\sum_Yl_{\mO_{X,\eta}}\left(\frac{\mO_{X,\eta}}{(f)}\right)\cdot Y$$
\end{defn}

If $\mO_{X,\eta}$ is a DVR, then $$l_{\mO_{X,\eta}}\left(\frac{\mO_{X,\eta}}{(f)}\right)=v(f)$$ where $v$ is the valuation of the DVR $\mO_{X,\eta}$. 

\begin{eg}{}{}\\
Let $R=\frac{k[x,y,z]}{(xy-z^2)}$. Then $\text{div}(y)=2\cdot\V(y,z)$. 
\begin{proof}\\
Let $X=\text{Spec}(R)$. Firstly, notice that $y\in\mO_X(X)$ is a globally regular function, hence $v_Z(y)\geq 0$ for all closed integral codimension one subschemes $Z$. I claim that if $v_Z(y)>0$, then $Z\subseteq\V(y)$. \\

Let $\eta$ be the generic point of $Z$. Let $p_\eta$ be the corresponding prime ideal. If $v_Z(y)>0$, then we have $y\in p_\eta R_{p_\eta}$ which is the maximal ideal of the localization $R_{p_\eta}$. Since $R$ is an integral domain, we must have $y\in p_\eta$ and so $p_\eta\in\V(y)$. Then $Z=\overline{p_\eta}^Z\subseteq\overline{p_\eta}^X=\V(y)$. \\

Thus any non-zero contribution of $\text{div}(y)$ must come from $Z\subseteq\V(y)$ where $Z$ has codimension $1$ in $X$. The only possibility is $Z=\V(y)=\V(y,z)$. Notice that $x$ is invertible in $R_{(y,z)}$. Hence we have $\text{div}(y)=\text{div}(z^2/x)=\text{div}(z^2)=2$ since $(z)$ is the unique maximal ideal of $R_{(y,z)}$. Hence $\text{div}(y)=2\cdot\V(y,z)$. 
\end{proof}
\end{eg}

\begin{lmm}{}{}\\
Let $X$ be a locally Noetherian integral scheme. Then the map $\text{div}:K(X)^\ast\to\text{Div}(X)$ given by $f\mapsto\text{div}(f)$ is a well defined group homomorphism. 
\end{lmm}

\begin{defn}{Principal Divisors}{}\\
Let $X$ be a locally Noetherian integral scheme. We say that a divisor $D$ on $X$ is principal if $D=\text{div}(f)$ for some function $f$. 
\end{defn}

\begin{defn}{The Divisor Class Group}{}\\
Let $X$ be a locally Noetherian integral scheme. Let $\text{Prin}(X)$ be the subgroup of all principal divisors of $X$. Define the divisor class group of $X$ as $$\text{Cl}(X)=\frac{\text{Div}(X)}{\text{Prin}(X)}$$ Two elements of the same coset are said to be linearly equivalent. 
\end{defn}

\begin{defn}{Degree Homomorphism}{}\\
Let $X$ be a locally Noetherian integral scheme. Define the degree homomorphism $$\deg:\text{Div}(X)\to\Z$$ by $D=\sum_Pn_P\cdot P\mapsto\sum_Pn_P$. For each divisor $D$, define the degree of $D$ to be $$\deg(D)=\sum_Pn_P$$
\end{defn}

\begin{defn}{Restrictions of Divisors}{}\\
Let $X$ be a locally Noetherian integral scheme. Let $D$ be a divisor on $X$. Let $U\subseteq X$ be an open subset. Define the restriction of $D$ to $U$ to be $$D|_U=\sum_{Y\subseteq U}v_Y(f)\cdot Y$$
\end{defn}

\begin{defn}{Locally Prinicpal Divisors}{}\\
Let $X$ be a locally Noetherian integral scheme. Let $D$ be a divisor on $X$. We say that $D$ is locally principal if there exists an open cover $\{U_i\;|\;i\in I\}$ of $X$ such that $D|_{U_i}$ is principal on $U_i$ for all $i\in I$. 
\end{defn}

\subsection{From Weil Divisors to Sheaves}
\begin{defn}{The Associated Sheaf of a Weil Divisor}{}\\
Let $X$ be a locally Noetherian integral scheme. Let $D\in\text{Div}(X)$ be a divisor of $X$. Define the associated sheaf $\mO_X(D)$ of $D$ to be the subsheaf of $Q_X$ given by $$\Gamma(U,\mO_X(D))=\{f\in K(X)\;|\;f=0\text{ or }\text{div}|_U(f)+D|_U\geq 0\}\cup\{0\}$$ for $U\subseteq X$ an open subset. 
\end{defn}

\begin{lmm}{}{}\\
Let $X$ be a locally Noetherian integral scheme. Let $D$ be a divisor on $X$. Then the following are true. 
\begin{itemize}
\item $\mO_X(D)$ is a quasi-coherent sheaf. 
\item $D$ is principal if and only if $\mO_X(D)\cong\mO_X$. 
\item $D$ is locally principal if and only if $\mO_X(D)$ is invertible (a line bundle). 
\end{itemize}
\end{lmm}

\begin{lmm}{}{}\\
Let $X$ be a locally Noetherian integral scheme. Let $D,D'$ be divisors on $X$. Then $D$ and $D'$ are linearly equivalent if and only if $\mO_X(D)\cong\mO_X(D')$. 
\end{lmm}

\subsection{Cartier Divisors}
\begin{defn}{The Sheaf of Total Quotient Rings}{} Let $(X,\mO_X)$ be a scheme. Define a presheaf $Q_X:\bold{Open}(X)\to\bold{Ring}$ as follows. 
\begin{itemize}
\item Each $U\subseteq X$ open subset is associated to the total quotient ring $Q_X(U)=Q(\mO_X(U))$ . 
\item For each inclusion $U\subseteq V$, define $Q_X(V)\to Q_X(U)$ by just the restriction map. 
\end{itemize}
Define the sheaf of total quotient rings to be the associated sheaf of the presheaf $K$. 
\end{defn}

\begin{defn}{Invertible Elements of a Sheaf}{} Let $(X,\mF)$ be a ringed space. Define the sheaf of invertible elements $$\mF^\ast:\bold{Open}\to\bold{Grp}$$ of $\mF$ as follows. 
\begin{itemize}
\item For $U\subseteq X$ an open set, define $\mF^\ast(U)=\text{The Group of Units of }\mF(U)$
\item For $U\subseteq V$, define $\mF^\ast(V)\to\mF^\ast(U)$ to just be the restriction map. 
\end{itemize}
\end{defn}

\begin{lmm}{}{}\\
Let $(X,\mO_X)$ be a scheme. Then there is an exact sequence of $\mO_X$-modules given by \\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	0 & {\mO_X^\ast} & {Q_X^\ast} & {\frac{Q_X^\ast}{\mO_X^\ast}} & 0
	\arrow[from=1-1, to=1-2]
	\arrow[from=1-2, to=1-3]
	\arrow[from=1-3, to=1-4]
	\arrow[from=1-4, to=1-5]
\end{tikzcd}} \\
\end{lmm}

\begin{defn}{Cartier Divisors}{}\\
Let $X$ be a scheme. Define the group of Cartier divisors to be the abelian group $$\text{CDiv}(X)=\Gamma(X,Q_X^\ast/\mO_X^\ast)$$
\end{defn}

Explicitly, a Cartier divisor consists of an open cover $\{U_u\;|\;i\in I\}$ and rational functions $f_i\in Q_X(U_i)$ such that for each $i,j\in I$, there exists $s\in\mO_X^\ast(U_i\cap U_j)$ such that $f_i=sf_j$. 

\begin{eg}{}{}\\
The set $\{(U_x,1/(x/y)-1),(U_y,1)\}$ is a Cartier divisor on $\Prj^1$. 
\end{eg}

\begin{defn}{Principal Cartier Divisors}{}\\
Let $(X,\mO_X)$ be a scheme. A Cartier divisor $f$ of $X$ is said to be principal if it is the image of the natural map $$\Gamma(X,Q_X^\ast)\to\Gamma(X,Q_X^\ast/\mO_X^\ast)=\text{CDiv}(X)$$
\end{defn}

\begin{lmm}{}{}\\
Let $X$ be a scheme. Let $D$ be a principal divisor on $X$. Then there exists $f\in\mO_X(X)$ such that $D$ is given by $\{(X,f)\}$. 
\end{lmm}

\begin{defn}{Effective Cartier Divisor}{}\\
Let $(X,\mO_X)$ be a scheme. A Cartier divisor $f$ of $X$ is said to be effective if it is the image of the natural map $$\Gamma(X,\mO_X\cap Q_X^\ast)\to\Gamma(X,Q_X^\ast/\mO_X^\ast)=\text{CDiv}(X)$$
\end{defn}

\begin{lmm}{}{}\\
Let $X$ be a scheme. Let $D$ be a Cartier divisor. on $X$ Then $D$ is given by the data $\{(U_i,f_i)\;|\;i\in I\}$ where $X=\bigcup_{i\in I}U_i$ is an open cover, $f_i\in Q_X(U_i)$ such that for $i,j\in I$, there exists $s\in\mO_X^\ast(U_i\cap U_j)$ such that $f_i=sf_j$. 
\end{lmm}

\begin{defn}{Cartier Data}{}\\
Let $X$ be a scheme. Let $D$ be a Cartier divisor on $X$. Define the Cartier data of $D$ to be $\{(U_i,f_i)\;|\;i\in I\}$ as above. 
\end{defn}

\begin{lmm}{}{}\\
Let $X$ be a scheme. Then the following are true regarding the Cartier divisors on $X$. 
\begin{itemize}
\item Let $\{(U_i,f_i)\;|\;i\in I\}$ and $\{(V_j,g_j\;|\;j\in J\}$ be Cartier divisors. Then their sum is given by $$\{(U_i,f_i)\;|\;i\in I\}+\{(V_j,g_j\;|\;j\in J\}=\{(U_i\cap V_j\;|\;f_ig_j)\;|\;i\in I,j\in J\}$$
\item Let $D$ be a Cartier divisor. Then $D$ is principal if and only if $\{(X,f)\}$ is a Cartier data for $D$. 
\item Let $D$ be a Cartier divisor. Then $D$ is effective if and only if $\{(U_i,f_i)\;|\;i\in I\}$ is a Cartier data for $D$ such that $f_i\in\mO(U_i)$. 
\end{itemize}
\end{lmm}

\begin{defn}{The Cartier Divisor Class Group}{}\\
Let $(X,\mO_X)$ be a scheme. Define the Cartier divisor class group to be the abelian group $$\text{CCl}(X)=\frac{\text{CDiv}(X)}{\sim}$$ where $D_1\sim D_2$ if and only if $D_1-D_2$ is a principal divisor. 
\end{defn}

\subsection{From Cartier Divisors to Sheaves}
\begin{defn}{The Associated Sheaf of a Cartier Divisor}{}\\
Let $(X,\mO_X)$ be a scheme. Let $D=\{(U_i,f_i\in K_X^\ast)\;|\;i\in I\}$ be a Cartier divisor on $X$. Define the associated sheaf $\mL(D)$ of $D$ to be the $\mO_X$-module by gluing the schemes $(U_i,\mO_X(U_i)[f_i^{-1}])$ for each $i$. 
\end{defn}

\begin{prp}{}{}\\
Let $X$ be a scheme. Let $D$ be a Cartier divisor of $X$. Then $\mL(D)$ is an invertible sheaf. 
\end{prp}

The proposition motivates us to investigate the relations between Cartier divisors and invertible sheaves on a scheme $X$. This leads to a very fruitful and satisfying result. 

\begin{thm}{}{} Let $X$ be a scheme. For any Cartier divisor $D$ of $X$, the association $D\mapsto\mL(D)$ gives a bijection $$\{\text{Cartier Divisors on }X\}\;\;\;\;\overset{1:1}{\longleftrightarrow}\;\;\;\;\{\text{Invertible Subsheaves of }Q_X\}$$
\end{thm}

\begin{prp}{}{} Let $X$ be a scheme. Let $D_1$ and $D_2$ be Cartier divisors of $X$. Then there is an isomorphism $$\mL(D_1-D_2)\cong\mL(D_1)\otimes\mL(D_2)^{-1}$$
\end{prp}

\begin{prp}{}{}\\
Let $X$ be a scheme. Then the map $$\text{CCl}(X)\to\text{Pic}(X)$$ defined by $[D]\mapsto\mL(D)$ is an injective group homomorphism. 
\end{prp}

When $X$ is integral, Cartier divisors and the Picard group is entirely the same invariant for $X$. 

\begin{prp}{}{}\\
Let $X$ be an integral scheme. Then the there is an isomorphism of groups $$\text{CCl}(X)\cong\text{Pic}(X)$$ defined by $[D]\mapsto\mL(D)$. 
\end{prp}

Recall that invertible sheaves correspond to line bundles. There is a more direct way to view Cartier divisors as line bundles. Indeed, one may check that for the Cartier data $\{(U_i,f_i)\;|\;i\in I\}$, the functions $g_{ij}=\frac{f_i}{f_j}$ satisfies the cocycle conditions. 

\subsection{Comparison of Weil and Cartier Divisors}
\begin{defn}{The Map from Cartier Divisors to Weil Divisors}{}\\
Let $X$ be a locally Noetherian integral scheme. Define the function $d:\text{CDiv}(X)\to\text{Div}(X)$ by $$\{(U_i,f_i)\;|\;i\in I\}\mapsto\text{ The unique divisor }D\text{ such that }D|_{U_i}=\text{div}|_{U_i}(f_i)$$
\end{defn}

\begin{prp}{}{}\\
Let $X$ be a locally Noetherian integral scheme. Then $d:\text{CDiv}(X)\to\text{Div}(X)$ is an injective group homomorphism. 
\end{prp}

\begin{prp}{}{}\\
Let $X$ be a normal Noetherian integral scheme. Let $D$ be a Weil divisor on $X$. Then $D\in\im(d)$ if and only if $D$ is locally principal. In this case, we have $\mO(D)=\mL(D)$. 
\end{prp}

\begin{prp}{}{}\\
Let $X$ be a normal Noetherian integral scheme. Then there is a bijection $$\text{Cl}(X)\;\;\overset{1:1}{\longleftrightarrow}\;\;\{\mF\;|\;\mF\text{ is a reflexive sheaf of rank }1\}/\cong$$
\end{prp}

If $X$ is normal and integral, then we have the following relation. 

\adjustbox{scale=1.0,center}{\begin{tikzcd}
	& {\text{Div}(X)} & {\text{Reflexive Sheaves}} \\
	{\text{CDiv}(X)} & {\text{Locally Principal}} & {\text{Invertible Sheaves}} \\
	& \begin{array}{c} \substack{\text{Invertible subsheaves}\\\text{of }\mK_X^\ast} \end{array}
	\arrow[from=1-2, to=1-3]
	\arrow["{1:1}", from=2-1, to=2-2]
	\arrow["{1:1}", from=2-1, to=3-2]
	\arrow[hook, from=2-2, to=1-2]
	\arrow[hook, from=2-2, to=2-3]
	\arrow[hook, from=2-3, to=1-3]
	\arrow[hook, from=3-2, to=2-3]
\end{tikzcd}} \\~\\

\begin{prp}{}{} Let $X$ be an integral, separated Noetherian scheme such that all local rings are UFD. Then the following are true. 
\begin{itemize}
\item There is an isomorphism $$\text{Div}(X)\cong\text{CDiv}(X)$$ induced by $d$. 
\item There is an isomorphism $$\text{Cl}(X)\cong\text{CCl}(X)$$ induced by $d$. 
\end{itemize}
\end{prp}

\begin{crl}{}{} Let $X$ be a scheme. If $X$ is Noetherian, integral, separated and that all local rings are UFDs, then there are natural isomorphisms $$\text{Cl}(X)\cong\text{CCl}(X)\cong\text{Pic}(X)$$
\end{crl}

\begin{defn}{Degree $0$ Picard Group}{}\\
Let $X$ be an integral, separated Noetherian scheme such that all local rings are UFD. Define the degree $0$ picard group to be $$\text{Pic}^0(X)=\ker(\deg:\text{Pic}(X)\cong\text{Cl}(X)\to\Z)$$
\end{defn}

\section{Classifying Maps to Projective Space}
\subsection{Base Points and Morphisms to Projective Space}
\begin{defn}{Base Points}{}\\
Let $X$ be a scheme. Let $\mL$ be an invertible sheaf. Let $p\in X$. We say that $p$ is a base point for $\mL$ if $s(p)=0$ for all $s\in\Gamma(X,\mL)$. 
\end{defn}

\begin{defn}{Base Point Free}{}\\
Let $X$ be a scheme. Let $\mL$ be an invertible sheaf. We say that $\mL$ is base point free if $\mL$ has no base points. 
\end{defn}

\begin{lmm}{}{}\\
Let $X$ be a scheme. Let $\mL$ be an invertible sheaf. Let $s_0,\dots,s_n\in\Gamma(X,\mL)$. Then the following are equivalent. 
\begin{itemize}
\item $(s_0,\dots,s_n)$ is basepoint free. 
\item $s_0,\dots,s_n$ has no common zeroes. 
\item $s_0,\dots,s_n$ generate $\mL(X)$. 
\end{itemize}
\end{lmm}

\begin{prp}{}{}\\
Let $R$ be a commutative ring. Let $X$ be a scheme over $R$. Then there is a one to one correspondence $$\{\varphi:X\to\Prj_R^n\;|\;\varphi\text{ is an }R\text{-morphism }\}\;\;\overset{1:1}{\longleftrightarrow}\;\;\frac{\left\{(\mL,s_0,\dots,s_n)\;\bigg{|}\;\substack{\mL\text{ is an invertible sheaf on }X\\\text{ and }s_0,\dots,s_n\in\mL(X)\text{ is base point free }}\right\}}{\sim}$$ given by $\varphi\mapsto(\varphi^\ast(\mO_{\Prj_R^n}(1)),\varphi^\ast(s_0),\dots,\varphi^\ast(s_n))$ and $(\mL,s_0,\dots,s_n)\sim(\mL',s_0',\dots,s_n')$ if and only if there exists an isomorphism $\mL\cong\mL'$ such that $s_i$ corresponds to $s_i'$. 
\end{prp}

\begin{prp}{}{}\\
Let $R$ be a commutative ring. Let $X$ be a scheme over $R$. Let $\varphi:X\to\Prj_R^n$ be an $R$-morphism corresponding to the invertible sheaf $\mL$ of $X$ and global sections $s_0,\dots,s_n$. Then $\varphi$ is a closed immersion if and only if the following are true. 
\begin{itemize}
\item For each $i$, if $U_i$ is the largest open set such that $s_i|_{U_i}$ generates $\mO_X|_{U_i}$, then $U_i$ is affine. 
\item For each $i$, the map of rings $R[x_0,\dots,x_n]\to\mO_X(U_i)$ defined by $x_j\mapsto x_j/x_i$ is surjective. 
\end{itemize}
\end{prp}

\subsection{Ample and Very Ample Line Bundles}
We freely interchange between line bundles and invertible sheaves. 

\begin{defn}{Very Ample Sheaves}{}\\
Let $X,Y$ be schemes. Let $f:X\to Y$ be a morphism. Let $\mL$ be an invertible sheaf of $\mO_X$-modules. We say that $\mL$ is very ample relative to $Y$ if there exists a closed embedding $i:X\to\Prj_Y^n$ such that $\mL\cong i^\ast\mO_Y(1)$
\end{defn}

If $Y=\text{Spec}(R)$ is affine, then 12.2.1 shows that $\mL$ is very ample if and only if it has global sections that generate $\mL$ such that the corresponding morphism is a closed embedding. 

\begin{defn}{Ample Sheaves}{}\\
Let $X$ be a Noetherian scheme. Let $\mL$ be an invertible sheaf of $\mO_X$-modules. We say that $\mL$ is ample if for every coherent sheaf $\mF$ of $X$, there exists $N\in\N^+$ such that $\mF\otimes\mL^{\otimes n}$ is very ample for all $n\geq N$. 
\end{defn}

The condition that $\mF\otimes\mL^{\otimes n}$ is globally generated for large $n$ can be thought of as saying that $\mL^{\otimes n}$ eventually trivializes any coherent sheaf, or that $\mL$ is positive enough to overcome any twisting present in $\mF$. This is why ample sheaves are the correct algebraic generalization of positive line bundles in complex geometry.

\section{Differentials}
\subsection{The Sheaf of Differentials}
\begin{lmm}{}{}\\
Let $X$ be a scheme over a scheme $S$. Let $\Delta:X\to X\times_SX$ be the diagonal morphism. Then there exists an open subset $\Delta(X)\subseteq U\subseteq X\times_SX$ such that $\Delta:X\to U$ is a closed immersion. 
\end{lmm}

\begin{defn}{The Sheaf of Relative Differentials}{}\\
Let $X,Y$ be schemes. Let $f:X\to Y$ be a morphism. Let $U\subseteq X\times_YX$ be the largest open subset such that $\Delta:X\to U$ is a closed immersion. Let $\mI$ be the ideal sheaf of $\Delta:X\to U$. Define the sheaf of relative differentials of $f$ to be $$\Omega_{X/Y}^1=\Delta^\ast(\mI/\mI^2)$$
\end{defn}

If $f$ is separated, then $\Delta:X\to X\times_YX$ is already a closed immersion and there is no need to take an open subset of $X\times_YX$ to form the ideal sheaf. In this case, we have $\Omega_{X/Y}^1=\mI/\mI^2$. \\

\begin{lmm}{}{}\\
Let $X,Y$ be schemes. Let $f:X\to Y$ be a morphism. Then $\Omega_{X/Y}^1$ is a quasicoherent $\mO_X$-module. 
\end{lmm}

There is an alternative way of defining the sheaf of relative differentials. 

\begin{lmm}{}{}\\
Let $X,Y$ be schemes. Let $f:X\to Y$ be a morphism. Let $V\subseteq Y$ be an affine open subset. Let $U\subseteq f^{-1}(V)$ be an affine open subset of $X$. Then there is a an isomorphism $$\Omega_{X/Y}^1\;|\;_U\cong\widetilde{\Omega_{\Gamma(U)/\Gamma(V)}^1}$$
\end{lmm}

\begin{eg}{}{}\\
Let $X=\A_Y^n$. Then $\Omega_{X/Y}^1$ is a free $\mO_X$-module of rank $n$. 
\end{eg}

	\arrow[from=1-3, to=1-4]
\end{tikzcd}}
\end{prp}

Let $X,Y,Z$ be schemes. Let $f:Y\to Z$ be a morphism. Let $i:X\to Y$ be a closed embedding with ideal sheaf $\mI$. Then there is an exact sequence of sheaves on $X$: \\

\subsection{The Tangent Sheaf}
\begin{defn}{The Relative Tangent Sheaf}{}\\
Let $X,Y$ be a schemes. Let $f:X\to Y$ be a morphism. Define the relative tangent sheaf of $X$ to be $$\mT_{X/Y}=\left(\Omega_{X/Y}^1\right)^\vee$$
\end{defn}

\begin{defn}{The Zariski (Co)Tangent Space}{}\\
Let $X$ be a scheme. Let $p\in X$. Let $m_p$ be the unique maximal ideal of $\mO_{X,p}$. 
\begin{itemize}
\item Define the Zariski tangent space at $p$ to be $$T_pX=\left(\frac{m_p}{m_p^2}\right)^\ast$$
\item Define the Zariski cotangent space at $p$ to be $$T_pX^\ast=\frac{m_p}{m_p^2}$$
\end{itemize}
\end{defn}

\begin{prp}{}{}\\\
Let $X$ be a scheme. Let $p\in X$. Then we have $$T_pX\cong\mT_{X,p}\otimes_{\mO_{X,p}}\kappa(p)$$
\end{prp}

\begin{crl}{}{}\\
Let $X$ be a scheme. Let $p\in X$. Then we have $$T_pX^\ast\cong\Omega_{X,p}^1\otimes_{\mO_{X,p}}\kappa(p)$$
\end{crl}

\begin{prp}{}{}\\
Let $k$ be a field. Let $X$ be a scheme of finite type over $k$. Let $p\in X(k)$ be a $k$-valued point. Then there is an isomorphism $$T_pX\cong\text{coker}(J_pX)$$
\end{prp}

\subsection{The Normal Bundle}
\begin{defn}{The Normal Sheaf}{}\\
Let $X,Y$ be schemes. Let $\iota:Y\to X$ be a closed embedding. Define the normal sheaf of $Y$ in $X$ by $$\mN_{Y/X}=\Hom_{\mO_Y}\left(\frac{\mI_Y}{\mI_Y^2},\mO_Y\right)$$
\end{defn}

\subsection{Flat Morphisms}
\begin{thm}{Miracle Flatness}{}\\
Let $k$ be a field. Let $X,Y$ be schemes over $k$ with pure dimension. Let $f:X\to Y$ be a morphism of $k$-schemes. Suppose that the following are true. 
\begin{itemize}
\item $X$ is Cohen-Macaulay. 
\item $Y$ is regular. 
\item Fibers of $f$ have dimension $\dim(X)-\dim(Y)$. 
\end{itemize}
Then $f$ is flat. 
\end{thm}

\subsection{Smooth Morphisms}
\begin{defn}{Smooth Morphisms}{}\\
Let $k$ be a field. Let $X,Y$ be schemes locally of finite type over $k$. Let $f:X\to Y$ be a morphism over $k$. We say that $f$ is smooth of relative dimension $n$ if the following are true. 
\begin{itemize}
\item $f$ is flat. 
\item If $X'\subseteq X$ and $Y'\subseteq Y$ are irreducible components such that $f(X')\subseteq Y'$, then $\dim(X')=\dim(Y')+n$. 
\item For all $p\in X$, we have $$\dim_{k(p)}(\Omega_{X/Y}^1\otimes k(p))=n$$
\end{itemize}
\end{defn}

Smooth morphisms are a direct generalization of smooth schemes. 

\begin{lmm}{}{}\\
Let $k$ be a field. Let $X$ be a scheme locally of finite type over $k$. Then the following are equivalent. 
\begin{itemize}
\item $X$ is smooth. 
\item The canonical morphism $X\to\text{Spec}(k)$ is smooth. 
\item $\Omega_X^1$ is locally free. 
\end{itemize}
\end{lmm}

Notice that if $X$ is irreducible, then the third criterion is equivalent to saying that $\dim_k(\Omega_{X/Y}^1)=n$. 

\begin{prp}{}{}\\
Let $k$ be an algebraically closed field. Let $X,Y$ be schemes locally of finite type over $k$. Let $f:X\to Y$ be a morphism over $k$. Then the following are equivalent. 
\begin{itemize}
\item $f$ is smooth of relative dimension $\dim(X)-\dim(Y)$. 
\item $\Omega_{X/Y}^1$ is locally free of rank $\dim(X)-\dim(Y)$. 
\item For all closed points $p\in X$, the induced map $T_pX\to T_{f(p)}Y$ is surjective. 
\end{itemize}
\end{prp}

\subsection{The Blowing Up Construction}
\begin{defn}{The Blow Up Along a Subscheme}{}\\
Let $X$ be a scheme. Let $Y\subseteq X$ be a closed subscheme of $X$. Define the blow up of $Y$ in $X$ to be $$\text{Bl}_X(Y)=\text{Proj}_X\left(\bigoplus_{i=0}^\infty\mI_Y\right)$$ together with the canonical projection $\pi:\text{Bl}_X(Y)\to X$. 
\end{defn}

\begin{prp}{}{}\\
Let $X$ be a scheme. Let $Y\subseteq X$ be a closed subscheme of $X$. Let $U=\text{Spec}(R)\subseteq X$ be an affine open subset. Then we have $$\pi^{-1}(U)\cong\text{Proj}\left(R[\I^S(Y\cap U)t]\right)$$
\end{prp}

\begin{prp}{The Universal Property of Blow Ups}{}\\
Let $X$ be a scheme. Let $Y\subseteq X$ be a closed subscheme of $X$. Then $\text{Bl}_X(Y)$ satisfies the following universal property: For any scheme $Z$ and a morphism $f:Z\to X$ such that $f^{-1}\mI_Y\in\bold{Mod}_{\mO_Z}$ is invertible, there exists a unique morphism $g:Z\to\text{Bl}_X(Y)$ such that the following diagram commutes: \\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	Z & {\text{Bl}_X(Y)} \\
	& X
	\arrow["{\exists!g}", dashed, from=1-1, to=1-2]
	\arrow["f"', from=1-1, to=2-2]
	\arrow["\pi", from=1-2, to=2-2]
\end{tikzcd}} \\
\end{prp}

\begin{prp}{}{}\\
Let $X$ be a scheme. Let $Y\subseteq X$ be a closed subscheme of $X$. Then $\pi:\text{Bl}_X(Y)\to X$ is proper. 
\end{prp}

\begin{lmm}{}{}\\
Let $X$ be a scheme. Let $Y\subseteq X$ be a closed subscheme of $X$. If $\mI_Y$ is invertible, then $\pi$ induces an isomorphism $\text{Bl}_X(Y)\cong X$. 
\end{lmm}

\subsection{The Exceptional Divisor}
\begin{defn}{The Exceptional Divisor}{}\\
Let $X$ be a scheme. Let $Y\subseteq X$ be a closed subscheme of $X$. Define the exceptional divisor of $Y$ in $X$ to be $$E_XY=\pi^{-1}(Y)\subseteq\text{Bl}_X(Y)$$
\end{defn}

\begin{lmm}{}{}\\
Let $X$ be a scheme. Let $Y\subseteq X$ be a closed subscheme of $X$. Then we have $$E_XY\cong\text{Proj}_X\left(\bigoplus_{k=0}^\infty\frac{\mI_Y^k}{\mI_Y^{k+1}}\right)$$
\end{lmm}

\begin{lmm}{}{}\\
Let $X$ be a scheme. Let $Y\subseteq X$ be a closed subscheme of $X$. Let $U=\text{Spec}(R)$ be an affine open subset of $X$. Then we have $$E_XY\cap\pi^{-1}(U)=\text{Proj}\left(\text{gr}_R(\I^S(Y\cap U))\right)$$
\end{lmm}

\begin{prp}{}{}\\
Let $X$ be a scheme. Let $Y\subseteq X$ be a closed subscheme of $X$. Then $E_XY$ is an effective Cartier divisor of $\text{Bl}_X(Y)$. 
\end{prp}

\begin{prp}{}{}\\
Let $X$ be a scheme. Let $Y\subseteq X$ be a closed subscheme of $X$. Then we have an isomorphism $$\text{Bl}_X(Y)\setminus E_XY\cong X\setminus Y$$ induced by $\pi|_{\text{Bl}_X(Y)\setminus E_XY}$. 
\end{prp}

\section{Cohomology of Schemes}
\subsection{Cech Cohomology of Schemes}
Let $X$ be a space. Let $\mU$ be an open cover of $X$. Let $\mF$ be a sheaf on $X$. Recall that there is a natural map $$\text{\v{H}}^p(X,\mU,\mF)\to H^p(X,\mF)$$ where $\text{\v{H}}^p(X,\mU,\mF)$ is the \v{C}ech cohomology of $X$ (the cohomology of the chain complex $C^\bullet(X,\mU,\mF)\in\bold{Ch}(\mA)$. 

\begin{prp}{}{}\\
Let $X$ be a Noetherian separated scheme. Let $\mU$ be an affine open cover of $X$. Let $\mF$ be a quasi-coherent sheaf of $\mO_X$-modules. Then there is a natural isomorphism $$\text{\v{H}}^i(X,\mU,\mF)\cong H^i(X,\mF)$$ for all $i>0$. 
\end{prp}

Recall that for any space $X$ and any sheaf $\mF$ on $X$, there are natural isomorphisms $$\varinjlim_{\mU}\text{\v{H}}^1(X,\mU,\mF)\cong H^1(X,\mF)$$

\subsection{Cohomology of a Noetherian Scheme}
\begin{thm}{Vanishing Theorem of Serre}{}\\
Let $(X,\mO_X)$ be a Noetherian scheme. Let $\mF$ be an $\mO_X$-module. Then we have $$H^i(X,\mF)=0$$ for all $i>\dim(X)$. 
\end{thm}

\begin{prp}{}{} Let $I$ be an injective module over a Noetherian ring $A$. Then the sheaf $\widetilde{I}$ on $\text{Spec}(A)$ is flasque. 
\end{prp}

\begin{prp}{}{}\\
Let $R$ be a Noetherian commutative ring. Let $\mF$ be an $\mO_{\text{Spec}(R)}$-module. Then we have $$H^i(\text{Spec}(R),\mF)=\begin{cases}
\mF(\text{Spec}(R)) & \text{ if } i=0\\
0 & \text{ otherwise }
\end{cases}$$
\end{prp}

Note that result is also true if we drop the requirement that $A$ is Noetherian. But the proof is more difficult. 

\begin{thm}{}{}\\
Let $X$ be a Noetherian scheme. Then the following are equivalent. 
\begin{itemize}
\item $X$ is an affine scheme
\item $H^i(X,\mF)=0$ for all quasi-coherent sheaves $\mF$ and all $i>0$
\item $H^1(X,\mI)=0$ for all coherent sheaves of ideals $\mI$
\end{itemize}
\end{thm}

\begin{prp}{}{}\\
Let $R$ be a Noetherian commutative ring. Let $X$ be a proper scheme over $\text{Spec}(R)$. Let $\mL$ be an invertible sheaf on $X$. Then $\mL$ is ample if and only if for every coherent sheaf $\mF$ on $X$, there exists $N\in\N$ such that $$H^i(X,\mF\otimes\mL^{\otimes n})=0$$ for all $i>0$ and $n\geq N$. 
\end{prp}

\section{Duality Theorems in Cohomology}
\subsection{The Ext Sheaf}
\begin{prp}{}{}\\
Let $X$ be a Notherian scheme. Let $\mF,\mG\in\bold{Mod}_{\mO_X}$. Suppose that $\mF$ is coherent. Then we have $$\mathscr{E}\text{xt}_{\mO_X}^i(\mF,\mG)_p\cong\text{Ext}_{\mO_{X,p}}^i(\mF_p,\mG_p)$$ for all $p\in X$. 
\end{prp}

\begin{prp}{}{}\\
Let $R$ be a Noetherian commutative ring. Let $M,N$ be $R$-modules. Suppose that $M$ is finitely generated. Then the following are true. 
\begin{itemize}
\item There is an isomorphism $$\text{Ext}_{\text{Spec}(R)}^i(\widetilde{M},\widetilde{N})\cong\text{Ext}_R^i(M,N)$$
\item There is an isomorphism $$\mathscr{E}\text{xt}_{\text{Spec}(R)}^i(\widetilde{M},\widetilde{N})\cong\widetilde{\text{Ext}_R^i(M,N)}$$
\end{itemize}
\end{prp}

\subsection{The Dualizing Sheaf}
\begin{defn}{The Dualizing Sheaf}{}\\
Let $k$ be a field. Let $X$ be a projective scheme over $k$. A dualizing sheaf of $X$ is a coherent sheaf of $\mO_X$-modules $\omega_X$ together with a map $t:H^{\dim(X)}(X,\omega_X)\to k$ such that the natural map $$\Hom(\mF,\omega_X)\times H^n(X,\mF)\to H^n(X,\omega_X)\overset{t}{\longrightarrow}k$$ is a perfect pairing for all $\mF\in\bold{Coh}_X$. 
\end{defn}

In other words, the dualizing sheaf represents the functor $$\mF\mapsto(H^{\dim(X)}(X,\mF))^\ast$$

\begin{prp}{}{}\\
Let $k$ be a field. Let $X$ be a projective scheme over $k$. If a dualizing sheaf exists for $X$, then it is unique up to unique isomorphism. 
\end{prp}

\begin{prp}{}{}\\
Let $k$ be a field. Let $X$ be a projective scheme over $k$. Then $X$ admits a dualizing sheaf given by $$\omega_X=\mathscr{E}\text{xt}_{\Prj_k^N}^r(\mO_X,\omega_{\Prj_k^N})$$ where we have chosen an embedding $X\to\Prj_k^N$ as a subscheme and $r=\text{codim}(X)$. 
\end{prp}

\subsection{Relation to Cohen-Macaulay}
But we want duality in all cohomology groups, not just the top one. 

\begin{prp}{}{}\\
Let $k$ be a field. Let $X$ be a projective scheme over $k$. Then for all $\mF\in\bold{Coh}_X$, there exists natural maps $$\theta^i:\text{Ext}^i(\mF,\omega_X)\to(H^{\dim(X)-i}(X,\mF))^\ast$$ such that $\theta^0$ is the isomorphism in the definition of the dualizing sheaf. 
\end{prp}

\begin{prp}{}{}\\
Let $k$ be a field. Let $X$ be a projective scheme over $k$. Then the following are equivalent. 
\begin{itemize}
\item $X$ is Cohen-Macaulay and equidimensional. 
\item We have $\theta^i:\text{Ext}^i(\mF,\omega_X)\to(H^{\dim(X)-i}(X,\mF))^\ast$ is an isomorphism for all $i$ and all $\mF\in\bold{Coh}_X$. 
\item We have $H^i(X,\mF(-k))=0$ for any $\mF\in\bold{Loc}_X$ and $i\leq\dim(X)-1$ and $q>N$ for some $N\in\N$. 
\end{itemize}
\end{prp}

\begin{crl}{}{}\\
Let $k$ be a field. Let $X$ be a projective Cohen-Macaulay scheme over $k$. Then there are natural isomorphisms $$H^i(X,\mF)\cong(H^{\dim(X)-i}(X,\mF^\vee\otimes\omega_X))^\ast$$ for any $\mF\in\bold{Loc}_X$. 
\end{crl}

\subsection{The Dualizing Complex}

If $X$ is Cohen-Macaulay, then the dualizing complex is concentrated in degree $\dim(X)$. 

\subsection{Relation to Gorenstein}
\begin{prp}{}{}\\
Let $k$ be a field. Let $X$ be a Cohen-Macaulay scheme of finite type over $k$. Then $X$ is Gorenstein if and only if $\omega_X$ is invertible. 
\end{prp}

\begin{defn}{The Associated Map of Divisors}{}\\
Let $k$ be a field. Let $X$ be a variety over $k$. Let $D$ be a Cartier divisor on $X$. Define the associated map of $D$ to be the map $F_D:X\to\Prj(\Gamma(X,\mL(D))^\ast)$ given by $$p\mapsto\left[\text{ev}_p\right]$$ where $\text{ev}_p:\Gamma(X,\mL(D))\to\mL(D)_p\otimes_{\mO_{X,p}}k(p)$ is the map given by $f\mapsto f\otimes 1$. 
\end{defn}

\begin{lmm}{}{}\\
Let $k$ be a field. Let $X$ be a variety over $k$. Let $D$ be a Cartier divisor on $X$. Let $s_0,\dots,s_n$ a basis for $\Gamma(X,\mL(D))$. Then the associated map is given by $$F_D(p)=[s_0(p):\cdots:s_n(p)]$$
\end{lmm}

\begin{lmm}{}{}\\
Let $k$ be a field. Let $X$ be a variety over $k$. Let $D$ be a Cartier divisor on $X$. Then the following are true. 
\begin{itemize}
\item $F_D$ is a rational map. 
\item $F_D$ is a morphism if and only if $\mL(D)$ is base point free. 
\end{itemize}
\end{lmm}

\begin{lmm}{}{}\\
Let $k$ be a field. Let $X$ be a variety over $k$. Let $D$ be a Cartier divisor on $X$. Suppose that $\mL(D)$ is base point free. Then $F_D$ corresponds to the morphism to projective space from the one to one correspondence to invertible sheaves. 
\end{lmm}

\begin{defn}{Complete Linear Systems}{}\\
Let $k$ be a field. Let $X$ be a variety over $k$. Let $D$ be a Cartier divisor on $X$. Define the complete linear system of $D$ to be $$\abs{D}=\{D'\in\text{CDiv}(X)\;|\;D\sim D'\text{ and }D'\text{ is effective }\}$$
\end{defn}

\begin{prp}{}{}\\
Let $k$ be an algebraically closed field. Let $X$ be a smooth projective variety over $k$. Let $D$ be an effective Cartier divisor on $X$. Then there is a bijection $$\Prj(\Gamma(X,\mL(D))^\ast)\;\;\overset{1:1}{\longleftrightarrow}\;\;\{E\in[D]\;|\;E\text{ is effective }\}$$ given by $[f]\mapsto D+\text{div}(f)$. 
\end{prp}

Notice that if $D$ itself is effective, then the inverse of the above bijection sends $D$ to the equivalence class of all constant functions in $\Prj(\mL(D))$.

\subsection{Blowing Up Varieties}

\end{thm}

\subsection{Sheaf Cohomology of Varieties}
If $X$ is affine, then we already know that $$H^n(X;\mF)=\begin{cases}

0 & \text{ otherwise }

	\arrow[from=2-3, to=2-4]

\end{itemize}

Now if $i:X\hookrightarrow\Prj_k^n$ is a hypersurface of degree $d$, then we also have an exact sequence: \\

\subsection{The Dualizing Sheaf and the Canonical Sheaf}
\begin{prp}{}{}\\
Let $k$ be an algebraically closed field. Let $X$ be a smooth projective variety over $k$. Then there is an isomorphism $$\omega_X\cong\mK_X$$
\end{prp}

\begin{crl}{}{}\\
Let $k$ be an algebraically closed field. Let $X$ be a smooth projective variety over $k$. Then there are natural isomorphisms $$H^q(X,\Omega_X^p)\cong(H^{\dim(X)-q}(X,\Omega_X^{\dim(X)-p}))^\ast$$
\end{crl}

\begin{prp}{}{}\\
Let $k$ be a field. Let $X$ be a normal and Cohen-Macaulay variety over $k$. Then there is an isomorphism $$\omega_X\cong\mK_X$$
\end{prp}

\begin{defn}{Fano, Calabi-Yau and Varieties of General Type}{}\\
Let $k$ be a field. Let $X$ be a smooth variety over $k$. 
\begin{itemize}
\item We say that $X$ is a fano variety if $-K_X$ is ample. 
\item We say that $X$ is Calabi-Yau if $K_X=0$. 
\item We say that $X$ is of general type if $K_X$ is ample. 
\end{itemize}
\end{defn}
