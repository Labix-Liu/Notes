\section{Metric Properties of Euclidean Space}
\subsection{Component-wise Properties}
\begin{thm}{Bolzano-Weierstrass Theorem}{} Any bounded sequence in $\R^n$ has a convergent subsequence. 
\begin{proof} Let $\{x_n\}\subset\R^n$ be bounded. By Bolzano-Weierstrass Theorem on $\R$, the sequence $\{x_{1,m}\}\subset\R$ has a convergent subsequence $\{x_{1,m_k}\}$. Apply Bolzano-Weierstrass Theorem to $\{x_{2,m_k}\}$ and keep going until you reach the $n$th component. We will end up with a subsequence that converges for all components and thus is convergent in $\R^n$. 
\end{proof}
\end{thm}

\begin{prp}{}{}\\
Let $m\in\N$. Let $(x_n)_{n\in\N}$ and $(y_n)_{n\in\N}$ be sequences in $\R^m$. Suppose that $(x_n)_{n\in\N}$ converges to $x\in\R^m$ and $(y_n)_{n\in\N}$ converges to $y\in\R^m$. Then the following are true. 
\begin{itemize}
\item If $\lambda,\mu\in\R$, then $(\lambda x_n+\mu y_n)_{n\in\N}$ converges to $\lambda x+\mu y$. 
\item $(x_ny_n)_{n\in\N}$ converges to $xy$. 
\end{itemize}
\end{prp}

\begin{thm}{Componentwise Continuity}{} Let $f:U\subseteq\R^n\to\R^m$ be a function. Then $f$ is continuous if and only if each of its components $f_1,\dots,f_n:U\to\R$ is continuous. 
\begin{proof}
Suppose that $f$ is continuous at $a\in\R^n$. Let $\epsilon>0$. Then there exists $\delta>0$ from continuity such that $\|x-a\|<\delta$ implies $\|f(x)-f(a)\|<\epsilon$. Then $\|x-a\|<\delta$ implies $$\|f_i(x)-f_i(a)\|\leq\|f(x)-f(a)\|<\epsilon$$ Thus each component is continuous. \\
Now let $f$ be component wise continuous at $a\in\R^n$. Let $\epsilon>0$. For each component $f_i:\R^n\to\R$, using continuity with $\frac{\epsilon}{\sqrt{n}}>0$there exists $\delta_i>0$ such that $\|x-a\|<\delta$ implies $\|f_i(x)-f_i(a)\|<\frac{\epsilon}{\sqrt{n}}$. Choose $\delta=\max\{\delta_1,\dots,\delta_n\}$. Then $\|x-a\|<\delta$ implies that 
\begin{align*}
\|f(x)-f(a)\|^2&=\sum_{k=1}^n\|f_k(x)-f_k(a)\|^2\\
&\leq n\max_{k\in\{1,\dots,n\}}\|f_k(x)-f_k(a)\|^2\\
\|f(x)-f(a)\|&\leq\sqrt{n}\max_{k\in\{1,\dots,n\}}\|f_k(x)-f_k(a)\|\\
&\leq\sqrt{n}\frac{\epsilon}{\sqrt{n}}\\
&=\epsilon
\end{align*}
Thus $f$ is continuous. 
\end{proof}
\end{thm}

\begin{prp}{}{} Let $U\subseteq\R^n$ be a subset. Let $f,g:U\to\R^m$ be continuous functions. Then the following are true. 
\begin{itemize}
\item If $\lambda,\mu\in\R$, then $\lambda f+\mu g$ is continuous. 
\item $fg$ is continuous. 
\item If $g(x)\neq 0$ for all $x\in U$, then $\frac{f}{g}$ is continuous. 
\end{itemize}
\end{prp}

\begin{defn}{Limits}{}\\
Let $U\subseteq\R^n$ be a subset. Let $f:U\to\R^m$ be a function. Let $c\in U$. We say that the limit of $f$ at $c$ is $L\in\R^m$ if for all $\varepsilon>0$, there exists $\delta>0$ such that $\abs{x-c}<\delta$ implies $$\abs{f(x)-L}<\varepsilon$$ In this case, we write $L=\lim_{x\to c}f(x)$. 
\end{defn}

\begin{lmm}{}{}\\
Let $U\subseteq\R^n$ be a subset. Let $f:U\to\R^m$ be a function. Let $c\in U$. Let $L=(L_1,\dots,L_m)\in\R^m$. Suppose that $f$ is given by $f(x)=(f_1(x),\dots,f_m(x))$ for $f_i:U\to\R$. Then $\lim_{x\to c}f(x)=L$ if and only if $\lim_{x\to c}f_i(x)=L_i$ for $1\leq i\leq m$. 
\end{lmm}

\begin{lmm}{}{}\\
Let $U\subseteq\R^n$ be a subset. Let $f:U\to\R^m$ be a function. Let $c\in U$. Then $f$ is continuous at $c$ if and only if $\lim_{x\to c}f(x)=f(c)$. 
\end{lmm}

\begin{thm}{Extreme Value Theorem}{}\\
Let $n,m\in\N\setminus\{0\}$. Let $K\subseteq\R^n$ be a closed and bounded subset. Let $f:K\to\R^m$ be a function. Then there exists $a,b\in K$ such that $$f(a)\leq f(x)\leq f(b)$$ for all $x\in K$. 
\begin{proof}
We know that $f(K)$ is closed and bounded. Thus it must have a supremum $M$ and infinum $m$ that are finite. Then we must have $(a_n)_{n\in\N}$ and $(b_n)_{n\in\N}$ such that $a_n\to m$ and $b_n\to M$ by definition of supremum and infinum. Since $f(K)$ is closed, $m,M\in f(K)$ and we are done. 
\end{proof}
\end{thm}

\subsection{Metric Structure on The Space of Linear Maps}
In the discussion of derivatives of functions of the form $f:U\to\R^n\to\R^m$, we will need to understand maps of the form $U\to\Hom_\R(\R^n,\R^m)$ and $U\to M_{m\times n}(\R)$. Recall from Linear Algebra that the codomain of both types of maps are isomorphic to $\R^{n\times m}$, and through the isomorphism we may introduce the notion of metrics and hence continuity to these maps. In order to understand the metric structure more explicitly, we shall define the metric structures of each of them more explicitly, and discuss their relations. \\

Recall from linear algebra that given a choice of basis $e_1,\dots,e_n$ of $\R^n$ and a choice of basis $f_1,\dots,f_m$ of $\R^m$, we can construct an explicit vector space isomorphism $$\Hom_\R(\R^n,\R^m)\cong M_{m\times n}(\R)$$ as follows. Given a linear map $T$, we may write $T(e_i)=\sum_{k=1}^ma_{ki}f_k$ for $a_{ki}\in\R$. Then the isomorphism is given by $$T\mapsto\begin{pmatrix}
a_{11} & \cdots & a_{1n}\\
\vdots & \ddots & \vdots\\
a_{m1} & \cdots & a_{mn}
\end{pmatrix}$$ We use the $\mu:\Hom_\R(\R^n,\R^m)\overset{\cong}{\longrightarrow}M_{m\times n}(\R)$ to denote the isomorphism when $e_1,\dots,e_n$ and $f_1,\dots,f_m$ are the standard basis vectors for $\R^n$ and $\R^m$ respectively. 

\begin{defn}{Operator Norm}{}\\
Define the operator norm of the vector space $\Hom_\R(\R^n,\R^m)$ to be the function $\|\cdot\|:\Hom_\R(\R^n,\R^m)\to\R$ given by $$\|T\|=\sup\{\abs{T(x)}\;|\;x\in\R^n\text{ such that }\abs{x}=1\}$$
\end{defn}

\begin{lmm}{}{} Let $T\in\Hom_\R(\R^n,\R^k)$ and $S\in\Hom_\R(\R^k,\R^m)$. Then $$\|S\circ T\|\leq\|S\|\|T\|$$ 
\begin{proof}
We have that 
\begin{align*}
\abs{S(T(x))}&\leq\|S\|\abs{T(x)}\\
&\leq\|S\|\|T\|\abs{x}
\end{align*}
Thus we are done. 
\end{proof}
\end{lmm}

\begin{defn}{The Operator Metric}{}\\
Define the operator metric $d:\Hom_\R(\R^n,\R^m)\times\Hom_\R(\R^n,\R^m)\to\R$ by the formula $$d(T,S)=\|T-S\|$$
\end{defn}

This gives a metric structure on $\Hom_\R(\R^n,\R^m)$ and we can now talk about functions $f:\R^k\to\Hom_\R(\R^n,\R^m)$ which are continuous. Similarly, we would like to talk about continuous functions of the form $\R^k\to M_{m\times n}(\R)$. 

\begin{defn}{The Frobenius Norm}{}\\
Define the Frobenius norm of the vector space $M_{m\times n}(\R)$ to be the function $\|\cdot\|:M_{m\times n}(\R)\to\R$ given by $$\|A\|_F=\left(\sum_{j=1}^n\sum_{i=1}^ma_{ij}^2\right)^{\frac{1}{2}}$$
\end{defn}

\begin{defn}{The Frobenius Metric}{}\\
Define the operator metric $d:M_{m\times n}(\R)\times M_{m\times n}(\R)\to\R$ by the formula $$d(A,B)=\|A-B\|_F$$
\end{defn}

\begin{lmm}{}{}\\
Let $T\in\Hom_\R(\R^n,\R^k)$ be a linear map. Let $A$ be the matrix of $T$ with respect to the standard basis. Then we have $$\frac{1}{\sqrt{n}}\|A\|_F\leq\|T\|\leq\|A\|_F$$
\begin{proof}\\
\end{proof}
\end{lmm}

\begin{lmm}{}{}\\
Let $U\subseteq\R^k$ be an open subset. Let $f:U\to\Hom_\R(\R^n,\R^m)$ be a function. Then $f$ is continuous if and only if $\mu\circ f:U\to M_{m\times n}(\R)$ is continuous. 
\begin{proof}\\
\end{proof}
\end{lmm}

This lemma will give us our condition for continuous derivatives in terms of the Jacobian matrix later. However, we may also interpret this lemma as follows: To check the continuity of a map of the form $U\to\Hom_\R(\R^n,\R^m)$, it suffices to check that each term in the matrix function $U\to M_{m\times n}(\R)$ is continuous. This is a practical way of checking that functions of the given type are continuous.

\section{Differentiation}
\subsection{Frechet Derivative}
\begin{defn}{Frechet Derivatives}{}\\
Let $U\subseteq\R^n$ be an open subset. Let $x\in U$. Let $f:U\to\R^m$ be a function. We say that $f$ is differentiable at $x$ if there exists a linear map $Df(x)\in\Hom_\R(\R^n,\R^m)$ such that $$\lim_{h\to 0}\frac{\abs{f(x +h)-f(x)-Df(x)h}}{\abs{h}}=0$$ In this case, we say that the derivative of $f$ at $x$ is $Df(x)$. 
\end{defn}

For each function $f$, $Df$ is a map from $U$ to $\Hom_\R(\R^n,\R^m)$, and $D$ is a map that sends differentiable functions $f:U\to\R^m$ to $Df:U\to\Hom_\R(\R^n,\R^m)$. 

\begin{eg}{}{}\\
Let $f:\R\to\R$ be the map given by $f(x)=x^2+5x$. Let $p\in\R$. Then $Df(p)=2p+5$. 
\end{eg}

\begin{prp}{}{}\\
Let $U\subseteq\R^n$ be an open subset. Let $x\in U$. Let $f:U\to\R^m$ be a function. Suppose that $f$ is differentiable at $x$ with derivatives $A$ and $B$ in $\Hom_\R(\R^n,\R^m)$. Then $A=B$. 
\begin{proof}
Suppose that $A,B$ both represent the derivative of $f:U\subseteq\R^n\to\R^m$. Then fix $\epsilon>0$. There exists $\delta_1,\delta_2$ such that $\abs{h}<\min\{\delta_1,\delta_2\}$ implies $$\frac{\abs{f(x+h)-f(x)-Ah}}{\abs{h}}<\epsilon$$ and $$\frac{\abs{f(x+h)-f(x)-Bh}}{\abs{h}}<\epsilon$$ Then we have 
\begin{align*}
\abs{(A-B)h}&=\abs{f(x+h)-f(x)-Bh-f(x+h)-f(x)-Ah}\\
&\leq\abs{f(x+h)-f(x)-Bh}+\abs{f(x+h)-f(x)-Ah}\\
&\leq\epsilon\abs{h}+\epsilon\abs{h}\\
&=2\epsilon\abs{h}
\end{align*}
Thus $A=B$ by definition of limit and we are done. 
\end{proof}
\end{prp}

\begin{prp}{}{}\\
Let $U\subseteq\R^n$ be an open subset. Let $x\in U$. Let $f:U\to\R^m$ be a function. If $f$ is differentiable at $x$, then $f$ is continuous at $x$. 
\begin{proof}
Since $f$ is continuous at $x$, there exists $A\in\R^{m\times n}$ such that $$\lim_{h\to 0}\frac{\abs{f(x+h)-f(x)-Ah}}{\abs{h}}=0$$ Fix $\epsilon>0$, there exists $\delta_1>0$ such that $\abs{h}<\delta_1$ 
\begin{align*}
\abs{f(x+h)-f(x)-Ah}&\leq\epsilon\abs{h}\\
\abs{f(x+h)-f(x)}&\leq\abs{Ah}+\epsilon\abs{h}\\
&<(\|A\|+\epsilon)\abs{h}
\end{align*}
Set $\delta_2=\min\left\{\delta_1,\frac{\epsilon}{\|A\|+\epsilon}\right\}$. Then $\abs{h}<\delta_2$ implies $$\abs{f(x+h)-f(x)}<(\|A\|+\epsilon)\delta_2<\epsilon$$ and we are done. 
\end{proof}
\end{prp}

Similar to continuity, differentiability in higher dimensions can be broken down by its individual components. This way we only need to show individual differentiability to save trouble. 

\begin{prp}{Componentwise Differentiability}{}\\
Let $U\subseteq\R^n$ be an open subset. Let $x\in U$. Let $f:U\to\R^m$ be a function whose components are given by $f=(f_1,\dots,f_m)$. Then $f$ is differentiable at $x$ if and only if $f_i:U\to\R$ is differentiable at $x$ for $1\leq i\leq m$. 
\end{prp}

\begin{eg}{}{}\\
Let $U\subseteq\R^n$ be an open subset. Let $f:U\to\R^m$ be the function given by $f(x)=c$ for some $c\in\R^m$. Then $f$ is differentiable at $U$ and $Df(x)=0$ for all $x\in U$. 
\end{eg}

\begin{eg}{}{}\\
Let $f:\R^n\to\R^m$ be the linear map given by $f(x)=Ax$ for some $A\in M_{n\times m}(\R)$. Then $f$ is differentiable and $Df(x)=A$. 
\end{eg}

\begin{prp}{}{}\\
Let $U\subseteq\R^n$ be an open subset. Let $x\in U$. Let $f,g:U\to\R^m$ be differentiable at $x$. Let $\lambda,\mu\in\R$. Then we have $$D(\lambda f+\mu g)(x)=\lambda Df(x)+\mu Dg(x)$$
\end{prp}

\begin{prp}{}{}\\
Let $U\subseteq\R^n$ be an open subset. Let $x\in U$. Let $f:U\to\R^m$ be differentiable at $x$. Let $g:V\subseteq f(U)\to\R^k$ be differentiable at $f(x)$. Then we have $$D(g\circ f)(x)=Dg(f(x))\cdot Df(x)$$
\end{prp}

\begin{prp}{}{}\\
Let $U,V\subseteq\R^n$ be open subsets. Let $f:U\to V$ be a bijective function. Suppose that $f$ is differentiable at $p\in U$ and $f^{-1}$ is differentiable at $f(p)$. Then $Df(p)$ is invertible and we have $$Df^{-1}(p)=(Df(p))^{-1}$$
\end{prp}

\subsection{Partial Derivatives and Directional Derivatives}
\begin{defn}{Partial Derivatives}{}\\
Let $e_1,\dots,e_n$ be the standard basis for $\R^n$. Let $U\subseteq\R^n$ be an open subset. Let $p\in U$. Let $f:U\to\R^m$ be a function whose components are given by $f=(f_1,\dots,f_m)$. Define the $j$th partial derivative of the $i$th component of $f$ at $p$ to be $$\frac{\partial f_i}{\partial x_j}\bigg{|}_p=\lim_{t\to 0}\frac{f_i(p+te_j)-f_i(p)}{t}=\frac{d}{dt}f(p+te_j)\bigg{|}_{t=0}$$
\end{defn}

Note that for a function $f:U\to\R^m$, the partial derivative $\frac{\partial f_i}{\partial x_j}\bigg{|}_p$ is a real number. 

\begin{eg}{}{}\\
Let $f:\R^2\to\R$ be the function defined by $$f(x,y)=\begin{cases}
\frac{xy}{(x^2+y^2)^2} & \text{ if }(x,y)\neq(0,0)\\
0 & \text{ if }x=y=0
\end{cases}$$
Then the following are true. 
\begin{itemize}
\item The $1$st partial derivative of $f$ at $(x,y)$ is given by $$\partial_{(1,0)}f(x,y)=\frac{y}{(x^2+y^2)}-\frac{4x^2y}{(x^2+y^2)^3}$$ for $(x,y)\in\R^2\setminus\{(0,0)\}$. 
\item The $2$nd partial derivative of $f$ at $(x,y)$ is given by $$\text{?}$$ for $(x,y)\in\R^2\setminus\{(0,0)\}$. 
\end{itemize}
\end{eg}

\begin{defn}{Jacobian Matrix}{}\\
Let $U\subseteq\R^n$ be an open subset. Let $p\in U$. Let $f:U\to\R^m$ be a function. Define the Jacobian of $f$ at $p$ to be the matrix $$Jf(p)=\begin{pmatrix}
\frac{\partial f_1}{\partial x_1}\bigg{|}_p & \cdots & \frac{\partial f_1}{\partial x_n}\bigg{|}_p\\
\vdots & \ddots & \vdots\\
\frac{\partial f_m}{\partial x_1}\bigg{|}_p & \cdots & \frac{\partial f_m}{\partial x_n}\bigg{|}_p
\end{pmatrix}$$
\end{defn}

\begin{defn}{Directional Derivative}{}\\
Let $U\subseteq\R^n$ be an open subset. Let $p\in U$ be a point. Let $f:U\to\R^m$ be a function. Let $v\in\R^n$ be a direction. Define the directional derivative of $f$ in the direction $v$ at $p$ to be $$\partial_vf(p)=\lim_{t\to 0}\frac{f(p+tv)-f(p)}{t}=\frac{d}{dt}f(p+tv)\bigg{|}_{t=0}$$ if the limit exists. 
\end{defn}

Note that $\partial_vf(p)$ is a vector in $\R^m$. 

\begin{lmm}{}{}\\
Let $e_1,\dots,e_n$ be the standard basis for $\R^n$. Let $U\subseteq\R^n$ be an open subset. Let $p\in U$ be a point. Let $f:U\to\R^m$ be a function whose components are given by $f=(f_1,\dots,f_m)$. Then we have $$\partial_{e_j}f(p)=\left(\frac{\partial f_1}{\partial x_j}\bigg{|}_p,\dots,\frac{\partial f_m}{\partial x_j}\bigg{|}_p\right)$$ for $1\leq j\leq n$. 
\end{lmm}

\begin{prp}{}{}\\
Let $U\subseteq\R^n$ be an open subset. Let $p\in U$. Let $f:U\to\R^m$ be a function. If $f$ is differentiable at $p$, then the following are true. 
\begin{itemize}
\item For any $v\in\R^n$, we have $Df(p)v=Jf(x)v=\partial_vf(x)$. 
\item For $\lambda,\mu\in\R$ and $v,w\in\R^m$, we have $$\partial_{\lambda v+\mu w}f(x)=\lambda\partial_vf(x)+\mu\partial_wf(x)$$
\end{itemize}
\begin{proof}
Suppose that $Df$ exists. Then using the definition of differentiability, we have that 
\begin{align*}
\lim_{t\to 0}\frac{f(x+tv)-f(x)-Df(x)(tv)}{t\abs{v}}&=0\\
\lim_{t\to 0}\frac{f(x+tv)-f(x)-tDf(x)v}{t}&=0\\
\lim_{t\to 0}\frac{f(x+tv)-f(x)}{t}&=\lim_{t\to 0}\frac{tDf(x)v}{t}\\
\lim_{t\to 0}\frac{f(x+tv)-f(x)}{t}&=Df(x)v\\
\partial_vf(x)&=Df(x)v
\end{align*}
Linearity follows since $Df(x)$ is matrix and matrix multiplication on a vector is linear. 
\end{proof}
\end{prp}

We should make absolutely clear distinction with the existence of frechet derivatives and the existence of directional derivatives. Even if the directional derivatives exists, as long as they are not linear, the matrix for the frechet derivative will not exist. In particular, it is possible to write out the Jacobian matrix as we see below, but this Jacobian matrix will not be equal to the matrix in the frechet derivative. \\

The above theorem is also useful for detecting non-differentiability. By verifying the directional derivatives are non-linear, one can use the contrapositive to prove that $f$ would not be differentiable. 

\begin{prp}{}{}\\
Let $U\subseteq\R^n$ be an open subset. Let $B_r(x_0)\subseteq U$ be a subset for some $r>0$ and $x_0\in U$. Let $f:U\to\R^m$ be a function. Suppose that the following are true. 
\begin{itemize}
\item The Jacobian $Jf(x)$ exists for all $x\in B_r(x_0)$. 
\item $Jf:B_r(x_0)\to M_{m\times n}(\R)$ is continuous. 
\end{itemize}
Then $f$ is differentiable. 
\begin{proof}
We show this for the case that $k=1$ since $f:\R^n\to\R^k$ is differentiable if and only if each component $f_1,\dots,f_k$ is differentiable. Thus now $f:\R^n\to\R$. Suppose that $\partial f$ is continuous in $B_r(x)$ for fixed $x$. This means that $\partial_{x_j}f_i$ is continuous for any $j\in\{1,\dots,n\}$ and $i\in\{1,\dots,k\}$. Take a point $x+h\in B_r(x)$. Define $h=\begin{pmatrix}
h_1\\ \vdots\\ h_n
\end{pmatrix}$ and $$h_k'=\begin{pmatrix}h_1\\\vdots\\ h_k\\0\\\vdots\\0\end{pmatrix}$$ Trivially define $h_0'=0$ and $h_0=0$ for convenience. Now $f(x+h_k')-f(x+h_{k-1}')$ is just a one variable function on the $k$th slot, thus we can apply the mean value theorem in real analysis and conclude that there exists $\theta_{k}\in(0,1)$ such that $f(x+h_k')-f(x+h_{k-1}')=\partial_{x_k}f(x+h_{k-1}'+(\theta_kh_k)e_k)h_k$. \\
Summing all these up, we have that 
\begin{align*}
\sum_{k=1}^nf(x+h_k')-f(x+h_{k-1}')&=\sum_{k=1}^n\partial_{x_k}f(x+h_{k-1}'+(\theta_kh_k)e_k)h_k\\
f(x+h)-f(x)&=\sum_{k=1}^n\partial_{x_k}f(x+h_{k-1}'+(\theta_kh_k)e_k)h_k
\end{align*}~\\

We now use the continuity of the partial derivatives. Given $\epsilon>0$, there $\delta_k>0$ such that $\abs{h}<\delta_k$ implies $\abs{\partial_{x_k}f(x+h)-\partial_{x_k}f(x)}<\epsilon$. Choose $\delta=\min\{\delta_1,\dots,\delta_n\}$. Since $\theta_k\in(0,1)$ for all $k\in\{1,\dots,n\}$, we must have $\abs{h_{k-1}'+(\theta_kh_k)e_k}<\abs{h}<\delta$. Thus
\begin{align*}
\abs{f(x+h)-f(x)-\sum_{k=1}^n\partial_{x_k}f(x)h_k}&=\abs{\sum_{k=1}^n\partial_{x_k}f(x+h_{k-1}'+(\theta_kh_k)e_k)h_k-\sum_{k=1}^n\partial_{x_k}f(x)h_k}\\
&<\epsilon\sum_{k=1}^n\abs{h_k}\\
&<\epsilon\sqrt{n}\abs{h}
\end{align*}
if $\abs{h}<\delta$. The second inequality is due to the fact that $\sum_{k=1}^nh_k\leq\sqrt{n}\abs{h}$. \\

Now define $A\in\Hom_\R(\R^n,\R)$ by $A(h)=(\partial f)h$. This makes sense in matrix multiplication since $\partial f\in M_{1\times n}(\R)$. Then $0<\abs{h}<\delta$ implies 
\begin{align*}
\frac{\abs{f(x+h)-f(x)-Ah}}{\abs{h}}&\leq\epsilon\sqrt{n}\frac{\abs{h}}{\abs{h}}\\
&=\epsilon\sqrt{n}
\end{align*} since $A(h)=\sum_{k=1}^n\partial_{x_k}f(x)h_k$. Since this is true for all $\epsilon$, the condition for differentiability is satisfied and we are done. 
\end{proof}
\end{prp}

Notice that in the above proof, in order to show that $f$ is differentiable at $x$, we need to know the partial derivatives of its neighbours. \\

The below theorem acts as a summary or recollection of the relationships between differentiability and continuous Jacobian Matrices. 

\begin{crl}{}{}\\
Let $U$ be an open subset of $\R^n$. Let $f:U\to\R^m$ be a function. Then $f$ is continuously differentiable if and only if $Jf:U\to M_{m\times n}(\R)$ is continuous. 
\end{crl}

\subsection{Inverse Function Theorem}
The inverse function theorem is a powerful multidimensional analog of finding the derivative of an inverse function. The function has a few conditions to satisfy in order for it to be applied. Since the proof is exceptionally long, I will split it to a number of theorems and prepositions. 
\begin{prp}{}{} Suppose that $\Psi:U\to V$ is a bijection which is differentiable at $x\in U$. Suppose that $\Psi^{-1}$ is differentiable at $y=\Psi(x)\in V$. Then $D\Psi(x)$ and $D\Psi^{-1}(y)$ are both invertible and $$(D\Psi^{-1})(y)=(D\Psi(\Psi^{-1}(y)))^{-1}$$ 
\begin{proof}
Differentiating the relation $\Psi(\Psi^{-1}(y))=y$ gives $$D\Psi(\Psi^{-1}(y))\circ D\Psi^{-1}(y)=I_n$$ which is the identity transformation thus we are done. 
\end{proof}
\end{prp}

\begin{lmm}{}{} $T\in\Hom_\R(\R^n,\R^k)$ is injective if and only if there exists $a>0$ such that $\abs{T(x)}\geq a\abs{x}$ for all $x\in\R^n$. 
\begin{proof}
Firstly suppose that there exists $a>0$ such that $\abs{T(x)}\geq a\abs{x}$ for all $x\in\R^n$. Let $T(x)=T(y)$. Then 
\begin{align*}
\abs{T(x-y)}&\geq a\abs{x-y}\\
\abs{T(x)-T(y)}&\geq a\abs{x-y}\\
0&\geq a\abs{x-y}
\end{align*}
Since $a>0$ we must have $x=y$. \\
Now suppose that for all $a>0$, $\abs{T(x)}<a\abs{x}$ for all $x\in\R^n$. This means that $\frac{\abs{T(x)}}{\abs{x}}<a$ for all $a>0$. This is precisely the definition of a limit. We can find a sequence $(x_j)_{j\in\N}$ such that $$\frac{\abs{T(x_j)}}{\abs{x_j}}\to 0$$ Now consider a new sequence defined by $y_j=\frac{x_j}{\abs{x_j}}$. Clearly $(y_j)_{j\in\N}\subset S^{n-1}$. But $S^{n-1}$ is compact by the Heine-Borel theorem. Thus there exists $(y_{j_m})_{m\in\N}$ such that it has its limit in $S^{n-1}$. But then $$T(y_j)=\frac{T(x_j)}{\abs{x_j}}$$ thus $\abs{T(y_j)}\to 0$. All the $y_j$ are clearly nonzero, this means that $T$ has a non-trivial kernel and thus $T$ is not injective. 
\end{proof}
\end{lmm}

We split the inverse function theorem into its injective and surjective part for better readability. 

\begin{prp}{Injective Part of Inverse Function Theorem}{} Let $U$ be an open subset of $\R^n$ and suppose that $\Psi:\R^n\to\R^m$ such that it is differentiable and its derivative is continuous in $U$. Assume that $D\Psi(p)$ is injective at a point $p\in U$. Then there exists $\delta>0$ such that $B_\delta(p)\subset U$ and such that $f$ is injective on $B_\delta(p)$. 
\begin{proof}
From the above lemma, $D\Psi(p)$ being injective means that there exists $a>0$ such that $$\abs{Df(p)h}\geq\epsilon\abs{h}$$ for all $h\in\R^n$. Since $Df:U\to\Hom_\R(\R^n,\R^k)$ is continuous, there exists $\delta>0$ such that $x\in B_\delta(p)\subset U$ implies $$\|Df(p)-Df(x)\|<\frac{1}{2}\epsilon$$ Define a new function $F:U\to\R^k$ by $F(x)=f(x)-Df(p)x$. Then $F$ is differentiable since $Df(p)x$ is just a linear transformation and we have $$DF(x)=Df(x)-Df(p)$$ Thus we now have $$\|DF(x)\|=\|Df(x)-Df(p)\|<\frac{1}{2}\epsilon$$ We can now apply the mean value inequality to get $$\abs{F(z)-F(x)}\leq\frac{1}{2}\epsilon\abs{z-x}$$ for all $x,z\in B_\delta(p)$. Finally we have 
\begin{align*}
\abs{f(x)-f(z)}&=\abs{Df(p)(x-z)-(F(z)-F(x))}\\
&\geq\epsilon\abs{x-z}-\frac{1}{2}\epsilon\abs{x-z}\\
&=\frac{1}{2}\epsilon\abs{x-z}
\end{align*}
This means that $x\neq z$ implies $f(x)\neq f(z)$ and we are done. 
\end{proof}
\end{prp}

\begin{prp}{Surjective Part of Inverse Function Theorem}{} Let $U$ be an open subset of $\R^n$ and suppose that $\Psi:\R^n\to\R^n$ such that it is differentiable and its derivative is continuous in $U$. Assume that $D\Psi(p)$ is surjective at a point $p\in U$. Then there exists $\rho>0$ such that $B_\rho(\Psi(p))\subset\Psi(U)$. 
\begin{proof}
By the rank nullity theorem, $D\Psi(p)$ is injective (surjectivity implies bijectivity in linear maps). By the above proposition, there exists $\epsilon>0$ such that $$\abs{D\Psi(p)h}\geq\epsilon\abs{h}$$ for all $h\in\R^n$. Again define $F:U\to\R^n$ by $F(x)=\Psi(x)-Df(p)x$. Then exactly the same as the above proof, we must have $\abs{F(x)-F(z)}\leq\frac{1}{2}\epsilon\abs{x-z}$ and $\abs{\Psi(x)-\Psi(z)}\geq\frac{1}{2}\epsilon\abs{x-z}$. Set $$K=\overline{B_{\frac{1}{2}\delta}(p)}=\{x\in\R^n|\abs{x-p}\leq\frac{1}{2}\delta\}$$ and $\partial K=\{x\in\R^n|\abs{x-p}=\frac{1}{2}\delta\}$. 
Now we have 
\begin{align*}
\abs{\Psi(x)-\Psi(p)}&\geq\frac{1}{2}\epsilon\abs{x-p}\\
&=\frac{1}{4}\epsilon\delta
\end{align*}
For all $x\in\partial K$. Set $\rho=\frac{1}{8}\epsilon\delta$ and fix $y\in B_\rho(\Psi(p))$. We will show that $$y\in\Psi(B_{\frac{1}{2}\delta}(p))$$ to finish the proof. Define a new function $\phi:K\to \R$ by $\phi(x)=\abs{\Psi(x)-y}$. Then $\phi$ is continuous by composition of continuous functions. By the extreme value theorem, there exists $c\in K$ such that $\phi(c)\leq\phi(x)$ for all $x\in K$. \\
Now I will show that $c\in B_{\frac{1}{2}\delta}(p)$ by showing that $c\notin\partial K$ and using the fact that $B_{\frac{1}{2}\delta}(p)=K\setminus\partial{K}$. $c\notin\partial K$ can be proved by $\phi(x)>\phi(p)$ for all $x\in\partial K$. Now if $x\in\partial K$, we have
\begin{align*}
\phi(x)&=\abs{\Psi(x)-y}\\
&\geq\abs{\Psi(x)-\Psi(p)}-\abs{y-\Psi(p)}\\
&\geq\frac{1}{4}\epsilon\delta-\frac{1}{8}\epsilon\delta\\
&=\rho\\
&>\abs{y-\Psi(p)}\\
&=\phi(p)
\end{align*}
Thus we are done with this part. \\
Now since $D\Psi(p)$ is surjective, there exists $h\in\R^n$ such that $D\Psi(p)h=y-\Psi(c)$ (by nuisances). Since $c\in B_{\frac{1}{2}\delta}(p)$, there exists $\nu>0$ such that $\abs{t}<\nu$ implies $\abs{c+th-p}<\frac{1}{2}\delta$. Thus we have 
\begin{align*}
\Psi(c+th)-y&=\Psi(c+th)-\Psi(c)-tD\Psi(p)h+(\Psi(c)-y+tD\Psi(p)h)\\
&=F(c+th)-F(c)+(1-t)(\Psi(c)-y)
\end{align*}
for $\abs{t}<\nu$. 
Now, we have that 
\begin{align*}
\phi(c)&\leq\phi(c+th)\\
&=\abs{\Psi(c+th)-y}\\
&\leq\frac{1}{2}t\abs{D\Psi(p)h}+(1-t)\abs{\Psi(c)-y}\\
&=\left(1-\frac{1}{2}t\right)(\Psi(c)-y)
\end{align*}
Since this holds for all $\abs{t}<\nu$, we must have that $\phi(c)=0$, which implies that $y=\Psi(c)$. Thus we have shown that $$B_\rho(\Psi(p))\subset\Psi(B_{\frac{1}{2}\delta}(p))\subset\Psi(U)$$
\end{proof}
\end{prp}

\begin{crl}{}{} Let $U\subset\R^n$ be open. Let $\Psi\in\mathcal{C}^1(U,\R^n)$ and suppose that $D\Psi(p)$ is invertible for all points $p\in U$, then $\Psi$ maps open subsets of $U$ to open subsets of $\R^n$. 
\begin{proof}
Let $V\subset U$ be open. By the above proposition applied to $\Psi|_V$, we have for all $p\in V$, there exists $\rho>0$ such that $$B_{\rho}(\Psi(p))\subset\Psi(V)$$, thus $\Psi(V)$ is open. 
\end{proof}
\end{crl}

\begin{thm}{Inverse Function Theorem}{} Let $U$ be an open subset of $\R^n$ and suppose that $\Psi\in\mathcal{C}^1(U,\R^n)$. Assume that $D\Psi(p)$ is invertible at a point $p\in U$. Let $q=\Psi(p)$. Then there exists a neighbourhood $N_p$ and $N_q$ such that $\Psi:N_p\to N_q$ is a bijection and $\Psi^{-1}:N_q\to N_p$ is continuously differentiable and $$(D\Psi^{-1})(y)=(D\Psi(\Psi^{-1}(y)))^{-1}$$ for all $y\in N_q$. 
\begin{proof}
Clearly $\Psi$ satisfies the above two propositions. Thus this means that there exists $\epsilon>0$ such that $\abs{Df(p)h}\geq\epsilon\abs{h}$ for all $h\in\R^n$ and that there exists $\delta>0$ such that $x\in B_\delta(p)\subset U$ implies $$\|Df(p)-Df(x)\|<\frac{1}{2}\epsilon$$ Using these two facts and the rank nullity theorem we deduce that $D\Psi(x)$ is invertible for all $x\in B_\delta(p)$. The above corollary implies that $\Psi(B_\delta(p))$ is open. Thus $\Psi|_{B_\delta(p)}:B_\delta(p)\to\Psi(B_\delta(p))$ is a bijection. \\
Now we show that $\Psi^{-1}$ is continuously differentiable. 
\end{proof}
\end{thm}

Note that there could be multiple $p$ such that $\Psi$ maps $p$ to $q$. The inverse function theorem guarantees that as long as $D\Psi(p)$ is invertible, then they will have bijective neighbourhoods. This is why we can have multiple branches for the inverse. A good example would be $y=x^2$. It has two inverses, namely $\sqrt{x}$ and $-\sqrt{x}$ precisely because of this reason. In this case the two neighbourhoods would be $\R^+$ and $\R^-$ respectively. \\

Furthermore, even if there are multiple $p$ mapping to $q$, not every $p$ could have a neighbourhood such that they are bijective because we must require the fact that $D\Psi(p)\neq 0$. 

\subsection{Implicit Function Theorem}
\begin{thm}{Implicit Function Theorem}{}\\
Let $U\subseteq\R^{n+l}$ be an open subset. Let $F:U\to\R^l$ be a function of class $\mC^1$. Let $p=(p_1,\dots,p_{n+l})\in U$ be a point. Let $c=F(p)$. Suppose that $$\begin{vmatrix}
\frac{\partial F_1}{\partial x_{n+1}}\bigg{|}_p & \cdots & \frac{\partial F_1}{\partial x_{n+l}}\bigg{|}_p\\
\vdots & \ddots & \vdots\\
\frac{\partial F_m}{\partial x_{n+1}}\bigg{|}_p & \cdots & \frac{\partial F_m}{\partial x_{n+l}}\bigg{|}_p
\end{vmatrix}\neq 0$$
Then there exists an open neighborhood $(p_1,\dots,p_n)\in U_p\subseteq\R^n$ and $g:U_p\to\R^l$ of class $\mC^1$ such that the following are true. 
\begin{itemize}
\item $g(p_1,\dots,p_n)=(p_{n+1},\dots,p_{n+l})$. 
\item $\{(x,g(x))\in\R^{n+l}\;|\;x\in N\}\subseteq U$. 
\item $f(x,g(x))=0$ for all $x\in U_p$. 
\item The Jacobian of $g$ is given by $$J_g(x)=-\begin{pmatrix}
\frac{\partial F_1}{\partial x_{n+1}}\bigg{|}_x & \cdots & \frac{\partial F_1}{\partial x_{n+l}}\bigg{|}_x\\
\vdots & \ddots & \vdots\\
\frac{\partial F_m}{\partial x_{n+1}}\bigg{|}_x & \cdots & \frac{\partial F_m}{\partial x_{n+l}}\bigg{|}_x
\end{pmatrix}^{-1}\begin{pmatrix}
\frac{\partial F_1}{\partial x_1}\bigg{|}_x & \cdots & \frac{\partial F_1}{\partial x_n}\bigg{|}_x\\
\vdots & \ddots & \vdots\\
\frac{\partial F_m}{\partial x_1}\bigg{|}_x & \cdots & \frac{\partial F_m}{\partial x_n}\bigg{|}_x
\end{pmatrix}$$ for all $x\in N$
\end{itemize}
\end{thm}

Beware that more often we seen that the invertible matrix is not necessarily on the right hand side of $\partial F$. It could consist of multiple columns of $\partial F$ simply because of the ordering of the variables which is completely by the writer's choice. Moreover, there may be more than one $g$ to convert the implicit function into an explicit one if you consider different variables for the domain and the codomain. 

\subsection{Higher Order Derivatives}
\begin{defn}{$k$-times Continuously Differentiable Functions}{}\\
Let $U\subseteq\R^n$ and $V\subseteq\R^m$ be open subsets. Let $f:U\to V$ be a function. We say that $f$ is of class $\mC^k$ if for $1\leq i\leq k-1$, the map $D^if:U\to\R^{mn^i}$ is differentiable and $D^kf:U\to\R^{mn^k}$ is continuous. 
\end{defn}

\begin{defn}{The Set of $k$-times Continuously Differentiable Functions}{}\\
Let $U\subseteq\R^n$ and $V\subseteq\R^m$ be open subsets. Define $$\mC^k(U,V)=\{f:U\to V\;|\;f\text{ is of class }\mC^k\}$$
\end{defn}

From here onwards, our function $f:\R^n\to\R$ will be a scalar function else the Hessian Matrix cannot be defined. Do note that general second order differential operators for $f:\R^n\to\R^m$ do exists. 

Let $f:U\subseteq\R^n\to\R^m$ be a function. If it is differentiable, then the partial derivatives are functions $\partial_if:U\to\R^m$ given by $$\partial_if(x)=\begin{pmatrix}
\partial_if_1(x)\\\vdots\\\partial f_m(x)
\end{pmatrix}$$

\begin{defn}{Higher Order Partial Derivatives}{}\\
Let $U\subseteq\R^n$ be an open subset. Let $f:U\to\R^m$ be a function. Define the partial derivatives of order $k$ inductively by $$\partial_{j_k\cdots j_1}f(x)=\partial_{j_k}(\partial_{j_{k-1}}\cdots\partial_{j_1})f(x)$$
\end{defn}

\begin{defn}{Hessian Matrix}{}\\
Let $U\subseteq\R^n$ be an open subset. Let $f:U\to\R$ be a function. Suppose that all second order partial derivatives of $f$ exists. Define the Hessian matrix of $f$ to be $$Hf(x)=\begin{pmatrix}
\partial_{11}f(x) & \cdots & \partial_{1n}f(x)\\
\vdots & \ddots & \vdots\\
\partial_{n1}f(x) & \cdots & \partial_{nn}f(x)
\end{pmatrix}$$
\end{defn}

\begin{prp}{}{}\\
Let $U\subseteq\R^n$ be an open subset. Let $f:U\to\R$ be a function of class $\mC^2$. Then the following are true. 
\begin{itemize}
\item $D(Df)(x)=Hf(x)$ for all $x\in U$. 
\item $Hf(x)$ is a symmetric matrix for all $x\in U$. 
\item $\partial_{ij}f(x)=\partial_{ji}f(x)$ for all $x\in U$. 
\end{itemize}
\end{prp}

\begin{prp}{}{}\\
Let $U\subseteq\R^n$ be an open subset. Let $f:U\to\R^m$ be a function of class $\mC^k$. Then the following are true. Let $1\leq j_1,\dots,j_m\leq n$ be integers. Let $\sigma\in\text{Sym}(\{j_1,\dots,j_m\})$ be a permutation. Then we have $$\partial_{j_k\cdots j_1}f(x)=\partial_{\sigma(j_k)\cdots\sigma(j_1)}f(x)$$
\end{prp}

\subsection{Jordan Measure}
\begin{defn}{Rectangles}{}\\
A rectangle in $\R^n$ is a set $R\subseteq\R^n$ of the form $$R=\prod_{j=1}^nI_j$$ where each $I_j\subseteq\R$ is an interval. 
\end{defn}

\begin{defn}{Volume of a Rectangle}{}\\
Let $R=I_1\times\cdots\times I_n\subseteq\R^n$ be a rectangle. Suppose that the endpoints of $I_j$ is given by $a_j,b_j\in\R$ for $a_j\leq b_j$. Define the volume of $R$ to be $$\text{Vol}(R)=\prod_{i=1}^n(b_i-a_i)$$
\end{defn}

\begin{defn}{Elementary Sets}{}\\
Let $S\subseteq\R^n$ be a subset. We say that $S$ is elementary if there exists rectangles $R_1,\dots,R_k\subseteq\R^n$ such that $$S=\bigcup_{i=1}^kR_i$$
\end{defn}

\begin{lmm}{}{}\\
Let $S,T\subseteq\R^n$ be elementary set. Then the following are true. 
\begin{itemize}
\item $S\cup T$ is an elementary set. 
\item $S\cap T$ is an elementary set. 
\item $S\setminus T$ is an elementary set. 
\end{itemize}
\end{lmm}

\begin{prp}{}{}\\
Let $S\subseteq\R^n$ be an elementary subset. Then the following are true. 
\begin{itemize}
\item There exists rectangles $R_1,\dots,R_k\subseteq\R^n$ such that $S=\bigcup_{i=1}^kR_i$ and $R_i\cap R_j=\emptyset$ for $1\leq i\neq j\leq k$. 
\item Suppose that $R_1,\dots,R_k\subseteq\R^n$ and $Q_1,\dots,Q_l\subseteq\R^n$ are rectangles such that $$S=\bigcup_{i=1}^kR_i=\bigcup_{i=1}^lQ_i$$ Then we have $$\sum_{i=1}^k\text{Vol}(R_i)=\sum_{i=1}^l\text{Vol}(Q_i)$$
\end{itemize}
\end{prp}

\begin{defn}{Volume of an Elementary Set}{}\\
Let $S\subseteq\R^n$ be an elementary subset. Let $R_1,\dots,R_k\subseteq\R^n$ be rectangles such that $S=\bigcup_{i=1}^kR_i$ and $R_i\cap R_j=\emptyset$ for $1\leq i\neq j\leq k$. Define the volume of $S$ to be $$\text{Vol}(S)=\sum_{i=1}^k\text{Vol}(R_i)$$
\end{defn}

\begin{defn}{Jordan Inner and Outer Measure}{}\\
Let $S\subseteq\R^n$ be a bounded subset. 
\begin{itemize}
\item Define the Jordan inner measure of $S$ to be $$\mu_\ast(S)=\sup\{\text{Vol}(E)\;|\;E\subseteq S\text{ and }E\text{ is an elementary set }\}$$
\item Define the Jordan outer measure of $S$ to be $$\mu^\ast(S)=\inf\{\text{Vol}(E)\;|\;S\subseteq E\text{ and }E\text{ is an elementary set }\}$$
\end{itemize}
\end{defn}

\begin{defn}{Jordan Measure}{}\\
Let $S\subseteq\R^n$ be a bounded subset. We say that $S$ is Jordan measurable if $\mu_\ast(S)=\mu^\ast(S)$. In this case, we say that the Jordan measure of $S$ is $$\mu(S)=\mu_\ast(S)=\mu^\ast(S)$$
\end{defn}

\begin{prp}{}{}\\
Let $S\subseteq\R^n$ be a bounded subset. Then $S$ is Jordan measurable if and only if for all $\varepsilon>0$, there exists elementary sets $A,B$ such that $A\subseteq S\subseteq B$ and $\text{Vol}(B\setminus A)<\varepsilon$. 
\end{prp}

\begin{defn}{Set of Jordan Measurable Sets}{}\\
Define the set of Jordan measurable sets in $\R^n$ to be $$\mJ(\R^n)=\{S\subseteq\R^n\;|\;S\text{ is Jordan measurable }\}$$
\end{defn}

\begin{prp}{}{}\\
The following are true regarding the Jordan measure $\mu:\mJ(\R^n)\to[0,\infty)$. 
\begin{itemize}
\item $\mu(\emptyset)=0$. 
\item If $S,T\in\mJ(\R^n)$, then $S\cup T,S\cap T\in\mJ(\R^n)$ and $$\mu(S\cup T)=\mu(S)+\mu(T)-\mu(S\cap T)$$
\item If $S,T\in\mJ(\R^n)$ and $S\subseteq T$, then $$\mu(S)\leq\mu(T)$$
\item If $S\in\mJ(\R^n)$ and $T\in\mJ(\R^m)$, then $S\times T\in\mJ(\R^{n+m})$ and $$\mu(S\times T)=\mu(S)\cdot\mu(T)$$
\end{itemize}
\end{prp}

\begin{defn}{The $k$th Inner and Outer Volumes of a Subset}{}\\
Let $S\subseteq\R^n$ be a bounded subset. 
\begin{itemize}
\item Define the $k$th inner volume of $S$ to be $$V_k^-(S)=\frac{1}{2^{nk}}\abs{\left\{(t_1,\dots,t_n)\in\Z^n\;\bigg{|}\;\left[\frac{t_1}{2^k},\frac{t_1+1}{2^k}\right)\times\cdots\times\left[\frac{t_n}{2^k},\frac{t_n+1}{2^k}\right)\subseteq S\right\}}$$
\item Define the $k$th outer volume of $S$ to be $$V_k^+(S)=\frac{1}{2^{nk}}\abs{\left\{(t_1,\dots,t_n)\in\Z^n\;\bigg{|}\;\left[\frac{t_1}{2^k},\frac{t_1+1}{2^k}\right)\times\cdots\times\left[\frac{t_n}{2^k},\frac{t_n+1}{2^k}\right)\cap S\neq\emptyset\right\}}$$
\end{itemize}
\end{defn}

\begin{prp}{}{}\\
Let $S\subseteq\R^n$ be a bounded subset. Then $S$ is Jordan measurable if and only if $(V_k^+(S)-V_k^-(S))_{k\in\N^+}$ tends to $0$. In this case, we have $$\text{Vol}(S)=\lim_{k\to\infty}V_k^-(S)=\lim_{k\to\infty}V_k^+(S)$$
\end{prp}

\subsection{Riemann Integral}
\begin{defn}{Subdivision}{}\\
Let $S\subseteq\R^n$ be a Jordan measurable set. A subdivision of $S$ consists of finitely many Jordan measurable sets $P=\{F_1,\dots,F_k\subseteq\R^n\}$ such that the following are true. 
\begin{itemize}
\item $S=\bigcup_{i=1}^kF_i$. 
\item $F_i^\circ\cap F_j^\circ=\emptyset$ for any $i\neq j$. 
\end{itemize}
\end{defn}

\begin{defn}{Refinements}{}\\
Let $S\subseteq\R^n$ be a Jordan measurable set. Let $P,Q$ be subdivisions of $S$. We say that $Q$ is a refinement of $P$ if $P\subseteq Q$. 
\end{defn}

\begin{prp}{}{}\\
Let $S\subseteq\R^n$ be a Jordan measurable set. Let $P,Q$ be subdivisions of $S$. Then there exists a subdivision $R$ of $S$ such that $R$ is a refinement of both $P$ and $Q$. 
\end{prp}

\begin{defn}{Lower and Upper Sums}{}\\
Let $S\subseteq\R^n$ be a Jordan measurable set. Let $f:S\to\R$ be a bounded function. Let $P$ be a subdivision of $S$. 
\begin{itemize}
\item Define the lower sum of $f$ with respect to $P$ to be $$L(f,P)=\sum_{F\in P}\inf_F(f)\cdot \mu(F)$$
\item Define the upper sum of $f$ with respect to $P$ to be $$U(f,P)=\sum_{F\in P}\sup_F(f)\cdot \mu(F)$$
\end{itemize}
\end{defn}

\begin{prp}{}{}\\
Let $S\subseteq\R^n$ be a Jordan measurable set. Let $f:S\to\R$ be a bounded function. Then the following are true. 
\begin{itemize}
\item If $m\leq f(x)\leq M$ for all $x\in S$, then we have $$m\cdot\mu(S)\leq L(f,P)\leq U(f,P)\leq M\cdot\mu(S)$$
\item If $P,Q$ are subdivisions of $S$, then we have $$L(f,P)\leq U(f,Q)$$
\end{itemize}
\end{prp}

\begin{defn}{Lower and Upper Integrals}{}\\
Let $S\subseteq\R^n$ be a Jordan measurable set. Let $f:S\to\R$ be a bounded function. 
\begin{itemize}
\item Define the lower integral of $f$ at $S$ to be $$\underline{\int}_Sf=\sup\{L(f,P)\;|\;P\text{ is a subdivision of }S\}$$
\item Define the upper integral of $f$ at $S$ to be $$\overline{\int}_Sf=\inf\{U(f,P)\;|\;P\text{ is a subdivision of }S\}$$
\end{itemize}
\end{defn}

\begin{defn}{Integrable Functions}{}\\
Let $S\subseteq\R^n$ be a Jordan measurable set. Let $f:S\to\R$ be a bounded function. We say that $f$ is integrable at $S$ if $$\underline{\int}_Sf=\overline{\int}_Sf$$ In this case, we say that the integral of $f$ at $S$ is the value $$\int_Sf=\underline{\int}_Sf=\overline{\int}_Sf$$
\end{defn}

\begin{prp}{}{}\\
Let $S\subseteq\R^n$ be a Jordan measurable set. Let $f:S\to\R$ be a bounded function. Then $f$ is integrable if and only if for all $\varepsilon>0$, there exists a subdivision $P$ of $S$ such that $$U(f,P)-L(f,P)<\varepsilon$$
\end{prp}

\begin{lmm}{}{}\\
Let $S\subseteq\R^n$ be a Jordan measurable set. Let $f:S\to\R$ be an integrable function. If $m\leq f(x)\leq M$ for all $x\in S$, then we have $$m\cdot\mu(S)\leq\int_f\leq M\cdot\mu(S)$$
\end{lmm}

\begin{lmm}{}{}\\
Let $S\subseteq\R^n$ be a Jordan measurable set. Let $f,g:S\to\R$ be integrable functions. Then the following are true. 
\begin{itemize}
\item For any $\lambda,\mu\in\R$, then $\lambda f+\mu g$ is integrable and $$\int_S\lambda f+\mu g=\lambda\int_Sf+\mu\int_Sg$$
\item If $f\leq g$, then we have $$\int_Sf\leq\int_Sg$$
\end{itemize}
\end{lmm}

\begin{prp}{}{}\\
Let $S,T\subseteq\R^n$ be Jordan measurable sets. Let $f:S\cup T\to\R$ be an integrable function. Then we have $$\int_{S\cup T}f=\int_Sf+\int_Tf-\int_{S\cap T}f$$
\end{prp}

\begin{prp}{}{}\\
Let $S\subseteq\R^n$ be a Jordan measurable set. Let $f:S\to\R$ be a bounded function. Then the following are true. 
\begin{itemize}
\item If $f$ is uniformly continuous, then $f$ is integrable. 
\item If $f$ is integrable, then $\abs{f}$ is integrable and we have $$\abs{\int_Sf}\leq\int_S\abs{f}$$
\end{itemize}
\end{prp}

\begin{prp}{}{}\\
Let $S\subseteq\R^n$ be a bounded subset. Then $S$ is Jordan measurable if and only if the indicator function $1_S:S\to\{0,1\}$ is integrable. In this case, we have $$\int_S1_S=\mu(S)$$
\end{prp}

\begin{prp}{}{}\\
Let $S\subseteq\R^n$ be a Jordan measurable set. Let $R\subseteq\R^n$ be a rectangle such that $S\subseteq R$. Let $f|^R:R\to\R$ be the function defined by $$f|^R(x)=\begin{cases}
f(x) & \text{ if }x\in S\\
0 & \text{ otherwise }
\end{cases}$$ Then we have $$\int_Sf=\int_Rf|^R$$
\end{prp}

\begin{thm}{Fubini's Theorem}{}\\
Let $S\subseteq\R^n$ and $T\subseteq\R^m$ be Jordan measurable sets. Let $f:S\times T\to\R$ be a bounded function. Suppose that $f(x,-):T\to\R$ is integrable for all $x\in S$. Then the function $x\mapsto\int_T f(x,-)$ is integrable, and we have $$\int_{S\times T}f=\int_S\int_Tf$$
\end{thm}

\begin{thm}{Change of Variables}{}\\
Let $S,T\subseteq\R^n$ be Jordan measurable sets. Let $f:S\to\R$ be a function. Let $g:T\to S$ be a bijective function of class $\mC^1$. Then $f$ is integrable if and only if $f\circ g$ is integrable. In this case, we have $$\int_Sf=\int_Tf\circ g\;\abs{Jg}$$
\end{thm}

\section{Differential Operators}
\subsection{Gradient}
\begin{defn}{Gradient}{}\\
Let $U\subseteq\R^n$ be an open subset. Let $f:U\to\R$ be a function of class $\mC^1$. Define the gradient of $f$ to be the function $\text{grad}(f):U\to\R^n$ given by $$\text{grad}(f)(p)=\begin{pmatrix}
\frac{\partial f}{\partial x_1}\bigg{|}_p\\\vdots\\\frac{\partial f}{\partial x_n}\bigg{|}_p
\end{pmatrix}=\begin{pmatrix}
\partial_{e_1}f(p)\\\vdots\\\partial_{e_n}f(p)
\end{pmatrix}$$ where $e_1,\dots,e_n\in\R^n$ is the standard basis vectors of $\R^n$. 
\end{defn}

Note: Implicit in our formula for $\text{grad}(f)$ is the choice of standard basis for $\R^n$. As we all things in linear algebra, the representation of a vector depends on the chosen basis. 

\begin{crl}{}{}\\
Let $U\subseteq\R^n$ be an open subset. Let $f:U\to\R$ be a function of class $\mC^1$. Then $\text{grad}(f):U\to\R^n$ is continuous. 
\end{crl}

Note: For a $\mC^1$-function $f:\R^n\to\R$, we have that $\text{grad}(f)(q)$ is orthogonal to $H=\{p\in\R^n\;|\;f(p)=0\}$ at the point $q\in H$. 

\begin{prp}{}{}\\
Let $U\subseteq\R^n$ be an open subset. Let $f,g:U\to\R$ be functions of class $\mC^1$. Then the following are true. 
\begin{itemize}
\item $\text{grad}(\lambda f+\mu g)=\lambda\text{grad}(f)+\mu\text{grad}(g)$ for any $\lambda,\mu\in\R$. 
\item $\text{grad}(fg)=f\text{grad}(g)+\text{grad}(f)g$. 
\item If $g\neq 0$, then we have $\text{grad}(f/g)=\frac{\text{grad}(f)g-f\text{grad}(g)}{g^2}$
\end{itemize}
\end{prp}

\subsection{Divergence}
\begin{defn}{Divergence}{}\\
Let $U\subseteq\R^n$ be an open subset. Let $F=(F_1,\dots,F_n):U\to\R^n$ be a function of class $\mC^1$. Define the divergence of $F$ to be the function $\text{div}(F):U\to\R$ given by $$\text{div}(F)(p)=\sum_{i=1}^n\frac{\partial f_i}{\partial x_i}\bigg{|}_p$$
\end{defn}

\begin{prp}{}{}\\
Let $U\subseteq\R^n$ be an open subset. Then the following are true regarding the divergence operator. 
\begin{itemize}
\item $\text{div}(\lambda F+\mu G)=\lambda\text{div}(F)+\mu\text{div}(G)$ for all $\mC^1$-functions $F,G:U\to\R^n$ and all $\lambda,\mu\in\R$. 
\item $\text{div}(hF)=h\text{div}(F)+\text{grad}(f)\cdot F$ for all $\mC^1$-functions $F:U\to\R^n$ and $h:U\to\R$. 
\end{itemize}
\end{prp}

Denote the set of all vector fields by $$\mathfrak{X}^k(U)=\{F:U\to\R^n\;|\;F\text{ is of class }\mC^k\}$$ We have defined operators $\text{grad}:\mC^1(U)\to\mathfrak{X}^0(U)$ and $\text{div}:\mathfrak{X}^1(U)\to\mC^0$. We provide a link between grad and div by establishing an operator $\mathfrak{X}^1(U)\to\mathfrak{X}^0(U)$. 

\subsection{Laplacian}
\begin{defn}{Laplacian}{}\\
Let $U\subseteq\R^n$ be an open subset. Let $f:U\to\R$ be a function of class $\mC^2$. Define the Laplacian of $f$ to be the function $\Delta:U\to\R$ given by $$\Delta(f)(p)=\text{div}(\text{grad}(f))(p)$$
\end{defn}

\begin{prp}{}{}\\
Let $U\subseteq\R^n$ be an open subset. Let $f,g:U\to\R$ be functions of class $\mC^2$. Then we have $$\Delta(\lambda f+\mu g)=\lambda\Delta(f)+\mu\Delta(g)$$ for all $\lambda,\mu\in\R$. 
\end{prp}

\end{defn}

\begin{prp}{}{}\\
Let $U\subseteq\R^n$ be an open subset. Let $f:U\to\R$ be a harmonic function. Then $f$ is analytic. 
\end{prp}

\subsection{Curl}
\begin{defn}{Curl}{}\\
Let $U\subseteq\R^3$ be an open subset. Let $F=(F_1,F_2,F_3):U\to\R^3$ be a $\mC^1$-function. Define the curl of $F$ to be the function $\text{curl}(F):U\to\R^3$ given by $$\text{curl}(F)(p)=\begin{pmatrix}
\frac{\partial F_3}{\partial x_2}\bigg{|}_p-\frac{\partial F_2}{\partial x_3}\bigg{|}_p\\
\frac{\partial F_1}{\partial x_3}\bigg{|}_p-\frac{\partial F_3}{\partial x_1}\bigg{|}_p\\
\frac{\partial F_2}{\partial x_1}\bigg{|}_p-\frac{\partial F_1}{\partial x_2}\bigg{|}_p
\end{pmatrix}$$
\end{defn}

\begin{prp}{}{}\\
Let $U\subseteq\R^n$ be an open subset. Then the following are true regarding the curl operator. 
\begin{itemize}
\item $\text{curl}(\lambda F+\mu G)=\lambda\text{curl}(F)+\mu\text{curl}(G)$ for all $\mC^1$-functions $F,G:U\to\R^n$ and all $\lambda,\mu\in\R$. 
\item $\text{curl}(hF)=h\text{curl}(F)+\text{grad}(f)\times F$ for all $\mC^1$-functions $F:U\to\R^n$ and $h:U\to\R$. 
\end{itemize}
\end{prp}

We relate the three operators by the following proposition. 

\begin{prp}{}{}\\
Let $U\subseteq\R^3$ be an open subset. Then the following are true regarding the three differential operators. 
\begin{itemize}
\item $\text{div}(F\times G)=\text{curl}(F)\cdot G-F\cdot\text{curl}(G)$ for all $\mC^1$-functions $F,G:U\to\R^3$. 
\item $\text{curl}(\text{grad}(f))=0$ for all $\mC^2$-functions $f:U\to\R$. 
\item $\text{div}(\text{curl}(F))=0$ for all $\mC^2$-functions $f:U\to\R^3$. 
\end{itemize}
\end{prp}

Thus when our functions are sufficiently continuously differentiable, we obtain a diagram\\~\\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	{\mC^\infty(U)} & {\mathfrak{X}^\infty(U)} & {\mathfrak{X}^\infty(U)} & {\mC^\infty(U)}
	\arrow["{\text{grad}}", from=1-1, to=1-2]
	\arrow["{\text{curl}}", from=1-2, to=1-3]
	\arrow["{\text{div}}", from=1-3, to=1-4]
\end{tikzcd}}\\
in which any two consecutive operators is $0$. 

\begin{defn}{The Curl in $\R^2$}{}\\
Let $U\subseteq\R^2$ be an open subset. Let $F:U\to\R^2$ be a function of class $\mC^1$. Define the curl of $F$ to be $$\text{curl}(F)(p)=\frac{\partial F_2}{\partial x_1}\bigg{|}_p-\frac{\partial F_1}{\partial x_2}\bigg{|}_p$$
\end{defn}

\section{Line Integrals}
\subsection{Regular Paths and the Arc Length Parametrization}
\begin{defn}{Regular Paths}{}\\
Let $r:[a,b]\to\R^n$ be a path. We say that $r$ is regular if $r'(t)\neq 0$ for all $t\in(a,b)$. 
\end{defn}

\begin{defn}{The Arc Length Function}{}\\
Let $r:[a,b]\to\R^n$ be a path. Define the arc length function of $r$ to be the function $s:[a,b]\to\R$ given by $$s(t)=\int_a^t\abs{r'(u)}\;du$$
\end{defn}

\begin{thm}{The Reparametrization Theorem}{}\\
Let $r:[a,b]\to\R^n$ be a regular path. Then there exists a reparametrization $s:[c,d]\to\R^n$ of $r$ such that $\abs{s'(t)}=1$ for all $t\in[c,d]$. 
\end{thm}

\begin{defn}{Tangent and Normal of a Regular Path}{}\\
Let $r:[a,b]\to\R^n$ be a regular path of class $\mC^2$. 
\begin{itemize}
\item Define the unit tangent vector of $r$ at $t\in(a,b)$ to be $$T_r(t)=\frac{r'(t)}{\abs{r'(t)}}$$
\item Define the unit normal vector of $r$ at $t\in(a,b)$ to be $$n_r(t)=\frac{T_r'(t)}{\abs{T_r'(t)}}$$
\end{itemize}
\end{defn}

Note: The normal always points towards the curve (it does not always point outwards). 

\begin{prp}{}{}\\
Let $r:[a,b]\to\R^n$ be a regular path of class $\mC^2$. Let $s:[a,b]\to\R^n$ be a orientation preserving regular reparametrization of $r$. Then the following are true. 
\begin{itemize}
\item $T_r(t)=T_s(t)$. 
\item $n_r(t)=n_s(t)$. 
\end{itemize}
\end{prp}

\subsection{Line Integrals}
\begin{defn}{Line Integrals over a $\mC^1$ Path}{}\\
Let $U\subseteq\R^n$ be an open subset. Let $F:U\to\R^n$ be a function. Let $r:[a,b]\to U$ be a $\mC^1$ path. Let $C=\im(r)$. Define the integral of $F$ along $r$ to be $$\int_CF\cdot dr=\int_{[a,b]}(F\circ r)\cdot Jr$$ if the integral exists. 
\end{defn}

\begin{defn}{Line Integrals over a Piecewise $\mC^1$ Path}{}\\
Let $U\subseteq\R^n$ be an open subset. Let $F:U\to\R^n$ be a function. Let $r:[a,b]\to U$ be a piecewise $\mC^1$ path whose pieces are given by $r_i:[a_i,a_{i+1}]$ for $1\leq i\leq n$. Let $C_i=\im(r_i)$. Define the line integral of $F$ along $r$ to be $$\int_CF\cdot dr=\sum_{i=1}^n\int_{C_i}F\cdot dr_i$$
\end{defn}

\subsection{The Fundamental Theorem of Line Integrals}
\begin{thm}{The Fundamental Theorem of Line Integrals}{}
Let $U\subseteq\R^n$ be an open subset. Let $f:U\to\R$ be a differentiable function. Let $r:[a,b]\to U$ be a differentiable function in $U$. Then we have $$\int_C\text{grad}(f)\cdot dr=f(r(b))-f(r(a))$$
\end{thm}

\begin{defn}{Conservative Vector Fields}{}\\
Let $U\subseteq\R^n$ be an open subset. Let $F:U\to\R^n$ be a function. We say that $F$ is conservative if there exists a differentiable function $f:U\to\R$ such that $F=\text{grad}(f)$. 
\end{defn}

\begin{prp}{}{}\\
Let $U\subseteq\R^n$ be an open subset. Let $F:U\to\R^n$ be a vector field. Then the following are equivalent. 
\begin{itemize}
\item $F$ is conservative. 
\item If $r,s:[a,b]\to U$ are differentiable paths in $U$ such that $r(a)=s(a)$ and $r(b)=s(b)$, then we have $$\int_{r([a,b])}F\cdot dr=\int_{s([a,b])}F\cdot ds$$
\item If $r:[a,b]\to U$ is a differentiable path in $U$ such that $r(a)=r(b)$, then we have $$\int_{r([a,b])}F\cdot dr=0$$
\end{itemize}
\end{prp}

\subsection{Green's Theorem}
\begin{defn}{Positively Oriented Regular Curves}{}\\
Let $r:[a,b]\to\R^2$ be a simple closed regular smooth curve. Let $D\subseteq\R^2$ be a bounded subset such that $\partial D=\im(r)$. We say that $r$ is positively oriented with respect to $D$ if the following are true. For all $t\in(a,b)$, there exists $M>0$ such that $$s\begin{pmatrix}
0 & -1\\
1 & 0
\end{pmatrix}r'(t)+r(t)\in D$$ for all $0\leq s<M$. 
\end{defn}

\begin{defn}{Positively Oriented Curves}{}\\
Let $r:[a,b]\to\R^2$ be a simple closed regular smooth curve. Let $D\subseteq\R^2$ be a bounded subset such that $\partial D=\im(r)$. We say that $r$ is positively oriented with respect to $D$ if there exists an orientation preserving reparametrization $s$ of $r$ such that $s$ is regular and $s$ is regular and positively oriented with respect to $D$. 
\end{defn}

\begin{thm}{Green's Theorem on the Plane}{}\\
Let $r:[a,b]\to\R^2$ be a piecewise-smooth simple closed curve whoses pieces are given by $r_1,\dots,r_k$. Let $D\subseteq\R^2$ be a bounded subset such that $\partial D=\im(r)$. Suppose that $r$ is positively oriented with respect to $D$. Let $U\subseteq\R^2$ be an open subset such that $D\subseteq U$. Let $F:U\to\R^2$ be a $\mC^1$-function. Then we have $$\iint_D\text{curl}(F)=\sum_{i=1}^k\int_{\im(r_i)}F\cdot dr_i$$
\end{thm}

By the Jordan Curve Theorem (proved in Algebraic Topology 2), every simple closed curve in $\R^2$ can be oriented in such a way.

\section{Surface Integrals}
\subsection{Parametric Surfaces}
\begin{defn}{Parametric Surface}{}\\
Let $S\subseteq\R^n$ be a subset. Let $r:D\to S$ be a surjective function of class $\mC^1$. We say that $S$ is a parametric surface parametrized by $r$ if the following are true. 
\begin{itemize}
\item There exists an open subset $D\subseteq U\subseteq\R^2$ and a function $f:U\to\R^n$ of class $\mC^1$ such that $f|_D=r$. 
\item $\partial_{e_1}r(p)$ and $\partial_{e_2}r(p)$ are linearly independent for all $p\in D^\circ$. 
\item $r|_{D^\circ}$ is injective. 
\end{itemize}
In this case we call $S$ a parametric surface. 
\end{defn}

Note: The general definition of a surface is that locally $S\subseteq\R^n$ looks like $\R^2$ (this appears in Introduction to Manifolds). In particular, there need not be a global parametrization of the surface. Therefore we are only considering a special kind of surfaces, although it can easily be generalized to any type of compact manifold. 

\begin{defn}{The Boundary of a Parametric Surface}{}\\
Let $S\subseteq\R^n$ be a parametric surface parametrized by $r:D\to S$. Define the boundary of $S$ to be $$\partial S=\left\{p\in S\;\bigg{|}\;\substack{\text{ there exists }f:U\cap S\to V\subseteq\R^2\text{ such that }p\in U\subseteq\R^n\text{ is open }\\f,f^{-1}\text{ are of class }\mC^1,f(p)=(0,0), f(U\cap S)\subseteq H^2}\right\}$$
\end{defn}

\begin{lmm}{}{}\\
Boundary of parametric surface is a curve
\end{lmm}

\begin{defn}{The Outward Pointing Unit Normal Vector}{}\\
Let $S\subseteq\R^3$ be a parametric surface parametrized by $r=(r_1,r_2,r_3):D\to S$. Let $p\in S$ be given by $p=r(u,v)$ for some $(u,v)\in D$. Define the outward pointing unit normal vector of $S$ at $p$ to be the unique vector $n_+(p)\in\R^3$ such that the following are true. 
\begin{itemize}
\item $\begin{pmatrix}\det(J(r_2,r_3))\\\det(J(r_3,r_1))\\\det(J(r_1,r_2))\end{pmatrix}\cdot n_+(p)=0$. 
\item There exists $M>0$ such that $p-\lambda n_+(p)\in S$ for all $0<\lambda<M$. 
\end{itemize}
\end{defn}

\begin{lmm}{}{}\\
Let $S\subseteq\R^3$ be a parametric surface parametrized by $r:D\to S$. Let $(u,v)\in D$. Then we have $$n_+(r(u,v))=\pm\frac{\partial_{e_1}r(u,v)\times\partial_{e_2}r(u,v)}{\abs{\partial_{e_1}r(u,v)\times\partial_{e_2}r(u,v)}}$$
\end{lmm}

\begin{defn}{Orientation}{}\\
Let $S\subseteq\R^3$ be a parametric surface parametrized by $r:D\to S$. We say that $S$ is orientable if the function $n_+:S\to\R^3$ is continuous. 
\end{defn}

Let $S\subseteq\R^3$ be a parametric surface parametrized by $r:D\to S$. Our definition of parametric surfaces imply that the map $$(s,t)\mapsto\frac{\partial_{e_1}r(u,v)\times\partial_{e_2}r(u,v)}{\abs{\partial_{e_1}r(u,v)\times\partial_{e_2}r(u,v)}}$$ This does not mean that every parametric surface is orientable because the definition requires that the domain of the outward normal function is $S$, not $D$. 

\begin{defn}{Surface Integrals}{}\\
Let $S\subseteq\R^3$ be a parametric surface parametrized by $r:D\to S$. Let $U\subseteq\R^3$ be an open subset such that $S\subseteq U$. Let $F:U\to\R^3$ be a function of class $\mC^1$. Define the surface integral of $F$ along $S$ to be $$\iint_S F\cdot dS=\iint_D(F\circ r)\cdot n_+$$
\end{defn}

Note the similarity with the line integral. The difference being that the tangent line $r'(t)$ in the line integral plays the role of $n_+$ in the surface integral.

\subsection{Divergence Theorem}
\begin{thm}{Divergence Theorem}{}\\
Let $\Omega\subseteq\R^3$ be a closed subset such that $\partial\Omega$ is a piecewise smooth parametric surface. Let $U\subseteq\R^3$ be an open subset such that $V\subseteq U$. Let $F:U\to\R^3$ be a function of class $\mC^1$. Then we have $$\iiint_\Omega\text{div}(F)=	\iint_{\partial\Omega}F\cdot d(\partial\Omega)$$
\end{thm}
