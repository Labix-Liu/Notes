\begin{defn}{Holomorphic Functions of Several Variables}{}\\
Let $U\subseteq\C^n$ be open. Let $f=(f_1,\dots,f_n):U\to\C$ be a function. Let $p\in U$. We say that $f$ is holomorphic at $p$ if the Cauchy-Riemann condition is satisfied: $$\frac{\partial f}{\partial\overline{z}_k}=\frac{1}{2}\left(\frac{\partial f}{\partial x_k}-\frac{1}{i}\frac{\partial f}{\partial y_k}\right)=0$$ for $1\leq k\leq n$. 
\end{defn}

\begin{defn}{Polydiscs}{}\\
Let $(a_1,\dots,a_n)\in\C^n$. Let $r_i>0$ for $1\leq i\leq n$. Define the polydisc at $(a_1,\dots,a_n)$ of polyradii $(r_1,\dots,r_n)$ to be $$\Delta_{(a_1,\dots,a_n)}(r_1,\dots,r_n)=\{(z_1,\dots,z_n)\in\C^n\;|\;\abs{z_i-a_i}<r_i\text{ for }1\leq i\leq n\}$$
\end{defn}

\begin{defn}{The Boundary Torus}{}\\
Let $(a_1,\dots,a_n)\in\C^n$. Let $r_i>0$ for $1\leq i\leq n$. Define the boundary torus of the polydisc $\Delta_{(a_1,\dots,a_n)}(r_1,\dots,r_n)$ to be $$T_{(a_1,\dots,a_n)}(r_1,\dots,r_n)=\{(z_1,\dots,z_n)\in\C^n\;|\;\abs{z_i-a_i}<r_i\text{ for }1\leq i\leq n\}$$
\end{defn}

The boundary torus is not quite the boundary of the polydisc: it is strictly smaller. 

\begin{thm}{Cauchy's Integral Formula}{}\\
Let $U\subseteq\C^n$ be open. Let $f:U\to\C$ be a holomorphic function. Let $\Delta_{(a_1,\dots,a_n)}(r_1,\dots,r_n)\subseteq U$. Then we have $$f(z_1,\dots,z_n)=\frac{1}{(2\pi i)^n}\int_{T_{(a_1,\dots,a_n)}(r_1,\dots,r_n)}\frac{f(w_1,\dots,w_n)}{(w_1-z_1)\cdots(w_n-z_n)}\;dw_1\cdots dw_n$$ for all $(z_1,\dots,z_n)\in\Delta_{(a_1,\dots,a_n)}(r_1,\dots,r_n)$. 
\end{thm}
