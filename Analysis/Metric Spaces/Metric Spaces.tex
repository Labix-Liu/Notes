\section{Metric Spaces}
\subsection{Basic Definitions}
\begin{defn}{Metric}{}\\
Let $X$ be a set. A metric on $X$ is a function $d:X\times X\to\R$ such that the following are true. 
\begin{itemize}
\item For all $x,y\in X$, we have $d(x,y)\geq 0$, and $d(x,y)=0$ if and only if $x=y$. 
\item $d(x,y)=d(y,x)$
\item $d(x,y)\leq d(x,z)+d(z,y)$
\end{itemize}
\end{defn}

\begin{defn}{Metric Space}{}\\
A metric space is an oredered pair $(X,d)$ where $X$ is a set and $d$ is a metric on $X$. 
\end{defn}

\begin{eg}{The Standard Euclidean Metric on $\R$}{}\\
The function $d:\R\times\R\to\R$ defined by $$d(x,y)=\sqrt{x^2+y^2}$$ defines a metric on $\R$, called the standard Euclidean metric. 
\end{eg}

\begin{eg}{Exotic Metrics in $\R^2$}{}\\
The following functions define metrics on $\R^2$. 
\begin{itemize}
\item Define the standard Euclidean metric on $\R^2$ by $$d((x_1,y_1),(x_2,y_2))=\sqrt{(x_1-x_2)^2+(y_1-y_2)^2}$$
\item Define the taxicab metric on $\R^2$ by $$d((x_1,y_1),(x_2,y_2))=\abs{x_1-x_2}+\abs{y_1-y_2}$$
\item Define the French railway metric on $\R^2$ by $$d((x_1,y_1),(x_2,y_2))=\begin{cases}
\sqrt{(x_1-x_2)^2+(y_1-y_2)^2} & \text{ if } x_2y_1=y_2x_1\\
\sqrt{x_1^2+x_2^2}+\sqrt{y_1^2+y_2^2} & \text{ otherwise }
\end{cases}$$
\item Define the jungle river metric on $\R^2$ by $$d((x_1,y_1),(x_2,y_2))=\begin{cases}
\abs{y_1-y_2} & \text{ if } x_1=x_2\\
\abs{y_1}+\abs{x_1-x_2}+\abs{y_2} & \text{ otherwise }
\end{cases}$$
\item Define the box metric on $\R^2$ by $$d((x_1,y_1),(x_2,y_2))=\max\{\abs{x_1-x_2},\abs{y_1,y_2}\}$$
\end{itemize}
\end{eg}

\begin{eg}{The Discrete Metric}{}\\
Let $X$ be a set. Define the discrete metric on $X$ to be the function $d:X\times X\to\R$ given by $$d(x,y)=\begin{cases}
0 & \text{ if }x=y\\
1 & \text{ otherwise }
\end{cases}$$
\end{eg}

\subsection{New Metrics from Old}
\begin{lmm}{Metric Subspace}{}\\
Let $(X,d)$ be a metric space. Let $U\subseteq X$ be a subset. Then $(U,d|_{U\times U})$ is also a metric space
\begin{proof}\\
$d|_{U\times U}$ has the same properties as $d$. 
\end{proof}
\end{lmm}

\begin{defn}{The p Metric}{} Let $(X,d_X)$ and $(Y,d_Y)$ be metric spaces. Let $p\in[1,\infty)$. Define the $p$ metric on $X\times Y$ to be the map $d_p:(X\times Y)\times(X\times Y)\to\R$ given by $$d_p((x_1,y_1),(x_2,y_2))=\left(d_X(x_1,x_2)^p+d_Y(y_1,y_2)^p\right)^{1/p}$$
\end{defn}

\begin{prp}{}{} Let $(X,d_X)$ and $(Y,d_Y)$ be metric spaces. Let $p\in[1,\infty)$. Then $d_p$ is a metric on $X\times Y$. 
\end{prp}

\begin{eg}{}{}\\
Let $n\in\N$. Then the function $d:\R^n\times\R^n\to\R$ defined by $$d((x_1,\dots,x_n),(y_1,\dots,y_n))=\sqrt{\sum_{i=1}^n(x_i-y_i)^2}$$ is a metric in $\R^n$, called the standard Euclidean metric of $\R^n$. 
\end{eg}

\subsection{Size and Distance of Subsets}
\begin{defn}{Open Balls}{}\\
Let $(X,d)$ be a metric space. Let $x\in X$. Let $r>0$. Define the open ball of radius $r$ at $x$ to be $$B_r(x)=\{y\in X\;|\;d(x,y)<r\}$$
\end{defn}

\begin{defn}{Bounded Subsets}{}\\
Let $X$ be a metric space. Let $U\subseteq X$ be a subset. We say that $U$ is bounded if there exists $x\in X$ and $r>0$ such that $U\subseteq B_r(x)$. 
\end{defn}

\begin{defn}{Distance between two Subsets}{}\\
Let $X$ be a metric space. Let $U,V\subseteq X$ be subsets. Define the distance between $U$ and $V$ to be $$\text{dist}(U,V)=\inf\{d(u,v)\;|\;u\in U,v\in V)\}$$
\end{defn}

\subsection{Open and Closed Subsets}
\begin{defn}{Open and Closed Sets}{}\\
Let $X$ be a metric space. Let $U\subseteq X$. 
\begin{itemize}
\item We say that $U$ is an open subset of $X$ if for all $x\in U$, there exists $r>0$ such that $B_r(x)\subseteq U$. 
\item We say that $U$ is closed if $X\setminus U$ is open. 
\end{itemize}
\end{defn}

\begin{eg}{}{}\\
Consider the following subsets of $\R$. 
\begin{itemize}
\item $(0,1)$ is open in $\R$. 
\item $[0,1]$ is closed in $\R$. 
\item $(0,1]$ and $[0,1)$ are neither open nor closed in $\R$. 
\end{itemize}
\end{eg}

\begin{eg}{}{}\\
Let $[0,1)$ inherit a metric subspace structure from $\R$. 
\begin{itemize}
\item $[1/2,1)$ is closed in $[0,1)$. 
\item $[0,1)$ is both open and closed in $[0,1)$. 
\end{itemize}
\end{eg}

\begin{lmm}{}{}\\
Let $X$ be a metric space. Let $x\in X$. Let $r>0$. Then $B_r(x)$ is an open subset of $X$. 
\begin{proof} Let $B_r(a)$ be our open ball. Let $x\in B_r(a)$. Then $$B_{(r-d(x,a))/2}(x)\subseteq B_r(a)$$ thus we are done. 
\end{proof}
\end{lmm}

\begin{prp}{}{}\\
Let $X$ be a metric space. Then the following are true. 
\begin{itemize}
\item Let $\{U_i\;|\;i\in I\}$ be a collection of open subsets of $X$. Then $\bigcup_{i\in I}U_i$ is an open subset of $X$. 
\item Let $U_1,\dots,U_n$ be open subsets of $X$. Then $\bigcap_{i=1}^nU_i$ is an open subset of $X$. 
\item Let $\{U_i\;|\;i\in I\}$ be a collection of closed subsets of $X$. Then $\bigcap_{i\in I}U_i$ is a closed subset of $X$. 
\item Let $U_1,\dots,U_n$ be closed subsets of $X$. Then $\bigcup_{i=1}^nU_i$ is a closed subset of $X$. 
\end{itemize} 
\begin{proof}\\
Suppose that $\{U_i\;|\;i\in I\}$ is a collection of open subsets of $X$. Let $x\in\bigcup_{i\in I}U_i$. Then there exists $i\in I$ such that $x\in U_i$. Since $U_i$ is open, there exists $r>0$ such that $B_r(x)\subseteq U_i$. Then $B_r(x)\subseteq\bigcup_{i\in I}U_i$ and so we are done. \\

Suppose that $U_1,\dots,U_n$ are open subsets of $X$. Let $x\in\bigcap_{i=1}^nU_i$. Then for $1\leq i\leq n$, there exists $r_i>0$ such that $B_{r_i}(x)\subseteq U_i$. Let $r=\min\{r_1,\dots,r_n\}$. Then for $1\leq i\leq n$, we have $B_r(x)\subseteq B_{r_i}(x)\subseteq U_i$. Hence $B_r(x)\subseteq\bigcap_{i=1}^nU_i$. Thus $\bigcap_{i=1}^nU_i$ is an open subset. \\
 
Suppose that $\{U_i\;|\;i\in I\}$ is a collection of closed subsets of $X$. Then $\{X\setminus U_i\;|\;i\in I\}$ is a collection of open subsets of $X$. Hence $$\bigcup_{i\in I}(X\setminus U_i)=X\setminus\bigcap_{i\in I}U_i$$ is an open subset of $X$. Hence $\bigcap_{i\in I}U_i$ is a closed subsets of $X$. \\

Suppose that $U_1,\dots,U_n$ are closed subsets of $X$. Then $X\setminus U_i$ for $1\leq i\leq n$ are open subsets. Hence $$\bigcap_{i=1}^n(X\setminus U_i)=X\setminus\bigcup_{i=1}^nU_i$$ is an open subset of $X$ by the above. Hence $\bigcup_{i=1}^nU_i$ is a closed subset of $X$. 
\end{proof}
\end{prp}

\subsection{Distance Preserving Maps}
\begin{defn}{Distance Preserving Maps}{}\\
Let $(X,d_X)$ and $(Y,d_Y)$ be metric spaces. Let $f:X\to Y$ be a function. We say that $f$ is distance preserving if for any $p,q\in X$, we have $$d_Y(f(p),f(q))=d_X(p,q)$$
\end{defn}

\begin{defn}{Isometries}{} Let $(X,d_1)$ and $(X,d_2)$ be metric spaces. Let $f:X\to Y$ be a distance preserving map. We say that $f$ is an isometry if there exists a distance preserving map $g:Y\to X$ such that $$g\circ f=\text{id}_X\;\;\;\;\text{ and }\;\;\;\;f\circ g=\text{id}_Y$$
\end{defn}

It is also common to use the term isometry to mean a distance preserving function and refer to our version of isometry as bijective isometries. 

\begin{lmm}{}{} Let $(X,d_1)$ and $(X,d_2)$ be metric spaces. Let $f:X\to Y$ be a distance preserving map. Then $f$ is an isometry if and only if $f$ is bijective. 
\begin{proof}\\
Clearly, if $f$ is an isometry then $f$ has an inverse function. Conversely, suppose that $f$ is bijective. Then we have $$d_X(f^{-1}(p),f^{-1}(q))=d_Y(f(f^{-1}(p)),f(f^{-1}(q)))=d_Y(p,q)$$ for any $p,q\in Y$ since $f$ is distance preserving. Hence $f^{-1}$ is distance preserving. 
\end{proof}
\end{lmm}

\begin{defn}{The Set of Isometries of a Metric Space}{}\\
Let $(X,d)$ be a metric space. Define the set of isometries of $X$ to be $$\text{Isom}(X,d)=\{f:X\to X\;|\;f\text{ is an isometry }\}$$
\end{defn}

\section{Limits and Continuity}
\subsection{Sequences}
\begin{defn}{Sequences}{}\\
Let $X$ be a metric space. A sequence in $X$ is a collection of elements $a_n\in X$ for each $n\in\N$. We denote the sequence by $(a_n)_{n\in\N}$. 
\end{defn}

\begin{defn}{Convergence}{}\\
Let $(X,d)$ be a metric space. Let $(x_n)_{n\in\N}$ be a sequence in $X$. Let $x\in X$. We say that the sequence converges to the limit $x$ if for all $\varepsilon>0$, there exists $N\in\N$ such that $d(x_n,x)<\varepsilon$ for all $n>N$. 
\end{defn}

\begin{lmm}{}{}\\
Let $X$ be a metric space. Let $(x_n)_{n\in\N}$ be a sequence in $X$. Suppose that $x,y\in X$ are the limits of the sequence, then $x=y$. 
\begin{proof}\\
Let $\varepsilon>0$. Since $(x_n)_{n\in\N}$ converges to $x$ and $y$, there exists $N_1,N_2\in\N$ such that $d(x_n,x)<\frac{\varepsilon}{2}$ and $d(x_m,y)<\frac{\varepsilon}{2}$ for all $n>N_1$ and $m>N_2$ respectively. Let $N=\max\{N_1,N_2\}$. Then we have $d(x_n,x)<\varepsilon$ and $d(x_n,y)<\frac{\varepsilon}{2}$ for all $n\in\N$. By the triangle inequality, we have $$d(x,y)<d(x_n,x)+d(x_n,y)=\frac{\varepsilon}{2}+\frac{\varepsilon}{2}=\varepsilon$$ Since $0\leq d(x,y)<\varepsilon$ for all $\varepsilon>0$, we deduce that $d(x,y)=0$. Hence $x=y$. 
\end{proof}
\end{lmm}

\begin{prp}{}{}\\
Let $X$ be a metric space. Let $U\subseteq X$. Then $U$ is closed if and only if for all sequence $(x_n)_{n\in\N}$ in $U$ that converges in $x\in X$, we have $x\in U$. 
\begin{proof} Suppose that $U$ is closed. Then $X\setminus U$ is open by definition. Let $\{x_n\}\subset U$ converge to $x\notin U$. Then $x\in X\setminus U$. By definition of convergence, for every $\epsilon>0$, there exists $N\in\N$ such that $x_n\in B_\epsilon(x)$ for $n>N$. But since $X\setminus U$ is open, there should be some $\epsilon$ such that $B_\epsilon(x)\subset X\setminus U$. In this case, we would have $x_n\in B_r(x)\subset X\setminus U$ which is a contradiction. \\
Suppose that the right hand side is true. Suppose for a contradiction that $X\setminus U$ is not open. Then for every $\epsilon>0$, $B_\epsilon(x)$ is not a subset of $X\setminus U$ for some $x\in X\setminus U$. Let $y_k\in B_{1/k}(x)$ but $y_k\notin X\setminus U$. Then $y_k\in U$ and $y_k\to x\in X\setminus U$, a contradiction. 
\end{proof}
\end{prp}

\begin{prp}{}{} Let $X,Y$ be metrics spaces. Let $((x_n,y_n))_{n\in\N}$ be a sequence in $X\times Y$. Let $(x,y)\in X\times Y$. Then $((x_n,y_n))_{n\in\N}$ converges to $(x,y)$ if and only if $(x_n)_{n\in\N}$ converges to $x$ and $(y_n)_{n\in\N}$ converges to $y$. 
\end{prp}

\begin{defn}{Cauchy Sequence}{} We say that $\{x_n\}\subset(X,d)$ is a Cauchy sequence if for every $\epsilon>0$, there exists some $N$ such that $d(x_n,x_m)<\epsilon$ for all $n,m>\epsilon$. 
\end{defn}

\begin{prp}{}{} Every convergent sequence is Cauchy. 
\begin{proof}
Let $(x_n)_{n\in\N}$ be a convergent sequence in a metric space $X$. Let $\epsilon>0$, then from convergence we have that for $d(x_n,x)<\frac{\epsilon}{2}$ for all $n>N$ for some $N\in\N$. Then choosing the same $N$, we have that $$d(x_n,x_m)\leq d(x_n,x)+d(x,x_m)<\frac{\epsilon}{2}+\frac{\epsilon}{2}=\epsilon$$ thus we are done. 
\end{proof}
\end{prp}

We now give the definition of a complete space in terms of Cauchy sequences. 

\subsection{Limits of Functions}
\begin{defn}{Accumulation Points}{}\\
Let $M$ be a metric space. Let $x\in M$. We say that $x$ is an accumulation point of $M$ if for all $r\in\R^+$, we have $$B_r(x)\setminus\{x\}\neq\emptyset$$
\end{defn}

\begin{defn}{Limits}{}\\
Let $(X,d_X),(Y,d_Y)$ be metric spaces. Let $f:X\to Y$ be a function. Let $p\in X$ be an accumulation point. We say that $L\in Y$ is the limit of $f$ at $p$ if for all $\varepsilon>0$, there exists $\delta>0$ such that for all $x\in M$ such that $0<d_X(x,p)<\delta$, we have $$d_Y(f(x),L)<\varepsilon$$ In this case, we write $L=\lim_{x\to p}f(x)$. 
\end{defn}

\subsection{Continuous Functions}
\begin{defn}{Continuity}{}\\
Let $(X,d_X),(Y,d_Y)$ be metric spaces. Let $f:X\to Y$ be a function. Let $p\in X$. We say that $f$ is continuous at $p$ if for all $\varepsilon>0$, there exists $\delta>0$ such that for any $x\in X$ such that $0<d_X(x,p)<\delta$, we have $d_Y(f(x),f(p))<\varepsilon$. 
\end{defn}

\begin{prp}{}{}\\
Let $X,Y$ be metric spaces. Let $f:X\to Y$ be a function. Let $p\in X$ be an accumulation point. Then the following are equivalent. 
\begin{itemize}
\item $f$ is continuous at $p$. 
\item For every sequence $(x_n)_{n\in\N}\subseteq X$ such that $(x_n)_{n\in\N}\to x$, we have $(f(x_n))_{n\in\N}\to f(p)$. 
\item $\lim_{x\to p}f(x)=f(p)$. 
\end{itemize}
\begin{proof}
\end{proof}
\end{prp}

\begin{prp}{}{}\\
Let $X,Y$ be metric spaces. Let $f:X\to Y$ be a function. Then the following are equivalent. 
\begin{itemize}
\item $f$ is continuous. 
\item For all $y\in Y$ and $r>0$, we have that $f^{-1}(B_r(y))$ is open in $X$. 
\item For all open subsets $V\subseteq Y$, we have that $f^{-1}(V)$ is open in $X$. 
\end{itemize}
\begin{proof}\\
Suppose that $f$ is continuous. Let $r>0$. By continuity, there exists $\delta>0$ such that $f(B_\delta(p))\subset B_r(f(p))$. This implies that $B_\delta(p)\subset f^{-1}(B_r(f(p))$. Thus $f^{-1}(B_r(f(p))$ is open. \\

Now suppose that for all $y\in Y$ and $r>0$, we have $f^{-1}(B_r(y))$ is open in $X$. Let $V\subseteq Y$ be an open subset. Let $x\in f^{-1}(V)$. Then there exists $r>0$ such that $B_r(f(x))\subseteq V$. This means that $x\in f^{-1}(B_r(f(x))\subseteq f^{-1}(V)$. Since $f^{-1}(B_r(f(x))$ is open, there exists $s>0$ such that $$B_s(x)\subseteq f^{-1}(B_r(f(x))\subseteq f^{-1}(V)$$ Since this is true for all $x\in f^{-1}(V)$, we conclude that $f^{-1}(V)$ is open. \\

Now suppose that $\Omega\subset V$ is open imply $f^{-1}(\Omega)$ is open. Let $p\in \Omega$. Then there exists $\epsilon>0$ such that $B_{\epsilon}(f(p))\subset\Omega$. By assumption, we must have $f^{-1}(B_{\epsilon}(f(p)))$ is open. The fact that this is open means there exists $\delta>0$ such that $B_{\delta}(p)\subset f^{-1}(B_{\epsilon}(f(p)))$. Then we have $$f(B_{\delta}(p))\subset B_{\epsilon}(f(p))$$ and we are done. 
\end{proof}
\end{prp}

\begin{eg}{}{}\\
Let $X$ be a metric space. Let $y\in X$. Prove that the map $d(-,y):X\to\R$ is continuous. 
\begin{answer}\\
Let $(a,b)\subseteq\R$. If the inverse image is empty, then we are done. So suppose that $z\in\{x\in X\;|\;a<d(x,y)<b\}$. Choose $0<r<b-d(z,y)$. Then for any $w\in B_r(z)$, we have $$d(w,y)\leq d(w,z)+d(z,y)<r+d(z,y)<b-d(z,y)+d(z,y)=b$$ Hence $B_r(z)\subseteq\{x\in X\;|\;a<d(x,y)<b\}$ and so the latter is an open set. By the above proposition we conclude. 
\end{answer}
\end{eg}

\begin{lmm}{}{}\\
Let $(X,d_1)$ and $(Y,d_2)$ be metric spaces. Let $f:X\to Y$ be a distance preserving map. Then $f$ is continuous. 
\begin{proof}
Let $p\in X$. Let $\varepsilon>0$. Choose $\delta=\varepsilon$, then for all $x\in X$ such that $0<d_X(x,p)<\delta$, we have $d_Y(f(x),f(p))=d_X(x,p)<\delta=\varepsilon$ so that $f$ is continuous. 
\end{proof}
\end{lmm}

\begin{prp}{}{}\\
Let $X,Y,Z$ be metric spaces. Let $f:X\to Y$ and $g:Y\to Z$ be functions. If $f$ and $g$ are continuous, then $g\circ f$ is continuous. 
\end{prp}

\subsection{Uniform Continuous Functions}
\begin{defn}{Uniformly Continuous}{} A map $f:X\to Y$ is uniformaly continuous if for every $\epsilon>0$, there exists $\delta>0$ such that $$d_X(x,y)<\delta\implies d_Y(f(x),f(y))<\epsilon$$ for any $x,y\in X$. 
\end{defn}

\begin{lmm}{}{} If $f:X\to Y$ is uniformly continuous then it is continuous. 
\end{lmm}

\subsection{Lipschitz Continuous Functions}
\begin{defn}{Lipschitz Continuous}{} Let $(X,d_X)$ and $(Y,d_Y)$ be metric spaces and suppose that $f:X\to Y$. We say that $f$ is a Lipschitz map if there is a constant $K\geq 0$ such that $$d_Y(f(x),f(y))\leq Kd(x,y)$$ for all $x,y$ in $X$. 
\end{defn}

\begin{lmm}{}{} If $f:X\to Y$ is Lipschitz continuous then it is uniformly continuous. 
\end{lmm}

\begin{defn}{Contraction Maps}{}\\
Let $X$ be a metric space. Let $f:X\to X$ be a function. We say that $f$ is a contraction map if $f$ is Lipschitz continuous with $K\in[0,1)$. 
\end{defn}

\section{Convergence in Metric Spaces}
\subsection{Complete Metric Spaces}
\begin{defn}{Complete Spaces}{}\\
Let $X$ be a metric space. We say that $X$ is complete if every Cauchy sequence in $X$ converges. 
\end{defn}

\begin{prp}{}{}\\
Let $X$ be a metric space. Let $Y\subseteq X$ be a subset. Then the following are true. 
\begin{itemize}
\item If $X$ is complete and $Y$ is closed, then $Y$ is complete. 
\item If $Y$ is complete, then $Y$ is closed. 
\end{itemize}
\begin{proof}\\
Suppose that $X$ is a complete metric space and $Y$ is closed. Let $(y_n)_{n\in\N}$ be a Cauchy sequence in $Y$. Since it is also a sequence in $X$, completeness of $X$ implies that it converges to some $x\in X$. But since $Y$ is closed, we must have $x\in Y$. Hence $Y$ is complete. \\

Now suppose that $Y$ is complete. Then for any convergent sequence $(y_n)_{n\in\N}$ in $Y$, it is Cauchy and hence converges to some $Y\in Y$. Thus $Y$ is closed. 
\end{proof}
\end{prp}

\begin{thm}{Cantor's Intersection Theorem}{} Let $X$ be a complete metric space. Let $S_1\supseteq S_2\supseteq\dots$ form a nested sequence of non-empty closed sets in $X$ with the property that $\text{diam}(S_n)\to 0$ as $n\to\infty$. Then $$\bigcap_{n=1}^\infty S_n\neq\emptyset$$ 
\begin{proof}
For each $N\in\N$, choose $x_N\in S_N$. Then for all $n>N$, $x_n\in S_N$. Thus for $n,m>N$, we have that $d(x_n,x_m)\leq\text{diam}(S_n)$. It follows that $(x_n)_{n\in\N}$ is Cauchy. Thus $x_n\to x$ for some $x\in X$. Since each $S_n$ is closed and $x_n\in S_N$ for all $n>N$, we must have that $x\in S_n$ for each $n$. Thus $x\in\bigcap_{k=1}^\infty S_n$ is nonempty. 
\end{proof}
\end{thm}

Below are a few examples of complete spaces. 

\begin{eg}{}{}\\
$\R^n$ is a complete metric space. 
\begin{proof}\\
Let $(x_k)_{k\in\N}$ be a Cauchy sequence in $\R^n$. Denote the $i$th component of $x_k$ by $x_{k,i}$. Then for every $\epsilon>0$, there exists $N$ such that $$\|x_k-x_m\|=\left(\sum_{i=1}^n\abs{x_{k,i}-x_{m,i}}^2\right)^{\frac{1}{2}}<\epsilon$$ for $k,m>N$. In particular, we have that each individual $$\abs{x_{k,i}-x_{m,i}}<\epsilon$$ for $m,n>N$. Thus $(x_{k,i})_{k\in\N}$ is a Cauchy sequence in $\R$. But we know that Cauchy sequences in $\R$ converges, thus $(x_{k,i})_{k\in\N}$ converges to $x_i\in\R$. Now define $x=(x_1,\dots,x_n)$, then $$\|x_k-x\|=\left(\abs{x_{k,i}-x_i}^2\right)^{\frac{1}{2}}<n\epsilon$$ by convergence of each individual component. Thus $(x_n)_{n\in\N}$ is a convergent sequence. \\
\end{proof}
\end{eg}

\subsection{Completion}
The goal of this section is to attempt to complete a metric space by adding in the missing limits of a metric space. 

\begin{defn}{Space of Bounded Real Functions}{}\\
Let $X$ be a metric space. Define the set of bounded real functions of $X$ to be $$B(X)=\{f:X\to\R\;|\;\abs{f}\leq M\text{ for some }M\in\R\}$$
\end{defn}

\begin{defn}{The Supremum Metric}{}\\
Let $X$ be a metric space. Define the supremum metric on $B(X)$ by the formula $$\|f-g\|_\infty=\sup_{x\in X}\abs{f(x)-g(x)}$$
\end{defn}

\begin{prp}{}{}\\
Let $X$ be a metric space. Then the metric space $B(X)$ whose metric induced by the supremum norm is complete. 
\begin{proof}
Let $(f_n)_{n\in\N}$ be a Cauchy sequence in $B(X)$. Then for every $\epsilon>0$, there exists $N$ such that $$\|f_n-f_m\|_{\infty}=\sup_{x\in X}\abs{f_n(x)-f_m(x)}<\epsilon$$ for all $n,m>N$. In particular, for each $x\in X$, the property of supremum implies that $\abs{f_n(x)-f_m(x)}<\epsilon$ for $n,m>N$. Thus $(f_n(x))_{n\in\N}\subset\R$ is Cauchy for each $x$. Since $\R$ is complete, $(f_n(x))_{n\in\N}$ converges for each $x\in X$. \\
Now define the function $f:X\to\R$ by $$f(x)=\lim_{n\to\infty}f_n(x)$$ Then fix $\epsilon>0$, we have that $$\abs{f_n(x)-f(x)}<\epsilon$$ for all $n>N$ by letting $m\to\infty$ from the fact that $\abs{f_n(x)-f_m(x)}<\epsilon$. This $N$ does not depend on $x$. Fix $\epsilon=1$, then there exists $N_1\in\N$ such that 
\begin{align*}
\abs{f(x)-f_n(x)}\leq\abs{f(x)-f_{N_1}(x)}\\
\leq 1+\abs{f_{N_1}(x)}
\end{align*} for all $x\in X$ and $n>N_1$ thus $f$ is bounded. This means that $f\in B(X)$ and that $\|f_n-f\|_\infty<\epsilon$ for all $n>N$. 
\end{proof}
\end{prp}

\begin{prp}{}{}\\
Let $X$ be a metric space. Then there exists a function $f:X\to B(X)$ such that map $f:X\to\im(f)$ is an isometry. 
\end{prp}

\subsection{The Contraction Mapping Theorem}
\begin{thm}{Contraction Mapping Theorem}{} Let $X$ be a nonempty complete metric space and suppose that $f:X\to X$ is a contraction. Then $f$ has a unique fixed point, meaning there is a unique $x\in X$ such that $f(x)=x$. 
\begin{proof}
Let $x_0\in X$ and define a sequence by $x_{n+1}=f(x_n)$ for $n\in\N$. Then we have that $$d(x_{n+1},x_n)\leq Kd(x_n,x_{n-1})\leq\dots\leq K^nd(x_1,x_0)$$ Then for any $k>n$, we have that 
\begin{align*}
d(x_k,x_n)&\leq\sum_{i=n}^{k-1}d(x_{i+1},x_i)\\
&\leq\sum_{i=n}^{k-1}K^id(x_1,x_0)\\
&\leq\frac{K^i}{1-K}d(x_1,x_0)
\end{align*}
This is Cauchy since we can choose $\epsilon>0$ such that $\frac{K^i}{1-K}<\epsilon$. Since $X$ is complete, we have that $x_n\to x$ for some $x\in X$. Since $f$ is continuous we have that $f(x_n)\to f(x)$. Now taking limits on $$x_{n+1}=f(x_n)$$ we have that $x=f(x)$. \\
To prove uniqueness, note that if $f(x)=x$ and $f(y)=y$, then $$d(x,y)=d(f(x),f(y))\leq Kd(x,y)$$ which implies that $(1-K)d(x,y)=0$. Thus $x=y$. 
\end{proof}
\end{thm}

Another name for this theorem would be Banach's Fixed Point Theorem. 

\begin{thm}{Picard-Lindelof Theorem}{} Let $f:\R^n\to\R^n$ be Lipschitz continuous with $$\abs{f(x)-f(y)}\leq L\abs{x-y}$$ where $x,y\in\R^n$. Then for any $x_0\in\R^n$, the differential equation $$\frac{dx}{dt}=f(x)$$ with initial condition $x(0)=x_0$ has a unique solution on $[-t,t]$ for any $Lt<1$. 
\end{thm}

\subsection{Cantor's Theorem}
\begin{thm}{}{} If $X$ is a complete metric space and $\{F_n|n\in\N\}$ is a collection of open dense subsets of $X$, then $$F=\bigcap_{k=1}^\infty F_nt$$ is dense in $X$. Equivalently, if $\{G_n|n\in\N\}$ is a collection of nowhere dense subsets of a nonempty complete metric space $X$, then $$\bigcup_{k=1}^\infty F_k\neq X$$
\end{thm}

\begin{lmm}{}{} The Cantor set is uncountable. 
\end{lmm}

\section{Paths in Metric Spaces}
\subsection{Basic Properties of Paths}
\begin{defn}{Paths in a Metric Space}{}\\
Let $X$ be a metric space. A path in $X$ is a continuous function $f:[a,b]\to X$. 
\end{defn}

\begin{defn}{Different Types of Paths}{}\\
Let $X$ be a metric space. Let $f:[a,b]\to X$ be a path in $X$. 
\begin{itemize}
\item We say that $r$ is simple if $f(t_1)=f(t_2)$ implies $t_1=t_2$ for all $t_1,t_2\in(a,b)$. 
\item We say that $r$ is closed if $f(a)=f(b)$. 
\end{itemize}
\end{defn}

\begin{defn}{Reparametrization}{}\\
Let $X$ be a metric space. Let $f:[a,b]\to X$ and $g:[c,d]\to X$ be paths in $X$. We say that $g$ is a reparametrization of $f$ if there exists a continuous function $\rho:[c,d]\to[a,b]$ such that $g=f\circ\rho$. 
\end{defn}

\begin{defn}{Orientation Preserving Reparametrizations}{}
Let $X$ be a metric space. Let $f:[a,b]\to X$ be a path in $X$. Let $g:[c,d]\to X$ be a reparametrization of $f$ via the continuous function $\rho:[c,d]\to[a,b]$. We say that $g$ is an orientation preserving reparametrization if $\rho'(t)>0$ for all $t\in[c,d]$. 
\end{defn}

\subsection{Geodesics}
\begin{defn}{Geodesics}{}\\
Let $X$ be a metric space. Let $I\subseteq\R$ be an interval. Let $\gamma:I\to\R$ be a path. We say that $\gamma$ is a geodesic if $\gamma$ is distance preserving. 
\end{defn}

\begin{defn}{Geodesic Metric Spaces}{} Let $(X,d)$ be a metric space. We say that $X$ is geodesic if for all $a,b\in X$, there exists a geodesic $\gamma:I\to X$ from $a$ to $b$. 
\end{defn}
