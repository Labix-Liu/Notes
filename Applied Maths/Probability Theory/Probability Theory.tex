\section{Foundations of Probability Theory}
\subsection{Definition of Probability}
\begin{defn}{Probability Space}{}\\
A probability space is a measure space $(\Omega,\mF,P:\mF\to[0,1])$. 
\end{defn}

Explicitly, a probability space is a triple $(\Omega,\mF,P)$ consisting of the following data: 
\begin{itemize}
\item $\Omega\neq\emptyset$ is a set called the sample space. 
\item $\mF\subseteq\mP(\Omega)$ is a $\sigma$-algebra called events.  \item $P:\mF\to[0,1]$ is a set function. 
\end{itemize}
such that the following are true: 
\begin{itemize}
\item $P(\Omega)=1$. 
\item If $\{A_n\;|\;n\in\N\}\subseteq\mF$ are pairwise disjoint, then $$P\left(\bigcup_{k=1}^\infty A_k\right)=\sum_{k=1}^{\infty}P(A_k)$$
\end{itemize}

\begin{defn}{Events and Probability}{}\\
Let $(\Omega,\mF,P)$ be a probability space. 
\begin{itemize}
\item An event of the space is an element of $\mF$. 
\item Let $A\in\mF$ be an event. The probability of $A$ is defined to be the value $P(A)$. 
\end{itemize}
\end{defn}

\begin{eg}{}{}\\
Let $\Omega$ be the set of all possible outcomes of flipping a coin ten times. Let $\mF=\mP(\Omega)$ be the set of subsets of $\Omega$. Let $P:\mF\to[0,1]$ be the function defined by $P(E)=\frac{\abs{E}}{\abs{\Omega}}$ for all $E\in\mF$. Then $(\Omega,\mF,P)$ is a probability space. 
\end{eg}

\begin{prp}{}{}\\
Let $(\Omega,\mF,P)$ be a probability space. Let $A,B\in\mF$ be events. Then the following are true. 
\begin{itemize}
\item $P(\Omega\setminus A)=1-P(A)$
\item $A\subset B\implies P(A)\leq P(B)$
\end{itemize} 
\begin{proof}\\
Let $A\subset B\subset\Omega$ be events in $\Omega$. 
\begin{itemize}
\item $A$ and $\Omega\setminus A$ are disjoint and $P(\Omega)=P(A)+P(\Omega\setminus A)$ and $P(\Omega\setminus A)=1-P(A)$
\item We have that $A$ and $B\setminus A$ are disjoint. Thus $P(B)=P(A)+P(B\setminus A)$. Since $P(B\setminus A)\geq 0$, we have $P(A)\leq P(B)$.  
\end{itemize}
\end{proof}
\end{prp}

\begin{eg}{The Birthday Problem}{}\\
Suppose that there are $k$ people in a room, all of whom are equally likely to have a birthday in one of the 365 days in a year. Suppose that the birthdays of the people in the room are independent of each other. What is the probability that two or more people in the room have the same birthday?
\begin{proof}\\
Let $\Omega=\{\{n_1,\dots,n_k\}\;|\;1\leq n_1,\dots,n_k\leq 365\}$. These indicate all the possible birthday configurations. Let $\mF=\mP(\Omega)$. This means that every birthday configuration we constructed is possible. By the assumptions of the question, we set $P(E)=\frac{\abs{E}}{\abs{\Omega}}$ for an event $E\in\mF$. Then we have $$P(\text{at least one birthday match})=1-P(\text{no birthday matches})=1-\frac{365!}{(365)^k(365-k)!}$$
\end{proof}
\end{eg}

\begin{thm}{Principle of Inclusion Exclusion}{}\\
Let $A,B\subset\Omega$ be a sample space and $P$ the probability measure. $$P(A\cup B)=P(A)+P(B)-P(A\cap B)$$
\begin{proof}\\
Note that 
\begin{align*}
A\cup(B\setminus A)&=A\cup(B\cap A^c)\\
&=(A\cup B)\cap(A\cup A^c)\\
&=A\cup B
\end{align*} Note also that $A\cap (B\setminus A)=\emptyset$. Thus $P(A\cup B)=P(A)+P(B\setminus A)=P(A)+P(B)-P(A\cap B)$
\end{proof}
\end{thm}

\begin{thm}{Extended Principle of Inclusion Exclusion}{} Let $A_k\subset\Omega$ be a sample space and $P$ the probability measure for all $k\leq n\in\mathbb{N}$. Then $$P\left(\bigcup_{k=1}^n A_k\right)=\sum_{k=1}^n(-1)^{k+1}\sum_{1\leq i_1\leq\dots\leq i_k\leq n}P(A_{i_1}\cap A_{i_2}\cap\dots\cap A_{i_k})$$
\end{thm}

\subsection{Conditional Probability}
\begin{defn}{Conditional Probability}{}\\
Let $(\Omega,\mF,P)$ be a probability space. Let $A,B\in\mF$ be events. Define the conditional probability of $A$ given $B$ to be $$P(A\;|\;B)=\frac{P(A\cap B)}{P(B)}$$
\end{defn}

\begin{prp}{Multiplication Rule}{}\\
Let $(\Omega,\mF,P)$ be a probability space. Let $A_1,\dots,A_n\in\mF$ be events. Then we have 
$$P\left(\bigcap_{i=1}^nA_i\right)=P(A_1)\cdot\prod_{i=2}^nP\left(A_i\;|\;\bigcap_{j=1}^{i-1}A_j\right)$$
\begin{proof}\\
Follow immediately by induction on the definition of conditional probability. 
\end{proof}
\end{prp}

\begin{thm}{Bayes' Rule}{}\\
Let $(\Omega,\mF,P)$ be a probability space. Let $A,B\in\mF$ be spaces such that $P(A),P(B)>0$. Then we have $$P(A|B)=\frac{P(B|A)P(A)}{P(B)}$$
\begin{proof}\\
We have that $P(A\cap B)=P(A|B)P(B)$ and $P(A\cap B)=P(B|A)P(A)$. 
\end{proof}
\end{thm}

\begin{thm}{Law of Total Probability}{}\\
Let $(\Omega,\mF,P)$ be a probability space. Let $A_1,\dots,A_n\in\mF$ be events that partition $\Omega$ such that $P(A_i)>0$ for $1\leq i\leq n$. Let $B\in\mF$ be an event. Then we have $$P(B)=\sum_{k=1}^nP(A_k)P(B|A_k)$$
\begin{proof}\\
Note that since $A_1,\dots,A_n$ is a partition of $\Omega$, $B\cap A_1,\dots,B\cap A_n$ is a partition of $B$. Then we have
\begin{align*}
\sum_{k=1}^nP(A_k)P(B|A_k)&=\sum_{k=1}^nP(B\cap A_k)\\
&=P\left(\bigcup_{k=1}^nB\cap A_k\right)\\
&=P(B\cap\Omega)\\
&=P(B)
\end{align*}
\end{proof}
\end{thm}

\begin{eg}{Gambler's Ruin}{}\\
A gambler starts with an initial fortune of £k for $1\leq k\leq n$. On each successive gamble, the gambler either wins £1 or loses £1 independent of past games, with a probability of $p$ and $1-p$ respectively for $p\in(0,1)$. What is the probability that the gambler reaches £n dollars before reaching £0?
\begin{proof}\\
The gambler is said to have lost the game if the gambler reaches £0 after any round. Let $a_i$ denote the probability that the gambler reaches £n dollars before reaching £0, when the gambler has £i. By the law of total probability, we obtain a recurrence relation $$a_i=pa_{i+1}+(1-p)a_{i-1}$$ This makes for a difference equation with boundary conditions $a_0=1$ and $a_n=0$. Solving it gives $$a_i=\begin{cases}
\frac{i}{n} & \text{ if }p=\frac{1}{2}\\
\frac{1-((1-p)/p)^i}{1-((1-p)/p)^n} & \text{ otherwise }
\end{cases}$$
\end{proof}
\end{eg}

Notice that the $p=1/2$ case is consistent with the $p\neq 1/2$ cases since $\lim_{p\to 1/2}\frac{1-((1-p)/p)^i}{1-((1-p)/p)^n}=\frac{i}{n}$. 

\subsection{Independent Events}
\begin{defn}{Independent Events}{}\\
Let $(\Omega,\mF,P)$ be a probability space. Let $A,B\in\mF$ be events. We say that $A$ and $B$ are independent if $$P(A\cap B)=P(A)P(B)$$
\end{defn}

\begin{prp}{}{}\\\
Let $(\Omega,\mF,P)$ be a probability space. Let $A,B\in\mF$ be events. If $A$ and $B$ are independent, then the following are true. 
\begin{itemize}
\item $\Omega\setminus A$ and $B$ are independent. 
\item $A$ and $\Omega\setminus B$ are independent. 
\item $\Omega\setminus A$ and $\Omega\setminus B$ are independent. 
\end{itemize}
\end{prp}

\section{Probability Distributions}
\subsection{Random Variables and its Distribution}
\begin{defn}{Random Variable}{}\\
Let $(\Omega,\mathcal{F},P)$ be a probability space. Let $(E,\mE)$ be a measurable space. An $(E,\mathcal{E})$ valued random variable is an $\mF$-measurable function $X:\Omega\to E$. 
\end{defn}

\begin{defn}{Discrete and Continuous Random Variables}{}\\
Let $(\Omega,\mF,P)$ be a probability space. Let $X:\Omega\to\R$ be a random variable. 
\begin{itemize}
\item We say that $X$ is discrete if $\im(X)$ is a countable subset of $\R$. 
\item We say that $X$ is continuous otherwise. 
\end{itemize}
\end{defn}

Recall that $X$ is an $\mF$-measurable function if $X^{-1}(B)\in\mF$ for $B\in\mE$. 

\begin{eg}{}{}\\
Let $\Omega$ be the the sample space of tossing three fair coins. Let $X:\Omega\to\R$ be the number of heads observed. Let $Y:\Omega\to\R$ be the number of tails observed. Then $X$ and $Y$ are identically distributed, but $X\neq Y$. 
\end{eg}

\begin{defn}{Probability Density Functions and Probability Mass Function}{}\\
Let $(\Omega,\mF,P)$ be a probability space. Let $X:\Omega\to\R$ be a random variable. 
\begin{itemize}
\item Suppose that $X$ is a discrete random variable. Define the probability mass function of $X$ to be the Radon–Nikodym derivative $$f_X=\frac{d X_\ast P}{d\#}$$ where $\#$ is the counting measure on $\R$. 
\item Suppose that $X$ is a continuous random variable. Define the probability density function of $X$ to be the Radon–Nikodym derivative $$f_X=\frac{d X_\ast P}{d\mu}$$ where $\mu$ is the Lebesgue measure. 
\end{itemize}
\end{defn}

Recall that this means that $f_X$ satisfies the property that $$P_X(A)=\int_A f_X d\mu$$ for any measurable set $A\subseteq\R$. In particular, if $A=\{a\}\subseteq\mF$, then we have $$f_X(a)=P_X(a)$$ The probability distribution function has its input as every measurable subset of $\R$, while the probability density function takes input as individual points of $\R$. They are really the same thing because having its probability be determined on singletons is sufficient to determine the probability of every measurable subset. 

\begin{prp}{}{}\\
Let $(\Omega,\mF,P)$ be a probability space. Let $f:X\to\R$ be a function. Then the following are true. 
\begin{itemize}
\item $f$ is the probability mass function of a discrete random variable if and only if $f\geq 0$ and $\sum_{x\in X}f_X=1$. 
\item $f$ is the probability density function of a continuous random variable if and only if $f\geq 0$ and $\int_{-\infty}^\infty f\;dx=1$. 
\end{itemize}
\end{prp}

\begin{prp}{}{}\\
Let $(\Omega,\mF,P)$ be a probability space. Let $X:\Omega\to\R$ be a random variable. Let $g:\R\to\R$ be a function. Then the following are true. 
\begin{itemize}
\item Suppose that $X$ is a discrete random variable. Then the probability density function of $Y=g\circ X$ is given by $$f_Y(y)=
\begin{cases}
\sum_{x\in g^{-1}(y)}f_X(x) & \text{ if }y\in\im(g)\\
0 & \text{ otherwise }
\end{cases}$$
\item Suppose that $X$ is a continuous random variable. Suppose that $g$ is strictly monotone and differentiable.  Then the probability density function of $Y=g\circ X$ is given by $$f_Y(y)=\begin{cases}
f_X(g^{-1}(y))\abs{\frac{d}{dy}(g^{-1}(y))} & \text{ if }y\in\im(g)\\
0 & \text{ otherwise }
\end{cases}$$
\end{itemize}
\end{prp}

\subsection{Cumulative Density Functions}
\begin{defn}{Cummulative Distribution Function}{}\\
Let $(\Omega,\mF,P)$ be a probability space. Let $X:\Omega\to\R$ be a random variable. Define the cummulative distribution function $F_X:\R\to\R$ of $X$ to be $$F_X(x)=P_X(X\leq x)$$
\end{defn}

\begin{prp}{}{}\\
Let $(\Omega,\mF,P)$ be a probability space. Let $X:\Omega\to\R$ be a random variable. Then the following are true. 
\begin{itemize}
\item $f_X=\frac{dF_X}{dx}$. 
\item $F_X(x)=\int_{-\infty}^xf_X(t)\;dt$
\end{itemize}
\end{prp}

\begin{prp}{}{}\\
Let $(\Omega,\mF,P)$ be a probability space. Let $X:\Omega\to\R$ be a random variable. Then the following are true regarding the cumulative distribution function $F_X$. 
\begin{itemize}
\item $F_X$ is monotonically increasing: $x\leq y\implies F_X(x)\leq F_X(y)$
\item $F_x$ is right continuous: If $(x_n)$ is a sequence such that $x_1\geq\dots\geq x_n\geq x_{n+1}\geq\dots\geq x$ and $(x_n)\to x$, then $F_X(x_n)\to F_X(x)$
\item $F_X(-\infty)=0$ and $F_X(\infty)=1$
\end{itemize}
\end{prp}

\begin{prp}{}{}\\
Let $(\Omega,\mF,P)$ be a probability space. Let $X:\Omega\to\R$ be a random variable. Then we have $$P(\{\omega\in\Omega\;|\;a<X(\omega)\leq b\})=F_X(b)-F_X(a)$$
\end{prp}

\subsection{Independent Variables and Identical Distributions}
\begin{defn}{Independent Random Variables}{}\\
Let $(\Omega,\mF,P)$ be a probability space. Let $(E,\mE)$ be a measurable space. Let $X,Y:\Omega\to E$ be random variables. We say that $X$ and $Y$ are independent if for any $A,B\in\mE$, we have that $X^{-1}(A)$ and $Y^{-1}(B)$ are independent events in $\mF$. 
\end{defn}

\begin{defn}{Probability Distribution}{}\\
Let $(\Omega,\mF,\Prj)$ be a probability space. Let $(E,\mE)$ be a measurable space. Let $X:\Omega\to E$ be a measurable function. Define the probability distribution of $X$ to be the pushforward measure $P\circ X^{-1}=P_X:\mE\to[0,1]$ defined by $$P_X(A)=P(X^{-1}(A))$$ for $A\in\mE$. 
\end{defn}

\begin{defn}{Identical Distributions}{}\\
Let $(\Omega,\mF,P)$ be a probability space. Let $X,Y:\Omega\to\R$ be random variables. We say that $X$ and $Y$ are identically distributions if $P_X=P_Y$. 
\end{defn}

\begin{lmm}{}{}\\
Let $(\Omega,\mF,P)$ be a probability space. Let $X,Y:\Omega\to\R$ be random variables. Then the following are equivalent. 
\begin{itemize}
\item $X$ and $Y$ are identically distributed. 
\item $f_X=f_Y$. 
\item $F_X=F_Y$. 
\end{itemize}
\end{lmm}

\subsection{Examples of Discrete Random Variables}
\begin{defn}{Discrete Uniform Distribution}{}\\
Let $(\Omega,\mF,P)$ be a probability space. Let $X:\Omega\to\R$ be a random variable. We say that $X$ has a discrete uniform distribution if the probability density function of $X$ is given by $$f_X(x)=\begin{cases}
\frac{1}{n} & \text{ if }x=1,\dots,n\\
0 & \text{otherwise}
\end{cases}$$ for some $n\in\N$. 
\end{defn}

\begin{defn}{Bernoulli Distribution}{}\\
Let $(\Omega,\mF,P)$ be a probability space. Let $X:\Omega\to\R$ be a random variable. We say that $X$ has a Bernoulli distribution if the probability density function of $X$ is given by $$f_X(x)=\begin{cases}
p & \text{ if }x=1\\
1-p & \text{ if }x=0\\
0 & \text{otherwise}
\end{cases}$$ for some $p\in [0,1]$. In this case we write $X\sim\text{Bern}(p)$. 
\end{defn}

\begin{defn}{Binomial Distribution}{}\\
Let $(\Omega,\mF,P)$ be a probability space. Let $X:\Omega\to\R$ be a random variable. We say that $X$ has a binomial distribution if the probability density function of $X$ is given by $$f_X(x)=\begin{cases}
\binom{n}{x}p^x(1-p)^{n-x} & \text{ if }x=0,\dots,n\\
0 & \text{ otherwise }
\end{cases}$$ for some $p\in[0,1]$ and $n\in\N$. In this case we write $X\sim\text{Binom}(n,p)$. 
\end{defn}

\begin{defn}{Poisson Distribution}{}\\
Let $(\Omega,\mF,P)$ be a probability space. Let $X:\Omega\to\R$ be a random variable. We say that $X$ has a Poisson distribution if the probability density function of $X$ is given by $$f_X(x)=\begin{cases}
\frac{\lambda^x}{x!}e^{-x} & \text{ if }x\in\N\\
0 & \text{ otherwise }
\end{cases}$$ for some $\lambda>0$. In this case we write $X\sim\text{Poisson}(\lambda)$. 
\end{defn}

\begin{defn}{Geometric Distribution}{}\\
Let $(\Omega,\mF,P)$ be a probability space. Let $X:\Omega\to\R$ be a random variable. We say that $X$ has a geometric distribution if the probability density function of $X$ is given by $$f_X(x)=\begin{cases}
p(1-p)^x & \text{ if }x\in\N\\
0 & \text{ otherwise }
\end{cases}$$ for some $p\in[0,1]$. In this case we write $X\sim\text{Geom}(p)$. 
\end{defn}

\section{Expectation and Variance}
\subsection{Expectations}
\begin{defn}{Expectations}{}\\
Let $(\Omega,\mF,P)$ be a probability space. Let $X:\Omega\to\R$ be a random variable. Define the expectation of $X$ to be $$E[X]=\int_\Omega XdP$$
\end{defn}

\begin{lmm}{}{}\\
Let $(\Omega,\mF,P)$ be a probability space. Let $X:\Omega\to\R$ be a random variable. Then we have $$E[X]=\int_{-\infty}^\infty xf_X(x)\;dx$$
\end{lmm}

\begin{eg}{}{}\\
Let $(\Omega,\mF,P)$ be a probability space. Let $X:\Omega\to\R$ be a random variable such that $X\sim\text{Binom}(n,p)$ for some $n\in\N$ and $p\in[0,1]$. Then we have $$E[X]=np$$
\begin{proof}\\
We have
\begin{align*}
E[X]&=\sum_{k=0}^nk\binom{n}{k}p^j(1-p)^{n-k}\\
&=\sum_{k=1}^n\frac{n!\cdot k}{k!(n-k)!}p^j(1-p)^{n-k}\\
&=\sum_{k=1}^nn\binom{n-1}{k-1}p^j(1-p)^{n-k}\\
&=\sum_{i=0}^{n-1}n\binom{n-1}{i}p^{i+1}(1-p)^{n-i-1}\tag{$i=k-1$}\\
&=np\sum_{i=0}^{n-1}\binom{n-1}{i}p^i(1-p)^{n-i-1}\\
&=np
\end{align*}
\end{proof}
\end{eg}

\begin{eg}{}{}\\
Let $(\Omega,\mF,P)$ be a probability space. Let $X:\Omega\to\R$ be a random variable such that $X\sim\text{Exp}(\lambda)$ for some $\lambda>0$. Then we have $$E[X]=\lambda$$
\begin{proof}\\
We have
\begin{align*}
E[X]&=\int_0^\infty\frac{1}{\lambda}xe^{-x/\lambda}\;dx\\
&=[-xe^{-x/\lambda}]_0^\infty+\int_0^\infty e^{-x/\lambda}\;dx\\
&=\int_0^\infty e^{-x/\lambda}\;dx\\
&=\lambda
\end{align*}
\end{proof}
\end{eg}

\begin{prp}{}{} Let $(\Omega,\mathcal{F},P)$ be a probability space. Let $X,Y:\Omega\to\R$ be random variables. Then the following are true. 
\begin{itemize}
\item If $X,Y$ are random variables and $a,b\in\R$, then $$E[aX+bY]=aE[X]+bE[Y]$$
\item If $P(X\geq Y)=1$, then $$E[X]\geq E[Y]$$
\end{itemize}
\end{prp}

\begin{prp}{}{} Let $(\Omega,\mathcal{F},P)$ be a probability space. Let $X,Y:\Omega\to\R$ be random variables. Then $X,Y$ are independent if and only if $$E[g(X)h(Y)]=E[g(X)]E[h(Y)]$$ for any two functions $g,h:\R\to\R$. 
\end{prp}

\subsection{Variance and Covariance}
\begin{defn}{Variance}{}\\
Let $(\Omega,\mathcal{F},P)$ be a probability space. Let $X:\Omega\to\R$ be a random variable. Define the variance of $X$ to be $$\text{Var}(X)=E[(X-E[X])^2]$$
\end{defn}

\begin{eg}{}{}\\
Let $(\Omega,\mF,P)$ be a probability space. Let $X:\Omega\to\R$ be a random variable such that $X\sim\text{Exp}(\lambda)$. Then we have $$\text{Var}(X)=\lambda^2$$
\end{eg}

\begin{lmm}{}{}\\
Let $(\Omega,\mathcal{F},P)$ be a probability space. Let $X:\Omega\to\R$ be a random variable. Then the following are true. 
\begin{itemize}
\item $\text{Var}(X)\geq 0$. 
\item $\text{Var}(X)=0$ if and only if $P_X(E[X])=1$. 
\item $\text{Var}(X)=E[X^2]-E[X]^2$
\item $\text{Var}(aX+b)=a^2\text{Var}(X)$ for any $a,b\in\R$. 
\end{itemize}
\end{lmm}

\begin{defn}{Covariance}{}\\
Let $(\Omega,\mF,P)$ be a probability space. Let $X,Y:\Omega\to\R$ be random variables. Define the covariance of $X$ and $Y$ to be $$\text{Cov}(X,Y)=E[(X-E(X))(Y-E(Y))]$$
\end{defn}

\begin{lmm}{}{}\\
Let $(\Omega,\mF,P)$ be a probability space. Let $X:\Omega\to\R$ be a random variable. Then we have $$\text{Cov}(X,X)=\text{Var}(X)$$
\end{lmm}

\begin{prp}{}{}\\
Let $(\Omega,\mF,P)$ be a probability space. Let $X,Y:\Omega\to\R$ be random variables. Then the following are true. 
\begin{itemize}
\item $\text{Cov}(X,Y)=\text{Cov}(Y,X)$. 
\item $\text{Cov}(X,Y)=E[XY]-E[X]E[Y]$. 
\item If $X,Y$ are independent, $\text{Cov}(X,Y)=0$. 
\end{itemize}
\end{prp}

\begin{lmm}{}{}\\
Let $(\Omega,\mF,P)$ be a probability space. Let $X,Y,Z:\Omega\to\R$ be random variables. Then we have $$\text{Cov}(aX+bY,Z)=a\text{Cov}(X,Z)+b\text{Cov}(Y,Z)$$ for any $a,b\in\R$. 
\end{lmm}

\begin{prp}{}{}\\
Let $(\Omega,\mF,P)$ be a probability space. Let $X_1,\dots,X_n:\Omega\to\R$ be random variables. Then we have $$\text{Var}\left(\sum_{i=1}^{n}X_i\right)=\sum_{i=1}^n\text{Var}\left(X_i\right)+2\sum_{1\leq i<j\leq n}\text{Cov}\left(X_i,X_j\right)$$
\end{prp}

\begin{prp}{}{}\\
Let $(\Omega,\mF,P)$ be a probability space. Let $X,Y:\Omega\to\R$ be random variables. Then we have $$\abs{\text{Cov}(X,Y)}\leq\sqrt{\text{Var}(X)\text{Var}(Y)}$$
\end{prp}

\subsection{Moments and Moment Generating Functions}
\begin{defn}{Moments}{}\\
Let $(\Omega,\mF,P)$ be a probability space. Let $X:\Omega\to\R$ be a random variable. Define the $n$th moment of $X$ to be $$\mu_n=E[X^n]$$
\end{defn}

\begin{defn}{Moment Generating Function}{}\\
Let $(\Omega,\mF,P)$ be a probability space. Let $X:\Omega\to\R$ be a random variable. Define the moment generating function of $X$ to be the function $M_X:\R\to\R$ given by $$M_X(t)=E[e^{tX}]$$ whenever the expectation is defined. 
\end{defn}

\begin{prp}{}{} Assume that $M_X$ exists in a neighbourhood of $0$, that is, there exists $\epsilon>0$ such that for all $t\in(-\epsilon,\epsilon)$ we have $M_X(t)<\infty$. Then for all $k\in\mathbb{N}$ the $k$th moment of $X$ exists, and $$E[X^k]=\frac{d^k}{dt^k}M_X(t)\bigg|_{t=0}$$
\begin{proof} We have that $E[X^k]=\int_{-\infty}^\infty x^kf_X(x)\,dx$ for any continuous cummulative probability. On the other hand, 
\begin{align*}
\frac{d^k}{dt^k}M_X(t)\bigg|_{t=0}&=\frac{d^k}{dt^k}\int_{-\infty}^\infty e^{tx}f_X(x)\,dx\bigg|_{t=0}\\
&=\int_{-\infty}^\infty\frac{\partial^k}{\partial t^k}e^{tx}f_X(x)\,dx\bigg|_{t=0}\\
&=\int_{-\infty}^\infty x^ke^{tx}f_X(x)\,dx\bigg|_{t=0}\\
&=\int_{-\infty}^\infty x^kf_X(x)\,dx\\
\end{align*}
\end{proof}
\end{prp}

\begin{prp}{}{}\\
Let $(\Omega,\mF,P)$ be a probability space. Let $X:\Omega\to\R$ be a random variable. Then the following are true regarding the moment generating function. 
\begin{itemize}
\item For any $a,b\in\R$, we have $M_{aX+b}(t)=e^{tb}M_X(at)$
\item If $X$, $Y$ are independent, then $M_{X+Y}=M_X\cdot M_Y$
\end{itemize}
\end{prp}

\begin{prp}{}{} Let $X$, $Y$ be two random variables. Assume that the moment generating functions of $X$, $Y$ exists and are finite on an interval of the form $(-\epsilon,\epsilon)$. Assume further that $M_X(t)=M_Y(t)$ for all $t\in(-\epsilon,\epsilon)$. Then $X$, $Y$ have the same distribution. 
\end{prp}

\subsection{Multiple Random Variables}
\begin{defn}{Joint Probability Density and Mass Functions}{}\\
Let $(\Omega,\mF,P)$ be a probability space. Let $X,Y:\Omega\to\R$ be random variables. 
\begin{itemize}
\item Suppose that $X$ and $Y$ are discrete random variables. Define the joint probability mass function of $X$ and $Y$ to be the Radon-Nikodym derivative $$f_{X,Y}=\frac{d(X,Y)_\ast P}{d\#}$$ where $\#$ is the counting measure on $\R^2$. 
\item Suppose that $X$ and $Y$ are continuous random variables. Define the joint probability density function of $X$ and $Y$ to be the Radon-Nikodym derivative $$f_{X,Y}=\frac{d(X,Y)_\ast P}{d\mu^2}$$ where $\mu^2$ is the Lebesgue measure on $\R^2$. 
\end{itemize}
\end{defn}

As with the case of pmfs with one random variable, we have $$f_{X,Y}(x,y)=P(X=x,Y=y)$$ for all $x,y\in\R$. 

\begin{lmm}{}{}\\
Let $(\Omega,\mF,P)$ be a probability space. Let $X,Y:\Omega\to\R$ be random variables. Then the following are equivalent. 
\begin{itemize}
\item $X$ and $Y$ are independent. 
\item $f_{X,Y}=f_Xf_Y$. 
\item There exists functions $g,h:\R\to\R$ such that $f_{X,Y}=gh$. 
\end{itemize}
\end{lmm}

\begin{prp}{Law of the Unconscious Staticians}{} Let $(\Omega,\mathcal{F},P)$ be a probability space. Let $X_1,\dots,X_n:\Omega\to\R$ be random variables. Let $g:\R^n\to\R$ be a function. Then we have $$E[g\circ(X_1,\dots,X_n)]=\int_{\R^n}g(x_1,\dots,x_n)f_{(X_1,\dots,X_n)}(x_1,\dots,x_n)\;dx_1\cdots dx_n$$
\end{prp}

\section{Conditional Probability Theory}
\subsection{Conditional Probability Density and Mass Functions}
\begin{defn}{Conditional Probability Density and Mass Functions}{}\\
Let $(\Omega,\mF,P)$ be a probability space. Let $X,Y:\Omega\to\R$ be random variables. Suppose that $f_Y(y)>0$ for some $y\in\Omega$. 
\begin{itemize}
\item Suppose that $X,Y$ are discrete. Define the conditional probability mass function of $X$ given $Y=y$ to be the function $$f_{X|Y}(x,y)=\frac{f_{X,Y}(x,y)}{f_X(x)}$$
\item Suppose that $X,Y$ are continuous. Define the conditional probability density function of $X$ given $Y=y$ to be the function $$f_{X|Y}(x,y)=\frac{f_{X,Y}(x,y)}{f_X(x)}$$
\end{itemize}
\end{defn}

\subsection{Conditional Expectations}
\begin{defn}{Conditional Expectations on Subalgebras}{} Let $(\Omega,\mF,P)$ be a probability space. Let $X:\Omega\to\R$ be a random variable. Let $\mH$ be a $\sigma$-subalgebra of $\mF$. Define $E[X\;|\;\mH]:\Omega\to\R$ to be a random variable such that the following are true. 
\begin{itemize}
\item $E[X\;|\;\mH]$ is $\mH$-measurable. 
\item For any $A\in\mH$, we have $E[X\cdot 1_A]=E[E[X\;|\;\mH]\cdot 1_A]$
\end{itemize}
\end{defn}

\begin{lmm}{}{} Let $(\Omega,\mF,P)$ be a probability space. Let $X:\Omega\to\R$ be a random variable. Let $\mH$ be a $\sigma$-subalgebra of $\mF$. Then the random variable $E[X\;|\;\mH]$ exists and is unique up to almost surely equality. 
\end{lmm}

\begin{lmm}{}{} Let $(\Omega,\mF,P)$ be a probability space. Let $X,Y:\Omega\to\R$ be random variables. Let $\mH$ be a $\sigma$-subalgebra of $\mF$. Then the following are true. 
\begin{itemize}
\item Stability: If $X$ is $\mH$-measurable, then $E[XY\;|\;\mH]=XE[Y\;|\;\mH]$. 
\item Independence: If $\sigma(X)$ and $\mH$ are independent, then $E[X\;|\;\mH]=E[X]$. 
\end{itemize}
\end{lmm}

\begin{defn}{Conditional Expectation on Random Variables}{} Let $(\Omega,\mF,P)$ be a probability space. Let $X,Y:\Omega\to\R$ be random variables. Define the conditional expectation of $X$ on $Y$ to be $$E[X\;|\;Y]=E[X\;|\;\sigma(Y)]$$
\end{defn}

\begin{lmm}{}{} Let $(\Omega,\mF,P)$ be a probability space. Let $X,Y:\Omega\to\R$ be random variables. Then we have $$E[X\;|\;Y](\omega)=E[X\;|\;Y=Y(\omega)]=\int_{-\infty}^\infty xf_{X|Y}(x,Y(\omega))\;dx$$
\end{lmm}

\section{Convergence of Random Variables}
\subsection{Different Notions of Convergences}
\begin{defn}{Convergence in Mean Square}{} We say that a sequence of random variables $X_1,X_2,\dots$ converges in mean square to a random variable $X$ if $$\lim_{n\to\infty}E[(X_n-X)^2]=0$$
\end{defn}

\begin{defn}{Convergence in Probability}{} We say that a sequence of random variables $X_1,X_2,\dots$ converges in probability to a random variable $X$ if for every $\epsilon>0$, we have $$\lim_{n\to\infty}P(\abs{X_n-X}>\epsilon)=0$$
\end{defn}

\begin{defn}{Convergence in Distribution}{} We say that a sequence of random variables $X_1,X_2,\dots$ converges in distribution to a random variable $X$ if $$\lim_{n\to\infty}F_{X_n}(x)=F_X(x)$$ for every $x$ in the set $C=\{x\in\mathbb{R}:F_X\text{ is continuous at } x\}$. 
\end{defn}

\begin{prp}{}{} Let $(\Omega,\mF,P)$ be a probability space. Let $X_n:\Omega\to\R$ be a sequence of random variables for $n\in\N\setminus\{0\}$. Let $X:\Omega\to\R$ also be a random variable. Then the following are true. 
\begin{itemize}
\item If $X_n$ converges in mean square to $X$, then $X_n$ converges in probability to $X$. 
\item If $X_n$ converges in probability to $X$, then $X_n$ converges in distribution to $X$. 
\end{itemize}
\end{prp}

\subsection{Law of Large Numbers}
\begin{thm}{Markov Inequality}{} Let $(\Omega,\mF,P)$ be a probability space. Let $X:\Omega\to\R$ be a random variable. If $E[X]<\infty$, then we have $$P(\abs{X}\geq a)\leq\frac{E[X]}{a}$$ for any $a>0$. 
\end{thm}

\begin{thm}{Chebyshev Inequality}{} Let $(\Omega,\mF,P)$ be a probability space. Let $X:\Omega\to\R$ be a random variable. If $E[X^2]<\infty$, then we have $$P(\abs{X}\geq a)\leq\frac{E[X^2]}{a}$$ for any $a>0$. 
\end{thm}

\begin{thm}{Weak law of large numbers}{} Let $(\Omega,\mF,P)$ be a probability space. Let $X_n:\Omega\to\R$ for $n\in\N\setminus\{0\}$ be a sequence of independently identically distributed random variables with mean $\mu$. Let $S_n=\frac{1}{n}\sum_{i=1}^nX_i$. Then we have $$\lim_{n\to\infty}P(\abs{S_n-\mu}>\varepsilon)=0$$ for all $\varepsilon>0$. In other words, $(S_n)_{n\in\N\setminus\{0\}}$ converges in probability to $\mu$. 
\end{thm}

\begin{thm}{Law of large numbers in mean square}{} Let $X_1,X_2,\dots$ be a sequence of independent random variable, each with mean $\mu$ and variance $\sigma^2$. Then $$\lim_{n\to\infty}\frac{X_1+\dots+X_n}{n}\to\mu$$ in mean square. 
\end{thm}

\subsection{Central Limit Theorem}
\begin{thm}{Central Limit Theorem}{} Let $(\Omega,\mF,P)$ be a probability space. Let $X_n:\Omega\to\R$ be a sequence of independent and identically distributed random variables for $n\in\N\setminus\{0\}$, each with mean $\mu$ and variance $\sigma^2\neq 0$. Let $S_n=X_1+\dots+X_n$. Then the standardized version of $S_n$, $$Z_n=\frac{S_n-E(S_n)}{\sqrt{\text{Var}(S_n)}}=\frac{S_n-n\mu}{\sigma\sqrt{n}}$$ converges in distribution as $n\to\infty$ to a Gaussian random variable with mean $0$ and variance $1$. That is, $$\lim_{n\to\infty}P(Z_n\leq x)=\lim_{n\to\infty}F_{Z_n}(x)=F_Y(y)=\int_{-\infty}^{x}-\frac{1}{\sqrt{2\pi}}e^{-y^2/2}$$
\end{thm}

\begin{defn}{The Space of Random Variables}{}\\
Let $(\Omega,\mathcal{F},P)$ be a probability space. Define the $\R$-vector space of (real-valued) random variables to be the set $$L^2(\Omega)=\{X:\Omega\to\R\;|\;X\text{ is a random variable and }E[X^2]<\infty\}$$ together with pointwise addition and scalar multiplication. 
\end{defn}

\begin{prp}{}{}\\
Let $(\Omega,\mathcal{F},P)$ be a probability space. Then $L^2(\Omega)$ is a Hilbert space with innner product given by $$\langle X,Y\rangle=E[XY]$$
\end{prp}
