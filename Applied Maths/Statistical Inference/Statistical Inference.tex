\section{Fundamentals of Statistics}
\subsection{Statistical Models}
\begin{defn}{Statistical Models}{}\\
Let $(\Omega,\mF)$ be a measurable space. A statistical model is a family of probability measures $$\mP=\{P_\theta:\mF\to[0,1]\;|\;\theta\in I\}$$ for some indexing set $I$. 
\begin{itemize}
\item $I$ is called the parameter space of the statistical model. 
\item We denote the corresponding probability distribution by $f_\theta$. 
\end{itemize}
\end{defn}

Let $X:\Omega\to\R$ be a random variable. For each probability measure $P_\theta$, the probability density function $f_X$ also depends on $\theta$, and we obtain a family of probability density functions $\{f_\theta\;|\;\theta\in I\}$. Some authors use this as the definition of statistical models, but our definition naturally implies this alternative one. 

\begin{defn}{Parametric Models}{}\\
Let $(\Omega,\mF)$ be a measurable space. Let $\mP$ be a statistical model indexed by $I$. We say that $\mP$ is parametric if $I\subseteq\R^k$ for some $k\in\N$. 
\end{defn}

\subsection{Random Samples and Statistics}
\begin{defn}{Random Samples}{}\\
Let $(\Omega,\mF,P)$ be a probability space. Let $X_1,\dots,X_n:\Omega\to\R$ be random variables. We say that they are a random sample of size $n$ from the population $f$ if the following are true. 
\begin{itemize}
\item $X_1,\dots,X_n$ are pairwise independent. 
\item The probability density (mass) function is given by $f_{X_i}=f$. 
\end{itemize}
\end{defn}

The fact that their probability density functions are equal imply that the random variables are identically distributed. 

\begin{defn}{Statistics}{}\\
Let $(\Omega,\mF,P)$ be a probability space. Let $X_1,\dots,X_n:\Omega\to\R$ be a random sample. A statistic of the random sample is a random variable $T(X_1,\dots,X_n)$ for some function $T:\R^n\to\R$. 
\end{defn}

\begin{defn}{Sample Mean}{}\\
Let $(\Omega,\mF,P)$ be a probability space. Let $X_1,\dots,X_n:\Omega\to\R$ be a random sample. Define the sample mean of the random sample to be the statistic defined by $$\overline{X}=\frac{1}{n}\sum_{i=1}^nX_i$$
\end{defn}

By the weak law of large numbers, we know that $\overline{X}$ converges in mean square to the mean of the random sample. 

\begin{defn}{Sample Variance}{}\\
Let $(\Omega,\mF,P)$ be a probability space. Let $X_1,\dots,X_n:\Omega\to\R$ be a random sample. Define the sample variance of the random sample to be the statistic defined by $$S^2=\frac{1}{n-1}\sum_{i=1}^n(X_i-\overline{X})^2$$
\end{defn}

\begin{defn}{Sample Standard Deviation}{}\\
Let $(\Omega,\mF,P)$ be a probability space. Let $X_1,\dots,X_n:\Omega\to\R$ be a random sample. Define the sample standard deviation of the random sample to be the statistic defined by $$S=\sqrt{S^2}$$
\end{defn}

\section{Principles of Data Reduction}
\subsection{Sufficiency Principle}
A sufficient statistic $T$ is a statistic where, informally speaking, all information about $P$ must come from $T$. 

\begin{defn}{Sufficient Statistics}{}\\
Let $(\Omega,\mF)$ be a measurable space. Let $\mP=\{P_\theta\;|\;\theta\in I\}$ be a parametric model. Let $X:\Omega\to\R^n$ be a random sample with distribution $f_\theta$. Let $T:\R^n\to\R$ be a statistic. We say that $T$ is sufficient for $I$ if the conditional probability density function $f_{X_1,\dots,X_n|T}$ is a constant function with respect to $\theta$. 
\end{defn}

\begin{eg}{}{}\\
Let $\Omega,\mF,P)$ be a probability space. Let $X_1,\dots,X_n:\Omega\to\R$ be a random sample of Bernoulli distribution with parameter $\theta\in(0,1)$ (Note that for each $\theta$, $P_{X_i}=f_{X_i}$ is different since the formula for $f_{X_i}$ depends on $\theta$). Then $T=\sum_{i=1}^nX_i$ is a sufficient statistic for $\theta$. 
\begin{proof}\\
Let $\theta\in\R$. We use the above lemma. We have
\begin{align*}
P_\theta(\bold{X}=\bold{x}\;|\;T(\bold{X})=T(\bold{x}))&=\frac{f_{X_1,\dots,X_n}(x_1,\dots,x_n)}{f_{T(X_1,\dots,X_n)}(x_1,\dots,x_n)}\\
&=\frac{\prod_{i=1}^nf_{X_i}(x_i)}{f_{X_1+\dots+X_n}(x_1,\dots,x_n)}\\
&=\frac{\prod_{i=1}^n\theta^{x_i}(1-\theta)^{1-x_i}}{\binom{n}{\sum_{i=1}^nx_i}\theta^{\sum_{i=1}^nx_i}(1-\theta)^{n-\sum_{i=1}^nx_i}}\\
&=\frac{\theta^{\sum_{i=1}^nx_i}(1-\theta)^{\sum_{i=1}^nx_i}}{\binom{n}{\sum_{i=1}^nx_i}\theta^{\sum_{i=1}^nx_i}(1-\theta)^{n-\sum_{i=1}^nx_i}}\\
&=\frac{1}{\binom{n}{\sum_{i=1}^nx_i}}
\end{align*}
\end{proof}
\end{eg}

\begin{thm}{Neyman-Fisher Factorization Theorem}{}\\
Let $(\Omega,\mF)$ be a measurable space. Let $\mP=\{P_\theta\;|\;\theta\in I\}$ be a parametric model. Let $X:\Omega\to\R^n$ be a random sample with distribution $f_\theta$. Let $T:\R^n\to\R$ be a statistic. Then $T$ is sufficient for $I$ if and only if there exists $g:\R\times I\to\R$ and $h:\R^n\to\R$ such that $$f_\theta(x)=g(T(x),\theta)h(x)$$ for all $x\in\im(X)$ and all $\theta\in I$. 
\end{thm}

\begin{defn}{Minimal Sufficient Statistics}{}\\
Let $(\Omega,\mF,P)$ be a probability space. Let $\{D_\theta\;|\;\theta\in I\}$ be a family of distributions. Let $X_\theta:\Omega\to\R^n$ be a random sample with distribution $D_\theta$. Let $T$ be a sufficient statistic of the random sample. We say that $T$ is minimal if for any other sufficient statistic $T'$, $T'(x)=T'(y)$ implies $T(x)=T(y)$. 
\end{defn}

\begin{defn}{Ancillary Statistics}{}\\
Let $(\Omega,\mF,P)$ be a probability space. Let $\{D_\theta\;|\;\theta\in I\}$ be a family of distributions. Let $X_\theta:\Omega\to\R^n$ be a random sample with distribution $D_\theta$. Let $T$ be a statistic of the random sample. We say that $T$ is ancillary if $T$ is a constant function with respect to $\theta$. 
\end{defn}

\section{Point Estimators}
Suppose some real life problem is modelling statistically by the probability space $(\Omega,\mF,P)$. To probe what $P$ looks like, we sample the space using random variables $X_1,\dots,X_n:\Omega\to\R$. Based on the sample, we may infer that $P$ may have parameter $\widehat{\theta}$ (naturally $\widehat{\theta}$ should be a function of $X_1,\dots,X_n$). We may then need to evaluate how accurate $\widehat{\theta}$ is compared to the actual parameter $\theta$. \\

There are two main branches in estimator theory: We need different methods for finding estimators, and we need to evaluate how accurate the estimators are. \\

There are also two types of estimators: Point estimators and interval estimators. 

\subsection{Basics of Point Estimators}
\begin{defn}{Point Estimators}{}\\
Let $(\Omega,\mF)$ be a measurable space. Let $\mP=\{P_\theta\;|\;\theta\in I\}$ be a parametric model. Let $X:\Omega\to\R^n$ be a random sample with distribution $f_\theta$. Let $T:\R^n\to\R$ be a statistic. We say that $T$ is a point estimator if $\im(T)\subseteq I$. 
\end{defn}

Estimators: A way to predict what the true parameter is (they are random variables because it depends on the actual measurements). 
Estimate: The value of an estimator of a given measurement. 
We also need a way to evaluate how good the estimators are (their input is a random variable and output is a real number). (In particular, one way to get a good estimator is to first find an method of evaluating estimators, then minimize it)

\subsection{Evaluating the Accuracy of Point Estimators}
\begin{defn}{Bias of a Point Estimator}{}\\
Let $(\Omega,\mF)$ be a measurable space. Let $\mP=\{P_\theta\;|\;\theta\in I\}$ be a parametric model. Let $X:\Omega\to\R^n$ be a random sample with distribution $f_\theta$. Let $\widehat{\theta}:\R^n\to I$ be a point estimator. Define the bias of $\widehat{\theta}$ to be $$\text{Bias}(\widehat{\theta})=E[\widehat{\theta}]-\theta$$
\end{defn}

\begin{defn}{Unbiased Point Estimators}{}\\
Let $(\Omega,\mF)$ be a measurable space. Let $\mP=\{P_\theta\;|\;\theta\in I\}$ be a parametric model. Let $X:\Omega\to\R^n$ be a random sample with distribution $f_\theta$. Let $\widehat{\theta}:\R^n\to I$ be a point estimator. We say that $\widehat{\theta}$ is unbiased if $\text{Bias}(\widehat{\theta})=0$. 
\end{defn}

\begin{defn}{Mean Squared Error}{}\\
Let $(\Omega,\mF)$ be a measurable space. Let $\mP=\{P_\theta\;|\;\theta\in I\}$ be a parametric model. Let $X:\Omega\to\R^n$ be a random sample with distribution $f_\theta$. Let $\widehat{\theta}:\R^n\to I$ be a point estimator. Define the mean squared error of $\widehat{\theta}$ to be $$\text{MSE}(\widehat{\theta})=E[(\widehat{\theta}-\theta)^2]$$
\end{defn}

\begin{thm}{The Bias Variance Trade Off}{}\\
Let $(\Omega,\mF,P)$ be a probability space. Let $X_1,\dots,X_n:\Omega\to\R$ be a random sample. Let $\widehat{\theta}$ be a point estimator of the random sample. Then we have $$\text{MSE}(\widehat{\theta})=\text{Var}(\widehat{\theta})+(\text{Bias}(\widehat{\theta}))^2$$
\end{thm}

\subsection{Constructing Point Estimators}
\begin{defn}{Method of Moments}{}\\
Let $(\Omega,\mF,P)$ be a probability space. Let $J_1,\dots,J_k\subseteq\R$ be indexing sets. Let $\{D_{\theta_1,\dots,\theta_k}\;|\;\theta_i\in J_i\}$ be a family of distributions. Let $X_{\theta_1,\dots,\theta_k}:\Omega\to\R^n$ be a random sample with distribution $D_{\theta_1,\dots,\theta_k}$ and probability distribution function $f_{\theta_1,\dots,\theta_k}$. Define the methods of moments estimator to be the statistics $\widehat{\theta}_1,\dots,\widehat{\theta}_k$ given by the solving the system of equations $$\frac{1}{n}\sum_{j=1}^n(X_{\theta_1,\dots,\theta_k})_j^i=E[D_{\widehat{\theta}_1,\dots,\widehat{\theta}_k}^i]$$ for $1\leq i\leq k$. 
\end{defn}

\begin{eg}{}{}\\
Let $(\Omega,\mF,P)$ be a probability space. Let $X:\Omega\to\R^n$ be a random sample with normal distribution $X_i\sim\text{Normal}(\theta_1,\theta_2)$. The method of moments estimator for $X$ is given by solving $$\frac{1}{n}\sum_{j=1}^n(X_{\theta_1,\dots,\theta_k})_j=E[\text{Normal}(\widehat{\theta}_1,\widehat{\theta}_2)]\;\;\;\text{ and }\;\;\;\frac{1}{n}\sum_{j=1}^n(X_{\theta_1,\dots,\theta_k})_j^2=E[\text{Normal}(\widehat{\theta}_1,\widehat{\theta}_2)^2]$$ which we may rewrite as $$\overline{X_{\theta_1,\dots,\theta_k}}=\widehat{\theta}_1\;\;\;\text{ and }\;\;\;\frac{1}{n}\sum_{j=1}^n(X_{\theta_1,\dots,\theta_k})_j^2=\widehat{\theta}_1^2+\widehat{\theta}_2^2$$ Omitting the dependency of the random variables on the parameters, we get $$\widehat{\theta}_1=\overline{X}\;\;\;\text{ and }\;\;\;\widehat{\theta}_2=\frac{1}{n}\sum_{j=1}^n(X_j-\overline{X})^2$$
\end{eg}

We are lucky in the above example because the estimator accurately recovers the actual parameters. But in general, this is not true. 

\begin{eg}{}{}\\
Let $(\Omega,\mF,P)$ be a probability space. Let $X:\Omega\to\R^n$ be a random sample with binomial distribution $X_i\sim\text{Binom}(\theta_1,\theta_2)$. The method of moments estimator for $X$ is given by solving $$\frac{1}{n}\sum_{j=1}^nX_j=E[\text{Binom}(\widehat{\theta}_1,\widehat{\theta}_2)]\;\;\;\text{ and }\;\;\;\frac{1}{n}\sum_{j=1}^nX_j^2=E[\text{Binom}(\widehat{\theta}_1,\widehat{\theta}_2)^2]$$ which we may rewrite as $$\overline{X}=\widehat{\theta}_1\widehat{\theta}_2\;\;\;\text{ and }\;\;\;\frac{1}{n}\sum_{j=1}^nX_j^2=\widehat{\theta}_1\widehat{\theta}_2(1-\widehat{\theta}_2)+\widehat{\theta}_1^2\widehat{\theta}_2^2$$ Solving the simultaneous equation gives $$\widehat{\theta}_1=\frac{\overline{X}^2}{\overline{X}-\frac{1}{n}\sum_{j=1}^n(X_j-\overline{X})^2}\;\;\;\text{ and }\;\;\;\widehat{\theta}_2=\frac{\overline{X}-\frac{1}{n}\sum_{j=1}^n(X_j-\overline{X})^2}{\overline{X}}=\frac{\overline{X}}{\widehat{\theta}_1}$$
\end{eg}

\begin{defn}{Maximum Likelihood Estimators}{}\\
Let $(\Omega,\mF)$ be a measurable space. Let $\mP=\{P_\theta\;|\;\theta\in I\}$ be a parametric model. Let $X:\Omega\to\R^n$ be a random sample with distribution $f_\theta$. Let $\widehat{\theta}:\R^n\to I$ be a point estimator. We say that $\widehat{\theta}$ is a maximum likelihood estimator if there exists $x\in\R^n$ such that $$L(\widehat{\theta}(x))(x)=\max\{L(\theta)(x):\theta\in I\}$$
\end{defn}

\begin{lmm}{}{}\\
Let $(\Omega,\mF)$ be a measurable space. Let $\mP=\{P_\theta\;|\;\theta\in I\}$ be a parametric model. Let $X:\Omega\to\R^n$ be a random sample with distribution $f_\theta$. If $L$ is differentiable, then the maximum likelihood estimator exists and is given by the root of the equation $$\nabla L=0$$
\end{lmm}

\section{Hypothesis Testing}
Let $(\Omega,\mF)$ be a measurable space. Let $\mP=\{P_\theta\;|\;\theta\in I\}$ be a parametric model. We are interested in the following question: Suppose we know that the parameter lies in one of two set $H_0,H_1\subseteq I$ with $H_0\cap H_1=\emptyset$. We wish to use random samples $X:\Omega\to\R^n$ to determine whether it lies in $H_0$ or $H_1$. 

\subsection{Basics of Hypothesis Testing}
\begin{defn}{Hypothesis}{}\\
Let $(\Omega,\mF)$ be a measurable space. Let $\mP=\{P_\theta\;|\;\theta\in I\}$ be a parametric model. A hypothesis of the parameter are two subsets $H_0,H_1\subseteq I$ such that $H_0\cap H_1=\emptyset$. 
\end{defn}

\begin{defn}{Test Functions}{}\\
Let $(\Omega,\mF)$ be a measurable space. Let $\mP=\{P_\theta\;|\;\theta\in I\}$ be a parametric model. Let $X:\Omega\to\R^n$ be a random sample with distribution $f_\theta$. A test function is a function $\delta:\im(X)\to\{0,1\}$. 
\end{defn}

A test function takes the value $0$ when the sample indicates that the parameter lies in $H_0$, and vice versa for $H_1$. A test function usually take values depending on a certain condition, so they are usually of the form $$\delta(x)=\begin{cases}
1 & \text{ if } T(x)\in C\\
0 & \text{ otherwise }
\end{cases}$$ for some statistic $T$ and some set $C\in\im(T)$. \\

\subsection{Evaluating the Accuracy of Hypothesis Tests}
Similar to point estimators, we also need to evaluate how accurate the test function $\delta$ is. In the case of hypothesis testing, we can easily classify the possible errors. 
\begin{itemize}
\item True parameter: $0$, decision: $0$. There is no error. 
\item True parameter: $0$, decision: $1$. We call this type I error. 
\item True parameter: $1$, decision: $0$. We call this type II error. 
\item True parameter $1$, decision: $1$. There is no error. 
\end{itemize}

\begin{defn}{Error Probabilities}{}\\
Let $(\Omega,\mF)$ be a measurable space. Let $\mP=\{P_\theta\;|\;\theta\in I\}$ be a parametric model. Let $X:\Omega\to\R^n$ be a random sample with distribution $f_\theta$. Let $H_0,H_1\subseteq I$ be a hypothesis. Let $\delta:\im(X)\to\{0,1\}$ be a test function. 
\begin{itemize}
\item Define the type I error probability to be the map $h_0:H_0\to[0,1]$ given by $$h_0(\theta)=P_\theta(\delta=1)$$
\item Define the type II error probability to be the map $h_1:H_1\to[0,1]$ given by $$h_1(\theta)=P_\theta(\delta=0)$$
\end{itemize}
\end{defn}

\subsection{Constructing Test Functions}
It is not possible to simultaneously minimize $h_0$ and $h_1$. Therefore one way to find a good test function is to put emphasis on $H_0$ before $H_1$. In practice most scientific experiments have a more important error to prevent from having false positives. 

\begin{defn}{The Neyman-Pearson Framework}{}\\
Let $(\Omega,\mF)$ be a measurable space. Let $\mP=\{P_\theta\;|\;\theta\in I\}$ be a parametric model. Let $X:\Omega\to\R^n$ be a random sample with distribution $f_\theta$. Let $H_0,H_1\subseteq I$ be a hypothesis. The Neyman-Pearson framework is the following algorithm that ???
\begin{enumerate}
\item Fix $\alpha\in(0,1)$ and call it the significance level of the test. 
\item Define the class of test functions that respect the level $$\mD(H_0,\alpha)=\{\delta:\im(X)\to\{0,1\}\;|\;\sup_{\theta\in H_0}P_\theta(\delta=1)\leq\alpha\}$$
\item Find the test function $\delta\in\mD(H_0,\alpha)$ that minimizes $h_1$. 
\end{enumerate}
\end{defn}

\subsection{Hypothesis Testing for One Dimensional Parameters}
\begin{defn}{Simple and Lateral Hypothesis}{}\\
Let $(\Omega,\mF)$ be a measurable space. Let $\mP=\{P_\theta\;|\;\theta\in I\}$ be a parametric model such that $I\subseteq\R$. Let $X:\Omega\to\R^n$ be a random sample with distribution $f_\theta$. Let $H_0,H_1\subseteq I$ be a hypothesis. 
\begin{itemize}
\item We say that the hypothesis is a simple vs simple if $H_0,H_1$ are one point sets. 
\item We say that the hypothesis is unilateral if $H_0=\{\theta\in I\;|\;\theta\leq\theta_0\}$ and $H_1=I\setminus H_0$ or $H_0=\{\theta\in I\;|\;\theta\geq\theta_0\}$ and $H_1=I\setminus H_0$ for some $\theta_0\in I$. 
\item We say that the hypothesis is simple vs bilateral if $H_0$ is a one point set and $H_1=I\setminus H_0$. 
\end{itemize}
\end{defn}

\begin{defn}{p-Value}{}\\
Let $(\Omega,\mF)$ be a measurable space. Let $\mP=\{P_\theta\;|\;\theta\in I\}$ be a parametric model such that $I\subseteq\R$. Let $X:\Omega\to\R^n$ be a random sample with distribution $f_\theta$. Let $H_0,H_1\subseteq I$ be a hypothesis that it is either simple vs simple, unilateral or simple vs bilateral. Let $\delta:\im(X)\to\{0,1\}$ be a test function of one of the following forms: $$\delta=1_{T>q_{1-\alpha}}\;\;\;\text{ or }\;\;\;\delta=1_{T\leq q_\alpha}$$ where $T$ is some statistic and $q_z$ is the $z$-quantile of the distribution $G_0(t)=P_{\theta_0}(T\leq t)$. Define the $p$-value of $X$ to be $$\inf\{\alpha\in(0,1)\;|\;\delta_\alpha=1\}$$
\end{defn}

\subsection{Basics of Interval Estimators}
\begin{defn}{Two Sided Confidence Intervals}{}\\
Let $(\Omega,\mF)$ be a measurable space. Let $\mP=\{P_\theta\;|\;\theta\in I\}$ be a parametric model such that $I\subseteq\R$. Let $X:\Omega\to\R^n$ be a random sample with distribution $f_\theta$. Let $\alpha\in(0,1)$. Let $L,U:\R^n\to\R$ be two statistics such that $$\inf_{\theta\in I}P_\theta(L\leq\theta\leq U)\geq 1-\alpha$$ Define the two sided confidence interval for $\theta$ with confidence level $1-\alpha$ to be the interval $$[L(X),U(X)]\subseteq\R$$
\end{defn}

Let $X,Y$ be random variables. Regression is about understanding the relationship of $Y$ given $X$. Clearly this is a question in conditional probability: If I observed $X=x$, what can I say about the distribution of $Y$? \\

Suppose we sample $p$ data points $(X_1,Y_1),\dots,(X_p,Y_p)$, i.e. we assume we have a random sample with distribution $(X,Y)$. We would like to assume that there is some function $f$ such that when $X_i=x_i$ and $Y_i=y_i$ are observed, then we have $y_i=f(x_i)$. In reality, if we also consider the fact that our observations may have measurement errors or inherent randomness, then we find that $y_i\in f(x_i)+\varepsilon$ where $\varepsilon$ is some random variable representing the measurement error / inherent randomness. Hence $Y_i=f(x_i)+\varepsilon$ is really the starting equation for us to use to find what $f$ is. Assume that $E[\varepsilon]=0$, that is on average our measurement error is $0$, we find that $E[Y_i]=f(x_i)$. Since this is a function of $x_i$, we may rewrite it as $E[Y_i\;|\;X_i=x_i]=f(x_i)$. This means that we are asking the question of on average, what is the relationship between the outcome $Y_i$ given $X_i=x_i$. \\

The above assumes that the relationship between $Y$ and $X$ is given by a function. In general, there are other types of relationships that we may ask for e.g. variance regression: asking for how the variance of $Y$ is related to $X$ instead of how the outcome $Y$ is related to $X$. 

\begin{defn}{Regression Models}{}\\
Let $(\Omega,\mF,P)$ be a probability space. Let $(X_1,Y_1),\dots,(X_p,Y_p):\Omega\to\R^n\times\R$ be a random sample. A regression model is a parameteric statistical model $\mP$ on the random variable $Y_i\;|\;X_i=x$. 
\end{defn}

Notice that $i$ does not matter in the regression model because $(X_i,Y_i)$ are independently identically distributed. 

\begin{defn}{Linear Regression Models}{}\\
Let $(\Omega,\mF,P)$ be a probability space. Let $(X_1,Y_1),\dots,(X_p,Y_p):\Omega\to\R^n\times\R$ be a random sample. Let $\mP$ be a regression model parameterized by $I\subseteq\R^k$. We say that $\mP$ is linear if $E[Y_i\;|\;X_i=x_i]$ is a linear function of the parameter $\theta\in I$. 
\end{defn}
