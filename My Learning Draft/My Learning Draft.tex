\subsection{Stanley-Reisner Rings}
\begin{thm}{}{}\\
Let $R$ be a commutative ring. Let $n\in\N$. There is a bijection $$\substack{\text{Abstract Simplicial}\\\text{Complexes on }\{1,\dots,n\}}\;\;\overset{1:1}{\leftrightarrow}\;\;\substack{\text{Squarefree monomial}\\\text{ideals of }R[x_1,\dots,x_n]}$$ given by sending a simplicial set $S$ to the ideal $\left\langle\prod_{j\in J}x_j\;|\;J\in\mP(\{1,\dots,n\})\setminus S\right\rangle$. 
\end{thm}

\begin{defn}{Stanley-Reisner Rings}{}\\
Let $R$ be a commutative ring. Let $n\in\N$. Let $S$ be an abstract simplicial complex on $\{1,\dots,n\}$. Define the Stanley-Reisner ring of $S$ to be $$R[S]=\frac{k[x_1,\dots,x_n]}{\left\langle\prod_{j\in J}x_j\;|\;J\in\mP(\{1,\dots,n\})\setminus S\right\rangle}$$ The ideal $I_S=\left\langle\prod_{j\in J}x_j\;|\;J\in\mP(\{1,\dots,n\})\setminus S\right\rangle$ is called the Stanley-Reisner ideal of $S$. 
\end{defn}

\begin{eg}{}{}\\
Let $R$ be a commutative ring. The Stanley-Reisner ring of the standard $n$-simplex $\Delta^n$ is given by $$R[\Delta^n]=k[x_1,\dots,x_n]$$
\end{eg}

\begin{eg}{}{}\\
Let $R$ be a commutative ring. The Stanley-Reisner ring of the abstract simplicial complex $S=\mP(\{1,2,3\})\cup\mP(\{1,3,4\})\cup\mP(\{2,4\})$ over $\{1,2,3,4\}$ is given by $$R[S]=\frac{k[x_1,x_2,x_3,x_4]}{(x_1x_2x_4,x_2x_3x_4)}$$
\end{eg}

\subsection{The Hilbert Series of Stanley-Reisner Rings}
\begin{thm}{Stanley}{}\\
Let $k$ be a field. Let $n\in\N$. Let $S$ be an abstract simplicial complex on $\{1,\dots,n\}$. Let $(f_0,\dots,f_n)$ be the face vector of $S$. Let $(h_0,\dots,h_n)$ be the $h$-vector of $S$. Then the Hilbert series of $k[S]$ is given by $$HS_{k[S]}(t)=\sum_{i=0}^{n+1}f_{k-1}\left(\frac{t^2}{1-t^2}\right)^k=\frac{1}{(1-t^2)^{n+1}}\sum_{i=0}^{n+1}h_it^{2i}$$
\end{thm}

\begin{eg}{}{}|\
Let $\Delta^n$ be the standard $n$-simplex. Then the following are true. 
\begin{itemize}
\item The Hilbert series for the Stanley-Reisner ring of $\Delta^n$ is given by $$HS_{k[\Delta^n]}(t)=\frac{1}{(1-t^2)^{n+1}}$$
\item The Hilbert series for the Stanley-Reisner ring of $\partial\Delta^n$ is given by $$HS_{k[\partial\Delta^n]}(t)=\frac{1}{(1-t^2)^{n+1}}\sum_{i=0}^{n+1}t^{2i}$$
\end{itemize}
\end{eg}

\subsection{The Homotopy Theory of Polyhedral Products}
\begin{defn}{Polyhedral Products}{}\\
Let $\mK$ be a simplicial complex on $\{1,\dots,n\}$. Let $(\bold{X},\bold{A})=\{(X_1,A_1),\dots,(X_n,A_n)\}$ be pairs of spaces. 
\begin{itemize}
\item Let $\sigma\in\mK$. Define $$(\bold{X},\bold{A})^\sigma=\left\{(x_1,\dots,x_n)\in\prod_{i=1}^nX_i\;\bigg{|}\;x_i\in A_i\text{ for }i\notin I\right\}$$
\item Define the polyhedral product of $(\bold{X},\bold{A})$ corresponding to $\mK$ to be $$(\bold{X},\bold{A})^\mK=\bigcup_{I\in\mK}(\bold{X},\bold{A})^\sigma$$
\end{itemize}
\end{defn}

\begin{prp}{}{}\\
Let $\mK$ be a simplicial complex on $\{1,\dots,n\}$. Let $(\bold{X},\bold{A})=\{(X_1,A_1),\dots,(X_n,A_n)\}$ be pairs of spaces. Then we have $$(\bold{X},\bold{A})^\mK=\colim_{\sigma\in\text{Cat}(\mK)}(\bold{X},\bold{A})^\sigma$$
\end{prp}

\begin{prp}{}{}\\
Let $\mK$ be a simplicial complex on $\{1,\dots,n\}$. Let $(\bold{X},\bold{A})$ and $(\bold{Y},\bold{B})$ be $n$ pairs of spaces. Let $f_i:(X_i,A_i)\to(Y_i,B_i)$ be a map of pair for $1\leq i\leq n)$. Then $(f_1,\dots,f_n)$ induces a map $$(\bold{X},\bold{A})^\mK\to(\bold{Y},\bold{B})^\mK$$
\end{prp}

\begin{prp}{}{}\\
Let $\mK,\mL$ be a simplicial complex on $\{1,\dots,n\}$. Let $(\bold{X},\bold{A})=\{(X_1,A_1),\dots,(X_n,A_n)\}$ be pairs of spaces. Let $f_i:(X_i,A_i)\to(Y_i,B_i)$ be a map of pair for $1\leq i\leq n)$. Then we have $$(\bold{X},\bold{A})^{\mK\ast\mL}=(\bold{X},\bold{A})^\mK\times(\bold{X},\bold{A})^\mL$$
\end{prp}

\begin{defn}{Polyhedral Smash Products}{}\\
Let $\mK$ be a simplicial complex on $\{1,\dots,n\}$. Let $(\bold{X},\bold{A})$ be pairs of spaces. 
\begin{itemize}
\item Let $\sigma\in\mK$. Define $$(\bold{X},\bold{A})^{\wedge\sigma}=\left\{(x_1,\dots,x_n)\in\bigwedge_{i=1}^nX_i\;\bigg{|}\;x_i\in A_i\text{ for }i\notin I\right\}$$
\item Define the polyhedral smash product of $(\bold{X},\bold{A})$ corresponding to $\mK$ to be $$(\bold{X},\bold{A})^\mK=\bigcup_{I\in\mK}(\bold{X},\bold{A})^{\wedge\sigma}$$
\end{itemize}
\end{defn}

\begin{thm}{Bahri, Bendersky, Cohen, Gitler}{}\\
Let $\mK$ be a simplicial complex on $\{1,\dots,n\}$. Let $(\bold{X},\bold{A})$ be $n$-pairs of CW complexes. Suppose that the inclusion maps $A_j\to X_j$ are null homotopic for $1\leq j\leq n$. Then there is a homotopy equivalence $$(\bold{X},\bold{A})^{\wedge\mK}\simeq\bigvee_{\sigma\in\mK}\abs{\text{lk}_\mK(\sigma)}\ast(\bold{X}_J,\bold{A}_J)^{\wedge\sigma}$$
\end{thm}

\begin{thm}{Bahri, Bendersky, Cohen, Gitler}{}\\
Let $\mK$ be a simplicial complex on $\{1,\dots,n\}$. Let $(\bold{X},\bold{A})$ be $n$-pairs of CW complexes. Then there is a homotopy equivalence $$\Sigma(\bold{X},\bold{A})^\mK\simeq\Sigma\left(\bigvee_{J\subseteq\{1,\dots,n\}}(\bold{X}_J,\bold{A}_J)^{\wedge\mK_J}\right)$$ that is natural in $(\bold{X},\bold{A})$, where $(\bold{X}_J,\bold{A}_J)=\{(X_j,A_j)\;|\;j\in J\}$. 
\end{thm}

\begin{crl}{}{}\\
Let $\mK$ be a simplicial complex on $\{1,\dots,n\}$. Let $(\bold{X},\bold{A})$ be $n$-pairs of CW complexes. Suppose that $A_i$ are null homotopic for all $i$. Then there is a homotopy equivalence $$\Sigma(\bold{X},\bold{A})^\mK\simeq\Sigma\left(\bigvee_{\sigma\in\mK}\bold{X}^{\wedge\sigma}\right)$$
\end{crl}

\begin{crl}{}{}\\
Let $\mK$ be a simplicial complex on $\{1,\dots,n\}$. Let $(\bold{X},\bold{A})$ be $n$-pairs of CW complexes. Suppose that $X_i$ are null homotopic for all $i$. Then there is a homotopy equivalence $$\Sigma(\bold{X},\bold{A})^\mK\simeq\Sigma\left(\bigvee_{J\notin\mK}\abs{\mK_J}\ast\bold{A}^{\wedge J}\right)$$
\end{crl}

\begin{defn}{Ring Structure in Hochster's Formula}{}\\
Let $\mK$ be a simplicial complex on $\{1,\dots,n\}$. Define a ring structure in $\bigoplus_{J\subseteq\{1,\dots,n\}}\widetilde{H}^\ast(\mK_\mJ;\Z)$ as follows. Let $\alpha_L$ be the cochain basis element in $C^p(\mK_I;\Z)$ corresponding to the simplex $L\subseteq I$. The ring structure is induced by the map $$C^p(\mK_I;\Z)\otimes_\Z C^q(\mK_\mJ;\Z)\to C^{p+q+1}(\mK_{I\cup J})$$ defined by $$\alpha_L\otimes\alpha_M\mapsto\begin{cases}
\varepsilon(L,I)\varepsilon(M,J)\varepsilon(L\cup M,I\cup J)\alpha_{L\amalg M} & \text{ if }I\cap J=\emptyset\\
0 & \text{ otherwise }
\end{cases}$$
where $\varepsilon(L,I)=\prod_{j\in L}\varepsilon(j,J)$ and $\varepsilon(j,J)=(-1)^{r-1}$ if $j$ is in the $r$th position in $J$. 
\end{defn}

\begin{prp}{}{}\\
Let $\mK$ be a simplicial complex on $\{1,\dots,n\}$. Then the following are true. 
\begin{itemize}
\item The ring structure in Hochster's formula is equal to the relative Kunneth theorem for $$H^\ast(\abs{\mK_I\ast\mK_J})=H^\ast(\Sigma(\abs{\mK_I}\wedge\abs{\mK_J}))=\widetilde{H}^{\ast-1}(\abs{K_I}\times\abs{K_J},\abs{K_I\vee K_J})$$ via the inclusion $\mK_{I\amalg J}\hookrightarrow\mK_I\ast\mK_J$. 
\item The natural isomorphism in the above prp induces a natural isomorphism of rings $$H^\ast(\mZ_\mK;\Z)\cong\bigoplus_{J\subseteq\{1,\dots,n\}}\widetilde{H}^\ast(\mK_J;\Z)$$
\end{itemize}
\end{prp}

There are three descriptions of the cohomology groups: $$H^k(\mZ_\mK;\Z)\cong\bigoplus_{2q-p=k}H_{-p}\left(\Lambda\left(\bigoplus_{i=1}^n\Z[\mK]u_i\right)\right)_{2q}=\bigoplus_{J\subseteq\{1,\dots,n\}}\widetilde{H}^{k-\abs{J}-1}(\mK_J;\Z)$$ For $L\subseteq J$ such that $\alpha_L$ is a cochain basis in the Hochster formula, we have $$[\varepsilon(L,J)\chi(J\setminus L,L)]\leftrightarrow[\varepsilon(L,J)u_{J\setminus L}v_L]\leftrightarrow\alpha_L$$

\begin{prp}{}{}\\

\subsection{Moment Angle Polytopes and Intersection of Quadrics}
Let $P\subseteq\R^n$ be a polytope defined by $$P=\{x\in\R^n\;|\;\langle a_i,x\rangle+b_i\geq 0\text{ for }1\leq i\leq m\}$$ such that $a_1,\dots,a_m$ span $\R^n$. Let $$A=\begin{pmatrix}
a_1\\
\vdots\\
a_m
\end{pmatrix}\in M_{m\times n}(\R)\;\;\text{ and }\;\;b=\begin{pmatrix}
b_1\\
\vdots\\
b_m
\end{pmatrix}\in\R^m$$ Recall that $$P=\{x\in\R^n\;|\;Ax+b\geq 0\}$$

\begin{defn}{Moment Angle Polytopes}{}\\
Let $P\subseteq\R^n$ be a polytope defined by $A\in M_{m\times n}(\R)$ and $b\in\R^m$ such that $a_1,\dots,a_m$ span $\R^n$. Let $i_P:\R^n\to\R^m$ be the map $i_P(x)=Ax-b$. Let $\mu:\C^m\to\R^m$ be the map $\mu(z_1,\dots,z_m)=(\abs{z_1}^2,\dots,\abs{z_m}^2)$. Define the moment angle polytope $\mZ_{A,b}$ to be the pullback \\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	{\mZ_{A,b}} & {\C^nm} \\
	{P} & {\R^m}
	\arrow[from=1-1, to=1-2]
	\arrow[from=1-1, to=2-1]
	\arrow["\mu", from=1-2, to=2-2]
	\arrow["i_P", hook, from=2-1, to=2-2]
\end{tikzcd}}\\
\end{defn}

\begin{prp}{}{}\\
Let $P\subseteq\R^n$ be a polytope defined by $A\in M_{m\times n}(\R)$ and $b\in\R^m$ such that $a_1,\dots,a_m$ span $\R^n$. Then there is a homeomorphism $$\mZ_{A,b}\cong\left\{z=(z_1,\dots,z_m)\in\C^m\;\bigg{|}\;\Gamma\begin{pmatrix}
\abs{z_1}^2\\
\vdots\\
\abs{z_m}^2
\end{pmatrix}=\Gamma b\right\}$$ where $\Gamma=\begin{pmatrix}
\gamma_1\\
\vdots\\
\gamma_{m-n}\end{pmatrix}$ and $\gamma_1,\dots,\gamma_{m-n}$ is a basis for $\ker(A^T)$. 
\end{prp}

This makes $Z_{A,b}$ an intersection of quadrics in $\R$, as well as an intersection of Hermitian quadrics (Hermitian since $\abs{z_i}^2=z_i\overline{z_i}$). 

\begin{defn}{Intersection of Quadrics}{}\\
Let $\Gamma\in M_{(m-n)\times n}(\R)$ be a matrix. Let $\delta\in\R^{m-n}$. Define the intersection of quadrics associated to $(\Gamma,\delta)$ by $$\mZ_{\Gamma,\delta}=\left\{z=(z_1,\dots,z_m)\in\C^m\;\bigg{|}\;\Gamma\begin{pmatrix}
\abs{z_1}^2\\
\vdots\\
\abs{z_m}^2
\end{pmatrix}=\delta\right\}$$
\end{defn}

\begin{lmm}{}{}\\
Let $\Gamma\in M_{(m-n)\times n}(\R)$ be a matrix. Let $\delta\in\R^{m-n}$. Then we have $$\frac{\mZ_{\Gamma,\delta}}{T^m}\cong\left\{y=(y_1,\dots,y_m)\in\R^m\;\bigg{|}\;\Gamma y=\delta\right\}$$
\end{lmm}

\begin{defn}{Associated Polytope}{}\\
Let $\Gamma\in M_{(m-n)\times n}(\R)$ be a matrix. Let $\delta\in\R^{m-n}$. Define the associated polytope of $\mZ_{\Gamma,\delta}$ to be $\mZ_{A,b}$ where 
\begin{itemize}
\item The columns $a_1,\dots,a_m$ of $A$ form a basis for $\ker(\Gamma^T)$. 
\item $b\in\R^m$ is such that $\Gamma b=\delta$. 
\end{itemize}
\end{defn}

\begin{thm}{}{}\\
There is a one to one correspondence $$\frac{\left\{\mZ_{A,b}\;\bigg{|}\;\substack{A\in M_{m\times n}(\R),b\in\R^m\text{ such that}\\\text{the columns }a_1,\dots,a_m\text{ of }A\text{ span }\R^n}\right\}}{\text{ Isomorphism of }\R^n}\;\;\overset{1:1}{\longleftrightarrow}\;\;\frac{\left\{\mZ_{\Gamma,\delta}\;\bigg{|}\;\substack{\Gamma\in M_{(m-n)\times n}(\R),\delta\in\R^{m-n}\text{ such that}\\\text{the columns }\gamma_1,\dots,\gamma_m\text{ of }\Gamma\text{ span }\R^{m-n}}\right\}}{\text{Linear isomorphism of }\R^{m-n}}$$
\end{thm}

\begin{prp}{}{}\\
Let $P\subseteq\R^n$ be a polytope defined by $A\in M_{m\times n}(\R)$ and $b\in\R^m$ such that $a_1,\dots,a_m$ span $\R^n$. Then $P$ is simple if and only if $\mZ_{A,b}$ is a smooth manifold of real dimension $m+n$. 
\end{prp}

\begin{prp}{}{}\\

\item There is a diffeomorphism $$\mZ_{\mK_P}\cong\#_{k=3}^{n+1}(S^k\times S^{2n+3-k})^{\# q_k}$$ for some $q_k$ determined by the Gale diagram of $P$. 
\end{itemize}
\end{thm}

\section{Research Notes}
\subsection*{The Foundation}
Moment angle complexes are $T^n$-equivariant objects defined using combinatorial data. More specifically, choose a simplicial complex $\mK$ on $\{1,\dots,n\}$. For each simplex $\sigma\in\mK$, define the space $$(D^2,S^1)^\sigma=\prod_{i=1}^nY_i\;\;\;\;\text{ where }\;\;\;\;Y_i=\begin{cases}
D^2 & \text{ if }i\in\sigma\\
S^1 & \text{ otherwise }
\end{cases}$$ In other words, $(D^2,S^1)^\sigma$ is a product of $D^2$ and $S^1$ depending on which vertices of $\mK$ lie in $\sigma$. For any inclusion of simplicies $\sigma\subseteq\tau$, there is an inclusion map $(D^2,S^1)^\sigma\hookrightarrow(D^2,S^1)^\tau$. Then the definition of the moment angle complex $\mZ_\mK$ of $\mK$ is just the gluing of products of $D^2$ and $S^1$ along these inclusions, or $$\mZ_\mK=\bigcup_{\sigma\in\mK}(D^2,S^1)^\sigma$$ Being combinatorial in nature, they form a testing ground for conjectures in topology. Moreover, in good cases (eg $\mK$ is a triangulation of $S^m$), the moment angle complex $\mZ_\mK$ has the structure of a smooth manifold and they form a testing ground for conjectures in geometry. 

These topological objects are also interesting because they lie at the intersection of a number of active fields. Other than moment angle complexes, there are other objects in topology and geometry that are defined in terms of combinatorial objects

\adjustbox{scale=1.0,center}{\begin{tikzcd}
	{\mZ_\mK} & {\mZ_P} & {V_P} \\
	{\text{(Topology)}} & {\text{(Symplectic Geometry)}} & {\text{(Algebraic Geometry)}}
\end{tikzcd}} \\

Moment angle polytopes $\mZ_P$ are intersection of Hermitian quadrics that define non-Kahler complex manifolds, also makes for a testing ground for theories in symplectic and complex geometry. 

Projective toric varieties $V_P$ are combinatorial objects in algebraic geometry formed by certain types of polytopes that are also, explicit enough to form a testing ground for conjectures in algebraic geometry. Choosing our polytopes carefully connects the three objects into a unified theory. 

\adjustbox{scale=1.0,center}{\begin{tikzcd}
	{U(\mK)} & {U(\Sigma_P)} & {U(\Sigma_P)/T^m} && {V_P} \\
	\\
	{\mZ_\mK} & {\mZ_P} & {\mZ_P/T^{m-n}} \\
	{\text{(Topology)}} & {\text{(Symplectic Geometry)}} & {\text{(Algebraic Geometry)}}
	\arrow["{\cong_{T^m}}", from=1-1, to=1-2]
	\arrow["{\simeq_{T^m}}", from=1-1, to=3-1]
	\arrow["{T^m\text{-Orbit}}", from=1-2, to=1-3]
	\arrow["{\simeq_{T^m}}", from=1-2, to=3-2]
	\arrow["\cong", from=1-3, to=1-5]
	\arrow["{\cong\text{diffeo}}"{description}, from=1-3, to=3-3]
	\arrow["{\cong_{T^m}}", from=3-1, to=3-2]
	\arrow["{T^{m-n}\text{-Orbit}}", from=3-2, to=3-3]
\end{tikzcd}} \\

In fact, the first connection between symplectic geometry and combinatorics came from the following theorem. 

\begin{thm}{The Atiyah–Guillemin–Sternberg convexity theorem (1982)}{}\\
Let $(M,\omega)$ be a compact connected symplectic manifold. Let $T^k\times M\to M$ be a Hamiltonian torus action. Then $\mu(M)$ is a convex polytope in $\text{Lie}(T^k)^\ast$ given by $$\mu(M)=\text{Conv}(\text{Fix}_{T^k}(M))$$
\end{thm}

Delzant in 1988 proved that if the action is more over effective and $M$, then these polytopes completely determine the manifold structure. This motivates the definition of quasitoric manifolds: Smooth manifolds $M^{2n}$ with a torus action $T^n$ that locally looks like the standard action, together with a homeomorphism of the orbit space $M/T^n\cong P$ with a polytope. In 2008, the following conjecture is made. 

\begin{conj}{Masuda–Suh (2008)}{}\\
If $M$ and $N$ are quasitoric manifolds such that $$H^\ast(M)\cong H^\ast(N)$$ Then $M$ and $N$ are homeomorphic (or diffeomorphic). 
\end{conj}

We are interested in the cohomological rigidty of moment angle complexes instead. The first observation in this direction is that Hochster's formula shows that the cohomology of a moment angle complex is completely determined by the simplicial complex: 

\begin{thm}{Hochster's Formula}{}\\
Let $\mK$ be a simplicial complex on $\{1,\dots,n\}$. Then there is an isomorphism of rings $$H^\ast(\mZ_\mK)\cong\bigoplus_{J\subseteq\{1,\dots,n\}}H^\ast(\mK_J)$$ where 
\begin{itemize}
\item $\mK_J$ is the full subcomplex of $\mK$ with the vertices in $J$. 
\item The additive isomorphism is given by $$H^n(\mZ_\mK)\cong\bigoplus_{J\subseteq\{1,\dots,n\}}H^{n-\abs{J}-1}(\mK_J)$$
\item The ring structure is given the map $H^i(\mK_I)\otimes H^j(\mK_J)\to H^{i+j+1}(\mK_{I\cup H_J})$ which sends products of generators to generators if $I\cap J=\emptyset$ and $0$ otherwise. 
\end{itemize}
\end{thm}

The second observation is that certain classes of simplicial complexes have the geometry of connected sums of products of spheres: 

\begin{thm}{}{}\\
Let $P$ be a stacked polytope. Let $n=\dim(P)$ and $m=\abs{V(P)}$. Then there is a diffeomorphism $$\mZ_{\partial P}\cong\#_{k=3}^{m-n+1}(S^k\times S^{m+n-k})^{(k-2)\binom{m-n}{k-1}}$$
\end{thm}

Among the class of topological spaces, spheres are homologically and homotopically the easiest to understand. It is natural to ask the question in what conditions does a moment angle complex has its cohomology only involving spheres, and in this case is it possible to lift it to a homotopy equivalence, a homeomorphism or a diffeomorphism. Currently my work focuses on moment angle complexes coming from simplicial polytopes of dimension $4$ with $8$ vertices.  

\subsection*{What is Known in the Simplicial 4-Polytopes and 8-Vertices Cases}
There are $39$ simplicial complexes in dimension $4$ with $8$ vertices (no ghost vertices). Of which $37$ of them come from the boundary of a simplicial $4$-polytope with $8$ vertices (polytopal spheres). These are classified in [An Enumeration of Simplicial 4-Polytopes with 8-Vertices] by Grunbaum and Sreedharan, and whose list was later corrected by my supervisor. 

$P_1^8$ is a stacked polytope where: 
\begin{itemize}
\item The starting simplex is $12356$. 
\item Vertex $4$ is stacked on the facet $1235$. 
\item Vertex $7$ is stacked on the facet $1236$. 
\item Vertex $8$ is stacked on the facet $1256$. 
\end{itemize}

$P_2^8$ is a stacked polytope where: 
\begin{itemize}
\item The starting simplex is $12356$. 
\item Vertex $4$ is stacked on the facet $1235$. 
\item Vertex $7$ is stacked on the facet $1236$. 
\item Vertex $8$ is stacked on the facet $1245$.

$P_3^8$ is a stacked polytope where: 
\begin{itemize}

\item Vertex $6$ is stacked on the facet $1235$. 
\item Vertex $7$ is stacked on the facet $1236$. 
\item Vertex $8$ is stacked on the facet $1234$. 
\end{itemize}

For a stacked polytope $P$, we have a theorem 

\begin{thm}{}{}\\
Let $P$ be a stacked polytope. Let $n=\dim(P)$ and $m=\abs{V(P)}$. Then there is a diffeomorphism $$\mZ_{\partial P}\cong\#_{k=3}^{m-n+1}(S^k\times S^{m+n-k})^{(k-2)\binom{m-n}{k-1}}$$
\end{thm}

Hence we conclude that $$\mZ_{\partial P_1^8}\cong\mZ_{\partial P_2^8}\cong\mZ_{\partial P_3^8}\cong(S^3\times S^9)^{\#6}\#(S^4\times S^8)^{\#8}\#(S^5\times S^7)^{\#3}$$

Also $P_{35}^8$, $P_{36}^8$ and $P_{37}^8$ are neighbourly polytopes (one such is cyclic), and we have the following theorem. 

\begin{thm}{Membrillo Solis, Theriault (2026)}{}\\
Let $\mK$ be a simplicial complex such that the following are true. 
\begin{itemize}
\item $\mK$ is a triangulation of $S^{2n+1}$. 
\item $\mK$ is neighbourly. 
\end{itemize}
Then there is a homotopy equivalence $$(D^k,S^{k-1})^\mK\simeq\#_{i=1}^j\left(S^{t_i}\times S^{s_i}\right)$$ for some $j\in\N^+$ and some $t_i,s_i\in\N^+$ for all $k\geq 2$. 
\end{thm}

So we have $$\mZ_{\partial P_{35}^8}\simeq\mZ_{\partial P_{36}^8}\simeq\mZ_{\partial P_{37}^8}\simeq(S^5\times S^7)^{\#16}\#(S^6\times S^6)^{\#15}$$

Also $P_{34}^8$ is a cross polytope (It is given by $P_{34}^8=S^0\ast S^0\ast S^0\ast S^0$). A direct computation shows that $$H^\ast(\mZ_{\partial P_{34}^8})\cong H^\ast(S^3\times S^3\times S^3\times S^3)$$ and the following theorem lifts this isomorphism into a homeomorphism. 

\begin{thm}{Fan, Cheng, Ma, Wang (2014)}{}\\

\section{Quasitoric Manifolds}
\subsection{The Davis-Januszkiewicz Classification Theorem}
\begin{defn}{Quasitoric Manifolds}{}\\
Let $M$ be a smooth manifold of dimension $2n$. Let $T^n$ be a torus acting smoothly on $M$. We say that $M$ is a quasitoric manifold if the following are true. 
\begin{itemize}
\item For each $p\in M$, there is a $T^n$-equivariant chart $(U,\varphi:U\to\C^n)$ where $\C^n$ is equipped with the standard torus action. 
\item There exists a convex polytope $P$ such that there is a homeomorphism $$M/T^n\cong P$$
\end{itemize}
\end{defn}

Toric symplectic manifolds are quasitoric manifolds. Moment angle manifolds are NOT quasitoric manifolds. 

\begin{defn}{Orbit Type Strata}{}\\
Let $M$ be a quasitoric manifold. Let $P$ be the underlying convex polytope. Let $H\leq T^n$ be a subgroup. Define the orbit type strata of $H$ to be $$M_{(H)}=\{x\in M\;|\;\text{Stab}_{T^n}(x)=H\}$$
\end{defn}

\begin{prp}{}{}\\
Let $M$ be a quasitoric manifold. Let $P$ be the underlying convex polytope. Then there is a bijection $$\{F\in\mF(P)\;|\;F\text{ has codimension }k\}\;\;\overset{1:1}{\leftrightarrow}\;\;\left\{N\subseteq M\;\bigg{|}\;\substack{N\text{ is an orbit type strata of subtori of }\\T^n\text{ of dimension }k}\right\}$$ given as follows. Let $\pi:M\to P$ be the quotient map. Then we have $$F_i\mapsto\pi^{-1}(\text{relint}(F))$$
\end{prp}

\begin{defn}{Primitive Vectors}{}\\
Let $v=(v_1,\dots,v_n)\in\Z^n$. We say that $v$ is primitive if $v\neq\lambda u$ for any $\lambda\in\Z\setminus\{0\}$ and $u\in\Z^n$. 
\end{defn}

\begin{defn}{The Characteristic Matrix}{}\\
Let $M$ be a quasitoric manifold. Let $P$ be the underlying convex polytope. Define the characteristic matrix $\Lambda$ as follows. For each facet $F_i$ of $P$ for $1\leq i\leq m$, the facet corresponds to a a circle $S_i^1\subseteq T^n$ by the above. Every circle $S_i^1\subseteq T^n$ corresponds to a unique primitive vector $v_i=(v_{1i},\dots,v_{1n})\in\Z^n$ up to sign. Then $$\Lambda=\begin{pmatrix}
v_{11} & \cdots & v_{1m}\\
\vdots & \ddots & \vdots\\
v_{n1} & \cdots & v_{nm}
\end{pmatrix}\in M_{n\times m}(\Z)$$
\end{defn}

\begin{defn}{Characteristic Pairs}{}\\
Let $P$ be a simple polytope of dimension $n$. Let $F_1,\dots,F_m$ be the facets of $P$. Let $\Lambda\in M_{n\times m}(\Z)$ be a matrix. Let $c_1,\dots,c_m$ be the columns of $\Lambda$. We say that $(P,\Lambda)$ is a characteristic pair if the following are true. 
\begin{itemize}
\item $c_i$ is a primitive vector for all $1\leq i\leq m$. 
\item If $F_{i_1}\cap\cdots\cap F_{i_n}$ is a vertex of $P$, then $$\det\begin{pmatrix}
| & & |\\
c_{i_1} & \cdots & c_{i_n}\\
| & & |
\end{pmatrix}=\pm1$$
\end{itemize}
\end{defn}

\begin{defn}{The Associated Quasitoric Manifold}{}\\
Let $(P,\Lambda)$ be a characteristic pair. Suppose that $\dim(P)=n$. Let $F_1,\dots,F_m$ be the facets of $P$. Let $c_1,\dots,c_m$ be the columns of $\Lambda$. Define the associated quasi-toric manifold of $(P,\Lambda)$ by $$M(P,\Lambda)=\frac{P\times T^n}{\sim}$$ where $(p,t_1)\sim(p,t_2)$ if $t_2^{-1}t_1\in\langle c_i\;|\;p\in F(p)\rangle\cong T^k\leq T^n$ and $F(p)$ is the unique facet such that $p\in\text{relint}(F(p))$ for any $p\in P$. 
\end{defn}

\begin{thm}{Davis-Januszkiewicz Classification Theorem}{}\\
There is a one to one bijection $$\frac{\{M\;|\;M\text{ is a quasitoric manifold}\}}{\cong}\;\;\overset{1:1}{\leftrightarrow}\;\;\frac{\{(P,\Lambda)\text{ is a characteristic pair}\}}{\sim}$$ where $(P,\Lambda)\sim(Q,\Gamma)$ if $P$ and $Q$ are combinatorially equivalent and there exists $A\in\text{GL}(n,\Z)$ with $\det(A)=\pm1$ such that $A\Lambda$ and $Q$ are equal up to permutation of columns. 
\end{thm}

\subsection{Equivalent Definitions}
\begin{defn}{Kahler Forms}{}\\
Let $(M,h)$ be a complex Hermitian manifold. Let $\omega$ be the associated form. We say that $\omega$ is a Kahler form if $d\omega=0$. In this case, we say that $M$ is a Kahler manifold. 
\end{defn}

\begin{lmm}{}{}\\
Let $M$ be a Kahler manifold. Then the following are true. 
\begin{itemize}
\item Let $g=\frac{1}{2}(h+\overline{h})$ be the associated Riemannian metric. Then $(M,g)$ is a Riemannian manifold. 
\item Let $\omega=\frac{i}{2}(h-\overline{h})$ be the associated $(1,1)$-form. Then $(M,\omega)$ is a symplectic manifold. 
\end{itemize}
\end{lmm}

\begin{prp}{}{}\\
Let $(M,\omega)$ be a symplectic manifold. Then $\omega$ is a Kahler form if and only if $M$ admits an integrable almost complex structure $J$. In this case, the following are true. 
\begin{itemize}
\item The Riemannian metric is given by $g(u,v)=\omega(u,Jv)$. 
\item The Hermitian metric is given by $h=g-i\omega$. 
\end{itemize}
\end{prp}

\begin{prp}{}{}\\
Let $(M,g)$ be a Riemannian manifold of dimension $2n$. Then $M$ admits a Kahler form if and only if $\text{Hol}(\nabla^g)\leq U(n)$. 
\end{prp}

\begin{thm}{Hodge Decomposition Theorem}{}\\
Let $M$ be a compact Kahler manifold. Then the following are true. 
\begin{itemize}
\item There is an isomorphism $$H_\text{dR}^k(M;\C)\cong\bigoplus_{p+q=k}H^{p,q}(M)$$ for all $k\in\N$. 
\item There is an isomorphism $$H^{p,q}(M)\cong\overline{H^{q,p}(M)}$$ for all $p,q\in\N$. 
\item $\dim_\R(H_\text{dR}^k(M))$ is even for all $k\in\N$. 
\end{itemize}
\end{thm}

0 & \text{ otherwise }
\end{cases}$$

Let $M$ be a compact Kahler manifold. Define the Hodge filtration of $M$ to be $$F^iH_\text{dR}^k(M;\C)=\bigoplus_{p\geq i}H^{p,q}(M)$$
\end{defn}

\begin{defn}{The Lefschetz Operator}{}\\
Let $M$ be a Kahler manifold. Let $\omega$ be its associated form. Define the Lefschetz operator to be the map $L:\Omega^k(M)\to\Omega^{k+2}(M)$ given by $$\alpha\mapsto\alpha\wedge\omega$$
\end{defn}

\begin{thm}{Hard Lefschetz Theorem}{}\\
Let $M$ be a compact Kahler manifold. Then the following are true. 
\begin{itemize}
\item The Lefschetz operator induces an isomorphism $$L^k:H_\text{dR}^{n-k}(M)\overset{\cong}{\longrightarrow} H_\text{dR}^{n+k}(M)$$
\end{itemize}
\end{thm}

\begin{defn}{Primitive Cohomology}{}\\
Let $M$ be a compact Kahler manifold. Let $n=\dim(M)$. Define the primitive cohomology of $M$ to be $$H_\text{prim}^i(M)=\ker(L^{n-i+1}:H_\text{dR}^i(M)\to H_\text{dR}^{2n-i+2}(M))$$
\end{defn}

\begin{defn}{Lefschetz Decomposition}{}\\

\end{defn}

	\arrow["\mu", from=1-6, to=2-5]
	\arrow["{1\otimes m}"', from=2-1, to=2-2]
	\arrow["m"', from=2-2, to=1-3]
\end{tikzcd}} \\
where $\alpha$ is the natural isomorphism from associativity of tensor products, and $\lambda,\mu$ is the natural isomorphism from the unitality of tensor products. 
\end{prp}

Equivalently, the above proposition is the same as saying $(A,m,u)$ is a monoid object in the monoidal category $(\bold{Mod}_R,\otimes_R,R)$. Motivated by the categorical description of an $R$-algebra, we define coalgebras as the formal dual of the above object in the following way. 

\begin{defn}{Coalgebras}{}\\
Let $R$ be a commutative ring. Let $C\in\bold{Mod}_R$ be an $R$-module. Let $\Delta:C\to C\otimes_RC$ and $\varepsilon:C\to R$ be $R$-module homomorphisms. We say that $(C,\Delta,\varepsilon)$ is a coalgebra if the following diagrams commute: \\~\\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	C & {C\otimes_RC} & {(C\otimes_RC)\otimes_RC} && C \\
	& {C\otimes_RC} & {C\otimes_R(C\otimes_RC)} & {R\otimes_RC} & {C\otimes C} & {C\otimes_RR}
	\arrow["\Delta", from=1-1, to=1-2]
	\arrow["\Delta"', from=1-1, to=2-2]
	\arrow["{\Delta\otimes 1}", from=1-2, to=1-3]
	\arrow["\alpha", from=1-3, to=2-3]
	\arrow["\lambda"', from=1-5, to=2-4]
	\arrow["\Delta", from=1-5, to=2-5]
	\arrow["\mu", from=1-5, to=2-6]
	\arrow["{1\otimes\Delta}"', from=2-2, to=2-3]
	\arrow["{\varepsilon\otimes\text{id}_A}", from=2-5, to=2-4]
	\arrow["{\text{id}_A\otimes\varepsilon}"', from=2-5, to=2-6]
\end{tikzcd}} \\
where $\alpha$ is the natural isomorphism from associativity of tensor products, and $\lambda,\mu$ is the natural isomorphism from the unitality of tensor products. 
\end{defn}

\begin{defn}{Bialgebras}{}\\
Let $R$ be a commutative ring. Let $B\in\bold{Mod}_R$ be an $R$-module. Let $m:B\otimes_RB\to B$, $u:R\to B$, $\Delta:B\to B\otimes_RB$ and $\varepsilon:B\to R$ be $R$-module homomorphisms. We say that $B$ is an $R$-bialgebra if the following are true. 
\begin{itemize}
\item $(B,m,u)$ is an $R$-algebra. 
\item $(B,\Delta,\varepsilon)$ is an $R$-coalgebra. 
\item Multiplication and comultiplication are compatible: \\~\\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	{B\otimes_RB} & B & {B\otimes_RB} \\
	{B\otimes_RB\otimes_RB\otimes_RB} && {B\otimes_RB\otimes_RB\otimes_RB}
	\arrow["m", from=1-1, to=1-2]
	\arrow["{\Delta\otimes\Delta}"', from=1-1, to=2-1]
	\arrow["\Delta", from=1-2, to=1-3]
	\arrow["{\text{id}\otimes\tau\otimes\text{id}}"', from=2-1, to=2-3]
	\arrow["{m\otimes m }"', from=2-3, to=1-3]
\end{tikzcd}} \\
where $\tau$ is the natural isomorphism from commutativity of tensor products. 
\item Multiplication and counit are compatible: \\~\\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	{B\otimes_RB} & B \\
	{R\otimes_RR} & R
	\arrow["m", from=1-1, to=1-2]
	\arrow["{\varepsilon\otimes\varepsilon}"', from=1-1, to=2-1]
	\arrow["\varepsilon", from=1-2, to=2-2]
	\arrow["\lambda"', from=2-1, to=2-2]
\end{tikzcd}} \\
where $\lambda$ is the natural isomorphism from unitality of tensor products. 
\item Comultiplication and unit are compatible: \\~\\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	R & B \\
	{R\otimes_RR} & {B\otimes_RB}
	\arrow["u", from=1-1, to=1-2]
	\arrow["\lambda"', from=1-1, to=2-1]
	\arrow["\Delta", from=1-2, to=2-2]
	\arrow["{\Delta\otimes\Delta}"', from=2-1, to=2-2]
\end{tikzcd}} \\
where $\lambda$ is the natural isomorphism from unitality of tensor products. 
\item Unit and counit are compatible: \\~\\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	R && R \\
	& B
	\arrow["{\text{id}}", from=1-1, to=1-3]
	\arrow["u"', from=1-1, to=2-2]
	\arrow["\varepsilon"', from=2-2, to=1-3]
\end{tikzcd}} \\
\end{itemize}
\end{defn}

\subsection{Hopf Algebra}
\begin{defn}{Hopf Algebras}{}\\
Let $R$ be a commutative ring. Let $(H,m,u,\Delta,\varepsilon)$ be an $R$-bialgebra. Let $S:H\to H$ be an $R$-module homomorphism. We say that $(H,m,u,\Delta,\varepsilon,S)$ is a Hopf algebra if the following diagram commutes: \\~\\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	& {H\otimes_RH} && {H\otimes_RH} \\
	H && R && H \\
	& {H\otimes_RH} && {H\otimes_RH}
	\arrow["{S\otimes\text{id}_H}", from=1-2, to=1-4]
	\arrow["m", from=1-4, to=2-5]
	\arrow["\Delta", from=2-1, to=1-2]
	\arrow["\varepsilon", from=2-1, to=2-3]
	\arrow["\Delta"', from=2-1, to=3-2]
	\arrow["u", from=2-3, to=2-5]
	\arrow["{\text{id}_H\otimes S}"', from=3-2, to=3-4]
	\arrow["m"', from=3-4, to=2-5]
\end{tikzcd}} \\
\end{defn}

\begin{defn}{Commmutative and Cocommutative}{}\\
Let $R$ be a commutative ring. Let $H$ be a Hopf algebra. 
\begin{itemize}
\item We say that $H$ is commutative if the underlying algebra is commutative. 
\item We say that $H$ is cocommutative if the underlying coalgebra is cocommutative. 
\end{itemize}
\end{defn}

\begin{prp}{}{}\\
Let $R$ be a commutative ring. Let $H$ be a Hopf algebra. If $H$ is a projective and finitely generated $R$-modue, then $\Hom_R(H,R)$ is a Hopf algebra. Moreover, the following are true. 
\begin{itemize}
\item $\Hom_R(M,R)$ is commutative if and only if $H$ is cocommutative. 
\item $\Hom_R(M,R)$ is cocommutative if and only if $H$ is commutative. 
\end{itemize}
\end{prp}

\subsection{Differential Graded Algebras}
\begin{defn}{Differential Graded Algebras}{}\\
Let $R$ be a commutative ring. A differential graded $R$-algebra consists of the following data. 
\begin{itemize}
\item A $\Z$-graded $R$-algebra $A$. 
\item An $R$-algebra homomorphism of degree $-1$ (called homological dgas) or of degree $1$ (called cohomological dgas). 
\end{itemize}
such that the following are true. 
\begin{itemize}
\item Differential: $d\circ d=0$. 
\item Graded Leibniz: $d(ab)=(da)b+(-1)^{\deg(a)}a(db)$ for $a,b$ homogeneous elements of $A$. 
\end{itemize}
\end{defn}

The differential property means that any (homological) differential graded algebra decomposes into a chain complex whose differential is given by $d$. 

\begin{defn}{Homomorphisms of Differential Graded Algebras}{}\\
Let $R$ be a commutative ring. Let $A,B$ be differential graded algebras. A morphism $\varphi:A\to B$ of differential graded algebras is a graded $R$-algebra homomorphism. 
\end{defn}

\begin{lmm}{}{}\\
Let $R$ be a commutative ring. Let $(A,d)$ be a cohomological differential graded algebra. Then $H^\ast(A,d)=\bigoplus_{n\in\Z}H^n(A,d)$ is a graded ring with multiplication defined by $$[u]\cdot[v]=[uv]$$ for $u\in H^n(A,d)$ and $v\in H^m(A,d)$. 
\begin{proof}\\
Notice that by the graded Leibniz rule, we have $d(uv)=(du)v+(-1)^nud(v)=0$ since $u$ and $v$ are cocycles. Hence $uv$ is also a cocycle. 
\end{proof}
\end{lmm}

\subsection{Massey Products}
\begin{defn}{Massey Triple Products}{}\\
Let $R$ be a commutative ring. Let $A$ be a differential graded algebra. Let $u\in A^p$, $v\in A^q$ and $w\in A^r$. Suppose that $[u]\cdot[v]=[v]\cdot[w]=0$. Define the Massey triple product of the three elements to be $$\langle[u],[v],[w]\rangle=\{[\overline{s}\cdot w+\overline{u}\cdot t]\in H^{p+q+r-1}(A)\;|\;s\in A^{p+q-1},t\in A^{q+r-1}\text{ and }ds=\overline{u}\cdot v,dt=\overline{v}\cdot w\}$$ where $\overline{a}=(-1)^{1+\deg(a)}a$ for any homogeneous element $a\in A$. 
\end{defn}

\begin{lmm}{}{}\\
Let $R$ be a commutative ring. Let $A$ be a differential graded algebra. Let $u\in A^p$, $v\in A^q$ and $w\in A^r$. Suppose that $[u]\cdot[v]=[v]\cdot[w]=0$. Then we have $$\langle[u],[v],[w]\rangle=\frac{H^{p+q+r-1}(A)}{[u]\cdot H^{q+r-1}(A)+H^{p+q-1}(A)\cdot[w]}$$
\end{lmm}

\begin{prp}{}{}\\

\begin{itemize}
\item $\langle u,v,w\rangle=(-1)^pz$. 
\item $\langle v,w,u\rangle=(-1)^{pq+pr+p}z$. 
\item $\langle w,u,v\rangle=0$. 
\end{itemize}
\end{prp}

\begin{defn}{The $I$-Torsion of a Module}{}\\
Let $R$ be a Noetherian commutative ring. Let $M$ be an $R$-module. Let $I$ be an ideal of $R$. Define the $I$-torsion of $M$ to be $$\Gamma_I(M)=\{m\in M\;|\;I^tm=0\text{ for some }t\in\N\}$$

\begin{defn}{$I$-Torsion Modules}{}\\
Let $R$ be a Noetherian commutative ring. Let $M$ be an $R$-module. Let $I$ be an ideal of $R$. We say that $M$ is $I$-torsion if $M=\Gamma_I(M)$. 
\end{defn}

\begin{defn}{The Torsion Functor}{}\\
Let $R$ be a Noetherian commutative ring. Let $I$ be an ideal of $R$. Define the $I$-torsion functor $$\Gamma_I:{_R}\bold{Mod}\to{_R}\bold{Mod}$$ by the following. 
\begin{itemize}
\item For each $R$-module $M$, the functor sends $M$ to the $I$-torsion $\Gamma_I(M)$. 
\item For each $R$-module homomorphism $f:M\to N$, the functor sends $f$ to $\Gamma_I(f)=f|_{\Gamma_I(M)}$. 
\end{itemize}
\end{defn}

\begin{lmm}{}{}\\
Let $R$ be a Noetherian commutative ring. Let $I$ be an ideal of $R$. Then $\Gamma_I$ is a left exact functor. 
\end{lmm}

\begin{defn}{Local Cohomology}{}\\
Let $R$ be a Noetherian commutative ring. Let $M$ be an $R$-module. Let $I$ be an ideal of $R$. Define the $n$th local cohomology of $M$ with support in $I$ to be $$H_I^n(M)=(R^n\Gamma_I)(M)$$
\end{defn}

\begin{prp}{}{}\\
Let $R$ be a Noetherian commutative ring. Let $M$ be an $R$-module. Let $I,J$ be ideals of $R$. If $\sqrt{I}=\sqrt{J}$, then the equality $\Gamma_I(M)=\Gamma_J(M)$ induces an equality $$H_I^n(M)=H_J^n(M)$$ for all $n\in\N$. 
\end{prp}

Thus we only need to consider the local cohomology with support in radical ideals. 

\begin{lmm}{}{}\\
Let $R$ be a Noetherian commutative ring. Let $M$ be an $R$-module. Let $I$ be an ideal of $R$. Then $H_I^n(M)$ is $I$-torsion for all $n\in\N$. 
\end{lmm}

\end{lmm}

\subsection{Progenerators}
\begin{defn}{Progenerators}{}\\
Let $R$ be a ring. Let $P$ be an $R$-module. We say that $P$ is a progenerator of $R$ if the following are true. 
\begin{itemize}
\item $P$ is finitely generated. 
\item $P$ is projective. 
\end{itemize}
\end{defn}

\begin{prp}{}{}\\
Let $R,S$ be rings. Let $F:{_R}\bold{Mod}\to{_S}\bold{Mod}$ and $G:{_S}\bold{Mod}\to{_R}\bold{Mod}$ be additive functors. Then $F$ and $G$ are inverses if and only if there exists an $(S,R)$-bimodule $P$ such that the following are true. 
\begin{itemize}
\item ${_S}P$ is a progenerator. 
\item $P_R$ is a progenerator. 
\item There are natural isomorphisms $F\cong P\otimes_R-$ and $G\cong\Hom({_S}P,-)$. 
\end{itemize}
\end{prp}

\begin{thm}{Morita}{}\\
Let $R,S$ be rings. Then the following are equivalent. 
\begin{itemize}
\item $R$ and $S$ are Morita equivalent. 
\item There exists a progenerator $R$-module $P_R$ such that $S\cong\text{End}_R(P_R)$. 
\item There exist an $(R,S)$-bimodule $M$ and a $(S,R)$-bimodule $N$ such that $M\otimes_SN\cong R$ as $(R,R)$-bimodules and $N\otimes_RM\cong S$ as $(S,S)$-bimodules. 
\end{itemize}
\end{thm}

\end{prp}

Let $R,S$ be rings. Then $R$ and $S$ are Morita equivalent if and only if ${_R}\bold{FGMod}\simeq{_S}\bold{FGMod}$. 
\end{prp}

\section{Complex Analytic Space}
\subsection{Complex Analytic Varieties}
\begin{defn}{Complex Analytic Varieties}{}\\
Let $Z\subseteq\C^n$. We say that $Z$ is a complex analytic variety if there exists an open set $Z\subseteq U\subseteq\C^n$ and $f_1,\dots,f_k\in\mO_{\C^n}(U)$ such that $$Z=\V(f_1,\dots,f_k)$$
\end{defn}

We use a subset of $C^n$ in the definition because in $\C^n$, locally defined functions cannot in general be defined globally (compare it to the real case, where they have partitions of unity and bump functions). \\

Indeed, for smooth functions on $\R^n$, a function on an open set $U$ can be modified (via bump functions) to produce a global function that agrees with it on a smaller open subset. You can always find a global function that captures the local behavior near any point. \\

For holomorphic functions on $\C^n$: no such modification is possible. A holomorphic function cannot be cut off or modified while remaining holomorphic, so local holomorphic data is genuinely trapped in its domain.

\begin{defn}{The Structure Sheaf}{} Let$ U\subseteq\C^n$ be an open subset. Let $f_1,\dots,f_k\in\mA_{\C^n}(U)$ be analytic functions on $U$. Let $Z=\V(f_1,\dots,f_k)$ be the vanishing locus of $f_1,\dots,f_k$. Define the structure sheaf $$\mO_Z:\bold{Open}(Z)\to\bold{CRing}$$ as follows. 
\begin{itemize}
\item For each open set $U\subseteq Z$, define $$\mO_Z(U)=\frac{\mO_{\C^n}(U)}{(f_1,\dots,f_k)}$$
\item For each inclusion $U\subseteq V\subseteq Z$ of open sets, define $$\mO_Z(V)\to\mO_Z(U)$$ to be the induced map of the restriction $\mO_{\C^n}(V)\to\mO_{\C^n}(U)$. 
\end{itemize}
\end{defn}

\subsection{Complex Analytic Subvarieties of Manifolds}
\begin{defn}{Complex Analytic Subvarieties}{}\\
Let $M$ be a complex manifold. Let $Z\subseteq M$ be a subset. We say that $Z$ is a complex analytic subvariety of $M$ if for each $p\in Z$, there exists a chart $(U,\phi)$ of $M$ such that $\phi(Z\cap U)$ is a complex analytic variety. 
\end{defn}

In particular, complex analytic subvarieties of $\C^n$ are just complex analytic varieties. 

\begin{thm}{Cartan's (Coherence) Theorem}{} Let $M$ be a complex manifold and let $A$ be an analytic subset of $M$. Then $\mI_A$ is a coherent sheaf of $\mO_M$-modules. 
\end{thm}

\subsection{Complex Analytic Spaces}
\begin{defn}{Complex Analytic Spaces}{} Let $(X,\mO_X)$ be a locally ringed space. We say that $X$ is a complex analytic space if for all $p\in X$, there exists an open neighbourhood $U\subseteq X$ such that $(U,\mO_X|_U)$ is a complex analytic space. 
\end{defn}

\begin{lmm}{}{}\\
Let $(X,\mO_X)$ be a complex analytic space. Then $X$ is a complex manifold if and only if for each $p\in X$, there exists an open neighbourhood $U\subseteq X$ such that there is an isomorphism $$(U,\mO_X|_U)\cong(V,\mO_{\C^n}|_V)$$ for some open subset $V\subseteq\C^n$. 
\end{lmm}

\begin{thm}{}{} Let $X$ be a complex space. Then $X_\text{reg}$ is dense and open in $X$. Moreover, $X_\text{reg}$ consists of a disjoin union of complex manifolds. 
\end{thm}

\begin{thm}{}{} Let $X$ be an irreducible complex space. Then every non-constant holomorphic function $f:X\to\C$ is an open map. 
\end{thm}

\begin{crl}{}{} Let $X$ be an irreducible compact complex space. Then every holomorphic function $f:X\to\C$ on $X$ is constant. 
\end{crl}

\begin{defn}{Complex Analytic Subvarieties}{}\\
Let $(X,\mO_X)$ be a complex analytic space. Let $Z\subseteq X$ be subset. We say that $Z$ is a complex analytic subvariety if for all $p\in X$, there exists an open neighbourhood $U\subseteq X$ and $f_1,\dots,f_k\in\mO_X(U)$ such that $$Z\cap U=\V_U(f_1,\dots,f_k)$$
\end{defn}

This definition generalizes complex analytic subvarieties of complex manifolds. 

\subsection{Connections to Algebraic Geometry}
Recall that the category of schemes and the category of analytic spaces are both full subcategories of locally ringed spaces. 

\begin{defn}{The Analytification of a Scheme}{}\\
Let $X$ be a scheme locally of finite type over $\C$. Define the analytification of $X$ to be the unique analytic space $X^\text{an}$ together with a morphism of locally ringed space $\alpha:X^\text{an}\to X$ satisfying the following universal property: For any analytic space $Y$ and any morphism of locally ringed spaces $f:Y\to X$, there exists a unique morphism of analytic spaces $g:Y\to X^\text{an}$ such that the following diagram commutes: \\~\\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	& Y \\
	{X^\text{an}} & X
	\arrow["{\exists!g}"', dashed, from=1-2, to=2-1]
	\arrow["f", from=1-2, to=2-2]
	\arrow["\alpha"', from=2-1, to=2-2]
\end{tikzcd}}\\
\end{defn}

Recall that the Yoneda embedding $h:\bold{LRS}\to\bold{Set}^{\bold{LRS}}$ is the functor sending a locally ringed space $X$ to the functor $h_X:\bold{LRS}\to\bold{Set}$ defined by $h_X(Y)=\Hom_{\bold{LRS}}(Y,X)$. Then the universal property of the analytification means that the functor $h_X$ restricted to analytic spaces $\bold{An}$ is representable. \\

One thinks of the morphism $X^\text{an}\to X$ as doing two things. It is an enlargement of the standard topology of $X^\text{an}$ to the Zariski topology of $X$, and it introduces generic points outside of $X(\C)$ (Indeed we will see that $X^\text{an}$ has the same underlying set as $X(\C)$). 

\begin{thm}{}{}\\
Let $X$ be a scheme locally of finite type over $\C$. Then the analytification of $X$ exists and is unique up to unique isomorphism. 
\end{thm}

\begin{defn}{The Analytification Functor}{}\\
Define the analytification functor $$(-)^\text{an}:\bold{Sch}_\C\to\bold{An}$$ as follows. 
\begin{itemize}
\item For each scheme $X$ locally of finite type over $\C$, we have $X^\text{an}$ is the analytification of $X$. 
\item For a morphism of schemes $f:X\to Y$ over $\C$, the universal propertyof $Y^\text{an}$ induces a morphism $X\to Y^\text{an}$. Precomposing with the canonical morphism $X^\text{an}\to Y^\text{an}$ gives the desired map $X^\text{an}\to Y^\text{an}$. 
\end{itemize}
\end{defn}

\begin{prp}{}{}\\
Let $X$ be a scheme locally of finite type over $\C$. Then the underlying set of $X^\text{an}$ is given by $X(\C)$. 
\end{prp}

\begin{prp}{}{}\\
Let $X$ be a scheme locally of finite type over $\C$. Then the following are true. 
\begin{itemize}
\item $X$ is separated if and only if $X^\text{an}$ is Hausdorff. 
\item $X$ is connected if and only if $X^\text{an}$ is connected. 
\item $X$ is smooth if and only if $X^\text{an}$ is a complex manifold. 
\item $X$ is proper if and only if $X^\text{an}$ is compact. 
\end{itemize}
\end{prp}

\begin{prp}{}{}\\
Let $X$ be a variety over $\C$. Then the following are true. 
\begin{itemize}
\item If $X$ is irreducible, then $X^\text{an}$ is connected. 
\item If $X$ is smooth and affine, then $X^\text{an}$ is a Stein manifold. 
\item If $X$ is smooth and projective, then $X^\text{an}$ is Kahler. 
\end{itemize}
\end{prp}

\begin{thm}{GAGA}{}\\
Let $X$ be a proper scheme of finite type over $\C$. Then the following are true. 
\begin{itemize}
\item Let $\mF$ be a sheaf of $\mO_X$-modules. Then $\mF^\text{an}$ is a sheaf of $\mO_{X^\text{an}}$-module. 
\item Let $\mF$ be a sheaf of $\mO_X$-modules. There is a natural isomorphism $$H^i(X,\mF)\cong H^i(X^\text{an},\mF^\text{an})$$
\item The analytification functor induces an equivalence of categories $$\bold{Coh}_{\mO_X}\cong\bold{Coh}_{\mO_{X^\text{an}}}$$
\item Let $Y\subseteq X^\text{an}$ be a closed complex analytic subspace. Then there exists a closed subvariety $Z$ of $X$ such that $Z^\text{an}=Y$. 
\end{itemize}
\end{thm}

\begin{crl}{Chow's Theorem}{}\\
Let $X$ be a closed complex analytic subspace of $\C\Prj^n$. Then $X$ is a projective variety over $\C$. 
\end{crl}

The important point is that: the theory of smooth projective complex varieties are subsumed by the theory of Kahler manifolds. Divisors and line bundles correspond bijectively thanks to Chow's theorem. 

\begin{thm}{Grothendieck}{}\\
Let $X$ be a smooth variety over $\C$. Then there is a natural isomorphism $$H_\text{dR}^n(X)\cong H_\text{Holo}^n(X^\text{an})\cong H_\text{dR}^n(X^\text{an})$$ induced by the canonical map $\alpha:X^\text{an}\to X$. 
\end{thm}

\begin{crl}{}{}\\
Let $X$ be a smooth variety over $\C$. Then the above isomorphism and de Rham's theorem induces a natural isomrophism $$H_\text{dR}^n(X)\cong H_\text{sing}^n(X^\text{an};\C)$$
\end{crl}

\begin{thm}{The Period Isomorphism}{}\\
Let $k$ be a subfield of $\C$. Let $X$ be a smooth variety over $k$. Then there is a natural isomorphism $$H_\text{dR}^\ast(X)\otimes_k\C\cong H_\text{sing}^\ast(X^\text{an};\Q)\otimes_\Q\C$$
\end{thm}

\subsection{Divisors}
\begin{defn}{Divisors on a Complex Manifold}{}\\
Let $X$ be a complex manifold. A divisor on $X$ is a formal linear combination $$D=\sum_{\substack{V\text{ is an analytic variety of }X\\\text{ of dimension }\dim(X)-1}}n_V\cdot V$$ where $n_V\in\Z$, and such that for each $p\in X$, there exists an open neighbourhood $U$ of $p$ such that $U$ meets finitely many $V$ in the summand. 
\end{defn}

The locally finite condition here is required because in Algebraic Geometry, Noetherianess is assumed, otherwise they are the same. 

\begin{defn}{Group of Divisors}{}\\
Let $X$ be a complex manifold. Define the group of divisors of $X$ to be the set $$\text{Div}(X)=\{D\;|\;D\text{ is a divisor on }X\}$$ together with addition given as a subgroup of $\prod_V\Z\cdot V$. 
\end{defn}

\begin{defn}{Effective Divisors}{}\\
Let $X$ be a complex manifold. Let $D=\sum_Vn_V\cdot V\in\text{Div}(X)$ be a divisor. We say that $D$ is effective if $n_V\geq 0$ for all $V$. 
\end{defn}

\begin{defn}{Divisors of Meromorphic Functions}{}\\
Let $X$ be a complex manifold. Let $f\in\mM_X^\ast$ be a meromorphic function. Define the divisor of $f$ to be $$\text{div}(f)=\sum_V\nu_V(f)\cdot V$$ where $\nu_V$ is the valuation of $\mO_{X,V}$. 
\end{defn}

\begin{defn}{Cartier Divisors on a Complex Manifold}{}\\
Let $X$ be a complex manifold. Define the group of divisors of $X$ to be $$\text{CDiv}(X)=H^0(X,\mM_X^\ast/\mO_X^\ast)=\Gamma(X,\mM_X^\ast/\mO_X^\ast)$$
\end{defn}

\begin{prp}{}{}\\
Let $X$ be a complex manifold. Then there is a natural isomorphism $$\text{Div}(X)\cong\text{CDiv}(X)$$
\end{prp}

\begin{prp}{}{}\\
Let $X$ be a complex manifold. Then the following diagram is commutative: \\

where the bottom horizontal map is induced by the long exact sequence of the short exact sequence $0\to\mO_X^\ast\to\mM_X^\ast\to\frac{\mM_X^\ast}{\mO_X^\ast}\to 0$. 
\end{prp}

By Serre's correspondence, holomorphic line bundles correspond to locally free sheaves of $\mO_M$-modules of rank $1$, and recall isomorphism classes of these objects form a group structure with the tensor product, and it is group isomorphic to $H^1(M,\mO_M^\ast)$. 

\begin{defn}{The Exponential Sequence}{}\\

\end{tikzcd}} \\
\end{defn}

Let $M$ be a complex manifold. Let $L$ be a holomorphic line bundle on $M$. Then the first chern class of $L$ is given by the image of $L$ in the homomorphism $$\text{Pic}(M)\cong H^1(M,\mO_M^\ast)\to H^2(M,\underline{\Z})\cong H^2(M,\Z)$$ induced by the long exact sequence of the exponential sequence. 
\end{prp}

\item $X$ admits a positive line bundle.

\end{itemize}

\end{crl}

Let $X$ be a compact Kahler manifold. Let $p:L\to X$ be a positive line bundle. Then we have $$H^q(X,\Omega^p(L))=0$$ for $p+q>\dim(X)$. 
\end{thm}

\end{thm}

\begin{thm}{The Lefschetz $(1,1)$ Theorem}{}\\

\end{thm}

\end{itemize}
\end{defn}

\begin{defn}{The Order of a Function at a Subvariety}{}\\
Let $k$ be a field. Let $X$ be a scheme of finite type over $k$. Let $W$ be an irreducible subvariety of $X$. Let $f/g\in K(W)^\ast$ for $f,g\in\mO_X(W)$ and $g\neq 0$. \end{defn}

\begin{defn}{The Group of $n$-Cycles}{}\\
Let $k$ be a field. Let $X$ be a scheme of finite type over $k$. Let $n\in\N$. Define the group of $n$-cycles on $X$ to be the free abelian group $$Z_n(X)=\Z\langle V\;|\;\substack{V\text{ is an }n\text{-dimensional}\\\text{irreducible subvariety of }X}\rangle$$
\end{defn}

\begin{lmm}{}{}\\
Let $k$ be a field. Let $X$ be a scheme of finite type over $k$. Let $W$ be an irreducible subvariety of $X$. Let $f\in K(W)^\ast$. Then $\text{div}(f)\in Z_{\dim(W)-1}(X)$. 
\end{lmm}

\begin{defn}{Cycles Rationally Equivalent to Zero}{}\\
Let $k$ be a field. Let $X$ be a scheme of finite type over $k$. Let $V\in Z_n(X)$. We say that $V$ is rationally equivalent to $0$ if there exists finitely many $n+1$-dimensional irreducible subvarieties $W_1,\dots,W_r$ of $X$ and $f_i\in K(W_i)^\ast$ such that $$V=\sum_{i=1}^n\text{div}(f_i)$$
\end{defn}

\begin{prp}{}{}\\

Then $V$ is rationally equivalent to $0$ i and only if there exists $(n+1)$-dimensional irreducible subvarieties $W_1,\dots,W_r$ of $X\times\Prj_k^1$ such that the projection maps $p_i:W_i\to\Prj_k^1$ are dominant and $V=\sum_{i=1}^r[p_i^{-1}(0)]-[p_i^{-1}(\infty)]$
\end{prp}

\begin{defn}{The Chow Group}{}\\
Let $k$ be a field. Let $X$ be a scheme of finite type over $k$. Define the $n$th Chow group of $X$ to be $$\text{CH}_n(X)=\frac{Z_n(X)}{\sim}$$ where we say that $V_1\sim V_2$ if $V_1-V_2$ is rationally equivalent to $0$. In this case, we say that $V_1$ and $V_2$ are rationally equivalent. 
\end{defn}

\section{Algebraic de Rham Cohomology}
\begin{defn}{Algebraic de Rham Cohomology}{}\\
Let $S$ be a scheme. Let $X$ be a scheme over $S$. Define the $n$th algebraic de Rham cohomology group by $$H_\text{dR}^n(X/S)=\H^n(X;\Omega_{X/S}^\bullet)$$
\end{defn}

\begin{lmm}{}{}\\
Let $S$ be a scheme. Let $X$ be a scheme over $S$. Then $\text{Tot}^\oplus\left(\mC^\bullet(X,\mU,\Omega_{X/S}^\bullet)\right)$ is an acyclic resolution of $\Omega_{X/S}^\bullet$. 
\end{lmm}

Thus one way to compute the de Rham cohomology would be $$H_\text{dR}^n(X/S)\cong H^n(\text{Tot}^\oplus(\mC^\bullet(X,\mU,\Omega_{X/S}^\bullet))$$

\begin{eg}{}{}\\
Let $U_0=\text{Spec}(R_0=k[x,y]/(y^2-f(x)))$ where $k$ is a field of characteristic $0$ and $f$ is separable and $\deg(f)=2g+1$ or $2g+2$. Let $U_1=\text{Spec}(R_1=k[u,v]/(v^2-h(u)))$ where $h(u)=u^{2g+2}h(1/u)$. Then we have $$H_\text{dR}^i(X/k)=H^i\left(R_0\oplus R_1\to R_0\frac{dx}{y}\oplus R_1\frac{du}{v}\oplus R_0[1/x]\to R_0[1/x]dx\right)$$ where the first map is given by $(w,z)\mapsto(dw,dz,w-z(1/x,y/x^{g+1}))$, the second map is given by $(wdx,zdu,g(x,y))\mapsto wdx-zdu-dg$ where $du=-z(1/x, y/x^{g+1})dx/x^2$. 
\begin{itemize}
\item $H_\text{dR}^0(X/k)\cong k$. 
\item $H_\text{dR}^2(X/k)\cong H_\text{dR}^1(X/k)$.

\section{The Nature of Mathematical Physics}
In mathematical physics, physical observations are given a mathematical formalism. A recurring theme is that mathematics often predict new phenomena beyond the original observation. 

\subsection*{Mechanics}
Mechanics is the part of physics that study the motion of bodies. Statics is the subject that studies equilibriums. Kinematics is concerned with general properties of motions regardless of the source. Dynamics study motion caused by forces or interactions. In regards to the system that is studied, mechanics can also be divided into material points, rigid bodies, elastic bodies and fluids. \\

Mechanics have received a number of revolutionary ideas that transformed the subject. Completely. In the modern day there are three main branches of mechanics: classical mechanics, relativistic mechanics and quantum mechanics. 

\subsubsection*{Classical Mechanics}
Theoretically, there are three different formalisms of classical mechanics that are related: Newtonian, Lagrangian and Hamiltonian. Newtonian mechanics is the easiest to formulate. Lagrangian is based on the least action principle, which is more sophisticated but is efficient for mechanical systems that are under constraints. Hamiltonian mechanics connects the bridge between classical mechanics and quantum mechanics. 

For a mathematical formalism of classical mechanics, we look for a mathematical model that describes the following physically observed phenomena. 
\begin{itemize}
\item Our space is three dimensional and time is one dimensional. 
\item There exists coordinate systems called inertial such that all laws of nature at all moments of time are the same in all inertial coordinates, and all coordinate systems in uniform rectilinear motion with respect to an inertial one are themselves inertial. 
\item The initial state of a mechanical system uniquely determines all of its motion. 
\end{itemize}

The limitations of classical mechanics come from Maxwell's equations, which says that electromagnetic waves travel at a fixed speed regardless of which inertial frame one uses. In particular, classical mechanics is a good enough estimate for motions of bodies that are significantly less than the speed of light, and whose bodies are macroscopic objects. 

\subsubsection*{Relativistic Mechanics}
For a mathematical formalism of relativistic mechanics, we look for a mathematical model that describes the following physically observed phenomena. 
\begin{itemize}
\item The Principle of Relativity: The laws of physics are the same in all inertial frames. 
\item The speed of light is the same for all inertial frames, regardless of the motion of the source or observer. 
\end{itemize}
A number of consequences appear because the speed of light being constant. 
\begin{itemize}
\item Time dilation: Moving clocks run slower. 
\item Length contraction: Moving objects are shortened along the direction of motion. 
\item Relativity of simultaneity: Two events simultaneous in one frame are not simultaneous in another. 
\item Speed do not add linearly, nothing with mass can have speed that is the speed of light. 
\item Mass energy equivalence: $E=mc^2$. 
\item Instantaneous interactions are impossible, and so must propagate at finite speed. 
\end{itemize}

Special relativity introduces the framework that is Lorentz geometry for the universe. This framework is one that is compatible with the observation that is the speed of light is the same for all inertial frames. In particular, it deduces the relativity of simultaneity, which means that instantaneous interactions are impossible. This introduces the concept of fields and field theory. \\

The limitations of relativistic mechanics come from general relativity, which says that gravity should be a result of the geometry, not a force. Relativistic mechanics does not take this to account. Also, classical field theory cannot handle the destruction and creation of particles.

\begin{defn}{Measure Preserving Systems}{}\\
Let $(X,\mF,\mu)$ be a measure space. Let $T:X\to X$ be a measurable function. We say that $T$ is a measure preserving function if $$\mu(T^{-1}(E))=\mu(E)$$ for all $E\in\mF$. In this case, we call $(X,\mF,\mu,T)$ a measure preserving system. 
\end{defn}

\begin{defn}{Probability Preserving System}{}\\
Let $(X,\mF,\mu,T)$ be a measure preserving system. We say that it is a probability preserving system if $(X,\mF,\mu)$ is a probability space. 
\end{defn}

\begin{thm}{Poincare Recurrence Theorem}{}\\
Let $(X,\mF,\mu,T)$ be a probability preserving system. Let $E\in\mF$. Then we have $$\mu(\{x\in E\;|\;\exists N\text{ such that }T^n(x)\notin E\text{ for all }n>N\})=0$$
\end{thm}

In other words, most points in $E$ go back to $E$ infinitely often after applying a certain number of $T$'s. 

\begin{defn}{Measure Theoretically Isomorphic}{}\\
Let $(X,\mF,\mu,T)$ and $(Y,\mE,\sigma,S)$ be measure preserving systems. Let $\pi:X\to Y$ be a measurable function. We say that $\pi$ is a measure theoretically isomorphism if the following are true. 
\begin{itemize}
\item There exists $A\subseteq X$ and $B\subseteq Y$ such that $\mu(X\setminus A)=\mu(Y\setminus B)=0$ and $\pi|_A:A\to B$ is invertible with a measurable inverse. 
\item For all $E\in\mE$, we have $\mu(\pi^{-1}(E))=\sigma(E)$. 
\item $S\circ\pi=\pi\circ T$. 
\end{itemize}
\end{defn}

\begin{defn}{The Euler-Lagrange Operator}{}\\
Let $p:E\to M$ be a smooth vector bundle over Minkowski spacetime. Let $\mL:J^1(E)\to\text{Den}(M)$ be a Lagrangian. Define the Euler-Lagrange operator $\text{EL}:\mC^\infty(E)\to\mC^\infty(\Hom(E,\text{Den}(M)))$ as follows. Let $(U,x^1,\cdots,x^4)$ be a chart of $M$. Let $(p^{-1}(U),x^i,y^j)$ be a local trivialization chart of $E$. Let $(p^{-1}(U),x^i,y^j,y_i^j)$ be a local chart on $J^1(E)$ where $y_i^j\in\mC^\infty(J^1(E))$ is such that $y_i^j(j_p^1\phi)=\frac{\partial y^j\circ\phi}{\partial x^i}\bigg{|}_p$. \end{defn}

To explain: 
\begin{itemize}
\item For $v\in E_p$, $y^j(v)$ picks out the $j$th coordinate of $v$ according to our local chart. 
\item $\frac{\partial L}{\partial y^j}$ is a smooth function of $J^1(E)$. We may compose with $j^1\phi:M\to J^1(E)$ to obtain a smooth function from $M$ to $\R$. 
\item Similarly, $\frac{\partial L}{\partial y_i^j}\circ j^1\phi$ is a smooth function from $M$ to $\R$. Using the local chart of $M$, we may differentiate this function with respect to $x^i$ to obtain a another smooth function from $M$ to $\R$. 
\item Notice that everything before the Minkowski density is indeed a real number. Hence $\text{EL}(\phi)_p(v)$ is indeed an element of $\text{Den}_p(M)\cong\R$. 
\end{itemize}

Let $p:E\to M$ be a smooth vector bundle over Minkowski spacetime. Let $\mL:J^1(E)\to\text{Den}(M)$ be a Lagrangian. Let $\phi\in\mC^\infty(E)$. Then we have $$\text{EL}(\phi)=0$$ if and only if $\delta S[\phi,\delta\phi]=0$ for all $\delta\phi\in\mC_{\partial\Omega}^\infty(E|_\Omega)$. 
\end{thm}

\section{Electromagnetism}
Classical field theory is developed to accommodate discoveries in the field of electromagnetism. We demonstrate the power of this language. 

\subsection*{The Observation of Charge}
Certain objects, when rubbed together, exert forces on one another across empty space. A glass rod rubbed with silk attracts a rubber rod rubbed with fur. Two glass rods rubbed with silk repel each other. Two rubber rods repel each other. \\

This observation suggests that matter can acquire a property which causes it to exert and respond to forces. We call this property \textit{electric charge}. The pattern of attraction and repulsion suggests that charge comes in two varieties. We label them positive and negative, with the convention that like charges repel and opposite charges attract. \\

A key further observation is that charge is \textit{conserved}: in any isolated process, the total charge never changes. When a glass rod is rubbed with silk, charge is not created --- it is transferred from one object to the other, leaving the total unchanged. 

\subsection*{Coulomb's Law and the Electric Field}
In 1785, Charles-Augustin de Coulomb measured the force between charged spheres using a torsion balance. He found that the force between two point charges $q_1$ and $q_2$ separated by a distance $r$ is: $$F=k_e\frac{q_1 q_2}{r^2}$$ where $k_e$ is a constant of proportionality. The force is repulsive if the charges have the same sign, and attractive if they have opposite signs. This is Coulomb's law. \\

The \textit{electric field} is the conceptual tool that makes sense of how this force is transmitted. Rather than saying that charge $q_1$ acts directly on charge $q_2$ at a distance, we say that $q_1$ modifies the space around it, producing an electric field $\E$. The charge $q_2$ then responds to the field it finds itself in, not directly to $q_1$. The field is defined operationally as the force per unit charge exerted on a small positive test charge: $$E(x)=\lim_{q \to 0}\frac{F(x)}{q}$$
The limit ensures we do not disturb the field we are trying to measure. The electric field is physically real: it carries energy and propagates at finite speed, so distant charges cannot influence each other instantaneously. 

\subsection*{Magnetism and Faraday's Discovery}
Magnetic phenomena were known long before electromagnetism was unified. Lodestones attract iron. Compass needles align with the Earth's field. For centuries, electricity and magnetism appeared to be entirely separate phenomena. \\

In 1820, Hans Christian Orsted discovered that a current-carrying wire deflects a compass needle placed nearby. A moving charge produces a \textit{magnetic field} $B$ in the space around it, just as a static charge produces an electric field. Andre-Marie Ampere then showed quantitatively that parallel currents attract, and anti-parallel currents repel. \\

In 1831, Michael Faraday made the reciprocal discovery: a changing magnetic field drives a current in a nearby wire. He described this in terms of \textit{lines of force} threading through a loop. When the number of these lines changes --- whether by moving a magnet, changing a current, or moving the loop itself --- a voltage is induced in the loop. This is electromagnetic induction. \\

Faraday also observed that magnetic field lines always form closed loops. No matter how a magnet is cut or arranged, one never finds an isolated magnetic ``charge'' (a \textit{monopole}) from which field lines emanate. This asymmetry between electricity and magnetism --- electric charges exist, magnetic charges do not --- is a fundamental observational fact. 

\subsection*{The Unification}
The deepest insight of the theory, clarified by Einstein's special relativity, is that electric and magnetic fields are not truly separate objects. They are two facets of a single \textit{electromagnetic field}, seen from different reference frames. \\

An observer watching a stationary charge sees only an electric field. A second observer moving relative to the first sees the same charge in motion --- a current --- and therefore sees both an electric and a magnetic field. The same physical situation produces different fields depending on the observer's state of motion. This is not a paradox; it means the split between $E$ and $B$ is frame-dependent, while the underlying electromagnetic field is the invariant object. 

\subsection*{Maxwell's Equations}
Experiment 1: Coulomb's Law: The electric field produced by a point charge $q$ at the origin is given by the map $E:\R^3\to\R^3$ where $$E(x)=\frac{q}{4\pi\varepsilon_0}\frac{x}{\abs{x}^2}$$ for some constant $\varepsilon_0\in\R$. 

Axiom 1: Gauss's Law: Denote $\rho$ by the charge per unit volume of $V\subseteq\R^3$. From the above we can deduce $$\int_SE\cdot dS=\frac{1}{\varepsilon_0}\int_V\rho dV$$, which equivalently says that $$\nabla\cdot E=\frac{\rho}{\varepsilon_0}$$


Experiment 2: $$\int_SB\cdot dS=0$$ for a closed surface $S$. 

Axiom 2: Gauss's Law: $$\nabla B=0$$

Experiment 3: Faraday observed $$\int_CE\cdot dl=-\frac{d}{dt}\int_\Sigma B\cdot dS$$ where $C$ bounds the surface $\Sigma$. 

Axiom 3: Faraday's Law: $$\nabla\times E=-\frac{\partial B}{\partial t}$$

Experiment 4: Ampere observed???

Axiom 4: Ampere Maxwell Law: $$\nabla\times B=\mu_0J+\mu_0\varepsilon_0\frac{\partial E}{\partial t}$$

Theorem: the above implies conservation of charge. 

Lorentz force law: $$F=q(E+v\times B)$$

Something called the field potential $F$ packs axiom 2 and 3 into a convenient package.

\section{The Logical Chain of Classical Field Theory and Gauge Theory}

\subsection*{Step 1: The Field}

We observe a physical quantity distributed over spacetime — a temperature, a velocity, an electromagnetic strength. Mathematically this is a fiber bundle $\pi: E \to M$, where $M$ is spacetime and the fiber $E_x$ is the space of possible values of the quantity at $x$. A \textbf{section} $\phi \in \Gamma(E)$ is one complete specification of the quantity at every point — a possible physical configuration.

\subsection*{Step 2: The Lagrangian as a Choice of Physics}

A Lagrangian density $\Lambda \in \Omega^n(J^kE)$ is a choice of action functional
\[
S[\phi] = \int_\Omega (j^k\phi)^*\Lambda
\]
Different Lagrangians encode different physical theories. The Euler-Lagrange equations — the equations of motion — are derived from $\Lambda$ by the variational principle. The physical sections are the critical points of $S$.

\subsection*{Step 3: Observing a Symmetry}

You compute the Euler-Lagrange equations and notice that the Lagrangian $\Lambda$ is invariant under some subgroup of automorphisms of $E$. An \textbf{automorphism} of $\pi: E \to M$ is a fiber-preserving diffeomorphism $\Phi: E \to E$. Such an automorphism acts on sections by
\[
\phi \mapsto \Phi \circ \phi
\]
and the Lagrangian being invariant means
\[
S[\Phi \circ \phi] = S[\phi] \quad \text{for all } \phi \in \Gamma(E)
\]
At this stage, the symmetry group acts the same way at every point — it is \textbf{global}. You observe this as an empirical fact about your Lagrangian. 

Symmetries of the action induces certain laws of physics, such as conservation of energy, momentum, charge etc. This is Noether's theorem. 

Another note: Rather than starting with a Lagrangian and observe the symmetries of its action, we may also start with a set of symmetries that we would like to enforce, apply it to all differential forms and retain only the invariants, and see what Lagrangian you would get. 

\subsection*{Step 4: The Absolute vs Relative Observation}
The global symmetry exists because the field takes values in a space that has \textbf{no preferred origin or frame} — the fiber $E_x$ has a group of self-maps (rotations, phase shifts, color rotations) that are physically unobservable. Different observers, or different conventions, would describe the same physical situation using sections related by this group action. The global symmetry is an artifact of having made consistent conventional choices everywhere at once. We need not restrict ourselves to consider symmetries of the entire space, and this is supported by physics: 

\subsubsection*{Argument 1: Special Relativity}

Special relativity says there is no notion of simultaneity — two spacelike separated points have no canonical notion of ``the same moment.'' A global transformation is therefore not a physically coherent operation in a relativistic theory. If you believe in special relativity, you are forced to allow your symmetry to act differently at different points. The global symmetry is then a special case of the local one, not the fundamental object.

\subsubsection*{Argument 2: The Gauge Principle}

The difference induced by the symmetry of the action is not observable. The phase is therefore a \textbf{convention} --- a choice made by an observer at $x$ to  describe the field. If the phase is a convention, then different observers at different points of spacetime should be free to make different conventional choices independently. There is no physical mechanism that would force their conventions to agree globally. The true symmetry group of the theory is therefore not the global group $G$ but the local group $mathcal{G} = \mathrm{Map}(M, G)$ of all smooth $G$-valued functions on spacetime.

This is the \textbf{gauge principle}: any redundancy in the description of a physical system should be a local redundancy. It is a postulate, not a theorem. Its justification is that it works --- every time it is applied, it produces a physically correct and predictive theory.

\subsection*{Step 5: Promoting to Local Symmetry}

This forces you to demand invariance not just under global $G$-transformations
\[
\phi(x) \mapsto g \cdot \phi(x), \quad g \in G \text{ constant}
\]
but under local $G$-transformations
\[
\phi(x) \mapsto g(x) \cdot \phi(x), \quad g: M \to G \text{ smooth}
\]

The Lagrangian, built from ordinary derivatives $\partial_\mu\phi$, immediately fails to be 
locally invariant. The reason is fundamental: $\partial_\mu\phi$ compares field values at 
nearby points $x$ and $x + dx$, which under local symmetry live in fibers with 
\textbf{independent conventions}. Subtracting them is meaningless — it is comparing 
quantities measured in different units.

\subsection*{Step 6: The Mathematics Is Now Forced}

To make sense of differentiation under local symmetry, we need a way to 
\textbf{compare vectors in nearby fibers} compatibly with the $G$-structure. But this is 
precisely the definition of a \textbf{connection}.
Where does this connection live? The answer is that the principal bundle is not introduced 
from outside — it is \textbf{already contained inside the vector bundle $E$}.

The fibers $E_x$ are vector spaces, but the isomorphism $E_x \cong \mathbb{R}^r$ is not 
canonical — it requires a choice of basis. A \textbf{frame} at $x$ is an ordered basis for 
$E_x$. The set of all frames at $x$ carries a natural free transitive right action of 
$GL(r, \mathbb{R})$, making it a $GL(r,\mathbb{R})$-torsor. Taking the disjoint union over 
all $x \in M$ produces the \textbf{frame bundle}
\[
P := \bigsqcup_{x \in M} \{\text{ordered bases for } E_x\}
\]
a principal $GL(r, \mathbb{R})$-bundle canonically constructed from $E$.

Now, the observation of a global $G$-symmetry in Step 3 is precisely the observation that 
the physically relevant frames are not arbitrary frames, but only those related by 
$G$-transformations. The structure group \textbf{reduces} from $GL(r, \mathbb{R})$ to $G$, 
producing a principal $G$-bundle
\[
P_G \subset P
\]
canonically determined by the symmetry. The original vector bundle is recovered as the 
associated bundle $E = P_G \times_G V$.

The connection demanded by local symmetry lives on $P_G$, because $P_G$ is the bundle whose 
fibers are $G$-torsors — they encode exactly the gauge freedom at each point. A connection 
on $P_G$ transports $G$-frames between nearby fibers, and therefore transports everything 
associated to those frames, including sections of $E = P_G \times_G V$.

\subsection*{Step 7: What Follows From The Maths}

Since the principal bundle is just a collection of frames at each fiber. A local section of the principal bundle is a choice of local coordinates. The mathematical fact that non-trivial principal bundles have no global sections can be interpreted as the fact that there is generally no globally consistent choice of coordinates because the symmetries now act locally. 

A connection in the principal bundle gives a covariant derivative in the associated bundle. This gives a way to compare phases across different time space.
