\section{Monomial Ideals}
\subsection{Basic Definitions}
\begin{defn}{Monomial Notation}{}\\
Let $R$ be a commutative ring. Let $u=(u_1,\dots,u_n)\in\N^n$. Define $$x^u=x_1^{u_1}\cdots x_n^{u_n}\in R[x_1,\dots,x_n]$$
\end{defn}

\begin{defn}{Monomial Ideals}{}\\
Let $R$ be a commutative ring. Let $I\subseteq R[x_1,\dots,x_n]$ be an ideal. We say that $I$ is a monomial ideal if $I$ is generated by monomials. 
\end{defn}

\begin{prp}{}{}\\
Let $R$ be a commutative ring. Let $I\subseteq R[x_1,\dots,x_n]$ be an ideal. Then the following are equivalent. 
\begin{itemize}
\item $I$ is a monomial ideal. 
\item Suppose that $f=\sum_{\alpha\in\N^n}c_\alpha x^\alpha\in I$. If $c_\alpha\neq 0$, then $x^\alpha\in I$. 
\item Suppose that $(c_1,\dots,c_n)\in(R^\ast)^n$. Then $(c_1,\dots,c_n)\cdot I=I$. 
\end{itemize}
\end{prp}

\begin{prp}{}{}\\
Let $R$ be a commutative ring. Let $m_1,\dots,m_k\in R[x_1,\dots,x_n]$ be monomials. Let $f=\sum_{\alpha\in\N^n}c_\alpha x^\alpha\in R[x_1,\dots,x_n]$. Then $f\in(m_1,\dots,m_k)$ if and only if $c_\alpha x^\alpha$ is a multiple of one of $m_1,\dots,m_k$. 
\end{prp}

\begin{prp}{}{}\\
Let $R$ be a commutative ring. Let $I\subseteq R[x_1,\dots,x_n]$ be a monomial ideal. Then there exists unique $u_1,\dots,u_m\in\N^n$ such that $$I=(x^{u_1},\dots,x^{u_m})$$
\end{prp}

\begin{prp}{}{}\\
Let $R$ be a commutative ring. Then the following are true. 
\begin{itemize}
\item The sum of monomial ideals is a monomial ideal. 
\item The product of monomial ideals is a monomial ideal. 
\item The intersection of monomial ideals is a monomial ideal. 
\end{itemize}
\end{prp}

\begin{defn}{Squarefree Monomials}{}\\
Let $R$ be a commutative ring. Let $u=(u_1,\dots,u_n)\in\N^n$. We say that $x^u$ is squarefree if $u_1,\dots,u_n\leq 1$. 
\end{defn}

\begin{defn}{Squarefree Monomial Ideals}{}\\
Let $R$ be a commutative ring. Let $I\subseteq R[x_1,\dots,x_n]$ be a monomial ideal. We say that $I$ is squarefree if it is generated by squarefree monomial. 
\end{defn}

\begin{prp}{}{}\\
Let $R$ be a commutative ring. Let $u_1,\dots,u_m\in\N^n$. Then we have $$\sqrt{(x^{u_1},\dots,x^{u_m})}=\left(x^{\min\{u_1,1\}},\dots,x^{\min\{u_m,1\}}\right)\in R[x_1,\dots,x_n]$$
\end{prp}

\begin{defn}{Quadratic Monomials}{}\\
Let $R$ be a commutative ring. Let $u=(u_1,\dots,u_n)\in\N^n$. We say that $x^u$ is squarefree if $\sum_{i=1}^nu_i=2$. 
\end{defn}

\begin{defn}{Quadratic Monomial Ideals}{}\\
Let $R$ be a commutative ring. Let $I\subseteq R[x_1,\dots,x_n]$ be a monomial ideal. We say that $I$ is quadratic if it is generated by quadratic monomial. 
\end{defn}

\begin{lmm}{}{}\\
Let $R$ be a commutative ring. Let $I\subseteq R[x_1,\dots,x_n]$ be an ideal. Then $I$ is a quadratic monomial ideal if and only if $\frac{R[x_1,\dots,x_n]}{I}$ is a quadratic algebra. 
\end{lmm}

\subsection{Monomial Matrices}
\begin{defn}{Partial Order on $\N^n$}{}\\
Let $x=(x_1,\dots,x_n),y=(y_1,\dots,y_n)\in\N^n$. We say that $x\succeq y$ if $x_i\geq y_i$ for all $i$. 
\end{defn}

\begin{defn}{Monomial Matrices}{}\\
Let $R$ be a commutative ring. Let $a_1,\dots,a_p,b_1,\dots,b_q\in\N^n$. Let $\phi:\bigoplus_{i=1}^pR[x_1,\dots,x_n](-a_i)\to\bigoplus_{j=1}^qR[x_1,\dots,x_n](-b_j)$ be a $R[x_1,\dots,x_n]$-module homomorphism. Let $A=(a_{ij})\in M_{q\times p}(R[x_1,\dots,x_n])$ be the matrix representing representing $\phi$. We say that $A$ is a monomial matrix if $A$ is of the form $$a_{ij}=\begin{cases}
\lambda_{ij}x^{a_i-b_j} & \text{ if }a_i\succeq b_j\\
0 & \text{ otherwise }
\end{cases}$$ for some $\lambda_{ij}\in R$. 
\end{defn}

\begin{defn}{Minimal Monomial Matrices}{}\\
Let $R$ be a commutative ring. Let $a_1,\dots,a_p,b_1,\dots,b_q\in\N^n$. Let $\phi:\bigoplus_{i=1}^pR[x_1,\dots,x_n](-a_i)\to\bigoplus_{j=1}^qR[x_1,\dots,x_n](-b_j)$ be a $R[x_1,\dots,x_n]$-module homomorphism. Let $A=(a_{ij})\in M_{q\times p}(R[x_1,\dots,x_n])$ be the matrix representing representing $\phi$. We say that $A$ is a minimal monomial matrix if $A$ is of the form $$a_{ij}=\begin{cases}
\lambda_{ij}x^{a_i-b_j} & \text{ if }a_i\succeq b_j\text{ and }a_i\neq b_j\\
0 & \text{ otherwise }
\end{cases}$$
\end{defn}

\begin{prp}{}{}\\
Let $R$ be a commutative ring. Let $I\subseteq R[x_1,\dots,x_n]$ be a monomial ideal. Let $(F_\bullet,d)$ be a graded free resolution of $I$. Then $F_\bullet$ is a minimal graded free resolution of $I$ if and only if for all $k\in\N$, the maps $d_k:F_k\to F_{k-1}$ are represented by minimal monomial matrices. 
\end{prp}

\section{The Stanley-Reisner Ring}
\subsection{Basic Definitions}
\begin{defn}{The Stanley-Reisner Ideal}{}\\
Let $\mK$ be a simplicial complex on $\{1,\dots,n\}$. Define the Stanley-Reisner ideal of $\mK$ to be $$I_\mK=\left\langle\prod_{j\in J}x_j\;\bigg{|}\;J\in\mP(\{1,\dots,n\})\setminus\mK\right\rangle\subseteq R[x_1,\dots,x_n]$$
\end{defn}

\begin{prp}{}{}\\
Let $R$ be a commutative ring. Let $n\in\N$. Let $\mK$ be an abstract simplicial complex on $\{1,\dots,n\}$. Then we have $$I_\mK=\bigcap_{J\in\mK}\langle x_i\;|\;i\notin J\rangle$$
\end{prp}

\begin{prp}{}{}\\
Let $R$ be a commutative ring. Let $n\in\N$. Then there is a one-to-one bijection $$\left\{\mK\subseteq\mP(\{1,\dots,n\})\;\bigg{|}\;\mK\text{ is a simplicial complex }\right\}\;\;\overset{1:1}{\longleftrightarrow}\;\;\left\{I\subseteq R[x_1,\dots,x_n]\;\bigg{|}\;\substack{I\text{ is a squarefree}\\\text{monomial ideal }}\right\}$$ given by $\mK\mapsto I_\mK$. 
\end{prp}

\begin{defn}{The Stanley-Reisner Ring}{}\\
Let $R$ be a commutative ring. Let $n\in\N$. Let $\mK$ be an abstract simplicial complex on $\{1,\dots,n\}$. Define the Stanley-Reisner ring of $\mK$ to be $$R[\mK]=\frac{R[x_1,\dots,x_n]}{I_\mK}$$
\end{defn}

\begin{eg}{}{}\\
Let $R$ be a commutative ring. The Stanley-Reisner ring of the standard $n$-simplex $\Delta^n$ is given by $$R[\Delta^n]=k[x_1,\dots,x_n]$$
\end{eg}

\begin{eg}{}{}\\
Let $R$ be a commutative ring. The Stanley-Reisner ring of the abstract simplicial complex $S=\mP(\{1,2,3\})\cup\mP(\{1,3,4\})\cup\mP(\{2,4\})$ over $\{1,2,3,4\}$ is given by $$R[S]=\frac{k[x_1,x_2,x_3,x_4]}{(x_1x_2x_4,x_2x_3x_4)}$$
\end{eg}

\begin{prp}{}{}\\
Let $R$ be a commutative ring. Let $\mK_1,\mK_2$ be simplicial complexes. Then we have $$R[\mK_1\ast\mK_2]\cong R[\mK_1]\otimes_RR[\mK_2]$$
\end{prp}

\begin{defn}{Induced Map of Stanley-Reisner Rings}{}\\
Let $R$ be a commutative ring. Let $\mK$ and $\mL$ be simplicial complexes on $\{1,\dots,n\}$ and $\{1,\dots,m\}$ respectively. Let $f:\mK\to\mL$ be a simplicial map. Define the induced map of Stanley-Reisner rings $f^\ast:R[\mL]\to R[\mK]$ by $$f^\ast(y_j)=\sum_{i\in f^{-1}(j)}x_i$$
\end{defn}

\begin{thm}{Bruns-Gubeladze}{}\\
Let $k$ be a field. Let $\mK$ and $\mL$ be simplicial complexes on $\{1,\dots,n\}$ and $\{1,\dots,m\}$ respectively. If $k[\mK]$ and $k[\mL]$ are isomorphic, then there exists a bijection $f:\{1,\dots,n\}\to\{1,\dots,m\}$ such that the induced map $f:\mK\to\mL$ is an isomorphism. 
\end{thm}

\begin{prp}{}{}\\
Let $R$ be a commutative ring. Let $\mK$ be a simplicial complex. Then we have $$\dim(R[\mK])=\dim(\mK)+1$$
\end{prp}

\begin{prp}{}{}\\
Let $R$ be a commutative ring. Let $\mK$ be a simplicial complex on $\{1,\dots,n\}$. Then the following are equivalent. 
\begin{itemize}
\item $R[\mK]$ is a quadratic algebra. 
\item $R[\mK]$ is a Koszul algebra. 
\item $I_\mK$ is a quadratic monomial ideal. 
\item $\mK$ is a flag complex. 
\end{itemize}
\end{prp}

\begin{thm}{Stanley}{}\\
Let $k$ be a field. Let $n\in\N^+$. Let $\mK$ be a simplicial complex on $\{1,\dots,n\}$. Let $(f_{-1},\dots,f_{d-1})$ be the face vector of $\mK$. Let $(h_0,\dots,h_d)$ be the $h$-vector of $\mK$. Then the following are true. 
\begin{itemize}
\item The Hilbert function of $k[\mK]$ is given by $$HF_{k[\mK]}(n)=\begin{cases}
\sum_{i=-1}^{d-1}f_i\binom{n-1}{i} & \text{ if }n\geq 1\\
0 & \text{ if }n=0
\end{cases}$$
\item The Hilbert series of $k[\mK]$ is given by $$HS_{k[\mK]}(t)=\sum_{i=-1}^{d-1}f_i\frac{t^{i+1}}{(1-t)^{i+1}}=\frac{1}{(1-t)^d}\sum_{i=0}^dh_it^i$$
\item The Hilbert polynomial of $k[\mK]$ is given by $$HP_{k[\mK]}(n)=\sum_{i=-1}^{d-1}f_i\binom{n-1}{i}$$
\end{itemize}
\end{thm}

\begin{eg}{}{}\\
Let $\Delta^n$ be the standard $n$-simplex. Then the following are true. 
\begin{itemize}
\item The Hilbert series for the Stanley-Reisner ring of $\Delta^n$ is given by $$HS_{k[\Delta^n]}(t)=\frac{1}{(1-t^2)^{n+1}}$$
\item The Hilbert series for the Stanley-Reisner ring of $\partial\Delta^n$ is given by $$HS_{k[\partial\Delta^n]}(t)=\frac{1}{(1-t^2)^{n+1}}\sum_{i=0}^{n+1}t^{2i}$$
\end{itemize}
\end{eg}

We take $k[x_1,\dots,x_n]$ to be graded by $\deg(x_i)=2$ so that $k[x_1,\dots,x_n]$ is graded-commutative as a graded $k$-algebra. It follows that $\text{Tor}_p^{k[x_1,\dots,x_n]}(k[\mK],k)$ also acquires a grading (in which the odd degree components are $0$).

Let $k$ be a field or $\Z$. Let $\mK$ be a simplicial complex on $\{1,\dots,n\}$. Then the following are true. 
\begin{itemize}
\item There is an graded isomorphism $k$-vector spaces $$\text{Tor}_p^{k[x_1,\dots,x_n]}(k[\mK],k)\cong H_p(K_\bullet(x_1,\dots,x_n;k[\mK]))$$
\item There is an isomorphism of graded $k$-algebras $$\text{Tor}_\ast^{k[x_1,\dots,x_n]}(k[\mK],k)\cong H_\ast(K_\bullet(x_1,\dots,x_n;k[\mK]))$$
\end{itemize}
\end{prp}

\begin{prp}{}{}\\
Let $k$ be a field or $\Z$. Let $\mK$ and $\mL$ be simplicial complexes on $\{1,\dots,n\}$ and $\{1,\dots,m\}$ respectively. Let $f:\mK\to\mL$ be a simplicial map. Then $f$ induces a graded $k$-algebra homomorphism $$f^\ast:\text{Tor}_\ast^{k[y_1,\dots,y_m]}(k[\mL],k)\to\text{Tor}_\ast^{k[x_1,\dots,n]}(k[\mK],k)$$
\end{prp}

\begin{thm}{Hochster I}{}\\
Let $k$ be a field or $\Z$. Let $\mK$ be a simplicial complex on $\{1,\dots,n\}$. Then the following are true. 
\begin{itemize}
\item There is an isomorphism of $k$-vector spaces $$\text{Tor}_p^{k[x_1,\dots,n]}(k[\mK],k)_{2j}\cong\bigoplus_{\substack{J\subseteq\{1,\dots,n\}\\\abs{J}=j}}\widetilde{H}^{j-i-1}(\mK_J;k)$$
\item There is an graded isomorphism of $k$-vector spaces $$\text{Tor}_p^{k[x_1,\dots,x_n]}(k[\mK],k)\cong\bigoplus_{J\subseteq\{1,\dots,n\}}\widetilde{H}^{p-\abs{J}-1}(\mK_J,k)$$
\end{itemize}
\end{thm}

\end{cases}$$

\end{defn}

\subsection{Cohen Macaulay Complexes}
\begin{defn}{Cohen Macaulay Complexes}{}\\
Let $\mK$ be a simplicial complex. Let $R$ be a commutative ring. We say that $\mK$ is Cohen Macaulay over $R$ if $R[\mK]$ is a Cohen-Macaulay ring. 
\end{defn}

See 1.3.6

\begin{prp}{}{}\\
Let $k$ be a field. Let $\mK$ be a simplicial complex on $\{1,\dots,n\}$. Then $k[\mK]$ is Cohen-Macaulay if and only if the following are true. 
\begin{itemize}
\item $\dim_k\left(\text{Tor}_p^{k[x_1,\dots,x_n]}(k[\mK],k)\right)=0$ for all $p\geq n-\dim(\mK)$. 
\item $\dim_k\left(\text{Tor}_{n-\dim(\mK)-1}^{k[x_1,\dots,x_n]}(k[\mK],k)\right)\neq 0$. 
\end{itemize}
\end{prp}

\begin{thm}{Reisner}{}\\
Let $k$ be a field or $\Z$. Let $\mK$ be a simplicial complex on $\{1,\dots,n\}$. Then $k[\mK]$ is Cohen-Macaulay if and only if $$\widetilde{H}_i\left(\text{lk}_\mK(I);k\right)=0\;\;\;\;\;\;\;\;\text{ for all }I\in\mK\;\text{ and }\;0\leq i\leq\dim(\text{lk}_\mK(I))-1$$
\end{thm}

\begin{thm}{Munkres, Stanley}{}\\
Let $k$ be a field or $\Z$. Let $\mK$ be a simplicial complex on $\{1,\dots,n\}$. Then $k[\mK]$ is Cohen-Macaulay if and only if $$\widetilde{H}_i\left(\abs{\mK},\abs{\mK}\setminus\{x\};k\right)=\widetilde{H}_i(\mK;k)=0\;\;\;\;\;\;\;\;\text{ for all }x\in\abs{\mK}\;\text{ and }\;0\leq i\leq\dim(\mK)-1$$
\end{thm}

\begin{crl}{}{}\\
Let $\mK$ be a triangulation of $S^n$. Let $k$ be a field. Then $k[\mK]$ is Cohen-Macaulay. 
\end{crl}

\begin{prp}{}{}\\
Let $k$ be a field. Let $\mK$ be a simplicial complex on $\{1,\dots,n\}$. Then $\mK$ is Cohen Macaulay over $k$ if and only if $I_{\mK^\vee}$ admits a linear resolution. 
\end{prp}

\begin{defn}{Sequential Cohen Macaulay Complexes}{}\\
Let $\mK$ be a simplicial complex. Let $R$ be a commutative ring. We say that $\mK$ is sequentially Cohen Macaulay over $R$ if $$\mK^{\langle i\rangle}=\langle F\in\mK\;|\;\dim(F)\geq i\rangle$$ is Cohen Macaulay for all $i$. 
\end{defn}

\begin{prp}{}{}\\
Let $\mK$ be a simplicial complex. Let $k$ be a field. Then $\mK$ is sequentially Cohen Macaulay over $k$ if and only if there is a filtration $$0=M_0\subset\cdots\subset M_r=k[\mK]$$ of graded $k[\mK]$-modules such that $\frac{M_j}{M_{j-1}}$ is Cohen Macaulay and $\dim(M_1/M_0)<\cdots<\dim(M_r/M_{r-1})$. 
\end{prp}

\begin{prp}{}{}\\
Let $\mK$ be a simplicial complex. Let $k$ be a field. Then $\mK$ is Cohen Macaulay over $k$ if and only if $\mK$ is sequentially Cohen Macaulay and pure. 
\end{prp}

\begin{prp}{}{}\\
Let $k$ be a field. Let $\mK$ be a simplicial complex on $\{1,\dots,n\}$. Then $\mK$ is sequentially Cohen Macaulay over $k$ if and only if $I_{\mK^\vee}$ admits a componentwise linear resolution. 
\end{prp}

\subsection{Gorenstein Complexes}
\begin{defn}{Gorenstein Complexes}{}\\
Let $R$ be a commutative ring. Let $\mK$ be a simplicial complex. We say that $\mK$ is Gorenstein over $R$ if $R[\mK]$ is a Gorestein ring. 
\end{defn}

This is equivalent to saying that $\text{Tor}_{m-\dim(\mK)-1}^{R[x_1,\dots,x_n]}(R[\mK],R)\cong R$ since Gorenstein rings are Cohen-Macaulay rings of type $1$. 

\begin{prp}{Stanley's Combinatorial Criterion}{}\\
Let $k$ be a field. Let $\mK$ be a Cohen-Macaulay simplicial complex over $\{1,\dots,n\}$. Then $\mK$ is Gorenstein over $k$ if and only if $$\dim_k(\widetilde{H}_{\dim(\text{lk}(\sigma))}(\text{lk}(\sigma);k))\leq 1$$
\end{prp}

\begin{prp}{}{}\\
Let $\mK$ be a simplicial complex. Then $\mK$ is Gorenstein over $\Z$ if and only if it is Gorenstein over $k$ for all fields $k$. 
\end{prp}

\begin{defn}{The Core of a Simplicial Complex}{}\\
Let $\mK$ be a simplicial complex. Define the core of $\mK$ to be $$\text{core}(mK)=\mK_{\{v\in V(\mK)\;|\;\text{st}(v)\neq\mK\}}$$
\end{defn}

\begin{defn}{Gorenstein* Complexes}{}\\
Let $R$ be a commutative ring. Let $\mK$ be a simplicial complex. We say that $\mK$ is a Gorenstein* complex if the following are true. 
\begin{itemize}
\item $\mK$ is Gorenstein over $R$. 
\item $\text{core}(\mK)=\mK$. 
\end{itemize}
\end{defn}

$A=\bigoplus_{i=0}^d$ is a Poincare algebra if the map $A^i\to\Hom(A^{d-i},A^d)$ given by $a\mapsto(b\mapsto ab)$ is an isomorphism. 

\begin{prp}{}{}\\
Let $R$ be a commutative ring. Let $\mK$ be a simplicial complex. Then the following are equivalent. 
\begin{itemize}
\item $\mK$ is a Gorenstein* complex. 
\item For any $\sigma\in\mK$, we have $$H^i(\text{lk}(\sigma))\cong\begin{cases}
R & \text{ if }i=0\text{ or }\dim(\text{lk}(\sigma))\\
0 & \text{ otherwise }
\end{cases}$$
\item For any $x\in\abs{\mK}$, we have $$H^i(\abs{\mK},\abs{\mK}\setminus\{x\};R)\cong\widetilde{H}^i(\mK;R)\cong\begin{cases}
R & \text{ if }i=\dim(\mK)\\
0 & \text{ otherwise }
\end{cases}$$
\item $\text{Tor}_\ast^{R[x_1,\dots,x_n]}(R[\mK],R)$ is a Poincare duality algebra. 
\end{itemize}
\end{prp}

\section{Lattice Ideals and Affine Semigroups}
\begin{defn}{Lattice Ideal}{}\\
Let $L\subseteq\Z^n$ be a lattice. Let $R$ be a commutative ring. Define the lattice ideal of $L$ in $R[x_1,\dots,x_n]$ to be $$I_L=\langle x^u-x^v\;|\;u,v\in\N^n\text{ and }u-v\in L\rangle$$
\end{defn}

\begin{prp}{}{}\\
Let $k$ be a field. Let $S$ be a finitely generated commutative monoid. Let $s_1,\dots,s_n$ be a set of generators of $S$. Let $\phi:\Z^n\to G(S)$ be the group homomorphism defined by $e_i\mapsto s_i$. Then there is a $k$-algebra isomorphism $$k[S]\cong\frac{k[x_1,\dots,x_n]}{I_{\ker(\phi)}}$$
\end{prp}

\begin{prp}{}{}\\
Let $k$ be a field. Let $S$ be a finitely generated commutative monoid. Let $s_1,\dots,s_n$ be a set of generators of $S$. Then we have $$\dim(k[S])=n-\rank(\ker(\phi))$$
\end{prp}

\begin{defn}{Affine Semi-group}{}\\
Let $S$ be a monoid. We say that $S$ is an affine semi-group if there exists an injective monoid homomorphism $S\to\N^d$ for some $d\in\N$. 
\end{defn}

\begin{prp}{}{}\\
Let $k$ be a field. Let $S$ be a finitely generated commutative monoid. Let $s_1,\dots,s_n$ be a set of generators of $S$. Let $\phi:\Z^n\to G(S)$ be the group homomorphism defined by $e_i\mapsto s_i$. Then the following are equivalent. 
\begin{itemize}
\item $I_{\ker(\phi)}$ is prime. 
\item $k[S]$ is an integral domain. 
\item $G(S)$ is a free abelian group. 
\item $S$ is an affine semi-group. 
\end{itemize}
\end{prp}

\section{Polyhedral Products}
\subsection{The Homotopy Theory of Polyhedral Products}
\begin{defn}{Polyhedral Products}{}\\
Let $\mK$ be a simplicial complex on $\{1,\dots,n\}$. Let $(\bold{X},\bold{A})=\{(X_1,A_1),\dots,(X_n,A_n)\}$ be pairs of spaces. 
\begin{itemize}
\item Let $\sigma\in\mK$. Define $$(\bold{X},\bold{A})^\sigma=\left\{(x_1,\dots,x_n)\in\prod_{i=1}^nX_i\;\bigg{|}\;x_i\in A_i\text{ for }i\notin I\right\}$$
\item Define the polyhedral product of $(\bold{X},\bold{A})$ corresponding to $\mK$ to be $$(\bold{X},\bold{A})^\mK=\bigcup_{I\in\mK}(\bold{X},\bold{A})^\sigma$$
\end{itemize}
\end{defn}

\begin{prp}{}{}\\
Let $\mK$ be a simplicial complex on $\{1,\dots,n\}$. Let $(\bold{X},\bold{A})=\{(X_1,A_1),\dots,(X_n,A_n)\}$ be pairs of spaces. Then we have $$(\bold{X},\bold{A})^\mK=\colim_{\sigma\in\text{Cat}(\mK)}(\bold{X},\bold{A})^\sigma$$
\end{prp}

In particular, the assignment $(-)^\mK:(\bold{Top}^2)^n\to\bold{Top}$ is functorial. It is also a homotopy functor. 

\begin{prp}{}{}\\
Let $\mK$ be a simplicial complex on $\{1,\dots,n\}$. Let $(\bold{X},\bold{A})$ and $(\bold{Y},\bold{B})$ be $n$ pairs of spaces. If $(X_i,A_i)$ and $(Y_i,B_i)$ are homotopy equivalent for $1\leq i\leq n$, then there is a homotopy equivalence $$(\bold{X},\bold{A})^\mK\simeq(\bold{Y},\bold{B})^\mK$$
\end{prp}

\begin{eg}{}{}\\

Notice that this is precisely the cohomology ring of $(S^3\times S^4)^{\#5}$. 
\end{eg}

There are three descriptions of the cohomology groups: $$H^k(\mZ_\mK;\Z)\cong\bigoplus_{2q-p=k}H_{-p}\left(\Lambda\left(\bigoplus_{i=1}^n\Z[\mK]u_i\right)\right)_{2q}=\bigoplus_{J\subseteq\{1,\dots,n\}}\widetilde{H}^{k-\abs{J}-1}(\mK_J;\Z)$$ For $L\subseteq J$ such that $\alpha_L$ is a cochain basis in the Hochster formula, we have $$[\varepsilon(L,J)\chi(J\setminus L,L)]\leftrightarrow[\varepsilon(L,J)u_{J\setminus L}v_L]\leftrightarrow\alpha_L$$

\begin{prp}{}{}\\
Let $\mK$ be a simplicial complex on $\{1,\dots.n\}$. Then the following are true. 
\begin{itemize}
\item $\text{rank}(H^0(C^{-\ast,0}(\mZ_\mK;\Z)))=1$. 
\item $\text{rank}(H^0(C^{-\ast,2q}(\mZ_\mK;\Z)))=0$ for $q\neq 0$.
\item $\text{rank}(H^{-p}(C^{-\ast,2q}(\mZ_\mK;\Z)))$ for $p>q$ or $q>n$. 
\item $\text{rank}(H^{-p}(C^{-\ast,2q}(\mZ_\mK;\Z)))$ for $p\geq q>0$ or $q-p>\dim(\mK)+1$. 
\item $\text{rank}(H^{-1}(C^{-\ast,4}(\mZ_\mK;\Z)))=\binom{n}{2}-f_1$ (recall that $f_1$ is the number of edges of $\mK$). 
\item $\text{rank}(H^{-(n-\dim(\mK)+1)}(C^{-\ast,2n}(\mZ_\mK;\Z)))=\text{rank}(\widetilde{H}^{\dim(\mK)}(\mK))$. 
\end{itemize}
\end{prp}

\begin{prp}{}{}\\
Let $\mK$ be a simplicial complex on $\{1,\dots.n\}$. Then the following are true. 
\begin{itemize}
\item $\text{rank}(H^0(\mZ_\mK;\Z))=1$. 
\item $\text{rank}(H^1(\mZ_\mK;\Z))=\text{rank}(H^2(\mZ_\mK;\Z))=0$. 
\item $\text{rank}(H^3(\mZ_\mK;\Z))=\binom{n}{2}-f_1$ (recall that $f_1$ is the number of edges of $\mK$). 
\item $\text{rank}(H^{n+\dim(\mK)+1}(\mZ_\mK;\Z))=\text{rank}(\widetilde{H}^{\dim(\mK)}(\mK))$. 
\end{itemize}
\end{prp}

\begin{lmm}{}{}\\
Let $\mK$ be a simplicial complex on $\{1,\dots,n\}$. If $\mK$ is a triangulation of $S^{m-1}$, then the top cohomology $H^{n+m}(\mZ_\mK;\Z)$ is generated by $[\chi(\{1,\dots,n\}\setminus I,I)]$ for any $I\in\mK$ and such that $\abs{I}=n$. 
\end{lmm}

If we use the Koszul complex as the cohomology model for $\mZ_\mK$, then the top cohomology is generated by $u_{\{1,\dots,n\}\setminus I}v_I$ for any $I\in\mK$ such that $\abs{I}=n$.

\section{Geometric Structures of Moment Angle Complexes}
\subsection{Moment Angle Polytopes and Intersection of Quadrics}
Let $P\subseteq\R^n$ be a polytope defined by $$P=\{x\in\R^n\;|\;\langle a_i,x\rangle+b_i\geq 0\text{ for }1\leq i\leq m\}$$ such that $a_1,\dots,a_m$ span $\R^n$. Let $$A=\begin{pmatrix}
a_1\\
\vdots\\
a_m
\end{pmatrix}\in M_{m\times n}(\R)\;\;\text{ and }\;\;b=\begin{pmatrix}
b_1\\
\vdots\\
b_m
\end{pmatrix}\in\R^m$$ Recall that $$P=\{x\in\R^n\;|\;Ax+b\geq 0\}$$

\begin{defn}{Moment Angle Polytopes}{}\\
Let $P\subseteq\R^n$ be a polytope defined by $A\in M_{m\times n}(\R)$ and $b\in\R^m$ such that $a_1,\dots,a_m$ span $\R^n$. Let $i_P:\R^n\to\R^m$ be the map $i_P(x)=Ax-b$. Let $\mu:\C^m\to\R^m$ be the map $\mu(z_1,\dots,z_m)=(\abs{z_1}^2,\dots,\abs{z_m}^2)$. Define the moment angle polytope $\mZ_{A,b}$ to be the pullback \\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	{\mZ_{A,b}} & {\C^nm} \\
	{P} & {\R^m}
	\arrow[from=1-1, to=1-2]
	\arrow[from=1-1, to=2-1]
	\arrow["\mu", from=1-2, to=2-2]
	\arrow["i_P", hook, from=2-1, to=2-2]
\end{tikzcd}}\\
\end{defn}

\begin{prp}{}{}\\
Let $P\subseteq\R^n$ be a polytope defined by $A\in M_{m\times n}(\R)$ and $b\in\R^m$ such that $a_1,\dots,a_m$ span $\R^n$. Then there is a homeomorphism $$\mZ_{A,b}\cong\left\{z=(z_1,\dots,z_m)\in\C^m\;\bigg{|}\;\Gamma\begin{pmatrix}
\abs{z_1}^2\\
\vdots\\
\abs{z_m}^2
\end{pmatrix}=\Gamma b\right\}$$ where $\Gamma=\begin{pmatrix}
\gamma_1\\
\vdots\\
\gamma_{m-n}\end{pmatrix}$ and $\gamma_1,\dots,\gamma_{m-n}$ is a basis for $\ker(A^T)$. 
\end{prp}

This makes $Z_{A,b}$ an intersection of quadrics in $\R$, as well as an intersection of Hermitian quadrics (Hermitian since $\abs{z_i}^2=z_i\overline{z_i}$). 

\begin{defn}{Intersection of Quadrics}{}\\
Let $\Gamma\in M_{(m-n)\times n}(\R)$ be a matrix. Let $\delta\in\R^{m-n}$. Define the intersection of quadrics associated to $(\Gamma,\delta)$ by $$\mZ_{\Gamma,\delta}=\left\{z=(z_1,\dots,z_m)\in\C^m\;\bigg{|}\;\Gamma\begin{pmatrix}
\abs{z_1}^2\\
\vdots\\
\abs{z_m}^2
\end{pmatrix}=\delta\right\}$$
\end{defn}

\begin{lmm}{}{}\\
Let $\Gamma\in M_{(m-n)\times n}(\R)$ be a matrix. Let $\delta\in\R^{m-n}$. Then we have $$\frac{\mZ_{\Gamma,\delta}}{T^m}\cong\left\{y=(y_1,\dots,y_m)\in\R^m\;\bigg{|}\;\Gamma y=\delta\right\}$$
\end{lmm}

\begin{defn}{Associated Polytope}{}\\
Let $\Gamma\in M_{(m-n)\times n}(\R)$ be a matrix. Let $\delta\in\R^{m-n}$. Define the associated polytope of $\mZ_{\Gamma,\delta}$ to be $\mZ_{A,b}$ where 
\begin{itemize}
\item The columns $a_1,\dots,a_m$ of $A$ form a basis for $\ker(\Gamma^T)$. 
\item $b\in\R^m$ is such that $\Gamma b=\delta$. 
\end{itemize}
\end{defn}

\begin{thm}{}{}\\
There is a one to one correspondence $$\frac{\left\{\mZ_{A,b}\;\bigg{|}\;\substack{A\in M_{m\times n}(\R),b\in\R^m\text{ such that}\\\text{the columns }a_1,\dots,a_m\text{ of }A\text{ span }\R^n}\right\}}{\text{ Isomorphism of }\R^n}\;\;\overset{1:1}{\longleftrightarrow}\;\;\frac{\left\{\mZ_{\Gamma,\delta}\;\bigg{|}\;\substack{\Gamma\in M_{(m-n)\times n}(\R),\delta\in\R^{m-n}\text{ such that}\\\text{the columns }\gamma_1,\dots,\gamma_m\text{ of }\Gamma\text{ span }\R^{m-n}}\right\}}{\text{Linear isomorphism of }\R^{m-n}}$$
\end{thm}

\begin{prp}{}{}\\
Let $P\subseteq\R^n$ be a polytope defined by $A\in M_{m\times n}(\R)$ and $b\in\R^m$ such that $a_1,\dots,a_m$ span $\R^n$. Then $P$ is simple if and only if $\mZ_{A,b}$ is a smooth manifold of real dimension $m+n$. 
\end{prp}

\subsection{Relation to Moment Angle Complexes}
\begin{prp}{}{}\\
Let $P\subseteq\R^n$ be a polytope defined by $A\in M_{m\times n}(\R)$ and $b\in\R^m$ such that $a_1,\dots,a_m$ span $\R^n$. Then there is a homotopy equivalence $$\mZ_{\mK_P}\simeq\mZ_{A,b}$$
\end{prp}

\begin{prp}{}{}\\
Let $P\subseteq\R^n$ be a polytope defined by $A\in M_{m\times n}(\R)$ and $b\in\R^m$ such that $a_1,\dots,a_m$ span $\R^n$. If $P$ is simple, then there is a $T^m$-equivariant homeomorphism $$\mZ_{\mK_P}\cong\mZ_{A,b}$$
\end{prp}

There is a canonical choice of intersection of quadrics in this case. 

\begin{prp}{}{}\\
Let $P\subseteq\R^n$ be a polytope defined by $A\in M_{m\times n}(\R)$ and $b\in\R^m$ such that $a_1,\dots,a_m$ span $\R^n$. Suppose that $P$ is simple. Then there is a $T^m$-equivariant diffeomorphism $$\mZ_{\mK_P}\cong\left \{z=(z_1,\dots,z_m)\in\C^m\;\bigg{|}\;\sum_{k=1}^m\abs{z_k}^2=1,\sum_{k=1}^mg_k\abs{z_k}^2=0\right\}$$ where $g_1,\dots,g_m\in\R^{m-n-1}$ is a combinatorial gale diagram of $P^\ast$. 
\end{prp}

\begin{thm}{de Medrano}{}\\
Let $P\subseteq\R^n$ be a simple polytope. Suppose that $\mZ_{\mK_P}$ is a non-empty and non-degenerate intersection of $3$ quadrics. Then exactly one of the following hold. 
\begin{itemize}
\item $P$ is combinatorially equivalent to a product of three simplicies, and there is a diffeomorphism $$\mZ_{\mK_P}\cong S^{n_1}\times S^{n_2}\times S^{n_3}$$ where $n_1,n_2,n_3$ are odd. 
\item There is a diffeomorphism $$\mZ_{\mK_P}\cong\#_{k=3}^{n+1}(S^k\times S^{2n+3-k})^{\# q_k}$$ for some $q_k$ determined by the Gale diagram of $P$. 
\end{itemize}
\end{thm}

\section{Fat Wedge Filtrations of Polyhedral Products}
\subsection{Basic Definitions}
\begin{defn}{The Classical Fat Wedge Filtration}{}\\
Let $X_1,\dots,X_n$ be pointed spaces. Define the fat wedge filtration of $\prod_{i=1}^nX_i$ at the $k$th step by $$T^k(X_1,\dots,X_n)=\left\{(x_1,\dots,x_n)\in\prod_{i=1}^nX_i\;\bigg{|}\;\abs{\{i\;|\;x_i=\ast\}}\geq n-k\right\}$$
\end{defn}

\begin{defn}{Fat Wedge Filtration of Polyhedral Products}{}\\
Let $\mK$ be a simplicial complex on $\{1,\dots,n\}$. Let $(\bold{X},\bold{A},\ast)$ be $n$-pairs of pointed spaces. Define the fat wedge filtration of the polyhedral product at the $k$th step by $$(\bold{X},\bold{A})^{\mK,k}=(\bold{X},\bold{A})^\mK\cap T^k(X_1,\dots,X_n)$$
\end{defn}

\subsection{Decomposition of the Fat Wedge Filtration}
We consider the fat wedge filtration of $\R\mZ_\mK$. 

\begin{prp}{}{}\\
Let $\mK$ be a simpicial complex on $\{1,\dots,n\}$. Then we have $$\R\mZ_\mK^i=\bigcup_{I\subseteq\{1,\dots,n\},\abs{I}=i}\R\mZ_{\mK_I}$$
\end{prp}

Recall that cubical complex of a simplicial complex. We know that $$(D^1,S^0)^\sigma=\bigcup_{\substack{\mu\subseteq\tau\subseteq\text{Vert}(\mK)\\\tau\setminus\mu=\sigma}}C_{\mu\subseteq\tau}$$ and $$\R\mZ_{\mK}=\bigcup_{\sigma\in\mK}\bigcup_{\substack{\mu\subseteq\tau\subseteq\text{Vert}(\mK)\\\tau\setminus\mu=\sigma}}C_{\mu\subseteq\tau}$$

\begin{prp}{}{}\\
Let $n\in\N^+$. Let $\mK$ be a simplicial complex on $n$. Let $J\subseteq\mK$. Then there is a relative homeomorphism $$(\abs{\text{Cone}(\text{Bd}(\mK_J))},\abs{\text{Bd}(\mK_J)})\to(\R\mZ_{\mK_J},\R\mZ_{\mK_J}^{\abs{J}-1})$$
\begin{proof}\\
Let $i_c:\abs{\Delta^n}\to I^{n+1}$ denote the cubical embedding. For each $J\subseteq\{1,\dots,n\}$, the embedding restricts to a map $i_c^J:\abs{\mK_J}\to I^J$ where $I^J=(I,0)^J$ is the moment angle complex of $I$. For each $\mu\subseteq\tau\subseteq J$, denote $C_{\mu\subseteq\tau}^J$ the $C_{\mu\subseteq\tau}$ face of $I^J$ (so $C_{\mu\subseteq\tau}^J$ still lives inside $I^J$ but some coordinates are $0$). We compute that $$\R\mZ_{\mK_J}^{\abs{J}-1}=\bigcup_{\substack{A\subseteq J,\abs{A}=\abs{J}-1}}\bigcup_{\sigma\in\mK_A}\bigcup_{\substack{\mu\subseteq\tau\subseteq A\\\tau\setminus\mu=\sigma}}C_{\sigma\subseteq\tau}^A=\abs{\text{Bd}(\mK_J)}$$ and similarly $$\R\mZ_{\mK_J}=\bigcup_{\sigma\in\mK_J}\bigcup_{\substack{\mu\subseteq\tau\subseteq J\\\tau\setminus\mu=\sigma}}C_{\mu\subseteq\tau}^J=\abs{\text{Cone}(\text{Bd}(\mK_J))}$$
\end{proof}
\end{prp}

Note: $f:(X,A)\to(Y,B)$ is a relative homeomorphism if $f:X\to Y$ is a homeomorphism and $\widetilde{f}:X/A\to Y/B$ is a homeomorphism. \\

Let us verify this for the case $\mK=\partial\Delta^2$. Pick $I=\{1\}$. Let $i_c:\abs{\mK}\to I^n$ be the map that is the cubical embedding. Now the relative version of the identity $$(D^1,S^0)^\sigma=\bigcup_{\substack{\mu\subseteq\tau\subseteq\text{Vert}(\mK)\\\tau\setminus\mu=\sigma}}C_{\mu\subseteq\tau}$$ says that if $\sigma\in\mK_I$, then $$(D^1,S^0)^\sigma=\bigcup_{\substack{\mu\subseteq\tau\subseteq\text{Vert}(\mK)\\\tau\setminus\mu=\sigma}}C_{\mu\subseteq\tau}^I$$ where we consider $(D^1,S^0)^\sigma$ as a subspace of $\mZ_\mK$ by considering $\mZ_{\mK_I}$ as a subspace of $\mZ_\mK$, and on the right $C_{\mu\subseteq\tau}^I$ is the usual meaning but restricted to the cube $(D^1)^I$ of $I^3$. \\

Restricting this map to $\mK_I$, this the map sending $\{1\}$ to $C_{\{1\},\{1\}}^I$ which is the point $-1\in C_{\{1\},\{1\}}^I=D^1$, but is the point $(-1,1,1)$ in $I^3$. \\

This induces a relative homeomorphism $$\coprod_{J\subseteq\{1,\dots,n\},\abs{J}=i}(\abs{\text{Cone}(\text{Bd}(\mK_J))},\abs{\text{Bd}(\mK_J)})\to(\R\mZ_\mK^i,\R\mZ_\mK^{i-1})$$ Let $\varphi_{\mK_J}$ denote the map $\abs{\text{Bd}(\mK_J)}\to\R\mZ_{\mK_J}^{i-1}$. Let $j_{\mK_I}$ denote the inclusion $\R\mZ_{\mK_I}^{i-1}\hookrightarrow\R\mZ_\mK^{i-1}$. 

\begin{prp}{}{}\\
Let $\mK$ be a simplicial complex on $\{1,\dots,n\}$. Then there is a pushout \\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	{\coprod_{I\subseteq\{1,\dots,n\},\abs{I}=i}\abs{\text{Bd}(\mK_I)}} &&& {\R\mZ_\mK^{i-1}} \\
	{\coprod_{I\subseteq\{1,\dots,n\},\abs{I}=i}\abs{\text{Cone}(\text{Bd}(\mK_I))}} &&& {\R\mZ_\mK^i}
	\arrow["{\coprod_{I\subseteq\{1,\dots,n\},\abs{I}=i}(j_{\mK_I}\circ\varphi_{\mK_I})}", from=1-1, to=1-4]
	\arrow[hook, from=1-1, to=2-1]
	\arrow[from=1-4, to=2-4]
	\arrow[from=2-1, to=2-4]
\end{tikzcd}} \\~\\
for $1\leq i\leq n$. 
\end{prp}

\begin{lmm}{}{}\\
Let $\mK$ be a simplicial complex on $\{1,\dots,n\}$. Let $\bold{X}=(X_1,\dots,X_n)$ be spaces. Then $(C\bold{X}_I,\bold{X}_I)^{\mK_I}$ is a retract of $(C\bold{X},\bold{X})^\mK$. 
\end{lmm}

\begin{prp}{}{}\\
Let $\mK$ be a simplicial complex on $\{1,\dots,n\}$. Let $\bold{X}=(X_1,\dots,X_n)$ be spaces. Then we have $$(C\bold{X},\bold{X})^{\mK,i}=\bigcup_{I\subseteq\{1,\dots,n\},\abs{I}=i}\mZ_{\mK_I}(C\bold{X}_I,\bold{X}_I)$$
\end{prp}

\begin{prp}{}{}\\
Let $\mK$ be a simplicial complex on $\{1,\dots,n\}$. Let $\bold{X}=(X_1,\dots,X_n)$ be spaces. Then there is a relative homeomorphism $$(\abs{\text{Cone}(\text{Bd}(\mK_I))},\abs{\text{Bd}(\mK_I)})\times(X^{\times I},T^{i-1}(\bold{X}_I))\to((C\bold{X}_I,\bold{X}_I)^{\mK_I},(C\bold{X}_I,\bold{X}_I)^{\mK_I,i-1})$$
\end{prp}

\subsection{Trivial Fat Wedge Filtrations}
\begin{defn}{Trivial Real Fat Wedge Filtration}{}\\
Let $\mK$ be a simplicial complex on $\{1,\dots,n\}$. Let $\bold{X}=(X_1,\dots,X_n)$ be spaces. We say that the fat wedge filtration of $\R\mZ_\mK$ is trivial if $\varphi_{\mK_I}$ is trivial for all $I\subseteq\{1,\dots,n\}$. 
\end{defn}

\begin{prp}{}{}\\
Let $\mK$ be a simplicial complex on $\{1,\dots,n\}$. If $\dim(\mK)\geq n-2$, then the fat wedge filtration of $\R\mZ_\mK$ is trivial. 
\end{prp}

\begin{thm}{}{}\\
Let $\mK$ be a simplicial complex on $\{1,\dots,n\}$. If the fat wedge filtration of $\R\mZ_\mK$ is trivial, then the BBCG decomposition desuspends and we have a homotopy equivalence $$(C\bold{X},\bold{X})^\mK\simeq\bigvee_{\emptyset\neq I\subseteq\{1,\dots,n\}}\abs{\Sigma\mK_I}\wedge\left(\bigwedge_{i\in I}X_i\right)$$
\end{thm}

Provable: the fat wedge filtration of $\mZ_\mK$ is a cone decomposition. Hence trivialness can also be defined. 

\begin{crl}{}{}\\
Let $\mK$ be a simplicial complex on $\{1,\dots,n\}$. If the fat wedge filtration of $\R\mZ_\mK$ is trivial, then the fat wedge filtration of $\mZ_\mK$ is trivial. 
\end{crl}

\begin{thm}{}{}\\
Let $\mK$ be a simplicial complex on $\{1,\dots,n\}$. Then the following are equivalent. 
\begin{itemize}
\item The fat wedge filtration of $\mZ_\mK$ is trivial. 
\item $\mZ_\mK$ is a co-H-space. 
\item The BBCG decomopsition desuspends to give $$\mZ_\mK\simeq\bigvee_{\emptyset\neq I\subseteq\{1,\dots,n\}}\Sigma^{\abs{I}+1}\abs{\mK_I}$$
\end{itemize}
\end{thm}

\begin{thm}{}{}\\
Let $\mK$ be a simplicial complex on $\{1,\dots,n\}$. If $\mK^\vee$ is sequentially Cohen Macaulay, then the fat wedge filtration of $\R\mZ_\mK$ is trivial. 
\end{thm}

\begin{thm}{}{}\\
Let $\mK$ be a simplicial complex on $\{1,\dots,n\}$. If $\mK$ is $\ceil{\frac{\dim(\mK)}{2}}$-neighbourly, then the fat wedge filtration of $\R\mZ_\mK$ is trivial. 
\end{thm}

\section{Quasitoric Manifolds}
\subsection{The Davis-Januszkiewicz Classification Theorem}
\begin{defn}{Quasitoric Manifolds}{}\\
Let $M$ be a smooth manifold of dimension $2n$. Let $T^n$ be a torus acting smoothly on $M$. We say that $M$ is a quasitoric manifold if the following are true. 
\begin{itemize}
\item For each $p\in M$, there is a $T^n$-equivariant chart $(U,\varphi:U\to\C^n)$ where $\C^n$ is equipped with the standard torus action. 
\item There exists a convex polytope $P$ such that there is a homeomorphism $$M/T^n\cong P$$
\end{itemize}
\end{defn}

Toric symplectic manifolds are quasitoric manifolds. Moment angle manifolds are NOT quasitoric manifolds. 

\begin{defn}{Orbit Type Strata}{}\\
Let $M$ be a quasitoric manifold. Let $P$ be the underlying convex polytope. Let $H\leq T^n$ be a subgroup. Define the orbit type strata of $H$ to be $$M_{(H)}=\{x\in M\;|\;\text{Stab}_{T^n}(x)=H\}$$
\end{defn}

\begin{prp}{}{}\\
Let $M$ be a quasitoric manifold. Let $P$ be the underlying convex polytope. Then there is a bijection $$\{F\in\mF(P)\;|\;F\text{ has codimension }k\}\;\;\overset{1:1}{\leftrightarrow}\;\;\left\{N\subseteq M\;\bigg{|}\;\substack{N\text{ is an orbit type strata of subtori of }\\T^n\text{ of dimension }k}\right\}$$ given as follows. Let $\pi:M\to P$ be the quotient map. Then we have $$F_i\mapsto\pi^{-1}(\text{relint}(F))$$
\end{prp}

\begin{defn}{Primitive Vectors}{}\\
Let $v=(v_1,\dots,v_n)\in\Z^n$. We say that $v$ is primitive if $v\neq\lambda u$ for any $\lambda\in\Z\setminus\{0\}$ and $u\in\Z^n$. 
\end{defn}

\begin{defn}{The Characteristic Matrix}{}\\
Let $M$ be a quasitoric manifold. Let $P$ be the underlying convex polytope. Define the characteristic matrix $\Lambda$ as follows. For each facet $F_i$ of $P$ for $1\leq i\leq m$, the facet corresponds to a a circle $S_i^1\subseteq T^n$ by the above. Every circle $S_i^1\subseteq T^n$ corresponds to a unique primitive vector $v_i=(v_{1i},\dots,v_{1n})\in\Z^n$ up to sign. Then $$\Lambda=\begin{pmatrix}
v_{11} & \cdots & v_{1m}\\
\vdots & \ddots & \vdots\\
v_{n1} & \cdots & v_{nm}
\end{pmatrix}\in M_{n\times m}(\Z)$$
\end{defn}

\begin{defn}{Characteristic Pairs}{}\\
Let $P$ be a simple polytope of dimension $n$. Let $F_1,\dots,F_m$ be the facets of $P$. Let $\Lambda\in M_{n\times m}(\Z)$ be a matrix. Let $c_1,\dots,c_m$ be the columns of $\Lambda$. We say that $(P,\Lambda)$ is a characteristic pair if the following are true. 
\begin{itemize}
\item $c_i$ is a primitive vector for all $1\leq i\leq m$. 
\item If $F_{i_1}\cap\cdots\cap F_{i_n}$ is a vertex of $P$, then $$\det\begin{pmatrix}

\begin{defn}{$I$-Torsion Modules}{}\\
Let $R$ be a Noetherian commutative ring. Let $M$ be an $R$-module. Let $I$ be an ideal of $R$. We say that $M$ is $I$-torsion if $M=\Gamma_I(M)$. 
\end{defn}

\begin{defn}{The Torsion Functor}{}\\
Let $R$ be a Noetherian commutative ring. Let $I$ be an ideal of $R$. Define the $I$-torsion functor $$\Gamma_I:{_R}\bold{Mod}\to{_R}\bold{Mod}$$ by the following. 
\begin{itemize}
\item For each $R$-module $M$, the functor sends $M$ to the $I$-torsion $\Gamma_I(M)$. 
\item For each $R$-module homomorphism $f:M\to N$, the functor sends $f$ to $\Gamma_I(f)=f|_{\Gamma_I(M)}$. 
\end{itemize}
\end{defn}

\begin{lmm}{}{}\\
Let $R$ be a Noetherian commutative ring. Let $I$ be an ideal of $R$. Then $\Gamma_I$ is a left exact functor. 
\end{lmm}

\begin{defn}{Local Cohomology}{}\\
Let $R$ be a Noetherian commutative ring. Let $M$ be an $R$-module. Let $I$ be an ideal of $R$. Define the $n$th local cohomology of $M$ with support in $I$ to be $$H_I^n(M)=(R^n\Gamma_I)(M)$$
\end{defn}

\begin{prp}{}{}\\
Let $R$ be a Noetherian commutative ring. Let $M$ be an $R$-module. Let $I,J$ be ideals of $R$. If $\sqrt{I}=\sqrt{J}$, then the equality $\Gamma_I(M)=\Gamma_J(M)$ induces an equality $$H_I^n(M)=H_J^n(M)$$ for all $n\in\N$. 
\end{prp}

Thus we only need to consider the local cohomology with support in radical ideals.

Let $R$ be a Noetherian commutative ring. Let $M$ be an $R$-module. Let $I$ be an ideal of $R$. Then $H_I^n(M)$ is $I$-torsion for all $n\in\N$. 
\end{lmm}

\begin{defn}{Progenerators}{}\\
Let $R$ be a ring. Let $P$ be an $R$-module. We say that $P$ is a progenerator of $R$ if the following are true. 
\begin{itemize}
\item $P$ is finitely generated. 
\item $P$ is projective. 
\end{itemize}
\end{defn}

\begin{prp}{}{}\\
Let $R,S$ be rings. Let $F:{_R}\bold{Mod}\to{_S}\bold{Mod}$ and $G:{_S}\bold{Mod}\to{_R}\bold{Mod}$ be additive functors. Then $F$ and $G$ are inverses if and only if there exists an $(S,R)$-bimodule $P$ such that the following are true. 
\begin{itemize}
\item ${_S}P$ is a progenerator. 
\item $P_R$ is a progenerator. 
\item There are natural isomorphisms $F\cong P\otimes_R-$ and $G\cong\Hom({_S}P,-)$. 
\end{itemize}
\end{prp}

\begin{thm}{Morita}{}\\
Let $R,S$ be rings. Then the following are equivalent. 
\begin{itemize}
\item $R$ and $S$ are Morita equivalent. 
\item There exists a progenerator $R$-module $P_R$ such that $S\cong\text{End}_R(P_R)$. 
\item There exist an $(R,S)$-bimodule $M$ and a $(S,R)$-bimodule $N$ such that $M\otimes_SN\cong R$ as $(R,R)$-bimodules and $N\otimes_RM\cong S$ as $(S,S)$-bimodules. 
\end{itemize}
\end{thm}

\section{The Nature of Mathematical Physics}
In mathematical physics, physical observations are given a mathematical formalism. A recurring theme is that mathematics often predict new phenomena beyond the original observation. 

\subsection*{Mechanics}
Mechanics is the part of physics that study the motion of bodies. Statics is the subject that studies equilibriums. Kinematics is concerned with general properties of motions regardless of the source. Dynamics study motion caused by forces or interactions. In regards to the system that is studied, mechanics can also be divided into material points, rigid bodies, elastic bodies and fluids. \\

Mechanics have received a number of revolutionary ideas that transformed the subject. Completely. In the modern day there are three main branches of mechanics: classical mechanics, relativistic mechanics and quantum mechanics. 

\subsubsection*{Classical Mechanics}
Theoretically, there are three different formalisms of classical mechanics that are related: Newtonian, Lagrangian and Hamiltonian. Newtonian mechanics is the easiest to formulate. Lagrangian is based on the least action principle, which is more sophisticated but is efficient for mechanical systems that are under constraints. Hamiltonian mechanics connects the bridge between classical mechanics and quantum mechanics. 

For a mathematical formalism of classical mechanics, we look for a mathematical model that describes the following physically observed phenomena. 
\begin{itemize}
\item Our space is three dimensional and time is one dimensional. 
\item There exists coordinate systems called inertial such that all laws of nature at all moments of time are the same in all inertial coordinates, and all coordinate systems in uniform rectilinear motion with respect to an inertial one are themselves inertial. 
\item The initial state of a mechanical system uniquely determines all of its motion. 
\end{itemize}

The limitations of classical mechanics come from Maxwell's equations, which says that electromagnetic waves travel at a fixed speed regardless of which inertial frame one uses. In particular, classical mechanics is a good enough estimate for motions of bodies that are significantly less than the speed of light, and whose bodies are macroscopic objects. 

\subsubsection*{Relativistic Mechanics}
For a mathematical formalism of relativistic mechanics, we look for a mathematical model that describes the following physically observed phenomena. 
\begin{itemize}
\item The Principle of Relativity: The laws of physics are the same in all inertial frames. 
\item The speed of light is the same for all inertial frames, regardless of the motion of the source or observer. 
\end{itemize}
A number of consequences appear because the speed of light being constant. 
\begin{itemize}
\item Time dilation: Moving clocks run slower. 
\item Length contraction: Moving objects are shortened along the direction of motion. 
\item Relativity of simultaneity: Two events simultaneous in one frame are not simultaneous in another. 
\item Speed do not add linearly, nothing with mass can have speed that is the speed of light. 
\item Mass energy equivalence: $E=mc^2$. 
\item Instantaneous interactions are impossible, and so must propagate at finite speed. 
\end{itemize}

Special relativity introduces the framework that is Lorentz geometry for the universe. This framework is one that is compatible with the observation that is the speed of light is the same for all inertial frames. In particular, it deduces the relativity of simultaneity, which means that instantaneous interactions are impossible. This introduces the concept of fields and field theory. \\

The limitations of relativistic mechanics come from general relativity, which says that gravity should be a result of the geometry, not a force. Relativistic mechanics does not take this to account. Also, classical field theory cannot handle the destruction and creation of particles.

	\arrow[from=1-4, to=1-5]

\begin{defn}{The Change of Rings Homomorphism}{}\\
Let $R,S$ be rings. Let $f:R\to S$ be a ring homomrphism. Define the change of rings homomorphism to be the map $K_0(f):K_0(R)\to K_0(S)$ given by $$[P]\mapsto[S\otimes_RP]$$
\end{defn}

Recall the Grothendieck completion of an abelian semi-group $G(-)$. 

\begin{prp}{}{}\\
Let $R$ be a ring. Let $P(R)=\{P\;|\;P\text{ is a finitely generated projective }R\text{-modules }\}$. Then there is a natural isomorphism $$K_0(R)\cong G(P(R),\oplus)$$
\end{prp}

\begin{prp}{Morita Equivalence}{}\\
Let $R$ be a ring. Then there is a natural isomorphism $$K_0(R)\cong K_0(M_n(R))$$ given by $[P]\mapsto R^n\otimes_RP$ where $R^n$ is considered as an $(M_n(R),R)$-bimodule.

\begin{prp}{}{}\\

\end{prp}

\section{Ergodic Theory}
\subsection{}
\begin{defn}{Measure Preserving Systems}{}\\
Let $(X,\mF,\mu)$ be a measure space. Let $T:X\to X$ be a measurable function. We say that $T$ is a measure preserving function if $$\mu(T^{-1}(E))=\mu(E)$$ for all $E\in\mF$. In this case, we call $(X,\mF,\mu,T)$ a measure preserving system. 
\end{defn}

\begin{defn}{Probability Preserving System}{}\\
Let $(X,\mF,\mu,T)$ be a measure preserving system. We say that it is a probability preserving system if $(X,\mF,\mu)$ is a probability space. 
\end{defn}

\begin{thm}{Poincare Recurrence Theorem}{}\\
Let $(X,\mF,\mu,T)$ be a probability preserving system. Let $E\in\mF$. Then we have $$\mu(\{x\in E\;|\;\exists N\text{ such that }T^n(x)\notin E\text{ for all }n>N\})=0$$
\end{thm}

In other words, most points in $E$ go back to $E$ infinitely often after applying a certain number of $T$'s. 

\begin{defn}{Measure Theoretically Isomorphic}{}\\
Let $(X,\mF,\mu,T)$ and $(Y,\mE,\sigma,S)$ be measure preserving systems. Let $\pi:X\to Y$ be a measurable function. We say that $\pi$ is a measure theoretically isomorphism if the following are true. 
\begin{itemize}
\item There exists $A\subseteq X$ and $B\subseteq Y$ such that $\mu(X\setminus A)=\mu(Y\setminus B)=0$ and $\pi|_A:A\to B$ is invertible with a measurable inverse. 
\item For all $E\in\mE$, we have $\mu(\pi^{-1}(E))=\sigma(E)$. 
\item $S\circ\pi=\pi\circ T$. 
\end{itemize}
\end{defn}

\subsection{}
\begin{defn}{Invariant Sets}{}\\
Let $(X,\mF,\mu,T)$ be a measure preserving system. Let $E\in\mF$. We say that $E$ is invariant if $$T^{-1}(E)=E$$
\end{defn}

\begin{defn}{Ergodic Systems}{}\\
Let $(X,\mF,\mu,T)$ be a measure preserving system. We say that the system is ergodic if for all invariant sets $E$, we have $\mu(E)=0$ or $\mu(X\setminus E)=0$. In this case, we say that $\mu$ is an ergodic measure. 
\end{defn}

\section{The Logical Chain of Classical Field Theory and Gauge Theory}

\subsection*{Step 1: The Field}

We observe a physical quantity distributed over spacetime — a temperature, a velocity, an electromagnetic strength. Mathematically this is a fiber bundle $\pi: E \to M$, where $M$ is spacetime and the fiber $E_x$ is the space of possible values of the quantity at $x$. A \textbf{section} $\phi \in \Gamma(E)$ is one complete specification of the quantity at every point — a possible physical configuration.

\subsection*{Step 2: The Lagrangian as a Choice of Physics}

A Lagrangian density $\Lambda \in \Omega^n(J^kE)$ is a choice of action functional
\[
S[\phi] = \int_\Omega (j^k\phi)^*\Lambda
\]
Different Lagrangians encode different physical theories. The Euler-Lagrange equations — the equations of motion — are derived from $\Lambda$ by the variational principle. The physical sections are the critical points of $S$.

\subsection*{Step 3: Observing a Symmetry}

You compute the Euler-Lagrange equations and notice that the Lagrangian $\Lambda$ is invariant under some subgroup of automorphisms of $E$. An \textbf{automorphism} of $\pi: E \to M$ is a fiber-preserving diffeomorphism $\Phi: E \to E$. Such an automorphism acts on sections by
\[
\phi \mapsto \Phi \circ \phi
\]
and the Lagrangian being invariant means
\[
S[\Phi \circ \phi] = S[\phi] \quad \text{for all } \phi \in \Gamma(E)
\]
At this stage, the symmetry group acts the same way at every point — it is \textbf{global}. You observe this as an empirical fact about your Lagrangian. 

Symmetries of the action induces certain laws of physics, such as conservation of energy, momentum, charge etc. This is Noether's theorem. 

Another note: Rather than starting with a Lagrangian and observe the symmetries of its action, we may also start with a set of symmetries that we would like to enforce, apply it to all differential forms and retain only the invariants, and see what Lagrangian you would get. 

\subsection*{Step 4: The Absolute vs Relative Observation}
The global symmetry exists because the field takes values in a space that has \textbf{no preferred origin or frame} — the fiber $E_x$ has a group of self-maps (rotations, phase shifts, color rotations) that are physically unobservable. Different observers, or different conventions, would describe the same physical situation using sections related by this group action. The global symmetry is an artifact of having made consistent conventional choices everywhere at once. We need not restrict ourselves to consider symmetries of the entire space, and this is supported by physics: 

\subsubsection*{Argument 1: Special Relativity}

Special relativity says there is no notion of simultaneity — two spacelike separated points have no canonical notion of ``the same moment.'' A global transformation is therefore not a physically coherent operation in a relativistic theory. If you believe in special relativity, you are forced to allow your symmetry to act differently at different points. The global symmetry is then a special case of the local one, not the fundamental object.

\subsubsection*{Argument 2: The Gauge Principle}

The difference induced by the symmetry of the action is not observable. The phase is therefore a \textbf{convention} --- a choice made by an observer at $x$ to  describe the field. If the phase is a convention, then different observers at different points of spacetime should be free to make different conventional choices independently. There is no physical mechanism that would force their conventions to agree globally. The true symmetry group of the theory is therefore not the global group $G$ but the local group $mathcal{G} = \mathrm{Map}(M, G)$ of all smooth $G$-valued functions on spacetime.

This is the \textbf{gauge principle}: any redundancy in the description of a physical system should be a local redundancy. It is a postulate, not a theorem. Its justification is that it works --- every time it is applied, it produces a physically correct and predictive theory.

\subsection*{Step 5: Promoting to Local Symmetry}

This forces you to demand invariance not just under global $G$-transformations
\[
\phi(x) \mapsto g \cdot \phi(x), \quad g \in G \text{ constant}
\]
but under local $G$-transformations
\[
\phi(x) \mapsto g(x) \cdot \phi(x), \quad g: M \to G \text{ smooth}
\]

The Lagrangian, built from ordinary derivatives $\partial_\mu\phi$, immediately fails to be 
locally invariant. The reason is fundamental: $\partial_\mu\phi$ compares field values at 
nearby points $x$ and $x + dx$, which under local symmetry live in fibers with 
\textbf{independent conventions}. Subtracting them is meaningless — it is comparing 
quantities measured in different units.

\subsection*{Step 6: The Mathematics Is Now Forced}

To make sense of differentiation under local symmetry, we need a way to 
\textbf{compare vectors in nearby fibers} compatibly with the $G$-structure. But this is 
precisely the definition of a \textbf{connection}.
Where does this connection live? The answer is that the principal bundle is not introduced 
from outside — it is \textbf{already contained inside the vector bundle $E$}.

The fibers $E_x$ are vector spaces, but the isomorphism $E_x \cong \mathbb{R}^r$ is not 
canonical — it requires a choice of basis. A \textbf{frame} at $x$ is an ordered basis for 
$E_x$. The set of all frames at $x$ carries a natural free transitive right action of 
$GL(r, \mathbb{R})$, making it a $GL(r,\mathbb{R})$-torsor. Taking the disjoint union over 
all $x \in M$ produces the \textbf{frame bundle}
\[
P := \bigsqcup_{x \in M} \{\text{ordered bases for } E_x\}
\]
a principal $GL(r, \mathbb{R})$-bundle canonically constructed from $E$.

Now, the observation of a global $G$-symmetry in Step 3 is precisely the observation that 
the physically relevant frames are not arbitrary frames, but only those related by 
$G$-transformations. The structure group \textbf{reduces} from $GL(r, \mathbb{R})$ to $G$, 
producing a principal $G$-bundle
\[
P_G \subset P
\]
canonically determined by the symmetry. The original vector bundle is recovered as the 
associated bundle $E = P_G \times_G V$.

The connection demanded by local symmetry lives on $P_G$, because $P_G$ is the bundle whose 
fibers are $G$-torsors — they encode exactly the gauge freedom at each point. A connection 
on $P_G$ transports $G$-frames between nearby fibers, and therefore transports everything 
associated to those frames, including sections of $E = P_G \times_G V$.

\subsection*{Step 7: What Follows From The Maths}

Since the principal bundle is just a collection of frames at each fiber. A local section of the principal bundle is a choice of local coordinates. The mathematical fact that non-trivial principal bundles have no global sections can be interpreted as the fact that there is generally no globally consistent choice of coordinates because the symmetries now act locally. 

A connection in the principal bundle gives a covariant derivative in the associated bundle. This gives a way to compare phases across different time space.

\section{Bordism and Cobordism}
\subsection{Classical Bordism}
\begin{defn}{Bordant Manifolds}{}\\
Let $M,N$ be smooth compact manifolds without boundary. A bordism between $M$ and $N$ is a smooth compact manifold $X$ such that $\partial X=M\amalg N$. In this case, we say that $M$ and $N$ are bordant. 
\end{defn}

\begin{prp}{}{}\\
Let $M,N$ be smooth compact manifolds without boundary. If $M$ and $N$ are diffeomorphic, then $M$ and $N$ are bordant. 
\end{prp}

\begin{defn}{The Bordism Groups}{}\\
Define the $n$th bordism group to be the set $$\Omega_n=\{[M]\;|\;[M]\text{ is a bordism class }\}$$ together with addition given by $[M]+[N]=[M\amalg N]$. 
\end{defn}

\begin{prp}{}{}\\
The following are true regarding the bordism groups. 
\begin{itemize}
\item $\Omega_0=\langle[\ast]\rangle\cong\Z/2\Z$. 
\item $\Omega_1=0$. 
\item $\Omega_2=\langle[\R\Prj^2]\rangle\cong\Z/2\Z$. 
\end{itemize}
\end{prp}

\begin{defn}{The Bordism Ring}{}\\
Define the bordism ring to be the abelian group $$\Omega=\bigoplus_{i=0}^\infty\Omega_n$$ together with multiplication given by $[M]\cdot[N]=[M\times N]$. 
\end{defn}

\begin{prp}{}{}\\
There is an isomorphism of commutative graded rings $$\Omega\cong\Z/2\Z[x_i\;|\;i\neq 2^k-1\text{ for some }k]$$
\end{prp}

\begin{prp}{}{}\\
Let $M,N$ be a smooth compact manifolds without boundary of dimension $n$. Then $M$ and $N$ are bordant if and only if $$\langle w_{i_1}(M)\cdots w_{i_k}(M),[M]\rangle=\langle w_{i_1}(N)\cdots w_{i_k}(N),[N]\rangle$$ for all partitions $(i_1,\dots,i_k)$ of $n$. 
\end{prp}

\subsection{Oriented Bordism}
\begin{defn}{Oriented Bordant Manifolds}{}\\
Let $M,N$ be smooth oriented compact manifolds without boundary. An oriented bordism between $M$ and $N$ is a smooth oriented compact manifold $X$ such that $\partial X=M\amalg\overline{N}$ where $\overline{N}$ has the inverse orientation of $M$. 
\end{defn}

\begin{defn}{The Oriented Bordism Groups}{}\\
Define the $n$th bordism group to be the set $$\Omega_n^{SO}=\{[M]\;|\;[M]\text{ is an oriented bordism class }\}$$ together with addition given by $[M]+[N]=[M\amalg N]$. 
\end{defn}

\begin{lmm}{}{}\\
Let $M$ be a smooth oriented compact manifold without boundary. Then we have $-[M]=[\overline{M}]$. 
\end{lmm}

\begin{defn}{The Oriented Bordism Ring}{}\\
Define the bordism ring to be the abelian group $$\Omega^{SO}=\bigoplus_{i=0}^\infty\Omega_n^{SO}$$ together with multiplication given by $[M]\cdot[N]=[M\times N]$. 
\end{defn}

\begin{prp}{}{}\\
There is an isomorphism of commutative graded rings $$\Omega^{SO}\otimes\Q\cong\Q[x_{4k}\;|\;k\in\N]$$ given by $[\C\Prj^{2k}]\mapsto x_{4k}$. 
\end{prp}

\begin{prp}{}{}\\
Let $t\in(\Omega^{SO})^\text{tor}$. Then we have $2t=0$. 
\end{prp}

\begin{prp}{}{}\\
There is an isomorphism of commutative graded rings $$\frac{\Omega^{SO}}{(\Omega^{SO})^\text{tor}}\cong\Z[x_{4k}\;|\;k\in\N]$$
\end{prp}

\begin{prp}{}{}\\
Let $M,N$ be a smooth oriented compact manifolds without boundary of dimension $n$. Then $M$ and $N$ are oriented bordant if and only if $$\langle w_{i_1}(M)\cdots w_{i_k}(M),[M]\rangle=\langle w_{i_1}(N)\cdots w_{i_k}(N),[N]\rangle$$ for all partitions $(i_1,\dots,i_k)$ of $n$ and $$\langle p_{j_1}(M)\cdots p_{j_k}(M),[M]\rangle=\langle p_{j_1}(N)\cdots p_{j_k}(N),[N]\rangle$$ for all partitions $(j_1,\dots,j_k)$ of $n/4$
\end{prp}

\begin{prp}{}{}\\
The following are true regarding the oriented bordism groups. 
\begin{itemize}
\item $\Omega_0^{SO}=\langle\ast\rangle\cong\Z$. 
\item $\Omega_1^{SO}=\Omega_2^{SO}=\Omega_3^{SO}=0$. 
\item $\Omega_4^{SO}=\langle[\C\Prj^2]\rangle\cong\Z$. 
\item $\Omega_5^{SO}=\langle[Y]\rangle\cong\Z/2\Z$ where $Y$ is the Dold manifold. 
\item $\Omega_6^{SO}=\Omega_7^{SO}=0$. 
\item $\Omega_8^{SO}=\langle[\C\Prj^2\times\C\Prj^2],[\C\Prj^4]\rangle\cong\Z^2$. 
\item $\Omega_n^{SO}\neq 0$ for all $n\geq 9$. 
\end{itemize}
\end{prp}

\section{Sites, Sieves and Grothendieck Topologies}
\subsection{Sites on a Category}
We now want to generalize the notion of sheaves to general categories, not just spaces. However, there is no notion of open sets in an arbitrary category. Grothendieck realized that the essence of sheaves comes from the data of the open covers $U=\bigcup_{i\in I}U_i$ for each open set $U$ in a space $X$. 

\begin{defn}{Covers}{} Let $\mC$ be a category. Let $U\in\mC$ be an object. A cover of $U$ is a collection of morphisms $$\{U_i\to U\;|\;i\in I\}$$
\end{defn}

\begin{defn}{Coverages (nLab)}{} Let $\mC$ be a category. A coverage $\text{Cov}(\mC)$ of $\mC$ consists a collection of covers for each $U\in\mC$, such that the following are true. 
\begin{itemize}
\item If $\{f_i:U_i\to U\;|\;i\in I\}$ is a cover of $U$ and $g:V\to U$ is a morphism in $\mC$, then there exists a cover $\{h_j:V_j\to V\}$ of $V$ such that every $g\circ h_j:V_j\to U$ factors through the cover of $U$. Or in other words, for each $h_j:V_j\to V$, there exists $U_i$ and $f_i:U_i\to U$ such that the following diagram commutes: \\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	{V_j} & {U_i} \\
	V & U
	\arrow["k", dashed, from=1-1, to=1-2]
	\arrow["{h_j}"', from=1-1, to=2-1]
	\arrow["{f_i}", from=1-2, to=2-2]
	\arrow["g"', from=2-1, to=2-2]
\end{tikzcd}}\\
\end{itemize}
\end{defn}

\begin{defn}{Coverages (Stacks)}{} Let $\mC$ be a category. A coverage $\text{Cov}(\mC)$ of $\mC$ consists a collection of covers for each $U\in\mC$, such that the following are true. 
\begin{itemize}
\item If $V\to U$ is an isomorphism then $\{V\to U\}\in\text{Cov}(\mC)$
\item If $\{U_i\to U\;|\;i\in I\}\in\text{Cov}(\mC)$ and $\{V_{i,j}\to U_i\;|\;j\in J_i\}\in\text{Cov}(\mC)$ for each $i\in I$, then $\{V_{i,j}\to U_i\to U\;|\;i,j\in I\}\in\text{Cov}(\mC)$
\item If $\{f_i:U_i\to U\;|\;i\in I\}\in\text{Cov}(\mC)$ and $g:V\to U$ is a morphism, then the pullback \\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	{U_i\times_UV} & {U_i} \\
	V & U
	\arrow["\text{proj.}", dashed, from=1-1, to=1-2]
	\arrow["{\text{proj.}}"', from=1-1, to=2-1]
	\arrow["{g}", from=1-2, to=2-2]
	\arrow["f_i"', from=2-1, to=2-2]
\end{tikzcd}}\\
exists for all $i\in I$ and $\{U_i\times_UV\to U\}\in\text{Cov}(\mC)$. 
\end{itemize}
\end{defn}

\begin{defn}{Sites (nLab)}{} A site is a category $\mC$ together with a coverage $\text{Cov}(\mC)$. 
\end{defn}

\begin{defn}{Continuous Functors}{} Let $\mC$ and $\mD$ be two sites. Let $F:\mC\to\mD$ be a functor. We say that $F$ is continuous if the following are true. 
\begin{itemize}
\item $F$ preserves fiber products
\item $F$ sends covers to covers
\end{itemize}
\end{defn}

\subsection{Basic Definition of Presheaves}
As with how we equipped to each variety $V$ its coordinate ring $k[V]$ which are functions on $V$, we want to equip to each spectrum some ring which are functions on them. It will not make sense that we can define functions on spectrums immediately. \\

In its full generality, sheaves are designed to encode all local information into one compact notation. It often has uses in complex geometry for its differential forms and smooth functions. In algebraic geometry it is fundamental for redefining the notion of a variety. 

\begin{defn}{The Category of Open Sets}{} Let $(X,\mT)$ be a topological space. Define the category $\bold{Open}(X)$ to consist of the following data. 
\begin{itemize}
\item The objects $\text{Ob}(\bold{Open}(X))=\mT$ are the open sets of $X$
\item For $U,V\subseteq X$ two open sets of $X$, the morphisms $$\Hom_{\bold{Open}(X)}(U,V)=\begin{cases}
\{\iota:U\hookrightarrow V\text{ the inclusion map}\} & \text{ if } U\subseteq V\\
\emptyset & \text{otherwise}
\end{cases}$$
\item Composition is given as the composition of functions. 
\end{itemize}
\end{defn}

Recall from Category Theory 1 that a presheaf on $\mC$ is just a contravariant functor $\mC\to\bold{Set}$. Using the category of open sets we can now define presheaves on spaces. 

\begin{defn}{Presheaves on a Space}{}\\
Let $X$ be a space. Let $\mC$ be a category. A $\mC$-presheaf on $X$ is a contravariant functor $$\mF:\bold{Open}(X)\to\mC$$
\end{defn}

As per the definition of a functor, for each inclusion map $\iota:V\to U$ of open sets in $X$, passing through a presheaf $\mF$ gives a unique morphism $\mF(\iota):\mF(U)\to\mF(V)$. We often denote $\mF(\iota)$ as $\text{res}_V^U$ or $(-)|_V$. 

\begin{eg}{The Constant Presheaf}{}\\
Let $X$ be a space. Let $S$ be a set. Define the constant presheaf with value $S$ to be the presheaf $\underline{S}_\text{pre}$ defined as follows. 
\begin{itemize}
\item For each open set $U\subseteq X$, we have $$\underline{S}_\text{pre}(U)=S$$
\item For each inclusion $V\subseteq U$ of open sets, define $$\text{res}_V^U=\text{id}_S$$
\end{itemize}
\end{eg}

\begin{defn}{The Stalk of a Presheaf}{} Let $\mF$ be a presheaf on a topological space $(X,\mathcal{T})$. Let $p\in X$. Consider the full subcategory $\mJ_p$ of $\bold{Open}(X)$ consisting of open sets in $X$ that contain $p$. Define the stalk of $\mF$ at $p$ to be the colimit $$\mF_p=\underset{\mJ_p}{\text{colim}}\mF(V_p)$$ 
\end{defn}

\begin{defn}{Morphism of Presheaves}{}\\
Let $X$ be a topological space. Let $\mC$ be a category. Let $\mF,\mG$ be $\mC$-presheaves on $X$. A morphism $\varphi:\mF\to\mG$ is a natural transformation of functors. 
\end{defn}

Notice that the natural transformation $\phi$ here takes every open set $U$ and maps it to a group homomorphism $\phi(U):\mF(U)\to\mG(U)$ if $\mF$ and $\mG$ are presheaves with values in $\bold{Grp}$. 

\begin{defn}{Induced Map of Stalks}{}\\
Let $X$ be a topological space. Let $\mC$ be a category. Let $\mF,\mG$ be $\mC$-presheaves on $X$. Let $p\in X$. Define the induced map of stalks $$\phi_p:\mF_p\to\mG_p$$ to be the map induced by the universal product of colimits. 
\end{defn}

\begin{defn}{Isomorphism of Presheaves}{}\\
Let $X$ be a topological space. Let $X$ be a topological space. Let $\mC$ be a category. Let $\mF,\mG$ be $\mC$-presheaves on $X$. Let $\varphi:\mF\to\mG$ be a morphism of presheaves. We say that $\varphi$ is an isomorphism if $\varphi$ is a natural isomorphism. In this case, we say that $\mF$ and $\mG$ are isomorphic. 
\end{defn}

\subsection{Sheaves with Values in a Set}
\begin{defn}{Sheaves}{} Let $X$ be a space. A $\bold{Set}$-sheaf on $X$ is a $\bold{Set}$-presheaf $\mF:\bold{Open}\to\bold{Sets}$ satisfying two additional properties
\begin{itemize}
\item Identity: If $\{U_i|i\in I\}$ is an open cover of $U$ and $\phi_1,\phi_2\in\mF(U)$ and $\phi_1|_{U_i}=\phi_2|_{U_i}$ for all $i$, then $\phi_1=\phi_2$
\item Gluing: If $\{U_i|i\in I\}$ is an open cover of $U$ and $\phi_i\in\mF(U_i)$ for all $i\in I$ such that $\phi_i|_{U_i\cap U_j}=\phi_j|_{U_i\cap U_j}$ for all $i,j\in I$, then there is exists some $\phi\in\mF(U)$ such that $\phi|_{U_i}=\phi_i$ for all $i\in I$. 
\end{itemize}
\end{defn}

\begin{eg}{}{}\\
Let $X,Y$ be spaces. Define a presheaf $\mF$ on $X$ by $$\mF(U)=\{f:U\to Y\;|\;f\text{ is continuous }\}$$ for $U\subseteq X$ an open set, together with restriction maps. Then $\mF$ is a sheaf. 
\begin{proof}\\
Identity: Suppose that $f,g\in\mF(U)$ for some open set $U\subseteq X$ and $f|_{U_i}=g|_{U_i}$ for $\{U_i\;|\;i\in I\}$ an open cover of $U$. Then for any $x\in U$, there exists $i\in I$ such that $x\in U_i$, and so $$f(x)=f|_{U_i}(x)=g|_{U_i}(x)=g(x)$$ Hence $f=g$. \\~\\

Gluing: Suppose that $f_i\in\mF(U_i)$ for $\{U_i\;|\;i\in I\}$ an open cover of an open set $U\subseteq X$ such that $f_i|_{U_i\cap U_j}=f_j|_{U_i\cap U_j}$. Define $f\in\mF(U)$ by $f(x)=f_i(x)$ for $x\in U_i$. This is well defined since $\{f_i\}$ agree on intersections. It remains to show that $f$ is continuous. Let $W\subseteq Y$ be an open set. I claim that $f^{-1}(W)=\bigcup_{i\in I}f_i^{-1}(W)$. Let $x\in f^{-1}(W)$. Then $f(x)\in W$. Since $x\in W\subseteq U=\bigcup_{i\in I}U_i$, there exists $i\in I$ such that $x\in U_i$. Hence $f_i(x)=f(x)\in W$. Thus $x\in f_i^{-1}(W)$. Conversely, suppose that $x\in f_i^{-1}(W)$. Then clearly $f(x)=f_i(x)\in W$ so that $x\in f^{-1}(W)$. Thus $f^{-1}(W)=\bigcup_{i\in I}f_i^{-1}(W)$. Since each $f_I$ is continuous and $W$ is an open set, $f_i^{-1}(W)$ is open and so the arbitrary union of $f_i^{-1}(W)$ is open. Hence $f^{-1}(W)$ is open. 
\end{proof}
\end{eg}

\begin{eg}{Locally Constant Sheaf}{}\\
Let $X$ be a space. Let $S$ be a set. Define the locally constant presheaf $\underline{S}$ of $X$ with values in $S$ as follows. 
\begin{itemize}
\item For each open set $U\subseteq X$, define $$\underline{S}(U)=\{f:U\to S\;|\;f\text{ is continuous where }S\text{ has the discrete topology }\}$$
\item For each inclusion $V\subseteq U$ of open subsets, define the induced map $\underline{S}(U)\to\underline{S}(V)$ by $f\mapsto f|_V$. 
\end{itemize}
The following are true regarding the locally constant sheaf. 
\begin{itemize}
\item $\underline{S}$ is a sheaf. 
\item Let $U\subseteq X$ be an open set. Let $\mB$ be the set of connected components of $U$. Then $$\underline{S}(U)\cong\prod_{C\in\mB}S$$
\item For all $p\in X$, $\underline{S}_p=S$. 
\end{itemize}
\begin{proof}\\
Let $Y$ be a connected component of $U$. Let $f\in\underline{S}(Y)$. I claim that $f$ is constant. Let $y_0\in Y$. Define $A=\{y\in Y\;|\;f(y)=f(y_0)\}=f^{-1}(f(y_0))$ and $B=\{y\in Y\;|\;f(y)\neq f(y_0)\}$. Clearly $Y=A\cup B$ and $A\cap B=\emptyset$. Both $A$ and $B$ are open sets since $S$ has a discrete topology. Since $Y$ is connected and $A$ is non-empty, this implies that $B=\emptyset$. Hence $Y=A$ and $f$ is constant on $Y$. Thus there is a bijection $\underline{S}(Y)\cong S$. \\

Define a function $\underline{S}(U)\to\prod_{C\in\mB}S$ as follows. For each connected component $C\in\mB$ of $U$, choose $x_C\in C$. Then the map is defined by $f\mapsto(f(x_C))$. This map is injective since $f(x_C)=g(x_C)$ implies that $f(C)=g(C)$ for all connected components $C\in\mB$. Hence $f=g$. It is surjective since for any choice of sequence $(x_C)_{C\in\mB}$, the map $f:U\to S$ defined by $f(x)=x_C$ for $x\in C$ is continuous because $S$ is discrete and $\{x_C\}$ is a closed set, and $f^{-1}(x_C)=C$ is a connected component and so is a closed set. 
\end{proof}
\end{eg}

\begin{prp}{Gluing Sheaves}{}\\
Let $X$ be a space. Let $\{U_i\;|\;i\in I\}$ be an open cover of $X$. Let $\mF_i$ be a sheaf of sets on $U_i$. Let $\phi_{i,j}:\mF_i|_{U_i\cap U_j}\to\mF_j|_{U_i\cap U_j}$ be a collection of morphism of sheaves for ranging $i,j\in I$ such that the following are true. 
\begin{itemize}
\item For each $i,j\in I$, $\phi_{i,j}$ is an isomorphism. 
\item For each $i,j,k\in I$, we have $\phi_{j,k}\circ\phi_{i,j}=\phi_{i,k}$. 
\end{itemize}
Then there exists a unique (up to unique isomoprhism) sheaf of sets $\mF$ on $X$ such that there are isomorphisms $\mF|_{U_i}\overset{\cong}{\to}\mF_i$ for all $i\in I$. 
\end{prp}

Let $X$ be a space. Let $\mS$ be a subcategory of $\bold{Open}(X)$. If $\mS\subseteq\bold{Open}(X)$ is filtered, then there is an equivalence of categories $$\bold{Sh}(X)=\text{Func}(\mS^\text{op},\bold{Set})$$
\end{prp}

\subsection{Sheaves with Values in a Category}
Presheaves are easily generalized to take values in an arbitrary category because they are simply defined to be contravariant functors. 

\begin{defn}{Presheaves with Values in a Category}{} Let $X$ be a space. Let $\mC$ be a category. A presheaf on $X$ with values in $\mC$ is a contravariant functor $$\mF:\bold{Open}(X)\to\mC$$
\end{defn}

However, it is hard to define sheaves with values on an arbitrary category $\mC$. This is because in our definition of sheaves on sets, we made use of the fact that morphisms in $\bold{Set}$ have a well defined notion of restriction. We will define sheaves on a general category by first considering sheaves with an underlying set. 

\begin{thm}{}{} Let $X$ be a space. Let $\mF$ be a presheaf with values in $\bold{Set}$. Then the following are equivalent characterizations. 
\begin{itemize}
\item $\mF$ is a sheaf
\item For any open covering $U=\bigcup_{i\in I}U_i$, the sequence \\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	0 & {\mF(U)} & {\prod_{i\in I}\mF(U_i)} & {\prod_{i,j\in I}\mF(U_i\cap U_j)}
	\arrow[from=1-1, to=1-2]
	\arrow["f", from=1-2, to=1-3]
	\arrow["g", from=1-3, to=1-4]
\end{tikzcd}}\\
where $f$ is given by the product of the unique maps $\mF(U)\to\mF(U_i)$ and $g$ is given by sending $(s_i)_{i\in I}$ to $(s_i|_{U_i\cap U_j}-s_j|_{U_i\cap U_j})_{i,j\in I}$ is exact. 
\item For any open covering $U=\bigcup_{i\in I}U_i$, the following diagram \\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	{\mF(U)} & {\prod_{i\in I}\mF(U_i)} & {\prod_{i,j\in I}\mF(U_i\cap U_j)}
	\arrow["f", from=1-1, to=1-2]
	\arrow["g", shift left, from=1-2, to=1-3]
	\arrow["h"', shift right, from=1-2, to=1-3]
\end{tikzcd}}\\
is an equalizer diagram, where $f$ is given by the product of the unique maps $\mF(U)\to\mF(U_i)$, $g$ is given by sending $(s_i)_{i\in I}$ to $(s_i|_{U_i\cap U_j})_{i,j\in I}$, $h$ is given by sending $(s_i)_{i\in I}$ to $(s_j|_{U_i\cap U_j})_{i,j\in I}$. 
\end{itemize} 
\begin{proof}~\\
We first note that the second and third condition are equivalent. Therefore we just have to show that the first condition is equivalent to the third one. Let $\mF$ be an abelian sheaf. We want to show that $\im(f)=\{x\in\prod_{i\in I}F(U_i)\;|\;g(x)=h(x)\}$. Therefore we begin with $x\in\im(f)$. Then $f(x)=(x_i)_{i\in I}$. But we have that $g((x_i)_{i\in I})=(x_i|_{U_i\cap U_j})_{i,j\in I}$ and $h((x_i)_{i\in I})=(x_j|_{U_i\cap U_j})_{i,j\in I}$. We want to show that for all $i,j\in I$, we have that $x_i|_{U_i\cap U_j}=x_j|_{U_i\cap U_j}$. But $x_i$ is just the restriction of $x$ to $U_i$ and $x_j$ is the restriction of $x$ to $U_j$. But by definition of a sheaf, restricting to $V_i$ and further to $V_{ij}$ is the same as directly restricting to $V_{ij}$. Similarly, restricting to $V_j$ and further to $V_{ij}$ is the same as directly restricting to $V_{ij}$. We conclude that applying $g$ and $h$ to $(x_i)_{i\in I}$ give the same element. Thus $x\in\{x\in\prod_{i\in I}F(U_i)\;|\;g(x)=h(x)\}$. \\

Now let $(y_i)_{i\in I}\in\{x\in\prod_{i\in I}F(U_i)\;|\;g(x)=h(x)\}$. This means that $y_i|_{U_i\cap U_j}=y_j|_{U_i\cap U_j}$ for all $i,j\in I$. By the gluing axiom, there exists $y\in C(U)$ such that $y|_{U_i}=y_i$. Thus $y$ is in the image of $f$ and we conclude that $y\in\im(f)$. \\

Finally, we also need to show that $f$ is injective. Suppose that $x,y\in\mF(U)$ is such that $x_i=y_i$ for all $i\in I$. Then by the identity axiom we conclude that $x=y$. Thus $f$ is injective. Thus the equalizer condition is satisfied. \\

Now suppose that the equalizer condition is satisfied. Then $f$ is injective. This means that for $x,y\in\mF(U)$ such that $x_i=y_i$ for all $i\in I$, then $x=y$. But this is precisely the identity axiom. Suppose that $\im(f)=\text{Eq}(g,h)$. Let $s_i\in\mF(U_i)$ such that $$s_i|_{U_i\cap U_j}=g(s_i)=h(s_i)=s_j|_{U_i\cap U_j}$$ for all $i,j\in I$. Then $(s_i)_{i\in I}\in\text{Eq}(g,h)=\im(f)$. Which means that there exists some $s\in\mF(U)$ such that $s|_{U_i}=s_i$. But this is the content of the gluing axiom. Since the identity and the gluing axiom are satisfied, we conclude that $\mF$ is a sheaf. 
\end{proof}
\end{thm}

Motivated by the above characterization, we define sheaves with values on a category with finite products as follows. 

\begin{defn}{Sheaves with Values in a Category}{} Let $X$ be a space. Let $\mC$ be a category that admits all products. Let $\mF$ be a presheaf on $X$ with values in $\mC$. We say that $\mF$ is a sheaf with values in $\mC$ if for every open covering $U=\bigcup_{i\in I}U_i$, the following diagram \\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	{\mF(U)} & {\prod_{i\in I}\mF(U_i)} & {\prod_{(i,j)\in I\times I}\mF(U_i\cap U_j)}
	\arrow["f", from=1-1, to=1-2]
	\arrow["g", shift left, from=1-2, to=1-3]
	\arrow["h"', shift right, from=1-2, to=1-3]
\end{tikzcd}}\\
is an equalizer diagram, where $f,g,h$ are defined as follows. 
\begin{itemize}
\item The morphism $f$ is the product $$\prod_{i\in I}\text{res}_{U_i}^U:\mF(U)\to\mF(U_i)$$
\item The morphism $g$ is the product $$\prod_{(i,j)\in I\times I}\text{res}_{U_i\cap U_j}^{U_i}:\mF(U_i)\to\mF(U_i\cap U_j)$$
\item The morphism $h$ is the product $$\prod_{(i,j)\in I\times I}\text{res}_{U_i\cap U_j}^{U_j}:\mF(U_j)\to\mF(U_i\cap U_j)$$
\end{itemize}
\end{defn}

Note: the double product is ordered so that the equalizing condition is the same as the gluing condition. 

\begin{prp}{}{}\\
Let $X$ be a space. Let $\mC$ be a category that admits all products. Let $F:\mC\to\bold{Set}$ be a functor such that the following are true. 
\begin{itemize}
\item $F$ is faithful. 
\item $F$ preserves limits. 
\item A morphism $f\in\mC$ is an isomorphism if and only if $F(f)\in\bold{Set}$ is an isomorphism. 
\end{itemize}
Let $\mF$ be a presheaf on $X$ with values in $\mC$. Then $\mF$ is a sheaf with values in $\mC$ if and only if $F\circ\mF$ is a sheaf of sets. 
\begin{proof}\\
Suppose that $\mF$ is a sheaf. Then the following diagram is an equalizer for any open set $U$ and any open cover $\{U_i\;|\;i\in I\}$ of $U$: \\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	{\mF(U)} & {\prod_{i\in I}\mF(U_i)} & {\prod_{(i,j)\in I\times I}\mF(U_i\cap U_j)}
	\arrow["f", from=1-1, to=1-2]
	\arrow["g", shift left, from=1-2, to=1-3]
	\arrow["h"', shift right, from=1-2, to=1-3]
\end{tikzcd}}\\
Since $F$ commutes with limits, the following diagram is an equalizer: \\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	{F\circ\mF(U)} & {\prod_{i\in I}F\circ\mF(U_i)} & {\prod_{(i,j)\in I\times I}F\circ\mF(U_i\cap U_j)}
	\arrow["f", from=1-1, to=1-2]
	\arrow["g", shift left, from=1-2, to=1-3]
	\arrow["h"', shift right, from=1-2, to=1-3]
\end{tikzcd}}\\
Hence $F\circ\mF$ is a sheaf. \\

Conversely, suppose that $F\circ\mF$ is a sheaf. Let $E$ be the equalizer of the following diagram: \\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	{\prod_{i\in I}F\circ\mF(U_i)} & {\prod_{(i,j)\in I\times I}F\circ\mF(U_i\cap U_j)}
	\arrow["g", shift left, from=1-1, to=1-2]
	\arrow["h"', shift right, from=1-1, to=1-2]
\end{tikzcd}}\\
By the universal property of products and the definition of a presheaf, we obtain a morphism $\mF(U)\to E$. Since $F$ preserves limits, we have that $F\circ\mF(U)$ and $F(E)$ are both equalizers of the same diagram, and so the morphism $F\circ\mF(U)\to F(E)$ is an isomorphism. By assumption on $F$, we conclude that $\mF(U)\cong E$. Hence $\mF$ is a sheaf. 
\end{proof}
\end{prp}

The following definition is inspired by the Stacks Project. 

\begin{defn}{Algebraic Categories}{}\\
Let $\mC$ be a category. We say that $\mC$ is an algebraic category if there exists a functor $F:\mC\to\bold{Set}$ such that the following are true. 
\begin{itemize}
\item $\mC$ admits all products and filtered colimits. 
\item $F$ is faithful (so that $\mC$ is a concrete category). 
\item $F$ preserves limits and filtered colimits. 
\item A morphism $f\in\mC$ is an isomorphism if and only if $F(f)\in\bold{Set}$ is an isomorphism. 
\end{itemize}
\end{defn}

This allows us to simplify statements about sheaves with values in categories whose objects are algebraic in nature. Some examples of algebraic categories include: $$\bold{Set},\bold{Grp},\bold{Ab},\bold{Ring},\bold{CRing},\bold{Mod}_R,\bold{Alg}_R$$ They are all equipped with a forgetful functor to sets that is faithful, is the right adjoint of the free functor (take $\text{id}:\bold{Set}\to\bold{Set}$ for $\bold{Set}$) and reflects isomorphisms. \\

We abuse notation and think of as if $\mF(U)$ are sets for each open set $U\subseteq X$ (in reality only $(F\circ\mF)(U)$ are sets). 

\begin{lmm}{}{}\\
Let $X$ be a topological space. Let $\mC$ be an algebraic category. Let $\mF$ be a $\mC$-presheaf on $X$. Let $p\in X$. Then we have $$\mF_p=\frac{\{(U,f)\;|\;U\subseteq X\text{ is an open neighborhood of }p, f\in\mF(U)\}}{\sim}$$ where $(U,f)\sim(V,g)$ if there exists $W\subseteq U\cap V$ an open neighborhood of $p$ such that $f|_W=g|_W$. 
\begin{proof}
If $\mF$ is a $\bold{Set}$-presheaf, the above would be an explicit construction of the filtered colimit. Since $\mC$ is algebraic, $F:\mC\to\bold{Set}$ preserves colimits and by abuse of notation we obtain the following set-theoretic description. 
\end{proof}
\end{lmm}

Think of the definition of stalks as follows: Treat $f$ and $g$ to be sections in $\mF(U_1)$ and $\mF(U_2)$ where $V\subseteq U_1\cap U_2$ is open and contains $x$. Then we think of $f$ and $g$ to be the same function in the stalk as long as they agree on some open set that contains $x$. \\

Indeed, since we do not care about the entirety of the domain of $f$ and $g$, and only care about what happens locally near $x$, it makes sense for us to treat them as a function when they appear to be the same locally. \\

Let $[(s,U)]\in\mF_p$ be an element of the stalk. Consider the following commutative diagram: \\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	{\mF(U)} & {\mG(U)} \\
	{\mF_p} & {\mG_p}
	\arrow[from=1-1, to=1-2]
	\arrow[from=1-1, to=2-1]
	\arrow[from=1-2, to=2-2]
	\arrow[from=2-1, to=2-2]
\end{tikzcd}} \\
The element $s\in\mF(U)$ maps vertically on the left to $[(s,U)]\in\mF_p$. Since the diagram is commutative, we see that the image of the element in $\mF_p$ from the induced map of stalks is $[(\phi(U)(s),U)]$. This is the formula for the induced map. 

\begin{prp}{}{}\\
Let $X$ be a topological space. Let $\mC$ be an algebraic category. Let $\mF,\mG$ be $\mC$-presheaves on $X$. Let $\varphi:\mF\to\mG$ be a morphism. Then $\varphi$ is an isomorphism if and only if for all $p\in X$, the induced map $$\varphi_p:\mF_p\to\mG_p$$ is an isomorphism. 
\begin{proof}\\
Suppose that $\phi$ is an isomorphism of sheaves. Then $\phi$ and $\phi^{-1}$ induces unique morphisms $\phi_p:\mF_p\to\mG_{X,p}$ and $\phi_p^{-1}:\mG_{X,p}\to\mF_p$ making the following diagram commute: \\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	{\mF(U)} & {\mG(U)} & {\mF(U)} & {\mG(U)} \\
	{\mF_p} & {\mG_{X,p}} & {\mF_p} & {\mG(V)}
	\arrow["{\phi(U)}", from=1-1, to=1-2]
	\arrow[from=1-1, to=2-1]
	\arrow["{\phi^{-1}(U)}", from=1-2, to=1-3]
	\arrow[from=1-2, to=2-2]
	\arrow["{\phi^{-1}(U)}", from=1-3, to=1-4]
	\arrow[from=1-3, to=2-3]
	\arrow[from=1-4, to=2-4]
	\arrow["{\exists!}"', dashed, from=2-1, to=2-2]
	\arrow["{\exists!}"', dashed, from=2-2, to=2-3]
	\arrow["{\exists!}"', dashed, from=2-3, to=2-4]
\end{tikzcd}} \\
On the other hand, the identity morphisms $\text{id}_\mF$ and $\text{id}_\mG$ also induces the identity map on the level of stalks. Thus we must have $\phi_p\circ\phi^{-1}_p=\text{id}_{\mG_{X,p}}$ and $\phi^{-1}_p\circ\phi_p=\text{id}_{\mF_p}$ . Hence the induced morphism of stalks is an isomorphism. \\

Now suppose that $\phi_p$ is an isomorphism for each $p\in X$. We show that $\phi(U)$ is bijective for all $U\subseteq X$ open. Let $s\in\mF(U)$ such that $\phi(U)(s)=0$. Then for each $p\in U$, $\phi(U)(s)$ considered as an element in $\mG_{X,p}$ is equal to $0$. Since $\phi_p$ is injective, $s=0$ in $\mF_p$. This means that there exists an open neighbourhood $V_p\subseteq U$ of $p$ such that $s\in\phi(V_p)$ is $0$ for each $p\in U$. Since $U$ is covered by the open neighbourhoods $V_p$ for $p\in U$, by the identity axiom we conclude that $s=0$ in $\phi(U)$. Thus $\phi$ is injective. \\

Now let $t\in\mG(U)$. Let $t_p$ be the corresponding element of $t$ in $\mG_{X,p}$ for each $p\in U$. Since $\phi_p$ is surjective, there exists $s_p\in\mF_p$ such that $\phi_p(s_p)=t_p$ for each $p\in U$. Suppose that on a neighbourhood $V_p$ of $p$, $s_p$ is represented as $r_p$. Then $\phi(r_p)\in\mG(V_p)$ and $t|_{V_p}$ are two elements whose germs in $\mG_{X,p}$ are equal. Hence there exists $W_p\subseteq V_p$ containing $p$ such that $\phi(r_p)=t|_{W_p}$ in $\mG(W_p)$. Now $U$ is covered by open sets of the form $W_p$ for each $p\in U$. Let $p\in W_p$ and $q\in W_q$. Then both $r_p|_{W_p\cap W_q}$ and $r_q|_{W_p\cap W_q}$ are two sections in $\mF(W_p\cap W_q)$ that are sent to $t|_{W_p\cap W_q}\in\mG(W_p\cap W_q)$ by $\phi(W_p\cap W_q)$. By injectivity, they are equal. We conclude that $$r_p|_{W_p\cap W_q}=r_q|_{W_p\cap W_q}$$ By the gluing axiom, we conclude that there exists a section $s\in\mF(U)$ such that $s|_{W_p}=r_p$. Now $\phi(s)$ and $t$ are two sections in $\mG(U)$ such that for any $p\in U$, $\phi(s)|_{W_p}=t|_{W_p}$. By the identity axiom, we conclude that $\phi(s)=t$. 
\end{proof}
\end{prp}

Note that the above proposition is very untrue for presheaves because as one can see, proving both injective and surjective requires to use of the sheaf axioms instead of just the presheaf datum. 

\begin{prp}{}{}\\
Let $X$ be a topological space. Let $\mC$ be an algebraic category. Let $\mF,\mG$ be $\mC$-presheaves on $X$. Let $\varphi,\psi:\mF\to\mG$ be morphisms. Then $\varphi=\phi$ if and only if for all $p\in X$, we have $\varphi_p=\phi_p$. 
\begin{proof}\\
Clearly if $\varphi(U)=\psi(U)$ for all open sets $U\subseteq X$, we must have that the induced map of colimits are equal. So suppose instead that $\varphi_p=\psi_p$ for all $p\in X$. Consider the following commutative diagram: \\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	{\mF(U)} & {\mG(U)} \\
	{\prod_{p\in U}\mF_p} & {\prod_{p\in U}\mG_p}
	\arrow[from=1-1, to=1-2]
	\arrow[from=1-1, to=2-1]
	\arrow[from=1-2, to=2-2]
	\arrow[from=2-1, to=2-2]
\end{tikzcd}} \\
where the top and bottom maps are either $\varphi$ or $\psi$ and their induced product map of stalks. \\

I claim that the vertical maps are injective. Suppose that $s,t\in\mF(U)$ are such that $([(s,U)])_{p\in U}=([(t,U)])_{p\in U}$. This means that for each $p\in U$, we have that $[(s,U)]=[(t,U)]\in\mF_p$. This means that there exists an open subset $p\in V_p\subseteq U$ such that $s|_{V_p}=t|_{V_p}$. Now the collection $\{V_p\;|\;p\in U\}$ forms an open cover of $U$. Since $\mF$ is a sheaf, the identity axiom implies that $s=t$ on $U$. Hence the map is injective. \\

Let $s\in\mF(U)$. Then the left vertical map sends $s$ to $([(s,U)])_{p\in U}$ and $\prod_{p\in U}\varphi_p=\prod_{p\in U}\psi_p$ sends the element to $([(\varphi(U)(s),U)])_{p\in U}=([(\psi(U)(s),U)])_{p\in U}$. Since the right vertical map is injective, we must have $\varphi(U)(s)=\psi(U)(s)$. Hence we have $\varphi=\psi$. 
\end{proof}
\end{prp}

\begin{defn}{Compatible Germs}{}\\
Let $X$ be a space. Let $\mC$ be an algebraic category. Let $\mF$ be an $\mC$-presheaf on $X$. Let $([(s_p,U_p)])_{p\in U}\in\prod_{p\in U}\mF_p$. We say that the element is a compatible germ if for all $p\in U$, there exists an open subset $p\in V_p\subseteq U$ such that $$[(s_p|_{V_p},V_p)]=[(s_q,U_q)]\in\mF_q$$ for all $q\in V_p$. 
\end{defn}

\begin{lmm}{}{}\\
Let $X$ be a space. Let $\mC$ be an algebraic category. Let $\mF$ be an $\mC$-presheaf on $X$. Let $([(s_p,U_p)])_{p\in U}\in\prod_{p\in U}\mF_p$. Then the element is a compatible germ if and only if there exists an open cover $U=\bigcup_{i\in I}U_i$ of $U$ together with elements $f_i\in\mF(U_i)$ such that for all $p\in U_i$ for all $i$, $[(f_i,U_i)]=[(s_p,U_p)]\in\mF_p$. 
\begin{proof}\\
Suppose that $([(s_p,U_p)])_{p\in U}\in\prod_{p\in U}\mF_p$ is a compatible germ. Then $\{U_p\;|\;p\in U\}$ is a valid open cover of $U$ together with the elements $s_p\in\mF(U_p)$ so that the latter condition is satisfied. \\

Conversely, suppose that the latter condition is satisfied. Let $p\in U$. Since $\{U_i\;|\;i\in I\}$ is an open cover of $U$, there exists $i\in I$ such that $p\in U_i$. Then for all $q\in U_i$, we have $[(f_i,U_i)]=[(s_q,U_q)]\in\mF_q$ by assumption. Hence we are done. 
\end{proof}
\end{lmm}

Thus not only are isomorphism of sheaves determined on the level of stalks, the morphism of sheaves themselves are determined on the level of stalks. 

\begin{defn}{Sheafification}{}\\
Let $X$ be a space. Let $\mC$ be an algebraic category. Let $\mF$ be an $\mC$-presheaf on $X$. Define the sheafification $$\mF^\text{sh}:\bold{Open}(X)\to\mA$$ of $\mF$ as follows. 
\begin{itemize}
\item For each open set $U\subseteq X$, $$\mF^\text{sh}(U)=\left\{([(s_p,U_p)])_{p\in U}\in\prod_{p\in U}\mF_p\;\bigg{|}\;([(s_p,U_p)])_{p\in U}\text{ is a compatible germ }\right\}$$
\item For each $V\subseteq U$ an inclusion, there is a unique morphism $\mF^\text{sh}(U)\to\mF^\text{sh}(V)$ that sends $([(s_p,U_p)])_{p\in U}\in\in\mF^\text(U)$ to $$([(s_p,U_p)])_{p\in V}$$
\end{itemize}
\end{defn}

\begin{lmm}{}{}\\
Let $X$ be a space. Let $\mC$ be an algebraic category. Let $\mF$ be an $\mC$-presheaf on $X$. Then $\mF^\text{sh}$ is a $\mC$-sheaf. 
\begin{proof}\\
We first show that $\mF^\text{sh}$ is a sheaf. Let $\{W_i\;|\;i\in I\}$ be an open cover for an open set $U\subseteq X$. \\

For the identity axiom, suppose that $([(s_p,U_p)])_{p\in U}$ and $([t_p,V_p])_{p\in U}$ in $\mF(U)$ are such that their restriction to $\mF^\text{sh}(W_i)$ are equal. This means that $([(s_p,U_p)])_{p\in W_i}=([(s_p,U_p)])_{p\in W_i}$ or in other words, that $[(s_p,U_p)]=[(t_p,V_p)]\in\mF_p$. Then since $\{W_i\;|\;i\in I\}$ covers $U$, we must have $([(s_p,U_p)])_{p\in U}=([t_p,V_p])_{p\in U}$ in $\mF(U)$. \\

For gluing, suppose that we are given $([(s_p,U_p)])_{p\in W_i}\in\mF^\text{sh}(W_i)$ for each $i\in I$ such that restricting to common intersections give equal elements. This means that the element $([(s_p,U_p)])_{p\in U}$ is a well defined element of $\mF^\text{sh}$. 
\end{proof}
\end{lmm}

\begin{eg}{}{}\\
Let $X$ be a space. Let $S$ be a set. Then we have $$\underline{S}=\left(\underline{S}_\text{pre}\right)^\text{sh}$$
\end{eg}

\begin{defn}{Universal Morphism of Sheafification}{}\\
Let $X$ be a space. Let $\mC$ be an algebraic category. Let $\mF$ be an $\mC$-presheaf on $X$. Define the universal morphism of sheafification $\theta_\mF$ as follows. For each open set $U\subseteq X$, define $$\theta_\mF(U):\mF(U)\to\prod_{p\in U}\mF_p$$ by $s\mapsto([(s,U)])_{p\in U}$. 
\end{defn}

\begin{lmm}{}{}\\
Let $X$ be a space. Let $\mC$ be an algebraic category. Let $\mF$ be an $\mC$-presheaf on $X$. Then the following are true. 
\begin{itemize}
\item $\im(\theta_\mF)=\mF^\text{sh}(U)$. 
\item If $\mF$ is a sheaf, then for any open subset $U\subseteq X$, $\theta_\mF(U):\mF\to\mF^\text{sh}$ is injective. 
\end{itemize}
\begin{proof}
\begin{itemize}
\item Clearly any element in the image of the map is a compatible germ. Now conversely, suppose that $([(s_p,U_p)])_{p\in U}$ is a compatible germ. Now $\{U_p\;|\;p\in U\}$ is an open cover of $U$. Let $p,q\in U$ be arbitrary. Let $r\in U_p\cap U_q$. Then there exists an open subset $V_r\subseteq U_p$ such that $[(s_p|_{V_r},V_r)]=[(s_q,U_q)]\in\mF_q$. This means that there exists an open subset $W_r\subseteq V_r\cap U_q$ such that $s_p|_{W_r}=s_q|_{W_r}$. Now since $\{W_r\;|\;r\in U_p\cap U_q\}$ covers $U_p\cap U_q$, we must have that $s_p|_{U_i\cap U_j}=s_q|_{U_i\cap U_j}$ by the identity axiom. Then by the gluability axiom, there exists $s\in\mF(U)$ such that $s|_{U_p}=s_p\in\mF(U_p)$. Now clearly $([(s,U)])_{p\in U}=([(s_p,U_p)])_{p\in U}$ since $s|_{U_p}=s_p\in\mF(U_p)$. Hence the map is surjective. 
\item Suppose that $s,t\in\mF(U)$ are such that $([(s,U)])_{p\in U}=([(t,U)])_{p\in U}$. This means that for each $p\in U$, we have that $[(s,U)]=[(t,U)]\in\mF_p$. This means that there exists an open subset $p\in V_p\subseteq U$ such that $s|_{V_p}=t|_{V_p}$. Now the collection $\{V_p\;|\;p\in U\}$ forms an open cover of $U$. Since $\mF$ is a sheaf, the identity axiom implies that $s=t$ on $U$. Hence the map is injective. 
\end{itemize}
\end{proof}
\end{lmm}

\begin{lmm}{}{}\\
Let $X$ be a space. Let $\mC$ be an algebraic category. Let $\mF$ be an $\mC$-presheaf on $X$. Then $\theta$ induces an isomorphism $$\theta_p:\mF_p\overset{\cong}{\longrightarrow}\mF_p^\text{sh}$$ of stalks for each $p\in X$. 
\begin{proof}\\
By the universal property of stalks as colimits, the morphism $\theta_p$ is defined by $[(s,U)]\mapsto[(([(s,U)])_{p\in U},U)]$. \\

Suppose that $[(([(s,U)])_{q\in U},U)]=[(([(t,U)])_{q\in U},U)]$. By the definition of the equivalence relation, we have that $([(s,U)])_{q\in U}$ and $([(t,U)])_{q\in U}$ agree on $U$. Hence for each $q\in U$, we have $[(s,U)]=[(t,U)]$. Again by the definition of the equivalence relation, we have $s=t$ on $U$. Hence $\theta_p$ is injective. \\

To show that $\theta_p$ is surjective, let $[(([(s_q,U_q)])_{q\in U},U)]\in\mF_p^\text{sh}$. Since $s$ is a compatible germ by definition, there exists an open cover $\{U_i\;|\;i\in I\}$ of $U$ together with elements $f_i\in\mF(U_i)$ such that $[(f_i,U_i)]=[(s_q,U_q)]\in\mF_p$ for all $q\in U_i$ for all $i$. In particular, for our already fixed choice of $p\in X$, there exists $i\in I$ and $f_i\in\mF(U_i)$ such that $[(f_i,U_i)]=[(s_p,U_p)]\in\mF_p$. Now $[(f_i,U_i)]\in\mF_p$ is mapped to $[(([(f_i,U_i)])_{q\in U_i},U_i)]$. I claim that $$[(([(f_i,U_i)])_{q\in U_i},U_i)]=[(([(s_q,U_q)])_{q\in U},U)]\in\mF_p^\text{sh}$$ Indeed, in the intersection $U_i=U_i\cap U$, we have $[(f_i,U_i)]=[(s_q,U_q)]$ for all $q\in U_i$. by definition. Hence we have $([(f_i,U_i)])_{q\in U_i}=([(s_q,U_q)])_{q\in U_i}$ and hence we have surjectivity. 
\end{proof}
\end{lmm}

\begin{prp}{The Universal Property of Sheafification}{}\\
Let $X$ be a space. Let $\mC$ be an algebraic category. Let $\mF$ be an $\mC$-presheaf on $X$. Then the sheafification $\mF^\text{sh}$ of $\mF$ together with the universal morphism $\theta$ satisfies the following universal property. For any sheaf $\mG$ on $X$ and morphism of presheaves $\varphi:\mF\to\mG$, there exists a unique morphism $\varphi^\text{sh}:\mF^\text{sh}\to\mG$ such that the following diagram commutes: \\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	\mF & {\mF^\text{sh}} \\
	& \mG
	\arrow["\theta", from=1-1, to=1-2]
	\arrow["\varphi"', from=1-1, to=2-2]
	\arrow["{\exists!\varphi^\text{sh}}", dashed, from=1-2, to=2-2]
\end{tikzcd}} \\
Moreover, $\mF^+$ is the unique (up to unique isomorphism) sheaf of $X$ satisfying the above property. 
\begin{proof}\\
Let $\mG$ be a sheaf. Let $\phi:\mF\to\mG$ be a morphism of presheaves. To define a morphism of sheaves $\mF^\text{sh}\to\mG$, we first define it on each open set $U$ and prove that the relevant square diagrams commute. So define $\phi^\text{sh}(U):\mF^\text{sh}(U)\to\mG(U)$ for each open set $U\subseteq X$ as follows. For $([(s_p,U_p)])_{p\in U})$ an element in $\prod_{p\in U}\mF_p$, the product map $\prod_{p\in U}\phi_p$ sends the our element to $([\phi(U_p)(s_p),U_p)])_{p\in U}$. Let $([(s_p,U_p)])_{p\in U}$ be a compatible germ. I claim that $([\phi(U_p)(s_p),U_p)])_{p\in U}$ is a compatible germ. By lemma 1.2.6, there exists an open cover $\{U_i\;|\;i\in I\}$ of $U$ together with elements in $f_i\in\mF(U_i)$ such that for all $i\in I$ and all $p\in U_i$, $[(f_i,U_i)]=[(s_p,U_p)]$. Now the elements $\phi(U_i)(f_i)$ for all $i$ proves the claim since $\phi_p$ maps $[(f_i,U_i)]$ to $[(\phi(U_i)(f_i),U_i)]$, $[(s_p,U_p)]\mapsto[(\phi(U_p)(s_p),U_p)]$ and $[(f_i,U_i)]=[(s_p,U_p)]\in\mF_p$ implies that $[(\phi(U_i)(f_i,U_i)]=[(\phi(U_p)(s_p),U_p)]$. By lemma 1.2.9, we obtain a unique element in $\mG(U)$. Hence we obtain a map $\phi^\text{sh}(U):\mF^\text{sh}(U)\to\mG(U)$. \\

Now we show that the composite $\mF\to\mF^\text{sh}\to\mG$ is equal to $\phi$. Notice that given $s\in\mF(U)$, it is sent to $([(s,U)]_{p\in U}$ in $\mF^\text{sh}(U)$ and then to $([(\phi(U)(s),U)])_{p\in U}$ in $\mG(U)$. Evidently, $\phi(U)(s)\in\mG(U)$ is the unique preimage of $([(\phi(U)(s),U)])_{p\in U}$ of the map in lemma 1.2.7. Hence $s\mapsto\phi(U)(s)$ matches up with the map of presheaves $\phi:\mF\to\mG$. \\

It remains to show uniqueness of the morphism $\phi^\text{sh}$. Suppose that $\psi:\mF^\text{sh}\to\mG$ is another morphism of sheaves such that $\phi=\psi\circ\theta$. By the above lemma, $\theta$ is an isomorphism at stalks. Hence for all $p\in X$, we have $$\phi_p\circ\theta_p^{-1}=\psi_p$$ On the other hand, the equation $\phi=\phi^\text{sh}\circ\theta$ gives $$\phi_p\circ\theta_p^{-1}=\phi_p^\text{sh}$$ Equating the two gives $\psi_p=\phi_p^\text{sh}$ for all $p\in X$. Since $\mF^\text{sh}$ and $\mG$ are sheaves, By 1.2.6 we conclude that $\psi=\phi^\text{sh}$ and so we are done. 
\end{proof}
\end{prp}

\begin{lmm}{}{}\\
Let $X$ be a space. Let $\mC$ be an algebraic category. Let $\mF$ be an $\mC$-presheaf on $X$. Then the universal morphism $\theta:\mF\to\mF^\text{sh}$ is an isomorphism. 
\begin{proof}\\
By 1.2.11, we have that $\theta_p$ is an isomorphism for all $p\in X$. Hence by 1.2.4, we have that $\theta$ is an isomorphism. 
\end{proof}
\end{lmm}

Thus the theory of compatible germs gives us an explicit way of understanding the elements of a sheaf by a collection of elements in the stalks. 

\begin{defn}{The Sheafification Functor}{}\\
Let $X$ be a space. Let $\mC$ be an algebraic category. Define the sheafification functor $$(-)^\text{sh}:\bold{P}\mC(X)\to\mC(X)$$ as follows. 
\begin{itemize}
\item For each presheaf $\mF$ on $X$, $\mF^\text{sh}$ is the sheafification of $\mF$. 
\item For each morphism of presheaves $\mF\to\mG$, the morphism of sheaves is given by the universal property of $\mF^\text{sh}$ applied to the composite $\mF\to\mG\to\mG^\text{sh}$. 
\end{itemize}
\end{defn}

\begin{prp}{}{}\\
Let $X$ be a space. Let $\mC$ be an algebraic category. Let $\text{Forget}:\mC(X)\to\bold{P}\mC(X)$ be the forgetful functor. Then there is an adjunction $$(-)^\text{sh}:\bold{P}\mC(X)\overset{\dashv}{\rightleftarrows}\mC(X):\text{Forget}$$
\begin{proof}\\
The universal property of sheafification gives a map from the right hom set to the left hom set. Conversely, precomposition with the universal morphism $\mF\to\mF^\text{sh}$ gives the map to the other direction. Evidently they are inverses of each other. It remains to show that this bijection is natural in both variables. In other words, we ought to show that the following two squares commute: \\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	{\Hom_{\bold{Set(X)}}(\mF^\text{sh},\mG)} & {\Hom_{\bold{PSet}(X)}(\mF,\text{Forget}(\mG))} \\
	{\Hom_{\bold{Set(X)}}(\mH^\text{sh},\mG)} & {\Hom_{\bold{PSet}(X)}(\mH,\text{Forget}(\mG))} \\
	{\Hom_{\bold{Set(X)}}(\mF^\text{sh},\mG)} & {\Hom_{\bold{PSet}(X)}(\mF,\text{Forget}(\mG))} \\
	{\Hom_{\bold{Set(X)}}(\mF^\text{sh},\mI)} & {\Hom_{\bold{PSet}(X)}(\mF,\text{Forget}(\mI))}
	\arrow["\cong", from=1-1, to=1-2]
	\arrow["{-\circ\varphi^\text{sh}}", from=2-1, to=1-1]
	\arrow["\cong"', from=2-1, to=2-2]
	\arrow["{-\circ\varphi}"', from=2-2, to=1-2]
	\arrow["\cong", from=3-1, to=3-2]
	\arrow["{\psi\circ-}"', from=3-1, to=4-1]
	\arrow["{\text{Forget}(\psi)\circ-}", from=3-2, to=4-2]
	\arrow["\cong"', from=4-1, to=4-2]
\end{tikzcd}} \\
for two morphisms $\varphi:\mF\to\mH$ and $\psi:\mG\to\mI$. Since $\bold{Sh}(X)$ is a full subcategory of $\bold{PSh}(X)$, clearly the square on the bottom commutes. For the top square, let $\sigma:\mH^\text{sh}\to\mG$ be a morphism. Following the left and top arrows, we obtain the morphism $\sigma\circ\varphi^\text{sh}\circ\theta_\mF$. Following the bottom and right arrows, we obtain the morphism $\sigma\circ\theta_\mH\circ\varphi$. I claim that $\varphi^\text{sh}\circ\theta_\mF=\theta_\mH\circ\varphi$. Indeed, the definition of $\varphi^\text{sh}$ as in def 2.1.10 is given by the universal property of sheafification of the composite $\theta_\mH\circ\varphi$. Hence precomposing $\varphi^\text{sh}$ with $\theta_\mF$ is precisely $\theta_\mH\circ\varphi$. Thus we are done. 
\end{proof}
\end{prp}

\subsection{The Category of (Pre)Sheaves as an Abelian Category}
\begin{lmm}{}{}\\
Let $X$ be a topological space. Let $\mA$ be an abelian category. Let $\ast$ be the zero object of $\mA$. Then the sheaf $\underline{\ast}$ is the zero object of $\mA(X)$. 
\begin{proof}
Let $\mF\in\mA(X)$ be an abelian sheaf. We first show the existence and uniqueness of a morphism $\underline{\ast}\to\mF$. Suppose that $\varphi:\underline{\ast}\to\mF$ is a morphism. Then for each open set $U\subseteq X$, there is a unique morphism $\varphi(U):\underline{\ast}(U)=\ast\to\mF(U)$. Hence $\varphi$ is unique. Existence is also evident. \\

Similarly, we show the existence and uniqueness of a morphism $\mF\to\underline{\ast}$. Suppose that $\varphi:\mF\to\underline{\ast}$ is a morphism. Then for each open set $U\subseteq X$, there is a unique morphism $\varphi(U):\mF(U)\to\ast=\underline{\ast}(U)$. Hence $\varphi$ is unique. Existence is also evident. 
\end{proof}
\end{lmm}

\begin{defn}{The Product Sheaf}{}\\
Let $X$ be a space. Let $\mC$ be a category that admits all products. Let $\mF,\mG$ be $\mC$-sheaves on $X$. Define the product $\mF\times\mG$ sheaf of $\mF$ and $\mG$ as follows. 
\begin{itemize}
\item For each open set $U\subseteq X$, define $(\mF\times\mG)(U)=\mF(U)\times\mG(U)$.

\begin{lmm}{}{}\\
Let $X$ be a space. Let $\mC$ be a category that admits all products. Let $\mF,\mG$ be $\mC$-sheaves on $X$. Then $\mF\times\mG$ is the categorical product of $\mF$ and $\mG$ in $\mC(X)$. 
\begin{proof}\\
Firstly, the identity and gluing axiom of $\mF\times\mG$ can be checked coordinate wise. Clearly, there are projection morphisms $\mF\times\mG\to\mF,\mG$ open set wise. Let $\mH$ be a sheaf with morphisms $\mH\to\mF,\mG$. Then by the universal product of products of abelian groups, there is a unique morphism $\mH(U)\to(\mF\times\mG)(U)$ for each open set $U\subseteq X$ such that the composition $\mH(U)\to(\mF\times\mG)(U)\to\mF(U),\mG(U)$ is equal to $\mH(U)\to\mF(U),\mG(U)$ respectively. Then they assemble into a morphism of sheaves $\mH\to\mF\times\mG$. This morphism is unique by construction. 
\end{proof}
\end{lmm}

\subsection{Exact Sequences of Sheaves}
From now on, we focus on the abelian category of sheaves and not presheaves. Therefore we will drop the sheafification symbol for cokernels and images since they mean the abelian categorical object that is the cokernel and the image. \\

Let $X$ be a space. Let $\mA$ be an abelian category that is an algebraic category. Since $\mA(X)$ is an abelian category, it makes sense to talk about chain complexes and in particular, exact sequences of sheaves. Recall that a sequence of sheaves $\mF\to\mG\to\mH$ is exact if $\im(\mF\to\mG)=\ker(\mG\to\mH)$. 

\begin{prp}{}{}\\
Let $X$ be a space. Let $\mA$ be an abelian category that is an algebraic category. Suppose the following \\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	\mF & \mG & \mH
	\arrow[from=1-1, to=1-2]
	\arrow[from=1-2, to=1-3]
\end{tikzcd}} \\~\\
is a sequence of $\mA$-sheaves on $X$. Then the sequence is exact if and only if the following sequence of stalks\\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	{\mF_p} & {\mG_p} & {\mH_p}
	\arrow[from=1-1, to=1-2]
	\arrow[from=1-2, to=1-3]
\end{tikzcd}} \\~\\
is exact for all $p\in X$. 
\begin{proof}
We have that $\im(\mF\to\mG)=\ker(\mG\to\mH)$ and taking stalks gives $\im(\mF_p\to\mG_p)=\ker(\mG_p\to\mH_p)$. Since taking stalks commute with images and kernels, we are done. Conversely, suppose that for all $p\in X$, we have $\im(\mF_p\to\mG_p)=\ker(\mG_p\to\mH_p)$. Since taking stalks commute with kernels and images, this show that $\im(\mF\to\mG)_p=\ker(\mG\to\mH)_p$. But two sheaves are isomorphic if and only if the stalks are isomorphic, we are done. 
\end{proof}
\end{prp}

Thus the functor $$(-)_p:\mA(X)\to\mA$$ of taking stalks is exact.

\begin{defn}{Direct Image Sheaf}{}\\
Let $X,Y$ be spaces. Let $f:X\to Y$ be a map. Let $\mC$ be category. Let $\mF$ be a $\mC$-presheaf on $X$. Define the direct image of sheaf $$f_\ast\mF:\bold{Open}(Y)\to\mC$$ as follows. 
\begin{itemize}
\item For each open set $U\subseteq Y$, define $f_\ast\mF(U)=\mF(f^{-1}(U))$. 
\item For each inclusion $V\subseteq U$, define the induced map $$\mF(f^{-1}(U))\to\mF(f^{-1}(V))$$ to be the map $\mF(\text{incl.}:f^{-1}(V)\hookrightarrow f^{-1}(U))$. 
\end{itemize}
\end{defn}

\begin{prp}{}{}\\
Let $X,Y$ be spaces. Let $f:X\to Y$ be a map. Let $\mC$ be an algebraic category. Let $\mF$ be a $\mC$-sheaf. Then $f_\ast\mF$ is a sheaf on $Y$. 
\begin{proof}\\
It is clear that $f_\ast\mF$ is a presheaf on $Y$. We need to check the identity and gluing axiom. By 1.3.4, without loss of generality we can assume that $\mC=\bold{Set}$. \\

Identity: Let $U\subseteq Y$ be an open set. Let $\{U_i\;|\;i\in I\}$ be an open cover of $U$. Denote the inclusion maps by $\iota_i:U_i\to U$. Let $s,t\in f_\ast\mF(U)$. Suppose that $f_\ast\mF(\iota_i)(s)=f_\ast\mF(\iota_i)(t)$ for all $i\in I$. This means that $s,t\in\mF(f^{-1}(U))$ is such that the images of $s,t$ in $\mF(f^{-1}(U_i))$ are equal for all $i\in I$. Since $\{f^{-1}(U_i)\}$ is an open cover of $f^{-1}(U)$ and $\mF$ is a sheaf on $X$, this means that $s=t$. \\

Gluing: Let $U\subseteq Y$ be an open set. Let $\{U_i\;|\;i\in I\}$ be an open cover of $U$. Suppose that $s_i\in f_\ast\mF(U_i)=\mF(f^{-1}(U_i))$ is such that the images of $s_i$ and $s_j$ in $f_\ast\mF(U_i\cap U_j)=\mF(f^{-1}(U_i\cap U_j))$ agree for all $i,j\in I$. Since $\{f^{-1}(U_i)\}$ is an open cover of $f^{-1}(U)$ and $\mF$ is a sheaf on $X$, we conclude that there exists $s\in\mF(f^{-1}(U))=f_\ast\mF(U)$ such that the image of $s$ in $\mF(f^{-1}(U_i))=f_\ast\mF(U_i)$ is $s_i$ for all $i\in I$. Thus the gluing axiom is satisfied. 
\end{proof}
\end{prp}

\begin{lmm}{}{}\\
Let $Y$ be a space. Let $U\subseteq Y$ be an open set. Let $\iota:U\hookrightarrow Y$ be the inclusion map. Let $\mC$ be an algebraic category. Let $\mG$ be a $\mC$-sheaf on $Y$. Then there is a natural isomorphism $$\iota^{-1}\mG\cong\mG|_U$$
\begin{proof}
Write $\iota^{-1}\mG_\text{pre}$ the inverse image sheaf before sheafification. Recall that before sheafification, for $W\subseteq U$ an open set we have $(\iota^{-1}\mG_\text{pre})(W)=\colim_{V\supseteq W}\mG(V)$. Since $W$ is the initial object in the category of open sets containing $W$, we have that he colimit is equal to the terminal object. Hence $(\iota^{-1}\mG_\text{pre})(W)=\mG(W)$. Since $\mG$ is already a sheaf, sheafification gives a natural isomorphism $\iota^{-1}\mG)\cong\iota^{-1}\mG_\text{pre}$ and the latter is equal to $\mG|_U$. 
\end{proof}
\end{lmm}

Let $X,Y$ be spaces. Let $f:X\to Y$ be a map. Let $\mC$ be an algebraic category. Then $\iota^{-1}:\mC(Y)\to\mC(X)$ is an exact functor. 
\end{prp}

\subsection{Operads as Objects with Compatibility Diagrams}
\begin{defn}{Symmetric Operads}{}\\
Let $\mC$ be a symmetric monoidal category. An operad consists on $\mC$ consists of the following data. 
\begin{itemize}
\item Objects $F(n)\in\mC$ for each $n\in\N$. 
\item Right action of the symmetric group $\rho_n:S_n\to\Hom_\mC(F(n),F(n))$. 
\item A unit $e:I\to F(1)$. 
\item Composition operations $\gamma:F(k)\otimes F(n_1)\otimes\cdots\otimes F(n_k)\to F(n_1+\cdots+n_k)$. 
\end{itemize}
such that the following diagram commutes: 
\begin{itemize}
\item Associativity: \\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	{F(k)\otimes\bigotimes_{i=1}^k\left(F(n_i)\otimes\bigotimes_{j=1}^{n_i}F(m_{ij})\right)} & {F\left(\sum_{i=1}^kn_i\right)\otimes\bigotimes_{j=1}^{\sum_{i=1}^kn_i}F(m_{ij})} \\
	{F(k)\otimes\bigotimes_{i=1}^kF\left(\sum_{j=1}^{n_i}m_{ij}\right)} & {F\left(\sum_{i=1}^k\sum_{j=1}^{n_i}m_{ij}\right)}
	\arrow["{\gamma\otimes\text{id}}", from=1-1, to=1-2]
	\arrow["{\text{id}\otimes\gamma}"', from=1-1, to=2-1]
	\arrow["\gamma", from=1-2, to=2-2]
	\arrow["\gamma"', from=2-1, to=2-2]
\end{tikzcd}} \\
\item Unit: The following diagram commutes \\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	{F(n)} & {F(1)\otimes F(n)} \\
	{F(n)\otimes F(1)^{\otimes n}} & {F(n)}
	\arrow["{e\otimes\text{id}_{F(n)}}", from=1-1, to=1-2]
	\arrow["{\text{id}\otimes e^{\otimes n}}"', from=1-1, to=2-1]
	\arrow["{\text{id}_{F(n)}}"{description}, from=1-1, to=2-2]
	\arrow["\gamma", from=1-2, to=2-2]
	\arrow["\gamma"', from=2-1, to=2-2]
\end{tikzcd}} \\
\item Compatibility with the symmetric group action: 
\end{itemize}
\end{defn}

Here we think of $F(n)$ as the set of all the allowed operations that has $n$ inputs. 

\begin{defn}{Operad Maps}{}\\
Let $\mC$ be a symmetric monoidal category. Let $O,P$ be operads. A morphism of operads consists of morphism $f_n\in\Hom_\mC(O(n),P(n))$ such that the following are true. 
\begin{itemize}
\item Associativity: \\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	{O(k)\otimes\bigotimes_{i=1}^k O(n_i)} & {P(k)\otimes\bigotimes_{i=1}^k P(n_i)} \\
	{O\left(\sum_{i=1}^kn_i\right)} & {P\left(\sum_{i=1}^kn_i\right)}
	\arrow["f", from=1-1, to=1-2]
	\arrow["\gamma"', from=1-1, to=2-1]
	\arrow["\gamma", from=1-2, to=2-2]
	\arrow["f"', from=2-1, to=2-2]
\end{tikzcd}} \\
\item Unit: The following diagram commutes \\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	I & {O(1)} \\
	& {P(1)}
	\arrow["e", from=1-1, to=1-2]
	\arrow["e"', from=1-1, to=2-2]
	\arrow["f", from=1-2, to=2-2]
\end{tikzcd}} \\
\item Compatibility with the symmetric group action: 
\end{itemize}
\end{defn}

\begin{defn}{Algebras over Operads}{}\\
Let $\mC$ be a symmetric monoidal category. Let $O$ be an operad on $\mC$. An algebra over $O$ consists of the following data. 
\begin{itemize}
\item An object $X\in\mC$. 
\item A collection of maps $\lambda:O(k)\otimes X^{\otimes k}\to X$
\end{itemize}
such that the following diagram commutes: 
\begin{itemize}
\item Associativity: \\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	{O(k)\otimes\bigotimes_{i=1}^k\left(O(n_i)\otimes V^{\otimes n_i}\right)} & {O\left(\sum_{i=1}^kn_i\right)\otimes V^{\otimes\sum_{i=1}^kn_i}} \\
	{O(k)\otimes V^{\otimes k}} & V
	\arrow["{\gamma\otimes\text{id}}", from=1-1, to=1-2]
	\arrow["{\text{id}\otimes\lambda}"', from=1-1, to=2-1]
	\arrow["\lambda", from=1-2, to=2-2]
	\arrow["\lambda"', from=2-1, to=2-2]
\end{tikzcd}} \\
\item Unit: The following diagram commutes \\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	V & {O(1)\otimes V} \\
	& V
	\arrow["{e\otimes\text{id}_V}", from=1-1, to=1-2]
	\arrow["{\text{id}}"', from=1-1, to=2-2]
	\arrow["\lambda", from=1-2, to=2-2]
\end{tikzcd}} \\
\item Compatibility with the symmetric group action: 
\end{itemize}
\end{defn}

\begin{defn}{The Canonical Operad of an Object}{}\\
Let $\mC$ be a symmetric monoidal category. Let $X\in\mC$ be an object. Define the canonical operad $\text{Op}(X)$ by the following data. 
\begin{itemize}
\item $\text{Op}(X)(n)=\Hom_\mC(X^{\otimes n},X)$. 
\item The symmetric group action is induced by shuffling the order of the copies of $X$ in $X^{\otimes n}$. 
\item The unit is the unique map $I\to\Hom_\mC(X,X)$. 
\item Composition is given by composition in $\mC$. 
\end{itemize}
\end{defn}

\begin{prp}{}{}\\
Let $\mC$ be a symmetric monoidal category. Let $O$ be an operad. Let $X\in\mC$ be an object. Then the data of an algebra $X$ over $O$ is equivalent to a monoid map $O\to\text{Op}(X)$. 
\end{prp}

\subsection{$\S$-Modules}
\begin{defn}{$\S$-Modules}{}\\
Let $R$ be a commutative ring. An $\S$-module is a collection $(M_n)_{n\in\N}$ where each $M_n$ is a right $R[S_n]$-module. 
\end{defn}

\begin{defn}{Morphism of $\S$-Modules}{}\\
Let $R$ be a commutative ring. Let $M$ and $N$ be $\S$-modules. Let $f_n:M_n\to N_n$ be a function. We say that $(f_n)_{n\in\N}$ is a morphism of $\S$-modules if each $f_n$ is a $k[S_n]$-module homomorphism. 
\end{defn}

\begin{defn}{The Associated Schur Functor}{}\\
Let $R$ be a commutative ring. Let $M$ be an $\S$-module. Define the associated Schur functor $\widetilde{M}:\bold{Mod}_R\to\bold{Mod}_R$ by $$\widetilde{M}(V)=\bigoplus_{n=0}^\infty M_n\otimes_{R[S_n]} V^{\otimes n}$$
\end{defn}

Induced map. 

Direct sum, tensor product, composition of functors. 

Category of $\S$-modules is monoidal. 

Symmetric operad is equivalent to monoid in the category of $\S$-modules. 

Algebraic Operads are maps from Schur functor to the base module.
