\section{More on Simplicial Homology}
\subsection{Relation to Singular Homology}
\begin{thm}{}{} Let $X$ be a topological space endowed with a $\Delta$-complex structure $(T,\abs{T}\cong X)$. The induced map $H_n^\Delta(X)\to H_n(X)$ by the map $s:\Delta^n\to X$ for each $s\in T^n$ is an isomorphism. 
\begin{proof}
Firstly, every $n$-simplex $s\in T$ induces a canonical continuous map $s:\Delta^n\to X$. This extends to a homomorphism $\Delta_n(T)\to C_n(X)$ and so to a chain map and descends to homology. \\

Denote $T^k$ the $\Delta$-sets of simplices in $T$ of dimension at most $k$. Define $\Delta_\bullet(T^k,T^{k-1})=\Delta_\bullet(T^k)/\Delta_\bullet(T^{k-1})$ and accordingly, $H_n(\Delta_\bullet(T^k,T^{k-1}))$. Then the short exact sequence of chain complexes \\
\adjustbox{scale=1,center}{\begin{tikzcd}
	0 & {\Delta_\bullet(T^{k-1})} & {\Delta_\bullet(T^k)} & {\Delta_\bullet(T^k,T^{k-1})} & 0
	\arrow[from=1-1, to=1-2]
	\arrow[from=1-2, to=1-3]
	\arrow[from=1-3, to=1-4]
	\arrow[from=1-4, to=1-5]
\end{tikzcd}} \\
Together with the inclusion maps and by naturality in theorem 1.4.3, give the following: \\
\adjustbox{scale=0.8,center}{\begin{tikzcd}
	\cdots & {H_{n+1}(T^k,T^{k-1})} & {H_n(T^{k-1})} & {H_n(T^{k-1})} & {H_n(T^k,T^{k-1})} & {H_{n-1}(T^{k-1})} & \cdots \\
	\cdots & {H_{n+1}(\abs{T^k},\abs{T^{k-1}})} & {H_n(\abs{T^{k-1}})} & {H_n(\abs{T^{k-1}})} & {H_n(\abs{T^k},\abs{T^{k-1}})} & {H_{n-1}(\abs{T^{k-1}})} & \cdots
	\arrow[from=1-1, to=1-2]
	\arrow[from=1-2, to=1-3]
	\arrow[from=1-2, to=2-2]
	\arrow[from=1-3, to=1-4]
	\arrow[from=1-3, to=2-3]
	\arrow[from=1-4, to=1-5]
	\arrow[from=1-4, to=2-4]
	\arrow[from=1-5, to=1-6]
	\arrow[from=1-5, to=2-5]
	\arrow[from=1-6, to=1-7]
	\arrow[from=1-6, to=2-6]
	\arrow[from=2-1, to=2-2]
	\arrow[from=2-2, to=2-3]
	\arrow[from=2-3, to=2-4]
	\arrow[from=2-4, to=2-5]
	\arrow[from=2-5, to=2-6]
	\arrow[from=2-6, to=2-7]
\end{tikzcd}} \\
When $k=0$, $\abs{T^0}$ is a discrete topological space on the set $T^0$, and the map $H_n(T^0)\to H_n(\abs{T^0})$ is an isomorphism. By induction and the five lemma, the middle map is an isomorphism provided that the first and fourth maps are isomorphisms. \\

Note that $\Delta_\bullet(T^{k-1})$ is a chain complex in degrees $k-1$, $k-2$, down to $0$. And $\Delta_\bullet(T^k)$ is the same chain complex with an additional term $\Z T_k$ in degree $k$. It follows that $\Delta_\bullet(T^k,T^{k-1})$ is the chain complex with $\Z T^k$ concentrated in degree $k$. In particular, we have that $$H_n(T^k,T^{k-1})=\begin{cases}
\Z T^k & \text{ if } n=k\\
0 & \text{ otherwise }
\end{cases}$$
By construction of the geometric realization, we have a homeomorphism $$\frac{\abs{T^k}}{\abs{T^{k-1}}}\cong\frac{\Delta^k\times T^k}{\partial\Delta^k\times T^k}\cong\bigvee_{T_k}\frac{\Delta^k}{\partial\Delta^k}$$ Using lemma 6.2.4 and lemma 6.2.5, we have that $$H_n(\abs{T^k},\abs{T^{k-1}})\cong\widetilde{H_n}(\abs{T^k}/\abs{T^{k-1}})\cong\begin{cases}
\Z T_k & \text{ if } n=k\\
0 & \text{ otherwisev }
\end{cases}$$
with generators corresponding to elements $s\in T^k$ via the relative cycles $s:\Delta^k\to\abs{T^k}$. It follows that an isomorphism occurs in the first and fourth position of the long exact sequence of relative homology groups. \\

If $T=T^n$ for some $n\in\N$, we are done. \\

Continuing the proof, let $z\in Z_n(T)$ be an $n$-cycle whose image in $H_n(\abs{T})$ vanishes. Then there exists $\tau\in C_{n+1}(\abs{T})$ with $\partial\tau=z$. Now since $\abs{T}$ is the geometric realization, we must have every compact subspace of $\abs{T}$ lying in some $\abs{T^k}$. Hence $\tau\in C_{n+1}(\abs{T^k})$ for some $k>n$. We deduce that $z$ maps to $0$ in $H_n(\abs{T^k})$. By the previous argument, we have taht $z=0\in H_n(T^k)=H_n(T)$. This shows injectivity of $H_n(T)\to H_n(\abs{T})$. \\

Similarly, let $\sigma\in Z_n(T)$ be an $n$-cycle. We have seen that $\sigma\in Z_n(\abs{T^k})$ for some $k>n$ and by the argument above, $[\sigma]$ comes from $H_n(T^k)=H_n(T)$. This shows surjectivity of $H_n(T)\to H_n(\abs{T})$ and so we are done. 
\end{proof}
\end{thm}

We can finally show that simplicial homology is independent of the choice of $\Delta$-complex structure. 

\begin{crl}{}{} The simplicial homology $H_\bullet^\Delta(X)$ depends on $X$ only and not on the $\Delta$-complex structure. 
\begin{proof}
Indeed the geometric realization of any two $\Delta$-complex structure is isomorphic to the same singular homology group. 
\end{proof}
\end{crl}

\begin{crl}{}{} Suppose $X$ has a $\Delta$-complex structure with simplicies in dimension $\leq k$ only. Then $H_n(X)=0$ for all $n>k$. 
\begin{proof}
Direct since singular homology coincides with simplicial homology. 
\end{proof}
\end{crl}

\section{Polyhedrons and Polytopes}
\subsection{Convex Geometry}
\begin{defn}{Convex Sets}{}\\
Let $C\subseteq\R^n$. We say that $C$ is convex if for all $x,y\in C$ and $t\in[0,1]$, we have $(1-t)x+ty\in C$. 
\end{defn}

\begin{defn}{Convex Hull}{}\\
Let $S\subseteq\R^n$ be a subset. Define the convex hull of $S$ to be $$\text{Conv}(S)=\left\{\sum_{k=1}^n\lambda_kx_k\;\bigg{|}\;x_k\in S\text{ and }\sum_{k=1}^n\lambda_k=1\right\}$$
\end{defn}

\begin{prp}{}{}\\
Let $S\subseteq\R^n$ be a subset. Then the following are sets are equal. 
\begin{itemize}
\item The convex hull $\text{Conv}(S)$. 
\item $\bigcap_{C\subseteq\R^n\text{ is convex}}C$. 
\item The smallest convex set containing $S$. 
\end{itemize}
\end{prp}

\subsection{Polyhedrons and its Properties}
Polytopes exists more generically in Euclidean space. We lose nothing discussing polytopes in $\R^n$ since Euclidean spaces maybe identified with $\R^n$ canonically. 

\begin{defn}{Halfspaces}{}\\
A halfspace of $\R^n$ is a subset of $\R^n$ of the form $$\{x\in\R^n\;|\;\langle a,x\rangle+b\geq 0\}$$ for some $a,b\in\R^n$. 
\end{defn}

\begin{defn}{Polyhedrons}{}\\
A polyhedron in $\R^n$ is an intersection of finitely many halfspaces. 
\end{defn}

\begin{defn}{Polytopes}{}\\
A polytope in $\R^n$ is a polyhedron that is bounded. 
\end{defn}

\begin{prp}{}{}\\
Let $P\subseteq\R^n$ be a subset. Then $P$ is a polytope if and only if there exists a finite subset $S\subseteq\R^n$ such that $P=\text{Conv}(S)$. 
\end{prp}

\begin{defn}{Dimension}{}\\
Let $P\subseteq\R^n$ be a polytope. Define the dimension of $P$ to be the dimension of the smallest affine subspace containing $P$. 
\end{defn}

\begin{defn}{Affinely Isomorphic}{}\\
Let $P_1\subseteq\R^{n_1}$ and $P_2\subseteq\R^{n_2}$ be polytopes. We say that $P_1$ and $P_2$ are affinely isomorphic if there exists an affine transformation $f:\R^{n_1}\to\R^{n_2}$ such that $f|_{P_1}$ is a bijection. 
\end{defn}

Note that $f$ itself need not be an affine isomorphism (i.e. an isomorphism of affine spaces). 

\subsection{The Faces of a Polyhedron}
\begin{defn}{Supporting Hyperplane}{}\\
Let $P\subseteq\R^n$ be a polyhedron. Let $H\subseteq\R^n$ be a hyperplane. We say that $H$ is a supporting hyperplane for $P$ if the following are true. 
\begin{itemize}
\item $P\cap H\neq\emptyset$. 
\item $P$ lies entirely in one of the halfspaces defined by $H$. 
\end{itemize}
\end{defn}

\begin{defn}{Faces}{}\\
Let $P\subseteq\R^n$ be a polyhedron. A face of $P$ is a subset of $P$ defined by $$F=P\cap H$$ where $H$ is a supporting hyperplane for $P$. 
\end{defn}

\begin{lmm}{}{}\\
Let $P\subseteq\R^n$ be a polyhedron. Let $F\subseteq P$ be a face. Then $F$ is a polytope. 
\end{lmm}

\begin{defn}{Vertices, Edges and Facets}{}\\
Let $P\subseteq\R^n$ be a polyhedron. Let $F\subseteq P$ be a face. 
\begin{itemize}
\item We say that $F$ is a vertex if $\dim(F)=0$. 
\item We say that $F$ is an edge if $\dim(F)=1$. 
\item We say that $F$ is a facet if $\dim(F)=\dim(P)-1$. 
\end{itemize}
\end{defn}

\begin{defn}{The Face Poset}{}\\
Let $P\subseteq\R^n$ be a polyhedron. Define the face poset $\mF(P)$ of $P$ to be the set of faces of $P$ together with inclusion. 
\end{defn}

\begin{defn}{The Face Vector}{}\\
Let $P\subseteq\R^n$ be a polytope of dimension $d$. Define the face vector $(f_0,\dots,f_{d-1})$ of $S$ by the formula $$f_i=\text{The number of }i\text{-dimensional faces of }P$$ for $0\leq i\leq d$ and by convention, $f_{-1}=f_d=1$. 
\end{defn}

\begin{lmm}{Euler's Formula}{}\\
Let $P\subseteq\R^n$ be a polytope of dimension $d$. Then we have $$\sum_{i=0}^{d-1}(-1)^if_i=1+(-1)^d$$
\end{lmm}

\begin{defn}{Combinatorially Equivalent}{}\\
Let $P_1\subseteq\R^{n_1}$ and $P_2\subseteq\R^{n_2}$ be polytopes. We say that $P_1$ and $P_2$ are combinatorially equivalent if there exists a inclusion preserving bijection between $\mF(P_1)$ and $\mF(P_2)$. In this case we write $P_1\simeq P_2$. 
\end{defn}

\section{Duality in Polytopes}
\subsection{Polar Sets}
\begin{defn}{Polar Sets}{}\\
Let $S\subseteq\R^n$ be a subset. Define the polar set of $S$ to be $$S^\ast=\{f\in(\R^n)^\ast\;|\;f(x)\leq 1\text{ for all }x\in S\}$$
\end{defn}

\begin{prp}{}{}\\
Let $P\subseteq\R^n$ be a polytope. If $0\in P$, then $P^\ast$ is a polytope. 
\end{prp}

\begin{eg}{}{}\\
The polar set of the cube $C^3=\{(a,b,c)\in\R^3\;|\;-1\leq a,b,c\leq 1\}$ is the octahedron $$(C^3)^\ast=\{(x,y,z)\in(\R^3)^\ast\;|\;\pm x+\pm y+\pm z\leq 1\}$$
\end{eg}

$P=(P^\ast)^\ast$. 

$\mF(P)^\text{op}\cong\mF(P^\ast)$. 

\subsection{Simplicial Polytopes}
\begin{defn}{Simplicial Polytopes}{}\\
Let $P\subseteq\R^n$ be a polytope. We say that $P$ is a simplicial polytope if all its facets are simplicies. 
\end{defn}

\begin{lmm}{}{}\\
Let $P\subseteq\R^n$ be a polytope. Then $\partial P$ is a simplicial complex. 
\end{lmm}

\begin{prp}{The Dehn-Sommerville Equation}{}\\
Let $P\subseteq\R^n$ be a simplicial polytope of dimension $d$. Then the following are true: $$\sum_{j=k}^{d-1}(-1)^j\binom{j+1}{k+1}f_j=(-1)^{d-1}f_k$$ for $-1\leq k\leq d-2$. 
\end{prp}

In particular, $\partial P$ is a simplicial complex. The notion of face vector coincides, and they enjoy both the Dehn-Sommerville equation and its relation with the $h$-vector. 

\subsection{Simple and Simplicial Polytopes}
\begin{defn}{Simple Polytopes}{}\\
Let $P\subseteq\R^n$ be a polytope. We say that $P$ is a simple polytope if for all vertices $v\in P$, there exists $n$ edges $e_1,\dots,e_n\subseteq P$ such that $v\in\partial e_i$ for all $i$. 
\end{defn}

\begin{lmm}{}{}\\
Let $P_1\subseteq\R^n$ and $P_2\subseteq\R^m$ be simple polytopes. Then the Cartesian product $P_1\times P_2$ is a simple polytope. 
\end{lmm}

\subsection{The Duality Between Simple and Simplicial Polytopes}
\begin{prp}{}{}\\
Let $P\subseteq\R^n$. Then $P$ is simple if and only $P^\ast$ is simplicial. 
\end{prp}

Duality of $d$-dimensional faces with $(n-d)$-dimensional faces. 

\begin{prp}{}{}\\
Let $P\subseteq\R^n$ be a polytope. If $P$ is both simple and simplicial, then $P$ belongs to one of the following types of polytopes. 
\begin{itemize}
\item $n=2$ (and so $P$ is a polygon). 
\item $P$ is a simplex. 
\end{itemize}
\end{prp}

\subsection{Nerve Complex}
\begin{defn}{The Nerve Complex}{}\\
Let $P\subseteq\R^n$ be a simple polytope. Define the nerve complex of $P$ to be the simplicial complex $$\mK_P=\partial(P^\ast)$$
\end{defn}

\begin{lmm}{}{}\\
Let $P\subseteq\R^n$ be a simple polytope. Then $\mK_P$ is a triangulation of $S^{n-1}$. 
\end{lmm}

\begin{lmm}{}{}\\
Let $P_1\subseteq\R^n$ and $P_2\subseteq\R^m$ be simple polytopes. Then we have $$\mK_{P_1\times P_2}\cong\mK_{P_1}\ast\mK_{P_2}$$
\end{lmm}

\section{Piecewise Linear Topology}
\subsection{Relation between Polyhedra and Simplicial Complexes}
\begin{defn}{Locally Finite Collections}{}\\
Let $X$ be a space. Let $P\subseteq\mP(X)$ be a collection of subsets. We say that $P$ is locally finite if for each $p\in X$, there exists a neighbourhood $U\subseteq X$ of $p$ such that the set $$\{V\in P\;|\;U\cap V\neq\emptyset\}$$ is finite. 
\end{defn}

\begin{prp}{}{}\\
Let $P$ be a polyhedra. Then $P$ is a locally finite union of simplexes. 
\end{prp}

Thus if $P$ is a polytope, then $P$ has the underlying structure of a simplicial complex. 

\subsection{Piecewise Linear Maps}
\begin{defn}{Piecewise Linear Maps between Polyhedra}{}\\
Let $P,Q$ be polyhedra. Let $f:P\to Q$ be a function. We say that $f$ is piecewise linear if there exists a locally finite decomposition into polyhedra $P=\bigcup_{\alpha\in I}P_\alpha$ such that for each $\alpha\in I$, the restriction $f|_{P_\alpha}$ is affine linear. 
\end{defn}

\begin{prp}{}{}\\
Let $P,Q$ be polyhedra. Let $f:P\to Q$ be a function. Then the following are equivalent. 
\begin{itemize}
\item $f$ is piecewise linear. 
\item For each $p\in P$, there exists $L\subseteq P$ such that $f(\lambda p+\mu x)=\lambda f(p)+\mu f(x)$ for all $\lambda,\mu\geq 0$ and $\lambda+\mu=1$ and $x\in L$. 
\item $\Gamma(f)=\{(x,f(x))\;|\;x\in P\}$ is a polyhedron. 
\item There exists locally finite decompositions of $P$ and $Q$ into simplexes such that $f$ sends simplexes to simplexes and $f$ restricted to each simplex is affine linear. 
\end{itemize}
\end{prp}

\begin{defn}{PL Homeomorphism}{}\\
Let $P,Q$ be polyhedra. Let $f:P\to Q$ be a piecewise linear function. We say that $f$ is a PL homeomorphism if there exists a piecewise linear function $g:Q\to P$ such that $$g\circ f=\text{id}_P\;\;\;\text{ and }\;\;\;f\circ g=\text{id}_Q$$
\end{defn}

\begin{lmm}{}{}\\
Let $P,Q$ be polyhedra. Let $f:P\to Q$ be a piecewise linear function. If $f$ is a homeomorphism, then $f$ is a piecewise linear homeomorphism. 
\end{lmm}

Motivated by the equivalent definitions of piecewise linear maps between polyhedra, we define piecewise linear maps between spaces in general. 

\begin{defn}{Piecewise Linear Maps in General}{}\\
Let $X,Y$ be spaces. Let $f:X\to Y$ be a function. We say that $f$ is piecewise linear if there exists locally finite triangulations of $X$ and $Y$ such that $f$ restricted to each simplex is affine linear. 
\end{defn}

\begin{lmm}{}{}\\
Let $S,T$ be simplicial complexes. Let $f:V(S)\to V(T)$ be a simplicial map. Then $\abs{f}:\abs{S}\to\abs{T}$ is piecewise linear. 
\end{lmm}

\begin{prp}{}{}\\
Let $S,T$ be simplicial complexes. Let $f:\abs{S}\to\abs{T}$ be a function. Then $f$ is piecewise linear if and only if there exists subdivisions $S'$ of $S$ and $T'$ of $T$ such that $f:\abs{S'}\to\abs{T'}$ is simplicial. 
\end{prp}

\subsection{Piecewise Linear Manifolds}
\begin{prp}{}{}\\
The set $$\Gamma=\{f:U\to V\;|\;U,V\subseteq\R^n\text{ are open }, f\text{ is a PL homeomorphism }\}$$ defines a pseudogroup of transformations on $\R^n$. 
\end{prp}

\begin{defn}{Piecewise Linear Manifolds}{}\\
A piecewise linear manifold is a manifold whose pseudogroup of transformations is given by $\Gamma$. 
\end{defn}

\subsection{Triangulated Manifolds}
\begin{defn}{Triangulated Manifolds}{}\\
Let $M$ be a topological manifold. We say that $M$ is a triangulated manifold if $M$ is triangulated as a topological space. 
\end{defn}

\begin{prp}{}{}\\
Let $X$ be a piecewise linear manifold. Then $X$ is a triangulated manifold. 
\end{prp}

\begin{prp}{}{}\\
Let $M$ be a triangulated manifold triangulated by the simplicial complex $K$. Then the following are equivalent. 
\begin{itemize}
\item $M$ is a PL manifold. 
\item For every $\emptyset\neq\sigma\in K$, we have that $\text{lk}(\sigma)$ is PL homeomorphic to $\abs{\Delta^d}$ where $d=\dim(M)-\abs{\sigma}$. 
\item For every $p\in V(K)$, we have that $\text{st}(p)$ is PL homeomorphic to $\abs{\Delta^{\dim(M)}}$. 
\end{itemize}
\end{prp}

The Hauptvermutung asks whether any two homeomorphic triangulated spaces are PL homemorphic (i.e. if $S,T$ are simplicial complexes and $\abs{S}$ is homeomorphic to $\abs{T}$, is there a PL homeomorphism between $\abs{S}$ and $\abs{T}$?). This is shown to be False by Milnor using Reidemeister torsion. \\

The manifold Hauptvermutung asks whether any two homeomorphic triangulated manifolds are PL homemorphic. This is also show to be false using an obstruction in cohomology. 

\begin{prp}{}{} Every finite CW complex is homotopy equivalent to a topological manifold. 
\end{prp}

\subsection{Pseudo-Manifolds}
\begin{defn}{Pseudo-Manifolds From Simplicial Complexes}{}\\
Let $\mK$ be a simplicial complex. We say that $\mK$ is a pseudo-manifold of dimension $n$ if the following are true. 
\begin{itemize}
\item All maximal simplicies of $\mK$ are of dimension $n$. 
\item Each $(n-1)$-simplex of $\mK$ is the face of two $n$-simplicies of $\mK$. 
\item For any two $n$-simplicies $\sigma,\sigma'\in\mK$, there exists $n$-simplicies $\sigma=\sigma_0,\dots,\sigma_k=\sigma'$ such that $\sigma_i\cap\sigma_{i+1}$ is an $(n-1)$-simplex for $0\leq i\leq k-1$. 
\end{itemize}
\end{defn}

\begin{defn}{Pseudo-Manifolds in General}{}\\
Let $X$ be a space. We say that $X$ is a pseudo-manifold if $X$ is triangulated by a simplicial complex $\mK$ that is a pseudo-manifold. 
\end{defn}

\begin{lmm}{}{}\\
Let $X$ be a pseudo-manifold of dimension $n$. Then $H_n(X)\cong\Z$ or $0$. 
\end{lmm}

\begin{lmm}{}{}\\
Let $\mK$ be a triangulation of $S^n$. Then the following are true. 
\begin{itemize}
\item $\mK$ is a pseudo-manifold. 
\item If $\mK$ is finite, then the generator of $H_n(\mK)$ is represented by the sum of $n$-simplicies of $\mK$. 
\end{itemize}
\end{lmm}

\section{Cones and Fans}
\subsection{Cones and its Properties}
\begin{defn}{Cones}{}\\
Let $C\subseteq\R^n$. We say that $C$ is a cone if $x\in C$ implies that $sx\in C$ for all $s\geq 0$. 
\end{defn}

\begin{defn}{Polyhedral Cones}{}\\
A cone in $\R^n$ is a subset $C\subseteq\R^n$ such that if $c_1,\dots,c_n\in C$, then $\sum_{i=1}^n\lambda_ic_i\in C$ for all $\lambda_1,\dots,\lambda_n\geq 0$. 
\end{defn}

\begin{defn}{Conical Hull}{}\\
Let $S\subseteq\R^n$ be finite. Define the conical hull of $S$ to be $$\text{Cone}(S)=\left\{\sum_{s\in S}\lambda_ss\;\bigg{|}\;\lambda_s\geq 0\right\}$$
\end{defn}

\begin{prp}{}{}\\
Let $C\subseteq\R^n$ be a subset. Then the following are equivalent. 
\begin{itemize}
\item $C$ is a polyhedral cone. 
\item There exists a finite subset $S\subseteq\R^n$ such that $C=\text{Cone}(S)$. 
\item $C$ is the intersection of finitely many halfspaces containing $0$ in their boundary. 
\end{itemize}
\end{prp}

Therefore cones are also polyhedrons. 

\begin{defn}{Dimension of a Cone}{}\\
Let $C\subseteq\R^n$ be a cone. Define the dimension of $C$ to be the dimension of the smallest affine subspace containing $C$. 
\end{defn}

\begin{defn}{Rational Cones}{}\\
Let $C\subseteq\R^n$ be a cone. We say that $C$ is rational if there exists a finite set $S\subseteq\Z^n$ such that $C=\text{Cone}(S)$. 
\end{defn}

\begin{defn}{Regular Cones}{}\\
Let $C\subseteq\R^n$ be a cone. We say that $C$ is regular if there exists a finite set $S\subseteq\Z^n$ such that the following are true. 
\begin{itemize}
\item $C=\text{Cone}(S)$. 
\item Elements of $S$ are $\Z$-linearly independent. 
\end{itemize}
\end{defn}

\begin{defn}{Simplicial Cones}{}\\
Let $C\subseteq\R^n$ be a cone. We say that $C$ is simplicial if there exists a finite set $S\subseteq\R^n$ such that the following are true. 
\begin{itemize}
\item $C=\text{Cone}(S)$. 
\item Elements of $S$ are $\R$-linearly independent. 
\end{itemize}
\end{defn}

\subsection{The Dual Cone}
\begin{defn}{The Dual of a Cone}{}\\
Let $C\in\R^n$ be a cone. Define the dual of $C$ to be $$C^\vee=\{x\in\R^n\;|\;\langle x,c\rangle\geq 0\text{ for all }c\in C\}$$
\end{defn}

\begin{defn}{Associated Hyperplane and Halfspace of a Linear Functional}{}\\
Let $\phi\in(\R^n)^\ast$. Define the associated hyperplane of $\phi$ to be $$H_\phi=\{x\in\R^n\;|\;\phi(x)=0\}$$ Define the associated halfspace of $\phi$ to be $$H_\phi^+=\{x\in\R^n\;|\;\phi(x)\geq 0\}$$
\end{defn}

\begin{lmm}{}{}\\
Let $C\in\R^n$ be a cone. Let $H$ be a hyperplane in $\R^n$. Then $H$ is a supporting hyperplane of $C$ if and only if there exists $\phi\in C^\vee$ such that $H=H_\phi$. 
\end{lmm}

\begin{lmm}{}{}\\
Let $C\in\R^n$ be a polyhedral cone. Then the following are true. 
\begin{itemize}
\item $C^\vee$ is a polyhedral cone.

\begin{enumerate}

For $I\subseteq\{1,\dots,n\}$, the condition that $\abs{z_j}^2=1$ for $j\in\{1,\dots,n\}\setminus I$ means that for those coordinates in $D^n$, they only consist of the boundary which is $S^1$. \item There are natural isomorphisms of cochain complexes $$\bigoplus_{J\subseteq\{1,\dots,n\}}C^{k-\abs{J}-1}(\mK_J;\Z)\cong C^\ast(\mZ_\mK;\Z)$$ given by $$\alpha_L\mapsto\varepsilon(L,J)\chi(J\setminus L,L)$$ where $\alpha_L$ is the cochain basis of Hochster's formula given by the simplex $L\subseteq J$ and $$\varepsilon(L,J)=\prod_{j\in L}\varepsilon(j,J)$$ and $\varepsilon(j,J)=(-1)^{r-1}$ if $j$ is in the $r$th position of $J$. 
\item There above natural isomorphisms induces natural isomorphisms $$H^{-i}(C^{-\ast,2j}(\mZ_\mK;\Z))\cong\bigoplus_{\substack{J\subseteq\{1,\dots,n\}\\\abs{J}=j}}\widetilde{H}^{j-i-1}(\mK_J;\Z)$$ where by convention we set $\widetilde{H}^{-1}(\mK_\emptyset)=\Z$. 
\item The above natural isomorphisms induces natural isomorphisms $$H^k(\mZ_\mK;\Z)\cong\bigoplus_{J\subseteq\{1,\dots,n\}}\widetilde{H}^{k-\abs{J}-1}(\mK_J;\Z)$$
\end{itemize}
\end{prp}

We are missing a description of the cup product in terms of Hochster's formula. It is given by the following definition. 

\begin{defn}{Ring Structure in Hochster's Formula}{}\\
Let $\mK$ be a simplicial complex on $\{1,\dots,n\}$. Define a ring structure in $\bigoplus_{J\subseteq\{1,\dots,n\}}\widetilde{H}^\ast(\mK_\mJ;\Z)$ as follows. Let $\alpha_L$ be the cochain basis element in $C^p(\mK_I;\Z)$ corresponding to the simplex $L\subseteq I$. The ring structure is induced by the map $$C^p(\mK_I;\Z)\otimes_\Z C^q(\mK_\mJ;\Z)\to C^{p+q+1}(\mK_{I\cup J})$$ defined by $$\alpha_L\otimes\alpha_M\mapsto\begin{cases}
\varepsilon(L,I)\varepsilon(M,J)\varepsilon(L\cup M,I\cup J)\alpha_{L\amalg M} & \text{ if }I\cap J=\emptyset\\
0 & \text{ otherwise }
\end{cases}$$
where $\varepsilon(L,I)=\prod_{j\in L}\varepsilon(j,J)$ and $\varepsilon(j,J)=(-1)^{r-1}$ if $j$ is in the $r$th position in $J$. 
\end{defn}

\begin{prp}{}{}\\
Let $\mK$ be a simplicial complex on $\{1,\dots,n\}$. Then the following are true. 
\begin{itemize}
\item The ring structure in Hochster's formula is equal to the relative Kunneth theorem for $$H^\ast(\abs{\mK_I\ast\mK_J})=H^\ast(\Sigma(\abs{\mK_I}\wedge\abs{\mK_J}))=\widetilde{H}^{\ast-1}(\abs{K_I}\times\abs{K_J},\abs{K_I\vee K_J})$$ via the inclusion $\mK_{I\amalg J}\hookrightarrow\mK_I\ast\mK_J$. 
\item The natural isomorphism in the above prp induces a natural isomorphism of rings $$H^\ast(\mZ_\mK;\Z)\cong\bigoplus_{J\subseteq\{1,\dots,n\}}\widetilde{H}^\ast(\mK_J;\Z)$$
\end{itemize}
\end{prp}

\subsection{Algebra, Coalgebra and Bialgebra}
Let $R$ be a commutative ring. Recall that an $R$-algebra is a ring $A$ together with a ring homomorphism $R\to A$. Equivalently, it is an $R$-module with multiplication that is associative and respects the $R$-module structure. 

\begin{prp}{}{}\\
Let $R$ be a commutative ring. Let $A\in\bold{Mod}_R$ be an $R$-module. Let $m:A\otimes_R A\to A$ and $u:R\to A$ be  $R$-module homomorphisms. Then $(A,m,u)$ is an associative algebra with multiplication given by $m$ and identity given $u(1)$ if and only if the following diagrams commute: \\~\\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	{(A\otimes_RA)\otimes_RA} & {A\otimes_RA} & A & {R\otimes_RA} & {A\otimes_RA} & {A\otimes R} \\
	{A\otimes_R(A\otimes_RA)} & {A\otimes_RA} &&& A
	\arrow["{m\otimes 1}", from=1-1, to=1-2]
	\arrow["\alpha"', from=1-1, to=2-1]
	\arrow["m", from=1-2, to=1-3]
	\arrow["{u\otimes\text{id}_A}", from=1-4, to=1-5]
	\arrow["\lambda"', from=1-4, to=2-5]
	\arrow["m"', from=1-5, to=2-5]
	\arrow["{\text{id}_A\otimes u}"', from=1-6, to=1-5]
	\arrow["\mu", from=1-6, to=2-5]
	\arrow["{1\otimes m}"', from=2-1, to=2-2]
	\arrow["m"', from=2-2, to=1-3]
\end{tikzcd}} \\
where $\alpha$ is the natural isomorphism from associativity of tensor products, and $\lambda,\mu$ is the natural isomorphism from the unitality of tensor products. 
\end{prp}

Equivalently, the above proposition is the same as saying $(A,m,u)$ is a monoid object in the monoidal category $(\bold{Mod}_R,\otimes_R,R)$. Motivated by the categorical description of an $R$-algebra, we define coalgebras as the formal dual of the above object in the following way. 

\begin{defn}{Coalgebras}{}\\
Let $R$ be a commutative ring. Let $C\in\bold{Mod}_R$ be an $R$-module. Let $\Delta:C\to C\otimes_RC$ and $\varepsilon:C\to R$ be $R$-module homomorphisms. We say that $(C,\Delta,\varepsilon)$ is a coalgebra if the following diagrams commute: \\~\\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	C & {C\otimes_RC} & {(C\otimes_RC)\otimes_RC} && C \\
	& {C\otimes_RC} & {C\otimes_R(C\otimes_RC)} & {R\otimes_RC} & {C\otimes C} & {C\otimes_RR}
	\arrow["\Delta", from=1-1, to=1-2]
	\arrow["\Delta"', from=1-1, to=2-2]
	\arrow["{\Delta\otimes 1}", from=1-2, to=1-3]
	\arrow["\alpha", from=1-3, to=2-3]
	\arrow["\lambda"', from=1-5, to=2-4]
	\arrow["\Delta", from=1-5, to=2-5]
	\arrow["\mu", from=1-5, to=2-6]
	\arrow["{1\otimes\Delta}"', from=2-2, to=2-3]
	\arrow["{\varepsilon\otimes\text{id}_A}", from=2-5, to=2-4]
	\arrow["{\text{id}_A\otimes\varepsilon}"', from=2-5, to=2-6]
\end{tikzcd}} \\
where $\alpha$ is the natural isomorphism from associativity of tensor products, and $\lambda,\mu$ is the natural isomorphism from the unitality of tensor products. 
\end{defn}

\begin{defn}{Bialgebras}{}\\
Let $R$ be a commutative ring. Let $B\in\bold{Mod}_R$ be an $R$-module. Let $m:B\otimes_RB\to B$, $u:R\to B$, $\Delta:B\to B\otimes_RB$ and $\varepsilon:B\to R$ be $R$-module homomorphisms. We say that $B$ is an $R$-bialgebra if the following are true. 
\begin{itemize}
\item $(B,m,u)$ is an $R$-algebra. 
\item $(B,\Delta,\varepsilon)$ is an $R$-coalgebra. 
\item Multiplication and comultiplication are compatible: \\~\\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	{B\otimes_RB} & B & {B\otimes_RB} \\
	{B\otimes_RB\otimes_RB\otimes_RB} && {B\otimes_RB\otimes_RB\otimes_RB}
	\arrow["m", from=1-1, to=1-2]
	\arrow["{\Delta\otimes\Delta}"', from=1-1, to=2-1]
	\arrow["\Delta", from=1-2, to=1-3]
	\arrow["{\text{id}\otimes\tau\otimes\text{id}}"', from=2-1, to=2-3]
	\arrow["{m\otimes m }"', from=2-3, to=1-3]
\end{tikzcd}} \\
where $\tau$ is the natural isomorphism from commutativity of tensor products. 
\item Multiplication and counit are compatible: \\~\\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	{B\otimes_RB} & B \\
	{R\otimes_RR} & R
	\arrow["m", from=1-1, to=1-2]
	\arrow["{\varepsilon\otimes\varepsilon}"', from=1-1, to=2-1]
	\arrow["\varepsilon", from=1-2, to=2-2]
	\arrow["\lambda"', from=2-1, to=2-2]
\end{tikzcd}} \\
where $\lambda$ is the natural isomorphism from unitality of tensor products. 
\item Comultiplication and unit are compatible: \\~\\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	R & B \\
	{R\otimes_RR} & {B\otimes_RB}
	\arrow["u", from=1-1, to=1-2]
	\arrow["\lambda"', from=1-1, to=2-1]
	\arrow["\Delta", from=1-2, to=2-2]
	\arrow["{\Delta\otimes\Delta}"', from=2-1, to=2-2]
\end{tikzcd}} \\
where $\lambda$ is the natural isomorphism from unitality of tensor products. 
\item Unit and counit are compatible: \\~\\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	R && R \\
	& B
	\arrow["{\text{id}}", from=1-1, to=1-3]
	\arrow["u"', from=1-1, to=2-2]
	\arrow["\varepsilon"', from=2-2, to=1-3]
\end{tikzcd}} \\
\end{itemize}
\end{defn}

\subsection{Hopf Algebra}
\begin{defn}{Hopf Algebras}{}\\
Let $R$ be a commutative ring. Let $(H,m,u,\Delta,\varepsilon)$ be an $R$-bialgebra. Let $S:H\to H$ be an $R$-module homomorphism. We say that $(H,m,u,\Delta,\varepsilon,S)$ is a Hopf algebra if the following diagram commutes: \\~\\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	& {H\otimes_RH} && {H\otimes_RH} \\
	H && R && H \\
	& {H\otimes_RH} && {H\otimes_RH}
	\arrow["{S\otimes\text{id}_H}", from=1-2, to=1-4]
	\arrow["m", from=1-4, to=2-5]
	\arrow["\Delta", from=2-1, to=1-2]
	\arrow["\varepsilon", from=2-1, to=2-3]
	\arrow["\Delta"', from=2-1, to=3-2]
	\arrow["u", from=2-3, to=2-5]
	\arrow["{\text{id}_H\otimes S}"', from=3-2, to=3-4]
	\arrow["m"', from=3-4, to=2-5]
\end{tikzcd}} \\
\end{defn}

\begin{defn}{Commmutative and Cocommutative}{}\\
Let $R$ be a commutative ring. Let $H$ be a Hopf algebra. 
\begin{itemize}
\item We say that $H$ is commutative if the underlying algebra is commutative. 
\item We say that $H$ is cocommutative if the underlying coalgebra is cocommutative. 
\end{itemize}
\end{defn}

\begin{prp}{}{}\\
Let $R$ be a commutative ring. Let $H$ be a Hopf algebra. If $H$ is a projective and finitely generated $R$-modue, then $\Hom_R(H,R)$ is a Hopf algebra. Moreover, the following are true. 
\begin{itemize}
\item $\Hom_R(M,R)$ is commutative if and only if $H$ is cocommutative. 
\item $\Hom_R(M,R)$ is cocommutative if and only if $H$ is commutative. 
\end{itemize}
\end{prp}

\subsection{Differential Graded Algebras}
\begin{defn}{Differential Graded Algebras}{}\\
Let $R$ be a commutative ring. A differential graded $R$-algebra consists of the following data. 
\begin{itemize}
\item A $\Z$-graded $R$-algebra $A$. 
\item An $R$-algebra homomorphism of degree $-1$ (called homological dgas) or of degree $1$ (called cohomological dgas). 
\end{itemize}
such that the following are true. 
\begin{itemize}
\item Differential: $d\circ d=0$. 
\item Graded Leibniz: $d(ab)=(da)b+(-1)^{\deg(a)}a(db)$ for $a,b$ homogeneous elements of $A$. 
\end{itemize}
\end{defn}

The differential property means that any (homological) differential graded algebra decomposes into a chain complex whose differential is given by $d$. 

\begin{defn}{Homomorphisms of Differential Graded Algebras}{}\\
Let $R$ be a commutative ring. Let $A,B$ be differential graded algebras. A morphism $\varphi:A\to B$ of differential graded algebras is a graded $R$-algebra homomorphism. 
\end{defn}

\begin{lmm}{}{}\\
Let $R$ be a commutative ring. Let $(A,d)$ be a cohomological differential graded algebra. Then $H^\ast(A,d)=\bigoplus_{n\in\Z}H^n(A,d)$ is a graded ring with multiplication defined by $$[u]\cdot[v]=[uv]$$ for $u\in H^n(A,d)$ and $v\in H^m(A,d)$. 
\begin{proof}\\
Notice that by the graded Leibniz rule, we have $d(uv)=(du)v+(-1)^nud(v)=0$ since $u$ and $v$ are cocycles. Hence $uv$ is also a cocycle. 
\end{proof}
\end{lmm}

\begin{defn}{Commutative Differential Graded Algebras}{}\\
Let $R$ be a commutative ring. Let $A$ be a differential graded algebra. We say that $A$ is commutative (cdga) if $A$ is graded commutative. 
\end{defn}

\begin{lmm}{}{}\\
Let $R$ be a commutative ring. Let $M$ be an $R$-module. Let $\varphi:M\to R$ be an $R$-module homomrphism. Then $K_\bullet(\varphi)$ is a commutative differential graded algebra. 
\end{lmm}

\section{Infinite Galois Theory}
\begin{prp}{}{}\\
Let $F$ be a field. Let $K/F$ be a Galois extension. Then the following are true. 
\begin{itemize}
\item The inverse limit of the system $\text{Gal}(L/F)$ for $L/F$ a finite Galois extension is $\text{Gal}(K/F)$. 
\item $\text{Gal}(K/F)$ is a profinite group. 
\item The projection maps $\text{Gal}(K/F)\to\text{Gal}(L/F)$ for $L/F$ a finite Galois extension is a surjective homomorphism. 
\item Let $F<L<K$ be a field extension. Then $K/L$ is a Galois extension and $\text{Gal}(K/L)$ is a closed subgroup of $\text{Gal}(K/F)$. 
\end{itemize}
\end{prp}

\begin{thm}{Krull}{}\\
Let $F$ be a field. Let $K/F$ be a field extension. Then there is an inclusion reversing bijection $$\left\{\substack{\text{Closed subgroups of}\\\text{Aut}_F(K)}\right\}\;\;\overset{\text{1:1}}{\longleftrightarrow}\;\;\left\{\substack{\text{Intermediary subfields}\\ F<L<K}\right\}$$ given as follows. 
\begin{itemize}
\item For a subgroup $H\leq\text{Aut}_F(K)$, $H\mapsto\text{Fix}_F(H)$. 
\item For an intermediary subfield $M$, $M\mapsto\text{Aut}_L(K)$. 
\end{itemize}
\end{thm}

\begin{prp}{}{}\\
Let $F$ be a field. Let $K/F$ be a Galois extension. Then the above bijection restricts to a bijection $$\left\{\substack{\text{Normal Subgroups}\\\text{of }\text{Aut}_K(L)}\right\}\;\;\overset{\text{1:1}}{\longleftrightarrow}\;\;\left\{\substack{\text{Normal Extensions}\\ K<M}\right\}$$ 
\end{prp}

\section{Local Cohomology}
\subsection{}
\begin{defn}{The $I$-Torsion of a Module}{}\\
Let $R$ be a Noetherian commutative ring. Let $M$ be an $R$-module. Let $I$ be an ideal of $R$. Define the $I$-torsion of $M$ to be $$\Gamma_I(M)=\{m\in M\;|\;I^tm=0\text{ for some }t\in\N\}$$
\end{defn}

\begin{defn}{$I$-Torsion Modules}{}\\
Let $R$ be a Noetherian commutative ring. Let $M$ be an $R$-module. Let $I$ be an ideal of $R$. We say that $M$ is $I$-torsion if $M=\Gamma_I(M)$. 
\end{defn}

\begin{defn}{The Torsion Functor}{}\\
Let $R$ be a Noetherian commutative ring. Let $I$ be an ideal of $R$. Define the $I$-torsion functor $$\Gamma_I:{_R}\bold{Mod}\to{_R}\bold{Mod}$$ by the following. 
\begin{itemize}
\item For each $R$-module $M$, the functor sends $M$ to the $I$-torsion $\Gamma_I(M)$. 
\item For each $R$-module homomorphism $f:M\to N$, the functor sends $f$ to $\Gamma_I(f)=f|_{\Gamma_I(M)}$. 
\end{itemize}
\end{defn}

\begin{lmm}{}{}\\
Let $R$ be a Noetherian commutative ring. Let $I$ be an ideal of $R$. Then $\Gamma_I$ is a left exact functor. 
\end{lmm}

\begin{defn}{Local Cohomology}{}\\
Let $R$ be a Noetherian commutative ring. Let $M$ be an $R$-module. Let $I$ be an ideal of $R$. Define the $n$th local cohomology of $M$ with support in $I$ to be $$H_I^n(M)=(R^n\Gamma_I)(M)$$
\end{defn}

\begin{prp}{}{}\\
Let $R$ be a Noetherian commutative ring. Let $M$ be an $R$-module. Let $I,J$ be ideals of $R$. If $\sqrt{I}=\sqrt{J}$, then the equality $\Gamma_I(M)=\Gamma_J(M)$ induces an equality $$H_I^n(M)=H_J^n(M)$$ for all $n\in\N$. 
\end{prp}

Thus we only need to consider the local cohomology with support in radical ideals. 

\begin{lmm}{}{}\\
Let $R$ be a Noetherian commutative ring. Let $M$ be an $R$-module. Let $I$ be an ideal of $R$. Then $H_I^n(M)$ is $I$-torsion for all $n\in\N$. 
\end{lmm}

\begin{lmm}{}{}\\
Let $R$ be a Noetherian commutative ring. Let $M_i$ be $R$-modules for each $i\in J$. Let $I$ be an ideal of $R$. Then there is a natural isomorphism $$H_I^n\left(\bigoplus_{i\in J}M_i\right)\cong\bigoplus_{i\in J}H_J^n(M_i)$$
\end{lmm}

\begin{prp}{}{}\\
Let $R$ be a Noetherian commutative ring. Let $M$ be an $R$-module. Let $I$ be an ideal of $R$.Then there are natural isomorphisms $$\varinjlim_{i\in\N}\text{Ext}_R^n(R/I^i,M)\cong H_I^n(M)$$ for all $n\in\N$. 
\end{prp}

\section{Morita Theory}
\subsection{}
\begin{defn}{Morita Equivalence}{}\\
Let $R,S$ be rings. We say that $R$ and $S$ are Morita equivalent if there exists an equivalence of categories $${_R}\bold{Mod}\simeq{_S}\bold{Mod}$$
\end{defn}

\begin{lmm}{}{}\\
Let $R,S$ be rings. Then $R$ and $S$ are Morita equivalent if and only if there exists an equivalence of categories $$\bold{Mod}_R\simeq\bold{Mod}_S$$
\end{lmm}

\begin{lmm}{}{}\\
Let $R,S$ be rings. Suppose that $R$ and $S$ are Morita equivalent. Let $F:{_R}\bold{Mod}\to{_S}\bold{Mod}$ be an equivalence. Then $F$ is additive. 
\end{lmm}

\subsection{Progenerators}
\begin{defn}{Progenerators}{}\\
Let $R$ be a ring. Let $P$ be an $R$-module. We say that $P$ is a progenerator of $R$ if the following are true. 
\begin{itemize}
\item $P$ is finitely generated. 
\item $P$ is projective. 
\end{itemize}
\end{defn}

\begin{prp}{}{}\\
Let $R,S$ be rings. Let $F:{_R}\bold{Mod}\to{_S}\bold{Mod}$ and $G:{_S}\bold{Mod}\to{_R}\bold{Mod}$ be additive functors. Then $F$ and $G$ are inverses if and only if there exists an $(S,R)$-bimodule $P$ such that the following are true. 
\begin{itemize}
\item ${_S}P$ is a progenerator. 
\item $P_R$ is a progenerator. 
\item There are natural isomorphisms $F\cong P\otimes_R-$ and $G\cong\Hom({_S}P,-)$. 
\end{itemize}
\end{prp}

\begin{thm}{Morita}{}\\
Let $R,S$ be rings. Then the following are equivalent. 
\begin{itemize}
\item $R$ and $S$ are Morita equivalent. 
\item There exists a progenerator $R$-module $P_R$ such that $S\cong\text{End}_R(P_R)$. 
\item There exist an $(R,S)$-bimodule $M$ and a $(S,R)$-bimodule $N$ such that $M\otimes_SN\cong R$ as $(R,R)$-bimodules and $N\otimes_RM\cong S$ as $(S,S)$-bimodules. 
\end{itemize}
\end{thm}

\subsection{Properties of Morita Equivalence}
\begin{prp}{}{}\\
Let $R,S$ be rings. If $R$ and $S$ are Morita equivalent, then the following are true. 
\begin{itemize}
\item There is a ring isomorphism $Z(R)\cong Z(S)$. 
\item $R/J(R)$ is Morita equivalent to $S/J(S)$. 
\end{itemize}
\end{prp}

\begin{prp}{}{}\\
Let $R,S$ be rings. Let $M$ be an $R$-module. Suppose that $R$ and $S$ are Morita equivalent via the functor $F:{_R}\bold{Mod}\to{_S}\bold{Mod}$. Then the following are true. 
\begin{itemize}
\item $M$ is injective if and only if $F(M)$ is injective. 
\item $M$ is projective if and only if $F(M)$ is projective. 
\item $M$ is simple if and only $F(M)$ is simple. 
\item $M$ is semisimple if and only if $F(M)$ is semisimple. 
\item $M$ is finitely generated if and only if $F(M)$ is finitely generated. 
\item $M$ is Noetherian if and only if $F(M)$ is Noetherian. 
\item $M$ is Artinian if and only if $F(M)$ is Artinian. 
\end{itemize}
\end{prp}

\begin{prp}{}{}\\
Let $R,S$ be rings. Suppose that $R$ and $S$ are Morita equivalent. Then the following are equivalent. 
\begin{itemize}
\item $R$ is simple if and only if $S$ is simple. 
\item $R$ is semisimple if and only if $S$ is semisimple. 
\item $R$ is Noetherian if and only if $S$ is Noetherian. 
\item $R$ is Artinian if and only if $S$ is Artinian. 
\end{itemize}
\end{prp}

\begin{prp}{}{}\\
Let $R,S$ be rings. Then $R$ and $S$ are Morita equivalent if and only if ${_R}\bold{FGMod}\simeq{_S}\bold{FGMod}$. 
\end{prp}
