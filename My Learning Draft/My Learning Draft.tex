\subsection{Stanley-Reisner Rings}
\begin{thm}{}{}\\
Let $R$ be a commutative ring. Let $n\in\N$. There is a bijection $$\substack{\text{Abstract Simplicial}\\\text{Complexes on }\{1,\dots,n\}}\;\;\overset{1:1}{\leftrightarrow}\;\;\substack{\text{Squarefree monomial}\\\text{ideals of }R[x_1,\dots,x_n]}$$ given by sending a simplicial set $S$ to the ideal $\left\langle\prod_{j\in J}x_j\;|\;J\in\mP(\{1,\dots,n\})\setminus S\right\rangle$. 
\end{thm}

\begin{defn}{Stanley-Reisner Rings}{}\\
Let $R$ be a commutative ring. Let $n\in\N$. Let $S$ be an abstract simplicial complex on $\{1,\dots,n\}$. Define the Stanley-Reisner ring of $S$ to be $$R[S]=\frac{k[x_1,\dots,x_n]}{\left\langle\prod_{j\in J}x_j\;|\;J\in\mP(\{1,\dots,n\})\setminus S\right\rangle}$$ The ideal $I_S=\left\langle\prod_{j\in J}x_j\;|\;J\in\mP(\{1,\dots,n\})\setminus S\right\rangle$ is called the Stanley-Reisner ideal of $S$. 
\end{defn}

\begin{eg}{}{}\\
Let $R$ be a commutative ring. The Stanley-Reisner ring of the standard $n$-simplex $\Delta^n$ is given by $$R[\Delta^n]=k[x_1,\dots,x_n]$$
\end{eg}

\begin{eg}{}{}\\
Let $R$ be a commutative ring. The Stanley-Reisner ring of the abstract simplicial complex $S=\mP(\{1,2,3\})\cup\mP(\{1,3,4\})\cup\mP(\{2,4\})$ over $\{1,2,3,4\}$ is given by $$R[S]=\frac{k[x_1,x_2,x_3,x_4]}{(x_1x_2x_4,x_2x_3x_4)}$$
\end{eg}

\subsection{The Hilbert Series of Stanley-Reisner Rings}
\begin{thm}{Stanley}{}\\
Let $k$ be a field. Let $n\in\N$. Let $S$ be an abstract simplicial complex on $\{1,\dots,n\}$. Let $(f_0,\dots,f_n)$ be the face vector of $S$. Let $(h_0,\dots,h_n)$ be the $h$-vector of $S$. Then the Hilbert series of $k[S]$ is given by $$HS_{k[S]}(t)=\sum_{i=0}^{n+1}f_{k-1}\left(\frac{t^2}{1-t^2}\right)^k=\frac{1}{(1-t^2)^{n+1}}\sum_{i=0}^{n+1}h_it^{2i}$$
\end{thm}

\begin{eg}{}{}|\
Let $\Delta^n$ be the standard $n$-simplex. Then the following are true. 
\begin{itemize}
\item The Hilbert series for the Stanley-Reisner ring of $\Delta^n$ is given by $$HS_{k[\Delta^n]}(t)=\frac{1}{(1-t^2)^{n+1}}$$
\item The Hilbert series for the Stanley-Reisner ring of $\partial\Delta^n$ is given by $$HS_{k[\partial\Delta^n]}(t)=\frac{1}{(1-t^2)^{n+1}}\sum_{i=0}^{n+1}t^{2i}$$
\end{itemize}
\end{eg}

\subsection{The Homotopy Theory of Polyhedral Products}
\begin{defn}{Polyhedral Products}{}\\
Let $\mK$ be a simplicial complex on $\{1,\dots,n\}$. Let $(\bold{X},\bold{A})=\{(X_1,A_1),\dots,(X_n,A_n)\}$ be pairs of spaces. 
\begin{itemize}
\item Let $\sigma\in\mK$. Define $$(\bold{X},\bold{A})^\sigma=\left\{(x_1,\dots,x_n)\in\prod_{i=1}^nX_i\;\bigg{|}\;x_i\in A_i\text{ for }i\notin I\right\}$$
\item Define the polyhedral product of $(\bold{X},\bold{A})$ corresponding to $\mK$ to be $$(\bold{X},\bold{A})^\mK=\bigcup_{I\in\mK}(\bold{X},\bold{A})^\sigma$$
\end{itemize}
\end{defn}

\begin{prp}{}{}\\
Let $\mK$ be a simplicial complex on $\{1,\dots,n\}$. Let $(\bold{X},\bold{A})=\{(X_1,A_1),\dots,(X_n,A_n)\}$ be pairs of spaces. Then we have $$(\bold{X},\bold{A})^\mK=\colim_{\sigma\in\text{Cat}(\mK)}(\bold{X},\bold{A})^\sigma$$
\end{prp}

\begin{prp}{}{}\\
Let $\mK$ be a simplicial complex on $\{1,\dots,n\}$. Let $(\bold{X},\bold{A})$ and $(\bold{Y},\bold{B})$ be $n$ pairs of spaces. Let $f_i:(X_i,A_i)\to(Y_i,B_i)$ be a map of pair for $1\leq i\leq n)$. Then $(f_1,\dots,f_n)$ induces a map $$(\bold{X},\bold{A})^\mK\to(\bold{Y},\bold{B})^\mK$$
\end{prp}

\begin{prp}{}{}\\
Let $\mK,\mL$ be a simplicial complex on $\{1,\dots,n\}$. Let $(\bold{X},\bold{A})=\{(X_1,A_1),\dots,(X_n,A_n)\}$ be pairs of spaces. Let $f_i:(X_i,A_i)\to(Y_i,B_i)$ be a map of pair for $1\leq i\leq n)$. Then we have $$(\bold{X},\bold{A})^{\mK\ast\mL}=(\bold{X},\bold{A})^\mK\times(\bold{X},\bold{A})^\mL$$
\end{prp}

\begin{defn}{Polyhedral Smash Products}{}\\
Let $\mK$ be a simplicial complex on $\{1,\dots,n\}$. Let $(\bold{X},\bold{A})$ be pairs of spaces. 
\begin{itemize}
\item Let $\sigma\in\mK$. Define $$(\bold{X},\bold{A})^{\wedge\sigma}=\left\{(x_1,\dots,x_n)\in\bigwedge_{i=1}^nX_i\;\bigg{|}\;x_i\in A_i\text{ for }i\notin I\right\}$$
\item Define the polyhedral smash product of $(\bold{X},\bold{A})$ corresponding to $\mK$ to be $$(\bold{X},\bold{A})^\mK=\bigcup_{I\in\mK}(\bold{X},\bold{A})^{\wedge\sigma}$$
\end{itemize}
\end{defn}

\begin{thm}{Bahri, Bendersky, Cohen, Gitler}{}\\
Let $\mK$ be a simplicial complex on $\{1,\dots,n\}$. Let $(\bold{X},\bold{A})$ be $n$-pairs of CW complexes. Suppose that the inclusion maps $A_j\to X_j$ are null homotopic for $1\leq j\leq n$. Then there is a homotopy equivalence $$(\bold{X},\bold{A})^{\wedge\mK}\simeq\bigvee_{\sigma\in\mK}\abs{\text{lk}_\mK(\sigma)}\ast(\bold{X}_J,\bold{A}_J)^{\wedge\sigma}$$
\end{thm}

\begin{thm}{Bahri, Bendersky, Cohen, Gitler}{}\\
Let $\mK$ be a simplicial complex on $\{1,\dots,n\}$. Let $(\bold{X},\bold{A})$ be $n$-pairs of CW complexes. Then there is a homotopy equivalence $$\Sigma(\bold{X},\bold{A})^\mK\simeq\Sigma\left(\bigvee_{J\subseteq\{1,\dots,n\}}(\bold{X}_J,\bold{A}_J)^{\wedge\mK_J}\right)$$ that is natural in $(\bold{X},\bold{A})$, where $(\bold{X}_J,\bold{A}_J)=\{(X_j,A_j)\;|\;j\in J\}$. 
\end{thm}

\begin{crl}{}{}\\
Let $\mK$ be a simplicial complex on $\{1,\dots,n\}$. Let $(\bold{X},\bold{A})$ be $n$-pairs of CW complexes. Suppose that $A_i$ are null homotopic for all $i$. Then there is a homotopy equivalence $$\Sigma(\bold{X},\bold{A})^\mK\simeq\Sigma\left(\bigvee_{\sigma\in\mK}\bold{X}^{\wedge\sigma}\right)$$
\end{crl}

\begin{crl}{}{}\\
Let $\mK$ be a simplicial complex on $\{1,\dots,n\}$. Let $(\bold{X},\bold{A})$ be $n$-pairs of CW complexes. Suppose that $X_i$ are null homotopic for all $i$. Then there is a homotopy equivalence $$\Sigma(\bold{X},\bold{A})^\mK\simeq\Sigma\left(\bigvee_{J\notin\mK}\abs{\mK_J}\ast\bold{A}^{\wedge J}\right)$$
\end{crl}

\begin{defn}{Ring Structure in Hochster's Formula}{}\\
Let $\mK$ be a simplicial complex on $\{1,\dots,n\}$. Define a ring structure in $\bigoplus_{J\subseteq\{1,\dots,n\}}\widetilde{H}^\ast(\mK_\mJ;\Z)$ as follows. Let $\alpha_L$ be the cochain basis element in $C^p(\mK_I;\Z)$ corresponding to the simplex $L\subseteq I$. The ring structure is induced by the map $$C^p(\mK_I;\Z)\otimes_\Z C^q(\mK_\mJ;\Z)\to C^{p+q+1}(\mK_{I\cup J})$$ defined by $$\alpha_L\otimes\alpha_M\mapsto\begin{cases}
\varepsilon(L,I)\varepsilon(M,J)\varepsilon(L\cup M,I\cup J)\alpha_{L\amalg M} & \text{ if }I\cap J=\emptyset\\
0 & \text{ otherwise }
\end{cases}$$
where $\varepsilon(L,I)=\prod_{j\in L}\varepsilon(j,J)$ and $\varepsilon(j,J)=(-1)^{r-1}$ if $j$ is in the $r$th position in $J$. 
\end{defn}

\begin{prp}{}{}\\
Let $\mK$ be a simplicial complex on $\{1,\dots,n\}$. Then the following are true. 
\begin{itemize}
\item The ring structure in Hochster's formula is equal to the relative Kunneth theorem for $$H^\ast(\abs{\mK_I\ast\mK_J})=H^\ast(\Sigma(\abs{\mK_I}\wedge\abs{\mK_J}))=\widetilde{H}^{\ast-1}(\abs{K_I}\times\abs{K_J},\abs{K_I\vee K_J})$$ via the inclusion $\mK_{I\amalg J}\hookrightarrow\mK_I\ast\mK_J$. 
\item The natural isomorphism in the above prp induces a natural isomorphism of rings $$H^\ast(\mZ_\mK;\Z)\cong\bigoplus_{J\subseteq\{1,\dots,n\}}\widetilde{H}^\ast(\mK_J;\Z)$$
\end{itemize}
\end{prp}

\subsection*{The Foundation}
lMoment angle complexes are $T^n$-equivariant objects defined using combinatorial data. More specifically, choose a simplicial complex $\mK$ on $\{1,\dots,n\}$. For each simplex $\sigma\in\mK$, define the space $$(D^2,S^1)^\sigma=\prod_{i=1}^nY_i$$ where $$Y_i=\begin{cases}
D^2 & \text{ if }i\in\sigma\\
S^1 & \text{ otherwise }
\end{cases}$$ In other words, $(D^2,S^1)^\sigma$ is a product of $D^2$ and $S^1$ depending on how many vertices of $\mK$ lie in $\sigma$. For any inclusion of simplicies $\sigma\subseteq\tau$, there is an inclusion map $(D^2,S^1)^\sigma\hookrightarrow(D^2,S^1)^\tau$. Then the definition of the moment angle complex $\mZ_\mK$ of $\mK$ is just the gluing of products of $D^2$ and $S^1$ along these inclusions, or $$\mZ_\mK=\bigcup_{\sigma\in\mK}(D^2,S^1)^\sigma$$ Being combinatorial in nature, they form a testing ground for conjectures in topology. Moreover, in good cases (eg $\mK$ is a triangulation of $S^m$), the moment angle complex $\mZ_\mK$ has the structure of a smooth manifold and they form a testing ground for conjectures in geometry. 

These topological objects are also interesting because they lie at the intersection of a number of active fields. Other than moment angle complexes, there are other objects in topology and geometry that are defined in terms of combinatorial objects

In fact, the first connection between symplectic geometry and combinatorics came from the following theorem. 

\begin{thm}{The Atiyah–Guillemin–Sternberg convexity theorem (1982)}{}\\
Let $(M,\omega)$ be a compact connected symplectic manifold. Let $T^k\times M\to M$ be a Hamiltonian torus action. Then $\mu(M)$ is a convex polytope in $\text{Lie}(T^k)^\ast$ given by $$\mu(M)=\text{Conv}(\text{Fix}_{T^k}(M))$$
\end{thm}

Delzant in 1988 proved that if the action is more over effective and $M$, then these polytopes completely determine the manifold structure. This motivates the definition of quasitoric manifolds: Smooth manifolds $M^{2n}$ with a torus action $T^n$ that locally looks like the standard action, together with a homeomorphism of the orbit space $M/T^n\cong P$ with a polytope. In 2008, the following conjecture is made. 

\begin{conj}{Masuda–Suh (2008)}{}\\
If $M$ and $N$ are quasitoric manifolds such that $$H^\ast(M)\cong H^\ast(N)$$ Then $M$ and $N$ are homeomorphic (or diffeomorphic). 
\end{conj}

\subsection*{The Focus of My Work}
The first observation is that Hochster's formula shows that the cohomology of a moment angle complex is completely determined by the simplicial complex: 

\begin{thm}{Hochster's Formula}{}\\
Let $\mK$ be a simplicial complex on $\{1,\dots,n\}$. Then there is an isomorphism of rings $$H^\ast(\mZ_\mK)\cong\bigoplus_{J\subseteq\{1,\dots,n\}}H^\ast(\mK_J)$$ where 
\begin{itemize}
\item $\mK_J$ is the full subcomplex of $\mK$ with the vertices in $J$. 
\item The additive isomorphism is given by $$H^n(\mZ_\mK)\cong\bigoplus_{J\subseteq\{1,\dots,n\}}H^{n-\abs{J}-1}(\mK_J)$$
\item The ring structure is given the map $H^i(\mK_I)\otimes H^j(\mK_J)\to H^{i+j+1}(\mK_{I\cup H_J})$ which sends products of generators to generators if $I\cap J=\emptyset$ and $0$ otherwise. 
\end{itemize}
\end{thm}

The second observation is that certain classes of simplicial complexes have the geometry of connected sums of products of spheres: 

\begin{thm}{}{}\\
Let $P$ be a stacked polytope. Let $n=\dim(P)$ and $m=\abs{V(P)}$. Then there is a diffeomorphism $$\mZ_{\partial P}\cong\#_{k=3}^{m-n+1}(S^k\times S^{m+n-k})^{(k-2)\binom{m-n}{k-1}}$$
\end{thm}

Among the class of topological spaces, spheres are homologically and homotopically the easiest to understand. It is natural to ask the question of in what conditions does a moment angle complex has its cohomology only involving spheres, and in this case is it possible to lift it to a homotopy equivalence, a homeomorphism or a diffeomorphism. Currently my work focuses on moment angle complexes coming from simplicial polytopes of dimension $4$ with $8$ vertices.

\item The homotopy types of moment angle complexes of $P_{35}^8,P_{36}^8,P_{37}^8$ are understood (they are the neighbourly polytopes of dim 4 and vert 8).

\end{itemize}

\subsubsection*{Analyzing the Cohomology}
If $\mK$ is a simplicial complex that is a triangulation of $S^n$. Then the following are true. 
\begin{itemize}
\item $\mK_{[m]\setminus J}$ is a deformation retract of $\mK\setminus\mK_J$. 
\item $\abs{\mK_J}$ is a compact locally contractible subspace of $\abs{\mK}$. 
\item By Alexander duality, we have $$\widetilde{H}^i(\mK_J)\cong\widetilde{H}_{n-i-1}(S^n\setminus\mK_J)\cong\widetilde{H}_{n-i-1}(\mK\setminus\mK_J)\cong\widetilde{H}_{n-i-1}(\mK_{[m]\setminus J})$$

and $\deg(a_i)=3$, $\deg(b_i)=4$, $\deg(c)=5$, $\deg(d_i)=6$, $\deg(e)=7$, $\deg(f_i)=8$, $\deg(h)=12$ (13 generators and 28 relations). \subsubsection*{Analyzing the Topology}
There are retractions $\mZ_{\mK_J}\hookrightarrow\mZ_\mK\to\mZ_{\mK_J}$ where $J\subseteq\{1,\dots,n\}$ induced by retracting $(D^2,S^1)$ to $(S^1,S^1)$ by mapping $D^2$ to the base point, and these are typically easier to compute. In particular, the retraction may factor through the $(n-1)$-skeleton, which may reveal information about the topology of the moment angle complex. In [Membrilo Solis, Theriault (2026)] and [Iriye (2018)], certain $\mZ_{\mK_J}$ are wedges of spheres which are useful to knowing the topology because the remainder of $\mZ_{\mK_J}$ in $\mZ_\mK$ may be understood by attaching maps of spheres to wedges of spheres, and if this is the case we may use Hilton Milnor theorem to analyze the components. 

In the general case, [BBCG] only decomposes the suspension of $\mZ_\mK$ as a wedge of spheres: $$\Sigma\mZ_\mK\simeq\bigvee_{J\notin\mK}\Sigma^{2+\abs{J}}\abs{\mK_J}$$ And work was done to desuspend this decomposition [Iriye, Kishimoto], the case for a shifted complex or disjoint points. 

To understand $\mZ_\mK$ it is also useful to analyze fibrations related to $\mZ_\mK$. The following are some classical fiber bundles: 
\begin{itemize}
\item $T^{m-n}\to\mZ_\mK\to M(P,\Lambda)$ to quasi-toric manifolds. 
\item $\mZ_\mK\to DJ(\mK)=(\C\Prj^\infty,\ast)^\mK\to BT^m$. 
\end{itemize}

\subsubsection*{Obstructions to Rigidity}
Massey products cannot be detected in cohomological isomorphisms so it is a prime method to showing two spaces are not homeomorphic. There are combinatorial descriptions [Denham, Suciu (2005)] for whitehead products to be non-trivial using the underlying graph of simplicial complexes. More generally the same can be said for $A_\infty$ structures on the cohomology ring.

\section{Basic Definitions}
\subsection{Symplectic Manifolds}
\begin{defn}{Symplectic Manifolds}{}\\
Let $M$ be a smooth manifold. Let $\omega\in\Omega^2(M)$ be a $2$-form. We say that $\omega$ is a symplectic form if the following are true. 
\begin{itemize}
\item $\omega$ is a symplectic structure on $TM$. 
\item $d\omega=0$. 
\end{itemize}
In this case we call $(M,\omega)$ a symplectic manifold. 
\end{defn}

\begin{lmm}{}{}\\
Let $(M,\omega)$ be a symplectic manifold. Then $\dim(M)$ is even. 
\end{lmm}

\begin{defn}{The Liouville Volume Form}{}\\
Let $(M,\omega)$ be a symplectic manifold of dimension $2n$. Define the Liouville volume form of $M$ by $$\mu_M=\frac{\omega^n}{n!}$$
\end{defn}

\begin{prp}{}{}\\
Let $(M,\omega)$ be a symplectic manifold. Then the following are true. 
\begin{itemize}
\item $\mu_M$ is a nowhere-vanishing differential form. 
\item $M$ is orientable. 
\end{itemize}
\end{prp}

\begin{eg}{}{}\\
The following are symplectic manifolds. 
\begin{itemize}
\item $\R^{2n}$ with coordinates $(x_1,\dots,x_n,y_1,\dots,y_n)$ and with symplectic form $$\sum_{i=1}^ndx_i\wedge dy_i$$
\item For all $g\geq 1$, the compact orientable surface $\Sigma_g$ with any $\omega$ such that $0\neq[\omega]\in H_\text{dR}^2(\Sigma_g)$. 
\end{itemize}
\end{eg}

\subsection{Standard Symplectic Coordinates}
\begin{thm}{Moser's Theorem Relative Version}{}\\
Let $M$ be a smooth manifold. Let $X$ be an embedded compact submanifold of $M$. Let $\iota:X\hookrightarrow M$ be the inclusion map. Let $\omega_0$ and $\omega_1$ be symplectic forms on $M$. If $\omega_0|_X=\omega_1|_X$, then there exists $X\subseteq U_0,U_1\subseteq M$ and a map $\varphi:U_0\to U_1$ such that the following are true. 
\begin{itemize}
\item $\iota=\varphi\circ\iota$. 
\item $\varphi$ is a diffeomorphism. 
\item $(\varphi|_X)^\ast\omega_1=\omega_0$. 
\end{itemize}
\end{thm}

\begin{thm}{Darboux (1982)}{}\\
Let $(M,\omega)$ be a symplectic manifold. Let $p\in M$. Then there exists a chart $(U,x^1,\dots,x^n,y^1,\dots,y^n)$ at $p$ such that $$\omega|_U=\sum_{i=1}^ndx^i\wedge dy^i$$
\end{thm}

\begin{defn}{Darboux Coordinates}{}\\
Let $(M,\omega)$ be a symplectic manifold. Let $(U,x^1,\dots,x^n,y^1,\dots,y^n)$ be a local chart of $M$. We say that the chart is Darboux if $$\omega|_U=\sum_{i=1}^ndx^i\wedge dy^i$$
\end{defn}

\subsection{Maps Preserving the Symplectic Structure}
\begin{defn}{Symplectic Maps}{}\\
Let $(M,\omega)$ and $(N,\sigma)$ be symplectic manifolds. Let $f:M\to N$ be a smooth map. 
\begin{itemize}
\item We say that $f$ is symplectic if $$f^\ast\sigma=\omega$$
\item We say that $f$ is a symplectomorphism if $f$ is symplectic and a diffeomorphism. 
\end{itemize}
\end{defn}

\begin{lmm}{}{}\\
Let $(M,\omega)$ and $(N,\sigma)$ be symplectic manifolds. Let $f:M\to N$ be a symplectic map. Then $f$ is an immersion. 
\end{lmm}

\begin{defn}{The Group of Symplectomorphisms}{}\\
Let $(M,\omega)$ be a symplectic manifold. Define the group of symplectomorphisms of $M$ to be the subgroup $$\text{Symp}(M,\omega)=\{f:M\to M\;|\;f\text{ is a symplectomorphism }\}\leq\text{Diff}(M)$$
\end{defn}

\subsection{Symplectic Structure on the Cotangent Bundle}
Through out this section we denote the projection map $T^\ast M\to M$ by $\pi_M$. 

\begin{defn}{The Canonical Hamiltonian Structure}{}\\
Let $M$ be a smooth manifold. Define the canonical Hamiltonian structure $\omega$ on $T^\ast M$ as follows. At the local chart $(U,q^1,\dots,q^n,p^1,\dots,p^n)$, $\omega$ is given by $$\omega|_U=\sum_{i=1}^ndq^i\wedge dp^i$$
\end{defn}

\begin{lmm}{}{}\\
Let $M$ be a smooth manifold. Let $\omega$ be the canonical Hamiltonian structure. Then $(T^\ast M,\omega)$ is a symplectic manifold. 
\end{lmm}

\begin{defn}{The Liouville 1-Form}{}\\
Let $M$ be a smooth manifold. Let $\pi:T^\ast M\to M$ be the bundle map. Define the Lioville $1$-form $\theta\in\Omega^1(T^\ast M)$ as follows. For each $q\in M$ and $p\in T_q^\ast M$, define $$\theta_{(q,p)}=p(\pi_M(v))$$ for any $v\in T_{(q,p)}(T^\ast M)$. 
\end{defn}

\begin{prp}{}{}\\
Let $M$ be a smooth manifold. Let $(U,q^1,\dots,q^n,p^1,\dots,p^n)$ be a chart of $T^\ast M$ where $(q_1,\dots,q_n)\in M$. Then $\theta$ is given in local coordinates by $$\theta|_U=\sum_{i=1}^np^idq^i$$
\end{prp}

\begin{prp}{}{}\\
Let $M$ be a smooth manifold. Let $\theta$ be the Liouville 1-form. Let $\omega$ be the canonical Hamitonian structure. Then we have $$\omega=-d\theta$$
\end{prp}

\begin{defn}{The Lift of a Diffeomorphism}{}\\
Let $M,N$ be smooth manifolds. Let $f:M\to N$ be a diffeomorphism. Define the lift of $f$ to be the map $f_\#:T^\ast M\to T^\ast N$ given by $$f_\#(p,v)=(f(p),w)$$ where $w\in T_p^\ast M$ is the unique element such that $v=(df_p)^\ast w$. 
\end{defn}

\begin{prp}{}{}\\
Let $M,N$ be smooth manifolds. Let $f:M\to N$ be a diffeomorphism. Then the following are true. 
\begin{itemize}
\item We have $f\circ\pi_M=\pi_N\circ f_\#$. 
\item $f_\#^\ast\theta_N=\theta_M$. 
\end{itemize}
\end{prp}

\section{Lagrangian Submanifolds}
\begin{defn}{Lagrangian Vector Subspaces}{}\\
Let $k$ be a field. Let $(V,\omega)$ be a symplectic vector space over $k$. Let $W\subseteq V$ be a subspace. We say that $W$ is Lagrangian if the following are true. 
\begin{itemize}
\item $\omega|_W=0$. 
\item $2\dim(W)=\dim(V)$. 
\end{itemize}
\end{defn}

\begin{defn}{Lagrangian Submanifolds}{}\\
Let $(M,\omega)$ be a symplectic manifold. Let $L$ be an embedded submanifold of $M$. We say that $L$ is Lagrangian if $T_pL$ is a Lagrangian subspace of $M$. 
\end{defn}

\begin{lmm}{}{}\\
Let $(M,\omega)$ be a symplectic manifold. Let $L$ be an embedded submanifold of $M$. Then $L$ is Lagrangian if and only if the following are true. 
\begin{itemize}
\item $\iota^\ast\omega=0$ where $\iota:L\hookrightarrow M$ is the inclusion map. 
\item $2\dim(L)=\dim(M)$. 
\end{itemize}
\end{lmm}

\begin{prp}{}{}\\
Let $M$ be a smooth manifold. Let $s\in\Omega^1(M)$. Then $\im(s)$ is a Lagrangian submanifold of $T^\ast M$ if and only if $ds=0$. 
\end{prp}

\begin{thm}{Weinstein Lagrangian Neighborhood Theorem}{}\\
Let $M$ be a smooth manifold. Let $X$ be an embedded compact submanifold of $M$. Let $\iota:X\hookrightarrow M$ be the inclusion map. Let $\omega_0$ and $\omega_1$ be symplectic forms on $M$ such that $X$ is a Lagrangian submanifold of $(M,\omega_0)$ and $(M,\omega_1)$. Then there exists $X\subseteq U_0,U_1\subseteq M$ and a map $\varphi:U_0\to U_1$ such that the following are true. 
\begin{itemize}
\item $\iota=\varphi\circ\iota$. 
\item $\varphi$ is a diffeomorphism. 
\item $(\varphi|_X)^\ast\omega_1=\omega_0$. 
\end{itemize}
\end{thm}

\subsection{Relation to Symplectomorphisms}
\begin{prp}{}{}\\
Let $(M,\omega)$ and $(N,\sigma)$ be symplectic manifolds. Let $f:M\to N$ be a diffeomorphism. Then $f$ is a symplectomorphism if and only if $\Gamma_f$ is a Lagrangian submanifold of $(M\times N,\text{proj}_M^\ast\omega-\text{proj}_N^\ast\sigma)$. 
\end{prp}

Recipe for producing symplectomorphisms. 
\begin{enumerate}
\item Start with a Lagrangian submanifold of $(T^\ast M\times T^\ast N,\omega\times\sigma)$. 
\item Send the Lagrangian through the twist map $\tau:T^\ast M\times T^\ast N\to T^\ast M\times T^\ast N$ given by $(x,v)\mapsto(x,-v)$. The image is then a Lagrangian. 
\item Check whether $\tau(\sigma)$ is the graph of some diffeomorphism $\varphi:M\to N$. 
\item If it is, then $\varphi$ is a symplectomorphism. 
\end{enumerate}

\subsection{Generating Functions}
\begin{defn}{Generating Functions}{}\\
Let $M,N$ be smooth manifolds. Let $f\in\mC^\infty(M\times N)$. Define $$Y_f=\{(p,q,\text{proj}_{T_p^\ast M}d_{(p,q)}f,-\text{proj}_{T_q^\ast M}d_{(p,q)}f)\in M\times N\times T^\ast M\times T^\ast N\;|\;(p,q)\in M\times N\}$$ We say that $f$ is a generating function if there exists a diffeomorphism $\varphi:T^\ast M\to T^\ast N$ such that $Y_f=\Gamma_\varphi$. 
\end{defn}

In this case, we know that $\varphi:T^\ast M\to T^\ast N$ is given by $$\varphi(x,\text{proj}_{T_p^\ast M}d_{(p,q)}f)=(y,-\text{proj}_{T_q^\ast M}d_{(p,q)}f)$$

\section{Calculus on Symplectic Manifolds}
\subsection{Symplectic Vector Fields}
\begin{defn}{Symplectic Vector Fields}{}\\
Let $(M,\omega)$ be a symplectic manifold. Let $X\in\mathfrak{X}(M)$ be a vector field. We say that $X$ is symplectic if $\iota_X(\omega)$ is a closed form. 
\end{defn}

\begin{prp}{}{}\\
Let $(M,\omega)$ be a symplectic manifold. Let $X\in\mathfrak{X}(M)$ be a vector field. Let $\theta^X$ be the flow of $X$. Then $X$ is symplectic if and only if $\theta^X(t,-)$ is a symplectomorphism for all $t$. 
\end{prp}

\subsection{Hamiltonian Vector Fields}
\begin{defn}{The Vector Field of a Function}{}\\
Let $(M,\omega)$ be a symplectic manifold. Let $H\in\mC^\infty(M)$. Define the associated vector field of $H$ to be the unique vector field $X_H\in\mathfrak{X}(M)$ such that $$\iota_{X_H}(\omega)=df$$ where $\iota$ is the interior product. 
\end{defn}

\begin{prp}{}{}\\
Let $(M,\omega)$ be a symplectic manifold. Let $H\in\mC^\infty(M)$. Let $(U,x^1,\dots,x^n,y^1,\dots,y^n)$ be a Darboux chart. Then the vector field of $H$ is given locally at the chart by $$(X_H)|_U=\sum_{i=1}^n\left(\frac{\partial H}{\partial y^i}\frac{\partial}{\partial x^i}-\frac{\partial H}{\partial x^i}\frac{\partial}{\partial y^i}\right)$$
\end{prp}

\begin{defn}{Hamiltonian Vector Fields}{}\\
Let $(M,\omega)$ be a symplectic manifold. Let $X\in\mathfrak{X}(M)$ be a vector field. We say that $X$ is Hamiltonian if there exists a smooth map $H\in\mC^\infty(M)$ such that $$X=X_H$$
\end{defn}

\begin{prp}{}{}\\
Let $(M,\omega)$ be a symplectic manifold. Let $X\in\mathfrak{X}(M)$ be a vector field. If $X$ is Hamiltonian, then $X$ is symplectic. 
\end{prp}

\begin{thm}{Liouville's Theorem}{}\\
Let $(M,\omega)$ be a symplectic manifold. Let $X\in\mathfrak{X}(M)$ be a complete Hamiltonian vector field. Let $\theta^X$ be the flow of $X$. Then we have $$\theta^X(t,-)^\ast\mu_M=\mu_M$$
\end{thm}

\subsection{The Poisson Structure}
\begin{defn}{The Poisson Bracket}{}\\
Let $(M,\omega)$ be a symplectic manifold. Define the poisson bracket of $M$ to be the map $\{-,-\}:\mC^\infty(M)\times\mC^\infty(M)\to\mC^\infty(M)$ by $$\{f,g\}=X_f(g)$$
\end{defn}

\begin{lmm}{}{}\\
Let $(M,\omega)$ be a symplectic manifold. Then the following are true. 
\begin{itemize}
\item $(M,\{-,-\})$ is a Lie algebra. 
\item Leibniz rule: $\{f,gh\}=\{f,g\}h+g\{f,h\}$. 
\end{itemize}
\end{lmm}

\begin{prp}{}{}\\
Let $(M,\omega)$ be a symplectic manifold. Then the map $\mC^\infty(M)\to\mathfrak{X}(M)$ given by $f\mapsto X_f$ is a Lie algebra homomorphism. 
\end{prp}

\subsection{Integrable Systems}
\begin{defn}{Hamiltonian Systems}{}\\
A Hamiltonian system is a triple $(M,\omega,H)$ where $(M,\omega)$ is a symplectic manifold and $H\in\mC^\infty(M)$. 
\end{defn}

\begin{defn}{Integrable Systems}{}\\
Let $(M,\omega,H)$ be a Hamiltonian system. Suppose that $\dim(M)=2n$. We say that it is integrable if there exists $f_i\in\mC^\infty(M)$ for $1\leq i\leq n$ and $f_1=H$ such that $$\{f_i,f_j\}=0$$ for all $i,j$. 
\end{defn}

\begin{thm}{Arnold-Liouville Theorem}{}\\
Let $(M,\omega,H,f_2,\dots,f_n)$ be an integrable system. Let $c\in\R$. Then $(H,f_2,\dots,f_n)^{-1}(c)$ is a Lagrangian submanifold of $M$. Moreover, if $M$ is compact and connected, then the following are true. 
\begin{itemize}
\item $(H,f_2,\dots,f_n)^{-1}(c)$ is diffeomorphic to $T^k$ for some $k\in\N^+$. 
\item Angle coordinates. 
\end{itemize}
\end{thm}

\subsection{Symplectic Actions}
\begin{defn}{Symplectic Actions}{}\\
Let $(M,\omega)$ be a symplectic manifold. Let $G$ be a Lie group. Let $\varphi:G\to\text{Diff}(M)$ be a smooth action. We say that the action is symplectic if $$\im(\varphi)\subseteq\text{Symp}(M)$$
\end{defn}

\subsection{Moment Maps}
\begin{defn}{Moment Maps}{}\\
Let $(M,\omega)$ be a symplectic manifold. Let $G$ be a Lie group with a symplectic action on $M$. A moment map of $M$ is a map $$\mu:M\to\text{Lie}(G)^\ast\cong\R^{\dim(G)}$$ such that the following are true. 
\begin{itemize}
\item Hamilton's condition: For all $\xi\in\text{Lie}(G)$, the fundamental vector field $\xi_M$ is Hamiltonian via the map $\langle\mu,\xi\rangle:M\to\R$. 
\item Equivariance: For all $g\in G$, we have $$\mu(g\cdot x)=\text{Ad}_g^\ast(\mu(x))$$ for all $x\in M$. 
\end{itemize}
\end{defn}

\subsection{Hamiltonian Actions}

\subsection{Hamiltonian Torus Actions}
\begin{defn}{Hamiltonian Torus Actions}{}\\
Let $(M,\omega)$ be a symplectic manifold. Let $T^k$ be a torus that acts on $M$ by symplectomorphisms. We say that the action is Hamiltonian if it admits a moment map $$\mu:M\to\text{Lie}(T^k)^\ast$$
\end{defn}

\begin{thm}{The Atiyah–Guillemin–Sternberg convexity theorem (1982)}{}\\
Let $(M,\omega)$ be a compact connected symplectic manifold. Let $T^k\times M\to M$ be a Hamiltonian torus action. Then $\mu(M)$ is a convex polytope in $\text{Lie}(T^k)^\ast$ given by $$\mu(M)=\text{Conv}(\text{Fix}_{T^k}(M))$$
\end{thm}

\begin{defn}{Toric Symplectic Manifold}{}\\
Let $(M,\omega)$ be a symplectic manifold of dimension $2n$. Let $T^n\times M\to M$ be a Hamiltonian torus action. We say that $M$ is a toric symplectic manifold if $M$ is compact connected and the action is faithful. 
\end{defn}

Delzant polytope: simple, rational, smooth

\begin{thm}{Delzant (1988)}{}\\
There is a one to one bijection $$\frac{\{(M,\omega)\;|\;(M,\omega)\text{ is a toric symplectic manifold }\}}{\cong}\;\;\overset{1:1}{\leftrightarrow}\;\;\{P\;|\;P\text{ is a Delzant polytope }\}$$
\end{thm}

\section{Quasitoric Manifolds}
\subsection{The Davis-Januszkiewicz Classification Theorem}
\begin{defn}{Quasitoric Manifolds}{}\\
Let $M$ be a smooth manifold of dimension $2n$. Let $T^n$ be a torus acting smoothly on $M$. We say that $M$ is a quasitoric manifold if the following are true. 
\begin{itemize}
\item For each $p\in M$, there is a $T^n$-equivariant chart $(U,\varphi:U\to\C^n)$ where $\C^n$ is equipped with the standard torus action. 
\item There exists a convex polytope $P$ such that there is a homeomorphism $$M/T^n\cong P$$
\end{itemize}
\end{defn}

Toric symplectic manifolds are quasitoric manifolds. Moment angle manifolds are NOT quasitoric manifolds. 

\begin{defn}{Orbit Type Strata}{}\\
Let $M$ be a quasitoric manifold. Let $P$ be the underlying convex polytope. Let $H\leq T^n$ be a subgroup. Define the orbit type strata of $H$ to be $$M_{(H)}=\{x\in M\;|\;\text{Stab}_{T^n}(x)=H\}$$
\end{defn}

\begin{prp}{}{}\\
Let $M$ be a quasitoric manifold. Let $P$ be the underlying convex polytope. Then there is a bijection $$\{F\in\mF(P)\;|\;F\text{ has codimension }k\}\;\;\overset{1:1}{\leftrightarrow}\;\;\left\{N\subseteq M\;\bigg{|}\;\substack{N\text{ is an orbit type strata of subtori of }\\T^n\text{ of dimension }k}\right\}$$ given as follows. Let $\pi:M\to P$ be the quotient map. Then we have $$F_i\mapsto\pi^{-1}(\text{relint}(F))$$
\end{prp}

\begin{defn}{Primitive Vectors}{}\\
Let $v=(v_1,\dots,v_n)\in\Z^n$. We say that $v$ is primitive if $v\neq\lambda u$ for any $\lambda\in\Z\setminus\{0\}$ and $u\in\Z^n$. 
\end{defn}

\begin{defn}{The Characteristic Matrix}{}\\
Let $M$ be a quasitoric manifold. Let $P$ be the underlying convex polytope. Define the characteristic matrix $\Lambda$ as follows. For each facet $F_i$ of $P$ for $1\leq i\leq m$, the facet corresponds to a a circle $S_i^1\subseteq T^n$ by the above. Every circle $S_i^1\subseteq T^n$ corresponds to a unique primitive vector $v_i=(v_{1i},\dots,v_{1n})\in\Z^n$ up to sign. Then $$\Lambda=\begin{pmatrix}
v_{11} & \cdots & v_{1m}\\
\vdots & \ddots & \vdots\\
v_{n1} & \cdots & v_{nm}
\end{pmatrix}\in M_{n\times m}(\Z)$$
\end{defn}

\begin{defn}{Characteristic Pairs}{}\\
Let $P$ be a simple polytope of dimension $n$. Let $F_1,\dots,F_m$ be the facets of $P$. Let $\Lambda\in M_{n\times m}(\Z)$ be a matrix. Let $c_1,\dots,c_m$ be the columns of $\Lambda$. We say that $(P,\Lambda)$ is a characteristic pair if the following are true. 
\begin{itemize}
\item $c_i$ is a primitive vector for all $1\leq i\leq m$. 
\item If $F_{i_1}\cap\cdots\cap F_{i_n}$ is a vertex of $P$, then $$\det\begin{pmatrix}
| & & |\\
c_{i_1} & \cdots & c_{i_n}\\
| & & |
\end{pmatrix}=\pm1$$
\end{itemize}
\end{defn}

\begin{defn}{The Associated Quasitoric Manifold}{}\\
Let $(P,\Lambda)$ be a characteristic pair. Suppose that $\dim(P)=n$. Let $F_1,\dots,F_m$ be the facets of $P$. Let $c_1,\dots,c_m$ be the columns of $\Lambda$. Define the associated quasi-toric manifold of $(P,\Lambda)$ by $$M(P,\Lambda)=\frac{P\times T^n}{\sim}$$ where $(p,t_1)\sim(p,t_2)$ if $t_2^{-1}t_1\in\langle c_i\;|\;p\in F(p)\rangle\cong T^k\leq T^n$ and $F(p)$ is the unique facet such that $p\in\text{relint}(F(p))$ for any $p\in P$. 
\end{defn}

\begin{thm}{Davis-Januszkiewicz Classification Theorem}{}\\
There is a one to one bijection $$\frac{\{M\;|\;M\text{ is a quasitoric manifold}\}}{\cong}\;\;\overset{1:1}{\leftrightarrow}\;\;\frac{\{(P,\Lambda)\text{ is a characteristic pair}\}}{\sim}$$ where $(P,\Lambda)\sim(Q,\Gamma)$ if $P$ and $Q$ are combinatorially equivalent and there exists $A\in\text{GL}(n,\Z)$ with $\det(A)=\pm1$ such that $A\Lambda$ and $Q$ are equal up to permutation of columns. 
\end{thm}

\subsection{Basic Structures}
\begin{prp}{}{}\\
Let $M$ be a quasitoric manifold with characteristic pair $(P,\Lambda)$. Let $F_1,\dots,F_m$ be the facets of $P$. Then there is an isomorphism of graded rings $$H^\ast(M;\Z)\cong\frac{\Z[x_1,\dots,x_m]}{I_\text{SR}+I_\Lambda}$$ where 
\begin{itemize}
\item $\deg(x_i)=2$ for $1\leq i\leq m$.

\subsection{Equivalent Definitions}
\begin{defn}{Kahler Forms}{}\\
Let $(M,h)$ be a complex Hermitian manifold. Let $\omega$ be the associated form. We say that $\omega$ is a Kahler form if $d\omega=0$. In this case, we say that $M$ is a Kahler manifold. 
\end{defn}

\begin{lmm}{}{}\\
Let $M$ be a Kahler manifold. Then the following are true. 
\begin{itemize}
\item Let $g=\frac{1}{2}(h+\overline{h})$ be the associated Riemannian metric. Then $(M,g)$ is a Riemannian manifold. 
\item Let $\omega=\frac{i}{2}(h-\overline{h})$ be the associated $(1,1)$-form. Then $(M,\omega)$ is a symplectic manifold. 
\end{itemize}
\end{lmm}

\begin{prp}{}{}\\
Let $(M,\omega)$ be a symplectic manifold. Then $\omega$ is a Kahler form if and only if $M$ admits an integrable almost complex structure $J$. In this case, the following are true. 
\begin{itemize}
\item The Riemannian metric is given by $g(u,v)=\omega(u,Jv)$. 
\item The Hermitian metric is given by $h=g-i\omega$. 
\end{itemize}
\end{prp}

\begin{prp}{}{}\\
Let $(M,g)$ be a Riemannian manifold of dimension $2n$. Then $M$ admits a Kahler form if and only if $\text{Hol}(\nabla^g)\leq U(n)$. 
\end{prp}

\begin{eg}{}{}\\
$\C\Prj^n$ together with the Fubini-Study metric is a Kahler manifold. 
\end{eg}

\begin{eg}{}{}\\
Every Riemann surfaces is a Kahler manifold. 
\end{eg}

\begin{thm}{Hodge Decomposition Theorem}{}\\
Let $M$ be a compact Kahler manifold. Then the following are true. 
\begin{itemize}
\item There is an isomorphism $$H_\text{dR}^k(M;\C)\cong\bigoplus_{p+q=k}H^{p,q}(M)$$ for all $k\in\N$. 
\item There is an isomorphism $$H^{p,q}(M)\cong\overline{H^{q,p}(M)}$$ for all $p,q\in\N$. 
\item $\dim_\R(H_\text{dR}^k(M))$ is even for all $k\in\N$. 
\end{itemize}
\end{thm}

\subsection{Algebra, Coalgebra and Bialgebra}
Let $R$ be a commutative ring. Recall that an $R$-algebra is a ring $A$ together with a ring homomorphism $R\to A$. Equivalently, it is an $R$-module with multiplication that is associative and respects the $R$-module structure. 

\begin{prp}{}{}\\
Let $R$ be a commutative ring. Let $A\in\bold{Mod}_R$ be an $R$-module. Let $m:A\otimes_R A\to A$ and $u:R\to A$ be  $R$-module homomorphisms. Then $(A,m,u)$ is an associative algebra with multiplication given by $m$ and identity given $u(1)$ if and only if the following diagrams commute: \\~\\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	{(A\otimes_RA)\otimes_RA} & {A\otimes_RA} & A & {R\otimes_RA} & {A\otimes_RA} & {A\otimes R} \\
	{A\otimes_R(A\otimes_RA)} & {A\otimes_RA} &&& A
	\arrow["{m\otimes 1}", from=1-1, to=1-2]
	\arrow["\alpha"', from=1-1, to=2-1]
	\arrow["m", from=1-2, to=1-3]
	\arrow["{u\otimes\text{id}_A}", from=1-4, to=1-5]
	\arrow["\lambda"', from=1-4, to=2-5]
	\arrow["m"', from=1-5, to=2-5]
	\arrow["{\text{id}_A\otimes u}"', from=1-6, to=1-5]
	\arrow["\mu", from=1-6, to=2-5]
	\arrow["{1\otimes m}"', from=2-1, to=2-2]
	\arrow["m"', from=2-2, to=1-3]
\end{tikzcd}} \\
where $\alpha$ is the natural isomorphism from associativity of tensor products, and $\lambda,\mu$ is the natural isomorphism from the unitality of tensor products. 
\end{prp}

Equivalently, the above proposition is the same as saying $(A,m,u)$ is a monoid object in the monoidal category $(\bold{Mod}_R,\otimes_R,R)$. Motivated by the categorical description of an $R$-algebra, we define coalgebras as the formal dual of the above object in the following way. 

\begin{defn}{Coalgebras}{}\\
Let $R$ be a commutative ring. Let $C\in\bold{Mod}_R$ be an $R$-module. Let $\Delta:C\to C\otimes_RC$ and $\varepsilon:C\to R$ be $R$-module homomorphisms. We say that $(C,\Delta,\varepsilon)$ is a coalgebra if the following diagrams commute: \\~\\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	C & {C\otimes_RC} & {(C\otimes_RC)\otimes_RC} && C \\
	& {C\otimes_RC} & {C\otimes_R(C\otimes_RC)} & {R\otimes_RC} & {C\otimes C} & {C\otimes_RR}
	\arrow["\Delta", from=1-1, to=1-2]
	\arrow["\Delta"', from=1-1, to=2-2]
	\arrow["{\Delta\otimes 1}", from=1-2, to=1-3]
	\arrow["\alpha", from=1-3, to=2-3]
	\arrow["\lambda"', from=1-5, to=2-4]
	\arrow["\Delta", from=1-5, to=2-5]
	\arrow["\mu", from=1-5, to=2-6]
	\arrow["{1\otimes\Delta}"', from=2-2, to=2-3]
	\arrow["{\varepsilon\otimes\text{id}_A}", from=2-5, to=2-4]
	\arrow["{\text{id}_A\otimes\varepsilon}"', from=2-5, to=2-6]
\end{tikzcd}} \\
where $\alpha$ is the natural isomorphism from associativity of tensor products, and $\lambda,\mu$ is the natural isomorphism from the unitality of tensor products. 
\end{defn}

\begin{defn}{Bialgebras}{}\\
Let $R$ be a commutative ring. Let $B\in\bold{Mod}_R$ be an $R$-module. Let $m:B\otimes_RB\to B$, $u:R\to B$, $\Delta:B\to B\otimes_RB$ and $\varepsilon:B\to R$ be $R$-module homomorphisms. We say that $B$ is an $R$-bialgebra if the following are true. 
\begin{itemize}
\item $(B,m,u)$ is an $R$-algebra. 
\item $(B,\Delta,\varepsilon)$ is an $R$-coalgebra. 
\item Multiplication and comultiplication are compatible: \\~\\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	{B\otimes_RB} & B & {B\otimes_RB} \\
	{B\otimes_RB\otimes_RB\otimes_RB} && {B\otimes_RB\otimes_RB\otimes_RB}
	\arrow["m", from=1-1, to=1-2]
	\arrow["{\Delta\otimes\Delta}"', from=1-1, to=2-1]
	\arrow["\Delta", from=1-2, to=1-3]
	\arrow["{\text{id}\otimes\tau\otimes\text{id}}"', from=2-1, to=2-3]
	\arrow["{m\otimes m }"', from=2-3, to=1-3]
\end{tikzcd}} \\
where $\tau$ is the natural isomorphism from commutativity of tensor products. 
\item Multiplication and counit are compatible: \\~\\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	{B\otimes_RB} & B \\
	{R\otimes_RR} & R
	\arrow["m", from=1-1, to=1-2]
	\arrow["{\varepsilon\otimes\varepsilon}"', from=1-1, to=2-1]
	\arrow["\varepsilon", from=1-2, to=2-2]
	\arrow["\lambda"', from=2-1, to=2-2]
\end{tikzcd}} \\
where $\lambda$ is the natural isomorphism from unitality of tensor products. 
\item Comultiplication and unit are compatible: \\~\\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	R & B \\
	{R\otimes_RR} & {B\otimes_RB}
	\arrow["u", from=1-1, to=1-2]
	\arrow["\lambda"', from=1-1, to=2-1]
	\arrow["\Delta", from=1-2, to=2-2]
	\arrow["{\Delta\otimes\Delta}"', from=2-1, to=2-2]
\end{tikzcd}} \\
where $\lambda$ is the natural isomorphism from unitality of tensor products. 
\item Unit and counit are compatible: \\~\\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	R && R \\
	& B
	\arrow["{\text{id}}", from=1-1, to=1-3]
	\arrow["u"', from=1-1, to=2-2]
	\arrow["\varepsilon"', from=2-2, to=1-3]
\end{tikzcd}} \\
\end{itemize}
\end{defn}

\subsection{Hopf Algebra}
\begin{defn}{Hopf Algebras}{}\\
Let $R$ be a commutative ring. Let $(H,m,u,\Delta,\varepsilon)$ be an $R$-bialgebra. Let $S:H\to H$ be an $R$-module homomorphism. We say that $(H,m,u,\Delta,\varepsilon,S)$ is a Hopf algebra if the following diagram commutes: \\~\\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	& {H\otimes_RH} && {H\otimes_RH} \\
	H && R && H \\
	& {H\otimes_RH} && {H\otimes_RH}
	\arrow["{S\otimes\text{id}_H}", from=1-2, to=1-4]
	\arrow["m", from=1-4, to=2-5]
	\arrow["\Delta", from=2-1, to=1-2]
	\arrow["\varepsilon", from=2-1, to=2-3]
	\arrow["\Delta"', from=2-1, to=3-2]
	\arrow["u", from=2-3, to=2-5]
	\arrow["{\text{id}_H\otimes S}"', from=3-2, to=3-4]
	\arrow["m"', from=3-4, to=2-5]
\end{tikzcd}} \\
\end{defn}

\begin{defn}{Commmutative and Cocommutative}{}\\
Let $R$ be a commutative ring. Let $H$ be a Hopf algebra. 
\begin{itemize}
\item We say that $H$ is commutative if the underlying algebra is commutative. 
\item We say that $H$ is cocommutative if the underlying coalgebra is cocommutative. 
\end{itemize}
\end{defn}

\begin{prp}{}{}\\
Let $R$ be a commutative ring. Let $H$ be a Hopf algebra. If $H$ is a projective and finitely generated $R$-modue, then $\Hom_R(H,R)$ is a Hopf algebra. Moreover, the following are true. 
\begin{itemize}
\item $\Hom_R(M,R)$ is commutative if and only if $H$ is cocommutative. 
\item $\Hom_R(M,R)$ is cocommutative if and only if $H$ is commutative. 
\end{itemize}
\end{prp}

\subsection{Differential Graded Algebras}
\begin{defn}{Differential Graded Algebras}{}\\
Let $R$ be a commutative ring. A differential graded $R$-algebra consists of the following data. 
\begin{itemize}
\item A $\Z$-graded $R$-algebra $A$. 
\item An $R$-algebra homomorphism of degree $-1$ (called homological dgas) or of degree $1$ (called cohomological dgas). 
\end{itemize}
such that the following are true. 
\begin{itemize}
\item Differential: $d\circ d=0$. 
\item Graded Leibniz: $d(ab)=(da)b+(-1)^{\deg(a)}a(db)$ for $a,b$ homogeneous elements of $A$. 
\end{itemize}
\end{defn}

The differential property means that any (homological) differential graded algebra decomposes into a chain complex whose differential is given by $d$. 

\begin{defn}{Homomorphisms of Differential Graded Algebras}{}\\
Let $R$ be a commutative ring. Let $A,B$ be differential graded algebras. A morphism $\varphi:A\to B$ of differential graded algebras is a graded $R$-algebra homomorphism. 
\end{defn}

\begin{lmm}{}{}\\
Let $R$ be a commutative ring. Let $(A,d)$ be a cohomological differential graded algebra. Then $H^\ast(A,d)=\bigoplus_{n\in\Z}H^n(A,d)$ is a graded ring with multiplication defined by $$[u]\cdot[v]=[uv]$$ for $u\in H^n(A,d)$ and $v\in H^m(A,d)$. 
\begin{proof}\\
Notice that by the graded Leibniz rule, we have $d(uv)=(du)v+(-1)^nud(v)=0$ since $u$ and $v$ are cocycles. Hence $uv$ is also a cocycle. 
\end{proof}
\end{lmm}

\end{lmm}

\end{prp}

\begin{defn}{The $I$-Torsion of a Module}{}\\
Let $R$ be a Noetherian commutative ring. Let $M$ be an $R$-module. Let $I$ be an ideal of $R$. Define the $I$-torsion of $M$ to be $$\Gamma_I(M)=\{m\in M\;|\;I^tm=0\text{ for some }t\in\N\}$$
\end{defn}

\begin{defn}{$I$-Torsion Modules}{}\\
Let $R$ be a Noetherian commutative ring. Let $M$ be an $R$-module. Let $I$ be an ideal of $R$. We say that $M$ is $I$-torsion if $M=\Gamma_I(M)$. 
\end{defn}

\begin{defn}{The Torsion Functor}{}\\
Let $R$ be a Noetherian commutative ring. Let $I$ be an ideal of $R$. Define the $I$-torsion functor $$\Gamma_I:{_R}\bold{Mod}\to{_R}\bold{Mod}$$ by the following. 
\begin{itemize}
\item For each $R$-module $M$, the functor sends $M$ to the $I$-torsion $\Gamma_I(M)$. 
\item For each $R$-module homomorphism $f:M\to N$, the functor sends $f$ to $\Gamma_I(f)=f|_{\Gamma_I(M)}$. 
\end{itemize}
\end{defn}

\begin{lmm}{}{}\\
Let $R$ be a Noetherian commutative ring. Let $I$ be an ideal of $R$. Then $\Gamma_I$ is a left exact functor. 
\end{lmm}

\begin{defn}{Local Cohomology}{}\\
Let $R$ be a Noetherian commutative ring. Let $M$ be an $R$-module. Let $I$ be an ideal of $R$. Define the $n$th local cohomology of $M$ with support in $I$ to be $$H_I^n(M)=(R^n\Gamma_I)(M)$$
\end{defn}

\begin{prp}{}{}\\
Let $R$ be a Noetherian commutative ring. Let $M$ be an $R$-module. Let $I,J$ be ideals of $R$. If $\sqrt{I}=\sqrt{J}$, then the equality $\Gamma_I(M)=\Gamma_J(M)$ induces an equality $$H_I^n(M)=H_J^n(M)$$ for all $n\in\N$. 
\end{prp}

Thus we only need to consider the local cohomology with support in radical ideals. 

\begin{lmm}{}{}\\
Let $R$ be a Noetherian commutative ring. Let $M$ be an $R$-module. Let $I$ be an ideal of $R$. Then $H_I^n(M)$ is $I$-torsion for all $n\in\N$. 
\end{lmm}

\begin{lmm}{}{}\\
Let $R$ be a Noetherian commutative ring. Let $M_i$ be $R$-modules for each $i\in J$. Let $I$ be an ideal of $R$. Then there is a natural isomorphism $$H_I^n\left(\bigoplus_{i\in J}M_i\right)\cong\bigoplus_{i\in J}H_J^n(M_i)$$
\end{lmm}

\begin{prp}{}{}\\
Let $R$ be a Noetherian commutative ring. Let $M$ be an $R$-module. Let $I$ be an ideal of $R$.Then there are natural isomorphisms $$\varinjlim_{i\in\N}\text{Ext}_R^n(R/I^i,M)\cong H_I^n(M)$$ for all $n\in\N$. 
\end{prp}

\begin{defn}{Morita Equivalence}{}\\
Let $R,S$ be rings. We say that $R$ and $S$ are Morita equivalent if there exists an equivalence of categories $${_R}\bold{Mod}\simeq{_S}\bold{Mod}$$
\end{defn}

\begin{lmm}{}{}\\
Let $R,S$ be rings. Then $R$ and $S$ are Morita equivalent if and only if there exists an equivalence of categories $$\bold{Mod}_R\simeq\bold{Mod}_S$$
\end{lmm}

\begin{lmm}{}{}\\
Let $R,S$ be rings. Suppose that $R$ and $S$ are Morita equivalent. Let $F:{_R}\bold{Mod}\to{_S}\bold{Mod}$ be an equivalence. Then $F$ is additive. 
\end{lmm}

\subsection{Progenerators}
\begin{defn}{Progenerators}{}\\
Let $R$ be a ring. Let $P$ be an $R$-module. We say that $P$ is a progenerator of $R$ if the following are true. 
\begin{itemize}
\item $P$ is finitely generated. 
\item $P$ is projective. 
\end{itemize}
\end{defn}

\begin{prp}{}{}\\
Let $R,S$ be rings. Let $F:{_R}\bold{Mod}\to{_S}\bold{Mod}$ and $G:{_S}\bold{Mod}\to{_R}\bold{Mod}$ be additive functors. Then $F$ and $G$ are inverses if and only if there exists an $(S,R)$-bimodule $P$ such that the following are true. 
\begin{itemize}
\item ${_S}P$ is a progenerator. 
\item $P_R$ is a progenerator. 
\item There are natural isomorphisms $F\cong P\otimes_R-$ and $G\cong\Hom({_S}P,-)$. 
\end{itemize}
\end{prp}

\begin{thm}{Morita}{}\\
Let $R,S$ be rings. Then the following are equivalent. 
\begin{itemize}
\item $R$ and $S$ are Morita equivalent. 
\item There exists a progenerator $R$-module $P_R$ such that $S\cong\text{End}_R(P_R)$. 
\item There exist an $(R,S)$-bimodule $M$ and a $(S,R)$-bimodule $N$ such that $M\otimes_SN\cong R$ as $(R,R)$-bimodules and $N\otimes_RM\cong S$ as $(S,S)$-bimodules. 
\end{itemize}
\end{thm}

\subsection{Properties of Morita Equivalence}
\begin{prp}{}{}\\
Let $R,S$ be rings. If $R$ and $S$ are Morita equivalent, then the following are true. 
\begin{itemize}
\item There is a ring isomorphism $Z(R)\cong Z(S)$. 
\item $R/J(R)$ is Morita equivalent to $S/J(S)$. 
\end{itemize}
\end{prp}

\begin{prp}{}{}\\
Let $R,S$ be rings. Let $M$ be an $R$-module. Suppose that $R$ and $S$ are Morita equivalent via the functor $F:{_R}\bold{Mod}\to{_S}\bold{Mod}$. Then the following are true. 
\begin{itemize}
\item $M$ is injective if and only if $F(M)$ is injective. 
\item $M$ is projective if and only if $F(M)$ is projective. 
\item $M$ is simple if and only $F(M)$ is simple. 
\item $M$ is semisimple if and only if $F(M)$ is semisimple. 
\item $M$ is finitely generated if and only if $F(M)$ is finitely generated. 
\item $M$ is Noetherian if and only if $F(M)$ is Noetherian. 
\item $M$ is Artinian if and only if $F(M)$ is Artinian. 
\end{itemize}
\end{prp}

\begin{prp}{}{}\\
Let $R,S$ be rings. Suppose that $R$ and $S$ are Morita equivalent. Then the following are equivalent. 
\begin{itemize}
\item $R$ is simple if and only if $S$ is simple. 
\item $R$ is semisimple if and only if $S$ is semisimple. 
\item $R$ is Noetherian if and only if $S$ is Noetherian. 
\item $R$ is Artinian if and only if $S$ is Artinian. 
\end{itemize}
\end{prp}

\begin{prp}{}{}\\
Let $R,S$ be rings. Then $R$ and $S$ are Morita equivalent if and only if ${_R}\bold{FGMod}\simeq{_S}\bold{FGMod}$. 
\end{prp}

\section{Complex Analytic Space}
\subsection{Complex Analytic Varieties}
\begin{defn}{Complex Analytic Varieties}{}\\
Let $Z\subseteq\C^n$. We say that $Z$ is a complex analytic variety if there exists an open set $Z\subseteq U\subseteq\C^n$ and $f_1,\dots,f_k\in\mO_{\C^n}(U)$ such that $$Z=\V(f_1,\dots,f_k)$$
\end{defn}

We use a subset of $C^n$ in the definition because in $\C^n$, locally defined functions cannot in general be defined globally (compare it to the real case, where they have partitions of unity and bump functions). \\

Indeed, for smooth functions on $\R^n$, a function on an open set $U$ can be modified (via bump functions) to produce a global function that agrees with it on a smaller open subset. You can always find a global function that captures the local behavior near any point. \\

For holomorphic functions on $\C^n$: no such modification is possible. A holomorphic function cannot be cut off or modified while remaining holomorphic, so local holomorphic data is genuinely trapped in its domain.

\begin{defn}{The Structure Sheaf}{} Let$ U\subseteq\C^n$ be an open subset. Let $f_1,\dots,f_k\in\mA_{\C^n}(U)$ be analytic functions on $U$. Let $Z=\V(f_1,\dots,f_k)$ be the vanishing locus of $f_1,\dots,f_k$. Define the structure sheaf $$\mO_Z:\bold{Open}(Z)\to\bold{CRing}$$ as follows. 
\begin{itemize}
\item For each open set $U\subseteq Z$, define $$\mO_Z(U)=\frac{\mO_{\C^n}(U)}{(f_1,\dots,f_k)}$$
\item For each inclusion $U\subseteq V\subseteq Z$ of open sets, define $$\mO_Z(V)\to\mO_Z(U)$$ to be the induced map of the restriction $\mO_{\C^n}(V)\to\mO_{\C^n}(U)$. 
\end{itemize}
\end{defn}

\subsection{Complex Analytic Subvarieties of Manifolds}
\begin{defn}{Complex Analytic Subvarieties}{}\\
Let $M$ be a complex manifold. Let $Z\subseteq M$ be a subset. We say that $Z$ is a complex analytic subvariety of $M$ if for each $p\in Z$, there exists a chart $(U,\phi)$ of $M$ such that $\phi(Z\cap U)$ is a complex analytic variety. 
\end{defn}

In particular, complex analytic subvarieties of $\C^n$ are just complex analytic varieties. 

\begin{thm}{Cartan's (Coherence) Theorem}{} Let $M$ be a complex manifold and let $A$ be an analytic subset of $M$. Then $\mI_A$ is a coherent sheaf of $\mO_M$-modules. 
\end{thm}

\subsection{Complex Analytic Spaces}
\begin{defn}{Complex Analytic Spaces}{} Let $(X,\mO_X)$ be a locally ringed space. We say that $X$ is a complex analytic space if for all $p\in X$, there exists an open neighbourhood $U\subseteq X$ such that $(U,\mO_X|_U)$ is a complex analytic space. 
\end{defn}

\begin{lmm}{}{}\\
Let $(X,\mO_X)$ be a complex analytic space. Then $X$ is a complex manifold if and only if for each $p\in X$, there exists an open neighbourhood $U\subseteq X$ such that there is an isomorphism $$(U,\mO_X|_U)\cong(V,\mO_{\C^n}|_V)$$ for some open subset $V\subseteq\C^n$. 
\end{lmm}

\begin{thm}{}{} Let $X$ be a complex space. Then $X_\text{reg}$ is dense and open in $X$. Moreover, $X_\text{reg}$ consists of a disjoin union of complex manifolds. 
\end{thm}

\begin{thm}{}{} Let $X$ be an irreducible complex space. Then every non-constant holomorphic function $f:X\to\C$ is an open map. 
\end{thm}

\begin{crl}{}{} Let $X$ be an irreducible compact complex space. Then every holomorphic function $f:X\to\C$ on $X$ is constant. 
\end{crl}

\begin{defn}{Complex Analytic Subvarieties}{}\\
Let $(X,\mO_X)$ be a complex analytic space. Let $Z\subseteq X$ be subset. We say that $Z$ is a complex analytic subvariety if for all $p\in X$, there exists an open neighbourhood $U\subseteq X$ and $f_1,\dots,f_k\in\mO_X(U)$ such that $$Z\cap U=\V_U(f_1,\dots,f_k)$$
\end{defn}

This definition generalizes complex analytic subvarieties of complex manifolds. 

\subsection{Connections to Algebraic Geometry}
Recall that the category of schemes and the category of analytic spaces are both full subcategories of locally ringed spaces. 

\begin{defn}{The Analytification of a Scheme}{}\\
Let $X$ be a scheme locally of finite type over $\C$. Define the analytification of $X$ to be the unique analytic space $X^\text{an}$ together with a morphism of locally ringed space $\alpha:X^\text{an}\to X$ satisfying the following universal property: For any analytic space $Y$ and any morphism of locally ringed spaces $f:Y\to X$, there exists a unique morphism of analytic spaces $g:Y\to X^\text{an}$ such that the following diagram commutes: \\~\\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	& Y \\
	{X^\text{an}} & X
	\arrow["{\exists!g}"', dashed, from=1-2, to=2-1]
	\arrow["f", from=1-2, to=2-2]
	\arrow["\alpha"', from=2-1, to=2-2]
\end{tikzcd}}\\
\end{defn}

Recall that the Yoneda embedding $h:\bold{LRS}\to\bold{Set}^{\bold{LRS}}$ is the functor sending a locally ringed space $X$ to the functor $h_X:\bold{LRS}\to\bold{Set}$ defined by $h_X(Y)=\Hom_{\bold{LRS}}(Y,X)$. Then the universal property of the analytification means that the functor $h_X$ restricted to analytic spaces $\bold{An}$ is representable. \\

One thinks of the morphism $X^\text{an}\to X$ as doing two things. It is an enlargement of the standard topology of $X^\text{an}$ to the Zariski topology of $X$, and it introduces generic points outside of $X(\C)$ (Indeed we will see that $X^\text{an}$ has the same underlying set as $X(\C)$). 

\begin{thm}{}{}\\
Let $X$ be a scheme locally of finite type over $\C$. Then the analytification of $X$ exists and is unique up to unique isomorphism. 
\end{thm}

\begin{defn}{The Analytification Functor}{}\\
Define the analytification functor $$(-)^\text{an}:\bold{Sch}_\C\to\bold{An}$$ as follows. 
\begin{itemize}
\item For each scheme $X$ locally of finite type over $\C$, we have $X^\text{an}$ is the analytification of $X$. 
\item For a morphism of schemes $f:X\to Y$ over $\C$, the universal propertyof $Y^\text{an}$ induces a morphism $X\to Y^\text{an}$. Precomposing with the canonical morphism $X^\text{an}\to Y^\text{an}$ gives the desired map $X^\text{an}\to Y^\text{an}$. 
\end{itemize}
\end{defn}

\begin{prp}{}{}\\
Let $X$ be a scheme locally of finite type over $\C$. Then the underlying set of $X^\text{an}$ is given by $X(\C)$. 
\end{prp}

\begin{prp}{}{}\\
Let $X$ be a scheme locally of finite type over $\C$. Then the following are true. 
\begin{itemize}
\item $X$ is separated if and only if $X^\text{an}$ is Hausdorff. 
\item $X$ is connected if and only if $X^\text{an}$ is connected. 
\item $X$ is smooth if and only if $X^\text{an}$ is a complex manifold. 
\item $X$ is proper if and only if $X^\text{an}$ is compact. 
\end{itemize}
\end{prp}

\begin{prp}{}{}\\
Let $X$ be a variety over $\C$. Then the following are true. 
\begin{itemize}
\item If $X$ is irreducible, then $X^\text{an}$ is connected. 
\item If $X$ is smooth and affine, then $X^\text{an}$ is a Stein manifold. 
\item If $X$ is smooth and projective, then $X^\text{an}$ is Kahler. 
\end{itemize}
\end{prp}

\begin{thm}{GAGA}{}\\
Let $X$ be a proper scheme of finite type over $\C$. Then the following are true. 
\begin{itemize}
\item Let $\mF$ be a sheaf of $\mO_X$-modules. Then $\mF^\text{an}$ is a sheaf of $\mO_{X^\text{an}}$-module. 
\item Let $\mF$ be a sheaf of $\mO_X$-modules. There is a natural isomorphism $$H^i(X,\mF)\cong H^i(X^\text{an},\mF^\text{an})$$
\item The analytification functor induces an equivalence of categories $$\bold{Coh}_{\mO_X}\cong\bold{Coh}_{\mO_{X^\text{an}}}$$
\item Let $Y\subseteq X^\text{an}$ be a closed complex analytic subspace. Then there exists a closed subvariety $Z$ of $X$ such that $Z^\text{an}=Y$. 
\end{itemize}
\end{thm}

\begin{crl}{Chow's Theorem}{}\\
Let $X$ be a closed complex analytic subspace of $\C\Prj^n$. Then $X$ is a projective variety over $\C$. 
\end{crl}

The important point is that: the theory of smooth projective complex varieties are subsumed by the theory of Kahler manifolds. Divisors and line bundles correspond bijectively thanks to Chow's theorem. 

\begin{thm}{Grothendieck}{}\\
Let $X$ be a smooth variety over $\C$. Then there is a natural isomorphism $$H_\text{dR}^n(X)\cong H_\text{Holo}^n(X^\text{an})\cong H_\text{dR}^n(X^\text{an})$$ induced by the canonical map $\alpha:X^\text{an}\to X$. 
\end{thm}

\begin{crl}{}{}\\
Let $X$ be a smooth variety over $\C$. Then the above isomorphism and de Rham's theorem induces a natural isomrophism $$H_\text{dR}^n(X)\cong H_\text{sing}^n(X^\text{an};\C)$$
\end{crl}

\begin{thm}{The Period Isomorphism}{}\\
Let $k$ be a subfield of $\C$. Let $X$ be a smooth variety over $k$. Then there is a natural isomorphism $$H_\text{dR}^\ast(X)\otimes_k\C\cong H_\text{sing}^\ast(X^\text{an};\Q)\otimes_\Q\C$$
\end{thm}

\subsection{Divisors}
\begin{defn}{Divisors on a Complex Manifold}{}\\
Let $X$ be a complex manifold. A divisor on $X$ is a formal linear combination $$D=\sum_{\substack{V\text{ is an analytic variety of }X\\\text{ of dimension }\dim(X)-1}}n_V\cdot V$$ where $n_V\in\Z$, and such that for each $p\in X$, there exists an open neighbourhood $U$ of $p$ such that $U$ meets finitely many $V$ in the summand. 
\end{defn}

The locally finite condition here is required because in Algebraic Geometry, Noetherianess is assumed, otherwise they are the same. 

\begin{defn}{Group of Divisors}{}\\
Let $X$ be a complex manifold. Define the group of divisors of $X$ to be the set $$\text{Div}(X)=\{D\;|\;D\text{ is a divisor on }X\}$$ together with addition given as a subgroup of $\prod_V\Z\cdot V$. 
\end{defn}

\begin{defn}{Effective Divisors}{}\\
Let $X$ be a complex manifold. Let $D=\sum_Vn_V\cdot V\in\text{Div}(X)$ be a divisor. We say that $D$ is effective if $n_V\geq 0$ for all $V$. 
\end{defn}

\begin{defn}{Divisors of Meromorphic Functions}{}\\
Let $X$ be a complex manifold. Let $f\in\mM_X^\ast$ be a meromorphic function. Define the divisor of $f$ to be $$\text{div}(f)=\sum_V\nu_V(f)\cdot V$$ where $\nu_V$ is the valuation of $\mO_{X,V}$. 
\end{defn}

\begin{defn}{Cartier Divisors on a Complex Manifold}{}\\
Let $X$ be a complex manifold. Define the group of divisors of $X$ to be $$\text{CDiv}(X)=H^0(X,\mM_X^\ast/\mO_X^\ast)=\Gamma(X,\mM_X^\ast/\mO_X^\ast)$$
\end{defn}

\begin{prp}{}{}\\
Let $X$ be a complex manifold. Then there is a natural isomorphism $$\text{Div}(X)\cong\text{CDiv}(X)$$
\end{prp}

where the bottom horizontal map is induced by the long exact sequence of the short exact sequence $0\to\mO_X^\ast\to\mM_X^\ast\to\frac{\mM_X^\ast}{\mO_X^\ast}\to 0$. 
\end{prp}

	\arrow[from=1-4, to=1-5]
\end{tikzcd}} \\
\end{defn}

Let $M$ be a complex manifold. Let $L$ be a holomorphic line bundle on $M$. Then the first chern class of $L$ is given by the image of $L$ in the homomorphism $$\text{Pic}(M)\cong H^1(M,\mO_M^\ast)\to H^2(M,\underline{\Z})\cong H^2(M,\Z)$$ induced by the long exact sequence of the exponential sequence. 
\end{prp}

\section{Compact Kahler Manifolds}
\subsection{Kodaira Embedding Theorem}

\begin{prp}{}{}\\
Let $X$ be a smooth projective variety over $\C$. Let $\mL$ be a line bundle over $X$. Then $\mL$ is ample if and only if $\mL^\text{an}$ is positive. 
\end{prp}

\begin{thm}{Kodaira Embedding}{}\\
Let $X$ be a compact complex manifold. Then the following are equivalent. 
\begin{itemize}
\item $X$ admits a holomorphic embedding into $\C\Prj^n$. 
\item $X$ admits an ample line bundle. 
\item $X$ admits a positive line bundle. 
\item $X$ is admits a Kahler metric $\omega$ such that $[\omega]\in H^{1,1}(X,\Z)$. 
\item There exists a projective variety $Z$ over $\C$ such that $Z^\text{an}=X$. 
\end{itemize}
\end{thm}

\begin{crl}{}{}\\
There is an equivalence of categories between smooth projective varieties over $\C$ and compact Kahler manifolds given by the analytification functor. 
\end{crl}

\begin{thm}{Kodaira-Nakano Vanishing Theorem}{}\\
Let $X$ be a compact Kahler manifold. Let $p:L\to X$ be a positive line bundle. Then we have $$H^q(X,\Omega^p(L))=0$$ for $p+q>\dim(X)$.

\begin{defn}{The Vertical Bundle}{}\\

\end{lmm}

\begin{defn}{Horizontal Distributions}{}\\
Let $M$ be a smooth manifold. Let $G$ be a Lie group. Let $\pi:P\to M$ be a smooth principal bundle over $G$. A horizontal distribution on $P$ is a subbundle $HP\subseteq TP$ such that the following are true. 
\begin{itemize}
\item $TP=VP\oplus HP$. 
\item $(HP)_{g\cdot p}=(L_g)_\ast(HP)_p$ for all $p\in P$ and $g\in G$. 
\end{itemize}
\end{defn}

Generalizes connections on vector bundles in the following way: For $p:E\to M$ a smooth vector bundle, a connection on $\text{Fr}(E)$ as a principal $\text{GL}(n,F)$ bundle is the same data as a connection on $p:E\to M$ as a vector bundle. 

\begin{defn}{Connection Forms on Principal Bundles}{}\\
Let $M$ be a smooth manifold. Let $G$ be a Lie group. Let $\pi:P\to M$ be a smooth principal bundle over $G$. A connection $1$-form on $P$ is a smooth section $\omega\in\mC^\infty(T^\ast P\otimes\text{Lie}(G))$ such that the following are true. 
\begin{itemize}
\item We have $\omega_p(\iota_p)=\text{id}_{\text{Lie}(G)}$ for all $p\in P$. 
\item We have $$\omega_{g\cdot p}((dL_g)_p(v))=\text{Ad}(g)(\omega_p(v))$$ for all $g\in G$, $p\in P$ and $v\in T_pP$. 
\end{itemize}
\end{defn}

Let $M$ be a smooth manifold. Let $G$ be a Lie group. Let $\pi:P\to M$ be a smooth principal bundle over $G$. Then the following are true.

\end{itemize}
\end{prp}

\section{The Logical Chain of Classical Field Theory and Gauge Theory}

\subsection*{Step 1: The Field}

We observe a physical quantity distributed over spacetime — a temperature, a velocity, an electromagnetic strength. Mathematically this is a fiber bundle $\pi: E \to M$, where $M$ is spacetime and the fiber $E_x$ is the space of possible values of the quantity at $x$. A \textbf{section} $\phi \in \Gamma(E)$ is one complete specification of the quantity at every point — a possible physical configuration.

\subsection*{Step 2: The Lagrangian as a Choice of Physics}

A Lagrangian density $\Lambda \in \Omega^n(J^kE)$ is a choice of action functional
\[
S[\phi] = \int_\Omega (j^k\phi)^*\Lambda
\]
Different Lagrangians encode different physical theories. The Euler-Lagrange equations — the equations of motion — are derived from $\Lambda$ by the variational principle. The physical sections are the critical points of $S$.

\subsection*{Step 3: Observing a Symmetry}

You compute the Euler-Lagrange equations and notice that the Lagrangian $\Lambda$ is invariant under some subgroup of automorphisms of $E$. An \textbf{automorphism} of $\pi: E \to M$ is a fiber-preserving diffeomorphism $\Phi: E \to E$. Such an automorphism acts on sections by
\[
\phi \mapsto \Phi \circ \phi
\]
and the Lagrangian being invariant means
\[
S[\Phi \circ \phi] = S[\phi] \quad \text{for all } \phi \in \Gamma(E)
\]
At this stage, the symmetry group acts the same way at every point — it is \textbf{global}. You observe this as an empirical fact about your Lagrangian. 

Symmetries of the action induces certain laws of physics, such as conservation of energy, momentum, charge etc. This is Noether's theorem. 

Another note: Rather than starting with a Lagrangian and observe the symmetries of its action, we may also start with a set of symmetries that we would like to enforce, apply it to all differential forms and retain only the invariants, and see what Lagrangian you would get. 

\subsection*{Step 4: The Absolute vs Relative Observation}
The global symmetry exists because the field takes values in a space that has \textbf{no preferred origin or frame} — the fiber $E_x$ has a group of self-maps (rotations, phase shifts, color rotations) that are physically unobservable. Different observers, or different conventions, would describe the same physical situation using sections related by this group action. The global symmetry is an artifact of having made consistent conventional choices everywhere at once. We need not restrict ourselves to consider symmetries of the entire space, and this is supported by physics: 

\subsubsection*{Argument 1: Special Relativity}

Special relativity says there is no notion of simultaneity — two spacelike separated points have no canonical notion of ``the same moment.'' A global transformation is therefore not a physically coherent operation in a relativistic theory. If you believe in special relativity, you are forced to allow your symmetry to act differently at different points. The global symmetry is then a special case of the local one, not the fundamental object.

\subsubsection*{Argument 2: The Gauge Principle}

The difference induced by the symmetry of the action is not observable. The phase is therefore a \textbf{convention} --- a choice made by an observer at $x$ to  describe the field. If the phase is a convention, then different observers at different points of spacetime should be free to make different conventional choices independently. There is no physical mechanism that would force their conventions to agree globally. The true symmetry group of the theory is therefore not the global group $G$ but the local group $mathcal{G} = \mathrm{Map}(M, G)$ of all smooth $G$-valued functions on spacetime.

This is the \textbf{gauge principle}: any redundancy in the description of a physical system should be a local redundancy. It is a postulate, not a theorem. Its justification is that it works --- every time it is applied, it produces a physically correct and predictive theory.

\subsection*{Step 5: Promoting to Local Symmetry}

This forces you to demand invariance not just under global $G$-transformations
\[
\phi(x) \mapsto g \cdot \phi(x), \quad g \in G \text{ constant}
\]
but under local $G$-transformations
\[
\phi(x) \mapsto g(x) \cdot \phi(x), \quad g: M \to G \text{ smooth}
\]

The Lagrangian, built from ordinary derivatives $\partial_\mu\phi$, immediately fails to be 
locally invariant. The reason is fundamental: $\partial_\mu\phi$ compares field values at 
nearby points $x$ and $x + dx$, which under local symmetry live in fibers with 
\textbf{independent conventions}. Subtracting them is meaningless — it is comparing 
quantities measured in different units.

\subsection*{Step 6: The Mathematics Is Now Forced}

To make sense of differentiation under local symmetry, we need a way to 
\textbf{compare vectors in nearby fibers} compatibly with the $G$-structure. But this is 
precisely the definition of a \textbf{connection}.
Where does this connection live? The answer is that the principal bundle is not introduced 
from outside — it is \textbf{already contained inside the vector bundle $E$}.

The fibers $E_x$ are vector spaces, but the isomorphism $E_x \cong \mathbb{R}^r$ is not 
canonical — it requires a choice of basis. A \textbf{frame} at $x$ is an ordered basis for 
$E_x$. The set of all frames at $x$ carries a natural free transitive right action of 
$GL(r, \mathbb{R})$, making it a $GL(r,\mathbb{R})$-torsor. Taking the disjoint union over 
all $x \in M$ produces the \textbf{frame bundle}
\[
P := \bigsqcup_{x \in M} \{\text{ordered bases for } E_x\}
\]
a principal $GL(r, \mathbb{R})$-bundle canonically constructed from $E$.

Now, the observation of a global $G$-symmetry in Step 3 is precisely the observation that 
the physically relevant frames are not arbitrary frames, but only those related by 
$G$-transformations. The structure group \textbf{reduces} from $GL(r, \mathbb{R})$ to $G$, 
producing a principal $G$-bundle
\[
P_G \subset P
\]
canonically determined by the symmetry. The original vector bundle is recovered as the 
associated bundle $E = P_G \times_G V$.

The connection demanded by local symmetry lives on $P_G$, because $P_G$ is the bundle whose

\begin{itemize}
\item $A$ is $P$-local. 
\item $A$ is a $S_P^{-1}\Z$-module by extension of scalars. 
\item The $P$-localization map $A\to A\otimes_\Z S_P^{-1}\Z$ is an isomorphism. 
\end{itemize}
\end{prp}

When $P=\emptyset$, then $S_P^{-1}\Z=\Q$ is just rationalization. 

\subsection{P-Local Spaces}
\begin{defn}{P-Local Space}{}\\
Let $X$ be a simply connected space. Let $P\subseteq\N$ be a set of prime numbers. We say that $X$ is $P$-local if $\pi_n(X)$ is a $P$-local group for all $n\geq 2$. 
\end{defn}

\begin{prp}{}{}\\
Let $X$ be a simply connected space. Let $P\subseteq\N$ be a set of prime numbers. Then the following are equivalent. 
\begin{itemize}
\item $X$ is $P$-local. 
\item $H_n(X,\ast)$ is $P$-local for all $n\geq 0$. 
\item $H_n(\Omega X,\ast)$ is $P$-local for all $n\geq 0$. 
\end{itemize}
\end{prp}

\begin{lmm}{}{}\\
Let $A$ be an abelian group. Let $n\geq 1$. Let $P\subseteq\N$ be a set of prime numbers. If $A$ is $P$-local, then we have $$H_\ast(X;\Z/p\Z)\cong H_\ast(\ast,\Z/p\Z)$$ for all $p\notin P$. 
\end{lmm}

\begin{defn}{P-Localization}{}\\
Let $X$ be a simply connected space. Let $P\subseteq\N$ be a set of prime numbers. A $P$-localization of $X$ is a space $X_P$ together with a map $i_p:X\to X_P$ such that $f$ induces an isomorphism $$\pi_n(X)\otimes_\Z S_P^{-1}\Z\to\pi_n(X_P)$$ for all $n\geq 2$. 
\end{defn}

\begin{prp}{}{}\\
Let $X$ be a simply connected space. Let $P\subseteq\N$ be a set of prime numbers. Suppose that $X_P$ exists. Then the following are true. 
\begin{itemize}
\item Suppose that $Z$ is a $P$-local space and $f:X\to Z$ is a map. Then there exists a map $g:X_P\to Z$ such that $g\circ i_p=f$. 
\item Suppose that $Z$ is a $P$-local space and $f_1,f_2:X\to Z$ are map. Suppose that $g_1,g_2:X_P\to Z$ are maps such that $g_1\circ i_p=f_1$ and $g_2\circ i_p=f_2$. Suppose that $H:X\times I\to Z$ is a homotopy from $f_1$ to $f_2$. Then there exists a homotopy $K:X_P\times I\to Z$ such that $K\circ i_p\times\text{id}_Z=H$. 
\end{itemize}
\end{prp}

\begin{prp}{}{}\\
Let $X$ be a simply connected space. Let $P\subseteq\N$ be a set of prime numbers. Then $X_P$ exists and is unique up to homotopy equivalence relative to $X$. 
\end{prp}

\end{prp}

\section{Newtonian Mechanics}
We begin with the assumption that we live in a smooth manifold of dimension $n$, together with an affine connection. 

\begin{defn}{Inertial Frame of References}{}\\
Let $M$ be a smooth manifold of dimension $n$. Let $\nabla$ be an affine connection on $M$. Let $(U,\varphi)$ be a chart of $M$. We say that $\varphi$ is an inertial frame of reference if $$\frac{d^2(\varphi\circ\gamma)}{dt^2}=0$$ for all geodesics $\gamma:I\to U$
\end{defn}

\begin{lmm}{}{}\\
Let $M$ be a smooth manifold of dimension $n$. Let $\nabla$ be an affine connection on $M$. If $M$ admits an inertial frame of reference, then $\nabla$ is flat and torsion-free. 
\end{lmm}

\begin{prp}{}{}\\
Let $M$ be a smooth manifold of dimension $n$. Let $\nabla$ be an torsion-free affine connection on $M$. Then $M$ admits a global inertial frame of reference if and only if $\nabla$ is flat and $M$ is simply connected. 
\end{prp}

Newton's first law: inertial frames exists. 

\subsection{Galilean Space}
\begin{defn}{Galilean Space-Time Structure}{}\\
Define a Galilean space-time structure to consist of the following data. 
\begin{itemize}
\item An affine space $(A,\R^4,+)$ called the universe. 
\item A linear map $\tau:\R^4\to\R$ called time. 
\item An inner product $\langle-,-\rangle:\ker(\tau)\times\ker(\tau)\to\R$. 
\end{itemize}
\end{defn}

Idea: There are no fixed point of reference in $A$ (no point of origin for both space and time). We measure time relatively by the linear map $\tau$. Elements of $\ker(\tau)$ lies in the same time frame and are called simultaneous events. We measure the spatial difference between simultaneous events by the usual inner product. 

\begin{defn}{Galilean Transformations}{}\\
Let $(A,\R^4,+,\tau,\langle-,-\rangle)$ and $(A',\R^4,+,\tau',[-,-])$ be Galilean spaces. Let $T:A\to A'$ be an affine transformation. We say that $T$ is a Galilean transformation if the following are true. 
\begin{itemize}
\item $\tau'(\overline{T}(b)-\overline{T}(a))=\tau(b-a)$ for all $a,b\in\R^4$. 
\item $[\overline{T}(b),\overline{T}(a)]=\langle a,b\rangle$ for all $a,b\in\ker(\tau)$. 
\end{itemize}
\end{defn}

\begin{defn}{Galilean Isomorphism}{}\\
Let $(A,\R^4,+,\tau,\langle-,-\rangle)$ and $(A',\R^4,+,\tau',[-,-])$ be Galilean spaces. Let $T:A\to A'$ be a Galilean transformation. We say that $T$ is an isomorphism if there exists a Galilean transformation $S$ such that $$S\circ T=\text{id}_A\;\;\text{ and }\;\;T\circ S=\text{id}_{A'}$$
\end{defn}

\begin{prp}{}{}\\
Let $(A,\R^4,+,\tau,\langle-,-\rangle)$ be a Galilean space. Then $A$ is isomorphic to $\R^4$ as a Galilean space. 
\end{prp}

\begin{defn}{The Galilean Group}{}\\
Let $(A,\R^4,+,\tau,\langle-,-\rangle)$ be a Galilean space. Define the Galilean group of $A$ to be the subgroup $$\text{Gal}(A)=\{T:A\to A\;|\;T\text{ is a Galilean transformation }\}\leq\text{Aff}(A)$$
\end{defn}

\begin{defn}{The Galilean Group for the Standard Galilean Space}{}\\
Define the Galilean group for $\R^4$ by $$\text{Gal}(3)=\{T:\R^4\to\R^4\;|\;T\text{ is a Galilean transformation }\}\leq\text{Aff}(\R^4)$$ Let $T\in\text{Gal}(3)$. 
\begin{itemize}
\item We say that $T$ is uniform motion if $T$ is of the form $$T(x,t)=(x+vt,t)$$ for some $v\in\R^3$. 
\item We say that $T$ is a translation if $T$ is of the form $$T(x,t)=(x+s_0,x+s_1)$$ for some $s_0\in\R^3$ and $s_1\in\R$. 
\item We say that $T$ is a rotation if $T$ is of the form $$T(x,t)=(Gx,t)$$ for some $G\in O(3,\R)$. 
\end{itemize}
\end{defn}

Decompose $\text{Gal}(3)$ into rotation + boost + translation

\begin{prp}{}{}\\
Let $T\in\text{Gal}(3)$. There exists a unique rotation $g_1\in\text{Gal}(3)$, a unique translation $g_2\in\text{Gal}(3)$ and a unique uniform motion $g_3\in\text{Gal}(3)$ such that $$T=g_1\circ g_2\circ g_3$$
\end{prp}

\begin{prp}{}{}\\
Let $(A,\R^4,+,\tau,\langle-,-\rangle)$ be a Galilean space. Then the following are true regarding inertial frames of references. 
\begin{itemize}
\item $A$ admits an inertial frame of reference. 
\item Let $\phi$ be an inertial frame of reference. Let $T\in\text{Gal}(A)$ be a Galilean transformation. Then $\phi\circ T$ is an inertial frame of reference. 
\item Let $\phi$ and $\psi$ be two inertial frames of references. Then there exists a unique $T\in\text{Gal}(A)$ such that $\psi=\phi\circ T$. 
\end{itemize}
\end{prp}

\subsection{Newton's Laws and Equation of Motion}
\begin{defn}{World Lines}{}\\
Let $(A,\R^4,+,\tau,\langle-,-\rangle)$ be a Galilean space. Let $r:\mC^\infty(\R,\R^3)$ be a map. We say that $r$ is a world line if there exists an inertial frame of reference $\varphi:A\to\R^4$ and a geodesic $\gamma:\R\to A$ such that $$r=\pi\circ\varphi\circ\gamma$$ where $\pi:\R^4\to\R^3$ is the projection map to the first three coordinates. 
\end{defn}

\begin{defn}{Newtonian System}{}\\
Let $(A,\R^4,+,\tau,\langle-,-\rangle)$ be a Galilean space. Let $N\in\N^+$. A Newtonian system of $N$ objects consists of the following data. 
\begin{itemize}
\item World lines $r_1,\dots,r_N\in\mC^\infty(\R,\R^3)$. 
\item Masses $m_1,\dots,m_N\in\R^+$. 
\item Force Laws $F_i:\R^{3N}\times\R^{3N}\times\R\to\R$ for $1\leq i\leq N$. 
\end{itemize}
such that $$m_i\frac{d^2r_i}{dt^2}=F_i\left(r_1,\dots,r_N,\frac{dr_1}{dt},\dots,\frac{dr_N}{dt},t\right)$$ for $1\leq i\leq N$. 
\end{defn}

The condition that $F_i$ is constant ration with $r''$ is Newton's second law. 

Postulate: Newton's principle of determinacy: global solution of of the ode exists given initial data (only local solution is guaranteed by Picard-Lindelof)

\begin{defn}{Galilean Newtonian System}{}\\
Let $(A,\R^4,+,\tau,\langle-,-\rangle)$ be a Galilean space. Let $N\in\N^+$. Let $(r_1,\dots,r_N,m_1,\dots,m_N,F_1,\dots,F_N)$ be a Newtonian system of $N$ objects. We say that the Newtonian system of $N$ objects is Galilean if $$F_i\left(Gr_1+vt+s_0,\dots,Gr_N+vt+s_0,G\frac{dr_1}{dt}+v,\dots,G\frac{dr_N}{dt}+v,t+s_1\right)=F_i\left(r_1,\dots,r_N,\frac{dr_1}{dt},\dots,\frac{dr_N}{dt},t\right)$$ for all $1\leq i\leq N$ and all $G\in O(3)$, $v,s_0\in\R^3$ and $s_1\in\R$. 
\end{defn}

Postulate: Galileo's principle of relativity. All observed forces are Galilean. (All Newtonian systems are Galilean)

\subsection{Gravitational Force}
\begin{defn}{System for Gravitational Force}{}\\
The system for gravitational force is a Newtonian system of $2$ objects whose force law is given by $$F_1=\frac{Gm_1m_2}{\abs{r_2-r_1}^3}(r_2-r_1)\;\;\text{ and }\;\;F_2=\frac{Gm_1m_2}{\abs{r_1-r_2}^3}(r_1-r_2)$$ where $G$ is the gravitational constant. 
\end{defn}

\begin{eg}{}{}\\
If the first object is a stone and the second object is earth, then $m_2>>m_1$. Then setting $F_1=m_1\frac{d^2r_1}{dt^2}$ and $F_2=m_2\frac{d^2r_2}{dt^2}$, we deduce that $$\frac{d^2r_2/dt^2}{d^2r_1/dt^2}=\frac{m_1}{m_2}\sim 0$$ This implies that and so $r_2(t)=k$ is approximately constant. Then the equation for $F_1$ reduces to $$\frac{d^2r_1}{dt^2}=-\frac{Gm_2}{\abs{r_1}^3}r_1$$ Assuming that the stone is close to the earth's surface, we have $\abs{r_1}\sim R$ where $R$ is the earth's radius. Write $r_1(t)=(x(t),y(t),z(t))$. In an appropriate inertial frame, we may decouple the equations to get $$\frac{d^2x}{dt^2}=0\;\;\text{ and }\;\;\frac{d^2y}{dt^2}=0\;\;\text{ and }\;\;\frac{d^2z}{dt^2}=-g$$ where $g=\frac{Gm_2}{R^2}$. Assuming that the stone is dropped from height $h_0$. Then solving the second order ODEs with the initial data gives $$r_1(t)=\left(0,0,h_0-\frac{1}{2}gt\right)$$ The stone hits the ground when $t=\sqrt{2h_0/g}$. 
\end{eg}

\section{Lagrangian Mechanics}
\subsection{Configuration Space}
\begin{defn}{Holonomic Constraints}{}\\
Let $N\in\N^+$. Let $(r_1,\dots,r_N,m_1,\dots,m_N,F_1,\dots,F_N)$ be a Newtonian system of $N$ objects. A holonomic constraint on the system is a function $f\in\mC^2(\R^{3N}\times\R,\R)$ such that $$f(r_1,\dots,r_N,t)=0$$
\end{defn}

\begin{defn}{Configuration Space}{}\\
Let $N\in\N^+$. Let $(r_1,\dots,r_N,m_1,\dots,m_N,F_1,\dots,F_N)$ be a Newtonian system of $N$ objects. Let $f_1,\dots,f_k$ be holonomic constrains on the system. Define the configuration space of the system to be $$Q=\{x\in\R^{3N}\;|\;f_j(x)=0\text{ for all }1\leq j\leq N\}$$
\end{defn}

Each point in the configuration space is not a point in spacetime, instead they represent a possible position for N particles to situate in space time. 

\begin{defn}{Trajectories}{}\\
Let $Q$ be a configuration space of $N$ objects. A trajectory on $Q$ is a smooth function $$\gamma:[t_1,t_2]\subseteq\R\to Q$$
\end{defn}

\begin{defn}{The Space of Trajectories}{}\\
Let $Q$ be a configuration space of $N$ objects. Define the space of trajectories of $Q$ to be $$\Omega Q=\{\gamma\;|\;\gamma\text{ is a trajectory }\}$$
\end{defn}

\subsection{The Euler Lagrange Operator}
\begin{defn}{Lagrangian}{}\\
Let $Q$ be a configuration space of $N$ objects. A Lagrangian on $Q$ is a smooth function $$L:TQ\times\R\to\R$$
\end{defn}

\begin{defn}{The Action Functional}{}\\
Let $Q$ be a configuration space of $N$ objects. Let $L:TQ\times\R\to\R$ be a Lagrangian. Let $\gamma:[t_1,t_2]\to Q$ be a trajectory. Define the action functional of $\gamma$ by $$S[\gamma]=\int_{t_1}^{t_2}L(\gamma(t),\gamma'(t),t)\;dt$$
\end{defn}

\begin{defn}{Space of Deformations}{}\\
Let $Q$ be a configuration space of $N$ objects. Let $\gamma:[t_1,t_2]\to Q$ be a trajectory. Define the space of deformation of $\gamma$ by $$\Omega_\gamma=\{\eta:[t_1,t_2]\to TQ\;|\;\eta(t)\in T_{\gamma(t)}Q\text{ and }\eta(t_1)=\eta(t_2)=0\}$$
\end{defn}

\begin{defn}{Variations}{}\\
Let $Q$ be a configuration space of $N$ objects. Let $L:TQ\times\R\to\R$ be a Lagrangian. Let $\gamma:[t_1,t_2]\to Q$ be a trajectory. Let $\eta\in\Omega_\gamma$ be a deformation. Define the variation of $\eta$ to be $$\delta S[\gamma](\eta)=\frac{d}{d\varepsilon}\bigg{|}_{\varepsilon=0}S[\gamma+\varepsilon\eta]$$
\end{defn}

Assumption: Hamilton's principle (principle of least / stationary action): the actual path taken by the system is the one for which $S[\gamma]$ is stationary under small perturbations (i.e. $\delta S[\gamma](\eta)=0$ for all $\eta$)

\begin{defn}{Euler-Lagrange Operator}{}\\
Let $Q$ be a configuration space of $N$ objects. Let $L:TQ\times\R\to\R$ be a Lagrangian. Define the Euler Lagrange operator to be the function $\text{EL}^i:\Omega\to\mC^0([t_1,t_2],\R^n)$ by $$\text{EL}^i(\gamma)(t)=\frac{\partial L}{\partial x^i}\bigg{|}_{\gamma(t)}-\frac{d}{dt}\frac{\partial L}{\partial x_i'}\bigg{|}_{\gamma(t)}$$ for any $\gamma\in\Omega$ where $(x_1',\dots,x_n',x^1,\dots,x^n)$ are the coordinates of $TQ$. 
\end{defn}

\begin{eg}{}{}\\
Suppose that $Q=\R^1$ is the configuration space of $1$ object. Let $L:TQ\times\R\to\R$ be the Lagrangian given by $$L(v,x,t)=\frac{1}{2}mv^2-V(x)$$ Then for any $\gamma:[0,1]\to Q$, the Euler Lagrange equation operator reads $$\text{EL}(\gamma)=-V'(\gamma(t))-m\gamma''(t)$$ Notice that setting $\text{EL}(\gamma)$ recovers Newton's second equation of motion where $F=-V'(\gamma(t))$. 
\end{eg}

\begin{thm}{}{}\\
Let $Q$ be a configuration space of $N$ objects. Let $L:TQ\times\R\to\R$ be a Lagrangian. Let $\gamma:[t_1,t_2]\to Q$ be a trajectory. Then we have $\delta S[\gamma](\eta)=0$ for all $\eta\in\Omega_\gamma$ if and only if $\text{EL}^i(\gamma)=0$ for all $i$. 
\end{thm}

\subsection{Symmetries and Noether's Theorem}
\begin{defn}{Infinitesimal Symmetry}{}\\
Let $Q$ be a configuration space of $N$ objects. Let $L:TQ\times\R\to\R$ be a Lagrangian. Let $\Phi:(-\varepsilon,\varepsilon)\times Q\to Q$ be a one parameter family of diffeomorphisms. We say that $\Phi$ is a symmetry of $L$ if $$L(\Phi_s,(T\Phi_s)(q'),t)=L(q,q',t)$$ for all $s\in(-\varepsilon,\varepsilon)$. 
\end{defn}

\begin{defn}{Infinitesimal Generator}{}\\
Let $Q$ be a configuration space of $N$ objects. Let $L:TQ\times\R\to\R$ be a Lagrangian. Let $\Phi:(-\varepsilon,\varepsilon)\times Q\to Q$ be a symmetry of $L$. Define the infinitesimal generator of $\Phi$ to be the vector field $$\xi=\frac{d}{ds}_{s=0}\Phi_s$$
\end{defn}

\begin{thm}{Noether's Theorem}{}\\
Let $Q$ be a configuration space of $N$ objects. Let $L:TQ\times\R\to\R$ be a Lagrangian. Let $\Phi:(-\varepsilon,\varepsilon)\times Q\to Q$ be a symmetry of $L$. Let $\gamma$ be a solution of $\text{EL}$. Then we have $$\frac{d}{dt}\frac{\partial L}{\partial x_i'}\xi^i\bigg{|}_{\gamma}=0$$
\end{thm}

\begin{defn}{The 0th K-Group}{}\\
Let $R$ be a ring. Define the $0$th $K$-group of $R$ to be the set $$K_0(R)=\frac{\{P\;|\;P\text{ is a finitely generated projective }R\text{-module }\}}{\cong}$$ together with addition given by $[P_1]+[P_2]=[P_3]$ if there is a short exact sequence \\
\adjustbox{scale=1.0,center}{\begin{tikzcd}
	0 & {P_1} & {P_3} & {P_2} & 0
	\arrow[from=1-1, to=1-2]
	\arrow[from=1-2, to=1-3]
	\arrow[from=1-3, to=1-4]
	\arrow[from=1-4, to=1-5]
\end{tikzcd}}\\~\\
\end{defn}

\begin{eg}{}{}\\
Suppose that $R$ is a PID. Then there is an isomorphism $$K_0(R)\cong\Z$$ given by $[n]\mapsto[R^n]$. 
\end{eg}

\begin{defn}{The Change of Rings Homomorphism}{}\\
Let $R,S$ be rings. Let $f:R\to S$ be a ring homomrphism. Define the change of rings homomorphism to be the map $K_0(f):K_0(R)\to K_0(S)$ given by $$[P]\mapsto[S\otimes_RP]$$
\end{defn}

Recall the Grothendieck completion of an abelian semi-group $G(-)$. 

\begin{prp}{}{}\\
Let $R$ be a ring. Let $P(R)=\{P\;|\;P\text{ is a finitely generated projective }R\text{-modules }\}$. Then there is a natural isomorphism $$K_0(R)\cong G(P(R),\oplus)$$
\end{prp}

\begin{prp}{Morita Equivalence}{}\\
Let $R$ be a ring. Then there is a natural isomorphism $$K_0(R)\cong K_0(M_n(R))$$ given by $[P]\mapsto R^n\otimes_RP$ where $R^n$ is considered as an $(M_n(R),R)$-bimodule. 
\end{prp}

\begin{prp}{}{}\\
Let $R,S$ be rings. Then there is a natural isomrophism $$K_0(R\times S)\cong K_0(R)\times K_0(S)$$ induced by the product of the projection maps. 
\end{prp}

\begin{eg}{}{}\\
Let $F$ be a field. Let $V$ be a vector space over $F$ of countable dimension. Then there is an isomorphism $$K_0(\text{End}_F(V))\cong\{1\}$$
\end{eg}

\begin{defn}{The Reduced 0th K-Group}{}\\
Let $R$ be a ring. Define the reduced $0$th $K$-group to be the quotient $$\widetilde{K}_0(R)=\frac{K_0(R)}{\langle[R^m]-[R^n]\;|\;m,n\in\N\rangle}$$
\end{defn}

\begin{lmm}{}{}\\
Let $R$ be a ring. Then there is a natural isomorphism $$\widetilde{K}_0(R)\cong\langle[R]\rangle\leq\widetilde{K}_0(R)$$
\end{lmm}

\begin{prp}{}{}\\
Let $G$ be a group. Let $R$ be an integral domain. Then there is an isomorphism $$K_0(R[G])\cong\widetilde{K}_0(R[G])\oplus\Z$$ give by $[P]\mapsto([P],\dim_{\text{Frac}(R)}(\text{Frac}(R)\otimes_{R[G]}P))$
\end{prp}

\begin{conj}{Farrell-Jones Conjecture}{}\\
Let $R$ be a regular ring. Let $G$ be a torsion free group. Then the inclusion map induces a natural isomorphism $$K_0(R)\cong K_0(R[G])$$
\end{conj}
