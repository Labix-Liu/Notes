\subsection{Background: Affine Varieties}
\begin{ex}{}{}\\
Suppose that $V$ is an affine variety such that $R=\C[V]=\frac{\C[x_1,\dots,x_n]}{\I(V)}$. Then $\I(V)$ is a radical ideal so $\C[V]$ is finitely generated with no non-zero nilpotents. \\

Conversely, suppose that $R$ is a finitely generated $\C$-algebra with no non-zero nilpotents. By the universal property of finitely generated $\C$-algebras, there exists an ideal $I\subseteq\C[x_1,\dots,x_n]$ such that $R\cong\frac{\C[x_1,\dots,x_n]}{I}$. Let $V=\V(I)$. Then $\I(V)=\sqrt{I}$ by the nullstellensatz and $\frac{\C[x_1,\dots,x_n]}{I}\cong\frac{\C[x_1,\dots,x_n]}{\sqrt{I}}$ so that $R$ is the coordinate ring of $V$. 
\end{ex}

\begin{ex}{}{}\\
\begin{enumerate}[label=(\alph*)]
\item Suppose that $a_1,a_2,c_1,c_2\in R$ and $b_1,b_2,d_1,d_2\in S$ such that $a_1/b_1\sim a_2/b_2$ and $c_1/d_1\sim c_2/d_2$. Then there exists $u,v\in S$ such that $u(a_1b_2-a_2b_1)=0=v(c_1d_2-c_2d_1)$. Then we have $$\frac{a_1}{b_1}+\frac{c_1}{d_1}=\frac{a_1d_1+b_1c_1}{b_1d_1}\;\;\text{ and }\;\;\frac{a_2}{b_2}+\frac{c_2}{d_2}=\frac{a_2d_2+b_2c_2}{b_2d_2}$$ Then notice that $$(a_1d_1+b_1c_1)(b_2d_2)-(a_2d_2+b_2c_2)(b_1d_1)=(d_1d_2)(a_1b_2-a_2b_1)+(b_1b_2)(c_1d_2-c_2d_1)$$ Multiplying $uv$ gives $0$. Hence addition is well defined. \\

For multiplication, we have $$\frac{a_1}{b_1}\frac{c_1}{d_1}=\frac{a_1c_1}{b_1d_1}\;\;\text{ and }\;\;\frac{a_2}{b_2}\frac{c_2}{d_2}=\frac{a_2c_2}{b_2d_2}$$ Then we have $$a_1c_1b_2d_2-a_2c_2b_1d_1=(c_1d_2)(a_1b_2-a_2b_1)-(a_2b_1)(c_1d_2-c_2d_1)$$ Multiplying $uv$ gives $0$. Hence multiplication is well defined. 
\item Suppose that $r\in R$ and $s\in S$ such that $(r/s)^n=0$ for some $n\in\N^+$. This means that there exists $u\in S$ such that $ur^n=0$. Multiplying $u^{n-1}$ on both sides give $(ur)^n=0$. Since $R$ has no nilpotents, we conclude that $ur=0$. Then this implies $r/s=0$. Thus $S^{-1}R$ has no nilpotents. 
\end{enumerate}
\end{ex}

\begin{ex}{}{}\\
Let $R$ be a UFD. Let $a/b\in\text{Frac}(R)$ that is integral. Assume that $a,b$ do not have common factors. Then there exists $r_0,\dots,r_{n-1}\in R$ such that $$\frac{a^n}{b^n}+r_{n-1}\frac{a^{n-1}}{b^{n-1}}+\dots+r_1\frac{a}{b}+r_0=0$$ Rearranging, we get $$a^n=-b\left(r_{n-1}a^{n-1}+\dots+r_1a^1b^{n-2}+r_0b^{n-1}\right)$$ This shows that any irreducible element dividing $b$ also divides $a^n$, and hence $a$. Since $a$ and $b$ do not have common factors, this means that no irreducible element divides $b$. Since $R$ is a UFD, $b$ must be a unit. Hence $a/b\in R$. 
\end{ex}
